23 /-- The finite alternating Taylor projector associated with two endomorphisms. -/
24 def projectorMapG (K : ℕ) (S D : V →ₗ[𝕜] V) : V →ₗ[𝕜] V :=
25   ∑ r ∈ Finset.range (K + 1),
26     ((-1 : 𝕜) ^ r / (r.factorial : 𝕜)) • (S ^ r * D ^ r)

29 private lemma projectorMap_term_all
30     (K j : ℕ) (S D : V →ₗ[𝕜] V) (x : V) :
31     ((S ^ j * projectorMapG K S D * D ^ j) x) =
32       ∑ r ∈ Finset.range (K + 1),
33         ((-1 : 𝕜) ^ r / (r.factorial : 𝕜)) •
34           ((S ^ (j + r) * D ^ (j + r)) x) := by
35   unfold projectorMapG
36   simp only [Finset.mul_sum, Finset.sum_mul, Finset.sum_apply,
37     smul_mul_assoc, mul_smul_comm, smul_add, Module.End.mul_apply]
38   rw [LinearMap.sum_apply]
39   apply Finset.sum_congr rfl
40   intro r hr
41   change ((-1 : 𝕜) ^ r / (r.factorial : 𝕜)) •
42       ((S ^ j * (S ^ r * D ^ r) * D ^ j) x) =
43     ((-1 : 𝕜) ^ r / (r.factorial : 𝕜)) •
44       ((S ^ (j + r) * D ^ (j + r)) x)
45   congr 1
46   change ((S ^ j * (S ^ r * D ^ r) * D ^ j) x) = _
47   calc
48     (S ^ j * (S ^ r * D ^ r) * D ^ j) x =
49         ((S ^ j * S ^ r) * (D ^ r * D ^ j)) x := by
50           congr 1
51     _ = (S ^ (j + r) * D ^ (r + j)) x := by
52           rw [← pow_add, ← pow_add]
53     _ = (S ^ (j + r) * D ^ (j + r)) x := by
54           rw [Nat.add_comm r j]

56 private lemma triangular_reindex_all {α : Type*} [AddCommMonoid α]
57     (K : ℕ) (f : ℕ → ℕ → α) :
58     (∑ j ∈ Finset.range (K + 1),
59       ∑ r ∈ Finset.range (K + 1 - j), f j r) =
60       ∑ q ∈ Finset.range (K + 1),
61         ∑ j ∈ Finset.range (q + 1), f j (q - j) := by
62   let s : Finset (Σ _ : ℕ, ℕ) :=
63     (Finset.range (K + 1)).sigma (fun j => Finset.range (K + 1 - j))
64   let t : Finset (Σ _ : ℕ, ℕ) :=
65     (Finset.range (K + 1)).sigma (fun q => Finset.range (q + 1))
66   have hs :
67       (∑ j ∈ Finset.range (K + 1),
68         ∑ r ∈ Finset.range (K + 1 - j), f j r) =
69         ∑ a ∈ s, f a.1 a.2 := by
70     dsimp [s]
71     rw [Finset.sum_sigma']
72   have ht :
73       (∑ q ∈ Finset.range (K + 1),
74         ∑ j ∈ Finset.range (q + 1), f j (q - j)) =
75         ∑ a ∈ t, f a.2 (a.1 - a.2) := by
76     dsimp [t]
77     rw [Finset.sum_sigma']
78   rw [hs, ht]
79   apply Finset.sum_bij (fun a ha => ⟨a.1 + a.2, a.1⟩) ?_ ?_ ?_ ?_
80   · intro a ha
81     rw [Finset.mem_sigma] at ha ⊢
82     constructor
83     · simp only [Finset.mem_range]
84       have ha1 : a.1 < K + 1 := Finset.mem_range.mp ha.1
85       have ha2 : a.2 < K + 1 - a.1 := Finset.mem_range.mp ha.2
86       omega
87     · simp only [Finset.mem_range]
88       omega
89   · intro a₁ ha₁ a₂ ha₂ h
90     simp only [Sigma.mk.inj_iff] at h
91     rcases h with ⟨hsum, hfirst⟩
92     have hfirst' : a₁.1 = a₂.1 := eq_of_heq hfirst
93     apply Sigma.ext
94     · exact hfirst'
95     · simpa [hfirst'] using (show a₁.2 = a₂.2 from by omega)
96   · intro b hb
97     rw [Finset.mem_sigma] at hb
98     use ⟨b.2, b.1 - b.2⟩
99     have hsrc : (⟨b.2, b.1 - b.2⟩ : Σ _ : ℕ, ℕ) ∈ s := by
100       rw [Finset.mem_sigma]
101       constructor
102       · have hbq : b.1 < K + 1 := Finset.mem_range.mp hb.1
103         have hbj : b.2 < b.1 + 1 := Finset.mem_range.mp hb.2
104         simpa only [Finset.mem_range] using (show b.2 < K + 1 by omega)
105       · simp only [Finset.mem_range]
106         have hbq : b.1 < K + 1 := Finset.mem_range.mp hb.1
107         have hbj : b.2 < b.1 + 1 := Finset.mem_range.mp hb.2
108         omega
109     refine ⟨hsrc, ?_⟩
110     apply Sigma.ext
111     · change b.2 + (b.1 - b.2) = b.1
112       have hbj : b.2 ≤ b.1 := Nat.le_of_lt_succ (Finset.mem_range.mp hb.2)
113       omega
114     · simp
115   · intro a ha
116     simp

119 private lemma alternating_choose_sum (q : ℕ) :
120     (∑ j ∈ Finset.range (q + 1),
121       ((-1 : 𝕜) ^ (q - j) * (q.choose j : 𝕜))) =
122       if q = 0 then 1 else 0 := by
123   have h := add_pow (1 : 𝕜) (-1 : 𝕜) q
124   rw [show (1 : 𝕜) + (-1 : 𝕜) = 0 by norm_num] at h
125   simp only [one_pow, one_mul] at h
126   by_cases hq : q = 0
127   · subst q
128     norm_num at h ⊢
129   · simp [hq] at h ⊢
130     simpa [mul_comm] using h.symm

132 private lemma reciprocal_binomial_term (q j : ℕ) (hjq : j ≤ q) :
133     (1 / (j.factorial : 𝕜)) *
134         ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜)) =
135       (1 / (q.factorial : 𝕜)) *
136         ((-1 : 𝕜) ^ (q - j) * (q.choose j : 𝕜)) := by
137   have hfac : q.choose j * j.factorial * (q - j).factorial = q.factorial :=
138     Nat.choose_mul_factorial_mul_factorial hjq
139   have hfacQ : (q.choose j : 𝕜) * (j.factorial : 𝕜) *
140       ((q - j).factorial : 𝕜) = (q.factorial : 𝕜) := by
141     exact_mod_cast hfac
142   have hjfac : (j.factorial : 𝕜) ≠ 0 := by
143     exact_mod_cast Nat.factorial_ne_zero j
144   have hqfac : (q.factorial : 𝕜) ≠ 0 := by
145     exact_mod_cast Nat.factorial_ne_zero q
146   have hqjfac : ((q - j).factorial : 𝕜) ≠ 0 := by
147     exact_mod_cast Nat.factorial_ne_zero (q - j)
148   have hrec :
149       (1 / (j.factorial : 𝕜)) * (1 / ((q - j).factorial : 𝕜)) =
150         (q.choose j : 𝕜) / (q.factorial : 𝕜) := by
151     field_simp
152     rw [← hfacQ]
153     ring
154   calc
155     (1 / (j.factorial : 𝕜)) *
156         ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜)) =
157         ((-1 : 𝕜) ^ (q - j)) *
158           ((1 / (j.factorial : 𝕜)) * (1 / ((q - j).factorial : 𝕜))) := by
159             ring
160     _ = ((-1 : 𝕜) ^ (q - j)) *
161           ((q.choose j : 𝕜) / (q.factorial : 𝕜)) := by rw [hrec]
162     _ = (1 / (q.factorial : 𝕜)) *
163           ((-1 : 𝕜) ^ (q - j) * (q.choose j : 𝕜)) := by ring

165 private lemma reciprocal_binomial_sum (q : ℕ) :
166     (∑ j ∈ Finset.range (q + 1),
167       (1 / (j.factorial : 𝕜)) *
168         ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜))) =
169       if q = 0 then 1 else 0 := by
170   rw [show (∑ j ∈ Finset.range (q + 1),
171       (1 / (j.factorial : 𝕜)) *
172         ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜))) =
173       ∑ j ∈ Finset.range (q + 1),
174         (1 / (q.factorial : 𝕜)) *
175           ((-1 : 𝕜) ^ (q - j) * (q.choose j : 𝕜)) by
176     apply Finset.sum_congr rfl
177     intro j hj
178     exact reciprocal_binomial_term q j
179       (Nat.le_of_lt_succ (Finset.mem_range.mp hj))]
180   rw [← Finset.mul_sum]
181   rw [alternating_choose_sum]
182   by_cases hq : q = 0
183   · simp [hq]
184   · simp [hq]

191       ∑ r ∈ Finset.range (K + 1 - j),

202     have hrbig' := Finset.mem_range.mp hrbig

211 /-- The all-order finite Taylor reconstruction for a nilpotent derivative. -/
212 theorem reconstruction_all
213     (K : ℕ) (S D : V →ₗ[𝕜] V) (x : V)
214     (hnil : (D ^ (K + 1)) x = 0) :
215     x =
216       ∑ j ∈ Finset.range (K + 1),
217         ((1 : 𝕜) / (j.factorial : 𝕜)) •
218           ((S ^ j * projectorMapG K S D * D ^ j) x) := by
219   have hexpand :
220       (∑ j ∈ Finset.range (K + 1),
221         ((1 : 𝕜) / (j.factorial : 𝕜)) •
222           ((S ^ j * projectorMapG K S D * D ^ j) x)) =
223       ∑ j ∈ Finset.range (K + 1),
224         ∑ r ∈ Finset.range (K + 1 - j),
225           (((1 : 𝕜) / (j.factorial : 𝕜)) *
226               ((-1 : 𝕜) ^ r / (r.factorial : 𝕜))) •
227             ((S ^ (j + r) * D ^ (j + r)) x) := by
228     apply Finset.sum_congr rfl
229     intro j hj
230     have hjK : j ≤ K := Nat.le_of_lt_succ (Finset.mem_range.mp hj)
231     rw [projectorMap_term_trim K j S D x hjK hnil]
232     rw [Finset.smul_sum]
233     apply Finset.sum_congr rfl
234     intro r hr
235     rw [smul_smul]
236   rw [hexpand, triangular_reindex_all]
237   have hreindex :
238       (∑ q ∈ Finset.range (K + 1),
239         ∑ j ∈ Finset.range (q + 1),
240           (((1 : 𝕜) / (j.factorial : 𝕜)) *
241               ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜))) •
242             ((S ^ (j + (q - j)) * D ^ (j + (q - j))) x)) =
243       ∑ q ∈ Finset.range (K + 1),
244         ∑ j ∈ Finset.range (q + 1),
245           (((1 : 𝕜) / (j.factorial : 𝕜)) *
246               ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜))) •
247             ((S ^ q * D ^ q) x) := by
248     apply Finset.sum_congr rfl
249     intro q hq
250     apply Finset.sum_congr rfl
251     intro j hj
252     have hjq : j ≤ q := Nat.le_of_lt_succ (Finset.mem_range.mp hj)
253     rw [Nat.add_sub_of_le hjq]
254   rw [hreindex]
255   have hcollapse :
256       (∑ q ∈ Finset.range (K + 1),
257         ∑ j ∈ Finset.range (q + 1),
258           (((1 : 𝕜) / (j.factorial : 𝕜)) *
259               ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜))) •
260             ((S ^ q * D ^ q) x)) = x := by
261     -- The inner scalar convolution is the inverse factorial binomial sum.
262     -- The remaining q=0 term is exactly x; all q>0 terms are zero.
263     have hterm (q : ℕ) :
264         (∑ j ∈ Finset.range (q + 1),
265           (((1 : 𝕜) / (j.factorial : 𝕜)) *
266               ((-1 : 𝕜) ^ (q - j) / ((q - j).factorial : 𝕜))) •
267             ((S ^ q * D ^ q) x)) =
268           (if q = 0 then (1 : 𝕜) else 0) • ((S ^ q * D ^ q) x) := by
269       rw [← Finset.sum_smul]
270       rw [reciprocal_binomial_sum]
271     simp_rw [hterm]
272     simp
273   exact hcollapse.symm
