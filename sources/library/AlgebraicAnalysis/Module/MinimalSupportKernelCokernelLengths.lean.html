23 private theorem localized_isFiniteLength_of_minimal_support
24     {R N : Type*} [CommRing R] [AddCommGroup N] [Module R N]
25     [IsNoetherianRing R] [Module.Finite R N]
26     (q : PrimeSpectrum R)
27     (hqmem : q ∈ Module.support R N)
28     (hq : ∀ p ∈ Module.support R N, p.asIdeal ≤ q.asIdeal →
29       q.asIdeal ≤ p.asIdeal) :
30     IsFiniteLength (Localization q.asIdeal.primeCompl)
31       (LocalizedModule q.asIdeal.primeCompl N) := by
32   have hqmem' : q ∈ Module.support R N := hqmem
33   have hqmem := hqmem'
34   have hmin : q.asIdeal ∈ (Module.annihilator R N).minimalPrimes := by
35     obtain ⟨p, hp, hpq⟩ :=
36       Ideal.exists_minimalPrimes_le
37         (Module.mem_support_iff_of_finite.mp hqmem)
38     have hpprime : p.IsPrime := hp.1.1
39     let p' : PrimeSpectrum R := ⟨p, hpprime⟩
40     have hp'supp : p' ∈ Module.support R N :=
41       Module.mem_support_iff_of_finite.mpr hp.1.2
42     have hqp : q.asIdeal ≤ p := hq p' hp'supp hpq
43     have heq : p = q.asIdeal := le_antisymm hpq hqp
44     simpa [heq] using hp
45   exact MinimalPrimeFiniteLengthLocalization.localizedModule_isFiniteLength
46     q.asIdeal hmin

48 theorem localized_kernel_and_cokernel_isFiniteLength
49     {R C E : Type*} [CommRing R] [CommRing C] [Algebra R C]
50     [AddCommGroup E] [Module C E] [Module R E] [IsScalarTower R C E]
51     [IsNoetherianRing R] [IsNoetherianRing C] [Module.Finite C E]
52     (f : Module.End C E)
53     [Module.Finite R (LinearMap.ker (f.restrictScalars R))]
54     [Module.Finite R (E ⧸ LinearMap.range (f.restrictScalars R))]
55     (q : PrimeSpectrum R)
56     (hqmem : q ∈ Module.support R (E ⧸ LinearMap.range (f.restrictScalars R)))
57     (hq : ∀ p ∈ Module.support R (E ⧸ LinearMap.range (f.restrictScalars R)),
58       p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal) :
59     IsFiniteLength (Localization q.asIdeal.primeCompl)
60         (LocalizedModule q.asIdeal.primeCompl
61           (E ⧸ LinearMap.range (f.restrictScalars R))) ∧
62       IsFiniteLength (Localization q.asIdeal.primeCompl)
63         (LocalizedModule q.asIdeal.primeCompl
64           (LinearMap.ker (f.restrictScalars R))) := by
65   have hc := localized_isFiniteLength_of_minimal_support q hqmem hq
66   have hksub :=
67     endomorphism_kernel_support_subset_cokernel_support_over_base
68       (R := R) (C := C) (E := E) f
69   by_cases hk : q ∈ Module.support R (LinearMap.ker (f.restrictScalars R))
70   · have hqk : ∀ p ∈ Module.support R (LinearMap.ker (f.restrictScalars R)),
71         p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal := by
72       intro p hp hpq
73       exact hq p (hksub hp) hpq
74     exact ⟨hc, localized_isFiniteLength_of_minimal_support q hk hqk⟩
75   · have hsub : Subsingleton (LocalizedModule q.asIdeal.primeCompl
76         (LinearMap.ker (f.restrictScalars R))) :=
77       not_nontrivial_iff_subsingleton.mp (by
78         intro h
79         exact hk (Module.mem_support_iff.mpr h))
80     let := hsub
81     exact ⟨hc, IsFiniteLength.of_subsingleton⟩
