8 # Comparison of base and coefficient localizations
9 
10 For an `R`-algebra `C`, localizing a `C`-module at the image of a submonoid
11 `S ≤ R` agrees with localizing it as an `R`-module at `S`.
12 -/

69 /-- The canonical equivalence between base and coefficient localizations. -/
70 noncomputable def localizedModuleComparison :
71     LocalizedModule S E ≃ₗ[Localization S] LocalizedModule SC E :=
72   let : IsLocalizedModule S
73       ((LocalizedModule.mkLinearMap SC E).restrictScalars R) :=
74     localizedModule_isLocalizedOverBase S
75   (IsLocalizedModule.linearEquiv S
76       (LocalizedModule.mkLinearMap S E)
77       ((LocalizedModule.mkLinearMap SC E).restrictScalars R)).extendScalarsOfIsLocalization
78         S (Localization S)

80 @[simp, nolint simpNF] theorem localizedModuleComparison_mkLinearMap (m : E) :
81     localizedModuleComparison S (LocalizedModule.mkLinearMap S E m) =
82       LocalizedModule.mkLinearMap SC E m := by
83   let : IsLocalizedModule S
84       ((LocalizedModule.mkLinearMap SC E).restrictScalars R) :=
85     localizedModule_isLocalizedOverBase S
86   exact IsLocalizedModule.linearEquiv_apply S
87     (LocalizedModule.mkLinearMap S E)
88     ((LocalizedModule.mkLinearMap SC E).restrictScalars R) m

90 @[simp] theorem localizedModuleComparison_mk (m : E) (s : S) :
91     localizedModuleComparison S (LocalizedModule.mk m s) =
92       (LocalizedModule.mk m (coefficientDenominator (C := C) S s) :
93         LocalizedModule SC E) := by
94   have h : LocalizedModule.mk m s =
95       Localization.mk (1 : R) s • LocalizedModule.mkLinearMap S E m := by
96     rw [LocalizedModule.mkLinearMap_apply, LocalizedModule.mk_smul_mk]
97     simp
98   rw [h, map_smul, localizedModuleComparison_mkLinearMap]
99   change algebraMap (Localization S) (Localization SC) (Localization.mk 1 s) •
100       (LocalizedModule.mk m (1 : SC) : LocalizedModule SC E) = _
101   rw [Localization.mk_eq_mk',
102     IsLocalization.algebraMap_mk' (R := R) (S := C)
103       (Rₘ := Localization S) (Sₘ := Localization SC)]
104   have hs : (⟨algebraMap R C s,
105       Algebra.mem_algebraMapSubmonoid_of_mem s⟩ : SC) =
106       coefficientDenominator (C := C) S s := Subtype.ext rfl
107   rw [map_one, hs]
108   rw [← Localization.mk_eq_mk', LocalizedModule.mk_smul_mk]
109   simp
