7 /-!
8 # Boundary exhaustion on the total target pages
9 
10 The concrete boundary inclusions give maps from the first target page to every
11 later target page.  Surjectivity of the underlying differential and exhaustive
12 filtration imply pointwise eventual vanishing; finite support then gives the
13 same statement on the external direct sum.
14 -/

27 private theorem boundaries_one_le (r : ℕ) (p : ℤ) :
28     K.boundaries 1 p ≤ K.boundaries (r + 1) p := by
29   induction r with
30   | zero => exact le_rfl
31   | succ r ihr =>
32       exact ihr.trans (K.boundaries_le_succ (r + 1) p)
33 
34 private theorem boundaries_mono {a b : ℕ} (hab : a ≤ b) (p : ℤ) :
35     K.boundaries a p ≤ K.boundaries b p := by
36   induction b, hab using Nat.le_induction with
37   | base => exact le_rfl
38   | succ b hb ih => exact ih.trans (K.boundaries_le_succ b p)
39 
40 /-- The quotient map induced by `B₁ ⊆ B_{r+1}` at one target component. -/
41 def targetBoundaryMap (r : ℕ) (p : ℤ) :
42     K.TargetPage 1 p →ₗ[k] K.TargetPage (r + 1) p :=
43   Submodule.mapQ _ _ LinearMap.id (by
44     intro x hx
45     exact K.boundaries_one_le r p hx)
46 
47 @[simp] theorem targetBoundaryMap_mk (r : ℕ) (p : ℤ) (x : K.G p) :
48     K.targetBoundaryMap r p (Submodule.Quotient.mk x) =
49       Submodule.Quotient.mk x :=
50   Submodule.mapQ_apply _ _ _ _
51 
52 theorem targetBoundaryMap_surjective (r : ℕ) (p : ℤ) :
53     Function.Surjective (K.targetBoundaryMap r p) := by
54   intro y
55   obtain ⟨x, rfl⟩ := Submodule.Quotient.mk_surjective
56     ((K.boundaries (r + 1) p).comap (K.G p).subtype) y
57   exact ⟨Submodule.Quotient.mk x, K.targetBoundaryMap_mk r p x⟩
58 
59 /-- The component kernels grow with the page index. -/
60 theorem targetBoundaryMap_ker_mono (r s : ℕ) (hrs : r ≤ s) (p : ℤ) :
61     LinearMap.ker (K.targetBoundaryMap r p) ≤
62       LinearMap.ker (K.targetBoundaryMap s p) := by
63   intro y hy
64   obtain ⟨x, rfl⟩ := Submodule.Quotient.mk_surjective
65     ((K.boundaries 1 p).comap (K.G p).subtype) y
66   change K.targetBoundaryMap r p (Submodule.Quotient.mk x) = 0 at hy
67   change K.targetBoundaryMap s p (Submodule.Quotient.mk x) = 0
68   rw [K.targetBoundaryMap_mk, Submodule.Quotient.mk_eq_zero] at hy ⊢
69   have hpage : r + 1 ≤ s + 1 := by omega
70   exact K.boundaries_mono hpage p hy
71 
72 /-- The total target map into page `r+1`. -/
73 def totalBoundaryMap (r : ℕ) :
74     K.TargetTotal 1 →ₗ[k] K.TargetTotal (r + 1) :=
75   DirectSum.lmap (K.targetBoundaryMap r)
76 
77 @[simp] theorem totalBoundaryMap_lof (r : ℕ) (p : ℤ)
78     (x : K.TargetPage 1 p) :
79     K.totalBoundaryMap r
80         (DirectSum.lof k ℤ (fun q : ℤ => K.TargetPage 1 q) p x) =
81       DirectSum.lof k ℤ (fun q : ℤ => K.TargetPage (r + 1) q) p
82         (K.targetBoundaryMap r p x) := by
83   rw [totalBoundaryMap]
84   exact DirectSum.lmap_of _ _ _
85 
86 theorem totalBoundaryMap_surjective (r : ℕ) :
87     Function.Surjective (K.totalBoundaryMap r) := by
88   apply (DirectSum.lmap_surjective _).mpr
89   exact K.targetBoundaryMap_surjective r
90 
91 theorem totalBoundaryMap_ker_mono (r s : ℕ) (hrs : r ≤ s) :
92     LinearMap.ker (K.totalBoundaryMap r) ≤
93       LinearMap.ker (K.totalBoundaryMap s) := by
94   intro x hx
95   apply LinearMap.mem_ker.mpr
96   apply DirectSum.ext_component k
97   intro p
98   have hlocal : K.targetBoundaryMap r p (x p) = 0 := by
99     apply LinearMap.mem_ker.mp
100     exact congrArg (fun z => z p) hx
101   have hker := K.targetBoundaryMap_ker_mono r s hrs p
102   exact LinearMap.mem_ker.mp (hker (LinearMap.mem_ker.mpr hlocal))

104 private theorem pointwise_boundary_exhaustion
105     (hG : ∀ z : M, ∃ s : ℤ, z ∈ K.G s)
106     (hf : Function.Surjective K.f) (p : ℤ) (y : K.TargetPage 1 p) :
107     ∃ r : ℕ, K.targetBoundaryMap r p y = 0 := by
108   obtain ⟨x, rfl⟩ := Submodule.Quotient.mk_surjective
109     ((K.boundaries 1 p).comap (K.G p).subtype) y
110   obtain ⟨n, hn⟩ := K.exists_mem_boundaries_of_surjective hG hf x.property
111   let r := n - 1
112   have hnr : n ≤ r + 1 := by dsimp [r]; omega
113   have hbound : K.boundaries n p ≤ K.boundaries (r + 1) p := by
114     rcases n with _ | n
115     · exact (K.boundaries_le_succ 0 p).trans (K.boundaries_one_le r p)
116     · simpa [r] using le_rfl
117   refine ⟨r, ?_⟩
118   rw [K.targetBoundaryMap_mk, Submodule.Quotient.mk_eq_zero]
119   exact hbound hn
120 
121 /-- Every finitely supported total target vector is killed at some page. -/
122 theorem totalBoundaryMap_eventually_zero
123     (hG : ∀ z : M, ∃ s : ℤ, z ∈ K.G s)
124     (hf : Function.Surjective K.f) (x : K.TargetTotal 1) :
125     ∃ r : ℕ, K.totalBoundaryMap r x = 0 := by
126   classical
127   induction x using DirectSum.induction_on with
128   | zero => exact ⟨0, by simp⟩
129   | of p y =>
130       obtain ⟨r, hr⟩ := K.pointwise_boundary_exhaustion hG hf p y
131       refine ⟨r, ?_⟩
132       rw [show DirectSum.of _ p y =
133         DirectSum.lof k ℤ (fun q : ℤ => K.TargetPage 1 q) p y from rfl,
134         K.totalBoundaryMap_lof, hr]
135       simp
136   | add x y hx hy =>
137       obtain ⟨r, hr⟩ := hx
138       obtain ⟨s, hs⟩ := hy
139       let t := max r s
140       refine ⟨t, ?_⟩
141       have hrt : r ≤ t := le_max_left _ _
142       have hst : s ≤ t := le_max_right _ _
143       have hxr := K.totalBoundaryMap_ker_mono r t hrt (LinearMap.mem_ker.mpr hr)
144       have hys := K.totalBoundaryMap_ker_mono s t hst (LinearMap.mem_ker.mpr hs)
145       rw [map_add]
146       rw [LinearMap.mem_ker.mp hxr, LinearMap.mem_ker.mp hys, add_zero]
