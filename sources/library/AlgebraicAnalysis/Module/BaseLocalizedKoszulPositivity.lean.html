33       (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R)))
34     (hqmin : ∀ p ∈ Module.support R
35         (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R)),
36       p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal)

38     Module.length (Localization q.asIdeal.primeCompl)
39         (LocalizedModule q.asIdeal.primeCompl
40           (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R))) >
41       Module.length (Localization q.asIdeal.primeCompl)

43           (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R))) := by

45   let SC : Submonoid C := Algebra.algebraMapSubmonoid C S
46   let fC : Module.End C E := LinearMap.lsmul C E x
47   let fR : Module.End R E := fC.restrictScalars R
48   have hfinite := localized_kernel_and_cokernel_isFiniteLength fC q hqmem hqmin

50       (LocalizedModule S (E ⧸ LinearMap.range fR)) := by
51     simpa only [S, fR, fC] using hfinite.1
52   have hkerfinite : IsFiniteLength (Localization S)
53       (LocalizedModule S (LinearMap.ker fR)) := by
54     simpa only [S, fR, fC] using hfinite.2
55   let : Module (Localization S) (LocalizedModule SC E) :=
56     Module.compHom _ (algebraMap (Localization S) (Localization SC))

60       (LocalizedModule SC E)

62     localizedModuleComparison S

64   let y : Localization SC := algebraMap C (Localization SC) x
65   let gC : Module.End (Localization SC) (LocalizedModule SC E) :=
66     LinearMap.lsmul (Localization SC) (LocalizedModule SC E) y
67   have heq (z : LocalizedModule S E) : e (gR z) = gC (e z) := by
68     rw [show e (gR z) = LocalizedModule.map SC fC (e z) by
69       exact localizedModuleComparison_natural S fC z]
70     induction e z using LocalizedModule.induction_on with
71     | _ m s => simp [gC, y, fC, LocalizedModule.smul'_mk]
72   let ek : LinearMap.ker gR ≃ₗ[Localization S]
73       LinearMap.ker (gC.restrictScalars (Localization S)) :=
74     LinearEquiv.ofBijective
75       (e.toLinearMap.domRestrict (LinearMap.ker gR) |>.codRestrict
76         (LinearMap.ker (gC.restrictScalars (Localization S))) (fun z => by
77           exact LinearMap.mem_ker.mpr (by simpa using (heq z).symm)))
78       (by
79         constructor
80         · intro a b hab
81           exact Subtype.ext (e.injective (congrArg Subtype.val hab))
82         · intro z
83           refine ⟨⟨e.symm z, ?_⟩, Subtype.ext (e.apply_symm_apply z)⟩
84           exact LinearMap.mem_ker.mpr (e.injective (by
85             rw [heq]
86             simpa using z.property)))
87   have herange : Submodule.map e.toLinearMap (LinearMap.range gR) =
88       LinearMap.range (gC.restrictScalars (Localization S)) := by
89     ext z
90     constructor
91     · rintro ⟨_, ⟨a, rfl⟩, rfl⟩
92       exact ⟨e a, by simpa using (heq a).symm⟩
93     · rintro ⟨a, rfl⟩

98         (gC.restrictScalars (Localization S))) :=

100   have hnontrivBase : Nontrivial

109         (localizedCokernelEquiv S fR).toEquiv.subsingleton_congr.mpr inferInstance
110       exact not_subsingleton_iff_nontrivial.mpr hnontrivBase inferInstance)
111   have hnontrivCokerC : Nontrivial
112       (LocalizedModule SC E ⧸ LinearMap.range
113         (gC.restrictScalars (Localization S))) :=
114     not_subsingleton_iff_nontrivial.mp (by
115       intro hs
116       exact not_subsingleton_iff_nontrivial.mpr hnontrivCokerR

118   have hrange : LinearMap.range gC = y • (⊤ : Submodule (Localization SC)

120     ext z
121     rw [LinearMap.mem_range, Submodule.mem_smul_pointwise_iff_exists]
122     simp only [Submodule.mem_top, true_and]
123     rfl
124   have hnontrivQuot : Nontrivial (QuotSMulTop y (LocalizedModule SC E)) := by
125     change Nontrivial (LocalizedModule SC E ⧸
126       y • (⊤ : Submodule (Localization SC) (LocalizedModule SC E)))
127     rw [← hrange]
128     exact hnontrivCokerC

130   have hkbase : IsFiniteLength (Localization S) (LinearMap.ker gR) :=
131     (localizedKernelEquiv S fR).isFiniteLength hkerfinite
132   have hkfinite : IsFiniteLength (Localization S)

139     ← ek.length_eq, ← (localizedKernelEquiv S fR).length_eq] at hpos
