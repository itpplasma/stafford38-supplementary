8 # Kernel support over a base ring
9 
10 A finite module over a commutative Noetherian coefficient algebra is Hopfian
11 over that algebra.  This gives the kernel--cokernel support inclusion after
12 restriction of scalars, without assuming finite generation over the base.
13 -/

17 theorem endomorphism_kernel_support_subset_cokernel_support_over_base
18     {R C E : Type*} [CommRing R] [CommRing C] [Algebra R C]
19     [AddCommGroup E] [Module C E] [Module R E] [IsScalarTower R C E]
20     [IsNoetherianRing C] [Module.Finite C E] (f : Module.End C E) :
21     Module.support R (LinearMap.ker (f.restrictScalars R)) ⊆
22       Module.support R (E ⧸ LinearMap.range (f.restrictScalars R)) := by
23   intro p hp
24   rw [Module.mem_support_iff_exists_annihilator] at hp ⊢
25   obtain ⟨x, hx⟩ := hp
26   let xC : LinearMap.ker f := ⟨x.1, by
27     exact x.2⟩
28   let I : Ideal C := (C ∙ xC).annihilator
29   have hdisj : Disjoint (I : Set C) (p.asIdeal.primeCompl.map (algebraMap R C)) := by
30     rw [Set.disjoint_left]
31     rintro c hc ⟨r, hr, rfl⟩
32     have hc' : algebraMap R C r • xC.1 = 0 := by
33       have hcx : algebraMap R C r • xC = 0 := by
34         exact (Submodule.mem_annihilator_span_singleton _ _).mp hc
35       exact congrArg Subtype.val hcx
36     have hrx : r • x = 0 := by
37       apply Subtype.ext
38       change r • x.1 = 0
39       simpa [xC, IsScalarTower.algebraMap_smul C r x.1] using hc'
40     have hrann : r ∈ (R ∙ x).annihilator := by
41       exact (Submodule.mem_annihilator_span_singleton _ _).mpr hrx
42     exact hr (hx hrann)
43   obtain ⟨q, hqprime, hIq, hqdisj⟩ := I.exists_le_prime_disjoint _ hdisj
44   let q' : PrimeSpectrum C := ⟨q, hqprime⟩
45   have hxC : q' ∈ Module.support C (LinearMap.ker f) := by
46     rw [Module.mem_support_iff_exists_annihilator]
47     exact ⟨xC, show (C ∙ xC).annihilator ≤ q from hIq⟩
48   have hcokerC : q' ∈ Module.support C (E ⧸ LinearMap.range f) :=
49     endomorphism_kernel_support_subset_cokernel_support f hxC
50   obtain ⟨y, hy⟩ := Module.mem_support_iff_exists_annihilator.mp hcokerC
51   refine ⟨y, ?_⟩
52   intro r hr
53   have hmap : algebraMap R C r ∈ (C ∙ y).annihilator := by
54     rw [Submodule.mem_annihilator_span_singleton]
55     have hry : r • y = 0 :=
56       (Submodule.mem_annihilator_span_singleton _ _).mp hr
57     simpa [IsScalarTower.algebraMap_smul C r y] using hry
58   by_contra hrp
59   exact Set.disjoint_left.mp hqdisj (hy hmap) ⟨r, hrp, rfl⟩
