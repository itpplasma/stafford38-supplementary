6 /-!
7 # Kernel and cokernel equivalences for module localization
8 
9 The actual localization functor preserves the two elementary constructions
10 needed by the successor pages.  The equivalences below are obtained from the
11 canonical submodule and quotient localization equivalences in Mathlib.
12 -/
13 
14 namespace AlgebraicAnalysis.LocalizedKernelCokernelEquivalences
15 
16 universe u v w
17 
18 variable {R : Type u} [CommRing R]
19 variable {U : Type v} {V : Type w}
20 variable [AddCommGroup U] [AddCommGroup V]
21 variable [Module R U] [Module R V]
22 
23 noncomputable section
24 
25 variable (S : Submonoid R) (f : U →ₗ[R] V)
26 
27 /-- The localization of an `R`-linear map, regarded as a map over the
28 localized ring. -/
29 noncomputable def localizedMap : LocalizedModule S U →ₗ[Localization S]
30     LocalizedModule S V :=
31   (LocalizedModule.map S f).extendScalarsOfIsLocalization S (Localization S)
32 
33 @[simp] theorem localizedMap_apply (x : LocalizedModule S U) :
34     localizedMap S f x = LocalizedModule.map S f x :=
35   rfl
36 
37 /-- Localization carries an actual linear equivalence to an equivalence over
38 the localized ring. -/
39 noncomputable def localizedEquiv (e : U ≃ₗ[R] V) :
40     LocalizedModule S U ≃ₗ[Localization S] LocalizedModule S V :=
41   LinearEquiv.ofBijective (localizedMap S e.toLinearMap) (by
42     constructor
43     · simpa [localizedMap] using
44         (LocalizedModule.map_injective S e.toLinearMap e.injective)
45     · simpa [localizedMap] using
46         (LocalizedModule.map_surjective S e.toLinearMap e.surjective))
47 
48 /-- Localization commutes with kernels. -/
49 noncomputable def localizedKernelEquiv :
50     LocalizedModule S (LinearMap.ker f) ≃ₗ[Localization S]
51       LinearMap.ker (localizedMap S f) := by
52   let e₁ : (LinearMap.ker f).localized' (Localization S) S
53       (LocalizedModule.mkLinearMap S U) ≃ₗ[Localization S]
54       LocalizedModule S (LinearMap.ker f) :=
55     Submodule.localizedEquiv S (LinearMap.ker f)
56   let e₂ : (LinearMap.ker f).localized' (Localization S) S
57       (LocalizedModule.mkLinearMap S U) ≃ₗ[Localization S]
58       LinearMap.ker (localizedMap S f) :=
59     LinearEquiv.ofEq _ _
60       (LinearMap.localized'_ker_eq_ker_localizedMap (Localization S) S
61         (LocalizedModule.mkLinearMap S U) (LocalizedModule.mkLinearMap S V) f)
62   exact e₁.symm.trans e₂
63 
64 /-- Localization commutes with cokernels. -/
65 noncomputable def localizedCokernelEquiv :
66     LocalizedModule S (V ⧸ LinearMap.range f) ≃ₗ[Localization S]
67       LocalizedModule S V ⧸ LinearMap.range (localizedMap S f) := by
68   let h : Submodule.localized S (LinearMap.range f) =
69       LinearMap.range (localizedMap S f) := by
70     have hm :
71         IsLocalizedModule.map S (LocalizedModule.mkLinearMap S U)
72             (LocalizedModule.mkLinearMap S V) f = LocalizedModule.map S f := by
73       ext x
74       induction x using LocalizedModule.induction_on with
75       | _ x s =>
76         rw [IsLocalizedModule.map_LocalizedModules]
77         exact (LocalizedModule.map_mk S f x s).symm
78     simpa [localizedMap, hm] using
79       (LinearMap.localized'_range_eq_range_localizedMap
80         (Localization S) S (LocalizedModule.mkLinearMap S U)
81         (LocalizedModule.mkLinearMap S V) f)
82   let e₁ : (LocalizedModule S V ⧸
83       Submodule.localized S (LinearMap.range f)) ≃ₗ[Localization S]
84       LocalizedModule S (V ⧸ LinearMap.range f) :=
85     localizedQuotientEquiv S (LinearMap.range f)
86   let e : LocalizedModule S V ≃ₗ[Localization S] LocalizedModule S V :=
87     LinearEquiv.refl _ _
88   exact e₁.symm.trans
89     (Submodule.Quotient.equiv _ _ e (by simpa [e] using h))
90 
91 end
92 end AlgebraicAnalysis.LocalizedKernelCokernelEquivalences
