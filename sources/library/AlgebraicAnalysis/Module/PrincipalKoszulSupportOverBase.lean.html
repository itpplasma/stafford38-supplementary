7 /-!
8 # Principal Koszul positivity after restriction of scalars
9 
10 Support and torsion are computed over the coefficient algebra `C`; the
11 resulting length inequality is measured over the base ring `R`.  No finite
12 generation of `E` over `R` is needed: only the first `R`-kernel has finite
13 length.
14 -/

31 theorem length_cokernel_gt_kernel_of_support_over_base
32     [IsNoetherianRing C] [Module.Finite C E]
33     (x : C) (p q : PrimeSpectrum C)
34     (hp : p ∈ Module.support C E) (hpq : p ≤ q)
35     (hxp : x ∉ p.asIdeal)
36     (hq : q ∈ Module.support C (QuotSMulTop x E))
37     (hfinite : IsFiniteLength R
38       (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R))) :
39     Module.length R (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R)) >
40       Module.length R (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R)) := by
41   let fC : Module.End C E := LinearMap.lsmul C E x
42   let fR : Module.End R E := fC.restrictScalars R
43   obtain ⟨n, hn⟩ := exists_stable_kernel_power_smul (E := E) x
44   have hpow : ∀ m, fR ^ m = (fC ^ m).restrictScalars R := by
45     intro m
46     induction m with
47     | zero => ext z; simp
48     | succ m ih =>
49         rw [pow_succ, pow_succ, ih]
50         ext z
51         simp [fR, LinearMap.restrictScalars_apply, Module.End.mul_apply]
52   have hstable : LinearMap.ker (fR ^ n) = LinearMap.ker (fR ^ (n + 1)) := by
53     rw [hpow, hpow, LinearMap.ker_restrictScalars, LinearMap.ker_restrictScalars]
54     exact congrArg (Submodule.restrictScalars R) hn
55   have hnonzeroC : Nontrivial (E ⧸ (LinearMap.ker (fC ^ n) ⊔ fC.range)) :=
56     residual_nontrivial_of_support x n p q hp hpq hxp hq
57   let PC : Submodule C E := LinearMap.ker (fC ^ n) ⊔ fC.range
58   let PR : Submodule R E := LinearMap.ker (fR ^ n) ⊔ fR.range
59   have hPR : PR = PC.restrictScalars R := by
60     dsimp [PR, PC]
61     rw [show fR ^ n = (fC ^ n).restrictScalars R from hpow n]
62     simp only [LinearMap.ker_restrictScalars, Submodule.restrictScalars_sup]
63     rfl
64   have hnonzeroR : Nontrivial (E ⧸ PR) := by
65     have hPC : PC ≠ (⊤ : Submodule C E) :=
66       Submodule.Quotient.nontrivial_iff.mp hnonzeroC
67     apply Submodule.Quotient.nontrivial_iff.mpr
68     intro htop
69     apply hPC
70     apply (Submodule.restrictScalars_eq_top_iff R C E).mp
71     rw [← hPR, htop]
72   have hresult :=
73     length_cokernel_gt_kernel_of_stable_power_and_nonzero
74       (R := R) fR n hfinite hstable hnonzeroR
75   simpa [fR, fC] using hresult
