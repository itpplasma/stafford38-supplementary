8 # Kernel support is contained in cokernel support
9 
10 The only input is the Hopfian property of a finite module over a commutative
11 Noetherian ring.  The proof is deliberately made at a prime, using the
12 actual `LocalizedModule` support definition.
13 -/

17 theorem endomorphism_kernel_support_subset_cokernel_support
18     {R E : Type*} [CommRing R] [AddCommGroup E] [Module R E]
19     [IsNoetherianRing R] [Module.Finite R E] (f : Module.End R E) :
20     Module.support R (LinearMap.ker f) ⊆
21       Module.support R (E ⧸ LinearMap.range f) := by
22   intro p hp
23   rw [Module.mem_support_iff'] at hp ⊢
24   by_contra h
25   have hnot : p ∉ Module.support R (E ⧸ LinearMap.range f) := by
26     rw [Module.mem_support_iff']
27     exact h
28   have h := Module.notMem_support_iff.mp hnot
29   have hsurj : Function.Surjective
30       (LocalizedModule.map p.asIdeal.primeCompl f) :=
31     (LinearMap.localizedMap_surjective_iff_subsingleton_localized_coker
32       p.asIdeal.primeCompl f).2 h
33   let : IsNoetherian (Localization p.asIdeal.primeCompl)
34       (LocalizedModule p.asIdeal.primeCompl E) := by infer_instance
35   have hinj : Function.Injective
36       (LocalizedModule.map p.asIdeal.primeCompl f) :=
37     IsNoetherian.injective_of_surjective_endomorphism _ hsurj
38   obtain ⟨x, hx⟩ := hp
39   let x' : LocalizedModule p.asIdeal.primeCompl E :=
40     LocalizedModule.mk x.1 1
41   have hxzero : x' = 0 := by
42     apply hinj
43     dsimp [x']
44     rw [LocalizedModule.map_mk]
45     simp [LinearMap.mem_ker.mp x.2]
46   have hxzero' :
47       IsLocalizedModule.mk' (LocalizedModule.mkLinearMap p.asIdeal.primeCompl E)
48         x.1 (1 : p.asIdeal.primeCompl) = 0 := by
49     rw [← IsLocalizedModule.mk_eq_mk']
50     exact hxzero
51   obtain ⟨s, hs⟩ :=
52     (IsLocalizedModule.mk'_eq_zero'
53       (LocalizedModule.mkLinearMap p.asIdeal.primeCompl E) (1 : p.asIdeal.primeCompl)).mp
54       hxzero'
55   have hs' : (s.1 : R) • x = 0 := by
56     apply Subtype.ext
57     exact hs
58   exact hx s.1 s.2 hs'
