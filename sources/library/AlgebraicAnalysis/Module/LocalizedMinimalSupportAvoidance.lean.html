8 /-!
9 # Transport of minimal-prime support avoidance through localization
10 
11 For a finite module, localization commutes with its annihilator.  The reverse
12 inclusion uses a finite generating set to clear all denominators at once.
13 Minimal-prime avoidance then follows from the ordinary minimal-prime
14 correspondence for a localization.
15 -/

23 /-- The annihilator of a finite module commutes with localization. -/
24 theorem annihilator_localizedModule
25     [Module.Finite C E] (S : Submonoid C) :
26     Module.annihilator (Localization S) (LocalizedModule S E) =
27       Ideal.map (algebraMap C (Localization S)) (Module.annihilator C E) := by
28   apply le_antisymm
29   · intro z hz
30     obtain ⟨⟨c, d⟩, rfl⟩ := IsLocalization.mk'_surjective S z
31     obtain ⟨gens, hgens⟩ := (inferInstance : Module.Finite C E)
32     have hclear : ∀ e ∈ gens, ∃ u : S, u • (c • e) = 0 := by
33       intro e he
34       have hzero :
35           IsLocalization.mk' (Localization S) c d • LocalizedModule.mk e 1 = 0 :=
36         Module.mem_annihilator.mp hz _
37       rw [IsLocalizedModule.mk_eq_mk'] at hzero
38       rw [IsLocalizedModule.mk'_smul_mk', IsLocalizedModule.mk'_eq_zero'] at hzero
39       exact hzero
40     choose u hu using hclear
41     let t : S := gens.attach.prod fun e => u e e.property
42     have htc : (t : C) * c ∈ Module.annihilator C E := by
43       rw [Module.mem_annihilator]
44       intro e
45       refine Submodule.span_induction ?_ (smul_zero _) ?_ ?_
46         (show e ∈ Submodule.span C (gens : Set E) by rw [hgens]; trivial)
47       · intro e he
48         obtain ⟨v, hv⟩ := Finset.dvd_prod_of_mem
49           (fun a : gens => u a a.property) (Finset.mem_attach gens ⟨e, he⟩)
50         have hue : (((u e he : S) : C) * c) • e = 0 := by
51           simpa only [mul_smul, Submonoid.smul_def] using hu e he
52         change (((t : S) : C) * c) • e = 0
53         rw [show t = gens.attach.prod (fun a => u a a.property) by rfl, hv,
54           Submonoid.coe_mul, mul_comm ((u e he : S) : C) (v : C),
55           mul_assoc, mul_smul, hue, smul_zero]
56       · intro e₁ e₂ _ _ he₁ he₂
57         rw [smul_add, he₁, he₂, zero_add]
58       · intro a e _ he
59         rw [smul_comm, he, smul_zero]
60     rw [IsLocalization.mk'_mem_map_algebraMap_iff]
61     exact ⟨t, t.property, htc⟩
62   · rw [Ideal.map_le_iff_le_comap]
63     intro c hc
64     rw [Ideal.mem_comap, Module.mem_annihilator]
65     intro y
66     induction y using LocalizedModule.induction_on with
67     | _ e d =>
68       rw [IsLocalizedModule.mk_eq_mk']
69       rw [← IsLocalization.mk'_one (M := S), IsLocalizedModule.mk'_smul_mk',
70         Module.mem_annihilator.mp hc e]
71       simp

73 /-- Minimal-prime avoidance survives localization. -/
74 @[nolint unusedArguments]
75 theorem localized_minimalPrime_avoids
76     [IsNoetherianRing C] [Module.Finite C E]
77     (S : Submonoid C) (x : C)
78     (havoid : ∀ p ∈ (Module.annihilator C E).minimalPrimes, x ∉ p) :
79     ∀ Q ∈ (Module.annihilator (Localization S)
80         (LocalizedModule S E)).minimalPrimes,
81       algebraMap C (Localization S) x ∉ Q := by
82   intro Q hQ
83   have hQmap : Q ∈
84       (Ideal.map (algebraMap C (Localization S))
85         (Module.annihilator C E)).minimalPrimes := by
86     rw [← annihilator_localizedModule S]
87     exact hQ
88   have hQunder :
89       Ideal.under C Q ∈ (Module.annihilator C E).minimalPrimes := by
90     rw [IsLocalization.minimalPrimes_map S
91       (Localization S) (Module.annihilator C E)] at hQmap
92     exact hQmap
93   intro hxQ
94   exact havoid (Ideal.under C Q) hQunder hxQ
95 
96 end
