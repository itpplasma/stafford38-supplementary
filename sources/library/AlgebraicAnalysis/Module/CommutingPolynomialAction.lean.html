7 # Polynomial actions generated by a commuting family of endomorphisms
8 
9 The target of `MvPolynomial.aeval` is required to be commutative.  A family of
10 pairwise commuting endomorphisms therefore gives an evaluation map by first
11 landing in the commutative subalgebra which it generates.
12 -/

23 private theorem range_commutative (v : σ → Module.End k V)
24     (hcomm : ∀ i j, Commute (v i) (v j)) :
25     ∀ x ∈ Set.range v, ∀ y ∈ Set.range v, x * y = y * x := by
26   rintro _ ⟨i, rfl⟩ _ ⟨j, rfl⟩
27   exact (hcomm i j).eq

29 /-- Evaluation of multivariate polynomials at a pairwise commuting family of
30 endomorphisms. -/
31 noncomputable def commutingPolynomialAction {σ : Type w}
32     (v : σ → Module.End k V)
33     (hcomm : ∀ i j, Commute (v i) (v j)) :
34     MvPolynomial σ k →ₐ[k] Module.End k V := by
35   let S : Subalgebra k (Module.End k V) := Algebra.adjoin k (Set.range v)
36   letI : IsMulCommutative S :=
37     Algebra.isMulCommutative_adjoin k (by
38     intro x hx y hy hne
39     rcases hx with ⟨i, rfl⟩
40     rcases hy with ⟨j, rfl⟩
41     exact hcomm i j)
42   exact
43     (S.val).comp
44       (MvPolynomial.aeval (fun i =>
45         ⟨v i, Algebra.subset_adjoin ⟨i, rfl⟩⟩))

47 @[simp] theorem commutingPolynomialAction_apply_X {σ : Type w}
48     (v : σ → Module.End k V) (hcomm : ∀ i j, Commute (v i) (v j)) (i : σ) :
49     commutingPolynomialAction v hcomm (MvPolynomial.X i) = v i := by
50   classical
51   simp [commutingPolynomialAction]

53 /-- Evaluation sends constants to scalar endomorphisms. This named
54 compatibility lemma is retained even though generic algebra-hom simplification
55 can also discharge its left-hand side. -/
56 @[simp, nolint simpNF] theorem commutingPolynomialAction_apply_C {σ : Type w}
57     (v : σ → Module.End k V) (hcomm : ∀ i j, Commute (v i) (v j)) (a : k) :
58     commutingPolynomialAction v hcomm (MvPolynomial.C a) = algebraMap k (Module.End k V) a := by
59   classical
60   simp [commutingPolynomialAction]

62 theorem commutingPolynomialAction_intertwines {σ : Type w}
63     {W : Type z} [AddCommGroup W] [Module k W]
64     (v : σ → Module.End k V) (w : σ → Module.End k W)
65     (hv : ∀ i j, Commute (v i) (v j))
66     (hw : ∀ i j, Commute (w i) (w j))
67     (f : V →ₗ[k] W)
68     (hgen : ∀ i, f.comp (v i) = (w i).comp f) (p : MvPolynomial σ k) :
69     f.comp (commutingPolynomialAction v hv p) =
70       (commutingPolynomialAction w hw p).comp f := by
71   classical
72   induction p using MvPolynomial.induction_on with
73   | C a =>
74       apply LinearMap.ext
75       intro x
76       simp only [LinearMap.comp_apply, commutingPolynomialAction_apply_C]
77       change f (a • x) = a • f x
78       exact f.map_smul a x
79   | add p q hp hq =>
80       rw [map_add, map_add, LinearMap.comp_add, LinearMap.add_comp, hp, hq]
81   | mul_X p i hp =>
82       apply LinearMap.ext
83       intro x
84       simp only [map_mul, LinearMap.comp_apply, Module.End.mul_apply,
85         commutingPolynomialAction_apply_X]
86       calc
87         f ((commutingPolynomialAction v hv p) (v i x)) =
88             (commutingPolynomialAction w hw p) (f (v i x)) :=
89           congrArg (fun g : V →ₗ[k] W => g (v i x)) hp
90         _ = (commutingPolynomialAction w hw p) (w i (f x)) := by
91           have hgi := congrArg (fun g : V →ₗ[k] W => g x) (hgen i)
92           change f (v i x) = w i (f x) at hgi
93           rw [hgi]

95 /-- The polynomial action as a module structure on the original space. -/
96 noncomputable def commutingPolynomialModule {σ : Type w}
97     (v : σ → Module.End k V) (hcomm : ∀ i j, Commute (v i) (v j)) :
98     Module (MvPolynomial σ k) V :=
99   Module.compHom V (commutingPolynomialAction v hcomm).toRingHom
