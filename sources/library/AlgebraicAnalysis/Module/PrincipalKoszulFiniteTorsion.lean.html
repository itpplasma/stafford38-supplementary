6 /-!
7 # Finite power torsion for a principal Koszul endomorphism
8 
9 This file isolates the reusable algebra needed after tangential localization.
10 The ambient module need not be finite over the ring in which lengths are
11 measured.  It is enough that the first kernel has finite length: all finite
12 power kernels then have finite length by devissage.
13 
14 The final theorem deliberately assumes that the actual residual quotient is
15 nonzero.  It proves only the strict length inequality and makes no
16 noncharacteristic-support or geometric nonvanishing claim.
17 -/

31 /-- If the first kernel of an endomorphism is Noetherian, then so is the
32 kernel of every finite power.  No finiteness hypothesis is imposed on the
33 ambient module. -/
34 theorem isNoetherian_kernel_power (f : Module.End R E)
35     [IsNoetherian R f.ker] (n : ℕ) : IsNoetherian R (f ^ n).ker := by
36   induction n with
37   | zero =>
38     rw [pow_zero, Module.End.one_eq_id, LinearMap.ker_id]
39     infer_instance
40   | succ n ih =>
41     let := ih
42     have hle : f.ker ≤ (f ^ (n + 1)).ker := by
43       intro z hz
44       rw [LinearMap.mem_ker, pow_succ, Module.End.mul_apply,
45         LinearMap.mem_ker.mp hz, map_zero]
46     let i : f.ker →ₗ[R] (f ^ (n + 1)).ker := Submodule.inclusion hle
47     let g : (f ^ (n + 1)).ker →ₗ[R] (f ^ n).ker := f.restrict (by
48       intro z hz
49       rw [LinearMap.mem_ker] at hz ⊢
50       simpa [pow_succ, Module.End.mul_apply] using hz)
51     have hexact : i.range = g.ker := by
52       ext z
53       constructor
54       · rintro ⟨w, rfl⟩
55         apply Subtype.ext
56         exact LinearMap.mem_ker.mp w.property
57       · intro hz
58         have hz0 : f z.val = 0 := congrArg Subtype.val (LinearMap.mem_ker.mp hz)
59         exact ⟨⟨z.val, hz0⟩, rfl⟩
60     exact isNoetherian_of_range_eq_ker i g hexact

62 /-- Finite length of the first kernel propagates to every finite power
63 kernel, even when the ambient module is not finite over `R`. -/
64 theorem isFiniteLength_kernel_power (f : Module.End R E)
65     (hfinite : IsFiniteLength R f.ker) (n : ℕ) :
66     IsFiniteLength R (f ^ n).ker := by
67   have hfirst := isFiniteLength_iff_isNoetherian_isArtinian.mp hfinite
68   let : IsNoetherian R f.ker := hfirst.1
69   let : IsArtinian R f.ker := hfirst.2
70   exact isFiniteLength_iff_isNoetherian_isArtinian.mpr
71     ⟨isNoetherian_kernel_power f n, isArtinian_kernel_power f n⟩

73 /-- A stable finite power kernel contributes the same length to the kernel and
74 cokernel.  If the actual quotient left after removing that torsion and the
75 range is nonzero, the principal Koszul Euler length is strictly positive.
76 
77 The nonzero quotient is an explicit input; this theorem does not manufacture
78 it from a support or noncharacteristic hypothesis. -/
79 theorem length_cokernel_gt_kernel_of_stable_power_and_nonzero
80     (f : Module.End R E) (n : ℕ)
81     (hfinite : IsFiniteLength R f.ker)
82     (hstable : LinearMap.ker (f ^ n) = LinearMap.ker (f ^ (n + 1)))
83     (hnonzero : Nontrivial (E ⧸ ((f ^ n).ker ⊔ f.range))) :
84     Module.length R (E ⧸ f.range) > Module.length R f.ker := by
85   have hfirstparts := isFiniteLength_iff_isNoetherian_isArtinian.mp hfinite
86   let : IsNoetherian R f.ker := hfirstparts.1
87   let : IsArtinian R f.ker := hfirstparts.2
88   let T : Submodule R E := (f ^ n).ker
89   have hTfinite : IsFiniteLength R T := isFiniteLength_kernel_power f hfinite n
90   have hTparts := isFiniteLength_iff_isNoetherian_isArtinian.mp hTfinite
91   let : IsNoetherian R T := hTparts.1
92   let : IsArtinian R T := hTparts.2
93   have hcomap : T.comap f = T := by
94     ext z
95     rw [Submodule.mem_comap]
96     change (f ^ n) (f z) = 0 ↔ (f ^ n) z = 0
97     rw [← Module.End.mul_apply, (Commute.self_pow f n).eq.symm,
98       Module.End.mul_apply]
99     simpa [Module.End.pow_apply, Function.iterate_succ_apply'] using
100       congrArg (fun K : Submodule R E ↦ z ∈ K) hstable.symm
101   let : Nontrivial (E ⧸ (T ⊔ f.range)) := hnonzero
102   have hpositive : 0 < Module.length R (E ⧸ (T ⊔ f.range)) :=
103     Module.length_pos
104   rw [length_cokernel_eq_kernel_add_regular_quotient f T hcomap]
105   simpa [add_comm] using
106     ENat.lt_add_left (Module.length_ne_top (R := R) (M := f.ker)) hpositive
