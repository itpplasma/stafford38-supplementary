9 # Uniform vanishing of an ascending family of boundary maps
10 
11 Pointwise eventual vanishing becomes uniform on a Noetherian source.  The
12 localized statement deliberately assumes Noetherianity only after

20 theorem exists_uniform_zero_of_noetherian
21     {R U : Type*} [Semiring R] [AddCommMonoid U] [Module R U]
22     {V : ℕ → Type*} [∀ r, AddCommMonoid (V r)] [∀ r, Module R (V r)]
23     (b : ∀ r, U →ₗ[R] V r)
24     (hmono : ∀ r s, r ≤ s → LinearMap.ker (b r) ≤ LinearMap.ker (b s))
25     (hpoint : ∀ u, ∃ r, b r u = 0) [IsNoetherian R U] :
26     ∃ r, b r = 0 := by
27   let F : ℕ →o Submodule R U :=
28     { toFun := fun r => LinearMap.ker (b r)
29       monotone' := fun r s hrs => hmono r s hrs }
30   obtain ⟨n, hn⟩ := monotone_stabilizes_iff_noetherian.mpr inferInstance F
31   refine ⟨n, LinearMap.ext fun u => ?_⟩
32   obtain ⟨r, hr⟩ := hpoint u
33   have hu : u ∈ F r := LinearMap.mem_ker.mpr hr
34   have hu' : u ∈ F (max n r) :=
35     (hmono r (max n r) (Nat.le_max_right _ _)) hu
36   have heq : F n = F (max n r) := hn _ (Nat.le_max_left _ _)
37   have : u ∈ F n := heq.symm ▸ hu'
38   exact LinearMap.mem_ker.mp this

40 theorem exists_uniform_subsingleton_of_noetherian
41     {R U : Type*} [Semiring R] [AddCommMonoid U] [Module R U]
42     {V : ℕ → Type*} [∀ r, AddCommMonoid (V r)] [∀ r, Module R (V r)]
43     (b : ∀ r, U →ₗ[R] V r)
44     (hmono : ∀ r s, r ≤ s → LinearMap.ker (b r) ≤ LinearMap.ker (b s))
45     (hpoint : ∀ u, ∃ r, b r u = 0)
46     (hsurj : ∀ r, Function.Surjective (b r)) [IsNoetherian R U] :
47     ∃ r, Subsingleton (V r) := by
48   obtain ⟨r, hr⟩ := exists_uniform_zero_of_noetherian b hmono hpoint
49   refine ⟨r, ?_⟩
50   constructor
51   intro x y
52   obtain ⟨u, rfl⟩ := hsurj r x
53   obtain ⟨v, rfl⟩ := hsurj r y
54   simp [hr]

62 private theorem localized_ker_mono
63     (b : ∀ r, U →ₗ[R] V r)
64     (hmono : ∀ r s, r ≤ s → LinearMap.ker (b r) ≤ LinearMap.ker (b s))
65     {r s : ℕ} (hrs : r ≤ s) :
66     LinearMap.ker (LocalizedModule.map S (b r)) ≤
67       LinearMap.ker (LocalizedModule.map S (b s)) := by
68   intro z hz
69   induction z using LocalizedModule.induction_on with
70   | _ u t =>
71       have hz' := LinearMap.mem_ker.mp hz
72       rw [LocalizedModule.map_mk] at hz'
73       change LocalizedModule.mk (b r u) t = 0 at hz'
74       apply LinearMap.mem_ker.mpr
75       rw [LocalizedModule.map_mk]
76       change LocalizedModule.mk (b s u) t = 0
77       rw [IsLocalizedModule.mk_eq_mk' (S := S)] at hz' ⊢
78       rw [IsLocalizedModule.mk'_eq_zero' (LocalizedModule.mkLinearMap S (V r))] at hz'
79       obtain ⟨a, ha⟩ := hz'
80       have hscaled : b r (a • u) = 0 := by
81         simpa [map_smul] using ha
82       have hscaled' : b s (a • u) = 0 :=
83         LinearMap.mem_ker.mp (hmono r s hrs (LinearMap.mem_ker.mpr hscaled))
84       rw [IsLocalizedModule.mk'_eq_zero']
85       exact ⟨a, by simpa [map_smul] using hscaled'⟩

87 theorem exists_uniform_zero_localized
88     (b : ∀ r, U →ₗ[R] V r)
89     (hmono : ∀ r s, r ≤ s → LinearMap.ker (b r) ≤ LinearMap.ker (b s))
90     (hpoint : ∀ u, ∃ r, b r u = 0)
91     [IsNoetherian (Localization S) (LocalizedModule S U)] :
92     ∃ r, LocalizedModule.map S (b r) = 0 := by
93   let B : ∀ r, LocalizedModule S U →ₗ[Localization S]
94       LocalizedModule S (V r) := fun r => LocalizedModule.map S (b r)
95   have hpoint' : ∀ u, ∃ r, B r u = 0 := by
96     intro u
97     induction u using LocalizedModule.induction_on with
98     | _ m s =>
99         obtain ⟨r, hr⟩ := hpoint m
100         refine ⟨r, ?_⟩
101         simp [B, LocalizedModule.map_mk, hr]
102   obtain ⟨r, hr⟩ := exists_uniform_zero_of_noetherian B
103     (fun r s hrs => by simpa only [B] using localized_ker_mono S b hmono hrs) hpoint'
104   exact ⟨r, by simpa [B] using hr⟩

106 theorem exists_uniform_subsingleton_localized
107     (b : ∀ r, U →ₗ[R] V r)
108     (hmono : ∀ r s, r ≤ s → LinearMap.ker (b r) ≤ LinearMap.ker (b s))
109     (hpoint : ∀ u, ∃ r, b r u = 0)
110     (hsurj : ∀ r, Function.Surjective (b r))
111     [IsNoetherian (Localization S) (LocalizedModule S U)] :
112     ∃ r, Subsingleton (LocalizedModule S (V r)) := by
113   let B : ∀ r, LocalizedModule S U →ₗ[Localization S]
114       LocalizedModule S (V r) := fun r => LocalizedModule.map S (b r)
115   have hsurj' : ∀ r, Function.Surjective (B r) := by
116     intro r
117     exact LocalizedModule.map_surjective S (b r) (hsurj r)
118   exact exists_uniform_subsingleton_of_noetherian B
119     (fun r s hrs => by simpa [B] using localized_ker_mono S b hmono hrs) (by
120       intro u
121       induction u using LocalizedModule.induction_on with
122       | _ m s =>
123           obtain ⟨r, hr⟩ := hpoint m
124           refine ⟨r, ?_⟩
125           simp [B, LocalizedModule.map_mk, hr]) hsurj'
