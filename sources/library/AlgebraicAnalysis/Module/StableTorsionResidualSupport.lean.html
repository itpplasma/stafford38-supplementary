8 # Support detects the residual after stable scalar torsion
9 
10 The support argument is most transparent before localization: the exact
11 sequence for the power kernel splits support into torsion and residual parts,
12 and `support_quotSMulTop` then applies at the larger prime.
13 -/

23 variable {R : Type u} {E : Type v}

26 abbrev scalarEnd (x : R) : E →ₗ[R] E := LinearMap.lsmul R E x
27 
28 @[nolint unusedArguments]
29 theorem residual_nontrivial_of_support
30     [IsNoetherianRing R] [Module.Finite R E]
31     (x : R) (n : ℕ) (p q : PrimeSpectrum R)
32     (hp : p ∈ Module.support R E) (hpq : p ≤ q)
33     (hxp : x ∉ p.asIdeal)
34     (hq : q ∈ Module.support R (QuotSMulTop x E)) :
35     Nontrivial (E ⧸ (LinearMap.ker (scalarEnd (E := E) x ^ n) ⊔
36       (scalarEnd (E := E) x).range)) := by
37   let f := scalarEnd (E := E) x
38   let T : Submodule R E := LinearMap.ker (f ^ n)
39   have hpower : ∀ m (z : E), (f ^ m) z = x ^ m • z := by
40     intro m
41     induction m with
42     | zero => intro z; simp
43     | succ m ih =>
44         intro z
45         simp [pow_succ, Module.End.mul_apply, ih, f, scalarEnd, smul_smul]
46   have hTann : x ^ n ∈ Module.annihilator R T := by
47     rw [Module.mem_annihilator]
48     intro z
49     have hz := z.property
50     apply Subtype.ext
51     change x ^ n • (z : E) = 0
52     change (f ^ n) (z : E) = 0 at hz
53     exact (hpower n (z : E)).symm.trans hz
54   have hpT : p ∉ Module.support R T := by
55     intro h
56     have hle := Module.annihilator_le_of_mem_support h
57     have hpow : x ^ n ∈ p.asIdeal := hle hTann
58     exact hxp (p.isPrime.mem_of_pow_mem n hpow)
59   have hexact : Function.Exact T.subtype T.mkQ :=
60     LinearMap.exact_subtype_mkQ T
61   have hsupport : Module.support R E =
62       Module.support R T ∪ Module.support R (E ⧸ T) :=
63     Module.support_of_exact hexact T.subtype_injective T.mkQ_surjective
64   have hpquot : p ∈ Module.support R (E ⧸ T) := by
65     have hpmem : p ∈ Module.support R T ∪ Module.support R (E ⧸ T) := by
66       rw [← hsupport]
67       exact hp
68     exact hpmem.resolve_left hpT
69   have hqquot : q ∈ Module.support R (E ⧸ T) :=
70     Module.mem_support_mono hpq hpquot
71   have hqzero : q ∈ PrimeSpectrum.zeroLocus ({x} : Set R) := by
72     rw [Module.support_quotSMulTop (M := E) x] at hq
73     exact hq.2
74   have hqres : q ∈ Module.support R (QuotSMulTop x (E ⧸ T)) := by
75     rw [Module.support_quotSMulTop]
76     exact ⟨hqquot, hqzero⟩
77   have hglobal : Nontrivial (QuotSMulTop x (E ⧸ T)) :=
78     Module.nonempty_support_iff.mp ⟨q, hqres⟩
79   have hmap : Submodule.map T.mkQ (f.range) =
80       x • (⊤ : Submodule R (E ⧸ T)) := by
81     ext z
82     constructor
83     · rintro ⟨y, ⟨w, rfl⟩, rfl⟩
84       simpa [f, scalarEnd] using
85         (Submodule.smul_mem_pointwise_smul (T.mkQ w) x (⊤ : Submodule R (E ⧸ T)) trivial)
86     · intro hz
87       have hz' : z ∈ Ideal.span ({x} : Set R) • (⊤ : Submodule R (E ⧸ T)) := by
88         rw [Submodule.ideal_span_singleton_smul]
89         exact hz
90       refine Submodule.smul_induction_on hz' ?_ (fun a b ha hb ↦ add_mem ha hb)
91       intro r hr z hz
92       rcases (Ideal.mem_span_singleton.mp hr) with ⟨c, rfl⟩
93       refine Submodule.Quotient.induction_on T z ?_
94       intro w
95       refine ⟨f (c • w), ⟨c • w, rfl⟩, ?_⟩
96       simp [f, scalarEnd, smul_smul, mul_comm]
97   have hequiv :
98       ((E ⧸ T) ⧸ Submodule.map T.mkQ f.range) ≃ₗ[R]
99         E ⧸ (T ⊔ f.range) :=
100     Submodule.quotientQuotientEquivQuotientSup T f.range
101   have hnontriv : Nontrivial ((E ⧸ T) ⧸ Submodule.map T.mkQ f.range) := by
102     rw [hmap]
103     exact Submodule.Quotient.nontrivial_iff.mpr (by
104       intro htop
105       exact not_subsingleton_iff_nontrivial.mpr hglobal
106         (Submodule.Quotient.subsingleton_iff.mpr htop))
107   let : Nontrivial ((E ⧸ T) ⧸ Submodule.map T.mkQ f.range) := hnontriv
108   have hfinal := Equiv.nontrivial hequiv.symm.toEquiv
109   simpa [T, f] using hfinal
