29 theorem targetBoundaryMap_naturality (r : ℕ) (p : ℤ) :
30     K.targetBoundaryMap r (p - d) ∘ₗ P.targetMap 1 p =
31       P.targetMap (r + 1) p ∘ₗ K.targetBoundaryMap r p := by
32   apply LinearMap.ext
33   intro y
34   refine Submodule.Quotient.induction_on
35     ((K.boundaries 1 p).comap (K.G p).subtype) y ?_
36   intro x
37   simp only [LinearMap.comp_apply, P.targetMap_mk, K.targetBoundaryMap_mk]
38 
39 theorem totalBoundaryMap_naturality (r : ℕ) :
40     K.totalBoundaryMap r ∘ₗ P.targetTotalMap 1 =
41       P.targetTotalMap (r + 1) ∘ₗ K.totalBoundaryMap r := by
42   apply LinearMap.ext
43   intro x
44   induction x using DirectSum.induction_on with
45   | zero => simp
46   | of p y =>
47       rw [show DirectSum.of (fun q : ℤ => K.TargetPage 1 q) p y =
48         DirectSum.lof k ℤ (fun q : ℤ => K.TargetPage 1 q) p y from rfl]
49       change K.totalBoundaryMap r (P.targetTotalMap 1
50         (DirectSum.lof k ℤ (fun q : ℤ => K.TargetPage 1 q) p y)) =
51         P.targetTotalMap (r + 1) (K.totalBoundaryMap r
52           (DirectSum.lof k ℤ (fun q : ℤ => K.TargetPage 1 q) p y))
53       rw [P.targetTotalMap_lof, K.totalBoundaryMap_lof,
54         K.totalBoundaryMap_lof, P.targetTotalMap_lof]
55       exact congrArg (fun z =>
56         DirectSum.lof k ℤ (fun q : ℤ => K.TargetPage (r + 1) q) (p - d) z)
57         (LinearMap.congr_fun (P.targetBoundaryMap_naturality r p) y)
58   | add x y hx hy => simpa using congrArg₂ (· + ·) hx hy
