8 # Algebraic hyperplane restriction
9 
10 For a finite module over a commutative ring, surjectivity of multiplication
11 by an element forces the module support to avoid the corresponding principal
12 hypersurface. The proof is the determinant trick and is independent of any
13 filtered or differential-operator application.
14 -/

22 /-- Degree-zero restriction to the principal hypersurface defined by `x`. -/
23 abbrev Restriction (x : R) : Type _ :=
24   M ⧸ (Ideal.span {x} • (⊤ : Submodule R M))
25 
26 /-- Restriction vanishes exactly when multiplication by `x` is surjective. -/
27 theorem restriction_subsingleton_iff_smul_surjective {x : R} :
28     Subsingleton (Restriction (M := M) x) ↔
29       Function.Surjective fun m : M ↦ x • m := by
30   have hquotient :
31       Subsingleton (Restriction (M := M) x) ↔
32         Ideal.span {x} • (⊤ : Submodule R M) = ⊤ := by
33     constructor
34     · intro h
35       apply Submodule.unique_quotient_iff_eq_top.mp
36       exact ⟨{
37         default := 0
38         uniq := fun y ↦ h.elim y 0 }⟩
39     · intro h
40       obtain ⟨hunique⟩ := Submodule.unique_quotient_iff_eq_top.mpr h
41       exact @Unique.instSubsingleton _ hunique
42   rw [hquotient]
43   constructor
44   · intro htop m
45     have hm : m ∈ x • (⊤ : Submodule R M) := by
46       rw [← Submodule.ideal_span_singleton_smul, htop]
47       exact Submodule.mem_top
48     rw [← Submodule.singleton_set_smul] at hm
49     obtain ⟨n, hn, hmn⟩ :=
50       (Submodule.mem_singleton_set_smul (⊤ : Submodule R M) x m).mp hm
51     exact ⟨n, hmn.symm⟩
52   · intro hx
53     apply top_unique
54     intro m hm
55     obtain ⟨n, rfl⟩ := hx m
56     exact Submodule.smul_mem_smul
57       (Ideal.subset_span (by simp))
58       (Submodule.mem_top : n ∈ (⊤ : Submodule R M))
59 
60 theorem restriction_subsingleton_of_smul_surjective {x : R}
61     (hx : Function.Surjective fun m : M ↦ x • m) :
62     Subsingleton (Restriction (M := M) x) :=
63   restriction_subsingleton_iff_smul_surjective.mpr hx
64 
65 /-- Determinant-trick certificate for a surjective scalar action. -/
66 theorem exists_annihilator_sub_one_mem_span_of_smul_surjective
67     [Module.Finite R M] {x : R}
68     (hx : Function.Surjective fun m : M ↦ x • m) :
69     ∃ r : R, r - 1 ∈ Ideal.span {x} ∧ ∀ m : M, r • m = 0 := by
70   obtain ⟨r, hr, hrann⟩ :=
71     Submodule.exists_sub_one_mem_and_smul_eq_zero_of_fg_of_le_smul
72       (Ideal.span {x}) (⊤ : Submodule R M) Module.Finite.fg_top (by
73         intro m hm
74         obtain ⟨n, rfl⟩ := hx m
75         exact Submodule.smul_mem_smul
76           (Ideal.subset_span (by simp))
77           (Submodule.mem_top : n ∈ (⊤ : Submodule R M)))
78   exact ⟨r, hr, fun m ↦ hrann m Submodule.mem_top⟩

80 /-- A support prime containing `x` is impossible when `x` acts surjectively. -/
81 theorem not_mem_support_of_smul_surjective_of_mem
82     [Module.Finite R M] {x : R}
83     (hx : Function.Surjective fun m : M ↦ x • m)
84     (p : PrimeSpectrum R) (hxp : x ∈ p.asIdeal) :
85     p ∉ Module.support R M := by
86   intro hp
87   obtain ⟨r, hrspan, hrann⟩ :=
88     exists_annihilator_sub_one_mem_span_of_smul_surjective hx
89   have hrp : r ∈ p.asIdeal :=
90     Module.annihilator_le_of_mem_support hp (Module.mem_annihilator.mpr hrann)
91   have hsubp : r - 1 ∈ p.asIdeal :=
92     (Ideal.span_le.mpr (by simpa using hxp)) hrspan
93   have hone : (1 : R) ∈ p.asIdeal := by
94     have := p.asIdeal.sub_mem hrp hsubp
95     simpa using this
96   exact p.2.ne_top ((Ideal.eq_top_iff_one p.asIdeal).mpr hone)

98 /-- The support of a finite module with surjective `x`-action avoids `V(x)`. -/
99 theorem support_disjoint_zeroLocus_of_smul_surjective
100     [Module.Finite R M] {x : R}
101     (hx : Function.Surjective fun m : M ↦ x • m) :
102     Disjoint (Module.support R M) (PrimeSpectrum.zeroLocus ({x} : Set R)) := by
103   rw [Set.disjoint_left]
104   intro p hp hpx
105   rw [PrimeSpectrum.mem_zeroLocus] at hpx
106   exact not_mem_support_of_smul_surjective_of_mem hx p (hpx (by simp)) hp
107 
108 /-- Restriction-vanishing form of support exclusion. -/
109 theorem support_disjoint_zeroLocus_of_restriction_subsingleton
110     [Module.Finite R M] {x : R}
111     (hx : Subsingleton (Restriction (M := M) x)) :
112     Disjoint (Module.support R M) (PrimeSpectrum.zeroLocus ({x} : Set R)) :=
113   support_disjoint_zeroLocus_of_smul_surjective
114     (restriction_subsingleton_iff_smul_surjective.mp hx)
115 
116 /-- Complement-inclusion form of support exclusion. -/
117 theorem support_subset_compl_zeroLocus_of_smul_surjective
118     [Module.Finite R M] {x : R}
119     (hx : Function.Surjective fun m : M ↦ x • m) :
120     Module.support R M ⊆ (PrimeSpectrum.zeroLocus ({x} : Set R))ᶜ := by
121   intro p hp hpx
122   exact Set.disjoint_left.mp
123     (support_disjoint_zeroLocus_of_smul_surjective hx) hp hpx
