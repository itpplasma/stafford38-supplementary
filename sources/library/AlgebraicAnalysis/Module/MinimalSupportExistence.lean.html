8 # Existence of a minimal support prime
9 
10 A nontrivial finite module over a commutative Noetherian ring has a prime in
11 its support which is minimal among the support primes.
12 -/

18 variable {R U : Type*} [CommRing R] [AddCommGroup U] [Module R U]
19 
20 @[nolint unusedArguments]
21 theorem exists_minimal_support_prime
22     [IsNoetherianRing R] [Module.Finite R U] [Nontrivial U] :
23     ∃ q : PrimeSpectrum R,
24       q ∈ Module.support R U ∧
25         ∀ p ∈ Module.support R U, p.asIdeal ≤ q.asIdeal →
26           q.asIdeal ≤ p.asIdeal := by
27   have hann : Module.annihilator R U ≠ ⊤ := by
28     intro h
29     have hs : Subsingleton U := Module.annihilator_eq_top_iff.mp h
30     exact not_subsingleton_iff_nontrivial.mpr inferInstance hs
31   obtain ⟨q, hq⟩ :=
32     (Module.annihilator R U).nonempty_minimalPrimes hann
33   let Q : PrimeSpectrum R := ⟨q, hq.1.1⟩
34   refine ⟨Q, Module.mem_support_iff_of_finite.mpr hq.1.2, ?_⟩
35   intro p hp hpq
36   exact hq.2
37     ⟨p.isPrime, Module.mem_support_iff_of_finite.mp hp⟩ hpq

41 end
