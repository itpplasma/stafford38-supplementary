9 /-!
10 # Finiteness from a monic annihilator
11 
12 A finite polynomial-ring module annihilated by a monic polynomial is finite
13 over the coefficient ring. This is the algebraic finiteness step for the
14 normal covariable; no characteristic-support assertion is assumed.
15 -/
16

19 open Polynomial

21 theorem finite_of_monic_annihilator {R E : Type*} [CommRing R]
22     [AddCommGroup E] [Module R E] [Module R[X] E]
23     [IsScalarTower R R[X] E] [Module.Finite R[X] E]
24     (g : R[X]) (hg : g.Monic) (hkill : ∀ z : E, g • z = 0) :
25     Module.Finite R E := by
26   have ht : Module.IsTorsionBySet R[X] E (Ideal.span {g}) := by
27     rw [Module.isTorsionBySet_span_singleton_iff]
28     exact hkill
29   let := ht.module
30   let : IsScalarTower R[X] (R[X] ⧸ Ideal.span {g}) E :=
31     ht.isScalarTower
32   let : IsScalarTower R (R[X] ⧸ Ideal.span {g}) E :=
33     ht.isScalarTower
34   let : Module.Finite (R[X] ⧸ Ideal.span {g}) E :=
35     Module.Finite.of_restrictScalars_finite R[X] (R[X] ⧸ Ideal.span {g}) E
36   let := hg.finite_quotient
37   exact Module.Finite.trans (R[X] ⧸ Ideal.span {g}) E

39 /-- A finite module killed by the polynomial variable is finite over the
40 coefficient ring. -/
41 theorem finite_of_variable_annihilates {R E : Type*} [CommRing R]
42     [AddCommGroup E] [Module R E] [Module R[X] E]
43     [IsScalarTower R R[X] E] [Module.Finite R[X] E]
44     (hkill : ∀ z : E, (X : R[X]) • z = 0) : Module.Finite R E :=
45   finite_of_monic_annihilator X monic_X hkill

47 /-- Both terms of the principal Koszul homology are finite over the
48 coefficient ring, although the ambient module need not be. -/
49 theorem finite_kernel_and_cokernel_variable {R E : Type*} [CommRing R]
50     [IsNoetherianRing R] [AddCommGroup E] [Module R E] [Module R[X] E]
51     [IsScalarTower R R[X] E] [Module.Finite R[X] E] :
52     Module.Finite R (LinearMap.ker (LinearMap.lsmul R[X] E X)) ∧
53       Module.Finite R (E ⧸ LinearMap.range (LinearMap.lsmul R[X] E X)) := by
54   let f : Module.End R[X] E := LinearMap.lsmul R[X] E X
55   constructor
56   · apply finite_of_variable_annihilates
57     intro z
58     apply Subtype.ext
59     exact LinearMap.mem_ker.mp z.property
60   · apply finite_of_variable_annihilates
61     intro z
62     induction z using Submodule.Quotient.induction_on with
63     | _ z =>
64       rw [← Submodule.Quotient.mk_smul, Submodule.Quotient.mk_eq_zero]
65       exact ⟨z, rfl⟩
66 
67 end AlgebraicAnalysis.MonicAnnihilatorFinite
