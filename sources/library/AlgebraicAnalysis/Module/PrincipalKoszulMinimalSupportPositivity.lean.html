7 /-!
8 # Principal Koszul positivity from minimal support
9 
10 The support primes used by the strict length argument are produced from the
11 finite `C`-module itself.  In particular, no finiteness of `E` over the base
12 ring `R` is needed.
13 -/

28 /-- Minimal support primes of `E` produce the ordered support pair needed by
29 the principal Koszul positivity argument. -/
30 theorem length_cokernel_gt_kernel_of_minimal_support
31     [IsNoetherianRing C] [Module.Finite C E]
32     (x : C)
33     (hmin : ∀ p ∈ (Module.annihilator C E).minimalPrimes,
34       x ∉ p)
35     (hnonzero : Nontrivial (QuotSMulTop x E))
36     (hfinite : IsFiniteLength R
37       (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R))) :
38     Module.length R (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R)) >
39       Module.length R (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R)) := by
40   have : Nontrivial (QuotSMulTop x E) := hnonzero
41   obtain ⟨q, hqquot⟩ := Module.nonempty_support_of_nontrivial
42     (R := C) (M := QuotSMulTop x E)
43   have hq := hqquot
44   rw [Module.support_quotSMulTop] at hq
45   have hqE : q ∈ Module.support C E := hq.1
46   have han : Module.annihilator C E ≤ q.asIdeal :=
47     Module.mem_support_iff_of_finite.mp hqE
48   obtain ⟨p, hpmin, hpq⟩ :=
49     Ideal.exists_minimalPrimes_le (I := Module.annihilator C E)
50       (J := q.asIdeal) han
51   let p' : PrimeSpectrum C := ⟨p, hpmin.1.1⟩
52   have hp' : p' ∈ Module.support C E :=
53     Module.mem_support_iff_of_finite.mpr hpmin.1.2
54   have hpq' : p' ≤ q := hpq
55   have hxp : x ∉ p'.asIdeal := by
56     exact hmin p hpmin
57   exact length_cokernel_gt_kernel_of_support_over_base
58     x p' q hp' hpq' hxp hqquot hfinite
