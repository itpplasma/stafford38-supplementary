6 /-!
7 # Finite intersections in a right Ore domain
8 
9 Right ideals are represented as left modules over the opposite ring.  The
10 right Ore condition is kept as an explicit common-right-multiple hypothesis;
11 this module does not depend on a particular localization construction.
12 -/
13 
14 namespace AlgebraicAnalysis.OreRightIntersection
15 
16 variable {R : Type*} [Ring R] [IsDomain R]
17 
18 /-- Every pair of nonzero elements has a nonzero common right multiple. -/
19 def RightOreCondition (R : Type*) [SemigroupWithZero R] : Prop :=
20   ∀ ⦃a b : R⦄, a ≠ 0 → b ≠ 0 →
21     ∃ x y : R, a * x = b * y ∧ a * x ≠ 0
22 
23 /-- A finite family of nonzero right ideals has a nonzero common element.
24 
25 The right ideals are left `Rᵐᵒᵖ`-submodules, so multiplying a member on the
26 right is expressed by scalar multiplication by `MulOpposite.op`. -/
27 theorem exists_mem_finset_rightIdeals
28     {ι : Type*} [DecidableEq ι]
29     (s : Finset ι) (I : ι → Submodule Rᵐᵒᵖ R)
30     (hI : ∀ i ∈ s, ∃ x ∈ I i, x ≠ 0)
31     (hOre : RightOreCondition R) :
32     ∃ x : R, x ≠ 0 ∧ ∀ i ∈ s, x ∈ I i := by
33   classical
34   induction s using Finset.induction_on with
35   | empty =>
36       exact ⟨1, one_ne_zero, by simp⟩
37   | @insert i s hi ih =>
38       obtain ⟨m, hm0, hm⟩ := ih (fun j hj => hI j (Finset.mem_insert_of_mem hj))
39       obtain ⟨n, hnI, hn0⟩ := hI i (Finset.mem_insert_self i s)
40       obtain ⟨x, y, hxy, hmx0⟩ := hOre hm0 hn0
41       refine ⟨m * x, hmx0, ?_⟩
42       intro j hj
43       rcases Finset.mem_insert.mp hj with hji | hj
44       · subst j
45         rw [hxy]
46         exact (I i).smul_mem (MulOpposite.op y) hnI
47       · exact (I j).smul_mem (MulOpposite.op x) (hm j hj)
48 
49 /- The theorem is intentionally stated with an explicit nonzero witness;
50    this is the usual meaning of ``the intersection is nonzero''. -/
51 #print axioms exists_mem_finset_rightIdeals
52 
53 end AlgebraicAnalysis.OreRightIntersection
