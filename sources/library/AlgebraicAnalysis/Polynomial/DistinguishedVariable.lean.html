10 /-!
11 # A distinguished variable in a homogeneous prime relation
12 
13 If a homogeneous polynomial uses one distinguished variable together with a
14 set of auxiliary variables, then subtracting its pure distinguished monomial
15 puts it in the ideal generated by the auxiliary variables.  Consequently, a
16 prime containing the polynomial and every auxiliary variable also contains
17 the distinguished variable, provided the pure coefficient is a unit.
18 -/

26 /-- Removing the pure distinguished monomial from a homogeneous polynomial
27 leaves an element of the ideal generated by its other allowed variables. -/
28 theorem sub_pureMonomial_mem_span_X
29     {P : MvPolynomial σ R} {N : ℕ} {t : σ} {s : Set σ} {c : R}
30     (hP : P.IsHomogeneous N)
31     (hvars : (P.vars : Set σ) ⊆ insert t s)
32     (hcoeff : P.coeff (Finsupp.single t N) = c) :
33     P - MvPolynomial.monomial (Finsupp.single t N) c ∈
34       Ideal.span (MvPolynomial.X '' s : Set (MvPolynomial σ R)) := by
35   classical
36   rw [MvPolynomial.mem_ideal_span_X_image]
37   intro m hm
38   by_contra haux
39   push_neg at haux
40   have hdiff :
41       (P - MvPolynomial.monomial (Finsupp.single t N) c).coeff m ≠ 0 :=
42     MvPolynomial.mem_support_iff.mp hm
43   by_cases hmt : m = Finsupp.single t N
44   · subst m
45     simp [hcoeff] at hdiff
46   have hmono :
47       (MvPolynomial.monomial (Finsupp.single t N) c).coeff m = 0 := by
48     simp [MvPolynomial.coeff_monomial, Ne.symm hmt]
49   have hmP : m ∈ P.support := by
50     apply MvPolynomial.mem_support_iff.mpr
51     simpa [hmono] using hdiff
52   have hsupp : m.support ⊆ {t} := by
53     intro i hi
54     have hiVars : i ∈ P.vars :=
55       (MvPolynomial.mem_vars i).mpr ⟨m, hmP, hi⟩
56     have hiAllowed : i = t ∨ i ∈ s := by
57       simpa only [Set.mem_insert_iff] using hvars hiVars
58     rcases hiAllowed with rfl | his
59     · simp
60     · exact False.elim ((Finsupp.mem_support_iff.mp hi) (haux i his))
61   have hmform : m = Finsupp.single t (m t) :=
62     Finsupp.support_subset_singleton.mp hsupp
63   have hdegree : m.degree = N := by
64     rw [Finsupp.degree_eq_weight_one]
65     exact hP (MvPolynomial.mem_support_iff.mp hmP)
66   rw [hmform, Finsupp.degree_single] at hdegree
67   exact hmt (by rw [hmform, hdegree])

69 /-- A prime containing a homogeneous relation, all auxiliary
70 variables, and a unit pure coefficient must contain the distinguished
71 variable. -/
72 theorem X_mem_of_homogeneous_mem_prime
73     {P : MvPolynomial σ R} {N : ℕ} {t : σ} {s : Set σ} {c : R}
74     (hP : P.IsHomogeneous N)
75     (hvars : (P.vars : Set σ) ⊆ insert t s)
76     (hcoeff : P.coeff (Finsupp.single t N) = c)
77     (hc : IsUnit c)
78     {p : Ideal (MvPolynomial σ R)} (hp : p.IsPrime)
79     (hPmem : P ∈ p)
80     (hsmem : ∀ i ∈ s, MvPolynomial.X i ∈ p) :
81     MvPolynomial.X t ∈ p := by
82   have hdiff :
83       P - MvPolynomial.monomial (Finsupp.single t N) c ∈ p :=
84     (Ideal.span_le.mpr (by
85       rintro _ ⟨i, hi, rfl⟩
86       exact hsmem i hi))
87       (sub_pureMonomial_mem_span_X hP hvars hcoeff)
88   have hmono : MvPolynomial.monomial (Finsupp.single t N) c ∈ p := by
89     have := p.sub_mem hPmem hdiff
90     simpa only [sub_sub_cancel] using this
91   rw [← MvPolynomial.C_mul_X_pow_eq_monomial] at hmono
92   have hC : IsUnit (MvPolynomial.C c : MvPolynomial σ R) :=
93     hc.map (MvPolynomial.C : R →+* MvPolynomial σ R)
94   have hpow : MvPolynomial.X t ^ N ∈ p :=
95     (p.unit_mul_mem_iff_mem hC).mp hmono
96   exact hp.mem_of_pow_mem N hpow
