9 # Finitely generated fraction fields
10 
11 A fraction field of a finitely generated domain is finitely generated as a
12 field extension.  The conclusion uses `IntermediateField.FG`, rather than
13 finite type as an algebra: a transcendental fraction field is generally not
14 finitely generated as an algebra over its ground field.
15 -/

23 /-- The fraction field of a finitely generated domain is a finitely generated
24 field extension of the ground field. -/
25 @[nolint unusedArguments]
26 theorem top_fg_of_finiteType_fractionRing
27     (k : Type u) (A : Type v) (K : Type w)
28     [Field k] [CommRing A] [IsDomain A] [Algebra k A]
29     [Field K] [Algebra A K] [IsFractionRing A K]
30     [Algebra k K] [IsScalarTower k A K]
31     [Algebra.FiniteType k A] :
32     (⊤ : IntermediateField k K).FG := by
33   classical
34   obtain ⟨s, hs⟩ := (Algebra.FiniteType.out (R := k) (A := A))
35   let f : A →ₐ[k] K := IsScalarTower.toAlgHom k A K
36   let fi : A ↪ K :=
37     ⟨f, IsFractionRing.injective A K⟩
38   let T : IntermediateField k K :=
39     IntermediateField.adjoin k (f '' (s : Set A))
40   have hmap : ∀ a : A, f a ∈ T := by
41     intro a
42     have ha : a ∈ Algebra.adjoin k (s : Set A) := by
43       rw [hs]
44       trivial
45     refine Algebra.adjoin_induction
46       (p := fun a _ ↦ f a ∈ T) ?_ ?_ ?_ ?_ ha
47     · intro x hx
48       exact IntermediateField.subset_adjoin k _ ⟨x, hx, rfl⟩
49     · intro x
50       simpa [f] using T.algebraMap_mem x
51     · intro x y _ _ hx hy
52       simpa using T.add_mem hx hy
53     · intro x y _ _ hx hy
54       simpa using T.mul_mem hx hy
55   have htop : T = ⊤ := by
56     rw [eq_top_iff]
57     intro z _
58     obtain ⟨a, b, hb, hab⟩ := IsFractionRing.div_surjective (A := A) z
59     rw [← hab]
60     exact T.div_mem (hmap a) (hmap b)
61   refine ⟨s.map fi, ?_⟩
62   simpa [T, fi, f, Finset.coe_map] using htop
63

65 
66 end
