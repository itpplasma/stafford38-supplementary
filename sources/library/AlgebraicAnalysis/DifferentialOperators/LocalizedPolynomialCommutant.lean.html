8 # The polynomial commutant after localization
9 
10 An endomorphism of a localization which commutes with the coordinate
11 multiplications is multiplication by its value at `1`.  The argument only
12 uses the localization presentation; it does not use a finite-order
13 hypothesis.
14 -/

31 theorem eq_multiplication_of_commute_coordinate
32     (S : Submonoid (MvPolynomial (Fin n) k))
33     [IsLocalization S B]
34     (P : Module.End k B)
35     (hcoord : ∀ i : Fin n, ∀ b : B,
36       P (algebraMap (MvPolynomial (Fin n) k) B (MvPolynomial.X i) * b) =
37         algebraMap (MvPolynomial (Fin n) k) B (MvPolynomial.X i) * P b) :
38     P = multiplication (k := k) (P 1) := by
39   have hpoly : ∀ a : MvPolynomial (Fin n) k, ∀ b : B,
40       P (algebraMap (MvPolynomial (Fin n) k) B a * b) =
41         algebraMap (MvPolynomial (Fin n) k) B a * P b := by
42     intro a
43     induction a using MvPolynomial.induction_on with
44     | C c =>
45         intro b
46         have hc : algebraMap (MvPolynomial (Fin n) k) B (MvPolynomial.C c) =
47             algebraMap k B c := by
48           calc
49             algebraMap (MvPolynomial (Fin n) k) B (MvPolynomial.C c) =
50                 algebraMap (MvPolynomial (Fin n) k) B
51                   (algebraMap k (MvPolynomial (Fin n) k) c) := by
52                     rw [MvPolynomial.algebraMap_eq]
53             _ = algebraMap k B c := by
54               exact (IsScalarTower.algebraMap_apply k
55                 (MvPolynomial (Fin n) k) B c).symm
56         rw [hc]
57         simpa [Algebra.smul_def] using P.map_smul c b
58     | add a a' ha ha' =>
59         intro b
60         simp only [map_add, map_mul, ha b, ha' b, add_mul]
61     | mul_X a i ha =>
62         intro b
63         calc
64           P (algebraMap _ B (a * MvPolynomial.X i) * b) =
65               P (algebraMap (MvPolynomial (Fin n) k) B a *
66                 (algebraMap (MvPolynomial (Fin n) k) B (MvPolynomial.X i) * b)) := by
67             rw [map_mul (algebraMap (MvPolynomial (Fin n) k) B)]
68             simp [mul_assoc]
69           _ = algebraMap _ B a *
70               P (algebraMap _ B (MvPolynomial.X i) * b) := ha _
71           _ = algebraMap _ B a *
72               (algebraMap _ B (MvPolynomial.X i) * P b) := by
73             rw [hcoord]
74           _ = algebraMap _ B (a * MvPolynomial.X i) * P b := by
75             rw [map_mul (algebraMap (MvPolynomial (Fin n) k) B)]
76             simp [mul_assoc]
77   apply LinearMap.ext
78   intro b
79   obtain ⟨⟨a, s⟩, hs⟩ := IsLocalization.surj S b
80   have hsa : IsUnit (algebraMap (MvPolynomial (Fin n) k) B (s : _)) :=
81     IsLocalization.map_units B s
82   refine hsa.mul_left_cancel ?_
83   simp only [multiplication_apply]
84   rw [← hpoly (s : MvPolynomial (Fin n) k) b]
85   rw [← mul_comm b, hs]
86   have ha1 : P (algebraMap (MvPolynomial (Fin n) k) B a) =
87       algebraMap (MvPolynomial (Fin n) k) B a * P 1 := by
88     simpa using hpoly a 1
89   rw [ha1]
90   rw [← hs]
91   simp [mul_assoc, mul_comm, mul_left_comm]
