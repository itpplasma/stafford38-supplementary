12 # The linear symplectic layer of the A₂ reduction
13 
14 This file isolates the part of the generic-monic reduction which is purely
15 linear algebra.  A Weyl presentation is encoded by a finite family `z` whose
16 pairwise commutators are the entries of a fixed skew matrix.  A matrix `M`
17 preserving that form gives a new family by linear combination, and the new
18 family has exactly the same Weyl commutators.
19 
20 The result is deliberately stated for an arbitrary finite index type and an
21 arbitrary symplectic form.  The A₂ instance is obtained with
22 `ι = Fin 2 ⊕ Fin 2` and `Matrix.J (Fin 2) k`, so this is not a list of
23 hand-picked coordinate changes.
24 
25 This is the algebraic prerequisite for applying Stafford's
26 ring-equivalence transport to a Weyl change of generators.  The final section
27 also descends a form-preserving change to a homomorphism of the presented
28 quotient and proves the A₂ symplectic change is invertible using the inverse
29 matrix and generator-extensionality. The PBW identification and generic monic
30 normalization belong to the downstream Weyl modules, outside this module's
31 linear-algebra scope. Reusable ringCommutator identities are imported from
32 AlgebraicAnalysis.
33 -/

48 lemma commutator_smul_smul (a b : k) (u v : A) :
49     ringCommutator (algebraMap k A a * u) (algebraMap k A b * v) =
50       algebraMap k A (a * b) * ringCommutator u v := by
51   have hub : u * algebraMap k A b = algebraMap k A b * u :=
52     (Algebra.commutes b u).symm
53   have hva : v * algebraMap k A a = algebraMap k A a * v :=
54     (Algebra.commutes a v).symm
55   simp only [ringCommutator]
56   calc
57     algebraMap k A a * u * (algebraMap k A b * v) -
58         algebraMap k A b * v * (algebraMap k A a * u) =
59       (algebraMap k A a * algebraMap k A b) * (u * v) -
60         (algebraMap k A b * algebraMap k A a) * (v * u) := by
61           rw [show algebraMap k A a * u * (algebraMap k A b * v) =
62               (algebraMap k A a * algebraMap k A b) * (u * v) by
63                 calc
64                   algebraMap k A a * u * (algebraMap k A b * v) =
65                       algebraMap k A a * (u * algebraMap k A b) * v := by
66                         noncomm_ring
67                   _ = algebraMap k A a * (algebraMap k A b * u) * v := by
68                         rw [hub]
69                   _ = (algebraMap k A a * algebraMap k A b) * (u * v) := by
70                         noncomm_ring,
71             show algebraMap k A b * v * (algebraMap k A a * u) =
72               (algebraMap k A b * algebraMap k A a) * (v * u) by
73                 calc
74                   algebraMap k A b * v * (algebraMap k A a * u) =
75                       algebraMap k A b * (v * algebraMap k A a) * u := by
76                         noncomm_ring
77                   _ = algebraMap k A b * (algebraMap k A a * v) * u := by
78                         rw [hva]
79                   _ = (algebraMap k A b * algebraMap k A a) * (v * u) := by
80                         noncomm_ring]
81     _ = algebraMap k A (a * b) * (u * v) -
82         algebraMap k A (b * a) * (v * u) := by
83           rw [_root_.map_mul, _root_.map_mul]
84     _ = algebraMap k A (a * b) * (u * v - v * u) := by
85           rw [mul_comm b a, mul_sub]

87 lemma commutator_linearCombination
88     (M : Matrix ι ι k) (z : ι → A) (omega : Matrix ι ι k) (i l : ι)
89     (hcomm : ∀ j n, ringCommutator (z j) (z n) =
90       algebraMap k A (omega j n)) :
91     ringCommutator (linearCombination M z i) (linearCombination M z l) =
92       algebraMap k A (∑ j, ∑ n, M i j * omega j n * M l n) := by
93   rw [linearCombination, linearCombination, commutator_sum_left]
94   simp_rw [commutator_sum_right, commutator_smul_smul, hcomm]
95   simp only [map_sum, map_mul, Finset.sum_mul, Finset.mul_sum]
96   congr 2
97   funext j
98   congr 1
99   funext n
100   calc
101     algebraMap k A (M i j) * algebraMap k A (M l n) *
102           algebraMap k A (omega j n) =
103         algebraMap k A (M i j) *
104           (algebraMap k A (M l n) * algebraMap k A (omega j n)) := by
105             rw [mul_assoc]
106     _ = algebraMap k A (M i j) *
107           (algebraMap k A (omega j n) * algebraMap k A (M l n)) := by
108             congr 1
109             rw [← _root_.map_mul, ← _root_.map_mul]
110             congr 1
111             ring
112     _ = algebraMap k A (M i j) * algebraMap k A (omega j n) *
113           algebraMap k A (M l n) := by
114             rw [← mul_assoc]

116 /-- A matrix preserving `omega` preserves all Weyl commutators. -/
117 theorem symplectic_linear_change_preserves_commutator
118     (M : Matrix ι ι k) (z : ι → A) (omega : Matrix ι ι k)
119     (hcomm : ∀ j n, ringCommutator (z j) (z n) =
120       algebraMap k A (omega j n))
121     (hM : M * omega * Matrix.transpose M = omega) :
122     ∀ i l, ringCommutator (linearCombination M z i) (linearCombination M z l) =
123       algebraMap k A (omega i l) := by
124   intro i l
125   rw [commutator_linearCombination M z omega i l hcomm]
126   have hentry : (∑ j, ∑ n, M i j * omega j n * M l n) = omega i l := by
127     calc
128       (∑ j, ∑ n, M i j * omega j n * M l n) =
129           (M * omega * Matrix.transpose M) i l := by
130             simp only [Matrix.mul_apply, Matrix.transpose_apply, Finset.sum_mul]
131             rw [Finset.sum_comm]
132       _ = omega i l := by rw [hM]
133   rw [hentry]

162 /-! ### Descent to the presented Weyl algebra
163 
164 `RingQuot` is Mathlib's universal quotient for a possibly noncommutative ring.
165 The following definitions use it to make the universal-property step
166 explicit.  This is still presentation-level algebra: identifying the
167 quotient with a PBW Weyl algebra remains separate, while the inverse-matrix
168 argument below proves the form-preserving A₂ map is an automorphism. -/
169 
170 theorem freeWeylGenerator_commutator
171     (omega : Matrix ι ι k) (i j : ι) :
172     ringCommutator (freeWeylGenerator omega i) (freeWeylGenerator omega j) =
173       algebraMap k (FreeWeyl k ι omega) (omega i j) := by
174   have h := RingQuot.mkAlgHom_rel k
175     (show freeWeylRelation omega
176       (FreeAlgebra.ι k i * FreeAlgebra.ι k j -
177         FreeAlgebra.ι k j * FreeAlgebra.ι k i)
178       (algebraMap k (FreeAlgebra k ι) (omega i j)) from ⟨i, j, rfl, rfl⟩)
179   simp only [freeWeylGenerator, ringCommutator, RingQuot.mkAlgHom_coe]
180   calc
181     RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i) *
182           RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k j) -
183         RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k j) *
184           RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i) =
185       RingQuot.mkAlgHom k (freeWeylRelation omega)
186         (FreeAlgebra.ι k i * FreeAlgebra.ι k j -
187           FreeAlgebra.ι k j * FreeAlgebra.ι k i) := by
188             rw [← map_mul, ← map_mul, ← map_sub]
189     _ = RingQuot.mkAlgHom k (freeWeylRelation omega)
190         ((algebraMap k (FreeAlgebra k ι)) (omega i j)) := h
191     _ = algebraMap k (FreeWeyl k ι omega) (omega i j) := by
192       have hmap : algebraMap k (FreeWeyl k ι omega) =
193           (RingQuot.mkAlgHom k (freeWeylRelation omega)).toRingHom.comp
194             (algebraMap k (FreeAlgebra k ι)) := by
195         apply RingHom.ext
196         intro c
197         change algebraMap k (FreeWeyl k ι omega) c =
198           RingQuot.mkAlgHom k (freeWeylRelation omega)
199             (algebraMap k (FreeAlgebra k ι) c)
200         exact (RingQuot.mkAlgHom k (freeWeylRelation omega)).commutes c |>.symm
201       rw [hmap]
202       rfl

243 theorem freeWeylSymplecticAlgHom_map_linearCombination
244     (M N : Matrix ι ι k) (omega : Matrix ι ι k)
245     (hpres : ∀ i j,
246       ringCommutator (Stafford.linearCombination M (freeWeylGenerator omega) i)
247           (Stafford.linearCombination M (freeWeylGenerator omega) j) =
248         algebraMap k (FreeWeyl k ι omega) (omega i j)) (i : ι) :
249     freeWeylSymplecticAlgHom M omega hpres
250         (Stafford.linearCombination N (freeWeylGenerator omega) i) =
251       Stafford.linearCombination (N * M) (freeWeylGenerator omega) i := by
252   change freeWeylSymplecticAlgHom M omega hpres
253       (∑ j, algebraMap k (FreeWeyl k ι omega) (N i j) *
254         freeWeylGenerator omega j) =
255     ∑ j, algebraMap k (FreeWeyl k ι omega) ((N * M) i j) *
256       freeWeylGenerator omega j
257   rw [map_sum]
258   simp_rw [map_mul, AlgHom.commutes,
259     freeWeylSymplecticAlgHom_generator M omega hpres]
260   simp_rw [Stafford.linearCombination]

262   rw [Finset.sum_comm]
263   apply Finset.sum_congr rfl
264   intro j hj

376   let g := a2_freeWeyl_symplectic_hom
