7 /-!
8 # Transport of Stafford certificates
9 
10 The right-ideal certificate is `1 = e * R + F * e * S`.
11 This module supplies the generic conditional reduction from monicization and
12 an exponent hypothesis, together with transport through algebra automorphisms
13 and nonzero scalars. These conditional interfaces are retained for current
14 imports; the unconditional theorem is proved in `Stafford38.FoundationClosure`.
15 -/
16 
17 namespace Stafford
18 namespace Reduction
19 
20 variable {k A : Type*} [Field k] [Ring A] [Algebra k A]
21 
22 /-- Historical reduction notation for the shared ring commutator. -/
23 abbrev ad (Y u : A) : A := AlgebraicAnalysis.ringCommutator Y u
24 
25 @[simp] theorem ad_eq_shared (Y u : A) :
26     ad Y u = AlgebraicAnalysis.ringCommutator Y u := rfl
27 
28 /-- The monic right-coefficient presentation `X ^ N + ∑_{i<N} X ^ i * β i`. -/
29 def monicElt (X : A) (N : ℕ) (β : Fin N → A) : A :=
30   X ^ N + ∑ i : Fin N, X ^ (i : ℕ) * β i
31 
32 /-- `e` is *Weyl-monic* if some Weyl pair presents it monically with all
33 coefficients commuting with the partner. -/
34 def WeylMonic (e : A) : Prop :=
35   ∃ (X Y : A) (N : ℕ) (β : Fin N → A),
36     ad Y X = 1 ∧ (∀ i, Commute Y (β i)) ∧ e = monicElt X N β
37 
38 /-! ## Transport lemmas -/
39 
40 /-- Stafford 3.8 transports along any ring automorphism. -/
41 theorem stafford38_of_ringEquiv (φ : A ≃+* A) {e : A} (h : Stafford38 (φ e)) :
42     Stafford38 e := by
43   obtain ⟨F, R, S, hFRS⟩ := h
44   refine ⟨φ.symm F, φ.symm R, φ.symm S, ?_⟩
45   have := congrArg φ.symm hFRS
46   simpa using this
47 
48 /-- Stafford 3.8 transports along scalar multiples (no nonvanishing needed: the
49 scalar is absorbed into the cofactors `R` and `S`). -/
50 theorem stafford38_smul (c : k) {e : A} (h : Stafford38 (c • e)) :
51     Stafford38 e := by
52   obtain ⟨F, R, S, hFRS⟩ := h
53   refine ⟨F, c • R, c • S, ?_⟩
54   have key : ∀ a b : A, (c • a) * b = a * (c • b) := by
55     intro a b
56     rw [Algebra.smul_def, Algebra.smul_def, ← mul_assoc, ← Algebra.commutes c a]
57   rw [hFRS, key e R]
58   congr 1
59   calc F * (c • e) * S = F * ((c • e) * S) := by rw [mul_assoc]
60     _ = F * (e * (c • S)) := by rw [key e S]
61     _ = F * e * (c • S) := by rw [mul_assoc]
62 
63 /-- A cofactor of the special form `Y ^ s` already witnesses Stafford 3.8. -/
64 theorem stafford38_of_power {e Y : A} {s : ℕ} {R S : A}
65     (h : (1 : A) = e * R + Y ^ s * e * S) : Stafford38 e :=
66   ⟨Y ^ s, R, S, h⟩
67 
68 /-! ## The two hypotheses -/
69 
70 /-- **Hypothesis M (monicization).**  Every nonzero element is, after an algebra
71 automorphism and a nonzero scalar, Weyl-monic. -/
72 def Monicization (k A : Type*) [Field k] [Ring A] [Algebra k A] : Prop :=
73   ∀ e : A, e ≠ 0 → ∃ (φ : A ≃ₐ[k] A) (c : k), c ≠ 0 ∧ WeylMonic (c • φ e)
74 
75 /-- **Hypothesis E (exponent).**  For every Weyl-monic presentation some power of
76 the Weyl partner is a Stafford cofactor. -/
77 def ExponentHypothesis (A : Type*) [Ring A] : Prop :=
78   ∀ (X Y : A) (N : ℕ) (β : Fin N → A), ad Y X = 1 → (∀ i, Commute Y (β i)) →
79     ∃ (s : ℕ) (R S : A),
80       (1 : A) = monicElt X N β * R + Y ^ s * monicElt X N β * S
81 
82 /-! ## The reduction -/
83 
84 /-- Hypothesis E gives Stafford 3.8 for every Weyl-monic element. -/
85 theorem stafford38_of_weylMonic (hE : ExponentHypothesis A) {e : A}
86     (he : WeylMonic e) : Stafford38 e := by
87   obtain ⟨X, Y, N, β, hweyl, hcomm, heq⟩ := he
88   obtain ⟨s, R, S, hs⟩ := hE X Y N β hweyl hcomm
89   rw [heq]
90   exact stafford38_of_power hs
91 
92 /--
93 **Stafford 3.8 from the two named hypotheses.**
94 
95 If every nonzero element monicizes (Hypothesis M) and every Weyl-monic element
96 admits a cofactor that is a power of the Weyl partner (Hypothesis E), then
97 
98     ∀ e ≠ 0, ∃ F R S, 1 = e * R + F * e * S.
99 -/
100 theorem stafford38_of_hypotheses
101     (hM : Monicization k A) (hE : ExponentHypothesis A) :
102     ∀ e : A, e ≠ 0 → Stafford38 e := by
103   intro e he
104   obtain ⟨φ, c, hc, hmonic⟩ := hM e he
105   have h1 : Stafford38 (c • φ e) := stafford38_of_weylMonic hE hmonic
106   have h2 : Stafford38 (φ e) := stafford38_smul (k := k) c h1
107   exact stafford38_of_ringEquiv (φ : A ≃+* A) h2
