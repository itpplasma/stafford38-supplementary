9 /-!
10 # Transport of Stafford certificates
11 
12 The right-ideal certificate is `1 = e * R + F * e * S`.
13 This module supplies the generic conditional reduction from monicization and
14 an exponent hypothesis, together with transport through algebra automorphisms
15 and nonzero scalars. These conditional interfaces are retained for current
16 imports; the unconditional theorem is proved in `Stafford38.FoundationClosure`.
17 -/
18 
19 namespace Stafford
20 namespace Reduction
21 
22 variable {k A : Type*} [Field k] [Ring A] [Algebra k A]
23 
24 /-- The monic right-coefficient presentation `X ^ N + ∑_{i<N} X ^ i * β i`. -/
25 def monicElt (X : A) (N : ℕ) (β : Fin N → A) : A :=
26   X ^ N + ∑ i : Fin N, X ^ (i : ℕ) * β i
27 
28 /-- `e` is *Weyl-monic* if some Weyl pair presents it monically with all
29 coefficients commuting with the partner. -/
30 def WeylMonic (e : A) : Prop :=
31   ∃ (X Y : A) (N : ℕ) (β : Fin N → A),
32     ringCommutator Y X = 1 ∧ (∀ i, Commute Y (β i)) ∧ e = monicElt X N β
33 
34 /-! ## Transport lemmas -/
35 
36 /-- Stafford 3.8 transports along any ring automorphism. -/
37 theorem stafford38_of_ringEquiv (φ : A ≃+* A) {e : A} (h : Stafford38 (φ e)) :
38     Stafford38 e := by
39   obtain ⟨F, R, S, hFRS⟩ := h
40   refine ⟨φ.symm F, φ.symm R, φ.symm S, ?_⟩
41   have := congrArg φ.symm hFRS
42   simpa using this
43 
44 /-- Stafford 3.8 transports along scalar multiples (no nonvanishing needed: the
45 scalar is absorbed into the cofactors `R` and `S`). -/
46 theorem stafford38_smul (c : k) {e : A} (h : Stafford38 (c • e)) :
47     Stafford38 e := by
48   obtain ⟨F, R, S, hFRS⟩ := h
49   refine ⟨F, c • R, c • S, ?_⟩
50   have key : ∀ a b : A, (c • a) * b = a * (c • b) := by
51     intro a b
52     rw [Algebra.smul_def, Algebra.smul_def, ← mul_assoc, ← Algebra.commutes c a]
53   rw [hFRS, key e R]
54   congr 1
55   calc F * (c • e) * S = F * ((c • e) * S) := by rw [mul_assoc]
56     _ = F * (e * (c • S)) := by rw [key e S]
57     _ = F * e * (c • S) := by rw [mul_assoc]
58 
59 /-- A cofactor of the special form `Y ^ s` already witnesses Stafford 3.8. -/
60 theorem stafford38_of_power {e Y : A} {s : ℕ} {R S : A}
61     (h : (1 : A) = e * R + Y ^ s * e * S) : Stafford38 e :=
62   ⟨Y ^ s, R, S, h⟩
63 
64 /-! ## The two hypotheses -/
65 
66 /-- **Hypothesis M (monicization).**  Every nonzero element is, after an algebra
67 automorphism and a nonzero scalar, Weyl-monic. -/
68 def Monicization (k A : Type*) [Field k] [Ring A] [Algebra k A] : Prop :=
69   ∀ e : A, e ≠ 0 → ∃ (φ : A ≃ₐ[k] A) (c : k), c ≠ 0 ∧ WeylMonic (c • φ e)
70 
71 /-- **Hypothesis E (exponent).**  For every Weyl-monic presentation some power of
72 the Weyl partner is a Stafford cofactor. -/
73 def ExponentHypothesis (A : Type*) [Ring A] : Prop :=
74   ∀ (X Y : A) (N : ℕ) (β : Fin N → A), ringCommutator Y X = 1 → (∀ i, Commute Y (β i)) →
75     ∃ (s : ℕ) (R S : A),
76       (1 : A) = monicElt X N β * R + Y ^ s * monicElt X N β * S
77 
78 /-! ## The reduction -/
79 
80 /-- Hypothesis E gives Stafford 3.8 for every Weyl-monic element. -/
81 theorem stafford38_of_weylMonic (hE : ExponentHypothesis A) {e : A}
82     (he : WeylMonic e) : Stafford38 e := by
83   obtain ⟨X, Y, N, β, hweyl, hcomm, heq⟩ := he
84   obtain ⟨s, R, S, hs⟩ := hE X Y N β hweyl hcomm
85   rw [heq]
86   exact stafford38_of_power hs
87 
88 /--
89 **Stafford 3.8 from the two named hypotheses.**
90 
91 If every nonzero element monicizes (Hypothesis M) and every Weyl-monic element
92 admits a cofactor that is a power of the Weyl partner (Hypothesis E), then
93 
94     ∀ e ≠ 0, ∃ F R S, 1 = e * R + F * e * S.
95 -/
96 theorem stafford38_of_hypotheses
97     (hM : Monicization k A) (hE : ExponentHypothesis A) :
98     ∀ e : A, e ≠ 0 → Stafford38 e := by
99   intro e he
100   obtain ⟨φ, c, hc, hmonic⟩ := hM e he
101   have h1 : Stafford38 (c • φ e) := stafford38_of_weylMonic hE hmonic
102   have h2 : Stafford38 (φ e) := stafford38_smul (k := k) c h1
103   exact stafford38_of_ringEquiv (φ : A ≃+* A) h2
104 
105 end Reduction
