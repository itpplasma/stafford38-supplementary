9 /-!
10 # A coordinate-momentum pair over a coefficient ring
11 
12 Two successive Ore extensions adjoin a central coordinate and then its
13 derivation momentum.  This file packages the resulting three canonical
14 generator families and proves their exact relations.
15 -/
16 
17 namespace Stafford38.OrePairStage
18 
19 open AlgebraicAnalysis
20 open AlgebraicAnalysis.OreAssociativity
21 open AlgebraicAnalysis.OreRightPBW
22 open Stafford38.OreCoordinateStage
23 
24 noncomputable section
25 
26 variable {B : Type*} [Ring B]
27 
28 /-- First adjoin a central coordinate, then a momentum differentiating it. -/
29 abbrev PairStage :=
30   NormalOre (coordinateDerivation :
31     OreDivisionDerivation (CoordinateStage (B := B)))
32 
33 abbrev innerD : OreDivisionDerivation B := zeroDerivation
34 
35 abbrev outerD : OreDivisionDerivation (CoordinateStage (B := B)) :=
36   coordinateDerivation
37 
38 /-- Embed the old coefficient ring through both Ore stages. -/
39 def pairCoefficient : B →+* PairStage (B := B) :=
40   (normalCoefficient outerD).comp (normalCoefficient innerD)
41 
42 /-- The newly adjoined coordinate. -/
43 def pairCoordinate : PairStage (B := B) :=
44   normalCoefficient outerD (normalVariable innerD)
45 
46 /-- The newly adjoined momentum. -/
47 def pairMomentum : PairStage (B := B) :=
48   normalVariable outerD
49 
50 /-- The checked right PBW basis for the momentum stage over the coordinate
51 stage.  This is a consumer of the reusable one-stage right-PBW interface; the
52 pair-specific generators and Weyl relation remain owned by Stafford38. -/
53 def pairStage_rightOrePBWBasis [Nontrivial B] :
54     Module.Basis ℕ (CoordinateStage (B := B))ᵐᵒᵖ (PairStage (B := B)) :=
55   rightOrePBWBasis coordinateDerivation
56 
57 /-- The new coordinate commutes with every old coefficient. -/
58 theorem pairCoordinate_mul_coefficient (b : B) :
59     pairCoordinate * pairCoefficient b = pairCoefficient b * pairCoordinate := by
60   change
61     normalCoefficient outerD (normalVariable innerD) *
62         normalCoefficient outerD (normalCoefficient innerD b) =
63       normalCoefficient outerD (normalCoefficient innerD b) *
64         normalCoefficient outerD (normalVariable innerD)
65   rw [← (normalCoefficient outerD).map_mul,
66     ← (normalCoefficient outerD).map_mul]
67   congr 1
68   have h := normalVariable_mul_coefficient innerD b
69   rw [zeroDerivation_apply, map_zero, add_zero] at h
70   exact h
71 
72 /-- The new momentum commutes with every old coefficient. -/
73 theorem pairMomentum_mul_coefficient (b : B) :
74     pairMomentum * pairCoefficient b = pairCoefficient b * pairMomentum := by
75   change
76     normalVariable outerD * normalCoefficient outerD (normalCoefficient innerD b) =
77       normalCoefficient outerD (normalCoefficient innerD b) * normalVariable outerD
78   have h := normalVariable_mul_coefficient outerD (normalCoefficient innerD b)
79   rw [coordinateDerivation_coefficient, map_zero, add_zero] at h
80   exact h
81 
82 /-- The new pair satisfies the Weyl relation in the manuscript convention. -/
83 theorem pairMomentum_mul_coordinate :
84     pairMomentum (B := B) * pairCoordinate =
85       pairCoordinate * pairMomentum + 1 := by
86   change
87     normalVariable outerD * normalCoefficient outerD (normalVariable innerD) =
88       normalCoefficient outerD (normalVariable innerD) * normalVariable outerD + 1
89   have h := normalVariable_mul_coefficient outerD
90     (normalVariable innerD : CoordinateStage (B := B))
91   rw [coordinateDerivation_variable, map_one] at h
92   exact h
93 
94 theorem pairCoordinate_commutator_momentum :
95     pairCoordinate (B := B) * pairMomentum - pairMomentum * pairCoordinate = -1 := by
96   rw [pairMomentum_mul_coordinate]
97   noncomm_ring

103 /-- The scalar algebra structure on the central-coordinate stage. -/
104 def coordinateStageAlgebra : Algebra k (CoordinateStage (B := B)) :=
105   Stafford38.OreScalarAlgebra.normalOreAlgebra innerD fun c => by
106     exact zeroDerivation_apply (algebraMap k B c)
107 
108 theorem coordinateDerivation_algebraMap (c : k) :
109     @coordinateDerivation B _
110       (@algebraMap k (CoordinateStage (B := B)) _ _ coordinateStageAlgebra c) = 0 := by
111   change coordinateDerivation
112     (normalCoefficient innerD (algebraMap k B c)) = 0
113   exact coordinateDerivation_coefficient (algebraMap k B c)
114 
115 /-- The scalar algebra structure on the full coordinate-momentum pair. -/
116 def pairStageAlgebra : Algebra k (PairStage (B := B)) := by
117   letI : Algebra k (CoordinateStage (B := B)) := coordinateStageAlgebra
118   exact Stafford38.OreScalarAlgebra.normalOreAlgebra outerD
119     coordinateDerivation_algebraMap
120 
121 theorem pairStageAlgebra_algebraMap :
122     @algebraMap k (PairStage (B := B)) _ _ pairStageAlgebra =
123       (pairCoefficient (B := B)).comp (algebraMap k B) := rfl
