7 /-!
8 # Universal property of a coordinate-momentum pair stage
9 
10 The construction is the universal property of `NormalOre` applied twice.
11 First a coefficient map and a commuting coordinate define a map from the
12 central-coordinate stage.  A momentum satisfying the derivation relation
13 against that map then defines a map from `PairStage`.
14 -/

28 /-- The Ore ambient determined by a coefficient map and a centralizing
29 coordinate image. -/
30 def coordinateAmbient (f : B →+* A) (X : A)
31     (hX : ∀ b, X * f b = f b * X) :
32     OreAmbient B A (zeroDerivation : OreDivisionDerivation B) where
33   embed := f
34   x := X
35   relation := by
36     intro b
37     simpa using hX b
38 
39 /-- The universal map from the central-coordinate stage. -/
40 def coordinateLift (f : B →+* A) (X : A)
41     (hX : ∀ b, X * f b = f b * X) :
42     CoordinateStage (B := B) →+* A :=
43   oreLift zeroDerivation (coordinateAmbient f X hX)
44 
45 @[simp] theorem coordinateLift_coefficient (f : B →+* A) (X : A)
46     (hX : ∀ b, X * f b = f b * X) (b : B) :
47     coordinateLift f X hX (normalCoefficient zeroDerivation b) = f b :=
48   oreLift_coefficient zeroDerivation (coordinateAmbient f X hX) b
49 
50 @[simp] theorem coordinateLift_variable (f : B →+* A) (X : A)
51     (hX : ∀ b, X * f b = f b * X) :
52     coordinateLift f X hX (normalVariable zeroDerivation) = X :=
53   oreLift_variable zeroDerivation (coordinateAmbient f X hX)
54 
55 /-- A map out of the coordinate stage is determined by the old coefficients
56 and the new coordinate. -/
57 theorem coordinateLift_unique (f : B →+* A) (X : A)
58     (hX : ∀ b, X * f b = f b * X)
59     (g : CoordinateStage (B := B) →+* A)
60     (hCoefficient : ∀ b, g (normalCoefficient zeroDerivation b) = f b)
61     (hVariable : g (normalVariable zeroDerivation) = X) :
62     g = coordinateLift f X hX :=
63   oreLift_unique zeroDerivation (coordinateAmbient f X hX) g
64     hCoefficient hVariable

66 private theorem momentum_mul_coordinate_pow (X P : A)
67     (hPX : P * X = X * P + 1) (n : ℕ) :
68     P * X ^ n = X ^ n * P + n • X ^ (n - 1) := by
69   induction n with
70   | zero => simp
71   | succ n ih =>
72       rw [pow_succ, ← mul_assoc, ih, add_mul, mul_assoc, hPX, mul_add]
73       by_cases hn : n = 0
74       · subst n
75         simp
76       · have hpow : X ^ (n - 1) * X = X ^ n := by
77           rw [← pow_succ, Nat.sub_add_cancel (Nat.one_le_iff_ne_zero.mpr hn)]
78         rw [nsmul_eq_mul, mul_assoc, hpow, ← nsmul_eq_mul]
79         simp only [Nat.add_sub_cancel, mul_one, mul_assoc, succ_nsmul]
80         abel

82 /-- The coefficient and generator relations imply the full coordinate
83 derivation relation.  No commutativity of either coefficient ring is used. -/
84 theorem coordinateLift_derivation_relation (f : B →+* A) (X P : A)
85     (hX : ∀ b, X * f b = f b * X)
86     (hP : ∀ b, P * f b = f b * P)
87     (hPX : P * X = X * P + 1) (z : CoordinateStage (B := B)) :
88     P * coordinateLift f X hX z =
89       coordinateLift f X hX z * P +
90         coordinateLift f X hX (coordinateDerivation z) := by
91   rcases normalForm_surjective zeroDerivation z with ⟨p, rfl⟩
92   induction p using Polynomial.induction_on' with
93   | add p q hp hq =>
94       rw [normalForm_add, (coordinateLift f X hX).map_add,
95         coordinateDerivation.map_add, (coordinateLift f X hX).map_add,
96         mul_add, add_mul, hp, hq]
97       abel
98   | monomial n b =>
99       rw [coordinateDerivation_normalForm, Polynomial.derivative_monomial,
100         normalForm_monomial, map_mul, map_pow,
101         coordinateLift_coefficient, coordinateLift_variable,
102         normalForm_monomial, map_mul, map_pow,
103         coordinateLift_coefficient, coordinateLift_variable]
104       rw [← mul_assoc, hP, mul_assoc, momentum_mul_coordinate_pow X P hPX n]
105       simp only [add_mul, mul_add, mul_assoc]
106       congr 1
107       rw [map_mul, map_natCast, nsmul_eq_mul]
108       simp only [mul_assoc]

110 /-- The outer Ore ambient determined by commuting coefficient images and a
111 coordinate-momentum Weyl pair. -/
112 def pairAmbient (f : B →+* A) (X P : A)
113     (hX : ∀ b, X * f b = f b * X)
114     (hP : ∀ b, P * f b = f b * P)
115     (hPX : P * X = X * P + 1) :
116     OreAmbient (CoordinateStage (B := B)) A coordinateDerivation where
117   embed := coordinateLift f X hX
118   x := P
119   relation := coordinateLift_derivation_relation f X P hX hP hPX

121 /-- The universal ring map from a coordinate-momentum pair stage. -/
122 def pairLift (f : B →+* A) (X P : A)
123     (hX : ∀ b, X * f b = f b * X)
124     (hP : ∀ b, P * f b = f b * P)
125     (hPX : P * X = X * P + 1) :
126     PairStage (B := B) →+* A :=
127   oreLift coordinateDerivation (pairAmbient f X P hX hP hPX)
128 
129 @[simp] theorem pairLift_coefficient (f : B →+* A) (X P : A)
130     (hX : ∀ b, X * f b = f b * X)
131     (hP : ∀ b, P * f b = f b * P)
132     (hPX : P * X = X * P + 1)
133     (b : B) :
134     pairLift f X P hX hP hPX (pairCoefficient b) = f b := by
135   change pairLift f X P hX hP hPX
136       (normalCoefficient coordinateDerivation
137         (normalCoefficient zeroDerivation b)) = f b
138   rw [pairLift, oreLift_coefficient]
139   change coordinateLift f X hX (normalCoefficient zeroDerivation b) = f b
140   exact coordinateLift_coefficient f X hX b
141 
142 @[simp] theorem pairLift_coordinate (f : B →+* A) (X P : A)
143     (hX : ∀ b, X * f b = f b * X)
144     (hP : ∀ b, P * f b = f b * P)
145     (hPX : P * X = X * P + 1) :
146     pairLift f X P hX hP hPX pairCoordinate = X := by
147   change pairLift f X P hX hP hPX
148       (normalCoefficient coordinateDerivation
149         (normalVariable zeroDerivation)) = X
150   rw [pairLift, oreLift_coefficient]
151   change coordinateLift f X hX (normalVariable zeroDerivation) = X
152   exact coordinateLift_variable f X hX
153 
154 @[simp] theorem pairLift_momentum (f : B →+* A) (X P : A)
155     (hX : ∀ b, X * f b = f b * X)
156     (hP : ∀ b, P * f b = f b * P)
157     (hPX : P * X = X * P + 1) :
158     pairLift f X P hX hP hPX pairMomentum = P := by
159   exact oreLift_variable coordinateDerivation (pairAmbient f X P hX hP hPX)

161 /-- A ring map out of `PairStage` is determined by the old coefficients and
162 the two newly adjoined generators. -/
163 theorem pairLift_unique (f : B →+* A) (X P : A)
164     (hX : ∀ b, X * f b = f b * X)
165     (hP : ∀ b, P * f b = f b * P)
166     (hPX : P * X = X * P + 1)
167     (g : PairStage (B := B) →+* A)
168     (hCoefficient : ∀ b, g (pairCoefficient b) = f b)
169     (hCoordinate : g pairCoordinate = X)
170     (hMomentum : g pairMomentum = P) :
171     g = pairLift f X P hX hP hPX := by
172   let inner : CoordinateStage (B := B) →+* A :=
173     g.comp (normalCoefficient coordinateDerivation)
174   have hInner : inner = coordinateLift f X hX := by
175     apply coordinateLift_unique f X hX
176     · intro b
177       exact hCoefficient b
178     · exact hCoordinate
179   apply oreLift_unique coordinateDerivation (pairAmbient f X P hX hP hPX)
180   · intro z
181     exact DFunLike.congr_fun hInner z
182   · exact hMomentum

191 /-- The pair lift as a scalar-preserving algebra map. -/
192 def pairLiftAlgHom (f : B →ₐ[k] A) (X P : A)
193     (hX : ∀ b, X * f b = f b * X)
194     (hP : ∀ b, P * f b = f b * P)
195     (hPX : P * X = X * P + 1) :
196     PairStage (B := B) →ₐ[k] A where
197   toRingHom := pairLift f.toRingHom X P hX hP hPX
198   commutes' c := by
199     rw [pairStageAlgebra_algebraMap]
200     exact pairLift_coefficient f.toRingHom X P hX hP hPX (algebraMap k B c)
201       |>.trans (f.commutes c)
202 
203 @[simp] theorem pairLiftAlgHom_coefficient (f : B →ₐ[k] A) (X P : A)
204     (hX : ∀ b, X * f b = f b * X)
205     (hP : ∀ b, P * f b = f b * P)
206     (hPX : P * X = X * P + 1)
207     (b : B) :
208     pairLiftAlgHom f X P hX hP hPX (pairCoefficient b) = f b := by
209   change pairLift f.toRingHom X P hX hP hPX (pairCoefficient b) = f b
210   exact pairLift_coefficient f.toRingHom X P hX hP hPX b

228 theorem pairLiftAlgHom_unique (f : B →ₐ[k] A) (X P : A)
229     (hX : ∀ b, X * f b = f b * X)
230     (hP : ∀ b, P * f b = f b * P)
231     (hPX : P * X = X * P + 1)
232     (g : PairStage (B := B) →ₐ[k] A)
233     (hCoefficient : ∀ b, g (pairCoefficient b) = f b)
234     (hCoordinate : g pairCoordinate = X)
235     (hMomentum : g pairMomentum = P) :
236     g = pairLiftAlgHom f X P hX hP hPX := by
237   apply AlgHom.ext
238   intro z
239   exact DFunLike.congr_fun
240     (pairLift_unique f.toRingHom X P hX hP hPX g.toRingHom
241       hCoefficient hCoordinate hMomentum) z
