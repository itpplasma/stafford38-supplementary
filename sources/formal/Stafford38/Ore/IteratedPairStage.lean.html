6 /-!
7 # Iterated coordinate-momentum pair stages
8 
9 This file recursively repeats the checked `PairStage` construction over an
10 arbitrary coefficient ring.  It records only the resulting tower and the
11 canonical data introduced at each successor; it does not identify the tower
12 with a presented Weyl algebra.
13 -/

23 /-- A type together with the ring structure used at the next Ore stage. -/
24 structure RingStage where
25   carrier : Type u
26   ring : Ring carrier
27 
28 variable (B : Type u) [Ring B]
29 
30 /-- The recursively constructed ring data after adjoining `n` Weyl pairs. -/
31 def iteratedPairData : Nat → RingStage
32   | 0 => ⟨B, inferInstance⟩
33   | n + 1 =>
34       let previous := iteratedPairData n
35       letI : Ring previous.carrier := previous.ring
36       ⟨PairStage (B := previous.carrier), inferInstance⟩
37 
38 /-- The ring obtained from `B` after recursively adjoining `n` checked
39 coordinate-momentum pairs. -/
40 def IteratedPairStage (n : Nat) : Type u :=
41   (iteratedPairData B n).carrier
42 
43 instance iteratedPairStageRing (n : Nat) : Ring (IteratedPairStage (B := B) n) :=
44   (iteratedPairData B n).ring
45 
46 /-- The zeroth stage is the original coefficient type. -/
47 theorem iteratedPairStage_zero : IteratedPairStage (B := B) 0 = B := rfl
48 
49 /-- Every successor is definitionally the checked `PairStage` construction
50 over its predecessor. -/
51 theorem iteratedPairStage_succ (n : Nat) :
52     IteratedPairStage (B := B) (n + 1) =
53       PairStage (B := IteratedPairStage (B := B) n) := rfl
54 
55 /-- The canonical embedding from stage `n` into stage `n + 1`. -/
56 def stageEmbedding (n : Nat) :
57     IteratedPairStage (B := B) n →+* IteratedPairStage (B := B) (n + 1) :=
58   pairCoefficient (B := IteratedPairStage (B := B) n)
59 
60 /-- The coordinate introduced at the successor of stage `n`. -/
61 def stageCoordinate (n : Nat) : IteratedPairStage (B := B) (n + 1) :=
62   pairCoordinate (B := IteratedPairStage (B := B) n)
63 
64 /-- The momentum introduced at the successor of stage `n`. -/
65 def stageMomentum (n : Nat) : IteratedPairStage (B := B) (n + 1) :=
66   pairMomentum (B := IteratedPairStage (B := B) n)
67 
68 /-- The coordinate introduced at stage `n + 1` commutes with the embedded
69 predecessor ring. -/
70 theorem stageCoordinate_mul_embedding (n : Nat)
71     (b : IteratedPairStage (B := B) n) :
72     stageCoordinate B n * stageEmbedding B n b =
73       stageEmbedding B n b * stageCoordinate B n :=
74   pairCoordinate_mul_coefficient b
75 
76 /-- The momentum introduced at stage `n + 1` commutes with the embedded
77 predecessor ring. -/
78 theorem stageMomentum_mul_embedding (n : Nat)
79     (b : IteratedPairStage (B := B) n) :
80     stageMomentum B n * stageEmbedding B n b =
81       stageEmbedding B n b * stageMomentum B n :=
82   pairMomentum_mul_coefficient b
83 
84 /-- The generators introduced at stage `n + 1` satisfy the checked Weyl
85 relation. -/
86 theorem stageMomentum_mul_coordinate (n : Nat) :
87     stageMomentum B n * stageCoordinate B n =
88       stageCoordinate B n * stageMomentum B n + 1 :=
89   pairMomentum_mul_coordinate
