26 theorem bernsteinDegree_eq_of_piece_of_principal_ne_zero
27     (k : Type u) [Field k] {n N : ℕ} {d : WeylAlg k n}
28     (hp : d ∈ bernsteinPiece k n N)
29     (hP : presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0) :
30     bernsteinDegree k d = N := by
31   rw [Stafford38FixedSourceChallengeTransport.bernsteinDegree_eq k d]
32   let f := presentedNormalFormLinearEquiv k n d
33   have hle : MvPolynomial.weightedTotalDegree (@bernsteinWeight n) f ≤ N := by
34     apply Finset.sup_le
35     intro m hm
36     exact (hp m (MvPolynomial.mem_support_iff.mp hm))
37   rcases MvPolynomial.support_nonempty.mpr hP with ⟨m, hm⟩
38   have hm : MvPolynomial.coeff m
39       (presentedPrincipalComponent k (@bernsteinWeight n) N d) ≠ 0 :=
40     MvPolynomial.mem_support_iff.mp hm
41   change AddMonoidAlgebra.coeff
42     (presentedPrincipalComponent k (@bernsteinWeight n) N d) m ≠ 0 at hm
43   have hcoeff := hm
44   change AddMonoidAlgebra.coeff
45     (presentedPrincipalComponent k (@bernsteinWeight n) N d) m ≠ 0 at hcoeff
46   rw [coeff_presentedPrincipalComponent] at hcoeff
47   have hweight : Finsupp.weight (@bernsteinWeight n) m = N := by
48     by_contra hne
49     have hz : AddMonoidAlgebra.coeff
50         (presentedPrincipalComponent k (@bernsteinWeight n) N d) m = 0 := by
51       rw [coeff_presentedPrincipalComponent]
52       simp [hne]
53     exact hm hz
54   have hfd : MvPolynomial.coeff m f ≠ 0 := by
55     rw [if_pos hweight] at hcoeff
56     simpa [f] using hcoeff
57   have hge : N ≤ MvPolynomial.weightedTotalDegree (@bernsteinWeight n) f := by
58     calc
59       N = Finsupp.weight (@bernsteinWeight n) m := hweight.symm
60       _ ≤ MvPolynomial.weightedTotalDegree (@bernsteinWeight n) f :=
61         MvPolynomial.le_weightedTotalDegree _ (MvPolynomial.mem_support_iff.mpr hfd)
62   exact Nat.le_antisymm hle hge

64 end
