22 /-! The (actual) Bernstein degree, read directly from checked PBW normal form. -/
23 def bernsteinDegree (k : Type u) [Field k] {n : ℕ}
24     (d : PresentedWeyl k n) : ℕ :=
25   MvPolynomial.weightedTotalDegree (@bernsteinWeight n)
26     (presentedNormalFormLinearEquiv k n d)

40 theorem bernsteinDegree_eq_of_piece_of_principal_ne_zero
41     (k : Type u) [Field k] {n N : ℕ} {d : PresentedWeyl k n}
42     (hp : d ∈ bernsteinPiece k n N)
43     (hP : presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0) :
44     bernsteinDegree k d = N := by
45   let f := presentedNormalFormLinearEquiv k n d
46   have hle : bernsteinDegree k d ≤ N := by
47     apply Finset.sup_le
48     intro m hm
49     exact (hp m (MvPolynomial.mem_support_iff.mp hm))
50   rcases MvPolynomial.support_nonempty.mpr hP with ⟨m, hm⟩
51   have hm : MvPolynomial.coeff m
52       (presentedPrincipalComponent k (@bernsteinWeight n) N d) ≠ 0 :=
53     MvPolynomial.mem_support_iff.mp hm
54   change AddMonoidAlgebra.coeff
55     (presentedPrincipalComponent k (@bernsteinWeight n) N d) m ≠ 0 at hm
56   have hcoeff := hm
57   change AddMonoidAlgebra.coeff
58     (presentedPrincipalComponent k (@bernsteinWeight n) N d) m ≠ 0 at hcoeff
59   rw [coeff_presentedPrincipalComponent] at hcoeff
60   have hweight : monomialWeight (@bernsteinWeight n) m = N := by
61     by_contra hne
62     have hz : AddMonoidAlgebra.coeff
63         (presentedPrincipalComponent k (@bernsteinWeight n) N d) m = 0 := by
64       rw [coeff_presentedPrincipalComponent]
65       simp [hne]
66     exact hm hz
67   have hfd : MvPolynomial.coeff m f ≠ 0 := by
68     rw [if_pos hweight] at hcoeff
69     simpa [f] using hcoeff
70   have hge : N ≤ bernsteinDegree k d := by
71     calc
72       N = monomialWeight (@bernsteinWeight n) m := hweight.symm
73       _ ≤ bernsteinDegree k d :=
74         MvPolynomial.le_weightedTotalDegree _ (MvPolynomial.mem_support_iff.mpr hfd)
75   exact Nat.le_antisymm hle hge
76 
77 end
