11 # Paper-facing scalar extension and quotient descent
12 
13 This file records the algebraic descent steps in the paper's proof.  The
14 coefficient extension is an arbitrary field extension (so an algebraic
15 closure is available as the paper's closed coefficient field).  It keeps the
16 actual filtered quotient, its associated graded annihilator, and the
17 unfiltered right quotient distinct.
18 -/

48 /-- Moving the field-extension factor to the left identifies the image of a
49 subspace quotient kernel with ordinary linear base change. -/
50 theorem comm_map_tensorSubmodule_range (J : Submodule k M) :
51     Submodule.map (TensorProduct.comm k M K).toLinearMap
52       (LinearMap.range (TensorProduct.map J.subtype
53         (LinearMap.id : K →ₗ[k] K))) = (J.baseChange K).restrictScalars k := by
54   have hcomp :
55       (TensorProduct.comm k M K).toLinearMap.comp
56           (TensorProduct.map J.subtype (LinearMap.id : K →ₗ[k] K)) =
57         ((J.subtype.baseChange K).restrictScalars k).comp
58           (TensorProduct.comm k J K).toLinearMap := by
59     ext j c
60     simp [LinearMap.comp_apply]
61   rw [← LinearMap.range_comp, hcomp, LinearMap.range_comp]
62   rw [LinearEquiv.range, Submodule.map_top]
63   change LinearMap.range ((J.subtype.baseChange K).restrictScalars k) =
64     Submodule.restrictScalars k (J.subtype.baseChange K).range
65   rw [LinearMap.range_restrictScalars]
66 
67 end TensorQuotient

69 /-- Scalar extension of the actual canonical quotient is the quotient by the
70 scalar-extended canonical right ideal.  The proof goes through the tensor
71 quotient equivalence and the already-checked PBW identification. -/
72 def canonicalRightQuotient_scalarExtension_equiv
73     (n N : ℕ) (d : WeylAlg k (n + 1)) :
74     TensorProduct k (WeylAlg k (n + 1) ⧸
75       rightIdealKSubmodule k
76         (canonicalRightIdeal (presentedCoordinate k n) d N)) K ≃ₗ[k]
77       RightQuotient
78         (canonicalRightIdeal (presentedCoordinate K n)
79           (presentedWeylScalarExtension (k := k) (K := K) (n + 1) d) N) := by
80   let I := canonicalRightIdeal (presentedCoordinate k n) d N
81   let J := rightIdealKSubmodule k I
82   let eK := pbwTensorEquiv (k := k) (K := K) (n + 1)
83   let IK := canonicalRightIdeal (presentedCoordinate K n)
84     (presentedWeylScalarExtension (k := k) (K := K) (n + 1) d) N
85   let JK := rightIdealKSubmodule K IK
86   let f := (TensorProduct.comm k (WeylAlg k (n + 1)) K).trans
87     (LinearEquiv.restrictScalars k eK)
88   have hf : f.toLinearMap =
89       (LinearEquiv.restrictScalars k eK).toLinearMap.comp
90         (TensorProduct.comm k (WeylAlg k (n + 1)) K).toLinearMap := by
91     ext z
92     rfl
93   let D := LinearMap.range (TensorProduct.map J.subtype
94     (LinearMap.id : K →ₗ[k] K))
95   have hden : Submodule.map f.toLinearMap D = JK.restrictScalars k := by
96     rw [hf, Submodule.map_comp]
97     rw [comm_map_tensorSubmodule_range]
98     calc
99       Submodule.map (LinearEquiv.restrictScalars k eK).toLinearMap
100         (Submodule.restrictScalars k (J.baseChange K)) =
101         Submodule.restrictScalars k
102           (Submodule.map eK.toLinearMap (J.baseChange K)) := by
103           symm
104           exact Submodule.restrictScalars_map eK.toLinearMap (J.baseChange K)
105       _ = Submodule.restrictScalars k JK := by
106         rw [pbwTensorEquiv_map_canonicalRightIdeal]
107   have hquot :
108       TensorProduct k (WeylAlg k (n + 1) ⧸ J) K ≃ₗ[k]
109         (WeylAlg K (n + 1) ⧸ JK.restrictScalars k) :=
110     (TensorProduct.quotientTensorEquiv K J).trans
111       (Submodule.Quotient.equiv D (JK.restrictScalars k) f hden)
112   exact hquot.trans
113     ((Submodule.Quotient.restrictScalarsEquiv k JK).trans
114       (LinearEquiv.restrictScalars k
115         (filteredRightQuotientEquivRightQuotient K IK)))

117 /-- Faithfully flat field extension descends vanishing of the actual right
118 quotient.  The scalar-extension equivalence above is the concrete bridge; the
119 last step is the standard faithful-flat reflection of a zero module. -/
120 theorem canonicalRightQuotient_subsingleton_of_scalarExtension
121     (n N : ℕ) (d : WeylAlg k (n + 1))
122     (hK : Subsingleton (RightQuotient
123       (canonicalRightIdeal (presentedCoordinate K n)
124         (presentedWeylScalarExtension (k := k) (K := K) (n + 1) d) N))) :
125     Subsingleton (RightQuotient
126       (canonicalRightIdeal (presentedCoordinate k n) d N)) := by
127   let I := canonicalRightIdeal (presentedCoordinate k n) d N
128   let e0 := filteredRightQuotientEquivRightQuotient k I
129   let e := (TensorProduct.congr e0 (LinearEquiv.refl k K)).trans
130     (canonicalRightQuotient_scalarExtension_equiv (k := k) (K := K) n N d)
131   have htensor : Subsingleton (TensorProduct k (RightQuotient I) K) :=
132     (Equiv.subsingleton_congr e.toEquiv).mpr hK
133   exact Module.FaithfullyFlat.rTensor_reflects_triviality k K (RightQuotient I)

135 /-- The paper's support descent, stated for the canonical quotient: empty
136 support after a field extension implies empty support before extension.  It
137 factors through vanishing of the actual quotient on both fields. -/
138 theorem canonicalSupport_empty_of_scalarExtension_empty
139     (n N : ℕ) (d : WeylAlg k (n + 1))
140     (hK : orderCharacteristicSupport K
141       (canonicalRightIdeal (presentedCoordinate K n)
142         (presentedWeylScalarExtension (k := k) (K := K) (n + 1) d) N) = ∅) :
143     orderCharacteristicSupport k
144       (canonicalRightIdeal (presentedCoordinate k n) d N) = ∅ := by
145   apply (orderCharacteristicSupport_eq_empty_iff_rightQuotient_subsingleton k
146     (canonicalRightIdeal (presentedCoordinate k n) d N)).mpr
147   apply canonicalRightQuotient_subsingleton_of_scalarExtension
148     (k := k) (K := K) n N d
149   exact rightQuotient_subsingleton_of_orderCharacteristicSupport_eq_empty K _ hK

151 /-- The canonical-support descent contract used by the official assembly,
152 proved through vanishing of the concrete canonical right quotient. -/
153 theorem canonicalSupportDescent_via_quotient :
154     Stafford38.CanonicalSupportVanishingReduction.CanonicalSupportDescent.{u} := by
155   intro hclosed
156   intro k _ _ n N d hN hd
157   let e := presentedWeylScalarExtension (k := k)
158     (K := AlgebraicClosure k) (n + 1)
159   have hdK : IsPBWMonicAt (AlgebraicClosure k)
160       (.inr (0 : Fin (n + 1))) N (e d) := by
161     exact presentedWeylScalarExtension_isPBWMonicAt
162       (k := k) (K := AlgebraicClosure k) n N hd
163   have hK : orderCharacteristicSupport (AlgebraicClosure k)
164       (canonicalRightIdeal (presentedCoordinate (AlgebraicClosure k) n)
165         (e d) N) = ∅ := hclosed (AlgebraicClosure k) n N (e d) hN hdK
166   exact canonicalSupport_empty_of_scalarExtension_empty
167     (k := k) (K := AlgebraicClosure k) n N d hK
