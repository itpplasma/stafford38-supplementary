10 /-!
11 # Principal components and leading-symbol multiplication
12 
13 Weighted homogeneous projection of the checked PBW normal form defines the
14 degree-`N` principal component of a filtered Weyl element. Exact one-pair
15 normal ordering proves that, for both the Bernstein and differential-order
16 filtrations, the principal component of a product is the product of the
17 principal components. No associated graded identification is assumed here.
18 -/

33 /-- The homogeneous component at the declared filtration bound `N`. It is zero
34 when the element has strictly smaller actual weighted degree. -/
35 def presentedPrincipalComponent {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ) :
36     WeylAlg k n →ₗ[k] SymbolRing k n :=
37   (MvPolynomial.weightedHomogeneousComponent w N).comp
38     (presentedNormalFormLinearEquiv k n).toLinearMap

51 theorem presentedPrincipalComponent_eq_zero_of_mem_of_lt {n L T : ℕ}
52     (w : PhaseVar n → ℕ) (z : WeylAlg k n)
53     (hz : z ∈ presentedWeightPiece k w L) (hLT : L < T) :
54     presentedPrincipalComponent k w T z = 0 := by
55   ext m
56   change AddMonoidAlgebra.coeff (presentedPrincipalComponent k w T z) m = 0
57   rw [coeff_presentedPrincipalComponent]
58   by_cases hm : Finsupp.weight w m = T
59   · rw [if_pos hm]
60     have hcoeff :
61         MvPolynomial.coeff m (presentedNormalFormLinearEquiv k n z) = 0 := by
62       by_contra hne
63       have hle := (mem_presentedWeightPiece k w L z).mp hz m hne
64       omega
65     change AddMonoidAlgebra.coeff (presentedNormalFormLinearEquiv k n z) m = 0 at hcoeff
66     rw [hcoeff]
67   · rw [if_neg hm]

133 theorem weightedHomogeneousComponent_monomial_mul {n T : ℕ}
134     (w : PhaseVar n → ℕ) (s : PhaseVar n →₀ ℕ)
135     (f : SymbolRing k n) :
136     MvPolynomial.weightedHomogeneousComponent w
137         (T + Finsupp.weight w s) (MvPolynomial.monomial s 1 * f) =
138       MvPolynomial.monomial s 1 *
139         MvPolynomial.weightedHomogeneousComponent w T f := by
140   classical
141   ext q
142   rw [MvPolynomial.coeff_weightedHomogeneousComponent,
143     MvPolynomial.coeff_monomial_mul',
144     MvPolynomial.coeff_monomial_mul']
145   by_cases hsq : s ≤ q
146   · rw [if_pos hsq, if_pos hsq, one_mul, one_mul,
147       MvPolynomial.coeff_weightedHomogeneousComponent]
148     have hsplit : s + (q - s) = q := by
149       rw [add_comm]
150       exact tsub_add_cancel_of_le hsq
151     have hweight :
152         Finsupp.weight w q =
153           Finsupp.weight w s + Finsupp.weight w (q - s) := by
154       rw [← hsplit]
155       simp [Finsupp.weight_apply, Finsupp.sum_add_index, add_mul]
156     rw [hweight]
157     by_cases hrest : Finsupp.weight w (q - s) = T
158     · simp [hrest, Nat.add_comm]
159     · have hne :
160           Finsupp.weight w s + Finsupp.weight w (q - s) ≠
161             T + Finsupp.weight w s := by
162         omega
163       simp [hrest, hne]
164   · rw [if_neg hsq, if_neg hsq]
165     simp

255 theorem phaseMonomial_succ_product (n : ℕ)
256     (a p c q : Fin (n + 1) → ℕ) :
257     MvPolynomial.monomial (phaseExponent a p) (1 : k) *
258         MvPolynomial.monomial (phaseExponent c q) 1 =
259       MvPolynomial.monomial
260           (extendPhaseExponent n (a 0 + c 0) (p 0 + q 0) 0) 1 *
261         MvPolynomial.rename Stafford38FixedSourceChallenge.oldIndex
262           (MvPolynomial.monomial
263               (phaseExponent (fun i => a i.succ) (fun i => p i.succ)) 1 *
264             MvPolynomial.monomial
265               (phaseExponent (fun i => c i.succ) (fun i => q i.succ)) 1) := by
266   have hexp :
267       phaseExponent a p + phaseExponent c q =
268         extendPhaseExponent n (a 0 + c 0) (p 0 + q 0) 0 +
269           (phaseExponent (fun i => a i.succ) (fun i => p i.succ) +
270             phaseExponent (fun i => c i.succ) (fun i => q i.succ)).mapDomain
271               Stafford38FixedSourceChallenge.oldIndex := by
272     rw [phaseExponent_succ_eq_extend, phaseExponent_succ_eq_extend]
273     simp [extendPhaseExponent, map_add, add_assoc, add_comm, add_left_comm]
274     rw [Finsupp.mapDomain_add]
275   rw [MvPolynomial.monomial_mul, MvPolynomial.monomial_mul,
276     MvPolynomial.rename_monomial, MvPolynomial.monomial_mul]
277   simp only [one_mul]
278   rw [hexp]

280 theorem presentedOrderedMonomial_mul_principal_bernstein :
281     ∀ (n : ℕ) (a p c q : Fin n → ℕ),
282       presentedPrincipalComponent k (@bernsteinWeight n)
283           (Finsupp.weight (@bernsteinWeight n) (phaseExponent a p) +
284             Finsupp.weight (@bernsteinWeight n) (phaseExponent c q))
285           (presentedOrderedMonomial k n a p *
286             presentedOrderedMonomial k n c q) =
287         MvPolynomial.monomial (phaseExponent a p) 1 *
288           MvPolynomial.monomial (phaseExponent c q) 1 := by
289   intro n
290   induction n with
291   | zero =>
292       intro a p c q
293       have hap : phaseExponent a p = 0 := by
294         ext i
295         exact Sum.elim Fin.elim0 Fin.elim0 i
296       have hcq : phaseExponent c q = 0 := by
297         ext i
298         exact Sum.elim Fin.elim0 Fin.elim0 i
299       rw [hap, hcq]
300       simp [presentedOrderedMonomial, Finsupp.weight_apply,
301         presentedPrincipalComponent, presentedNormalFormLinearEquiv_one,
302         weightedHomogeneousComponent_monomial]
303   | succ n ih =>
304       intro a p c q
305       let L :=
306         Finsupp.weight (@bernsteinWeight n)
307             (phaseExponent (fun i => a i.succ) (fun i => p i.succ)) +
308           Finsupp.weight (@bernsteinWeight n)
309             (phaseExponent (fun i => c i.succ) (fun i => q i.succ))
310       have htarget :
311           Finsupp.weight (@bernsteinWeight (n + 1)) (phaseExponent a p) +
312               Finsupp.weight (@bernsteinWeight (n + 1)) (phaseExponent c q) =
313             L + (a 0 + c 0) + (p 0 + q 0) := by
314         rw [monomialWeight_phaseExponent_succ_bernstein,
315           monomialWeight_phaseExponent_succ_bernstein]
316         simp only [L]
317         omega
318       rw [htarget]
319       change
320         presentedPrincipalComponent k (@bernsteinWeight (n + 1))
321             (L + (a 0 + c 0) + (p 0 + q 0))
322             (presentedCoefficientOrdered k n (a 0) (p 0)
323                 (presentedOrderedMonomial k n
324                   (fun i => a i.succ) (fun i => p i.succ)) *
325               presentedCoefficientOrdered k n (c 0) (q 0)
326                 (presentedOrderedMonomial k n
327                   (fun i => c i.succ) (fun i => q i.succ))) = _
328       rw [presentedCoefficientOrdered_mul]
329       simp only [map_sum]
330       rw [Finset.sum_eq_single 0]
331       · simp only [Nat.zero_le, if_true, Nat.choose_zero_right,
332           Nat.descFactorial_zero, mul_one, one_nsmul, Nat.sub_zero]
333         rw [presentedPrincipalComponent_coefficientOrdered_bernstein]
334         have hih := ih
335           (fun i => a i.succ) (fun i => p i.succ)
336           (fun i => c i.succ) (fun i => q i.succ)
337         change presentedPrincipalComponent k (@bernsteinWeight n) L
338             (presentedOrderedMonomial k n
339                 (fun i => a i.succ) (fun i => p i.succ) *
340               presentedOrderedMonomial k n
341                 (fun i => c i.succ) (fun i => q i.succ)) = _ at hih
342         rw [hih]
343         exact (phaseMonomial_succ_product k n a p c q).symm
344       · intro i hi hi0
345         have hib : i ≤ p 0 := by
346           have := Finset.mem_range.mp hi
347           omega
348         by_cases hic : i ≤ c 0
349         · simp only [if_pos hic]
350           have hold :
351               presentedOrderedMonomial k n
352                     (fun j => a j.succ) (fun j => p j.succ) *
353                   presentedOrderedMonomial k n
354                     (fun j => c j.succ) (fun j => q j.succ) ∈
355                 bernsteinPiece k n L := by
356             simpa [L] using
357               presentedOrderedMonomial_mul_mem_bernsteinPiece k n
358                 (fun j => a j.succ) (fun j => p j.succ)
359                 (fun j => c j.succ) (fun j => q j.succ)
360           have hterm := presentedCoefficientOrdered_mem_bernsteinPiece k n L
361             (a 0 + c 0 - i) (p 0 + q 0 - i)
362             (presentedOrderedMonomial k n
363                 (fun j => a j.succ) (fun j => p j.succ) *
364               presentedOrderedMonomial k n
365                 (fun j => c j.succ) (fun j => q j.succ)) hold
366           have hlt :
367               L + (a 0 + c 0 - i) + (p 0 + q 0 - i) <
368                 L + (a 0 + c 0) + (p 0 + q 0) := by
369             omega
370           have hz := presentedPrincipalComponent_eq_zero_of_mem_of_lt k
371             (@bernsteinWeight (n + 1)) _ hterm hlt
372           rw [map_nsmul, hz, nsmul_zero]
373         · simp [hic]
374       · simp
