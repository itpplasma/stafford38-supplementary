8 /-!
9 # Principal components and leading-symbol multiplication
10 
11 Weighted homogeneous projection of the checked PBW normal form defines the
12 degree-`N` principal component of a filtered Weyl element. Exact one-pair
13 normal ordering proves that, for both the Bernstein and differential-order
14 filtrations, the principal component of a product is the product of the
15 principal components. No associated graded identification is assumed here.
16 -/

36 /-- The homogeneous component at the declared filtration bound `N`. It is zero
37 when the element has strictly smaller actual weighted degree. -/
38 def presentedPrincipalComponent {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ) :
39     PresentedWeyl k n →ₗ[k] SymbolRing k n :=
40   (MvPolynomial.weightedHomogeneousComponent w N).comp
41     (presentedNormalFormLinearEquiv k n).toLinearMap

55 theorem presentedPrincipalComponent_eq_zero_of_mem_of_lt {n L T : ℕ}
56     (w : PhaseVar n → ℕ) (z : PresentedWeyl k n)
57     (hz : z ∈ presentedWeightPiece k w L) (hLT : L < T) :
58     presentedPrincipalComponent k w T z = 0 := by
59   ext m
60   change AddMonoidAlgebra.coeff (presentedPrincipalComponent k w T z) m = 0
61   rw [coeff_presentedPrincipalComponent]
62   by_cases hm : monomialWeight w m = T
63   · rw [if_pos hm]
64     have hcoeff :
65         MvPolynomial.coeff m (presentedNormalFormLinearEquiv k n z) = 0 := by
66       by_contra hne
67       have hle := (mem_presentedWeightPiece k w L z).mp hz m hne
68       omega
69     change AddMonoidAlgebra.coeff (presentedNormalFormLinearEquiv k n z) m = 0 at hcoeff
70     rw [hcoeff]
71   · rw [if_neg hm]

138 theorem weightedHomogeneousComponent_monomial_mul {n T : ℕ}
139     (w : PhaseVar n → ℕ) (s : PhaseVar n →₀ ℕ)
140     (f : SymbolRing k n) :
141     MvPolynomial.weightedHomogeneousComponent w
142         (T + monomialWeight w s) (MvPolynomial.monomial s 1 * f) =
143       MvPolynomial.monomial s 1 *
144         MvPolynomial.weightedHomogeneousComponent w T f := by
145   classical
146   ext q
147   rw [MvPolynomial.coeff_weightedHomogeneousComponent,
148     MvPolynomial.coeff_monomial_mul',
149     MvPolynomial.coeff_monomial_mul']
150   by_cases hsq : s ≤ q
151   · rw [if_pos hsq, if_pos hsq, one_mul, one_mul,
152       MvPolynomial.coeff_weightedHomogeneousComponent]
153     have hsplit : s + (q - s) = q := by
154       rw [add_comm]
155       exact tsub_add_cancel_of_le hsq
156     have hweight :
157         monomialWeight w q =
158           monomialWeight w s + monomialWeight w (q - s) := by
159       rw [← hsplit]
160       simp [monomialWeight, Finsupp.sum_add_index, add_mul]
161     rw [finsupp_weight_eq_monomialWeight,
162       finsupp_weight_eq_monomialWeight]
163     rw [hweight]
164     by_cases hrest : monomialWeight w (q - s) = T
165     · simp [hrest, Nat.add_comm]
166     · have hne :
167           monomialWeight w s + monomialWeight w (q - s) ≠
168             T + monomialWeight w s := by
169         omega
170       simp [hrest, hne]
171   · rw [if_neg hsq, if_neg hsq]
172     simp

262 theorem phaseMonomial_succ_product (n : ℕ)
263     (a p c q : Fin (n + 1) → ℕ) :
264     MvPolynomial.monomial (phaseExponent a p) (1 : k) *
265         MvPolynomial.monomial (phaseExponent c q) 1 =
266       MvPolynomial.monomial
267           (extendPhaseExponent n (a 0 + c 0) (p 0 + q 0) 0) 1 *
268         MvPolynomial.rename oldIndex
269           (MvPolynomial.monomial
270               (phaseExponent (fun i => a i.succ) (fun i => p i.succ)) 1 *
271             MvPolynomial.monomial
272               (phaseExponent (fun i => c i.succ) (fun i => q i.succ)) 1) := by
273   have hexp :
274       phaseExponent a p + phaseExponent c q =
275         extendPhaseExponent n (a 0 + c 0) (p 0 + q 0) 0 +
276           (phaseExponent (fun i => a i.succ) (fun i => p i.succ) +
277             phaseExponent (fun i => c i.succ) (fun i => q i.succ)).mapDomain
278               oldIndex := by
279     rw [phaseExponent_succ_eq_extend, phaseExponent_succ_eq_extend]
280     simp [extendPhaseExponent, map_add, add_assoc, add_comm, add_left_comm]
281     rw [Finsupp.mapDomain_add]
282   rw [MvPolynomial.monomial_mul, MvPolynomial.monomial_mul,
283     MvPolynomial.rename_monomial, MvPolynomial.monomial_mul]
284   simp only [one_mul]
285   rw [hexp]

287 theorem presentedOrderedMonomial_mul_principal_bernstein :
288     ∀ (n : ℕ) (a p c q : Fin n → ℕ),
289       presentedPrincipalComponent k (@bernsteinWeight n)
290           (monomialWeight (@bernsteinWeight n) (phaseExponent a p) +
291             monomialWeight (@bernsteinWeight n) (phaseExponent c q))
292           (presentedOrderedMonomial k n a p *
293             presentedOrderedMonomial k n c q) =
294         MvPolynomial.monomial (phaseExponent a p) 1 *
295           MvPolynomial.monomial (phaseExponent c q) 1 := by
296   intro n
297   induction n with
298   | zero =>
299       intro a p c q
300       have hap : phaseExponent a p = 0 := by
301         ext i
302         exact Sum.elim Fin.elim0 Fin.elim0 i
303       have hcq : phaseExponent c q = 0 := by
304         ext i
305         exact Sum.elim Fin.elim0 Fin.elim0 i
306       rw [hap, hcq]
307       simp [presentedOrderedMonomial, monomialWeight,
308         presentedPrincipalComponent, presentedNormalFormLinearEquiv_one,
309         weightedHomogeneousComponent_monomial]
310   | succ n ih =>
311       intro a p c q
312       let L :=
313         monomialWeight (@bernsteinWeight n)
314             (phaseExponent (fun i => a i.succ) (fun i => p i.succ)) +
315           monomialWeight (@bernsteinWeight n)
316             (phaseExponent (fun i => c i.succ) (fun i => q i.succ))
317       have htarget :
318           monomialWeight (@bernsteinWeight (n + 1)) (phaseExponent a p) +
319               monomialWeight (@bernsteinWeight (n + 1)) (phaseExponent c q) =
320             L + (a 0 + c 0) + (p 0 + q 0) := by
321         rw [monomialWeight_phaseExponent_succ_bernstein,
322           monomialWeight_phaseExponent_succ_bernstein]
323         simp only [L]
324         omega
325       rw [htarget]
326       change
327         presentedPrincipalComponent k (@bernsteinWeight (n + 1))
328             (L + (a 0 + c 0) + (p 0 + q 0))
329             (presentedCoefficientOrdered k n (a 0) (p 0)
330                 (presentedOrderedMonomial k n
331                   (fun i => a i.succ) (fun i => p i.succ)) *
332               presentedCoefficientOrdered k n (c 0) (q 0)
333                 (presentedOrderedMonomial k n
334                   (fun i => c i.succ) (fun i => q i.succ))) = _
335       rw [presentedCoefficientOrdered_mul]
336       simp only [map_sum]
337       rw [Finset.sum_eq_single 0]
338       · simp only [Nat.zero_le, if_true, Nat.choose_zero_right,
339           Nat.descFactorial_zero, mul_one, one_nsmul, Nat.sub_zero]
340         rw [presentedPrincipalComponent_coefficientOrdered_bernstein]
341         have hih := ih
342           (fun i => a i.succ) (fun i => p i.succ)
343           (fun i => c i.succ) (fun i => q i.succ)
344         change presentedPrincipalComponent k (@bernsteinWeight n) L
345             (presentedOrderedMonomial k n
346                 (fun i => a i.succ) (fun i => p i.succ) *
347               presentedOrderedMonomial k n
348                 (fun i => c i.succ) (fun i => q i.succ)) = _ at hih
349         rw [hih]
350         exact (phaseMonomial_succ_product k n a p c q).symm
351       · intro i hi hi0
352         have hib : i ≤ p 0 := by
353           have := Finset.mem_range.mp hi
354           omega
355         by_cases hic : i ≤ c 0
356         · simp only [if_pos hic]
357           have hold :
358               presentedOrderedMonomial k n
359                     (fun j => a j.succ) (fun j => p j.succ) *
360                   presentedOrderedMonomial k n
361                     (fun j => c j.succ) (fun j => q j.succ) ∈
362                 bernsteinPiece k n L := by
363             simpa [L] using
364               presentedOrderedMonomial_mul_mem_bernsteinPiece k n
365                 (fun j => a j.succ) (fun j => p j.succ)
366                 (fun j => c j.succ) (fun j => q j.succ)
367           have hterm := presentedCoefficientOrdered_mem_bernsteinPiece k n L
368             (a 0 + c 0 - i) (p 0 + q 0 - i)
369             (presentedOrderedMonomial k n
370                 (fun j => a j.succ) (fun j => p j.succ) *
371               presentedOrderedMonomial k n
372                 (fun j => c j.succ) (fun j => q j.succ)) hold
373           have hlt :
374               L + (a 0 + c 0 - i) + (p 0 + q 0 - i) <
375                 L + (a 0 + c 0) + (p 0 + q 0) := by
376             omega
377           have hz := presentedPrincipalComponent_eq_zero_of_mem_of_lt k
378             (@bernsteinWeight (n + 1)) _ hterm hlt
379           rw [map_nsmul, hz, nsmul_zero]
380         · simp [hic]
381       · simp
