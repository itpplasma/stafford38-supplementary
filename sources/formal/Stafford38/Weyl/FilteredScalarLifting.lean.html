12 /-!
13 # Filtered scalar lifting for the presented Weyl algebra
14 
15 This file isolates the filtered linear-algebra step which is not supplied by
16 unfiltered PBW spanning.  The point is that an element of the intersection of
17 two scalar-extended subspaces has a bounded source representative.  The proof
18 uses the exact kernel statement for tensoring a quotient map; no target
19 initial generators and no strictness assertion are used.
20 
21 The small finite-dimensional example at the end records the cancellation
22 falsifier: an arbitrary high-degree representation need not be termwise
23 bounded, even though a bounded representation exists.
24 -/

83 /-!
84 Tensoring with a flat coefficient module preserves intersections of
85 submodules.  The proof is deliberately phrased through the quotient map of
86 `q`: the reverse inclusion is a kernel statement, not an appeal to an
87 unfiltered spanning equality.

89 theorem baseChange_inf_eq [SMulCommClass R S S]
90     [Module.Flat R S] (p q : Submodule R M) :
91     p.baseChange S ⊓ q.baseChange S = (p ⊓ q).baseChange S := by
92   apply le_antisymm
93   · intro z hz
94     let f : p →ₗ[R] M ⧸ q := q.mkQ.comp p.subtype
95     have hqzero :
96         TensorProduct.AlgebraTensorModule.lTensor S S q.mkQ z = 0 := by
97       have hzq := hz.2
98       rw [Submodule.baseChange_eq_span] at hzq
99       refine Submodule.span_induction ?_ ?_ ?_ ?_ hzq
100       · intro a ha
101         rcases ha with ⟨m, hm, rfl⟩
102         have hmker : m ∈ LinearMap.ker q.mkQ := by
103           rw [Submodule.ker_mkQ q]
104           exact hm
105         change (1 : S) ⊗ₜ[R] q.mkQ m = 0
106         rw [LinearMap.mem_ker.mp hmker]
107         simp
108       · simp
109       · intro a b _ _ ha hb
110         simpa only [map_add, ha, hb, add_zero] using rfl
111       · intro c a _ ha
112         simpa only [map_smul, ha, smul_zero] using rfl
113     have hpRange : z ∈ LinearMap.range (p.subtype.baseChange S) := by
114       rw [range_baseChange_subtype p]
115       exact hz.1
116     obtain ⟨y, hy⟩ := hpRange
117     change TensorProduct.AlgebraTensorModule.lTensor S S p.subtype y = z at hy
118     have hyker : y ∈ LinearMap.ker
119         (TensorProduct.AlgebraTensorModule.lTensor S S f) := by
120       apply LinearMap.mem_ker.mpr
121       calc
122         TensorProduct.AlgebraTensorModule.lTensor S S f y =
123             TensorProduct.AlgebraTensorModule.lTensor S S q.mkQ
124               (TensorProduct.AlgebraTensorModule.lTensor S S p.subtype y) := by
125                 rw [TensorProduct.AlgebraTensorModule.lTensor_comp]
126                 rfl
127         _ = TensorProduct.AlgebraTensorModule.lTensor S S q.mkQ z := by
128               rw [hy]
129         _ = 0 := hqzero
130     have hyspan : y ∈
131         LinearMap.range (TensorProduct.AlgebraTensorModule.lTensor S S
132           (LinearMap.ker f).subtype) := by
133       rw [← Module.Flat.ker_lTensor_eq (S := S) (M := S) f]
134       exact hyker
135     obtain ⟨v, hv⟩ := hyspan
136     change (LinearMap.ker f).subtype.baseChange S v = y at hv
137     let g : LinearMap.ker f →ₗ[R] ↥(p ⊓ q) :=
138       { toFun := fun u =>
139           ⟨(u : M), ⟨u.1.property, by
140             have hu : (u : M) ∈ LinearMap.ker q.mkQ := by
141               change q.mkQ (u : M) = 0
142               exact u.property
143             exact Eq.mp (congrArg (fun r : Submodule R M => (u : M) ∈ r)
144               (Submodule.ker_mkQ q)) hu⟩⟩
145         map_add' := by
146           intro u v
147           ext
148           simp
149         map_smul' := by
150           intro a u
151           ext
152           simp }
153     have hcomp :
154         p.subtype.comp (LinearMap.ker f).subtype =
155           (p ⊓ q).subtype.comp g := by
156       ext u
157       rfl
158     have hz' : z = (p ⊓ q).subtype.baseChange S
159         (g.baseChange S v) := by
160       calc
161         z = p.subtype.baseChange S y := hy.symm
162         _ = p.subtype.baseChange S
163             ((LinearMap.ker f).subtype.baseChange S v) := by rw [hv]
164         _ = (p.subtype.comp (LinearMap.ker f).subtype).baseChange S v := by
165               rw [LinearMap.baseChange_comp]
166               rfl
167         _ = ((p ⊓ q).subtype.comp g).baseChange S v := by rw [hcomp]
168         _ = (p ⊓ q).subtype.baseChange S (g.baseChange S v) := by
169               rw [LinearMap.baseChange_comp]
170               rfl
171     rw [hz']
172     rw [← range_baseChange_subtype (p ⊓ q)]
173     exact ⟨g.baseChange S v, rfl⟩
174   · intro z hz
175     exact ⟨baseChange_mono inf_le_left hz,
176       baseChange_mono inf_le_right hz⟩

195 theorem pbwTensorEquiv_tmul_one (n : Nat) (z : WeylAlg k n) :

480 /-!
481 An unfiltered representation cannot be truncated term by term.  Here the
482 low piece is the first coordinate axis, while the source span is all of
483 `ℚ × ℚ`.  The displayed representation of `(1, 0)` has two summands outside
484 the low piece whose second coordinates cancel.  The preceding flat
485 intersection theorem is the surviving filtration-on-span replacement.
486 -/
487 theorem cancellation_falsifier :
488     let F : Submodule ℚ (ℚ × ℚ) :=
