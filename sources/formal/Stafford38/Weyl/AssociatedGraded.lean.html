10 /-!
11 # Associated graded pieces of the PBW filtrations
12 
13 The strict lower piece in degree `N` is the kernel of degree-`N` principal
14 component projection. Restricting that projection to the filtered piece is
15 surjective onto the weighted-homogeneous symbol submodule, so each filtered
16 quotient is canonically linearly equivalent to its symbol component. The
17 external graded multiplication is a downstream construction.
18 -/

34 /-- The part of a filtration piece strictly below degree `N`; degree zero has
35 zero lower piece. -/
36 def presentedStrictLowerPiece {n : ℕ} (w : PhaseVar n → ℕ) :
37     (N : ℕ) → Submodule k (WeylAlg k n)
38   | 0 => ⊥
39   | N + 1 => presentedWeightPiece k w N
40 
41 theorem presentedStrictLowerPiece_le {n N : ℕ}
42     (w : PhaseVar n → ℕ) :
43     presentedStrictLowerPiece k w N ≤ presentedWeightPiece k w N := by
44   cases N with
45   | zero => exact bot_le
46   | succ N => exact presentedWeightPiece_mono k w (Nat.le_succ N)

48 theorem presentedPrincipalComponent_eq_zero_iff_mem_strictLower
49     {n N : ℕ} (w : PhaseVar n → ℕ) (z : WeylAlg k n)
50     (hz : z ∈ presentedWeightPiece k w N) :
51     presentedPrincipalComponent k w N z = 0 ↔
52       z ∈ presentedStrictLowerPiece k w N := by
53   constructor
54   · intro hp
55     cases N with
56     | zero =>
57         rw [presentedStrictLowerPiece, Submodule.mem_bot]
58         apply (presentedNormalFormLinearEquiv k n).injective
59         ext m
60         change AddMonoidAlgebra.coeff
61           ((presentedNormalFormLinearEquiv k n) z) m =
62             AddMonoidAlgebra.coeff ((presentedNormalFormLinearEquiv k n) 0) m
63         have hle := (mem_presentedWeightPiece k w 0 z).mp hz
64         change ∀ m, AddMonoidAlgebra.coeff
65           ((presentedNormalFormLinearEquiv k n) z) m ≠ 0 → Finsupp.weight w m ≤ 0 at hle
66         have hpc := congrArg (fun f : SymbolRing k n => AddMonoidAlgebra.coeff f m) hp
67         change AddMonoidAlgebra.coeff (presentedPrincipalComponent k w 0 z) m = 0 at hpc
68         rw [coeff_presentedPrincipalComponent] at hpc
69         by_cases hc :
70             AddMonoidAlgebra.coeff (presentedNormalFormLinearEquiv k n z) m = 0
71         · rw [hc]
72           simp
73         · have hw : Finsupp.weight w m = 0 := Nat.eq_zero_of_le_zero (hle m hc)
74           simpa [hw] using hpc
75     | succ N =>
76         rw [presentedStrictLowerPiece, mem_presentedWeightPiece]
77         intro m hm
78         change AddMonoidAlgebra.coeff (presentedNormalFormLinearEquiv k n z) m ≠ 0 at hm
79         have hle := (mem_presentedWeightPiece k w (N + 1) z).mp hz m hm
80         have hpc := congrArg (fun f : SymbolRing k n => AddMonoidAlgebra.coeff f m) hp
81         change AddMonoidAlgebra.coeff (presentedPrincipalComponent k w (N + 1) z) m = 0 at hpc
82         rw [coeff_presentedPrincipalComponent] at hpc
83         by_contra hnot
84         have heq : Finsupp.weight w m = N + 1 := by omega
85         simp [heq, hm] at hpc
86   · intro hlower
87     cases N with
88     | zero =>
89         rw [presentedStrictLowerPiece, Submodule.mem_bot] at hlower
90         subst z
91         exact map_zero _
92     | succ N =>
93         exact presentedPrincipalComponent_eq_zero_of_mem_of_lt k w z
94           hlower (Nat.lt_succ_self N)

96 /-- The principal component restricted to the degree-`N` filtration piece,
97 with codomain narrowed to the degree-`N` homogeneous symbol submodule. -/
98 def principalComponentOnPiece {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ) :
99     presentedWeightPiece k w N →ₗ[k]
100       MvPolynomial.weightedHomogeneousSubmodule k w N where
101   toFun z := ⟨presentedPrincipalComponent k w N z,
102     MvPolynomial.weightedHomogeneousComponent_mem w
103       (presentedNormalFormLinearEquiv k n z) N⟩
104   map_add' x y := by
105     ext
106     simp [presentedPrincipalComponent]
107   map_smul' c x := by
108     ext
109     simp [presentedPrincipalComponent]

111 theorem mem_ker_principalComponentOnPiece_iff {n N : ℕ}
112     (w : PhaseVar n → ℕ) (z : presentedWeightPiece k w N) :
113     z ∈ LinearMap.ker (principalComponentOnPiece k w N) ↔
114       (z : WeylAlg k n) ∈ presentedStrictLowerPiece k w N := by
115   rw [LinearMap.mem_ker]
116   constructor
117   · intro h
118     have hp : presentedPrincipalComponent k w N z = 0 :=
119       congrArg (fun f : MvPolynomial.weightedHomogeneousSubmodule k w N =>
120         (f : SymbolRing k n)) h
121     exact (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
122       k w z z.property).mp hp
123   · intro hlower
124     apply Subtype.ext
125     exact (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
126       k w z z.property).mpr hlower

128 /-- The kernel of principal-component projection is exactly the strict lower
129 filtration piece, viewed inside the degree-`N` piece. -/
130 theorem ker_principalComponentOnPiece_eq {n N : ℕ}
131     (w : PhaseVar n → ℕ) :
132     LinearMap.ker (principalComponentOnPiece k w N) =
133       (presentedStrictLowerPiece k w N).comap
134         (presentedWeightPiece k w N).subtype := by
135   ext z
136   exact mem_ker_principalComponentOnPiece_iff k w z

138 theorem principalComponentOnPiece_surjective {n N : ℕ}
139     (w : PhaseVar n → ℕ) :
140     Function.Surjective (principalComponentOnPiece k w N) := by
141   intro f
142   let z : WeylAlg k n :=
143     (presentedNormalFormLinearEquiv k n).symm (f : SymbolRing k n)
144   have hz : z ∈ presentedWeightPiece k w N := by
145     rw [mem_presentedWeightPiece]
146     intro m hm
147     have hnormal : presentedNormalFormLinearEquiv k n z = (f : SymbolRing k n) := by
148       exact (presentedNormalFormLinearEquiv k n).apply_symm_apply f
149     rw [hnormal] at hm
150     have hw := f.property hm
151     exact le_of_eq hw
152   refine ⟨⟨z, hz⟩, ?_⟩
153   apply Subtype.ext
154   change presentedPrincipalComponent k w N z = (f : SymbolRing k n)
155   rw [presentedPrincipalComponent, LinearMap.comp_apply]
156   change MvPolynomial.weightedHomogeneousComponent w N
157       (presentedNormalFormLinearEquiv k n z) = (f : SymbolRing k n)
158   rw [show presentedNormalFormLinearEquiv k n z = (f : SymbolRing k n) from
159     (presentedNormalFormLinearEquiv k n).apply_symm_apply f]
160   exact f.property.weightedHomogeneousComponent_same

170 /-- The degree-`N` associated graded piece, presented canonically as the
171 quotient by the kernel of principal-component projection. The preceding
172 kernel theorem identifies this kernel with the strict lower filtration. -/
173 abbrev presentedAssociatedGradedPiece {n : ℕ}
174     (w : PhaseVar n → ℕ) (N : ℕ) :=
175   (presentedWeightPiece k w N) ⧸
176     LinearMap.ker (principalComponentOnPiece k w N)

180     AddCommGroup (presentedAssociatedGradedPiece k w N) :=
181   @Submodule.Quotient.addCommGroup k (presentedWeightPiece k w N) _

183     (presentedWeightPiece k w N).module

194 /-- Every associated graded piece is linearly equivalent to the corresponding
195 weighted-homogeneous symbol submodule. -/
196 def presentedAssociatedGradedPieceEquiv {n : ℕ}
197     (w : PhaseVar n → ℕ) (N : ℕ) :
198     presentedAssociatedGradedPiece k w N ≃ₗ[k]
199       MvPolynomial.weightedHomogeneousSubmodule k w N :=
200   @LinearMap.quotKerEquivOfSurjective k
201     (presentedWeightPiece k w N)
202     (MvPolynomial.weightedHomogeneousSubmodule k w N) _
203     (presentedWeightPiece k w N).addCommGroup
204     (MvPolynomial.weightedHomogeneousSubmodule k w N).addCommGroup
205     (presentedWeightPiece k w N).module
206     (MvPolynomial.weightedHomogeneousSubmodule k w N).module
207     (principalComponentOnPiece k w N)
208     (principalComponentOnPiece_surjective k w)
