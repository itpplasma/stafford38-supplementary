9 # Universal property of the presented Weyl algebra
10 
11 This factors the quotient lift used by the existing symplectic maps: any
12 family satisfying the prescribed commutators induces a unique algebra map.
13 -/
14 
15 namespace Stafford38.WeylUniversal
16 
17 open Stafford
18 open AlgebraicAnalysis
19 
20 noncomputable section
21 
22 variable {k ι A : Type*} [Field k] [Ring A] [Algebra k A]
23 
24 def freeWeylGeneratorMap (z : ι → A) : FreeAlgebra k ι →ₐ[k] A :=
25   FreeAlgebra.lift k z
26 
27 theorem freeWeylGeneratorMap_respects (omega : Matrix ι ι k) (z : ι → A)
28     (hz : ∀ i j, AlgebraicAnalysis.ringCommutator (z i) (z j) = algebraMap k A (omega i j)) :
29   ∀ ⦃a b : FreeAlgebra k ι⦄,
30       freeWeylRelation omega a b →
31         freeWeylGeneratorMap z a = freeWeylGeneratorMap z b
32   | _, _, ⟨i, j, rfl, rfl⟩ => by
33       simp only [freeWeylGeneratorMap, map_sub, map_mul,
34         FreeAlgebra.lift_ι_apply, AlgHom.commutes]
35       exact hz i j
36 
37 /-- The algebra map induced by a family with the prescribed commutators. -/
38 def freeWeylLift (omega : Matrix ι ι k) (z : ι → A)
39     (hz : ∀ i j, AlgebraicAnalysis.ringCommutator (z i) (z j) = algebraMap k A (omega i j)) :
40     FreeWeyl k ι omega →ₐ[k] A :=
41   RingQuot.liftAlgHom k
42     ⟨freeWeylGeneratorMap z, freeWeylGeneratorMap_respects omega z hz⟩
43 
44 @[simp] theorem freeWeylLift_generator (omega : Matrix ι ι k) (z : ι → A)
45     (hz : ∀ i j, AlgebraicAnalysis.ringCommutator (z i) (z j) = algebraMap k A (omega i j)) (i : ι) :
46     freeWeylLift omega z hz (freeWeylGenerator omega i) = z i := by
47   simp [freeWeylLift, freeWeylGenerator, freeWeylGeneratorMap,
48     RingQuot.liftAlgHom_mkAlgHom_apply]
49 
50 /-- Algebra maps out of the presented Weyl algebra are determined by the
51 canonical generators. -/
52 theorem freeWeyl_algHom_ext (omega : Matrix ι ι k)
53     (f g : FreeWeyl k ι omega →ₐ[k] A)
54     (h : ∀ i, f (freeWeylGenerator omega i) =
55       g (freeWeylGenerator omega i)) : f = g := by
56   apply RingQuot.ringQuot_ext' k
57   apply FreeAlgebra.hom_ext
58   funext i
59   exact h i
60 
61 #print axioms freeWeylLift
62 #print axioms freeWeyl_algHom_ext
63 
64 end
65 end Stafford38.WeylUniversal
