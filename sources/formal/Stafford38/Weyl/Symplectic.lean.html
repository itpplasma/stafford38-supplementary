8 # General-rank linear symplectic changes of Weyl generators
9 
10 A matrix preserving the standard symplectic form defines an algebra
11 endomorphism of the presented Weyl algebra. Two form-preserving matrices that
12 are mutual inverses define a checked algebra equivalence. The inverse is part
13 of the input, which keeps this theorem independent of matrix inversion APIs
14 and makes the exact direction of generator substitution explicit.
15 -/

30 def standardSymplecticHpres {n : ℕ}
31     (M : Matrix (Fin n ⊕ Fin n) (Fin n ⊕ Fin n) k)
32     (hM : M * standardForm k n * Matrix.transpose M = standardForm k n) :
33     ∀ i j,
34       AlgebraicAnalysis.ringCommutator
35           (Stafford.linearCombination M
36             (freeWeylGenerator (standardForm k n)) i)
37           (Stafford.linearCombination M
38             (freeWeylGenerator (standardForm k n)) j) =
39         algebraMap k (FreeWeyl k (Fin n ⊕ Fin n) (standardForm k n))
40           (standardForm k n i j) := by
41   exact symplectic_linear_change_preserves_commutator M
42     (freeWeylGenerator (standardForm k n)) (standardForm k n)
43     (fun i j => freeWeylGenerator_commutator (standardForm k n) i j) hM

45 def standardSymplecticAlgHom {n : ℕ}
46     (M : Matrix (Fin n ⊕ Fin n) (Fin n ⊕ Fin n) k)
47     (hM : M * standardForm k n * Matrix.transpose M = standardForm k n) :
48     FreeWeyl k (Fin n ⊕ Fin n) (standardForm k n) →ₐ[k]
49       FreeWeyl k (Fin n ⊕ Fin n) (standardForm k n) :=
50   freeWeylSymplecticAlgHom M (standardForm k n)
51     (standardSymplecticHpres k M hM)

63 theorem standardSymplecticAlgHom_comp_eq_id {n : ℕ}
64     (M N : Matrix (Fin n ⊕ Fin n) (Fin n ⊕ Fin n) k)
65     (hM : M * standardForm k n * Matrix.transpose M = standardForm k n)
66     (hN : N * standardForm k n * Matrix.transpose N = standardForm k n)
67     (hNM : N * M = 1) :
68     (standardSymplecticAlgHom k M hM).comp
69         (standardSymplecticAlgHom k N hN) =
70       AlgHom.id k (FreeWeyl k (Fin n ⊕ Fin n) (standardForm k n)) := by
71   apply RingQuot.ringQuot_ext' k
72   apply FreeAlgebra.hom_ext
73   funext i
74   change standardSymplecticAlgHom k M hM
75       (standardSymplecticAlgHom k N hN
76         (freeWeylGenerator (standardForm k n) i)) =
77     freeWeylGenerator (standardForm k n) i
78   calc
79     _ = Stafford.linearCombination (N * M)
80         (freeWeylGenerator (standardForm k n)) i := by
81       simpa [standardSymplecticAlgHom] using
82         freeWeylSymplecticAlgHom_comp_generator M N (standardForm k n)
83           (standardSymplecticHpres k M hM)
84           (standardSymplecticHpres k N hN) i
85     _ = _ := by
86       rw [hNM]
87       exact freeWeylLinearCombination_one (standardForm k n) i

89 /-- A general-rank symplectic change with an explicitly certified inverse. -/
90 def standardSymplecticAlgEquivOfInverse {n : ℕ}
91     (M N : Matrix (Fin n ⊕ Fin n) (Fin n ⊕ Fin n) k)
92     (hM : M * standardForm k n * Matrix.transpose M = standardForm k n)
93     (hN : N * standardForm k n * Matrix.transpose N = standardForm k n)
94     (hMN : M * N = 1) (hNM : N * M = 1) :
95     FreeWeyl k (Fin n ⊕ Fin n) (standardForm k n) ≃ₐ[k]
96       FreeWeyl k (Fin n ⊕ Fin n) (standardForm k n) :=
97   AlgEquiv.mk
98     { toFun := standardSymplecticAlgHom k M hM
99       invFun := standardSymplecticAlgHom k N hN
100       left_inv := fun x => DFunLike.congr_fun
101         (standardSymplecticAlgHom_comp_eq_id k N M hN hM hMN) x
102       right_inv := fun x => DFunLike.congr_fun
103         (standardSymplecticAlgHom_comp_eq_id k M N hM hN hNM) x }
104     (standardSymplecticAlgHom k M hM).map_mul
105     (standardSymplecticAlgHom k M hM).map_add
106     (standardSymplecticAlgHom k M hM).commutes
