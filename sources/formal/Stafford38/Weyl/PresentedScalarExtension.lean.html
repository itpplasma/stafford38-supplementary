22 /-!
23 # Coefficient extension for the presented Weyl algebra
24 
25 This file records the concrete map needed before any characteristic-support
26 descent can be attempted.  The source is the quotient presentation over `k`
27 and the target is the same presentation over an extension field `K`.  The map
28 is defined by the quotient universal property; in particular, no PBW
29 identification or base-change theorem is used in its definition.
30 
31 The file also proves injectivity, PBW normal-form and filtration transport,
32 PBW monicity, canonical-right-ideal transport in the usable direction, and the
33 resulting source-to-target order-initial-ideal inclusion.  The reverse filtered
34 comparison is recorded as an explicit contract below and proved in
35 `FilteredScalarLifting.lean`; no base-change equality is treated as
36 definitional.
37 -/

389 /-- Every source order-initial generator maps to a target order-initial
390 generator.  This is the comparison needed for descent; it is intentionally a
391 one-way inclusion and makes no claim that the target initial ideal is the
392 extension of the source initial ideal. -/

394     (n N : Nat) (d : WeylAlg k (n + 1)) :
395     (orderInitialIdeal k
396         (canonicalRightIdeal (presentedCoordinate k n) d N)).map
397         (symbolScalarExtension (k := k) (K := K) (n + 1)).toRingHom ≤
398       orderInitialIdeal K
399         (canonicalRightIdeal (presentedCoordinate K n)
400           (presentedWeylScalarExtension (k := k) (K := K) (n + 1) d) N) := by
401   rw [Ideal.map_le_iff_le_comap]
402   apply Ideal.span_le.mpr
403   intro P hP
404   rcases hP with ⟨L, z, hz, hI, rfl⟩
405   apply Ideal.subset_span

437 theorem target_mem_scalarImageSpan (n : Nat) (z : WeylAlg K n) :
438     z ∈ scalarImageSpan (k := k) (K := K) n := by
439   let f := presentedNormalFormLinearEquiv K n z
440   have hreconstruct :
441       z = ∑ m ∈ f.support,
442         MvPolynomial.coeff m f • presentedPBWBasis K n m := by
443     apply (presentedNormalFormLinearEquiv K n).injective
444     simp only [map_sum, map_smul, presentedNormalFormLinearEquiv_basis]
445     change f = _
446     calc
447       f = ∑ m ∈ f.support,
448           MvPolynomial.monomial m (MvPolynomial.coeff m f) :=
449         MvPolynomial.as_sum f
450       _ = _ := by
451         apply Finset.sum_congr rfl
452         intro m hm
453         rw [MvPolynomial.smul_monomial]
454         simp
455   rw [hreconstruct]
456   apply Submodule.sum_mem
457   intro m hm
458   apply Submodule.smul_mem
459   apply Submodule.subset_span
460   exact ⟨presentedPBWBasis k n m,
461     presentedWeylScalarExtension_basis (k := k) (K := K) n m⟩

471 theorem scalarImageRightIdealSpan_mul_right (n : Nat)
472     (I : RightIdeal (WeylAlg k n))
473     {a : WeylAlg K n}
474     (ha : a ∈ scalarImageRightIdealSpan (k := k) (K := K) n I)
475     (b : WeylAlg K n) :
476     a * b ∈ scalarImageRightIdealSpan (k := k) (K := K) n I := by
477   let e := presentedWeylScalarExtension (k := k) (K := K) n
478   let S := scalarImageRightIdealSpan (k := k) (K := K) n I
479   have hbase : ∀ (z : WeylAlg k n), z ∈ I →
480       ∀ b : WeylAlg K n, e z * b ∈ S := by
481     intro z hz b
482     have hb : b ∈ scalarImageSpan (k := k) (K := K) n :=
483       target_mem_scalarImageSpan (k := k) (K := K) n b
484     refine Submodule.span_induction ?_ ?_ ?_ ?_ hb
485     · intro c hc
486       rcases hc with ⟨c, hc, rfl⟩
487       have hzc : z * c ∈ I := by
488         change MulOpposite.op c • z ∈ I
489         exact I.smul_mem _ hz
490       change e z * e c ∈ S
491       rw [← map_mul]
492       exact Submodule.subset_span ⟨z * c, hzc, rfl⟩
493     · simp [S, scalarImageRightIdealSpan]
494     · intro b₁ b₂ _ _ hb₁ hb₂
495       simpa [mul_add] using S.add_mem hb₁ hb₂
496     · intro c b _ hb
497       rw [Algebra.smul_def, ← mul_assoc,

558 /-- The exact filtered statement for support descent.  It asks that
559 every target order-initial generator be represented by the scalar extension
560 of source initial generators.  It is stronger than the unfiltered PBW span
561 proved above and is deliberately not identified with a definitional
562 base-change equality.  `FilteredScalarLifting.lean` proves this contract by
563 flat base change of the ideal/order-piece intersection. -/
564 def FilteredInitialLifting : Prop :=
565   ∀ (k : Type u) [Field k] [CharZero k] (n N : ℕ)
566     (d : WeylAlg k (n + 1)),
567     orderInitialIdeal (AlgebraicClosure k)
568         (canonicalRightIdeal (presentedCoordinate (AlgebraicClosure k) n)
569           (presentedWeylScalarExtension (k := k)
570             (K := AlgebraicClosure k) (n + 1) d) N) ≤
571       (orderInitialIdeal k
572         (canonicalRightIdeal (presentedCoordinate k n) d N)).map
573         (scalarPolynomialMap (k := k) (K := AlgebraicClosure k)
574           (PhaseVar (n + 1)))
