8 # Computing the canonical Bernstein degree
9

12 degree function or importing the terminal theorem assembly.

32 /-- Inserting an old free word and passing to the quotient agrees with the
33 development's rank-shift embedding of the old quotient. -/
34 theorem mkAlgHom_freeOldMap (n : ℕ)
35     (x : FreeAlgebra k (Stafford38Challenge.PhaseVar n)) :
36     RingQuot.mkAlgHom k
37         (Stafford38Challenge.relation (k := k) (n := n + 1)
38           (Matrix.J (Fin (n + 1)) k))
39         (Stafford38FixedSourceChallenge.freeOldMap k n x) =
40       previousWeylEmbedding k n
41         (RingQuot.mkAlgHom k
42           (Stafford38Challenge.relation (k := k) (n := n)
43             (Matrix.J (Fin n) k)) x) := by
44   have h :
45       (RingQuot.mkAlgHom k
46           (Stafford38Challenge.relation (k := k) (n := n + 1)
47             (Matrix.J (Fin (n + 1)) k))).comp
48           (Stafford38FixedSourceChallenge.freeOldMap k n) =
49         (previousWeylEmbedding k n).comp
50           (RingQuot.mkAlgHom k
51             (Stafford38Challenge.relation (k := k) (n := n)
52               (Matrix.J (Fin n) k))) := by
53     apply FreeAlgebra.hom_ext
54     funext i
55     simp only [Function.comp_apply, AlgHom.comp_apply,
56       Stafford38FixedSourceChallenge.freeOldMap, FreeAlgebra.lift_ι_apply]
57     change freeWeylGenerator (Matrix.J (Fin (n + 1)) k)
58         (Stafford38FixedSourceChallenge.oldIndex i) =
59       previousWeylEmbedding k n (freeWeylGenerator (Matrix.J (Fin n) k) i)
60     rw [previousWeylEmbedding_generator]
61     rfl
62   exact DFunLike.congr_fun h x

64 /-- The challenge ordered words are the development's ordered PBW words. -/
65 theorem orderedMonomial_eq :
66     ∀ (n : ℕ) (a p : Fin n → ℕ),
67       Stafford38FixedSourceChallenge.orderedMonomial k n a p =
68         presentedOrderedMonomial k n a p := by
69   intro n
70   induction n with
71   | zero =>
72       intro a p
73       simp only [Stafford38FixedSourceChallenge.orderedMonomial,
74         Stafford38FixedSourceChallenge.freeOrderedMonomial,
75         presentedOrderedMonomial, map_one]
76   | succ n ih =>
77       intro a p
78       have htail := ih (fun i => a i.succ) (fun i => p i.succ)
79       rw [Stafford38FixedSourceChallenge.orderedMonomial] at htail
80       simp only [Stafford38FixedSourceChallenge.orderedMonomial,
81         Stafford38FixedSourceChallenge.freeOrderedMonomial, map_mul, map_pow,
82         presentedOrderedMonomial]
83       rw [mkAlgHom_freeOldMap, htail]
84       rfl

86 /-- The Bernstein weight of a split exponent is the challenge's total phase
87 degree. -/
88 theorem monomialWeight_bernstein_phaseExponent {n : ℕ} (a p : Fin n → ℕ) :
89     Finsupp.weight (@bernsteinWeight n) (phaseExponent a p) =
90       Stafford38FixedSourceChallenge.phaseDegree a p := by

92   unfold bernsteinWeight Stafford38FixedSourceChallenge.phaseDegree
93   rw [Finsupp.sum_fintype _ _ (fun _ => by simp)]
94   simp [Fintype.sum_sum_type]

96 /-- The challenge filtration is the development's Bernstein filtration. -/
97 theorem bernsteinPiece_eq (n N : ℕ) :
98     Stafford38FixedSourceChallenge.bernsteinPiece k n N =
99       Stafford38.WeylFiltration.bernsteinPiece k n N := by
100   rw [Stafford38FixedSourceChallenge.bernsteinPiece,
101     Stafford38.WeylFiltration.bernsteinPiece, presentedWeightPiece_eq_span]
102   congr 1
103   ext z
104   simp only [presentedWeightBasisSet, Set.mem_ofPred_eq]
105   constructor
106   · rintro ⟨a, p, hdeg, rfl⟩
107     refine ⟨phaseExponent a p, ?_, ?_⟩
108     · rw [monomialWeight_bernstein_phaseExponent]
109       exact hdeg
110     · rw [presentedPBWBasis_apply, orderedMonomial_eq]
111       rfl
112   · rintro ⟨m, hdeg, rfl⟩
113     refine ⟨fun i => m (.inl i), fun i => m (.inr i), ?_, ?_⟩
114     · rw [← monomialWeight_bernstein_phaseExponent, phaseExponent_split]
115       exact hdeg
116     · rw [presentedPBWBasis_apply, orderedMonomial_eq]

118 /-- The challenge's intrinsic degree is the development's checked PBW
119 normal-form degree. -/
120 theorem bernsteinDegree_eq {n : ℕ} (d : WeylAlg k n) :
121     Stafford38FixedSourceChallenge.bernsteinDegree k d =
122       (MvPolynomial.weightedTotalDegree (@bernsteinWeight n)
123         (presentedNormalFormLinearEquiv k n d)) := by
124   have hmem' : d ∈ presentedWeightPiece k (@bernsteinWeight n)
125       ((MvPolynomial.weightedTotalDegree (@bernsteinWeight n)
126         (presentedNormalFormLinearEquiv k n d))) :=
127     (mem_presentedWeightPiece k _ _ d).mpr fun m hm =>
128       MvPolynomial.le_weightedTotalDegree _
129         (MvPolynomial.mem_support_iff.mpr hm)
130   have hmem : d ∈ Stafford38FixedSourceChallenge.bernsteinPiece k n
131       ((MvPolynomial.weightedTotalDegree (@bernsteinWeight n)
132         (presentedNormalFormLinearEquiv k n d))) := by
133     rw [bernsteinPiece_eq]
134     exact hmem'
135   have hle : ∀ N, d ∈ Stafford38FixedSourceChallenge.bernsteinPiece k n N →
136       (MvPolynomial.weightedTotalDegree (@bernsteinWeight n)
137         (presentedNormalFormLinearEquiv k n d)) ≤ N := by
138     intro N hN
139     rw [bernsteinPiece_eq] at hN
140     have hN' : d ∈ presentedWeightPiece k (@bernsteinWeight n) N := hN
141     rw [mem_presentedWeightPiece] at hN'
142     apply Finset.sup_le
143     intro m hm
144     exact hN' m (MvPolynomial.mem_support_iff.mp hm)
145   apply le_antisymm
146   · exact Nat.sInf_le hmem
147   · exact le_csInf ⟨_, hmem⟩ hle
