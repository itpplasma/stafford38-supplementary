10 /-!
11 # Bernstein-symbol compatibility for linear Weyl changes
12 
13 An algebra map whose generator images have Bernstein degree one preserves the
14 Bernstein filtration. Its action on every principal component is obtained by
15 commutatively substituting the degree-one principal symbols of those images.
16 For a symplectic linear Weyl map, this induced substitution is exactly the
17 phase-space linear action defined in `Characteristic.LinearAction`.
18 -/

187 theorem algHom_preserves_bernsteinPiece {n r N : ℕ}
188     (f : PresentedWeyl k n →ₐ[k] PresentedWeyl k r)
189     (hgen : ∀ i : PhaseVar n,
190       f (freeWeylGenerator (standardForm k n) i) ∈
191         bernsteinPiece k r 1)
192     {z : PresentedWeyl k n} (hz : z ∈ bernsteinPiece k n N) :
193   f z ∈ bernsteinPiece k r N := by
194   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hz
195   induction hz using Submodule.span_induction with
196   | mem b hb =>
197       obtain ⟨m, hm, rfl⟩ := hb
198       rw [presentedPBWBasis_apply]
199       have h := algHom_orderedMonomial_mem_bernstein_all k n f hgen
200         (fun i => m (.inl i)) (fun i => m (.inr i))
201       rw [phaseExponent_split] at h
202       exact presentedWeightPiece_mono k bernsteinWeight hm h
203   | zero =>
204       rw [map_zero]
205       exact (bernsteinPiece k r N).zero_mem
206   | add x y hx hy ihx ihy =>
207       rw [map_add]
208       exact (bernsteinPiece k r N).add_mem ihx ihy
209   | smul c x hx ih =>
210       rw [map_smul]
211       exact (bernsteinPiece k r N).smul_mem c ih

278 theorem algHom_orderedMonomial_principal_succ
279     {n r : ℕ}
280     (ih : ∀ (f : PresentedWeyl k n →ₐ[k] PresentedWeyl k r)
281       (_hgen : ∀ i : PhaseVar n,
282         f (freeWeylGenerator (standardForm k n) i) ∈
283           bernsteinPiece k r 1)
284       (g : PhaseVar n → SymbolRing k r),
285       (∀ i,
286         presentedPrincipalComponent k (@bernsteinWeight r) 1
287             (f (freeWeylGenerator (standardForm k n) i)) = g i) →
288       ∀ (a p : Fin n → ℕ),
289         presentedPrincipalComponent k (@bernsteinWeight r)
290             (monomialWeight (@bernsteinWeight n) (phaseExponent a p))
291             (f (presentedOrderedMonomial k n a p)) =
292           MvPolynomial.aeval g
293             (MvPolynomial.monomial (phaseExponent a p) (1 : k)))
294     (f : PresentedWeyl k (n + 1) →ₐ[k] PresentedWeyl k r)
295     (hgen : ∀ i : PhaseVar (n + 1),
296       f (freeWeylGenerator (standardForm k (n + 1)) i) ∈
297         bernsteinPiece k r 1)
298     (g : PhaseVar (n + 1) → SymbolRing k r)
299     (hprincipal : ∀ i,
300       presentedPrincipalComponent k (@bernsteinWeight r) 1
301           (f (freeWeylGenerator (standardForm k (n + 1)) i)) = g i)
302     (a p : Fin (n + 1) → ℕ) :
303     presentedPrincipalComponent k (@bernsteinWeight r)
304         (monomialWeight (@bernsteinWeight (n + 1)) (phaseExponent a p))
305         (f (presentedOrderedMonomial k (n + 1) a p)) =
306       MvPolynomial.aeval g
307         (MvPolynomial.monomial (phaseExponent a p) (1 : k)) := by
308   let fold : PresentedWeyl k n →ₐ[k] PresentedWeyl k r :=
309     f.comp (previousWeylEmbedding k n)
310   let gold : PhaseVar n → SymbolRing k r := fun i => g (oldIndex i)
311   have hfold : ∀ i : PhaseVar n,
312       fold (freeWeylGenerator (standardForm k n) i) ∈
313         bernsteinPiece k r 1 := by
314     intro i
315     simpa [fold, oldGenerator] using hgen (oldIndex i)
316   have hfoldprincipal : ∀ i,
317       presentedPrincipalComponent k (@bernsteinWeight r) 1
318           (fold (freeWeylGenerator (standardForm k n) i)) = gold i := by
319     intro i
320     simpa [fold, gold, oldGenerator] using hprincipal (oldIndex i)
321   let aold : Fin n → ℕ := fun i => a i.succ
322   let pold : Fin n → ℕ := fun i => p i.succ
323   let d := monomialWeight (@bernsteinWeight n) (phaseExponent aold pold)
324   have holdmem := algHom_orderedMonomial_mem_bernstein_all k n fold hfold aold pold
325   have holdprincipal := ih fold hfold gold hfoldprincipal aold pold
326   have hx := hgen (.inl 0)
327   have hp := hgen (.inr 0)
328   have hxa := pow_mem_bernsteinPiece k hx (a 0)
329   have hpp := pow_mem_bernsteinPiece k hp (p 0)
330   have hfirst := presentedPrincipalComponent_mul_bernstein k holdmem hxa
331   rw [pow_principal_bernstein k hx] at hfirst
332   have hfirstmem := mul_mem_bernsteinPiece k holdmem hxa
333   have hsecond := presentedPrincipalComponent_mul_bernstein k hfirstmem hpp
334   rw [pow_principal_bernstein k hp, hfirst, holdprincipal,
335     hprincipal (.inl 0), hprincipal (.inr 0)] at hsecond
336   have hf : f (presentedOrderedMonomial k (n + 1) a p) =
337       fold (presentedOrderedMonomial k n aold pold) *
338         f (presentedCoordinate k n) ^ a 0 *
339         f (presentedMomentum k n) ^ p 0 := by
340     rw [presentedOrderedMonomial, map_mul, map_mul, map_pow, map_pow]
341     rfl
342   rw [hf, monomialWeight_phaseExponent_succ_bernstein]
343   change presentedPrincipalComponent k (@bernsteinWeight r)
344       (d + a 0 + p 0)
345       (fold (presentedOrderedMonomial k n aold pold) *
346         f (presentedCoordinate k n) ^ a 0 *
347         f (presentedMomentum k n) ^ p 0) = _
348   rw [phaseMonomial_succ_decompose]
349   simp only [map_mul, map_pow, MvPolynomial.aeval_X]
350   rw [MvPolynomial.aeval_rename]
351   exact hsecond

372 theorem algHom_principal_compatibility {n r N : ℕ}
373     (f : PresentedWeyl k n →ₐ[k] PresentedWeyl k r)
374     (hgen : ∀ i : PhaseVar n,
375       f (freeWeylGenerator (standardForm k n) i) ∈
376         bernsteinPiece k r 1)
377     (g : PhaseVar n → SymbolRing k r)
378     (hprincipal : ∀ i,
379       presentedPrincipalComponent k (@bernsteinWeight r) 1
380           (f (freeWeylGenerator (standardForm k n) i)) = g i)
381     {z : PresentedWeyl k n} (hz : z ∈ bernsteinPiece k n N) :
382     presentedPrincipalComponent k (@bernsteinWeight r) N (f z) =
383       MvPolynomial.aeval g
384         (presentedPrincipalComponent k (@bernsteinWeight n) N z) := by
385   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hz
386   induction hz using Submodule.span_induction with
387   | mem b hb =>
388       obtain ⟨m, hm, rfl⟩ := hb
389       let d := monomialWeight (@bernsteinWeight n) m
390       have hmapmem : f (presentedPBWBasis k n m) ∈ bernsteinPiece k r d := by
391         rw [presentedPBWBasis_apply]
392         have h := algHom_orderedMonomial_mem_bernstein_all k n f hgen
393           (fun i => m (.inl i)) (fun i => m (.inr i))
394         rw [phaseExponent_split] at h
395         exact h
396       by_cases hd : d = N
397       · subst N
398         rw [presentedPrincipalComponent_basis, if_pos rfl,
399           presentedPBWBasis_apply]
400         have h := algHom_orderedMonomial_principal_all k n f hgen g hprincipal
401           (fun i => m (.inl i)) (fun i => m (.inr i))
402         rw [phaseExponent_split] at h
403         exact h
404       · have hdlt : d < N := lt_of_le_of_ne hm hd
405         rw [presentedPrincipalComponent_eq_zero_of_mem_of_lt k
406           (@bernsteinWeight r) (f (presentedPBWBasis k n m)) hmapmem hdlt,
407           presentedPrincipalComponent_basis, if_neg hd, map_zero]
408   | zero => simp
409   | add x y hx hy ihx ihy =>
410       simp only [map_add]
411       rw [ihx, ihy]
412   | smul c x hx ih =>
413       simp only [map_smul]
414       rw [ih]
