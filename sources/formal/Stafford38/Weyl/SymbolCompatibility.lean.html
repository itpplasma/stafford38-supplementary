14 /-!
15 # Bernstein-symbol compatibility for linear Weyl changes
16 
17 An algebra map whose generator images have Bernstein degree one preserves the
18 Bernstein filtration. Its action on every principal component is obtained by
19 commutatively substituting the degree-one principal symbols of those images.
20 For a symplectic linear Weyl map, this induced substitution is exactly the
21 phase-space linear action defined in `Characteristic.LinearAction`.
22 -/

191 theorem algHom_preserves_bernsteinPiece {n r N : ℕ}
192     (f : WeylAlg k n →ₐ[k] WeylAlg k r)
193     (hgen : ∀ i : PhaseVar n,
194       f (freeWeylGenerator (standardForm k n) i) ∈
195         bernsteinPiece k r 1)
196     {z : WeylAlg k n} (hz : z ∈ bernsteinPiece k n N) :
197   f z ∈ bernsteinPiece k r N := by
198   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hz
199   induction hz using Submodule.span_induction with
200   | mem b hb =>
201       obtain ⟨m, hm, rfl⟩ := hb
202       rw [presentedPBWBasis_apply]
203       have h := algHom_orderedMonomial_mem_bernstein_all k n f hgen
204         (fun i => m (.inl i)) (fun i => m (.inr i))
205       rw [phaseExponent_split] at h
206       exact presentedWeightPiece_mono k bernsteinWeight hm h
207   | zero =>
208       rw [map_zero]
209       exact (bernsteinPiece k r N).zero_mem
210   | add x y hx hy ihx ihy =>
211       rw [map_add]
212       exact (bernsteinPiece k r N).add_mem ihx ihy
213   | smul c x hx ih =>
214       rw [map_smul]
215       exact (bernsteinPiece k r N).smul_mem c ih

282 theorem algHom_orderedMonomial_principal_succ
283     {n r : ℕ}
284     (ih : ∀ (f : WeylAlg k n →ₐ[k] WeylAlg k r)
285       (_hgen : ∀ i : PhaseVar n,
286         f (freeWeylGenerator (standardForm k n) i) ∈
287           bernsteinPiece k r 1)
288       (g : PhaseVar n → SymbolRing k r),
289       (∀ i,
290         presentedPrincipalComponent k (@bernsteinWeight r) 1
291             (f (freeWeylGenerator (standardForm k n) i)) = g i) →
292       ∀ (a p : Fin n → ℕ),
293         presentedPrincipalComponent k (@bernsteinWeight r)
294             (Finsupp.weight (@bernsteinWeight n) (phaseExponent a p))
295             (f (presentedOrderedMonomial k n a p)) =
296           MvPolynomial.aeval g
297             (MvPolynomial.monomial (phaseExponent a p) (1 : k)))
298     (f : WeylAlg k (n + 1) →ₐ[k] WeylAlg k r)
299     (hgen : ∀ i : PhaseVar (n + 1),
300       f (freeWeylGenerator (standardForm k (n + 1)) i) ∈
301         bernsteinPiece k r 1)
302     (g : PhaseVar (n + 1) → SymbolRing k r)
303     (hprincipal : ∀ i,
304       presentedPrincipalComponent k (@bernsteinWeight r) 1
305           (f (freeWeylGenerator (standardForm k (n + 1)) i)) = g i)
306     (a p : Fin (n + 1) → ℕ) :
307     presentedPrincipalComponent k (@bernsteinWeight r)
308         (Finsupp.weight (@bernsteinWeight (n + 1)) (phaseExponent a p))
309         (f (presentedOrderedMonomial k (n + 1) a p)) =
310       MvPolynomial.aeval g
311         (MvPolynomial.monomial (phaseExponent a p) (1 : k)) := by
312   let fold : WeylAlg k n →ₐ[k] WeylAlg k r :=
313     f.comp (previousWeylEmbedding k n)
314   let gold : PhaseVar n → SymbolRing k r := fun i => g (Stafford38FixedSourceChallenge.oldIndex i)
315   have hfold : ∀ i : PhaseVar n,
316       fold (freeWeylGenerator (standardForm k n) i) ∈
317         bernsteinPiece k r 1 := by
318     intro i
319     simpa [fold, oldGenerator] using hgen (Stafford38FixedSourceChallenge.oldIndex i)
320   have hfoldprincipal : ∀ i,
321       presentedPrincipalComponent k (@bernsteinWeight r) 1
322           (fold (freeWeylGenerator (standardForm k n) i)) = gold i := by
323     intro i
324     simpa [fold, gold, oldGenerator] using hprincipal (Stafford38FixedSourceChallenge.oldIndex i)
325   let aold : Fin n → ℕ := fun i => a i.succ
326   let pold : Fin n → ℕ := fun i => p i.succ
327   let d := Finsupp.weight (@bernsteinWeight n) (phaseExponent aold pold)
328   have holdmem := algHom_orderedMonomial_mem_bernstein_all k n fold hfold aold pold
329   have holdprincipal := ih fold hfold gold hfoldprincipal aold pold
330   have hx := hgen (.inl 0)
331   have hp := hgen (.inr 0)
332   have hxa := pow_mem_bernsteinPiece k hx (a 0)
333   have hpp := pow_mem_bernsteinPiece k hp (p 0)
334   have hfirst := presentedPrincipalComponent_mul_bernstein k holdmem hxa
335   rw [pow_principal_bernstein k hx] at hfirst
336   have hfirstmem := mul_mem_bernsteinPiece k holdmem hxa
337   have hsecond := presentedPrincipalComponent_mul_bernstein k hfirstmem hpp
338   rw [pow_principal_bernstein k hp, hfirst, holdprincipal,
339     hprincipal (.inl 0), hprincipal (.inr 0)] at hsecond
340   have hf : f (presentedOrderedMonomial k (n + 1) a p) =
341       fold (presentedOrderedMonomial k n aold pold) *
342         f (presentedCoordinate k n) ^ a 0 *
343         f (presentedMomentum k n) ^ p 0 := by
344     rw [presentedOrderedMonomial, map_mul, map_mul, map_pow, map_pow]
345     rfl
346   rw [hf, monomialWeight_phaseExponent_succ_bernstein]
347   change presentedPrincipalComponent k (@bernsteinWeight r)
348       (d + a 0 + p 0)
349       (fold (presentedOrderedMonomial k n aold pold) *
350         f (presentedCoordinate k n) ^ a 0 *
351         f (presentedMomentum k n) ^ p 0) = _
352   rw [phaseMonomial_succ_decompose]
353   simp only [map_mul, map_pow, MvPolynomial.aeval_X]
354   rw [MvPolynomial.aeval_rename]
355   exact hsecond

376 theorem algHom_principal_compatibility {n r N : ℕ}
377     (f : WeylAlg k n →ₐ[k] WeylAlg k r)
378     (hgen : ∀ i : PhaseVar n,
379       f (freeWeylGenerator (standardForm k n) i) ∈
380         bernsteinPiece k r 1)
381     (g : PhaseVar n → SymbolRing k r)
382     (hprincipal : ∀ i,
383       presentedPrincipalComponent k (@bernsteinWeight r) 1
384           (f (freeWeylGenerator (standardForm k n) i)) = g i)
385     {z : WeylAlg k n} (hz : z ∈ bernsteinPiece k n N) :
386     presentedPrincipalComponent k (@bernsteinWeight r) N (f z) =
387       MvPolynomial.aeval g
388         (presentedPrincipalComponent k (@bernsteinWeight n) N z) := by
389   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hz
390   induction hz using Submodule.span_induction with
391   | mem b hb =>
392       obtain ⟨m, hm, rfl⟩ := hb
393       let d := Finsupp.weight (@bernsteinWeight n) m
394       have hmapmem : f (presentedPBWBasis k n m) ∈ bernsteinPiece k r d := by
395         rw [presentedPBWBasis_apply]
396         have h := algHom_orderedMonomial_mem_bernstein_all k n f hgen
397           (fun i => m (.inl i)) (fun i => m (.inr i))
398         rw [phaseExponent_split] at h
399         exact h
400       by_cases hd : d = N
401       · subst N
402         rw [presentedPrincipalComponent_basis, if_pos rfl,
403           presentedPBWBasis_apply]
404         have h := algHom_orderedMonomial_principal_all k n f hgen g hprincipal
405           (fun i => m (.inl i)) (fun i => m (.inr i))
406         rw [phaseExponent_split] at h
407         exact h
408       · have hdlt : d < N := lt_of_le_of_ne hm hd
409         rw [presentedPrincipalComponent_eq_zero_of_mem_of_lt k
410           (@bernsteinWeight r) (f (presentedPBWBasis k n m)) hmapmem hdlt,
411           presentedPrincipalComponent_basis, if_neg hd, map_zero]
412   | zero => simp
413   | add x y hx hy ihx ihy =>
414       simp only [map_add]
415       rw [ihx, ihy]
416   | smul c x hx ih =>
417       simp only [map_smul]
418       rw [ih]
