11 /-!
12 # The associated graded algebra of the presented Weyl algebra
13 
14 The degree-wise filtered quotients assemble into an external direct sum. Its
15 canonical symbol map is an algebra equivalence with the commutative symbol
16 polynomial ring. For the Bernstein and differential-order filtrations, the
17 transported multiplication is proved to be the multiplication induced by Weyl
18 products of filtered representatives.
19 -/

52 /-- The external direct sum of all filtered quotient pieces. This is a new
53 type so its algebra structure can be transported without changing the
54 canonical module structures on the individual quotient pieces. -/
55 def PresentedAssociatedGraded {n : ℕ} (w : PhaseVar n → ℕ) :=
56   DirectSum ℕ (fun N => presentedAssociatedGradedPiece k w N)
57 
58 /-- The direct sum of the piecewise principal-component equivalences, followed
59 by weighted-homogeneous recomposition. -/
60 def presentedAssociatedGradedRawLinearEquiv {n : ℕ} (w : PhaseVar n → ℕ) :
61     DirectSum ℕ (fun N => presentedAssociatedGradedPiece k w N) ≃ₗ[k]
62       SymbolRing k n := by
63   letI := MvPolynomial.weightedDecomposition k w
64   exact (DFinsupp.mapRange.linearEquiv fun N =>
65       presentedAssociatedGradedPieceEquiv k w N).trans
66     (DirectSum.decomposeLinearEquiv
67       (MvPolynomial.weightedHomogeneousSubmodule k w)).symm
68 
69 def presentedAssociatedGradedRawEquiv {n : ℕ} (w : PhaseVar n → ℕ) :
70     PresentedAssociatedGraded k w ≃ SymbolRing k n :=
71   (presentedAssociatedGradedRawLinearEquiv k w).toEquiv

73 /-- The commutative ring structure transported through the global symbol
74 equivalence. The representative multiplication theorems below identify it
75 with the multiplication induced by the two Weyl filtrations used downstream. -/
76 noncomputable instance presentedAssociatedGradedCommRing {n : ℕ}
77     (w : PhaseVar n → ℕ) : CommRing (PresentedAssociatedGraded k w) :=
78   Equiv.commRing (presentedAssociatedGradedRawEquiv k w)

80 noncomputable instance presentedAssociatedGradedAlgebra {n : ℕ}
81     (w : PhaseVar n → ℕ) : Algebra k (PresentedAssociatedGraded k w) :=
82   Equiv.algebra k (presentedAssociatedGradedRawEquiv k w)

84 /-- The associated graded algebra is the commutative symbol polynomial
85 algebra. -/
86 def presentedAssociatedGradedAlgEquiv {n : ℕ} (w : PhaseVar n → ℕ) :
87     PresentedAssociatedGraded k w ≃ₐ[k] SymbolRing k n :=
88   Equiv.algEquiv k (presentedAssociatedGradedRawEquiv k w)

90 /-- Insert one quotient piece into the external direct sum. -/
91 def presentedAssociatedGradedOf {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ)
92     (z : presentedAssociatedGradedPiece k w N) :
93     PresentedAssociatedGraded k w :=
94   DirectSum.of (fun N => presentedAssociatedGradedPiece k w N) N z

96 @[simp] theorem presentedAssociatedGradedAlgEquiv_of {n : ℕ}
97     (w : PhaseVar n → ℕ) (N : ℕ)
98     (z : presentedAssociatedGradedPiece k w N) :
99     presentedAssociatedGradedAlgEquiv k w
100         (presentedAssociatedGradedOf k w N z) =
101       (presentedAssociatedGradedPieceEquiv k w N z : SymbolRing k n) := by
102   letI := MvPolynomial.weightedDecomposition k w
103   change presentedAssociatedGradedRawLinearEquiv k w
104       (DirectSum.of (fun N => presentedAssociatedGradedPiece k w N) N z) = _
105   rw [presentedAssociatedGradedRawLinearEquiv, LinearEquiv.trans_apply]
106   have hmap :
107       (DFinsupp.mapRange.linearEquiv fun N =>
108           presentedAssociatedGradedPieceEquiv k w N)
109           (DirectSum.of (fun N => presentedAssociatedGradedPiece k w N) N z) =
110         DirectSum.of (fun N =>
111           MvPolynomial.weightedHomogeneousSubmodule k w N) N
112           (presentedAssociatedGradedPieceEquiv k w N z) := by
113     rw [DFinsupp.mapRange.linearEquiv_apply]
114     exact DFinsupp.mapRange_single
115   calc
116     _ = (DirectSum.decomposeLinearEquiv
117         (MvPolynomial.weightedHomogeneousSubmodule k w)).symm
118           (DirectSum.of (fun N =>
119             MvPolynomial.weightedHomogeneousSubmodule k w N) N
120             (presentedAssociatedGradedPieceEquiv k w N z)) :=
121       congrArg _ hmap
122     _ = _ := by
123       rw [DirectSum.decomposeLinearEquiv_symm_apply,
124         DirectSum.decompose_symm_of]

126 /-- Homogeneous insertion is additive even though the ring structure on the
127 new external-direct-sum type is transported from the symbol ring. -/
128 theorem presentedAssociatedGradedOf_add {n N : ℕ}
129     (w : PhaseVar n → ℕ)
130     (x y : presentedAssociatedGradedPiece k w N) :
131     presentedAssociatedGradedOf k w N (x + y) =
132       presentedAssociatedGradedOf k w N x +
133         presentedAssociatedGradedOf k w N y := by
134   apply (presentedAssociatedGradedAlgEquiv k w).injective
135   rw [map_add, presentedAssociatedGradedAlgEquiv_of,
136     presentedAssociatedGradedAlgEquiv_of,
137     presentedAssociatedGradedAlgEquiv_of]
138   exact congrArg Subtype.val
139     ((presentedAssociatedGradedPieceEquiv k w N).map_add x y)

168 /-- On Bernstein-homogeneous classes, external graded multiplication is
169 represented by multiplication in the Weyl algebra. -/
170 theorem presentedAssociatedGradedMk_mul_bernstein {n N M : ℕ}
171     (x : bernsteinPiece k n N) (y : bernsteinPiece k n M) :
172     presentedAssociatedGradedMk k bernsteinWeight x *
173         presentedAssociatedGradedMk k bernsteinWeight y =
174       presentedAssociatedGradedMk k bernsteinWeight
175         (bernsteinMulRepresentative k x y) := by
176   apply (presentedAssociatedGradedAlgEquiv k bernsteinWeight).injective
177   rw [map_mul,
178     presentedAssociatedGradedAlgEquiv_mk (n := n) k bernsteinWeight x,
179     presentedAssociatedGradedAlgEquiv_mk (n := n) k bernsteinWeight y,
180     presentedAssociatedGradedAlgEquiv_mk (n := n) k bernsteinWeight
181       (bernsteinMulRepresentative k x y)]
182   simpa [bernsteinMulRepresentative] using
183     (presentedPrincipalComponent_mul_bernstein k x.property y.property).symm
