10 /-!
11 # Weighted PBW truncations
12 
13 The ordered PBW basis gives intrinsic finite-degree subspaces of the presented
14 Weyl algebra.  This file constructs the truncations for an arbitrary weight on
15 the named generators, proves their coefficient characterization, monotonicity,
16 exhaustiveness, and basis-monomial membership, and specializes them to the
17 Bernstein and order weights.  Exact normal ordering then proves multiplicative
18 closure.  Leading symbols and the associated graded algebra remain downstream.
19 -/

33 /-- Extend an old exponent vector by exponents for the newest coordinate and
34 momentum. -/
35 def extendPhaseExponent (n a p : ℕ) (m : PhaseVar n →₀ ℕ) :
36     PhaseVar (n + 1) →₀ ℕ :=
37   Finsupp.single (.inr (0 : Fin (n + 1))) p +
38     Finsupp.single (.inl (0 : Fin (n + 1))) a +
39     Finsupp.mapDomain Stafford38FixedSourceChallenge.oldIndex m

41 theorem monomialWeight_mono {n : ℕ} {w₁ w₂ : PhaseVar n → ℕ}
42     (h : ∀ i, w₁ i ≤ w₂ i) (m : PhaseVar n →₀ ℕ) :
43     Finsupp.weight w₁ m ≤ Finsupp.weight w₂ m := by
44   apply Finsupp.sum_le_sum
45   intro i _
46   exact Nat.mul_le_mul_left _ (h i)

48 /-- Symbols supported on monomials of weighted degree at most `N`. -/
49 def symbolWeightPiece {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ) :
50     Submodule k (SymbolRing k n) where
51   carrier := {f | ∀ m, MvPolynomial.coeff m f ≠ 0 → Finsupp.weight w m ≤ N}
52   zero_mem' m hm := by simp at hm
53   add_mem' {f g} hf hg m hm := by
54     by_cases hfm : MvPolynomial.coeff m f = 0
55     · apply hg m
56       simpa [MvPolynomial.coeff_add, hfm] using hm
57     · exact hf m hfm
58   smul_mem' c f hf m hm := by
59     apply hf m
60     intro hfm
61     apply hm
62     simp [MvPolynomial.coeff_smul, hfm]

64 @[simp] theorem mem_symbolWeightPiece {n : ℕ} (w : PhaseVar n → ℕ)
65     (N : ℕ) (f : SymbolRing k n) :
66     f ∈ symbolWeightPiece k w N ↔
67       ∀ m, MvPolynomial.coeff m f ≠ 0 → Finsupp.weight w m ≤ N :=

70 theorem symbolWeightPiece_mono {n : ℕ} (w : PhaseVar n → ℕ)
71     {N M : ℕ} (hNM : N ≤ M) :
72     symbolWeightPiece k w N ≤ symbolWeightPiece k w M := by
73   intro f hf m hm

78     symbolWeightPiece k w₂ N ≤ symbolWeightPiece k w₁ N := by

82 @[simp] theorem monomial_mem_symbolWeightPiece_iff {n : ℕ}
83     (w : PhaseVar n → ℕ) (N : ℕ) (m : PhaseVar n →₀ ℕ) :
84     MvPolynomial.monomial m (1 : k) ∈ symbolWeightPiece k w N ↔
85       Finsupp.weight w m ≤ N := by
86   constructor

88     apply h m

96 /-- The sum of the weights of all supported monomials is a simple exhaustive
97 bound.  It is not intended to be the minimal weighted degree. -/
98 def symbolSupportWeightBound {n : ℕ} (w : PhaseVar n → ℕ)
99     (f : SymbolRing k n) : ℕ :=
100   f.support.sum (Finsupp.weight w)

102 theorem mem_symbolWeightPiece_supportBound {n : ℕ}
103     (w : PhaseVar n → ℕ) (f : SymbolRing k n) :
104     f ∈ symbolWeightPiece k w (symbolSupportWeightBound k w f) := by
105   intro m hm
106   apply Finset.single_le_sum (fun _ _ => Nat.zero_le _)
107   exact MvPolynomial.mem_support_iff.mpr hm

154 /-- The coefficient-support definition of a weighted truncation agrees with
155 the span of exactly the bounded ordered PBW basis vectors. -/
156 theorem presentedWeightPiece_eq_span {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ) :
157     presentedWeightPiece k w N =
158       Submodule.span k (presentedWeightBasisSet k w N) := by
159   apply le_antisymm
160   · intro a ha
161     let f := presentedNormalFormLinearEquiv k n a
162     have hreconstruct :
163         a = ∑ m ∈ f.support,
164           MvPolynomial.coeff m f • presentedPBWBasis k n m := by
165       apply (presentedNormalFormLinearEquiv k n).injective
166       simp only [map_sum, map_smul,
167         presentedNormalFormLinearEquiv_basis]
168       change f = _
169       calc
170         f = ∑ m ∈ f.support,
171             MvPolynomial.monomial m (MvPolynomial.coeff m f) :=
172           MvPolynomial.as_sum f
173         _ = _ := by
174           apply Finset.sum_congr rfl
175           intro m hm
176           rw [MvPolynomial.smul_monomial]
177           simp
178     rw [hreconstruct]
179     apply Submodule.sum_mem
180     intro m hm
181     apply Submodule.smul_mem
182     apply Submodule.subset_span
183     exact ⟨m, ha m (MvPolynomial.mem_support_iff.mp hm), rfl⟩
184   · apply Submodule.span_le.mpr
185     intro a ha
186     obtain ⟨m, hm, rfl⟩ := ha
187     exact (presentedPBWBasis_mem_weightPiece_iff k w N m).mpr hm

289 /-- Appending a normal-ordered newest pair increases Bernstein degree by at
290 most the sum of the two new exponents. -/
291 theorem presentedCoefficientOrdered_mem_bernsteinPiece
292     (n N a p : ℕ) (z : WeylAlg k n)
293     (hz : z ∈ bernsteinPiece k n N) :
294     presentedCoefficientOrdered k n a p z ∈
295       bernsteinPiece k (n + 1) (N + a + p) := by
296   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hz
297   induction hz using Submodule.span_induction with
298   | mem z hz =>
299       obtain ⟨m, hm, rfl⟩ := hz
300       rw [bernsteinPiece, mem_presentedWeightPiece,
301         presentedCoefficientOrdered_basis_normal]
302       intro q hq
303       simp only [MvPolynomial.coeff_monomial] at hq
304       split at hq
305       · next h =>
306           subst q
307           rw [← monomialWeight_extend_bernstein]
308           omega
309       · contradiction
310   | zero =>
311       change presentedCoefficientOrderedLinear k n a p 0 ∈ _
312       rw [map_zero]
313       exact Submodule.zero_mem _
314   | add x y hx hy ihx ihy =>
315       change presentedCoefficientOrderedLinear k n a p (x + y) ∈ _
316       rw [map_add]
317       exact Submodule.add_mem _ ihx ihy
318   | smul c x hx ih =>
319       change presentedCoefficientOrderedLinear k n a p (c • x) ∈ _
320       rw [map_smul]
321       exact Submodule.smul_mem _ c ih

323 /-- Appending a normal-ordered newest pair increases differential order by at
324 most the new momentum exponent; coordinate powers have order zero. -/
325 theorem presentedCoefficientOrdered_mem_orderPiece
326     (n N a p : ℕ) (z : WeylAlg k n)
327     (hz : z ∈ orderPiece k n N) :
328     presentedCoefficientOrdered k n a p z ∈
329       orderPiece k (n + 1) (N + p) := by
330   rw [orderPiece, presentedWeightPiece_eq_span] at hz
331   induction hz using Submodule.span_induction with
332   | mem z hz =>
333       obtain ⟨m, hm, rfl⟩ := hz
334       rw [orderPiece, mem_presentedWeightPiece,
335         presentedCoefficientOrdered_basis_normal]
336       intro q hq
337       simp only [MvPolynomial.coeff_monomial] at hq
338       split at hq
339       · next h =>
340           subst q
341           rw [← monomialWeight_extend_order]
342           omega
343       · contradiction
344   | zero =>
345       change presentedCoefficientOrderedLinear k n a p 0 ∈ _
346       rw [map_zero]
347       exact Submodule.zero_mem _
348   | add x y hx hy ihx ihy =>
349       change presentedCoefficientOrderedLinear k n a p (x + y) ∈ _
350       rw [map_add]
351       exact Submodule.add_mem _ ihx ihy
352   | smul c x hx ih =>
353       change presentedCoefficientOrderedLinear k n a p (c • x) ∈ _
354       rw [map_smul]
355       exact Submodule.smul_mem _ c ih

357 /-- Products of ordered PBW monomials satisfy the Bernstein degree bound. -/
358 theorem presentedOrderedMonomial_mul_mem_bernsteinPiece :
359     ∀ (n : ℕ) (a p c q : Fin n → ℕ),
360       presentedOrderedMonomial k n a p * presentedOrderedMonomial k n c q ∈
361         bernsteinPiece k n
362           (Finsupp.weight (@bernsteinWeight n) (phaseExponent a p) +
363             Finsupp.weight (@bernsteinWeight n) (phaseExponent c q)) := by
364   intro n
365   induction n with
366   | zero =>
367       intro a p c q
368       have hap : phaseExponent a p = 0 := by
369         ext i
370         exact Sum.elim Fin.elim0 Fin.elim0 i
371       have hcq : phaseExponent c q = 0 := by
372         ext i
373         exact Sum.elim Fin.elim0 Fin.elim0 i
374       rw [hap, hcq]
375       simp only [Finsupp.weight_apply, Finsupp.sum_zero_index, zero_add]
376       simp only [presentedOrderedMonomial, one_mul]
377       change (1 : WeylAlg k 0) ∈ bernsteinPiece k 0 0
378       rw [bernsteinPiece, mem_presentedWeightPiece]
379       intro m hm
380       have hone :
381           presentedNormalFormLinearEquiv k 0 (1 : WeylAlg k 0) = 1 :=
382         presentedNormalFormLinearEquiv_one k 0
383       rw [hone] at hm
384       have hm0 : m = 0 := by
385         symm
386         simpa [MvPolynomial.coeff_one] using hm
387       subst m
388       simp [Finsupp.weight_apply]
389   | succ n ih =>
390       intro a p c q
391       change
392         presentedCoefficientOrdered k n (a 0) (p 0)
393               (presentedOrderedMonomial k n
394                 (fun i => a i.succ) (fun i => p i.succ)) *
395             presentedCoefficientOrdered k n (c 0) (q 0)
396               (presentedOrderedMonomial k n
397                 (fun i => c i.succ) (fun i => q i.succ)) ∈ _
398       rw [presentedCoefficientOrdered_mul]
399       apply Submodule.sum_mem
400       intro i hi
401       have hib : i ≤ p 0 := by
402         have := Finset.mem_range.mp hi
403         omega
404       by_cases hic : i ≤ c 0
405       · simp only [if_pos hic]
406         apply Submodule.smul_mem
407         have hterm := presentedCoefficientOrdered_mem_bernsteinPiece k
408           n
409           (Finsupp.weight (@bernsteinWeight n)
410               (phaseExponent (fun i => a i.succ) (fun i => p i.succ)) +
411             Finsupp.weight (@bernsteinWeight n)
412               (phaseExponent (fun i => c i.succ) (fun i => q i.succ)))
413           (a 0 + c 0 - i) (p 0 + q 0 - i)
414           (presentedOrderedMonomial k n
415               (fun i => a i.succ) (fun i => p i.succ) *
416             presentedOrderedMonomial k n
417               (fun i => c i.succ) (fun i => q i.succ))
418           (ih (fun i => a i.succ) (fun i => p i.succ)
419             (fun i => c i.succ) (fun i => q i.succ))
420         apply presentedWeightPiece_mono k bernsteinWeight _ hterm
421         rw [monomialWeight_phaseExponent_succ_bernstein,
422           monomialWeight_phaseExponent_succ_bernstein]
423         omega
424       · simp [hic]

495 /-- The Bernstein PBW truncations are multiplicatively filtered. -/
496 theorem mul_mem_bernsteinPiece {n N M : ℕ} {x y : WeylAlg k n}
497     (hx : x ∈ bernsteinPiece k n N) (hy : y ∈ bernsteinPiece k n M) :
498     x * y ∈ bernsteinPiece k n (N + M) := by
499   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hx hy
500   apply Submodule.span_induction₂
501       (p := fun x y _ _ => x * y ∈ bernsteinPiece k n (N + M))
502       (ha := hx) (hb := hy)
503   · intro x y hx hy
504     obtain ⟨m, hm, rfl⟩ := hx
