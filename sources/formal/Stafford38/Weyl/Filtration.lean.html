7 /-!
8 # Weighted PBW truncations
9 
10 The ordered PBW basis gives intrinsic finite-degree subspaces of the presented
11 Weyl algebra.  This file constructs the truncations for an arbitrary weight on
12 the named generators, proves their coefficient characterization, monotonicity,
13 exhaustiveness, and basis-monomial membership, and specializes them to the
14 Bernstein and order weights.  Exact normal ordering then proves multiplicative
15 closure.  Leading symbols and the associated graded algebra remain downstream.
16 -/

30 /-- Weighted degree of a commutative phase-space monomial. -/
31 def monomialWeight {n : ℕ} (w : PhaseVar n → ℕ)
32     (m : PhaseVar n →₀ ℕ) : ℕ :=
33   m.sum fun i e => e * w i
34 
35 /-- Extend an old exponent vector by exponents for the newest coordinate and
36 momentum. -/
37 def extendPhaseExponent (n a p : ℕ) (m : PhaseVar n →₀ ℕ) :
38     PhaseVar (n + 1) →₀ ℕ :=
39   Finsupp.single (.inr (0 : Fin (n + 1))) p +
40     Finsupp.single (.inl (0 : Fin (n + 1))) a +
41     Finsupp.mapDomain oldIndex m

43 theorem monomialWeight_mono {n : ℕ} {w₁ w₂ : PhaseVar n → ℕ}
44     (h : ∀ i, w₁ i ≤ w₂ i) (m : PhaseVar n →₀ ℕ) :
45     monomialWeight w₁ m ≤ monomialWeight w₂ m := by
46   apply Finsupp.sum_le_sum
47   intro i _
48   exact Nat.mul_le_mul_left _ (h i)

50 /-- Symbols supported on monomials of weighted degree at most `N`. -/
51 def symbolWeightPiece {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ) :
52     Submodule k (SymbolRing k n) where
53   carrier := {f | ∀ m, MvPolynomial.coeff m f ≠ 0 → monomialWeight w m ≤ N}
54   zero_mem' m hm := by simp at hm
55   add_mem' {f g} hf hg m hm := by
56     by_cases hfm : MvPolynomial.coeff m f = 0
57     · apply hg m
58       simpa [MvPolynomial.coeff_add, hfm] using hm
59     · exact hf m hfm
60   smul_mem' c f hf m hm := by
61     apply hf m
62     intro hfm
63     apply hm
64     simp [MvPolynomial.coeff_smul, hfm]

66 @[simp] theorem mem_symbolWeightPiece {n : ℕ} (w : PhaseVar n → ℕ)
67     (N : ℕ) (f : SymbolRing k n) :
68     f ∈ symbolWeightPiece k w N ↔
69       ∀ m, MvPolynomial.coeff m f ≠ 0 → monomialWeight w m ≤ N :=
70   Iff.rfl

72 theorem symbolWeightPiece_mono {n : ℕ} (w : PhaseVar n → ℕ)
73     {N M : ℕ} (hNM : N ≤ M) :
74     symbolWeightPiece k w N ≤ symbolWeightPiece k w M := by
75   intro f hf m hm
76   exact (hf m hm).trans hNM

78 theorem symbolWeightPiece_antitone_weight {n : ℕ}
79     {w₁ w₂ : PhaseVar n → ℕ} (h : ∀ i, w₁ i ≤ w₂ i) (N : ℕ) :
80     symbolWeightPiece k w₂ N ≤ symbolWeightPiece k w₁ N := by
81   intro f hf m hm
82   exact (monomialWeight_mono h m).trans (hf m hm)

84 @[simp] theorem monomial_mem_symbolWeightPiece_iff {n : ℕ}
85     (w : PhaseVar n → ℕ) (N : ℕ) (m : PhaseVar n →₀ ℕ) :
86     MvPolynomial.monomial m (1 : k) ∈ symbolWeightPiece k w N ↔
87       monomialWeight w m ≤ N := by
88   constructor
89   · intro h
90     apply h m
91     simp [MvPolynomial.coeff_monomial]
92   · intro hm q hq
93     simp only [MvPolynomial.coeff_monomial] at hq
94     split at hq
95     · next h => simpa [h] using hm
96     · contradiction

98 /-- The sum of the weights of all supported monomials is a simple exhaustive
99 bound.  It is not intended to be the minimal weighted degree. -/
100 def symbolSupportWeightBound {n : ℕ} (w : PhaseVar n → ℕ)
101     (f : SymbolRing k n) : ℕ :=
102   f.support.sum (monomialWeight w)

104 theorem mem_symbolWeightPiece_supportBound {n : ℕ}
105     (w : PhaseVar n → ℕ) (f : SymbolRing k n) :
106     f ∈ symbolWeightPiece k w (symbolSupportWeightBound k w f) := by
107   intro m hm
108   apply Finset.single_le_sum (fun _ _ => Nat.zero_le _)
109   exact MvPolynomial.mem_support_iff.mpr hm

156 /-- The coefficient-support definition of a weighted truncation agrees with
157 the span of exactly the bounded ordered PBW basis vectors. -/
158 theorem presentedWeightPiece_eq_span {n : ℕ} (w : PhaseVar n → ℕ) (N : ℕ) :
159     presentedWeightPiece k w N =
160       Submodule.span k (presentedWeightBasisSet k w N) := by
161   apply le_antisymm
162   · intro a ha
163     let f := presentedNormalFormLinearEquiv k n a
164     have hreconstruct :
165         a = ∑ m ∈ f.support,
166           MvPolynomial.coeff m f • presentedPBWBasis k n m := by
167       apply (presentedNormalFormLinearEquiv k n).injective
168       simp only [map_sum, map_smul,
169         presentedNormalFormLinearEquiv_basis]
170       change f = _
171       calc
172         f = ∑ m ∈ f.support,
173             MvPolynomial.monomial m (MvPolynomial.coeff m f) :=
174           MvPolynomial.as_sum f
175         _ = _ := by
176           apply Finset.sum_congr rfl
177           intro m hm
178           rw [MvPolynomial.smul_monomial]
179           simp
180     rw [hreconstruct]
181     apply Submodule.sum_mem
182     intro m hm
183     apply Submodule.smul_mem
184     apply Submodule.subset_span
185     exact ⟨m, ha m (MvPolynomial.mem_support_iff.mp hm), rfl⟩
186   · apply Submodule.span_le.mpr
187     intro a ha
188     obtain ⟨m, hm, rfl⟩ := ha
189     exact (presentedPBWBasis_mem_weightPiece_iff k w N m).mpr hm

291 /-- Appending a normal-ordered newest pair increases Bernstein degree by at
292 most the sum of the two new exponents. -/
293 theorem presentedCoefficientOrdered_mem_bernsteinPiece
294     (n N a p : ℕ) (z : PresentedWeyl k n)
295     (hz : z ∈ bernsteinPiece k n N) :
296     presentedCoefficientOrdered k n a p z ∈
297       bernsteinPiece k (n + 1) (N + a + p) := by
298   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hz
299   induction hz using Submodule.span_induction with
300   | mem z hz =>
301       obtain ⟨m, hm, rfl⟩ := hz
302       rw [bernsteinPiece, mem_presentedWeightPiece,
303         presentedCoefficientOrdered_basis_normal]
304       intro q hq
305       simp only [MvPolynomial.coeff_monomial] at hq
306       split at hq
307       · next h =>
308           subst q
309           rw [← monomialWeight_extend_bernstein]
310           omega
311       · contradiction
312   | zero =>
313       change presentedCoefficientOrderedLinear k n a p 0 ∈ _
314       rw [map_zero]
315       exact Submodule.zero_mem _
316   | add x y hx hy ihx ihy =>
317       change presentedCoefficientOrderedLinear k n a p (x + y) ∈ _
318       rw [map_add]
319       exact Submodule.add_mem _ ihx ihy
320   | smul c x hx ih =>
321       change presentedCoefficientOrderedLinear k n a p (c • x) ∈ _
322       rw [map_smul]
323       exact Submodule.smul_mem _ c ih

342           subst q

345       · contradiction
346   | zero =>

359 /-- Products of ordered PBW monomials satisfy the Bernstein degree bound. -/
360 theorem presentedOrderedMonomial_mul_mem_bernsteinPiece :
361     ∀ (n : ℕ) (a p c q : Fin n → ℕ),
362       presentedOrderedMonomial k n a p * presentedOrderedMonomial k n c q ∈
363         bernsteinPiece k n
364           (monomialWeight (@bernsteinWeight n) (phaseExponent a p) +
365             monomialWeight (@bernsteinWeight n) (phaseExponent c q)) := by
366   intro n
367   induction n with
368   | zero =>
369       intro a p c q
370       have hap : phaseExponent a p = 0 := by
371         ext i
372         exact Sum.elim Fin.elim0 Fin.elim0 i
373       have hcq : phaseExponent c q = 0 := by
374         ext i
375         exact Sum.elim Fin.elim0 Fin.elim0 i
376       rw [hap, hcq]
377       simp only [monomialWeight, Finsupp.sum_zero_index, zero_add]
378       simp only [presentedOrderedMonomial, one_mul]
379       change (1 : PresentedWeyl k 0) ∈ bernsteinPiece k 0 0
380       rw [bernsteinPiece, mem_presentedWeightPiece]
381       intro m hm
382       have hone :
383           presentedNormalFormLinearEquiv k 0 (1 : PresentedWeyl k 0) = 1 :=
384         presentedNormalFormLinearEquiv_one k 0
385       rw [hone] at hm
386       have hm0 : m = 0 := by
387         symm
388         simpa [MvPolynomial.coeff_one] using hm
389       subst m
390       simp [monomialWeight]
391   | succ n ih =>
392       intro a p c q
393       change
394         presentedCoefficientOrdered k n (a 0) (p 0)
395               (presentedOrderedMonomial k n
396                 (fun i => a i.succ) (fun i => p i.succ)) *
397             presentedCoefficientOrdered k n (c 0) (q 0)
398               (presentedOrderedMonomial k n
399                 (fun i => c i.succ) (fun i => q i.succ)) ∈ _
400       rw [presentedCoefficientOrdered_mul]
401       apply Submodule.sum_mem
402       intro i hi
403       have hib : i ≤ p 0 := by
404         have := Finset.mem_range.mp hi
405         omega
406       by_cases hic : i ≤ c 0
407       · simp only [if_pos hic]
408         apply Submodule.smul_mem
409         have hterm := presentedCoefficientOrdered_mem_bernsteinPiece k
410           n
411           (monomialWeight (@bernsteinWeight n)
412               (phaseExponent (fun i => a i.succ) (fun i => p i.succ)) +
413             monomialWeight (@bernsteinWeight n)
414               (phaseExponent (fun i => c i.succ) (fun i => q i.succ)))
415           (a 0 + c 0 - i) (p 0 + q 0 - i)
416           (presentedOrderedMonomial k n
417               (fun i => a i.succ) (fun i => p i.succ) *
418             presentedOrderedMonomial k n
419               (fun i => c i.succ) (fun i => q i.succ))
420           (ih (fun i => a i.succ) (fun i => p i.succ)
421             (fun i => c i.succ) (fun i => q i.succ))
422         apply presentedWeightPiece_mono k bernsteinWeight _ hterm
423         rw [monomialWeight_phaseExponent_succ_bernstein,
424           monomialWeight_phaseExponent_succ_bernstein]
425         omega
426       · simp [hic]

497 /-- The Bernstein PBW truncations are multiplicatively filtered. -/
498 theorem mul_mem_bernsteinPiece {n N M : ℕ} {x y : PresentedWeyl k n}
499     (hx : x ∈ bernsteinPiece k n N) (hy : y ∈ bernsteinPiece k n M) :
500     x * y ∈ bernsteinPiece k n (N + M) := by
501   rw [bernsteinPiece, presentedWeightPiece_eq_span] at hx hy
502   apply Submodule.span_induction₂
503       (p := fun x y _ _ => x * y ∈ bernsteinPiece k n (N + M))
504       (ha := hx) (hb := hy)
505   · intro x y hx hy
506     obtain ⟨m, hm, rfl⟩ := hx
507     obtain ⟨r, hr, rfl⟩ := hy
508     rw [presentedPBWBasis_apply, presentedPBWBasis_apply]
