9 # Coordinate commutators on arbitrary PBW sums

12 presented Weyl elements.  The proof packages both sides as linear maps and

31 /-- Commutation with the newest coordinate, followed by extraction of one
32 order-homogeneous component, as a linear map. -/
33 def coordinateCommutatorPrincipalLinear (n T : ℕ) :
34     WeylAlg k (n + 1) →ₗ[k] SymbolRing k (n + 1) where
35   toFun w :=
36     presentedPrincipalComponent k (@orderWeight (n + 1)) T
37       (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) w)
38   map_add' a b := by
39     change presentedPrincipalComponent k orderWeight T
40         (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) (a + b)) =
41       presentedPrincipalComponent k orderWeight T
42           (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) a) +
43         presentedPrincipalComponent k orderWeight T
44           (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) b)
45     rw [show AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) (a + b) =
46         AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) a +
47           AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) b by
48       simp only [AlgebraicAnalysis.ringCommutator, AlgebraicAnalysis.ringCommutator,
49         mul_add, add_mul]
50       abel]
51     exact LinearMap.map_add _ _ _
52   map_smul' c a := by
53     change presentedPrincipalComponent k orderWeight T
54         (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) (c • a)) =
55       c • presentedPrincipalComponent k orderWeight T
56         (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) a)
57     rw [show AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) (c • a) =
58         c • AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) a by
59       simp only [AlgebraicAnalysis.ringCommutator, AlgebraicAnalysis.ringCommutator,
60         Algebra.mul_smul_comm,
61         Algebra.smul_mul_assoc, smul_sub]]
62     exact LinearMap.map_smul _ _ _

64 /-- The negative Poisson derivative of the next order-homogeneous component,
65 as a linear map. -/
66 def coordinatePoissonPrincipalLinear (n T : ℕ) :
67     WeylAlg k (n + 1) →ₗ[k] SymbolRing k (n + 1) :=
68   (-((MvPolynomial.pderiv (R := k)
69       (Sum.inr (0 : Fin (n + 1)) : PhaseVar (n + 1))).toLinearMap)).comp
70     (presentedPrincipalComponent k (@orderWeight (n + 1)) (T + 1))

72 /-- A PBW basis vector in `n+1` pairs is an ordered newest-pair block whose
73 coefficient is the corresponding PBW basis vector in `n` pairs. -/
74 theorem presentedPBWBasis_succ_eq_coefficientOrdered
75     (n : ℕ) (m : PhaseVar (n + 1) →₀ ℕ) :
76     presentedPBWBasis k (n + 1) m =
77       presentedCoefficientOrdered k n
78         (m (.inl (0 : Fin (n + 1))))
79         (m (.inr (0 : Fin (n + 1))))
80         (presentedPBWBasis k n
81           (phaseExponent
82             (fun i => m (.inl i.succ))
83             (fun i => m (.inr i.succ)))) := by
84   rw [presentedPBWBasis_apply, presentedOrderedMonomial,
85     presentedPBWBasis_apply]
86   rfl

88 theorem extendPhaseExponent_sub_newestMomentum_general
89     (n a p : ℕ) (m : PhaseVar n →₀ ℕ) :
90     extendPhaseExponent n a p m -
91         Finsupp.single (.inr (0 : Fin (n + 1))) 1 =
92       extendPhaseExponent n a (p - 1) m := by
93   classical
94   have hnew : (Finsupp.mapDomain Stafford38FixedSourceChallenge.oldIndex m)
95       (.inr (0 : Fin (n + 1))) = 0 := by
96     apply Finsupp.mapDomain_notin_range
97     rintro ⟨i, hi⟩
98     cases i <;> simp [Stafford38FixedSourceChallenge.oldIndex, Fin.succ_ne_zero] at hi
99   ext i
100   rcases i with i | i
101   · by_cases hi : i = 0
102     · subst i
103       simp [extendPhaseExponent, Stafford38FixedSourceChallenge.oldIndex,
104         add_comm, add_left_comm, add_assoc]
105     · simp [extendPhaseExponent, Stafford38FixedSourceChallenge.oldIndex, hi,
106         add_comm, add_left_comm, add_assoc]
107   · by_cases hi : i = 0
108     · subst i
109       simp [extendPhaseExponent, Stafford38FixedSourceChallenge.oldIndex, hnew]
110     · have h0i : (0 : Fin (n + 1)) ≠ i := Ne.symm hi
111       simp [extendPhaseExponent, Stafford38FixedSourceChallenge.oldIndex, hi, h0i,
112         add_comm, add_left_comm, add_assoc]

124 theorem coordinate_symbol_linear_maps_eq (n T : ℕ) :
125     coordinateCommutatorPrincipalLinear k n T =
126       coordinatePoissonPrincipalLinear k n T := by
127   apply Module.Basis.ext (presentedPBWBasis k (n + 1))
128   intro m
129   let a := m (.inl (0 : Fin (n + 1)))
130   let p := m (.inr (0 : Fin (n + 1)))
131   let mold : PhaseVar n →₀ ℕ :=
132     phaseExponent
133       (fun i => m (.inl i.succ))
134       (fun i => m (.inr i.succ))
135   rw [presentedPBWBasis_succ_eq_coefficientOrdered k n m]
136   change presentedPrincipalComponent k orderWeight T
137       (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n)
138         (presentedCoefficientOrdered k n a p
139           (presentedPBWBasis k n mold))) =
140     -MvPolynomial.pderiv (.inr (0 : Fin (n + 1)))
141       (presentedPrincipalComponent k orderWeight (T + 1)
142         (presentedCoefficientOrdered k n a p
143           (presentedPBWBasis k n mold)))
144   rw [presentedCoordinate_commutator_coefficientOrdered]
145   simp only [map_neg, map_smul]
146   change -((p : k) •
147       MvPolynomial.weightedHomogeneousComponent orderWeight T
148         (presentedNormalFormLinearEquiv k (n + 1)
149           (presentedCoefficientOrdered k n a (p - 1)
150             (presentedPBWBasis k n mold)))) =
151     -MvPolynomial.pderiv (.inr (0 : Fin (n + 1)))
152       (MvPolynomial.weightedHomogeneousComponent orderWeight (T + 1)
153         (presentedNormalFormLinearEquiv k (n + 1)
154           (presentedCoefficientOrdered k n a p
155             (presentedPBWBasis k n mold))))
156   rw [show presentedNormalFormLinearEquiv k (n + 1)
157         (presentedCoefficientOrdered k n a (p - 1)
158           (presentedPBWBasis k n mold)) =
159         MvPolynomial.monomial
160           (extendPhaseExponent n a (p - 1) mold) 1 from
161       presentedCoefficientOrdered_basis_normal k n a (p - 1) mold,
162     weightedHomogeneousComponent_monomial,
163     show presentedNormalFormLinearEquiv k (n + 1)
164         (presentedCoefficientOrdered k n a p
165           (presentedPBWBasis k n mold)) =
166         MvPolynomial.monomial (extendPhaseExponent n a p mold) 1 from
167       presentedCoefficientOrdered_basis_normal k n a p mold,
168     weightedHomogeneousComponent_monomial]
169   let q := extendPhaseExponent n a p mold
170   let q' := extendPhaseExponent n a (p - 1) mold
171   change -((p : k) •
172       (if Finsupp.weight orderWeight q' = T then
173         MvPolynomial.monomial q' 1 else 0)) =
174     -MvPolynomial.pderiv (.inr (0 : Fin (n + 1)))
175       (if Finsupp.weight orderWeight q = T + 1 then
176         MvPolynomial.monomial q 1 else 0)
177   by_cases hp : p = 0
178   · simp [hp, q, q', MvPolynomial.pderiv_monomial,
179       extendPhaseExponent_newestMomentum_apply]
180     split <;> simp [MvPolynomial.pderiv_monomial,
181       extendPhaseExponent_newestMomentum_apply]
182   · have hp0 : 0 < p := Nat.pos_of_ne_zero hp
183     have hweight :
184         Finsupp.weight orderWeight q =
185           Finsupp.weight orderWeight q' + 1 := by
186       dsimp [q, q']
187       rw [← monomialWeight_extend_order, ← monomialWeight_extend_order]
188       omega
189     by_cases hq : Finsupp.weight orderWeight q = T + 1
190     · have hq' : Finsupp.weight orderWeight q' = T := by omega
191       rw [if_pos hq, if_pos hq', MvPolynomial.pderiv_monomial,
192         extendPhaseExponent_newestMomentum_apply,
193         extendPhaseExponent_sub_newestMomentum_general]
194       rw [MvPolynomial.smul_monomial]
195       simp [q']
196     · have hq' : Finsupp.weight orderWeight q' ≠ T := by
197         intro h
198         apply hq
199         omega
200       rw [if_neg hq, if_neg hq']
201       simp

203 /-- For every presented Weyl element and every homogeneous order degree, the
204 symbol of commutation with the newest coordinate is the negative Poisson
205 bracket with its coordinate symbol.  No filtration-bound hypothesis on `w`
206 is required. -/
207 theorem principalComponent_coordinate_commutator_arbitrary
208     (n T : ℕ) (w : WeylAlg k (n + 1)) :
209     presentedPrincipalComponent k (@orderWeight (n + 1)) T
210         (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) w) =
211       -poissonBracket (MvPolynomial.X (.inl (0 : Fin (n + 1))))
212         (presentedPrincipalComponent k (@orderWeight (n + 1)) (T + 1) w) := by
213   have h := LinearMap.congr_fun (coordinate_symbol_linear_maps_eq k n T) w
214   change _ = -MvPolynomial.pderiv (.inr (0 : Fin (n + 1)))
215       (presentedPrincipalComponent k orderWeight (T + 1) w) at h
216   rw [poissonBracket_newestCoordinate]
217   exact h
