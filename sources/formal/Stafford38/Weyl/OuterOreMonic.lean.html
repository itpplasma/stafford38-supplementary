7 /-!
8 # PBW monicity as concrete outer-Ore monicity
9 
10 The successor-rank iterated Weyl tower is an outer Ore extension in the newest
11 momentum. This file exposes its concrete coefficient-left polynomial, proves
12 the exact coefficient formula relating the nested PBW normal form to that
13 polynomial, and turns the checked PBW coefficient-one bound into
14 `Polynomial.Monic`.
15 -/

118 theorem coeff_flattenPairSymbols_monomial (n a p a' p' : ℕ)
119     (m : PhaseVar n →₀ ℕ) (r : SymbolRing k n) :
120     AddMonoidAlgebra.coeff
121         (flattenPairSymbols k n
122           (Polynomial.monomial p' (Polynomial.monomial a' r)))
123         (pairExponent n a p m) =
124       if p' = p ∧ a' = a then AddMonoidAlgebra.coeff r m else 0 := by
125   rw [flattenPairSymbols_monomial]
126   rw [MvPolynomial.X_pow_eq_monomial, MvPolynomial.X_pow_eq_monomial,
127     MvPolynomial.monomial_mul, mul_one]
128   have hcoeff := MvPolynomial.coeff_monomial_mul'
129     (pairExponent n a p m)
130     (Finsupp.single (.inr (0 : Fin (n + 1))) p' +
131       Finsupp.single (.inl (0 : Fin (n + 1))) a')
132     (1 : k) (MvPolynomial.rename oldIndex r)
133   rw [hcoeff]
134   classical
135   by_cases hp : p' = p
136   · subst p'
137     by_cases ha : a' = a
138     · subst a'
139       have hle :
140           (Finsupp.single (.inr (0 : Fin (n + 1))) p +
141               Finsupp.single (.inl (0 : Fin (n + 1))) a) ≤
142             pairExponent n a p m := by
143         intro i
144         simp only [pairExponent, Finsupp.add_apply]
145         omega
146       rw [if_pos hle]
147       have hsub :
148           (pairExponent n a p m -
149             (Finsupp.single (.inr (0 : Fin (n + 1))) p +
150               Finsupp.single (.inl (0 : Fin (n + 1))) a)) =
151             Finsupp.mapDomain oldIndex m := by
152         rw [pairExponent]
153         exact add_tsub_cancel_left _ _
154       rw [hsub, MvPolynomial.coeff_rename_mapDomain oldIndex oldIndex_injective]
155       simp
156     · by_cases hle :
157           ((Finsupp.single (.inr (0 : Fin (n + 1))) p +
158               Finsupp.single (.inl (0 : Fin (n + 1))) a') ≤
159             pairExponent n a p m)
160       · rw [if_pos hle]
161         have haa : a' ≤ a := by
162           simpa [pairExponent, Finsupp.single_apply] using
163             hle (.inl (0 : Fin (n + 1)))
164         have hz : (MvPolynomial.rename oldIndex r).coeff
165           ((pairExponent n a p m) -
166               (Finsupp.single (.inr (0 : Fin (n + 1))) p +
167                 Finsupp.single (.inl (0 : Fin (n + 1))) a')) = 0 := by
168           apply MvPolynomial.coeff_rename_eq_zero
169           intro u hu
170           have hnew := DFunLike.congr_fun hu (.inl (0 : Fin (n + 1)))
171           simp [pairExponent] at hnew
172           omega
173         change MvPolynomial.coeff
174           (pairExponent n a p m -
175             (Finsupp.single (.inr (0 : Fin (n + 1))) p +
176               Finsupp.single (.inl (0 : Fin (n + 1))) a'))
177           (MvPolynomial.rename oldIndex r) = 0 at hz
178         change 1 * MvPolynomial.coeff
179           (pairExponent n a p m -
180             (Finsupp.single (.inr (0 : Fin (n + 1))) p +
181               Finsupp.single (.inl (0 : Fin (n + 1))) a'))
182           (MvPolynomial.rename oldIndex r) = _
183         rw [hz]
184         simp [ha]
185       · rw [if_neg hle]
186         simp [ha]
187   · by_cases hle :
188         ((Finsupp.single (.inr (0 : Fin (n + 1))) p' +
189             Finsupp.single (.inl (0 : Fin (n + 1))) a') ≤
190           pairExponent n a p m)
191     · rw [if_pos hle]
192       have hpp : p' ≤ p := by
193         simpa [pairExponent, Finsupp.single_apply] using
194           hle (.inr (0 : Fin (n + 1)))
195       have hz : (MvPolynomial.rename oldIndex r).coeff
196           ((pairExponent n a p m) -
197             (Finsupp.single (.inr (0 : Fin (n + 1))) p' +
198               Finsupp.single (.inl (0 : Fin (n + 1))) a')) = 0 := by
199         apply MvPolynomial.coeff_rename_eq_zero
200         intro u hu
201         have hnew := DFunLike.congr_fun hu (.inr (0 : Fin (n + 1)))
202         simp [pairExponent] at hnew
203         omega
204       change MvPolynomial.coeff
205         (pairExponent n a p m -
206           (Finsupp.single (.inr (0 : Fin (n + 1))) p' +
207             Finsupp.single (.inl (0 : Fin (n + 1))) a'))
208         (MvPolynomial.rename oldIndex r) = 0 at hz
209       change 1 * MvPolynomial.coeff
210         (pairExponent n a p m -
211           (Finsupp.single (.inr (0 : Fin (n + 1))) p' +
212             Finsupp.single (.inl (0 : Fin (n + 1))) a'))
213         (MvPolynomial.rename oldIndex r) = _
214       rw [hz]
215       simp [hp]
216     · rw [if_neg hle]
217       simp [hp]

219 theorem coeff_flattenPairSymbols (n a p : ℕ)
220     (m : PhaseVar n →₀ ℕ)
221     (q : Polynomial (Polynomial (SymbolRing k n))) :
222     AddMonoidAlgebra.coeff (flattenPairSymbols k n q)
223         (pairExponent n a p m) =
224       AddMonoidAlgebra.coeff ((q.coeff p).coeff a) m := by
225   induction q using Polynomial.induction_on' with
226   | add q₁ q₂ h₁ h₂ =>
227       change AddMonoidAlgebra.coeff (flattenPairSymbols k n q₁)
228           (pairExponent n a p m) =
229         AddMonoidAlgebra.coeff ((q₁.coeff p).coeff a) m at h₁
230       change AddMonoidAlgebra.coeff (flattenPairSymbols k n q₂)
231           (pairExponent n a p m) =
232         AddMonoidAlgebra.coeff ((q₂.coeff p).coeff a) m at h₂
233       calc
234         AddMonoidAlgebra.coeff (flattenPairSymbols k n (q₁ + q₂))
235             (pairExponent n a p m) =
236             AddMonoidAlgebra.coeff (flattenPairSymbols k n q₁)
237                 (pairExponent n a p m) +
238               AddMonoidAlgebra.coeff (flattenPairSymbols k n q₂)
239                 (pairExponent n a p m) := by
240                   rw [map_add, AddMonoidAlgebra.coeff_add, Finsupp.add_apply]
241         _ = AddMonoidAlgebra.coeff ((q₁.coeff p).coeff a) m +
242               AddMonoidAlgebra.coeff ((q₂.coeff p).coeff a) m := by rw [h₁, h₂]
243         _ = AddMonoidAlgebra.coeff (((q₁ + q₂).coeff p).coeff a) m := by
244               rw [Polynomial.coeff_add, Polynomial.coeff_add]
245               exact (Finsupp.add_apply
246                 (AddMonoidAlgebra.coeff ((q₁.coeff p).coeff a))
247                 (AddMonoidAlgebra.coeff ((q₂.coeff p).coeff a)) m).symm
248   | monomial p' r =>
249       induction r using Polynomial.induction_on' with
250       | add r₁ r₂ h₁ h₂ =>
251           change AddMonoidAlgebra.coeff
252               (flattenPairSymbols k n (Polynomial.monomial p' r₁))
253               (pairExponent n a p m) =
254             AddMonoidAlgebra.coeff
255               (((Polynomial.monomial p' r₁).coeff p).coeff a) m at h₁
256           change AddMonoidAlgebra.coeff
257               (flattenPairSymbols k n (Polynomial.monomial p' r₂))
258               (pairExponent n a p m) =
259             AddMonoidAlgebra.coeff
260               (((Polynomial.monomial p' r₂).coeff p).coeff a) m at h₂
261           rw [map_add (Polynomial.monomial p') r₁ r₂]
262           simp only [map_add, AddMonoidAlgebra.coeff_add,
263             Finsupp.add_apply, Polynomial.coeff_monomial, Polynomial.coeff_add]
264           by_cases hp : p' = p
265           · subst p'
266             simp [h₁, h₂]
267           · rw [h₁, h₂]
268             simp [Polynomial.coeff_monomial, hp]
269       | monomial a' s =>
270           rw [coeff_flattenPairSymbols_monomial]
271           simp only [Polynomial.coeff_monomial]
272           by_cases hp : p' = p
273           · subst p'
274             by_cases ha : a' = a
275             · subst a'
276               simp
277             · simp [Polynomial.coeff_monomial, ha]
278           · simp [hp]

280 theorem coeff_polynomialMapRangeLinearEquiv
281     {R S : Type*} [Semiring R] [Semiring S]
282     [Algebra k R] [Algebra k S] (e : R ≃ₗ[k] S)
283     (q : Polynomial R) (p : ℕ) :
284     (polynomialMapRangeLinearEquiv e q).coeff p = e (q.coeff p) := by
285   induction q using Polynomial.induction_on' with
286   | add q₁ q₂ h₁ h₂ => simp [map_add, h₁, h₂]
287   | monomial n r =>
288       rw [polynomialMapRangeLinearEquiv_monomial]
289       by_cases h : n = p <;> simp [Polynomial.coeff_monomial, h]

302 theorem nested_coeff_eq_zero_of_outer_exponent_gt (n N : ℕ)
303     {d : PresentedWeyl k (n + 1)}
304     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
305     {p : ℕ} (hp : N < p) :
306     (presentedNestedNormalForm k n d).coeff p = 0 := by
307   ext a m
308   simp only [Polynomial.coeff_zero]
309   change AddMonoidAlgebra.coeff
310     (((presentedNestedNormalForm k n d).coeff p).coeff a) m = 0
311   rw [← coeff_flattenPairSymbols k n a p m]
312   rw [flatten_presentedNestedNormalForm]
313   exact coeff_normalForm_eq_zero_of_exponent_gt k
314     (t := .inr (0 : Fin (n + 1))) (m := pairExponent n a p m)
315     hd.1 (by simpa using hp)

317 theorem nested_coeff_eq_one_at_bound (n N : ℕ)
318     {d : PresentedWeyl k (n + 1)}
319     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
320     (presentedNestedNormalForm k n d).coeff N = 1 := by
321   ext a m
322   change AddMonoidAlgebra.coeff
323     (((presentedNestedNormalForm k n d).coeff N).coeff a) m =
324       AddMonoidAlgebra.coeff ((1 : Polynomial (SymbolRing k n)).coeff a) m
325   rw [← coeff_flattenPairSymbols k n a N m]
326   rw [flatten_presentedNestedNormalForm]
327   by_cases ha : a = 0
328   · subst a
329     by_cases hm : m = 0
330     · subst m
331       simpa [pairExponent] using hd.2
332     · have hmap : Finsupp.mapDomain oldIndex m ≠ 0 := by
333         exact (Finsupp.mapDomain_injective oldIndex_injective).ne hm
334       have hdegree : N < (pairExponent n 0 N m).degree := by
335         rw [degree_pairExponent]
336         have hpos : 0 < (Finsupp.mapDomain oldIndex m).degree := by
337           exact Nat.pos_of_ne_zero
338             ((Finsupp.degree_eq_zero_iff _).not.mpr hmap)
339         omega
340       have hz : (presentedNormalFormLinearEquiv k (n + 1) d).coeff
341           (pairExponent n 0 N m) = 0 := by
342         by_contra hc
343         have hle := (mem_presentedWeightPiece k (@bernsteinWeight (n + 1)) N d).mp
344           hd.1 (pairExponent n 0 N m) hc
345         have : (pairExponent n 0 N m).degree ≤ N := by
346           change (pairExponent n 0 N m).sum (fun _ e => e) ≤ N
347           simpa [monomialWeight, bernsteinWeight] using hle
348         omega
349       change MvPolynomial.coeff (pairExponent n 0 N m)
350         (presentedNormalFormLinearEquiv k (n + 1) d) = _
351       rw [hz]
352       simp [Polynomial.coeff_one, MvPolynomial.coeff_one, Ne.symm hm]
353   · have hdegree : N < (pairExponent n a N m).degree := by
354       rw [degree_pairExponent]
355       omega
356     have hz : (presentedNormalFormLinearEquiv k (n + 1) d).coeff
357         (pairExponent n a N m) = 0 := by
358       by_contra hc
359       have hle := (mem_presentedWeightPiece k (@bernsteinWeight (n + 1)) N d).mp
360         hd.1 (pairExponent n a N m) hc
361       have : (pairExponent n a N m).degree ≤ N := by
362         change (pairExponent n a N m).sum (fun _ e => e) ≤ N
363         simpa [monomialWeight, bernsteinWeight] using hle
364       omega
365     change MvPolynomial.coeff (pairExponent n a N m)
366       (presentedNormalFormLinearEquiv k (n + 1) d) = _
367     rw [hz]
368     simp [Polynomial.coeff_one, ha]

385 theorem outer_coeff_eq_one_at_bound (n N : ℕ)
386     {d : PresentedWeyl k (n + 1)}
387     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
388     (presentedOuterPolynomial k n d).coeff N = 1 := by
389   have h := nested_coeff_eq_one_at_bound k n N hd

391   apply (coordinateCoefficientNormalForm k n).injective
392   rw [h, coordinateCoefficientNormalForm_one]

394 theorem outer_coeff_eq_zero_of_exponent_gt (n N : ℕ)
395     {d : PresentedWeyl k (n + 1)}
396     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
397     {p : ℕ} (hp : N < p) :
398     (presentedOuterPolynomial k n d).coeff p = 0 := by

404 theorem presentedOuterPolynomial_monic (n N : ℕ)
405     {d : PresentedWeyl k (n + 1)}
406     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
407     (presentedOuterPolynomial k n d).Monic := by
408   apply Polynomial.monic_of_degree_le N
409   · rw [Polynomial.degree_le_iff_coeff_zero]
410     intro p hp
411     exact outer_coeff_eq_zero_of_exponent_gt k n N hd (by simpa using hp)
412   · exact outer_coeff_eq_one_at_bound k n N hd
