9 /-!
10 # PBW monicity as concrete outer-Ore monicity
11 
12 The successor-rank iterated Weyl tower is an outer Ore extension in the newest
13 momentum. This file exposes its concrete coefficient-left polynomial, proves
14 the exact coefficient formula relating the nested PBW normal form to that
15 polynomial, and turns the checked PBW coefficient-one bound into
16 `Polynomial.Monic`.

221 theorem coeff_flattenPairSymbols (n a p : ℕ)
222     (m : PhaseVar n →₀ ℕ)
223     (q : Polynomial (Polynomial (SymbolRing k n))) :
224     AddMonoidAlgebra.coeff (flattenPairSymbols k n q)
225         (pairExponent n a p m) =
226       AddMonoidAlgebra.coeff ((q.coeff p).coeff a) m := by
227   induction q using Polynomial.induction_on' with
228   | add q₁ q₂ h₁ h₂ =>
229       change AddMonoidAlgebra.coeff (flattenPairSymbols k n q₁)
230           (pairExponent n a p m) =
231         AddMonoidAlgebra.coeff ((q₁.coeff p).coeff a) m at h₁
232       change AddMonoidAlgebra.coeff (flattenPairSymbols k n q₂)
233           (pairExponent n a p m) =
234         AddMonoidAlgebra.coeff ((q₂.coeff p).coeff a) m at h₂
235       calc
236         AddMonoidAlgebra.coeff (flattenPairSymbols k n (q₁ + q₂))
237             (pairExponent n a p m) =
238             AddMonoidAlgebra.coeff (flattenPairSymbols k n q₁)
239                 (pairExponent n a p m) +
240               AddMonoidAlgebra.coeff (flattenPairSymbols k n q₂)
241                 (pairExponent n a p m) := by
242                   rw [map_add, AddMonoidAlgebra.coeff_add, Finsupp.add_apply]
243         _ = AddMonoidAlgebra.coeff ((q₁.coeff p).coeff a) m +
244               AddMonoidAlgebra.coeff ((q₂.coeff p).coeff a) m := by rw [h₁, h₂]
245         _ = AddMonoidAlgebra.coeff (((q₁ + q₂).coeff p).coeff a) m := by
246               rw [Polynomial.coeff_add, Polynomial.coeff_add]
247               exact (Finsupp.add_apply
248                 (AddMonoidAlgebra.coeff ((q₁.coeff p).coeff a))
249                 (AddMonoidAlgebra.coeff ((q₂.coeff p).coeff a)) m).symm
250   | monomial p' r =>
251       induction r using Polynomial.induction_on' with
252       | add r₁ r₂ h₁ h₂ =>
253           change AddMonoidAlgebra.coeff
254               (flattenPairSymbols k n (Polynomial.monomial p' r₁))
255               (pairExponent n a p m) =
256             AddMonoidAlgebra.coeff
257               (((Polynomial.monomial p' r₁).coeff p).coeff a) m at h₁
258           change AddMonoidAlgebra.coeff
259               (flattenPairSymbols k n (Polynomial.monomial p' r₂))
260               (pairExponent n a p m) =
261             AddMonoidAlgebra.coeff
262               (((Polynomial.monomial p' r₂).coeff p).coeff a) m at h₂
263           rw [map_add (Polynomial.monomial p') r₁ r₂]
264           simp only [map_add, AddMonoidAlgebra.coeff_add,
265             Finsupp.add_apply, Polynomial.coeff_monomial, Polynomial.coeff_add]
266           by_cases hp : p' = p
267           · subst p'
268             simp [h₁, h₂]
269           · rw [h₁, h₂]
270             simp [Polynomial.coeff_monomial, hp]
271       | monomial a' s =>
272           rw [coeff_flattenPairSymbols_monomial]
273           simp only [Polynomial.coeff_monomial]
274           by_cases hp : p' = p
275           · subst p'
276             by_cases ha : a' = a
277             · subst a'
278               simp
279             · simp [Polynomial.coeff_monomial, ha]
280           · simp [hp]

282 theorem coeff_polynomialMapRangeLinearEquiv
283     {R S : Type*} [Semiring R] [Semiring S]
284     [Algebra k R] [Algebra k S] (e : R ≃ₗ[k] S)
285     (q : Polynomial R) (p : ℕ) :
286     (polynomialMapRangeLinearEquiv e q).coeff p = e (q.coeff p) := by
287   induction q using Polynomial.induction_on' with
288   | add q₁ q₂ h₁ h₂ => simp [map_add, h₁, h₂]
289   | monomial n r =>
290       rw [polynomialMapRangeLinearEquiv_monomial]
291       by_cases h : n = p <;> simp [Polynomial.coeff_monomial, h]

304 theorem nested_coeff_eq_zero_of_outer_exponent_gt (n N : ℕ)
305     {d : WeylAlg k (n + 1)}
306     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
307     {p : ℕ} (hp : N < p) :
308     (presentedNestedNormalForm k n d).coeff p = 0 := by

312     (((presentedNestedNormalForm k n d).coeff p).coeff a) m = 0

316     (t := .inr (0 : Fin (n + 1))) (m := pairExponent n a p m)
317     hd.1 (by simpa using hp)

319 theorem nested_coeff_eq_one_at_bound (n N : ℕ)
320     {d : WeylAlg k (n + 1)}
321     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
322     (presentedNestedNormalForm k n d).coeff N = 1 := by
323   ext a m
324   change AddMonoidAlgebra.coeff
325     (((presentedNestedNormalForm k n d).coeff N).coeff a) m =
326       AddMonoidAlgebra.coeff ((1 : Polynomial (SymbolRing k n)).coeff a) m
327   rw [← coeff_flattenPairSymbols k n a N m]
328   rw [flatten_presentedNestedNormalForm]
329   by_cases ha : a = 0
330   · subst a
331     by_cases hm : m = 0
332     · subst m
333       simpa [pairExponent] using hd.2
334     · have hmap : Finsupp.mapDomain Stafford38FixedSourceChallenge.oldIndex m ≠ 0 := by
335         exact (Finsupp.mapDomain_injective oldIndex_injective).ne hm
336       have hdegree : N < (pairExponent n 0 N m).degree := by
337         rw [degree_pairExponent]
338         have hpos : 0 < (Finsupp.mapDomain Stafford38FixedSourceChallenge.oldIndex m).degree := by
339           exact Nat.pos_of_ne_zero
340             ((Finsupp.degree_eq_zero_iff _).not.mpr hmap)
341         omega
342       have hz : (presentedNormalFormLinearEquiv k (n + 1) d).coeff
343           (pairExponent n 0 N m) = 0 := by
344         by_contra hc
345         have hle := (mem_presentedWeightPiece k (@bernsteinWeight (n + 1)) N d).mp
346           hd.1 (pairExponent n 0 N m) hc
347         have : (pairExponent n 0 N m).degree ≤ N := by
348           change (pairExponent n 0 N m).sum (fun _ e => e) ≤ N
349           simpa [Finsupp.weight_apply, bernsteinWeight] using hle
350         omega
351       change MvPolynomial.coeff (pairExponent n 0 N m)
352         (presentedNormalFormLinearEquiv k (n + 1) d) = _
353       rw [hz]
354       simp [Polynomial.coeff_one, MvPolynomial.coeff_one, Ne.symm hm]
355   · have hdegree : N < (pairExponent n a N m).degree := by
356       rw [degree_pairExponent]
357       omega
358     have hz : (presentedNormalFormLinearEquiv k (n + 1) d).coeff
359         (pairExponent n a N m) = 0 := by
360       by_contra hc
361       have hle := (mem_presentedWeightPiece k (@bernsteinWeight (n + 1)) N d).mp
362         hd.1 (pairExponent n a N m) hc
363       have : (pairExponent n a N m).degree ≤ N := by
