42 private theorem polynomialMapRangeLinearEquiv_derivative
43     {R S : Type u} [Ring R] [CommRing S] [Algebra k R] [Algebra k S]
44     (e : R ≃ₗ[k] S) (q : Polynomial R) :
45     polynomialMapRangeLinearEquiv e q.derivative =
46       (polynomialMapRangeLinearEquiv e q).derivative := by
47   apply Polynomial.ext
48   intro a
49   calc
50     _ = e ((q.derivative).coeff a) :=
51       coeff_polynomialMapRangeLinearEquiv k e q.derivative a
52     _ = e (q.coeff (a + 1) * (a + 1)) := by
53       rw [Polynomial.coeff_derivative]
54     _ = e (q.coeff (a + 1)) * (a + 1) := by
55       calc
56             _ = e ((a + 1) • q.coeff (a + 1)) := by
57               rw [← Nat.cast_smul_eq_nsmul R (a + 1) (q.coeff (a + 1)), smul_eq_mul]
58               congr 1
59               simpa only [Nat.cast_succ] using
60                 (Nat.cast_commute (a + 1) (q.coeff (a + 1))).eq.symm
61         _ = (a + 1) • e (q.coeff (a + 1)) := map_nsmul e _ _
62         _ = _ := by
63               rw [← Nat.cast_smul_eq_nsmul S (a + 1) (e (q.coeff (a + 1))), smul_eq_mul]
64               simpa only [Nat.cast_succ] using
65                 (Nat.cast_commute (a + 1) (e (q.coeff (a + 1)))).eq
66     _ = _ := by
67       rw [Polynomial.coeff_derivative,
68         coeff_polynomialMapRangeLinearEquiv k]

175 @[simp] theorem mem_coordinateBernsteinPiece (n L : ℕ)
176     (z : CoordinateStage (B := IteratedPairStage k n)) :
177     z ∈ coordinateBernsteinPiece k n L ↔
178       ∀ a m,
179         MvPolynomial.coeff m
180           ((coordinateCoefficientNormalForm k n z).coeff a) ≠ 0 →
181           a + Finsupp.weight (@bernsteinWeight n) m ≤ L := by
182   change coordinateStageToPresented k n z ∈
183     Stafford38.WeylFiltration.bernsteinPiece k (n + 1) L ↔ _
184   rw [Stafford38.WeylFiltration.bernsteinPiece, mem_presentedWeightPiece,
185     coordinateStageToPresented_normalForm]
186   constructor
187   · intro h a m hm
188     have hnonzero : MvPolynomial.coeff (pairExponent n a 0 m)
189         (flattenPairSymbols k n
190           (Polynomial.C (coordinateCoefficientNormalForm k n z))) ≠ 0 := by
191       change AddMonoidAlgebra.coeff
192         (flattenPairSymbols k n
193           (Polynomial.C (coordinateCoefficientNormalForm k n z)))
194         (pairExponent n a 0 m) ≠ 0
195       rw [coeff_flattenPairSymbols]
196       simpa [Polynomial.coeff_C] using hm
197     have hbound := h (pairExponent n a 0 m) hnonzero
198     have hweight : Finsupp.weight (@bernsteinWeight (n + 1))
199         (pairExponent n a 0 m) =
200           Finsupp.weight (@bernsteinWeight n) m + a := by
201       simpa [pairExponent, extendPhaseExponent, add_comm, add_left_comm,
202         add_assoc] using
203           (monomialWeight_extend_bernstein n a 0 m).symm
204     omega
205   · intro h q hq
206     let aq : Fin (n + 1) → ℕ := fun i => q (.inl i)
207     let pq : Fin (n + 1) → ℕ := fun i => q (.inr i)
208     let old : PhaseVar n →₀ ℕ :=
209       phaseExponent (fun i => aq i.succ) (fun i => pq i.succ)
210     have hqsplit : q = pairExponent n (aq 0) (pq 0) old := by
211       have hphase : phaseExponent aq pq = q := by
212         ext i <;> cases i <;> simp [phaseExponent, aq, pq]
213       rw [← hphase, phaseExponent_succ]
214       rfl
215     have hpzero : pq 0 = 0 := by
216       by_contra hp
217       have hz : MvPolynomial.coeff q
218           (flattenPairSymbols k n
219             (Polynomial.C (coordinateCoefficientNormalForm k n z))) = 0 := by
220         change AddMonoidAlgebra.coeff
221           (flattenPairSymbols k n
222             (Polynomial.C (coordinateCoefficientNormalForm k n z))) q = 0
223         rw [hqsplit, coeff_flattenPairSymbols]
224         simp [Polynomial.coeff_C, hp]
225       exact hq hz
226     have hsource : MvPolynomial.coeff old
227         ((coordinateCoefficientNormalForm k n z).coeff (aq 0)) ≠ 0 := by
228       have := hq
229       change AddMonoidAlgebra.coeff
230         (flattenPairSymbols k n
231           (Polynomial.C (coordinateCoefficientNormalForm k n z))) q ≠ 0 at this
232       rw [hqsplit, coeff_flattenPairSymbols] at this
233       simpa [Polynomial.coeff_C, hpzero] using this
234     have hbound := h (aq 0) old hsource
235     have hweight : Finsupp.weight (@bernsteinWeight (n + 1)) q =
236         Finsupp.weight (@bernsteinWeight n) old + aq 0 := by
237       rw [hqsplit]
238       simpa [pairExponent, extendPhaseExponent, hpzero, add_comm,
239         add_left_comm, add_assoc] using
240           (monomialWeight_extend_bernstein n (aq 0) 0 old).symm
241     omega

280     {z : CoordinateStage (B := IteratedPairStage k n)}
281     (hz : z ∈ coordinateBernsteinPiece k n L) :
282     coordinateDerivation z ∈ coordinateBernsteinPiece k n L := by

285   intro a m hm

422 private theorem rightPBWBasis_repr_monomial
423     {B : Type*} [Ring B] [Nontrivial B]
424     (D : OreDivisionDerivation B) (l : ℕ) (b : B) :
425     (rightOrePBWBasis D).repr (normalForm D (Polynomial.monomial l b)) =
426       ∑ ij ∈ Finset.HasAntidiagonal.antidiagonal l,
427         l.choose ij.1 •
428           (if Even ij.1 then Finsupp.single ij.2 (MulOpposite.op
429               ((D^[ij.1]) b))
430           else -Finsupp.single ij.2 (MulOpposite.op

432   rw [normalForm_monomial_reverse]
433   simp only [map_sum, map_nsmul]
434   apply Finset.sum_congr rfl

441     rw [rightPBWBasis_repr_single, if_pos he]

622     (d : WeylAlg k (n + 1))
623     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
624     ((pairStage_rightOrePBWBasis (B := IteratedPairStage k n)).repr

669 /-- The PBW monic normal form with right coefficients, including the degree
670 bound on every lower coefficient. -/
671 theorem presentedOuter_right_monic_decomposition (n N : ℕ)
672     (d : WeylAlg k (n + 1))
673     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
674     ∃ a : ℕ → CoordinateStage (B := IteratedPairStage k n),
675       (∀ j, j < N → a j ∈ coordinateBernsteinPiece k n (N - j)) ∧
676       presentedToIterated k (n + 1) d =
677         rightPBWMonomial (coordinateDerivation :
678           OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n))) N +
679         ∑ j ∈ Finset.range N,
680           MulOpposite.op (a j) • rightPBWMonomial (coordinateDerivation :
681             OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n))) j := by
682   let D : OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)) :=
683     coordinateDerivation
684   let b := pairStage_rightOrePBWBasis (B := IteratedPairStage k n)
685   let c := b.repr (normalForm D (presentedOuterPolynomial k n d))
686   have hnormal : normalForm D (presentedOuterPolynomial k n d) =
687       presentedToIterated k (n + 1) d := by
688     dsimp [D, presentedOuterPolynomial]
689     exact (normalFormAddEquiv coordinateDerivation).apply_symm_apply _
690   have hcTop : c N = MulOpposite.op (1 : CoordinateStage
691       (B := IteratedPairStage k n)) := by
692     simpa [c, b, D, hnormal, pairStage_rightOrePBWBasis] using
693       presentedOuterRightBasis_coeff_at_bound k n N d hd
694   have hcAbove (j : ℕ) (hNj : N < j) : c j = 0 := by
695     simpa [c, b, D, hnormal, pairStage_rightOrePBWBasis] using
696       presentedOuterRightBasis_coeff_zero_of_gt k n N j d hd hNj
697   have heraseSupport : (Finsupp.erase N c).support ⊆ Finset.range N := by
698     intro j hj
699     rw [Finsupp.support_erase, Finset.mem_erase] at hj
700     rcases hj with ⟨hne, hmem⟩
701     have hcnz : c j ≠ 0 := Finsupp.mem_support_iff.mp hmem
702     have hjN : j < N := by
703       by_contra hjN
704       have hNj : N < j := by omega
705       exact hcnz (hcAbove j hNj)
706     exact Finset.mem_range.mpr hjN
707   have heraseSum :
708       Finsupp.sum (Finsupp.erase N c) (fun j a => a • b j) =
709         ∑ j ∈ Finset.range N, (Finsupp.erase N c) j • b j := by
710     exact Finsupp.sum_of_support_subset (Finsupp.erase N c) heraseSupport
711       (fun j a => a • b j) (by intro j hj; simp)
712   have hdecomp : b.repr.symm c = c N • b N +
713       ∑ j ∈ Finset.range N, c j • b j := by
714     have hsingle : (Finsupp.linearCombination
715         (MulOpposite (CoordinateStage (B := IteratedPairStage k n))) b
716         (Finsupp.single N (c N))) = c N • b N := by
717       rw [Finsupp.linearCombination_apply]
718       exact Finsupp.sum_single_index (by simp)
719     calc
720       b.repr.symm c = Finsupp.linearCombination
721           (MulOpposite (CoordinateStage (B := IteratedPairStage k n))) b c :=
722             b.repr_symm_apply c
723       _ = Finsupp.linearCombination
724           (MulOpposite (CoordinateStage (B := IteratedPairStage k n))) b
725           (Finsupp.single N (c N) + Finsupp.erase N c) := by
726             rw [Finsupp.single_add_erase]
727       _ = _ := by
728             rw [map_add]
729             rw [hsingle]
730             rw [Finsupp.linearCombination_apply]
731             rw [heraseSum]
732             congr 1
733             apply Finset.sum_congr rfl
734             intro j hj
735             rw [Finsupp.erase_apply]
736             have hjlt : j < N := Finset.mem_range.mp hj
737             have hjne : j ≠ N := by omega
738             simp [hjne]
739   have hback : presentedToIterated k (n + 1) d = b.repr.symm c := by
740     dsimp [c]
741     rw [← hnormal]
742     exact (b.repr.symm_apply_apply _).symm
743   have hbasis (j : ℕ) : b j = rightPBWMonomial D j := by
744     simp [b, D, pairStage_rightOrePBWBasis]
745   have hdecomp' : b.repr.symm c = c N • rightPBWMonomial D N +
746       ∑ j ∈ Finset.range N, c j • rightPBWMonomial D j := by
747     calc
748       b.repr.symm c = c N • b N + ∑ j ∈ Finset.range N, c j • b j := hdecomp
749       _ = c N • rightPBWMonomial D N +
750           ∑ j ∈ Finset.range N, c j • rightPBWMonomial D j := by
751             rw [hbasis N]
752             congr 1
753             apply Finset.sum_congr rfl
754             intro j hj
755             rw [hbasis j]
756   let a : ℕ → CoordinateStage (B := IteratedPairStage k n) :=
757     fun j => MulOpposite.unop (c j)
758   refine ⟨a, ?_, ?_⟩
759   · intro j hj
760     dsimp [a]
761     simpa [c, b, D, hnormal, pairStage_rightOrePBWBasis] using
762       (presentedOuterRightBernsteinBound k n N d hd) j
763   · rw [hback, hdecomp', hcTop]
764     simp [a, D] <;> rfl
