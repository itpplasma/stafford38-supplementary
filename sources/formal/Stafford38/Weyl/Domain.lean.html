7 /-!
8 # The presented Weyl algebra is a domain
9 
10 The Bernstein principal component is multiplicative. A nonzero element has a
11 nonzero top Bernstein component, so the product of two nonzero elements cannot
12 vanish.
13 -/

24 /-- The product of two nonzero elements of a Weyl algebra is nonzero. -/
25 theorem mul_ne_zero {k : Type u} [Field k] {n : ℕ}
26     {a b : Stafford38Challenge.WeylAlg k n} (ha : a ≠ 0) (hb : b ≠ 0) :
27     a * b ≠ 0 := by
28   obtain ⟨N, haN, hsa⟩ := exists_top_bernstein_piece k ha
29   obtain ⟨M, hbM, hsb⟩ := exists_top_bernstein_piece k hb
30   have htop := presentedPrincipalComponent_mul_bernstein k haN hbM
31   have hsymbol :
32       presentedPrincipalComponent k (@bernsteinWeight n) N a *
33           presentedPrincipalComponent k (@bernsteinWeight n) M b ≠ 0 :=
34     _root_.mul_ne_zero hsa hsb
35   intro hab
36   have hzero :
37       presentedPrincipalComponent k (@bernsteinWeight n) (N + M) (a * b) = 0 := by
38     simp [hab]
39   rw [htop] at hzero
40   exact hsymbol hzero
41 
42 #print axioms mul_ne_zero
43 
44 end Stafford38.WeylDomain
