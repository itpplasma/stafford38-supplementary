11 /-!
12 # Transport of the canonical right quotient
13 
14 The recursive Ore tower and the presented Weyl algebra are related by an
15 algebra equivalence.  This file transports the *literal* two-generator right
16 ideal and its right quotient across that equivalence.  The final theorem is
17 the presented-Weyl form of the already proved `PairStage` surjectivity.
18 -/

39 instance rightModuleEquiv_invPair {A B : Type*} [Ring A] [Ring B]
40     [Algebra k A] [Algebra k B] (e : A ≃ₐ[k] B) :
41     RingHomInvPair
42       ((AlgEquiv.op e).toRingEquiv : Aᵐᵒᵖ →+* Bᵐᵒᵖ)
43       ((AlgEquiv.op e).toRingEquiv.symm : Bᵐᵒᵖ →+* Aᵐᵒᵖ) :=
44   RingHomInvPair.of_ringEquiv (AlgEquiv.op e).toRingEquiv
45 
46 instance rightModuleEquiv_invPair_symm {A B : Type*} [Ring A] [Ring B]
47     [Algebra k A] [Algebra k B] (e : A ≃ₐ[k] B) :
48     RingHomInvPair
49       ((AlgEquiv.op e).toRingEquiv.symm : Bᵐᵒᵖ →+* Aᵐᵒᵖ)
50       ((AlgEquiv.op e).toRingEquiv : Aᵐᵒᵖ →+* Bᵐᵒᵖ) :=
51   RingHomInvPair.of_ringEquiv (AlgEquiv.op e).toRingEquiv.symm

53 /-- The algebra equivalence, regarded as a semilinear equivalence of regular
54 right modules.  The scalar ring is transported on opposites. -/
55 def rightModuleEquiv {A B : Type*} [Ring A] [Ring B] [Algebra k A]
56     [Algebra k B] (e : A ≃ₐ[k] B) :
57     @LinearEquiv Aᵐᵒᵖ Bᵐᵒᵖ _ _ (AlgEquiv.op e).toRingEquiv
58       (AlgEquiv.op e).toRingEquiv.symm _ _ A B _ _ _ _ :=
59   { e.toAddEquiv with
60     map_smul' := by
61       intro c a
62       change e (a * c.unop) = e a * e c.unop
63       exact map_mul e a c.unop }
64 
65 /-- The inverse regular-right-module equivalence, with the inverse opposite
66 scalar map made explicit for quotient composition. -/
67 def rightModuleEquiv_symm {A B : Type*} [Ring A] [Ring B] [Algebra k A]
68     [Algebra k B] (e : A ≃ₐ[k] B) :
69     @LinearEquiv Bᵐᵒᵖ Aᵐᵒᵖ _ _ (AlgEquiv.op e).toRingEquiv.symm
70       (AlgEquiv.op e).toRingEquiv _ _ B A _ _ _ _ :=
71   { e.symm.toAddEquiv with
72     map_smul' := by
73       intro c b
74       change e.symm (b * c.unop) = e.symm b * e.symm c.unop
75       exact map_mul e.symm b c.unop }

77 /-- Image transport of a literal right ideal along an algebra equivalence. -/
78 def transportedRightIdeal {A B : Type*} [Ring A] [Ring B] [Algebra k A]
79     [Algebra k B] (e : A ≃ₐ[k] B) (I : RightIdeal A) : RightIdeal B :=
80   { carrier := e '' I
81     zero_mem' := by
82       exact ⟨0, I.zero_mem, by simp⟩
83     add_mem' := by
84       rintro y z ⟨y', hy', rfl⟩ ⟨z', hz', rfl⟩
85       refine ⟨y' + z', I.add_mem hy' hz', ?_⟩
86       simp
87     smul_mem' := by
88       rintro b y ⟨a, ha, rfl⟩
89       let c : Aᵐᵒᵖ := (AlgEquiv.op e).symm b
90       refine ⟨c • a, I.smul_mem c ha, ?_⟩
91       have h := (rightModuleEquiv e).map_smulₛₗ c a
92       calc
93         e (c • a) = (AlgEquiv.op e).toRingEquiv c • e a := h
94         _ = b • e a := by
95           rw [show (AlgEquiv.op e).toRingEquiv c = b by
96             exact (AlgEquiv.op e).apply_symm_apply b] }

100 /-- The literal canonical right ideal in the presented Weyl algebra. -/
101 abbrev presentedCanonicalRightIdeal (n N : ℕ) (d : WeylAlg k (n + 1)) :
102     RightIdeal (WeylAlg k (n + 1)) :=
103   canonicalRightIdeal (presentedCoordinate k n) d N
104 
105 /-- Its literal right quotient. -/
106 abbrev PresentedCanonicalRightQuotient (n N : ℕ)
107     (d : WeylAlg k (n + 1)) :=
108   RightQuotient (presentedCanonicalRightIdeal (k := k) n N d)

110 theorem transported_canonicalRightIdeal
111     {A B : Type*} [Ring A] [Ring B] [Algebra k A] [Algebra k B]
112     (e : A ≃ₐ[k] B) (x d : A) (N : ℕ) :
113     transportedRightIdeal e (canonicalRightIdeal x d N) =
114       canonicalRightIdeal (e x) (e d) N := by
115   apply le_antisymm
116   · intro y hy
117     change y ∈ e '' canonicalRightIdeal x d N at hy
118     rcases hy with ⟨z, hz, rfl⟩
119     refine Submodule.span_induction ?_ ?_ ?_ ?_ hz
120     · intro z hz
121       simp only [Set.mem_insert_iff, Set.mem_singleton_iff] at hz
122       rcases hz with hz | hz
123       · subst z
124         exact firstGenerator_mem (e x) (e d) N
125       · subst z
126         simpa [map_mul, map_pow] using
127           (secondGenerator_mem (e x) (e d) N)
128     · simp
129     · intro a b ha hb hha hhb
130       simpa using (canonicalRightIdeal (e x) (e d) N).add_mem hha hhb
131     · intro c a ha hha
132       change (rightModuleEquiv e) (c • a) ∈
133         canonicalRightIdeal (e x) (e d) N
134       rw [(rightModuleEquiv e).map_smulₛₗ c a]
135       exact (canonicalRightIdeal (e x) (e d) N).smul_mem _ hha
136   · intro y hy
137     change y ∈ e '' canonicalRightIdeal x d N
138     refine ⟨e.symm y, ?_, by simp⟩
139     refine Submodule.span_induction ?_ ?_ ?_ ?_ hy
140     · intro z hz
141       simp only [Set.mem_insert_iff, Set.mem_singleton_iff] at hz
142       rcases hz with hz | hz
143       · subst z
144         simpa using (firstGenerator_mem x d N)
145       · subst z
146         simpa [map_mul, map_pow] using (secondGenerator_mem x d N)
147     · simp
148     · intro a b ha hb hha hhb
149       simpa using (canonicalRightIdeal x d N).add_mem hha hhb
150     · intro c a ha hha
151       change (rightModuleEquiv_symm e) (c • a) ∈
152         canonicalRightIdeal x d N
153       rw [(rightModuleEquiv_symm e).map_smulₛₗ c a]
154       exact (canonicalRightIdeal x d N).smul_mem _ hha

156 theorem presented_canonicalRightIdeal_map
157     (n N : ℕ) (d : WeylAlg k (n + 1)) :
158     transportedRightIdeal (presentedIteratedEquiv k (n + 1))
159         (presentedCanonicalRightIdeal (k := k) n N d) =
160       canonicalRightIdeal (pairCoordinate (B := IteratedPairStage k n))
161         (presentedToIterated k (n + 1) d) N := by
162   have h := transported_canonicalRightIdeal
163       (e := presentedIteratedEquiv k (n + 1))
164       (x := presentedCoordinate k n) (d := d) (N := N)
165   change transportedRightIdeal (presentedIteratedEquiv k (n + 1))
166       (canonicalRightIdeal (presentedCoordinate k n) d N) = _
167   have hx : presentedIteratedEquiv k (n + 1) (presentedCoordinate k n) =
168       pairCoordinate (B := IteratedPairStage k n) := by
169     change presentedToIterated k (n + 1) (presentedCoordinate k n) =
170       pairCoordinate (B := IteratedPairStage k n)
171     simpa only [stageCoordinate] using (presentedToIterated_coordinate k n)
172   rw [hx] at h
173   exact h

175 /-- The quotient map induced by the transported ideal. -/
176 def transportedRightQuotientMap {A B : Type*} [Ring A] [Ring B]
177     [Algebra k A] [Algebra k B] (e : A ≃ₐ[k] B) (I : RightIdeal A) :
178     @LinearMap Aᵐᵒᵖ Bᵐᵒᵖ _ _ (AlgEquiv.op e).toRingEquiv
179       (RightQuotient I) (RightQuotient (transportedRightIdeal e I)) _ _ _ _ :=
180   Submodule.mapQ I (transportedRightIdeal e I) (rightModuleEquiv e) (by
181     intro a ha
182     exact ⟨a, ha, by rfl⟩)

185     {A B : Type*} [Ring A] [Ring B] [Algebra k A] [Algebra k B]
186     (e : A ≃ₐ[k] B) (I : RightIdeal A) (a : A) :
187   transportedRightQuotientMap e I (qmk I a) =
188       qmk (transportedRightIdeal e I) (e a) := by

191 /-- The quotient equivalence induced by the transported ideal. -/
192 def transportedRightQuotientEquiv {A B : Type*} [Ring A] [Ring B]
193     [Algebra k A] [Algebra k B] (e : A ≃ₐ[k] B) (I : RightIdeal A) :
194     @LinearEquiv Aᵐᵒᵖ Bᵐᵒᵖ _ _ (AlgEquiv.op e).toRingEquiv
195       (AlgEquiv.op e).toRingEquiv.symm _ _
196       (RightQuotient I) (RightQuotient (transportedRightIdeal e I)) _ _ _ _ := by
197   apply LinearEquiv.ofLinear (transportedRightQuotientMap e I)
198     (Submodule.mapQ (transportedRightIdeal e I) I
199       (rightModuleEquiv_symm e) (by
200         intro b hb
201         change b ∈ e '' I at hb
202         rcases hb with ⟨a, ha, rfl⟩
203         simpa [rightModuleEquiv_symm] using ha))
204   · apply LinearMap.ext
205     intro q
206     refine Submodule.Quotient.induction_on
207       (transportedRightIdeal e I) q ?_
208     intro b
209     change transportedRightQuotientMap e I
210         (qmk I (e.symm b)) = qmk (transportedRightIdeal e I) b
211     rw [transportedRightQuotientMap_qmk]
212     simp
213   · apply LinearMap.ext
214     intro q
215     refine Submodule.Quotient.induction_on I q ?_
216     intro a
217     simp only [LinearMap.coe_comp, Function.comp_apply]
218     change qmk I (e.symm (e a)) = qmk I a
219     simp

227 theorem transportedRightQuotientEquiv_rightMul
228     {A B : Type*} [Ring A] [Ring B] [Algebra k A] [Algebra k B]
229     (e : A ≃ₐ[k] B) (I : RightIdeal A) (a : A) (q : RightQuotient I) :
230     transportedRightQuotientEquiv e I (rightMul I a q) =
231       rightMul (transportedRightIdeal e I) (e a)
232         (transportedRightQuotientEquiv e I q) := by
233   refine Submodule.Quotient.induction_on I q ?_
234   intro b
235   change transportedRightQuotientEquiv e I
236       (MulOpposite.op a • Submodule.Quotient.mk b) = _
237   rw [LinearEquiv.map_smulₛₗ]

240 /-! ## Presented-Weyl surjectivity -/
241 
242 theorem presentedCanonicalRightQuotient_rightMul_coordinate_surjective
243     [Algebra ℚ k] (n N : ℕ) {d : WeylAlg k (n + 1)}
244     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
245     Function.Surjective
246       (rightMul (presentedCanonicalRightIdeal (k := k) n N d)
247         (presentedCoordinate k n)) := by
248   let e := presentedIteratedEquiv k (n + 1)
249   let I := presentedCanonicalRightIdeal (k := k) n N d
250   let f := transportedRightQuotientEquiv e I
251   have hmap : transportedRightIdeal e I =
252       canonicalRightIdeal (pairCoordinate (B := IteratedPairStage k n))
253         (presentedToIterated k (n + 1) d) N := by
254     exact presented_canonicalRightIdeal_map (k := k) n N d
255   have hsurj :=
256     Stafford38.PaperEulerGrading.presentedCanonicalQuotient_rightMul_coordinate_surjective
257       k n N hd
258   have hsurj' : Function.Surjective
259       (rightMul (transportedRightIdeal e I)
260         (e (presentedCoordinate k n))) := by
261     rw [hmap]
262     have hx : e (presentedCoordinate k n) =
263         pairCoordinate (B := IteratedPairStage k n) := by
264       change presentedToIterated k (n + 1) (presentedCoordinate k n) =
265         pairCoordinate (B := IteratedPairStage k n)
266       simpa only [stageCoordinate] using (presentedToIterated_coordinate k n)
267     rw [hx]
268     exact hsurj
269   intro q
270   obtain ⟨r, hr⟩ := hsurj' (f q)
271   refine ⟨f.symm r, ?_⟩
272   apply f.injective
273   rw [transportedRightQuotientEquiv_rightMul]
274   rw [f.apply_symm_apply]
275   exact hr
