11 /-!
12 # From principal-axis normalization to PBW monicity
13 
14 Axis restriction identifies the degree-`N` coefficient with the selected
15 pure-variable coefficient. The same coefficient passes from the principal
16 component to the checked PBW normal form; together with the Bernstein bound
17 this yields a coefficient-one pure momentum term and excludes higher
18 momentum powers. `OuterOreMonic` converts the result to `Polynomial.Monic`;
19 `PaperRightMonic` supplies the paper-facing right-coefficient decomposition
20 and lower-coefficient bounds.
21 -/

38 theorem coeff_axisPolynomial {n : ℕ} (t : PhaseVar n)
39     (P : SymbolRing k n) (N : ℕ) :
40     MvPolynomial.coeff (Finsupp.single () N) (axisPolynomial k t P) =
41       MvPolynomial.coeff (Finsupp.single t N) P := by
42   induction P using MvPolynomial.induction_on' with
43   | monomial m c =>
44       by_cases hsub : ∀ i ∈ m.support, i = t
45       · have hm : m = Finsupp.single t (m t) := by
46           ext i
47           by_cases hit : i = t
48           · subst i
49             simp
50           · have hi : i ∉ m.support := by
51               intro him
52               exact hit (hsub i him)
53             rw [Finsupp.notMem_support_iff.mp hi]
54             rw [Finsupp.single_apply]
55             simp [Ne.symm hit]
56         rw [hm]
57         rw [axisPolynomial, MvPolynomial.aeval_monomial]
58         rw [Finsupp.prod_single_index (by simp)]
59         simp [MvPolynomial.X_pow_eq_monomial]
60         by_cases hmt : m t = N
61         · simp [hmt]
62         · have hs : Finsupp.single t (m t) ≠ Finsupp.single t N := by
63             intro h
64             exact hmt (Finsupp.single_injective t h)
65           simp [hmt, hs]
66       · push_neg at hsub
67         rcases hsub with ⟨i, hi, hit⟩
68         have hmne : m ≠ Finsupp.single t N := by
69           intro hm
70           subst m
71           simp [Finsupp.single_apply, Ne.symm hit] at hi
72         rw [axisPolynomial, MvPolynomial.aeval_monomial]
73         have hz : m.prod
74             (fun i e => (if i = t then MvPolynomial.X ()
75               else (0 : MvPolynomial Unit k)) ^ e) = 0 := by
76           rw [Finsupp.prod]
77           apply Finset.prod_eq_zero hi
78           have hmi : m i ≠ 0 := Finsupp.mem_support_iff.mp hi
79           simp [hit, hmi]
80         rw [hz, mul_zero]
81         simp [MvPolynomial.coeff_monomial, hmne]
82   | add p q hp hq =>
83       change AddMonoidAlgebra.coeff (axisPolynomial k t (p + q))
84         (Finsupp.single () N) =
85         AddMonoidAlgebra.coeff (p + q) (Finsupp.single t N)
86       simp [map_add, hp, hq]

88 theorem coeff_principal_pure_eq_normalForm {n N : ℕ}
89     (t : PhaseVar n) (d : WeylAlg k n) :
90     MvPolynomial.coeff (Finsupp.single t N)
91         (presentedPrincipalComponent k (@bernsteinWeight n) N d) =
92     MvPolynomial.coeff (Finsupp.single t N)
93         (presentedNormalFormLinearEquiv k n d) := by
94   change AddMonoidAlgebra.coeff
95       (presentedPrincipalComponent k (@bernsteinWeight n) N d)
96       (Finsupp.single t N) =
97     AddMonoidAlgebra.coeff (presentedNormalFormLinearEquiv k n d)
98       (Finsupp.single t N)
99   rw [coeff_presentedPrincipalComponent]
100   simp [Finsupp.weight_apply, bernsteinWeight]

102 def IsPBWMonicAt {n : ℕ} (t : PhaseVar n) (N : ℕ)
103     (d : WeylAlg k n) : Prop :=
104   d ∈ bernsteinPiece k n N ∧
105     MvPolynomial.coeff (Finsupp.single t N)
106       (presentedNormalFormLinearEquiv k n d) = 1

108 theorem coeff_normalForm_eq_zero_of_exponent_gt {n N : ℕ}
109     {t : PhaseVar n} {d : WeylAlg k n}
110     (hd : d ∈ bernsteinPiece k n N) {m : PhaseVar n →₀ ℕ}
111     (hmt : N < m t) :
112     MvPolynomial.coeff m (presentedNormalFormLinearEquiv k n d) = 0 := by
113   by_contra hcoeff
114   have hweight := (mem_presentedWeightPiece k (@bernsteinWeight n) N d).mp
115     hd m hcoeff
116   have hle : m t ≤ Finsupp.weight (@bernsteinWeight n) m := by
117     simpa [Finsupp.weight_apply, bernsteinWeight] using
118       Finsupp.single_eval_le_sum m (g := id) rfl (fun _ => Nat.zero_le _) t
119   omega

121 structure NormalizedPBWChartData {n : ℕ} (t : PhaseVar n) (N : ℕ)
122     (d : WeylAlg k n) where
123   M : Matrix (PhaseVar n) (PhaseVar n) k
124   Ninv : Matrix (PhaseVar n) (PhaseVar n) k
125   c : k
126   hM : M * standardForm k n * Matrix.transpose M = standardForm k n
127   hNinv : Ninv * standardForm k n * Matrix.transpose Ninv = standardForm k n
128   hMN : M * Ninv = 1
129   hNM : Ninv * M = 1
130   hc : c ≠ 0
131   mem_piece : normalizedSymplecticImage k M hM c d ∈ bernsteinPiece k n N
132   pure_coeff : MvPolynomial.coeff (Finsupp.single t N)
133       (presentedNormalFormLinearEquiv k n
134         (normalizedSymplecticImage k M hM c d)) = 1

136 theorem HasNormalizedSymplecticChart.toNormalizedPBWChartData
137     {n N : ℕ} {t : PhaseVar n} {d : WeylAlg k n}
138     (hchart : HasNormalizedSymplecticChart k t N d) :
139     Nonempty (NormalizedPBWChartData k t N d) := by
140   rcases hchart with ⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc, hpiece, haxis⟩
141   refine ⟨⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc, hpiece, ?_⟩⟩
142   rw [← coeff_principal_pure_eq_normalForm k]
143   rw [← coeff_axisPolynomial k]
144   exact haxis

146 theorem HasNormalizedSymplecticChart.purePBWCoefficient
147     {n N : ℕ} {t : PhaseVar n} {d : WeylAlg k n}
148     (hchart : HasNormalizedSymplecticChart k t N d) :
149     ∃ d' : WeylAlg k n, IsPBWMonicAt k t N d' := by
150   rcases hchart with ⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc, hpiece, haxis⟩

152   rw [← coeff_principal_pure_eq_normalForm k]
153   rw [← coeff_axisPolynomial k]
154   exact haxis

186 theorem normalizedPBWChartData_fields {n N : ℕ} {t : PhaseVar n}
187     {d : WeylAlg k n} (D : NormalizedPBWChartData k t N d) :
188     D.M * standardForm k n * Matrix.transpose D.M = standardForm k n ∧
189       D.Ninv * standardForm k n * Matrix.transpose D.Ninv = standardForm k n ∧
190       D.M * D.Ninv = 1 ∧ D.Ninv * D.M = 1 ∧ D.c ≠ 0 ∧
191       normalizedSymplecticImage k D.M D.hM D.c d ∈ bernsteinPiece k n N ∧
192       MvPolynomial.coeff (Finsupp.single t N)
193         (presentedNormalFormLinearEquiv k n
194           (normalizedSymplecticImage k D.M D.hM D.c d)) = 1 :=
195   ⟨D.hM, D.hNinv, D.hMN, D.hNM, D.hc, D.mem_piece, D.pure_coeff⟩
