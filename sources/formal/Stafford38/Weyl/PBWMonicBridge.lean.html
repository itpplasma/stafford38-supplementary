7 /-!
8 # From principal-axis normalization to PBW monicity
9 
10 Axis restriction identifies the degree-`N` coefficient with the selected
11 pure-variable coefficient. The same coefficient passes from the principal
12 component to the checked PBW normal form; together with the Bernstein bound
13 this yields a coefficient-one pure momentum term and excludes higher
14 momentum powers. `OuterOreMonic` converts the result to `Polynomial.Monic`;
15 `PaperRightMonic` supplies the paper-facing right-coefficient decomposition
16 and lower-coefficient bounds.
17 -/

34 theorem coeff_axisPolynomial {n : ℕ} (t : PhaseVar n)
35     (P : SymbolRing k n) (N : ℕ) :
36     MvPolynomial.coeff (Finsupp.single () N) (axisPolynomial k t P) =
37       MvPolynomial.coeff (Finsupp.single t N) P := by
38   induction P using MvPolynomial.induction_on' with
39   | monomial m c =>
40       by_cases hsub : ∀ i ∈ m.support, i = t
41       · have hm : m = Finsupp.single t (m t) := by
42           ext i
43           by_cases hit : i = t
44           · subst i
45             simp
46           · have hi : i ∉ m.support := by
47               intro him
48               exact hit (hsub i him)
49             rw [Finsupp.notMem_support_iff.mp hi]
50             rw [Finsupp.single_apply]
51             simp [Ne.symm hit]
52         rw [hm]
53         rw [axisPolynomial, MvPolynomial.aeval_monomial]
54         rw [Finsupp.prod_single_index (by simp)]
55         simp [MvPolynomial.X_pow_eq_monomial]
56         by_cases hmt : m t = N
57         · simp [hmt]
58         · have hs : Finsupp.single t (m t) ≠ Finsupp.single t N := by
59             intro h
60             exact hmt (Finsupp.single_injective t h)
61           simp [hmt, hs]
62       · push_neg at hsub
63         rcases hsub with ⟨i, hi, hit⟩
64         have hmne : m ≠ Finsupp.single t N := by
65           intro hm
66           subst m
67           simp [Finsupp.single_apply, Ne.symm hit] at hi
68         rw [axisPolynomial, MvPolynomial.aeval_monomial]
69         have hz : m.prod
70             (fun i e => (if i = t then MvPolynomial.X ()
71               else (0 : MvPolynomial Unit k)) ^ e) = 0 := by
72           rw [Finsupp.prod]
73           apply Finset.prod_eq_zero hi
74           have hmi : m i ≠ 0 := Finsupp.mem_support_iff.mp hi
75           simp [hit, hmi]
76         rw [hz, mul_zero]
77         simp [MvPolynomial.coeff_monomial, hmne]
78   | add p q hp hq =>
79       change AddMonoidAlgebra.coeff (axisPolynomial k t (p + q))
80         (Finsupp.single () N) =
81         AddMonoidAlgebra.coeff (p + q) (Finsupp.single t N)
82       simp [map_add, hp, hq]

84 theorem coeff_principal_pure_eq_normalForm {n N : ℕ}
85     (t : PhaseVar n) (d : PresentedWeyl k n) :
86     MvPolynomial.coeff (Finsupp.single t N)
87         (presentedPrincipalComponent k (@bernsteinWeight n) N d) =
88     MvPolynomial.coeff (Finsupp.single t N)
89         (presentedNormalFormLinearEquiv k n d) := by
90   change AddMonoidAlgebra.coeff
91       (presentedPrincipalComponent k (@bernsteinWeight n) N d)
92       (Finsupp.single t N) =
93     AddMonoidAlgebra.coeff (presentedNormalFormLinearEquiv k n d)
94       (Finsupp.single t N)
95   rw [coeff_presentedPrincipalComponent]
96   simp [monomialWeight, bernsteinWeight]

98 def IsPBWMonicAt {n : ℕ} (t : PhaseVar n) (N : ℕ)
99     (d : PresentedWeyl k n) : Prop :=
100   d ∈ bernsteinPiece k n N ∧
101     MvPolynomial.coeff (Finsupp.single t N)
102       (presentedNormalFormLinearEquiv k n d) = 1

104 theorem coeff_normalForm_eq_zero_of_exponent_gt {n N : ℕ}
105     {t : PhaseVar n} {d : PresentedWeyl k n}
106     (hd : d ∈ bernsteinPiece k n N) {m : PhaseVar n →₀ ℕ}
107     (hmt : N < m t) :
108     MvPolynomial.coeff m (presentedNormalFormLinearEquiv k n d) = 0 := by
109   by_contra hcoeff
110   have hweight := (mem_presentedWeightPiece k (@bernsteinWeight n) N d).mp
111     hd m hcoeff
112   have hle : m t ≤ monomialWeight (@bernsteinWeight n) m := by
113     simpa [monomialWeight, bernsteinWeight] using
114       Finsupp.single_eval_le_sum m (g := id) rfl (fun _ => Nat.zero_le _) t
115   omega

117 structure NormalizedPBWChartData {n : ℕ} (t : PhaseVar n) (N : ℕ)
118     (d : PresentedWeyl k n) where
119   M : Matrix (PhaseVar n) (PhaseVar n) k
120   Ninv : Matrix (PhaseVar n) (PhaseVar n) k
121   c : k
122   hM : M * standardForm k n * Matrix.transpose M = standardForm k n
123   hNinv : Ninv * standardForm k n * Matrix.transpose Ninv = standardForm k n
124   hMN : M * Ninv = 1
125   hNM : Ninv * M = 1
126   hc : c ≠ 0
127   mem_piece : normalizedSymplecticImage k M hM c d ∈ bernsteinPiece k n N
128   pure_coeff : MvPolynomial.coeff (Finsupp.single t N)
129       (presentedNormalFormLinearEquiv k n
130         (normalizedSymplecticImage k M hM c d)) = 1

132 theorem HasNormalizedSymplecticChart.toNormalizedPBWChartData
133     {n N : ℕ} {t : PhaseVar n} {d : PresentedWeyl k n}
134     (hchart : HasNormalizedSymplecticChart k t N d) :
135     Nonempty (NormalizedPBWChartData k t N d) := by
136   rcases hchart with ⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc, hpiece, haxis⟩
137   refine ⟨⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc, hpiece, ?_⟩⟩
138   rw [← coeff_principal_pure_eq_normalForm k]
139   rw [← coeff_axisPolynomial k]
140   exact haxis

150   exact haxis

182 theorem normalizedPBWChartData_fields {n N : ℕ} {t : PhaseVar n}
183     {d : PresentedWeyl k n} (D : NormalizedPBWChartData k t N d) :
184     D.M * standardForm k n * Matrix.transpose D.M = standardForm k n ∧
185       D.Ninv * standardForm k n * Matrix.transpose D.Ninv = standardForm k n ∧
186       D.M * D.Ninv = 1 ∧ D.Ninv * D.M = 1 ∧ D.c ≠ 0 ∧
187       normalizedSymplecticImage k D.M D.hM D.c d ∈ bernsteinPiece k n N ∧
188       MvPolynomial.coeff (Finsupp.single t N)
189         (presentedNormalFormLinearEquiv k n
190           (normalizedSymplecticImage k D.M D.hM D.c d)) = 1 :=
191   ⟨D.hM, D.hNinv, D.hMN, D.hNM, D.hc, D.mem_piece, D.pure_coeff⟩
