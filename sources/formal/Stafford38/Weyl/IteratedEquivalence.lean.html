11 # The presented Weyl algebra maps to the iterated Ore construction

34 /-- The scalar algebra structure inherited recursively by the iterated Ore

37     (n : Nat) → Algebra k (IteratedPairStage k n)
38   | 0 => by

50 /-- The canonical successor embedding as a scalar-preserving algebra map. -/
51 def stageAlgHom (n : Nat) :
52     IteratedPairStage k n →ₐ[k] IteratedPairStage k (n + 1) where
53   toRingHom := stageEmbedding k n
54   commutes' c := by
55     change pairCoefficient (algebraMap k (IteratedPairStage k n) c) =
56       @algebraMap k (PairStage (B := IteratedPairStage k n)) _ _
57         (pairStageAlgebra (k := k) (B := IteratedPairStage k n)) c
58     rw [pairStageAlgebra_algebraMap
59       (k := k) (B := IteratedPairStage k n)]
60     rfl

62 /-- Coordinates in the final stage, ordered from newest to oldest. -/
63 def iteratedCoordinate :

65   | 0, i => Fin.elim0 i

67       Fin.cases (stageCoordinate k n)

106 /-- An element commuting with the image of every presented Weyl generator
107 commutes with the image of the whole presented algebra. -/
108 theorem commutes_freeWeyl_image_of_generators
109     {ι A : Type*} [Ring A] [Algebra k A]
110     (omega : Matrix ι ι k)
111     (f : FreeWeyl k ι omega →ₐ[k] A) (a : A)
112     (h : ∀ i, a * f (freeWeylGenerator omega i) =
113       f (freeWeylGenerator omega i) * a) :
114     ∀ w, a * f w = f w * a := by
115   intro w
116   rcases RingQuot.mkAlgHom_surjective k (freeWeylRelation omega) w with
117     ⟨q, rfl⟩
118   let g : FreeAlgebra k ι →ₐ[k] A :=
119     f.comp (RingQuot.mkAlgHom k (freeWeylRelation omega))
120   change a * g q = g q * a
121   induction q using FreeAlgebra.induction k ι with
122   | grade0 c =>
123       rw [g.commutes]
124       exact (Algebra.commutes c a).symm
125   | grade1 i =>
126       change a * f (freeWeylGenerator omega i) =
127         f (freeWeylGenerator omega i) * a
128       exact h i
129   | mul u v hu hv =>
130       rw [map_mul]
131       calc
132         a * (g u * g v) = (a * g u) * g v := by rw [mul_assoc]
133         _ = (g u * a) * g v := by rw [hu]
134         _ = g u * (a * g v) := by rw [mul_assoc]
135         _ = g u * (g v * a) := by rw [hv]
136         _ = (g u * g v) * a := by rw [mul_assoc]
137   | add u v hu hv =>
138       rw [map_add, mul_add, add_mul, hu, hv]

140 /-- All recursively adjoined coordinates commute. -/
141 theorem iteratedCoordinate_commutator (n : Nat) :
142     ∀ i j : Fin n,
143       commutator (iteratedCoordinate k n i) (iteratedCoordinate k n j) = 0 := by
144   induction n with
145   | zero =>
146       intro i
147       exact Fin.elim0 i
148   | succ n ih =>
149       intro i j
150       refine Fin.cases ?_ (fun i => ?_) i
151       · refine Fin.cases ?_ (fun j => ?_) j
152         · simp [AlgebraicAnalysis.ringCommutator]
153         · simp only [iteratedCoordinate_zero, iteratedCoordinate_succ]
154           exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _)
155       · refine Fin.cases ?_ (fun j => ?_) j
156         · simp only [iteratedCoordinate_zero, iteratedCoordinate_succ]
157           exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _).symm
158         · simp only [iteratedCoordinate_succ]
159           calc
160             commutator
161                 (stageAlgHom k n (iteratedCoordinate k n i))
162                 (stageAlgHom k n (iteratedCoordinate k n j)) =
163               stageAlgHom k n
164                 (commutator (iteratedCoordinate k n i)
165                   (iteratedCoordinate k n j)) :=
166                 (map_commutator (stageAlgHom k n).toRingHom _ _).symm
167             _ = 0 := by rw [ih i j, map_zero]

198 /-- Coordinate-momentum commutators are the negative Kronecker delta, in
199 the convention used by `Matrix.J`. -/
200 theorem iteratedCoordinate_momentum_commutator (n : Nat) :
201     ∀ i j : Fin n,
202       commutator (iteratedCoordinate k n i) (iteratedMomentum k n j) =
203         if i = j then -1 else 0 := by
204   induction n with
205   | zero =>
206       intro i
207       exact Fin.elim0 i
208   | succ n ih =>
209       intro i j
210       refine Fin.cases ?_ (fun i => ?_) i
211       · refine Fin.cases ?_ (fun j => ?_) j
212         · change stageCoordinate k n * stageMomentum k n -
213               stageMomentum k n * stageCoordinate k n = -1
214           rw [stageMomentum_mul_coordinate]
215           noncomm_ring
216         · simp only [iteratedCoordinate_zero, iteratedMomentum_succ]
217           exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _)
218       · refine Fin.cases ?_ (fun j => ?_) j
219         · simp only [iteratedCoordinate_succ, iteratedMomentum_zero]
220           simp only [Fin.succ_ne_zero, ↓reduceIte]
221           exact sub_eq_zero.mpr (stageMomentum_mul_embedding k n _).symm
222         · simp only [iteratedCoordinate_succ, iteratedMomentum_succ]
223           calc
224             commutator
225                 (stageAlgHom k n (iteratedCoordinate k n i))
226                 (stageAlgHom k n (iteratedMomentum k n j)) =
227               stageAlgHom k n
228                 (commutator (iteratedCoordinate k n i)
229                   (iteratedMomentum k n j)) :=
230                 (map_commutator (stageAlgHom k n).toRingHom _ _).symm
231             _ = if i.succ = j.succ then -1 else 0 := by
232               rw [ih i j]
233               by_cases h : i = j <;> simp [h]

235 /-- The combined generator family realizes the standard symplectic
236 commutator matrix. -/
237 theorem iteratedGenerator_commutator (n : Nat) :
238     ∀ a b : Fin n ⊕ Fin n,
239       commutator (iteratedGenerator k n a) (iteratedGenerator k n b) =
240         algebraMap k (IteratedPairStage k n) (Matrix.J (Fin n) k a b) := by
241   intro a b
242   cases a with
243   | inl i =>
244       cases b with
245       | inl j =>
246           change commutator (iteratedCoordinate k n i)
247               (iteratedCoordinate k n j) = _
248           rw [iteratedCoordinate_commutator]
249           simp [Matrix.J]
250       | inr j =>
251           change commutator (iteratedCoordinate k n i)
252               (iteratedMomentum k n j) = _
253           rw [iteratedCoordinate_momentum_commutator]
254           by_cases h : i = j <;> simp [Matrix.J, h]
255   | inr i =>
256       cases b with
257       | inl j =>
258           change commutator (iteratedMomentum k n i)
259               (iteratedCoordinate k n j) = _
260           rw [commutator_swap,
261             iteratedCoordinate_momentum_commutator]
262           by_cases h : j = i
263           · simp [Matrix.J, h]
264           · have hij : i ≠ j := fun e => h e.symm
265             simp [Matrix.J, h, hij]
266       | inr j =>
267           change commutator (iteratedMomentum k n i)
268               (iteratedMomentum k n j) = _
269           rw [iteratedMomentum_commutator]
270           simp [Matrix.J]

342 theorem presentedCoordinate_commutes_oldGenerator (n : Nat)
343     (i : Fin n ⊕ Fin n) :
344     presentedCoordinate k n * oldGenerator k n i =
345       oldGenerator k n i * presentedCoordinate k n := by
346   apply sub_eq_zero.mp
347   have h := freeWeylGenerator_commutator
348     (k := k) (Matrix.J (Fin (n + 1)) k)
349     (.inl (0 : Fin (n + 1))) (oldIndex i)
350   cases i with
351   | inl i =>
352       change Stafford.commutator
353         (freeWeylGenerator (Matrix.J (Fin (n + 1)) k) (.inl 0))
354         (freeWeylGenerator (Matrix.J (Fin (n + 1)) k) (.inl i.succ)) =
355           algebraMap k (PresentedWeyl k (n + 1)) 0 at h
356       change Stafford.commutator
357         (freeWeylGenerator (Matrix.J (Fin (n + 1)) k) (.inl 0))
358         (freeWeylGenerator (Matrix.J (Fin (n + 1)) k) (.inl i.succ)) = 0
359       rw [map_zero] at h
360       exact h
361   | inr i =>
362       have hi : (0 : Fin (n + 1)) ≠ i.succ := i.succ_ne_zero.symm
363       simp only [oldIndex, Stafford38FixedSourceChallenge.oldIndex] at h
364       have hJ : Matrix.J (Fin (n + 1)) k (.inl 0) (.inr i.succ) = 0 := by
365         simp [Matrix.J, hi]
366       rw [hJ] at h
367       change Stafford.commutator
368         (freeWeylGenerator (Matrix.J (Fin (n + 1)) k) (.inl 0))
369         (freeWeylGenerator (Matrix.J (Fin (n + 1)) k) (.inr i.succ)) = 0
370       rw [map_zero] at h
371       exact h

422 theorem presentedMomentum_mul_coordinate (n : Nat) :
423     presentedMomentum k n * presentedCoordinate k n =
424       presentedCoordinate k n * presentedMomentum k n + 1 := by
425   have h := freeWeylGenerator_commutator
426     (k := k) (Matrix.J (Fin (n + 1)) k)
427     (.inr (0 : Fin (n + 1))) (.inl (0 : Fin (n + 1)))
428   change Stafford.commutator (presentedMomentum k n)
429     (presentedCoordinate k n) = algebraMap k (PresentedWeyl k (n + 1)) 1 at h
430   rw [map_one] at h
431   change presentedMomentum k n * presentedCoordinate k n -
432     presentedCoordinate k n * presentedMomentum k n = 1 at h

434     presentedMomentum k n * presentedCoordinate k n =
435         (presentedMomentum k n * presentedCoordinate k n -
436           presentedCoordinate k n * presentedMomentum k n) +
437             presentedCoordinate k n * presentedMomentum k n := by noncomm_ring
438     _ = 1 + presentedCoordinate k n * presentedMomentum k n := by rw [h]

441 /-- The recursive map from the Ore tower back to the quotient presentation. -/
442 def iteratedToPresented :
443     (n : Nat) → IteratedPairStage k n →ₐ[k] PresentedWeyl k n
444   | 0 => Algebra.ofId k (PresentedWeyl k 0)
445   | n + 1 => by
446       letI : Algebra k (IteratedPairStage k n) :=
447         iteratedPairStageAlgebra k n
448       letI : Algebra k (PairStage (B := IteratedPairStage k n)) :=
449         pairStageAlgebra
450       change PairStage (B := IteratedPairStage k n) →ₐ[k]
451         PresentedWeyl k (n + 1)
452       let f : IteratedPairStage k n →ₐ[k] PresentedWeyl k (n + 1) :=
453         (previousWeylEmbedding k n).comp (iteratedToPresented n)
454       exact pairLiftAlgHom f (presentedCoordinate k n) (presentedMomentum k n)
455         (fun b => presentedCoordinate_commutes_previous k n
456           (iteratedToPresented n b))
457         (fun b => presentedMomentum_commutes_previous k n
458           (iteratedToPresented n b))
459         (presentedMomentum_mul_coordinate k n)

506   unfold presentedMomentum

510 /-- Mapping from the Ore tower to the presentation and back is the identity. -/
511 theorem presentedToIterated_comp_iteratedToPresented :
512     ∀ n : Nat,
513       (presentedToIterated k n).comp (iteratedToPresented k n) =
514         AlgHom.id k (IteratedPairStage k n) := by
515   intro n
516   induction n with
517   | zero =>
518       apply AlgHom.ext
519       intro c
520       change k at c
521       change presentedToIterated k 0
522         (algebraMap k (PresentedWeyl k 0) c) = c
523       rw [(presentedToIterated k 0).commutes]
524       rfl
525   | succ n ih =>
526       letI : Algebra k (IteratedPairStage k n) :=
527         iteratedPairStageAlgebra k n
528       letI : Algebra k (PairStage (B := IteratedPairStage k n)) :=
529         pairStageAlgebra
530       let f : IteratedPairStage k n →ₐ[k] IteratedPairStage k (n + 1) :=
531         stageAlgHom k n
532       let X : IteratedPairStage k (n + 1) := stageCoordinate k n
533       let P : IteratedPairStage k (n + 1) := stageMomentum k n
534       let g : IteratedPairStage k (n + 1) →ₐ[k]
535           IteratedPairStage k (n + 1) :=
536         (presentedToIterated k (n + 1)).comp
537           (iteratedToPresented k (n + 1))
538       have hgCoefficient : ∀ b, g (stageEmbedding k n b) = f b := by
539         intro b
540         dsimp [g, f]
541         rw [iteratedToPresented_coefficient]
542         have hp := DFunLike.congr_fun (presentedToIterated_previous k n)
543           (iteratedToPresented k n b)
544         calc
545           presentedToIterated k (n + 1)
546               (previousWeylEmbedding k n (iteratedToPresented k n b)) =
547               stageAlgHom k n
548                 (presentedToIterated k n (iteratedToPresented k n b)) := hp
549           _ = stageAlgHom k n b := by
550             have hib : presentedToIterated k n (iteratedToPresented k n b) = b := by
551               change ((presentedToIterated k n).comp
552                 (iteratedToPresented k n)) b = b
553               rw [ih]
554               rfl
555             rw [hib]
556       have hgCoordinate : g (stageCoordinate k n) = X := by
557         dsimp [g, X]
558         rw [iteratedToPresented_coordinate, presentedToIterated_coordinate]
559       have hgMomentum : g (stageMomentum k n) = P := by
560         dsimp [g, P]
561         rw [iteratedToPresented_momentum, presentedToIterated_momentum]
562       have hg : g = pairLiftAlgHom f X P
563           (stageCoordinate_mul_embedding k n)
564           (stageMomentum_mul_embedding k n)
565           (stageMomentum_mul_coordinate k n) :=
566         pairLiftAlgHom_unique f X P
567           (stageCoordinate_mul_embedding k n)
568           (stageMomentum_mul_embedding k n)
569           (stageMomentum_mul_coordinate k n) g
570           hgCoefficient hgCoordinate hgMomentum
571       have hid : AlgHom.id k (IteratedPairStage k (n + 1)) =
572           pairLiftAlgHom f X P
573             (stageCoordinate_mul_embedding k n)
574             (stageMomentum_mul_embedding k n)
575             (stageMomentum_mul_coordinate k n) :=
576         pairLiftAlgHom_unique f X P
577           (stageCoordinate_mul_embedding k n)
578           (stageMomentum_mul_embedding k n)
579           (stageMomentum_mul_coordinate k n)
580           (AlgHom.id k (IteratedPairStage k (n + 1)))
581           (fun _ => rfl) rfl rfl
582       exact hg.trans hid.symm

584 /-- The recursive inverse sends every iterated generator back to its named
585 generator in the quotient presentation. -/
586 theorem iteratedToPresented_generator :
587     ∀ (n : Nat) (i : Fin n ⊕ Fin n),
588       iteratedToPresented k n (iteratedGenerator k n i) =
589         freeWeylGenerator (Matrix.J (Fin n) k) i := by
590   intro n
591   induction n with
592   | zero =>
593       intro i
594       exact Sum.elim Fin.elim0 Fin.elim0 i
595   | succ n ih =>
596       intro i
597       cases i with
598       | inl i =>
599           refine Fin.cases ?_ (fun j => ?_) i
600           · change iteratedToPresented k (n + 1) (stageCoordinate k n) = _
601             rw [iteratedToPresented_coordinate]
602             rfl
603           · change iteratedToPresented k (n + 1)
604               (stageEmbedding k n (iteratedCoordinate k n j)) = _
605             rw [iteratedToPresented_coefficient]
606             change previousWeylEmbedding k n
607               (iteratedToPresented k n (iteratedGenerator k n (.inl j))) = _
608             rw [ih (.inl j), previousWeylEmbedding_generator]
609             rfl
610       | inr i =>
611           refine Fin.cases ?_ (fun j => ?_) i
612           · change iteratedToPresented k (n + 1) (stageMomentum k n) = _
613             rw [iteratedToPresented_momentum]
614             rfl
615           · change iteratedToPresented k (n + 1)
616               (stageEmbedding k n (iteratedMomentum k n j)) = _
617             rw [iteratedToPresented_coefficient]
618             change previousWeylEmbedding k n
619               (iteratedToPresented k n (iteratedGenerator k n (.inr j))) = _
620             rw [ih (.inr j), previousWeylEmbedding_generator]
621             rfl
