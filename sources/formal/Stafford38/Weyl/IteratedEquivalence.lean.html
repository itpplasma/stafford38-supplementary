54 /-- The canonical successor embedding as a scalar-preserving algebra map. -/
55 def stageAlgHom (n : Nat) :
56     IteratedPairStage k n →ₐ[k] IteratedPairStage k (n + 1) where
57   toRingHom := stageEmbedding k n
58   commutes' c := by
59     change pairCoefficient (algebraMap k (IteratedPairStage k n) c) =
60       @algebraMap k (PairStage (B := IteratedPairStage k n)) _ _
61         (pairStageAlgebra (k := k) (B := IteratedPairStage k n)) c

63       (k := k) (B := IteratedPairStage k n)]

110 /-- An element commuting with the image of every presented Weyl generator
111 commutes with the image of the whole presented algebra. -/
112 theorem commutes_freeWeyl_image_of_generators
113     {ι A : Type*} [Ring A] [Algebra k A]
114     (omega : Matrix ι ι k)
115     (f : FreeWeyl k ι omega →ₐ[k] A) (a : A)
116     (h : ∀ i, a * f (freeWeylGenerator omega i) =
117       f (freeWeylGenerator omega i) * a) :
118     ∀ w, a * f w = f w * a := by
119   intro w
120   rcases RingQuot.mkAlgHom_surjective k (freeWeylRelation omega) w with
121     ⟨q, rfl⟩
122   let g : FreeAlgebra k ι →ₐ[k] A :=
123     f.comp (RingQuot.mkAlgHom k (freeWeylRelation omega))
124   change a * g q = g q * a
125   induction q using FreeAlgebra.induction k ι with
126   | grade0 c =>
127       rw [g.commutes]
128       exact (Algebra.commutes c a).symm
129   | grade1 i =>
130       change a * f (freeWeylGenerator omega i) =
131         f (freeWeylGenerator omega i) * a
132       exact h i
133   | mul u v hu hv =>
134       rw [map_mul]
135       calc
136         a * (g u * g v) = (a * g u) * g v := by rw [mul_assoc]
137         _ = (g u * a) * g v := by rw [hu]
138         _ = g u * (a * g v) := by rw [mul_assoc]
139         _ = g u * (g v * a) := by rw [hv]
140         _ = (g u * g v) * a := by rw [mul_assoc]
141   | add u v hu hv =>
142       rw [map_add, mul_add, add_mul, hu, hv]

144 /-- All recursively adjoined coordinates commute. -/
145 theorem iteratedCoordinate_commutator (n : Nat) :
146     ∀ i j : Fin n,
147       ringCommutator (iteratedCoordinate k n i) (iteratedCoordinate k n j) = 0 := by
148   induction n with
149   | zero =>
150       intro i
151       exact Fin.elim0 i
152   | succ n ih =>
153       intro i j
154       refine Fin.cases ?_ (fun i => ?_) i
155       · refine Fin.cases ?_ (fun j => ?_) j
156         · simp [AlgebraicAnalysis.ringCommutator]
157         · simp only [iteratedCoordinate_zero, iteratedCoordinate_succ]
158           exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _)
159       · refine Fin.cases ?_ (fun j => ?_) j
160         · simp only [iteratedCoordinate_zero, iteratedCoordinate_succ]
161           exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _).symm
162         · simp only [iteratedCoordinate_succ]
163           calc
164             ringCommutator
165                 (stageAlgHom k n (iteratedCoordinate k n i))
166                 (stageAlgHom k n (iteratedCoordinate k n j)) =
167               stageAlgHom k n
168                 (ringCommutator (iteratedCoordinate k n i)
169                   (iteratedCoordinate k n j)) :=
170                 (map_commutator (stageAlgHom k n).toRingHom _ _).symm
171             _ = 0 := by rw [ih i j, map_zero]

202 /-- Coordinate-momentum commutators are the negative Kronecker delta, in
203 the convention used by `Matrix.J`. -/
204 theorem iteratedCoordinate_momentum_commutator (n : Nat) :
205     ∀ i j : Fin n,
206       ringCommutator (iteratedCoordinate k n i) (iteratedMomentum k n j) =
207         if i = j then -1 else 0 := by
208   induction n with
209   | zero =>
210       intro i
211       exact Fin.elim0 i
212   | succ n ih =>
213       intro i j
214       refine Fin.cases ?_ (fun i => ?_) i
215       · refine Fin.cases ?_ (fun j => ?_) j
216         · change stageCoordinate k n * stageMomentum k n -
217               stageMomentum k n * stageCoordinate k n = -1
218           rw [stageMomentum_mul_coordinate]
219           noncomm_ring
220         · simp only [iteratedCoordinate_zero, iteratedMomentum_succ]
221           exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _)
222       · refine Fin.cases ?_ (fun j => ?_) j
223         · simp only [iteratedCoordinate_succ, iteratedMomentum_zero]
224           simp only [Fin.succ_ne_zero, ↓reduceIte]
225           exact sub_eq_zero.mpr (stageMomentum_mul_embedding k n _).symm
226         · simp only [iteratedCoordinate_succ, iteratedMomentum_succ]
227           calc
228             ringCommutator
229                 (stageAlgHom k n (iteratedCoordinate k n i))
230                 (stageAlgHom k n (iteratedMomentum k n j)) =
231               stageAlgHom k n
232                 (ringCommutator (iteratedCoordinate k n i)
233                   (iteratedMomentum k n j)) :=
234                 (map_commutator (stageAlgHom k n).toRingHom _ _).symm
235             _ = if i.succ = j.succ then -1 else 0 := by
236               rw [ih i j]
237               by_cases h : i = j <;> simp [h]

239 /-- The combined generator family realizes the standard symplectic
240 ringCommutator matrix. -/
241 theorem iteratedGenerator_commutator (n : Nat) :
242     ∀ a b : Fin n ⊕ Fin n,
243       ringCommutator (iteratedGenerator k n a) (iteratedGenerator k n b) =
244         algebraMap k (IteratedPairStage k n) (Matrix.J (Fin n) k a b) := by
245   intro a b
246   cases a with
247   | inl i =>
248       cases b with
249       | inl j =>
250           change ringCommutator (iteratedCoordinate k n i)
251               (iteratedCoordinate k n j) = _
252           rw [iteratedCoordinate_commutator]
253           simp [Matrix.J]
254       | inr j =>
255           change ringCommutator (iteratedCoordinate k n i)
256               (iteratedMomentum k n j) = _
257           rw [iteratedCoordinate_momentum_commutator]
258           by_cases h : i = j <;> simp [Matrix.J, h]
259   | inr i =>
260       cases b with
261       | inl j =>
262           change ringCommutator (iteratedMomentum k n i)
263               (iteratedCoordinate k n j) = _
264           rw [commutator_swap,
265             iteratedCoordinate_momentum_commutator]
266           by_cases h : j = i
267           · simp [Matrix.J, h]
268           · have hij : i ≠ j := fun e => h e.symm
269             simp [Matrix.J, h, hij]
270       | inr j =>
271           change ringCommutator (iteratedMomentum k n i)
272               (iteratedMomentum k n j) = _
273           rw [iteratedMomentum_commutator]
274           simp [Matrix.J]

419     presentedMomentum k n * presentedCoordinate k n =

437 /-- The recursive map from the Ore tower back to the quotient presentation. -/
438 def iteratedToPresented :
439     (n : Nat) → IteratedPairStage k n →ₐ[k] WeylAlg k n
440   | 0 => Algebra.ofId k (WeylAlg k 0)
441   | n + 1 => by
442       letI : Algebra k (IteratedPairStage k n) :=
443         iteratedPairStageAlgebra k n
444       letI : Algebra k (PairStage (B := IteratedPairStage k n)) :=
445         pairStageAlgebra
446       change PairStage (B := IteratedPairStage k n) →ₐ[k]
447         WeylAlg k (n + 1)
448       let f : IteratedPairStage k n →ₐ[k] WeylAlg k (n + 1) :=
449         (previousWeylEmbedding k n).comp (iteratedToPresented n)
450       exact pairLiftAlgHom f (presentedCoordinate k n) (presentedMomentum k n)
451         (fun b => presentedCoordinate_commutes_previous k n
452           (iteratedToPresented n b))
453         (fun b => presentedMomentum_commutes_previous k n
454           (iteratedToPresented n b))
455         (presentedMomentum_mul_coordinate k n)

507 theorem presentedToIterated_comp_iteratedToPresented :
508     ∀ n : Nat,
509       (presentedToIterated k n).comp (iteratedToPresented k n) =
510         AlgHom.id k (IteratedPairStage k n) := by
511   intro n
512   induction n with
513   | zero =>
514       apply AlgHom.ext
515       intro c
516       change k at c
517       change presentedToIterated k 0
518         (algebraMap k (WeylAlg k 0) c) = c
519       rw [(presentedToIterated k 0).commutes]
520       rfl
521   | succ n ih =>
522       letI : Algebra k (IteratedPairStage k n) :=
523         iteratedPairStageAlgebra k n
524       letI : Algebra k (PairStage (B := IteratedPairStage k n)) :=
525         pairStageAlgebra
526       let f : IteratedPairStage k n →ₐ[k] IteratedPairStage k (n + 1) :=
527         stageAlgHom k n
528       let X : IteratedPairStage k (n + 1) := stageCoordinate k n
529       let P : IteratedPairStage k (n + 1) := stageMomentum k n
530       let g : IteratedPairStage k (n + 1) →ₐ[k]
531           IteratedPairStage k (n + 1) :=
532         (presentedToIterated k (n + 1)).comp
533           (iteratedToPresented k (n + 1))
534       have hgCoefficient : ∀ b, g (stageEmbedding k n b) = f b := by
535         intro b
536         dsimp [g, f]
537         rw [iteratedToPresented_coefficient]
538         have hp := DFunLike.congr_fun (presentedToIterated_previous k n)
539           (iteratedToPresented k n b)
540         calc
541           presentedToIterated k (n + 1)
542               (previousWeylEmbedding k n (iteratedToPresented k n b)) =
543               stageAlgHom k n
544                 (presentedToIterated k n (iteratedToPresented k n b)) := hp
545           _ = stageAlgHom k n b := by
546             have hib : presentedToIterated k n (iteratedToPresented k n b) = b := by
547               change ((presentedToIterated k n).comp
548                 (iteratedToPresented k n)) b = b
549               rw [ih]
550               rfl
551             rw [hib]
552       have hgCoordinate : g (stageCoordinate k n) = X := by
553         dsimp [g, X]
554         rw [iteratedToPresented_coordinate, presentedToIterated_coordinate]
555       have hgMomentum : g (stageMomentum k n) = P := by
556         dsimp [g, P]
557         rw [iteratedToPresented_momentum, presentedToIterated_momentum]
558       have hg : g = pairLiftAlgHom f X P
559           (stageCoordinate_mul_embedding k n)
560           (stageMomentum_mul_embedding k n)
561           (stageMomentum_mul_coordinate k n) :=
562         pairLiftAlgHom_unique f X P
563           (stageCoordinate_mul_embedding k n)
564           (stageMomentum_mul_embedding k n)
565           (stageMomentum_mul_coordinate k n) g
566           hgCoefficient hgCoordinate hgMomentum
567       have hid : AlgHom.id k (IteratedPairStage k (n + 1)) =
568           pairLiftAlgHom f X P
569             (stageCoordinate_mul_embedding k n)
570             (stageMomentum_mul_embedding k n)
571             (stageMomentum_mul_coordinate k n) :=
572         pairLiftAlgHom_unique f X P
573           (stageCoordinate_mul_embedding k n)
574           (stageMomentum_mul_embedding k n)
575           (stageMomentum_mul_coordinate k n)
576           (AlgHom.id k (IteratedPairStage k (n + 1)))
577           (fun _ => rfl) rfl rfl
578       exact hg.trans hid.symm

580 /-- The recursive inverse sends every iterated generator back to its named
581 generator in the quotient presentation. -/
582 theorem iteratedToPresented_generator :
583     ∀ (n : Nat) (i : Fin n ⊕ Fin n),
584       iteratedToPresented k n (iteratedGenerator k n i) =
585         freeWeylGenerator (Matrix.J (Fin n) k) i := by
586   intro n
587   induction n with
588   | zero =>
589       intro i
590       exact Sum.elim Fin.elim0 Fin.elim0 i
591   | succ n ih =>
592       intro i
593       cases i with
594       | inl i =>
595           refine Fin.cases ?_ (fun j => ?_) i
596           · change iteratedToPresented k (n + 1) (stageCoordinate k n) = _
597             rw [iteratedToPresented_coordinate]
598             rfl
599           · change iteratedToPresented k (n + 1)
600               (stageEmbedding k n (iteratedCoordinate k n j)) = _
601             rw [iteratedToPresented_coefficient]
602             change previousWeylEmbedding k n
603               (iteratedToPresented k n (iteratedGenerator k n (.inl j))) = _
604             rw [ih (.inl j), previousWeylEmbedding_generator]
605             rfl
606       | inr i =>
607           refine Fin.cases ?_ (fun j => ?_) i
608           · change iteratedToPresented k (n + 1) (stageMomentum k n) = _
609             rw [iteratedToPresented_momentum]
610             rfl
611           · change iteratedToPresented k (n + 1)
612               (stageEmbedding k n (iteratedMomentum k n j)) = _
613             rw [iteratedToPresented_coefficient]
614             change previousWeylEmbedding k n
615               (iteratedToPresented k n (iteratedGenerator k n (.inr j))) = _
616             rw [ih (.inr j), previousWeylEmbedding_generator]
617             rfl
