33 private theorem weightedHomogeneousComponent_monomial_mul_eq_zero_of_lt
34     {n T : ℕ} (w : PhaseVar n → ℕ) (s : PhaseVar n →₀ ℕ)
35     (f : SymbolRing k n) (hT : T < Finsupp.weight w s) :
36     MvPolynomial.weightedHomogeneousComponent w T
37         (MvPolynomial.monomial s 1 * f) = 0 := by
38   classical
39   ext q
40   change AddMonoidAlgebra.coeff
41     (MvPolynomial.weightedHomogeneousComponent w T
42       (MvPolynomial.monomial s 1 * f)) q = 0
43   rw [MvPolynomial.coeff_weightedHomogeneousComponent,
44     MvPolynomial.coeff_monomial_mul']
45   by_cases hsq : s ≤ q
46   · rw [if_pos hsq, one_mul]
47     have hsplit : s + (q - s) = q := by
48       rw [add_comm]
49       exact tsub_add_cancel_of_le hsq
50     have hweight :
51         Finsupp.weight w q =
52           Finsupp.weight w s + Finsupp.weight w (q - s) := by
53       rw [← hsplit]
54       simp [Finsupp.weight_apply, Finsupp.sum_add_index, add_mul]
55     have hne : Finsupp.weight w q ≠ T := by omega
56     rw [if_neg hne]
57   · rw [if_neg hsq]

226 /-- For ordered monomials of positive total differential order, the component
227 one below the top order of their product is exactly the one-contraction term. -/
228 theorem presentedOrderedMonomial_mul_firstContraction :
229     ∀ (n : ℕ) (a p c q : Fin n → ℕ),
230       0 < Finsupp.weight (@orderWeight n) (phaseExponent a p) +
231           Finsupp.weight (@orderWeight n) (phaseExponent c q) →
232       presentedPrincipalComponent k (@orderWeight n)
233           (Finsupp.weight (@orderWeight n) (phaseExponent a p) +
234             Finsupp.weight (@orderWeight n) (phaseExponent c q) - 1)
235           (presentedOrderedMonomial k n a p *
236             presentedOrderedMonomial k n c q) =
237         pbwFirstContraction k (phaseExponent a p) (phaseExponent c q) := by
238   intro n
239   induction n with
240   | zero =>
241       intro a p c q hpos
242       have hap : phaseExponent a p = 0 := by
243         ext i
244         exact Sum.elim Fin.elim0 Fin.elim0 i
245       have hcq : phaseExponent c q = 0 := by
246         ext i
247         exact Sum.elim Fin.elim0 Fin.elim0 i
248       rw [hap, hcq] at hpos
249       simp [Finsupp.weight_apply] at hpos
250   | succ n ih =>
251       intro a p c q hpos
252       let mold :=
253         phaseExponent (fun i => a i.succ) (fun i => p i.succ)
254       let qold :=
255         phaseExponent (fun i => c i.succ) (fun i => q i.succ)
256       let z :=
257         presentedOrderedMonomial k n (fun i => a i.succ) (fun i => p i.succ) *
258           presentedOrderedMonomial k n (fun i => c i.succ) (fun i => q i.succ)
259       let L := Finsupp.weight (@orderWeight n) mold +
260         Finsupp.weight (@orderWeight n) qold
261       let t := p 0 + q 0
262       have htarget :
263           Finsupp.weight (@orderWeight (n + 1)) (phaseExponent a p) +
264               Finsupp.weight (@orderWeight (n + 1)) (phaseExponent c q) =
265             L + t := by
266         rw [monomialWeight_phaseExponent_succ_order,
267           monomialWeight_phaseExponent_succ_order]
268         dsimp [L, t, mold, qold]
269         omega
270       have htotal : 0 < L + t := by
271         rwa [htarget] at hpos
272       have hzOrder : z ∈ orderPiece k n L := by
273         dsimp [z, L, mold, qold]
274         exact presentedOrderedMonomial_mul_mem_orderPiece k n
275           (fun i => a i.succ) (fun i => p i.succ)
276           (fun i => c i.succ) (fun i => q i.succ)
277       have hzeroTerm :
278           presentedPrincipalComponent k (@orderWeight (n + 1)) (L + t - 1)
279               (presentedCoefficientOrdered k n (a 0 + c 0) t z) =
280             MvPolynomial.monomial
281                 (extendPhaseExponent n (a 0 + c 0) t 0) 1 *
282               MvPolynomial.rename Stafford38FixedSourceChallenge.oldIndex
283                 (pbwFirstContraction k mold qold) := by
284         by_cases hL : L = 0
285         · have ht : 0 < t := by omega
286           have hvanish :
287               presentedPrincipalComponent k (@orderWeight (n + 1))
288                   (L + t - 1)
289                   (presentedCoefficientOrdered k n (a 0 + c 0) t z) = 0 := by
290             rw [presentedPrincipalComponent, LinearMap.comp_apply]
291             change MvPolynomial.weightedHomogeneousComponent
292                 (@orderWeight (n + 1)) (L + t - 1)
293                 (presentedNormalFormLinearEquiv k (n + 1)
294                   (presentedCoefficientOrdered k n (a 0 + c 0) t z)) = 0
295             rw [presentedNormalFormLinearEquiv_previous_ordered_monomial]
296             apply weightedHomogeneousComponent_monomial_mul_eq_zero_of_lt k
297             rw [monomialWeight_extend_zero_order]
298             omega
299           have hmold : Finsupp.weight (@orderWeight n) mold = 0 := by
300             dsimp [L] at hL
301             omega
302           rw [hvanish,
303             pbwFirstContraction_eq_zero_of_orderWeight_eq_zero
304               k mold qold hmold, map_zero, mul_zero]
305         · have hLpos : 0 < L := Nat.pos_of_ne_zero hL
306           have hih := ih
307             (fun i => a i.succ) (fun i => p i.succ)
308             (fun i => c i.succ) (fun i => q i.succ) hLpos
309           change presentedPrincipalComponent k (@orderWeight n) (L - 1) z =
310               pbwFirstContraction k mold qold at hih
311           rw [show L + t - 1 = (L - 1) + t by omega,
312             presentedPrincipalComponent_coefficientOrdered_order, hih]
313       rw [htarget]
314       change presentedPrincipalComponent k (@orderWeight (n + 1)) (L + t - 1)
315           (presentedCoefficientOrdered k n (a 0) (p 0)
316               (presentedOrderedMonomial k n
317                 (fun i => a i.succ) (fun i => p i.succ)) *
318             presentedCoefficientOrdered k n (c 0) (q 0)
319               (presentedOrderedMonomial k n
320                 (fun i => c i.succ) (fun i => q i.succ))) = _
321       rw [presentedCoefficientOrdered_mul]
322       simp only [map_sum]
323       have hmem0 : 0 ∈ Finset.range (p 0 + 1) := by simp
324       rw [Finset.sum_eq_add_sum_sdiff_singleton_of_mem hmem0]
325       simp only [Nat.zero_le, if_true, Nat.choose_zero_right,
326         Nat.descFactorial_zero, mul_one, one_nsmul, Nat.sub_zero]
327       change presentedPrincipalComponent k (@orderWeight (n + 1)) (L + t - 1)
328             (presentedCoefficientOrdered k n (a 0 + c 0) t z) + _ = _
329       rw [hzeroTerm]
330       rw [pbwFirstContraction_phaseExponent_succ]
331       by_cases hp0 : p 0 = 0
332       · have hempty : Finset.range (p 0 + 1) \ {0} = ∅ := by
333           ext i
334           simp [hp0]
335         rw [hempty]
336         simp [MvPolynomial.pderiv_monomial, phaseExponent, hp0,
337           t, mold, qold]
338       · have hp : 0 < p 0 := Nat.pos_of_ne_zero hp0
339         by_cases hc0 : c 0 = 0
340         · have hrest :
341               ∑ i ∈ Finset.range (p 0 + 1) \ {0},
342                   presentedPrincipalComponent k (@orderWeight (n + 1))
343                     (L + t - 1)
344                     (if i ≤ c 0 then
345                       ((p 0).choose i * (c 0).descFactorial i) •
346                         presentedCoefficientOrdered k n
347                           (a 0 + c 0 - i) (p 0 + q 0 - i) z
348                     else 0) = 0 := by
349             apply Finset.sum_eq_zero
350             intro i hi
351             have hi' := Finset.mem_sdiff.mp hi
352             have hi0 : i ≠ 0 := by simpa using hi'.2
353             have hnot : ¬ i ≤ c 0 := by omega
354             simp [hnot]
355           rw [hrest]
356           simp [MvPolynomial.pderiv_monomial, phaseExponent, hc0,
357             t, mold, qold]
358         · have hc : 0 < c 0 := Nat.pos_of_ne_zero hc0
359           have honeMem : 1 ∈ Finset.range (p 0 + 1) \ {0} := by
360             simp [hp]
361           have hone :
362               presentedPrincipalComponent k (@orderWeight (n + 1))
363                 (L + t - 1)
364                 (((p 0).choose 1 * (c 0).descFactorial 1) •
365                   presentedCoefficientOrdered k n
366                     (a 0 + c 0 - 1) (p 0 + q 0 - 1) z) =
367               MvPolynomial.pderiv (.inr (0 : Fin (n + 1)))
368                     (MvPolynomial.monomial (phaseExponent a p) 1) *
369                 MvPolynomial.pderiv (.inl (0 : Fin (n + 1)))
370                     (MvPolynomial.monomial (phaseExponent c q) 1) := by
371             simp only [map_nsmul]
372             rw [show L + t - 1 = L + (p 0 + q 0 - 1) by omega,
373               presentedPrincipalComponent_coefficientOrdered_order]
374             have htop := presentedOrderedMonomial_mul_principal_order k n
375               (fun i => a i.succ) (fun i => p i.succ)
376               (fun i => c i.succ) (fun i => q i.succ)
377             change presentedPrincipalComponent k (@orderWeight n) L z =
378                 MvPolynomial.monomial mold 1 *
379                   MvPolynomial.monomial qold 1 at htop
380             rw [htop]
381             exact firstContraction_newest_term k n a p c q hp hc
382           rw [Finset.sum_eq_single 1]
383           · simp only [show 1 ≤ c 0 from hc, if_true]
384             rw [hone]
385             abel
386           · intro i hi hi1
387             have hi' := Finset.mem_sdiff.mp hi
388             have hirange : i ∈ Finset.range (p 0 + 1) := hi'.1
389             have hi0 : i ≠ 0 := by simpa using hi'.2
390             have hip : i ≤ p 0 := by
391               have := Finset.mem_range.mp hirange
392               omega
393             by_cases hic : i ≤ c 0
394             · simp only [if_pos hic, map_nsmul]
395               have hterm := presentedCoefficientOrdered_mem_orderPiece k n L
396                 (a 0 + c 0 - i) (p 0 + q 0 - i) z hzOrder
397               have hlt : L + (p 0 + q 0 - i) < L + t - 1 := by
398                 dsimp [t]
399                 omega
400               have hvanish :=
401                 presentedPrincipalComponent_eq_zero_of_mem_of_lt k
402                   (@orderWeight (n + 1)) _ hterm hlt
403               rw [hvanish, nsmul_zero]
404             · simp [hic]
405           · exact fun h => (h honeMem).elim

407 /-- The exact first-contraction coefficient for arbitrary PBW basis vectors.
408 The positivity hypothesis says that the two monomials have positive combined
409 differential order, so the component one below the top is unambiguous. -/
410 theorem presentedPBWBasis_mul_firstContraction {n : ℕ}
411     (m q : PhaseVar n →₀ ℕ)
412     (hpos : 0 < Finsupp.weight (@orderWeight n) m +
413       Finsupp.weight (@orderWeight n) q) :
414     presentedPrincipalComponent k (@orderWeight n)
415         (Finsupp.weight (@orderWeight n) m +
416           Finsupp.weight (@orderWeight n) q - 1)
417         (presentedPBWBasis k n m * presentedPBWBasis k n q) =
418       pbwFirstContraction k m q := by
419   have hpos' :
420       0 < Finsupp.weight (@orderWeight n)
421             (phaseExponent (fun i => m (.inl i)) (fun i => m (.inr i))) +
422           Finsupp.weight (@orderWeight n)
423             (phaseExponent (fun i => q (.inl i)) (fun i => q (.inr i))) := by
424     simpa only [phaseExponent_split] using hpos
425   rw [presentedPBWBasis_apply, presentedPBWBasis_apply,
426     ← phaseExponent_split m, ← phaseExponent_split q]
427   exact presentedOrderedMonomial_mul_firstContraction k n
428     (fun i => m (.inl i)) (fun i => m (.inr i))
429     (fun i => q (.inl i)) (fun i => q (.inr i)) hpos'

431 /-- The unconditional two-element subprincipal ringCommutator formula. -/
432 theorem principalComponent_commutator_eq_neg_poisson
433     {n r s : ℕ} {a b : WeylAlg k n}
434     (ha : a ∈ orderPiece k n r) (hb : b ∈ orderPiece k n s) :
435     presentedPrincipalComponent k (@orderWeight n) (r + s - 1)
436         (AlgebraicAnalysis.ringCommutator a b) =
437       -poissonBracket
