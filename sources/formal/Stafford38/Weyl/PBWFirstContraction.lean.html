31 private theorem weightedHomogeneousComponent_monomial_mul_eq_zero_of_lt
32     {n T : ℕ} (w : PhaseVar n → ℕ) (s : PhaseVar n →₀ ℕ)
33     (f : SymbolRing k n) (hT : T < monomialWeight w s) :
34     MvPolynomial.weightedHomogeneousComponent w T
35         (MvPolynomial.monomial s 1 * f) = 0 := by
36   classical
37   ext q
38   change AddMonoidAlgebra.coeff
39     (MvPolynomial.weightedHomogeneousComponent w T
40       (MvPolynomial.monomial s 1 * f)) q = 0
41   rw [MvPolynomial.coeff_weightedHomogeneousComponent,
42     MvPolynomial.coeff_monomial_mul']
43   by_cases hsq : s ≤ q
44   · rw [if_pos hsq, one_mul]
45     have hsplit : s + (q - s) = q := by
46       rw [add_comm]
47       exact tsub_add_cancel_of_le hsq
48     have hweight :
49         monomialWeight w q =
50           monomialWeight w s + monomialWeight w (q - s) := by
51       rw [← hsplit]
52       simp [monomialWeight, Finsupp.sum_add_index, add_mul]
53     have hne : monomialWeight w q ≠ T := by omega
54     rw [finsupp_weight_eq_monomialWeight, if_neg hne]
55   · rw [if_neg hsq]
56     simp

224 /-- For ordered monomials of positive total differential order, the component
225 one below the top order of their product is exactly the one-contraction term. -/
226 theorem presentedOrderedMonomial_mul_firstContraction :
227     ∀ (n : ℕ) (a p c q : Fin n → ℕ),
228       0 < monomialWeight (@orderWeight n) (phaseExponent a p) +
229           monomialWeight (@orderWeight n) (phaseExponent c q) →
230       presentedPrincipalComponent k (@orderWeight n)
231           (monomialWeight (@orderWeight n) (phaseExponent a p) +
232             monomialWeight (@orderWeight n) (phaseExponent c q) - 1)
233           (presentedOrderedMonomial k n a p *
234             presentedOrderedMonomial k n c q) =
235         pbwFirstContraction k (phaseExponent a p) (phaseExponent c q) := by
236   intro n
237   induction n with
238   | zero =>
239       intro a p c q hpos
240       have hap : phaseExponent a p = 0 := by
241         ext i
242         exact Sum.elim Fin.elim0 Fin.elim0 i
243       have hcq : phaseExponent c q = 0 := by
244         ext i
245         exact Sum.elim Fin.elim0 Fin.elim0 i
246       rw [hap, hcq] at hpos
247       simp [monomialWeight] at hpos
248   | succ n ih =>
249       intro a p c q hpos
250       let mold :=
251         phaseExponent (fun i => a i.succ) (fun i => p i.succ)
252       let qold :=
253         phaseExponent (fun i => c i.succ) (fun i => q i.succ)
254       let z :=
255         presentedOrderedMonomial k n (fun i => a i.succ) (fun i => p i.succ) *
256           presentedOrderedMonomial k n (fun i => c i.succ) (fun i => q i.succ)
257       let L := monomialWeight (@orderWeight n) mold +
258         monomialWeight (@orderWeight n) qold
259       let t := p 0 + q 0
260       have htarget :
261           monomialWeight (@orderWeight (n + 1)) (phaseExponent a p) +
262               monomialWeight (@orderWeight (n + 1)) (phaseExponent c q) =
263             L + t := by
264         rw [monomialWeight_phaseExponent_succ_order,
265           monomialWeight_phaseExponent_succ_order]
266         dsimp [L, t, mold, qold]
267         omega
268       have htotal : 0 < L + t := by
269         rwa [htarget] at hpos
270       have hzOrder : z ∈ orderPiece k n L := by
271         dsimp [z, L, mold, qold]
272         exact presentedOrderedMonomial_mul_mem_orderPiece k n
273           (fun i => a i.succ) (fun i => p i.succ)
274           (fun i => c i.succ) (fun i => q i.succ)
275       have hzeroTerm :
276           presentedPrincipalComponent k (@orderWeight (n + 1)) (L + t - 1)
277               (presentedCoefficientOrdered k n (a 0 + c 0) t z) =
278             MvPolynomial.monomial
279                 (extendPhaseExponent n (a 0 + c 0) t 0) 1 *
280               MvPolynomial.rename oldIndex
281                 (pbwFirstContraction k mold qold) := by
282         by_cases hL : L = 0
283         · have ht : 0 < t := by omega
284           have hvanish :
285               presentedPrincipalComponent k (@orderWeight (n + 1))
286                   (L + t - 1)
287                   (presentedCoefficientOrdered k n (a 0 + c 0) t z) = 0 := by
288             rw [presentedPrincipalComponent, LinearMap.comp_apply]
289             change MvPolynomial.weightedHomogeneousComponent
290                 (@orderWeight (n + 1)) (L + t - 1)
291                 (presentedNormalFormLinearEquiv k (n + 1)
292                   (presentedCoefficientOrdered k n (a 0 + c 0) t z)) = 0
293             rw [presentedNormalFormLinearEquiv_previous_ordered_monomial]
294             apply weightedHomogeneousComponent_monomial_mul_eq_zero_of_lt k
295             rw [monomialWeight_extend_zero_order]
296             omega
297           have hmold : monomialWeight (@orderWeight n) mold = 0 := by
298             dsimp [L] at hL
299             omega
300           rw [hvanish,
301             pbwFirstContraction_eq_zero_of_orderWeight_eq_zero
302               k mold qold hmold, map_zero, mul_zero]
303         · have hLpos : 0 < L := Nat.pos_of_ne_zero hL
304           have hih := ih
305             (fun i => a i.succ) (fun i => p i.succ)
306             (fun i => c i.succ) (fun i => q i.succ) hLpos
307           change presentedPrincipalComponent k (@orderWeight n) (L - 1) z =
308               pbwFirstContraction k mold qold at hih
309           rw [show L + t - 1 = (L - 1) + t by omega,
310             presentedPrincipalComponent_coefficientOrdered_order, hih]
311       rw [htarget]
312       change presentedPrincipalComponent k (@orderWeight (n + 1)) (L + t - 1)
313           (presentedCoefficientOrdered k n (a 0) (p 0)
314               (presentedOrderedMonomial k n
315                 (fun i => a i.succ) (fun i => p i.succ)) *
316             presentedCoefficientOrdered k n (c 0) (q 0)
317               (presentedOrderedMonomial k n
318                 (fun i => c i.succ) (fun i => q i.succ))) = _
319       rw [presentedCoefficientOrdered_mul]
320       simp only [map_sum]
321       have hmem0 : 0 ∈ Finset.range (p 0 + 1) := by simp
322       rw [Finset.sum_eq_add_sum_sdiff_singleton_of_mem hmem0]
323       simp only [Nat.zero_le, if_true, Nat.choose_zero_right,
324         Nat.descFactorial_zero, mul_one, one_nsmul, Nat.sub_zero]
325       change presentedPrincipalComponent k (@orderWeight (n + 1)) (L + t - 1)
326             (presentedCoefficientOrdered k n (a 0 + c 0) t z) + _ = _
327       rw [hzeroTerm]
328       rw [pbwFirstContraction_phaseExponent_succ]
329       by_cases hp0 : p 0 = 0
330       · have hempty : Finset.range (p 0 + 1) \ {0} = ∅ := by
331           ext i
332           simp [hp0]
333         rw [hempty]
334         simp [MvPolynomial.pderiv_monomial, phaseExponent, hp0,
335           t, mold, qold]
336       · have hp : 0 < p 0 := Nat.pos_of_ne_zero hp0
337         by_cases hc0 : c 0 = 0
338         · have hrest :
339               ∑ i ∈ Finset.range (p 0 + 1) \ {0},
340                   presentedPrincipalComponent k (@orderWeight (n + 1))
341                     (L + t - 1)
342                     (if i ≤ c 0 then
343                       ((p 0).choose i * (c 0).descFactorial i) •
344                         presentedCoefficientOrdered k n
345                           (a 0 + c 0 - i) (p 0 + q 0 - i) z
346                     else 0) = 0 := by
347             apply Finset.sum_eq_zero
348             intro i hi
349             have hi' := Finset.mem_sdiff.mp hi
350             have hi0 : i ≠ 0 := by simpa using hi'.2
351             have hnot : ¬ i ≤ c 0 := by omega
352             simp [hnot]
353           rw [hrest]
354           simp [MvPolynomial.pderiv_monomial, phaseExponent, hc0,
355             t, mold, qold]
356         · have hc : 0 < c 0 := Nat.pos_of_ne_zero hc0
357           have honeMem : 1 ∈ Finset.range (p 0 + 1) \ {0} := by
358             simp [hp]
359           have hone :
360               presentedPrincipalComponent k (@orderWeight (n + 1))
361                 (L + t - 1)
362                 (((p 0).choose 1 * (c 0).descFactorial 1) •
363                   presentedCoefficientOrdered k n
364                     (a 0 + c 0 - 1) (p 0 + q 0 - 1) z) =
365               MvPolynomial.pderiv (.inr (0 : Fin (n + 1)))
366                     (MvPolynomial.monomial (phaseExponent a p) 1) *
367                 MvPolynomial.pderiv (.inl (0 : Fin (n + 1)))
368                     (MvPolynomial.monomial (phaseExponent c q) 1) := by
369             simp only [map_nsmul]
370             rw [show L + t - 1 = L + (p 0 + q 0 - 1) by omega,
371               presentedPrincipalComponent_coefficientOrdered_order]
372             have htop := presentedOrderedMonomial_mul_principal_order k n
373               (fun i => a i.succ) (fun i => p i.succ)
374               (fun i => c i.succ) (fun i => q i.succ)
375             change presentedPrincipalComponent k (@orderWeight n) L z =
376                 MvPolynomial.monomial mold 1 *
377                   MvPolynomial.monomial qold 1 at htop
378             rw [htop]
379             exact firstContraction_newest_term k n a p c q hp hc
380           rw [Finset.sum_eq_single 1]
381           · simp only [show 1 ≤ c 0 from hc, if_true]
382             rw [hone]
383             abel
384           · intro i hi hi1
385             have hi' := Finset.mem_sdiff.mp hi
386             have hirange : i ∈ Finset.range (p 0 + 1) := hi'.1
387             have hi0 : i ≠ 0 := by simpa using hi'.2
388             have hip : i ≤ p 0 := by
389               have := Finset.mem_range.mp hirange
390               omega
391             by_cases hic : i ≤ c 0
392             · simp only [if_pos hic, map_nsmul]
393               have hterm := presentedCoefficientOrdered_mem_orderPiece k n L
394                 (a 0 + c 0 - i) (p 0 + q 0 - i) z hzOrder
395               have hlt : L + (p 0 + q 0 - i) < L + t - 1 := by
396                 dsimp [t]
397                 omega
398               have hvanish :=
399                 presentedPrincipalComponent_eq_zero_of_mem_of_lt k
400                   (@orderWeight (n + 1)) _ hterm hlt
401               rw [hvanish, nsmul_zero]
402             · simp [hic]
403           · exact fun h => (h honeMem).elim

405 /-- The exact first-contraction coefficient for arbitrary PBW basis vectors.
406 The positivity hypothesis says that the two monomials have positive combined
407 differential order, so the component one below the top is unambiguous. -/
408 theorem presentedPBWBasis_mul_firstContraction {n : ℕ}
409     (m q : PhaseVar n →₀ ℕ)
410     (hpos : 0 < monomialWeight (@orderWeight n) m +
411       monomialWeight (@orderWeight n) q) :
412     presentedPrincipalComponent k (@orderWeight n)
413         (monomialWeight (@orderWeight n) m +
414           monomialWeight (@orderWeight n) q - 1)
415         (presentedPBWBasis k n m * presentedPBWBasis k n q) =
416       pbwFirstContraction k m q := by
417   have hpos' :
418       0 < monomialWeight (@orderWeight n)
419             (phaseExponent (fun i => m (.inl i)) (fun i => m (.inr i))) +
420           monomialWeight (@orderWeight n)
421             (phaseExponent (fun i => q (.inl i)) (fun i => q (.inr i))) := by
422     simpa only [phaseExponent_split] using hpos
423   rw [presentedPBWBasis_apply, presentedPBWBasis_apply,
424     ← phaseExponent_split m, ← phaseExponent_split q]
425   exact presentedOrderedMonomial_mul_firstContraction k n
426     (fun i => m (.inl i)) (fun i => m (.inr i))
427     (fun i => q (.inl i)) (fun i => q (.inr i)) hpos'

429 /-- The unconditional two-element subprincipal commutator formula. -/
430 theorem principalComponent_commutator_eq_neg_poisson
431     {n r s : ℕ} {a b : PresentedWeyl k n}
432     (ha : a ∈ orderPiece k n r) (hb : b ∈ orderPiece k n s) :
