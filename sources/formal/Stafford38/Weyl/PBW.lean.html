13 /-!
14 # Ordered PBW basis from Ore normal forms
15 
16 The checked iterated Ore normal form gives a scalar-linear equivalence with
17 the commutative polynomial symbol space.  Transport across the independently
18 proved presented/iterated algebra equivalence yields a free-module basis
19 indexed by commutative monomials, without using monomial independence in the
20 quotient presentation.  Recursive all-degree formulas identify those basis
21 vectors with ordered products of the named Weyl generators.
22 -/

44 /-- Reindex the two newest symbol variables followed by the old variables to
45 the standard newest-first coordinate/momentum indexing. -/
46 def phaseSuccEquiv (n : Nat) :
47     ((PUnit.{1} ⊕ PUnit.{1}) ⊕ PhaseVar n) ≃ PhaseVar (n + 1) where
48   toFun
49     | .inl (.inl _) => .inr 0
50     | .inl (.inr _) => .inl 0
51     | .inr (.inl i) => .inl i.succ
52     | .inr (.inr i) => .inr i.succ
53   invFun
54     | .inl i => Fin.cases (.inl (.inr PUnit.unit))
55         (fun j => .inr (.inl j)) i
56     | .inr i => Fin.cases (.inl (.inl PUnit.unit))
57         (fun j => .inr (.inr j)) i
58   left_inv x := by
59     rcases x with (⟨⟨u⟩ | ⟨u⟩⟩ | ⟨i | i⟩) <;> rfl
60   right_inv x := by
61     rcases x with (i | i)
62     · exact Fin.cases rfl (fun _ => rfl) i
63     · exact Fin.cases rfl (fun _ => rfl) i

214 /-- Ordered monomials in the iterated tower: all older pairs occur first, and
215 within each pair the coordinate power precedes the momentum power. -/
216 def iteratedOrderedMonomial :
217     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → IteratedPairStage k n
218   | 0, _, _ => 1
219   | n + 1, a, p =>
220       stageEmbedding k n
221           (iteratedOrderedMonomial n (fun i => a i.succ) (fun i => p i.succ)) *
222         stageCoordinate k n ^ a 0 * stageMomentum k n ^ p 0

483 /-- Append powers of the newest named coordinate and momentum to an arbitrary
484 old presentation element in normal order. -/

486     WeylAlg k (n + 1) :=
487   previousWeylEmbedding k n z * presentedCoordinate k n ^ a *

490 /-- Normal-form coordinates of an arbitrary old presentation element followed
491 by powers of the newest named coordinate and momentum. -/
492 theorem presentedNormalFormLinearEquiv_previous_ordered
493     (n a p : ℕ) (z : WeylAlg k n) :
494     presentedNormalFormLinearEquiv k (n + 1)
495         (presentedCoefficientOrdered k n a p z) =
496       MvPolynomial.X (.inr (0 : Fin (n + 1))) ^ p *
497         MvPolynomial.X (.inl (0 : Fin (n + 1))) ^ a *
498         MvPolynomial.rename Stafford38FixedSourceChallenge.oldIndex
499           (presentedNormalFormLinearEquiv k n z) := by
500   letI : Algebra k (IteratedPairStage k n) := iteratedPairStageAlgebra k n
501   letI : Algebra k (CoordinateStage (B := IteratedPairStage k n)) :=
502     coordinateStageAlgebra
503   letI : Algebra k (PairStage (B := IteratedPairStage k n)) := pairStageAlgebra
504   rw [presentedCoefficientOrdered]
505   rw [presentedNormalFormLinearEquiv, LinearEquiv.trans_apply,
506     AlgEquiv.toLinearEquiv_apply]
507   have hmap : presentedIteratedEquiv k (n + 1)
508         (previousWeylEmbedding k n z *
509           presentedCoordinate k n ^ a * presentedMomentum k n ^ p) =
510       presentedIteratedEquiv k (n + 1) (previousWeylEmbedding k n z) *
511         presentedIteratedEquiv k (n + 1) (presentedCoordinate k n) ^ a *
512         presentedIteratedEquiv k (n + 1) (presentedMomentum k n) ^ p := by
513     simp only [map_mul, map_pow]
514   rw [hmap]
515   change iteratedNormalFormLinearEquiv k (n + 1)
516       (presentedToIterated k (n + 1) (previousWeylEmbedding k n z) *
517         presentedToIterated k (n + 1) (presentedCoordinate k n) ^ a *
518         presentedToIterated k (n + 1) (presentedMomentum k n) ^ p) = _
519   have hprevious := DFunLike.congr_fun (presentedToIterated_previous k n) z
520   rw [AlgHom.comp_apply, AlgHom.comp_apply] at hprevious
521   rw [hprevious, presentedToIterated_coordinate,
522     presentedToIterated_momentum]
523   change iteratedNormalFormLinearEquiv k (n + 1)
524     (iteratedCoefficientOrdered k n a p (presentedToIterated k n z)) = _
525   rw [iteratedNormalFormLinearEquiv_coefficient_ordered]

528 /-- Exact normal-ordering formula for two newest-pair blocks.  Every correction
529 term lowers both the coordinate and momentum exponents by the same amount. -/
530 theorem presentedCoefficientOrdered_mul
531     (n a b c d : ℕ) (z₁ z₂ : WeylAlg k n) :
532     presentedCoefficientOrdered k n a b z₁ *
533         presentedCoefficientOrdered k n c d z₂ =
534       ∑ i ∈ Finset.range (b + 1),
535         if i ≤ c then
536           (b.choose i * c.descFactorial i) •
537             presentedCoefficientOrdered k n (a + c - i) (b + d - i) (z₁ * z₂)
538         else 0 := by
539   let X := presentedCoordinate k n
540   let P := presentedMomentum k n
541   let Z₁ := previousWeylEmbedding k n z₁
542   let Z₂ := previousWeylEmbedding k n z₂
543   have hX : Commute X Z₂ := presentedCoordinate_commutes_previous k n z₂
544   have hP : Commute P Z₂ := presentedMomentum_commutes_previous k n z₂
545   have hblock : Commute (X ^ a * P ^ b) Z₂ :=
546     (hX.pow_left a).mul_left (hP.pow_left b)
547   have hPX : P * X - X * P = 1 := by
548     rw [show P * X = X * P + 1 from presentedMomentum_mul_coordinate k n]
549     noncomm_ring
550   rw [presentedCoefficientOrdered, presentedCoefficientOrdered]
551   change (Z₁ * X ^ a * P ^ b) * (Z₂ * X ^ c * P ^ d) = _
552   calc
553     (Z₁ * X ^ a * P ^ b) * (Z₂ * X ^ c * P ^ d) =
554         Z₁ * ((X ^ a * P ^ b) * Z₂) * (X ^ c * P ^ d) := by
555       noncomm_ring
556     _ = (Z₁ * Z₂ * X ^ a) * (P ^ b * X ^ c) * P ^ d := by
557       rw [hblock.eq]
558       noncomm_ring
559     _ = (Z₁ * Z₂ * X ^ a) *
560           (∑ i ∈ Finset.range (b + 1),
561             if i ≤ c then
562               (b.choose i * c.descFactorial i) •
563                 (X ^ (c - i) * P ^ (b - i))
564             else 0) * P ^ d := by
565       rw [OreAmbient.commutator_pow_mul_pow P X hPX b c]
566     _ = _ := by
567       rw [Finset.mul_sum, Finset.sum_mul]
568       apply Finset.sum_congr rfl
569       intro i hi
570       have hib : i ≤ b := by
571         have := Finset.mem_range.mp hi
572         omega
573       by_cases hic : i ≤ c
574       · simp only [if_pos hic, OreAmbient.mul_nsmul_left,
575           OreAmbient.nsmul_mul_right, mul_nsmul]
576         congr 1
577         congr 1
578         change (Z₁ * Z₂ * X ^ a) * (X ^ (c - i) * P ^ (b - i)) * P ^ d =
579           previousWeylEmbedding k n (z₁ * z₂) * X ^ (a + c - i) *
580             P ^ (b + d - i)
581         rw [map_mul]
582         rw [show a + c - i = a + (c - i) by omega,
583           show b + d - i = (b - i) + d by omega,
584           pow_add, pow_add]
585         noncomm_ring
586       · simp [hic]

589 -- exponent loss are part of the public normal-ordering statement.
590 example (n a b c d : ℕ) (z₁ z₂ : WeylAlg k n) :

600 /-- Ordered monomials built recursively from the named generators in the
601 quotient presentation.  Older coordinate/momentum pairs occur first; within
602 each pair the coordinate power precedes the momentum power. -/
603 def presentedOrderedMonomial :
604     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → WeylAlg k n
605   | 0, _, _ => 1
606   | n + 1, a, p =>
607       previousWeylEmbedding k n
608           (presentedOrderedMonomial n
609             (fun i => a i.succ) (fun i => p i.succ)) *
610         presentedCoordinate k n ^ a 0 * presentedMomentum k n ^ p 0

612 /-- The presentation/Ore equivalence preserves the recursively ordered
613 monomials, not merely the degree-one generators. -/
614 theorem presentedIteratedEquiv_orderedMonomial :
615     ∀ (n : ℕ) (a p : Fin n → ℕ),
616       presentedIteratedEquiv k n (presentedOrderedMonomial k n a p) =
617         iteratedOrderedMonomial k n a p := by
618   intro n
619   induction n with
620   | zero =>
621       intro a p
622       simp [presentedOrderedMonomial, iteratedOrderedMonomial]
623   | succ n ih =>
624       intro a p
625       rw [presentedOrderedMonomial, iteratedOrderedMonomial, map_mul, map_mul,
626         map_pow, map_pow, presentedIteratedEquiv]
627       change
628         presentedToIterated k (n + 1)
629               (previousWeylEmbedding k n
630                 (presentedOrderedMonomial k n
631                   (fun i => a i.succ) (fun i => p i.succ))) *
632             presentedToIterated k (n + 1) (presentedCoordinate k n) ^ a 0 *
633           presentedToIterated k (n + 1) (presentedMomentum k n) ^ p 0 = _
634       have hprevious := DFunLike.congr_fun
635         (presentedToIterated_previous k n)
636         (presentedOrderedMonomial k n
637           (fun i => a i.succ) (fun i => p i.succ))
638       rw [AlgHom.comp_apply, AlgHom.comp_apply] at hprevious
639       have hih := ih (fun i => a i.succ) (fun i => p i.succ)
640       change presentedToIterated k n
641           (presentedOrderedMonomial k n
642             (fun i => a i.succ) (fun i => p i.succ)) = _ at hih
643       rw [hprevious, hih, presentedToIterated_coordinate,
644         presentedToIterated_momentum]
645       rfl
