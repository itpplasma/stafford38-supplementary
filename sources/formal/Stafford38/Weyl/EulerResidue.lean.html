8 # The positive Euler residue in the canonical right ideal
9 
10 This module fixes the right-ideal orientation of the Euler Bezout identity
11 and proves normalized remainder-residue facts. `PaperEulerGrading` supplies
12 the PBW dictionary and right-quotient surjectivity route and is the official
13 consumer of this Euler residue.
14 -/

23 variable {A : Type*} [Ring A]

25 /-- The literal right ideal generated by `d` and `x^N d`. -/
26 def canonicalRightIdeal (x d : A) (N : ℕ) : Submodule Aᵐᵒᵖ A :=
27   Submodule.span Aᵐᵒᵖ ({d, x ^ N * d} : Set A)
28 
29 lemma firstGenerator_mem (x d : A) (N : ℕ) :
30     d ∈ canonicalRightIdeal x d N := by
31   apply Submodule.subset_span
32   simp
33 
34 lemma secondGenerator_mem (x d : A) (N : ℕ) :
35     x ^ N * d ∈ canonicalRightIdeal x d N := by
36   apply Submodule.subset_span
37   simp

39 /-- Expanding a monic element around the pure momentum power transports the
40 Euler Bezout identity to an exact positive-residue identity. -/
41 theorem positiveEulerResidue_identity
42     (x p d sigma u v : A) (N : ℕ)
43     (hd : d = p ^ N + sigma)
44     (hbezout : (p ^ N * x ^ N) * u + (x ^ N * p ^ N) * v = 1) :
45     d * (x ^ N * u) + (x ^ N * d) * v =
46       1 + ((sigma * x ^ N) * u + (x ^ N * sigma) * v) := by
47   rw [hd]
48   calc
49     (p ^ N + sigma) * (x ^ N * u) +
50         (x ^ N * (p ^ N + sigma)) * v =
51       ((p ^ N * x ^ N) * u + (x ^ N * p ^ N) * v) +
52         ((sigma * x ^ N) * u + (x ^ N * sigma) * v) := by
53           noncomm_ring
54     _ = 1 + ((sigma * x ^ N) * u + (x ^ N * sigma) * v) := by
55       rw [hbezout]
56 
57 /-- For a Weyl pair and a monic decomposition `d = p^N + sigma`, the explicit
58 positive Euler residue belongs to the literal right ideal `dA + x^N dA`.
59 The cofactors come from the ordinary Bezout identity for the two disjoint
60 Euler products. -/
61 theorem positiveEulerResidue_mem
62     [Algebra ℚ A] (x p d sigma : A) (N : ℕ)
63     (hweyl : p * x = x * p + 1)
64     (hd : d = p ^ N + sigma) :
65     ∃ u v : A,
66       1 + ((sigma * x ^ N) * u + (x ^ N * sigma) * v) ∈
67         canonicalRightIdeal x d N := by
68   obtain ⟨u, v, hbezout⟩ :=
69     euler_products_right_bezout_exists x p hweyl N
70   refine ⟨u, v, ?_⟩
71   have hfirst : d * (x ^ N * u) ∈ canonicalRightIdeal x d N := by
72     change MulOpposite.op (x ^ N * u) • d ∈ canonicalRightIdeal x d N
73     exact (canonicalRightIdeal x d N).smul_mem _
74       (firstGenerator_mem x d N)
75   have hsecond : (x ^ N * d) * v ∈ canonicalRightIdeal x d N := by
76     change MulOpposite.op v • (x ^ N * d) ∈ canonicalRightIdeal x d N
77     exact (canonicalRightIdeal x d N).smul_mem _
78       (secondGenerator_mem x d N)
79   rw [← positiveEulerResidue_identity x p d sigma u v N hd hbezout]
80   exact (canonicalRightIdeal x d N).add_mem hfirst hsecond
81 
82 /-- The same residue statement with the two cofactors retained explicitly as
83 rational polynomials in the Euler element `x*p`. -/
84 theorem positiveEulerResidue_polynomial_mem
85     [Algebra ℚ A] (x p d sigma : A) (N : ℕ)
86     (hweyl : p * x = x * p + 1)
87     (hd : d = p ^ N + sigma) :
88     ∃ u v : Polynomial ℚ,
89       1 +
90           ((sigma * x ^ N) * eulerPolynomialEval x p u +
91             (x ^ N * sigma) * eulerPolynomialEval x p v) ∈
92         canonicalRightIdeal x d N := by
93   obtain ⟨u, v, hbezout⟩ :=
94     euler_products_polynomial_right_bezout_exists x p hweyl N
95   refine ⟨u, v, ?_⟩
96   have hfirst : d * (x ^ N * eulerPolynomialEval x p u) ∈
97       canonicalRightIdeal x d N := by
98     change MulOpposite.op (x ^ N * eulerPolynomialEval x p u) • d ∈
99       canonicalRightIdeal x d N
100     exact (canonicalRightIdeal x d N).smul_mem _
101       (firstGenerator_mem x d N)
102   have hsecond : (x ^ N * d) * eulerPolynomialEval x p v ∈
103       canonicalRightIdeal x d N := by
104     change MulOpposite.op (eulerPolynomialEval x p v) • (x ^ N * d) ∈
105       canonicalRightIdeal x d N
106     exact (canonicalRightIdeal x d N).smul_mem _
107       (secondGenerator_mem x d N)
108   rw [← positiveEulerResidue_identity x p d sigma
109     (eulerPolynomialEval x p u) (eulerPolynomialEval x p v) N hd hbezout]
110   exact (canonicalRightIdeal x d N).add_mem hfirst hsecond
