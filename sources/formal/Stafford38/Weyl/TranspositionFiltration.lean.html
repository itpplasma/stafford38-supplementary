11 /-!
12 # Filtration and symbol transport under Weyl transposition
13 
14 The Weyl transposition reverses products, fixes coordinates, and negates
15 momenta.  This file proves directly from the checked PBW filtration that it
16 preserves both differential order and Bernstein degree.  No filtered
17 `D`-module or characteristic-variety theorem is used here.
18 -/

53 @[simp] theorem transpose_pow (n : Nat) (a : WeylAlg k n) (r : Nat) :
54     transpose k n (a ^ r) = transpose k n a ^ r := by
55   induction r with
56   | zero => simp [transpose]
57   | succ r ih =>
58       rw [pow_succ, transpose_mul, ih]
59       calc
60         transpose k n a * transpose k n a ^ r =
61             transpose k n a ^ r * transpose k n a :=
62           ((Commute.refl (transpose k n a)).pow_right r).eq
63         _ = transpose k n a ^ (r + 1) := (pow_succ _ _).symm

65 theorem transpose_previousWeylEmbedding (n : Nat) (a : WeylAlg k n) :
66     transpose k (n + 1) (previousWeylEmbedding k n a) =
67       previousWeylEmbedding k n (transpose k n a) := by
68   let lhs : WeylAlg k n →ₐ[k] (WeylAlg k (n + 1))ᵐᵒᵖ :=
69     (transpositionHom k (n + 1)).comp (previousWeylEmbedding k n)
70   let rhs : WeylAlg k n →ₐ[k] (WeylAlg k (n + 1))ᵐᵒᵖ :=
71     (AlgHom.op (previousWeylEmbedding k n)).comp (transpositionHom k n)
72   have h : lhs = rhs := by
73     apply Stafford38.WeylUniversal.freeWeyl_algHom_ext
74     intro i
75     dsimp [lhs, rhs]
76     cases i with
77     | inl i =>
78         rw [previousWeylEmbedding_generator,
79           show freeWeylGenerator (Matrix.J (Fin n) k) (.inl i) =
80             coordinate k n i from rfl,
81           transpositionHom_coordinate]
82         change transpositionHom k (n + 1) (oldGenerator k n (.inl i)) =
83           op (previousWeylEmbedding k n (coordinate k n i))
84         rw [show oldGenerator k n (.inl i) = coordinate k (n + 1) i.succ from rfl,
85           transpositionHom_coordinate]
86         have hp := previousWeylEmbedding_generator k n (.inl i)
87         change previousWeylEmbedding k n (coordinate k n i) =
88           coordinate k (n + 1) i.succ at hp
89         exact congrArg op hp.symm
90     | inr i =>
91         rw [previousWeylEmbedding_generator,
92           show freeWeylGenerator (Matrix.J (Fin n) k) (.inr i) =
93             momentum k n i from rfl,
94           transpositionHom_momentum]
95         change transpositionHom k (n + 1) (oldGenerator k n (.inr i)) =
96           op (previousWeylEmbedding k n (-momentum k n i))
97         rw [show oldGenerator k n (.inr i) = momentum k (n + 1) i.succ from rfl,
98           transpositionHom_momentum, map_neg]
99         have hp := previousWeylEmbedding_generator k n (.inr i)
100         change previousWeylEmbedding k n (momentum k n i) =
101           momentum k (n + 1) i.succ at hp
102         exact congrArg op (congrArg Neg.neg hp.symm)
103   have ha := DFunLike.congr_fun h a
104   exact congrArg MulOpposite.unop ha

211       transpose k n (presentedOrderedMonomial k n a p) ∈
212         bernsteinPiece k n
213           (Finsupp.weight (@bernsteinWeight n) (phaseExponent a p)) := by

222       simpa [presentedOrderedMonomial, Finsupp.weight_apply, bernsteinPiece,

279 theorem transpose_mem_bernsteinPiece {n N : Nat} {a : WeylAlg k n}
280     (ha : a ∈ bernsteinPiece k n N) :
281     transpose k n a ∈ bernsteinPiece k n N := by
282   rw [bernsteinPiece, presentedWeightPiece_eq_span] at ha
283   change transpose k n a ∈ presentedWeightPiece k bernsteinWeight N
284   induction ha using Submodule.span_induction with
285   | mem b hb =>
286       obtain ⟨m, hm, rfl⟩ := hb
287       rw [presentedPBWBasis_apply]
288       have hmono := transpose_orderedMonomial_mem_bernsteinPiece k n
289         (fun i => m (.inl i)) (fun i => m (.inr i))
290       exact presentedWeightPiece_mono k bernsteinWeight hm
291         (by simpa [bernsteinPiece, phaseExponent_split] using hmono)
292   | zero =>
293       rw [transpose_zero]
294       exact (presentedWeightPiece k bernsteinWeight N).zero_mem
295   | add x y hx hy ihx ihy =>
296       rw [transpose_add]
297       exact (presentedWeightPiece k bernsteinWeight N).add_mem ihx ihy
298   | smul c x hx ih =>
299       rw [transpose_smul]
300       exact (presentedWeightPiece k bernsteinWeight N).smul_mem c ih

426     ∀ (n : Nat) (a p : Fin n → Nat),
427       presentedPrincipalComponent k (@bernsteinWeight n)
428           (Finsupp.weight (@bernsteinWeight n) (phaseExponent a p))

461         monomialWeight_phaseExponent_succ_bernstein]

603 theorem principal_transpose_orderedMonomial_order :
604     ∀ (n : Nat) (a p : Fin n → Nat),
605       presentedPrincipalComponent k (@orderWeight n)
606           (Finsupp.weight (@orderWeight n) (phaseExponent a p))
607           (transpose k n (presentedOrderedMonomial k n a p)) =
608         symbolTransposition k
609           (MvPolynomial.monomial (phaseExponent a p) (1 : k)) := by
610   intro n
611   induction n with
612   | zero =>
613       intro a p
614       have he : phaseExponent a p = 0 := by
615         ext i
616         exact Sum.elim Fin.elim0 Fin.elim0 i
617       rw [he]
618       simp [presentedOrderedMonomial, transpose, Finsupp.weight_apply,
619         presentedPrincipalComponent, presentedNormalFormLinearEquiv_one,
620         weightedHomogeneousComponent_monomial, symbolTransposition]
621   | succ n ih =>
622       intro a p
623       let oldA : Fin n → Nat := fun i => a i.succ
624       let oldP : Fin n → Nat := fun i => p i.succ
625       let oldD := Finsupp.weight (@orderWeight n) (phaseExponent oldA oldP)
626       have hold := previousWeylEmbedding_mem_orderPiece k
627         (transpose_orderedMonomial_mem_orderPiece k n oldA oldP)
628       have hx := coordinate_pow_mem_orderPiece k
629         (0 : Fin (n + 1)) (a 0)
630       have hp := momentum_pow_mem_orderPiece k
631         (0 : Fin (n + 1)) (p 0)
632       have hxo := mul_mem_orderPiece k hx hold
633       rw [presentedOrderedMonomial, transpose_mul, transpose_mul,
634         transpose_pow, transpose_pow,
635         show presentedMomentum k n = momentum k (n + 1) 0 from rfl,
636         show presentedCoordinate k n = coordinate k (n + 1) 0 from rfl,
637         transpose_momentum, transpose_coordinate,
638         transpose_previousWeylEmbedding,
639         monomialWeight_phaseExponent_succ_order]
640       change presentedPrincipalComponent k (@orderWeight (n + 1))
641           (oldD + p 0)
642           ((-momentum k (n + 1) 0) ^ p 0 *
643             (coordinate k (n + 1) 0 ^ a 0 *
644               previousWeylEmbedding k n
645                 (transpose k n (presentedOrderedMonomial k n oldA oldP)))) = _
646       rw [show oldD + p 0 = p 0 + (0 + oldD) by omega,
647         presentedPrincipalComponent_mul_order k hp hxo,
648         presentedPrincipalComponent_mul_order k hx hold,
649         principal_negMomentum_pow_order,
650         principal_coordinate_pow_order,
651         principal_previous_order, ih oldA oldP]
652       rw [Stafford38.WeylSymbolCompatibility.phaseMonomial_succ_decompose]
653       simp only [map_mul, map_pow, symbolTransposition_coordinate,
654         symbolTransposition_momentum, symbolTransposition_rename_oldIndex]
655       ring
