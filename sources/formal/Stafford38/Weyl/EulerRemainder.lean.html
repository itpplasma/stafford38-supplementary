11 /-!
12 # Positive outer-Ore remainders
13 
14 An element whose outer momentum support is strictly below `N` acquires a
15 right coordinate factor after multiplication by `x^N` on either side.  The
16 cofactors remain in the concrete Euler subring.
17 -/

45 /-- Both products by `x^N` have a right `x` factor with cofactor in the Euler
46 subring. -/
47 def HasPositiveCoordinateFactor (N : ℕ) (z : PairStage (B := B)) : Prop :=
48   ∃ U V : pairEulerSubring B,
49     z * pairCoordinate ^ N = (U : PairStage (B := B)) * pairCoordinate ∧
50     pairCoordinate ^ N * z = (V : PairStage (B := B)) * pairCoordinate

69 lemma pairCoordinate_pow_mul_coefficient (b : B) : ∀ N : ℕ,
70     pairCoordinate ^ N * pairCoefficient b =
71       pairCoefficient b * pairCoordinate ^ N
72   | 0 => by simp
73   | N + 1 => by
74       calc
75         pairCoordinate ^ (N + 1) * pairCoefficient b =
76             (pairCoordinate ^ N * pairCoordinate) * pairCoefficient b := by
77               rw [pow_succ]
78         _ = pairCoordinate ^ N * (pairCoordinate * pairCoefficient b) := by
79               rw [mul_assoc]
80         _ = pairCoordinate ^ N * (pairCoefficient b * pairCoordinate) := by
81               rw [pairCoordinate_mul_coefficient]
82         _ = (pairCoordinate ^ N * pairCoefficient b) * pairCoordinate := by
83               rw [mul_assoc]
84         _ = pairCoefficient b * pairCoordinate ^ (N + 1) := by
85               rw [pairCoordinate_pow_mul_coefficient b N, pow_succ, mul_assoc]

88 /-- A single ordered monomial with outer momentum exponent below `N` has the
89 two required positive factorizations. -/
90 theorem pairOrderedMonomial_hasPositiveCoordinateFactor
91     (b : B) (a j N : ℕ) (hjN : j < N) :
92     HasPositiveCoordinateFactor B N
93       (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) := by
94   obtain ⟨f, hf⟩ := exists_eulerPolynomial_d_pow_mul_x_pow
95     (A := PairStage (B := B)) pairCoordinate pairMomentum
96     pairMomentum_mul_coordinate j
97   have hpoly : eulerPolynomialEval pairCoordinate pairMomentum f ∈
98       pairEulerSubring B :=
99     eulerPolynomialEval_mem_eulerSubring (pairOldSubring B)
100       pairCoordinate pairMomentum (rational_mem_pairOldSubring B) f
101   have hb : pairCoefficient b ∈ pairEulerSubring B :=
102     coefficient_mem_eulerSubring (pairOldSubring B)
103       pairCoordinate pairMomentum ⟨pairCoefficient b, ⟨b, rfl⟩⟩
104   have hxa : pairCoordinate ^ a ∈ pairEulerSubring B :=
105     (pairEulerSubring B).pow_mem
106       (coordinate_mem_eulerSubring (pairOldSubring B)
107         pairCoordinate pairMomentum) a
108   have hxleft : pairCoordinate ^ (N - j - 1) ∈ pairEulerSubring B :=
109     (pairEulerSubring B).pow_mem
110       (coordinate_mem_eulerSubring (pairOldSubring B)
111         pairCoordinate pairMomentum) (N - j - 1)
112   let U : pairEulerSubring B :=
113     ⟨pairCoefficient b * pairCoordinate ^ a *
114         eulerPolynomialEval pairCoordinate pairMomentum f *
115           pairCoordinate ^ (N - j - 1),
116       (pairEulerSubring B).mul_mem
117         ((pairEulerSubring B).mul_mem
118           ((pairEulerSubring B).mul_mem hb hxa) hpoly) hxleft⟩
119   obtain ⟨V, hV⟩ :=
120     pairOrderedMonomial_eq_eulerSubring_mul_coordinate B b (N + a) j
121       (by omega)
122   refine ⟨U, V, ?_, ?_⟩
123   · have hpowN : (pairCoordinate (B := B)) ^ N =
124         pairCoordinate ^ j * pairCoordinate ^ (N - j) := by
125       rw [← pow_add, Nat.add_sub_of_le (Nat.le_of_lt hjN)]
126     have hpowDiff : (pairCoordinate (B := B)) ^ (N - j) =
127         pairCoordinate ^ (N - j - 1) * pairCoordinate := by
128       calc
129         pairCoordinate ^ (N - j) =
130             pairCoordinate ^ ((N - j - 1) + 1) := by congr 1; omega
131         _ = pairCoordinate ^ (N - j - 1) * pairCoordinate := by
132           rw [pow_add, pow_one]
133     change
134       (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) *
135           pairCoordinate ^ N =
136         (pairCoefficient b * pairCoordinate ^ a *
137             eulerPolynomialEval pairCoordinate pairMomentum f *
138               pairCoordinate ^ (N - j - 1)) * pairCoordinate
139     calc
140       (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) *
141           pairCoordinate ^ N =
142         pairCoefficient b * pairCoordinate ^ a *
143           (pairMomentum ^ j * pairCoordinate ^ j) *
144             pairCoordinate ^ (N - j) := by
145               rw [hpowN]
146               simp only [mul_assoc]
147       _ = pairCoefficient b * pairCoordinate ^ a *
148           eulerPolynomialEval pairCoordinate pairMomentum f *
149             pairCoordinate ^ (N - j) := by
150               rw [hf]
151       _ = (pairCoefficient b * pairCoordinate ^ a *
152             eulerPolynomialEval pairCoordinate pairMomentum f *
153               pairCoordinate ^ (N - j - 1)) * pairCoordinate := by
154               rw [hpowDiff]
155               simp only [mul_assoc]
156   · calc
157       pairCoordinate ^ N *
158           (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) =
159         pairCoefficient b * pairCoordinate ^ (N + a) * pairMomentum ^ j := by
160           calc
161             pairCoordinate ^ N *
162                 (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) =
163               (pairCoordinate ^ N * pairCoefficient b) *
164                 pairCoordinate ^ a * pairMomentum ^ j := by
165                   simp only [mul_assoc]
166             _ = (pairCoefficient b * pairCoordinate ^ N) *
167                 pairCoordinate ^ a * pairMomentum ^ j := by
168                   rw [pairCoordinate_pow_mul_coefficient B b N]
169             _ = pairCoefficient b * pairCoordinate ^ (N + a) *
170                 pairMomentum ^ j := by
171                   rw [pow_add]
172                   simp only [mul_assoc]
173       _ = (V : PairStage (B := B)) * pairCoordinate := hV

175 private lemma coordinateCoefficientTerm_hasPositive
176     (c : CoordinateStage (B := B)) (j N : ℕ) (hjN : j < N) :
177     HasPositiveCoordinateFactor B N
178       (normalCoefficient coordinateDerivation c * pairMomentum ^ j) := by
179   let q := (normalFormAddEquiv (zeroDerivation (B := B))).symm c
180   have hc : c = normalForm (zeroDerivation (B := B)) q := by
181     exact (normalFormAddEquiv (zeroDerivation (B := B))).apply_symm_apply c |>.symm
182   rw [hc]
183   induction q using Polynomial.induction_on' with
184   | add q r hq hr =>
185       rw [normalForm_add,
186         (normalCoefficient (coordinateDerivation (B := B))).map_add, add_mul]
187       exact hasPositiveCoordinateFactor_add (B := B) hq hr
188   | monomial a b =>
189       rw [normalForm_monomial,
190         (normalCoefficient (coordinateDerivation (B := B))).map_mul,
191         (normalCoefficient (coordinateDerivation (B := B))).map_pow]
192       exact pairOrderedMonomial_hasPositiveCoordinateFactor B b a j N hjN

194 /-- Every element whose outer momentum support lies below `N` has both
195 positive right-coordinate factorizations. -/
196 theorem hasPositiveCoordinateFactor_of_outer_support
197     (z : PairStage (B := B)) (N : ℕ)
198     (hsupport : ∀ j ∈
199       ((normalFormAddEquiv coordinateDerivation).symm z).support, j < N) :
200     HasPositiveCoordinateFactor B N z := by
201   let q := (normalFormAddEquiv coordinateDerivation).symm z
202   have hz : z = normalForm coordinateDerivation q := by
203     exact (normalFormAddEquiv coordinateDerivation).apply_symm_apply z |>.symm
204   rw [hz, ← Polynomial.sum_monomial_eq q]
205   change HasPositiveCoordinateFactor B N
206     (normalFormAddHom (coordinateDerivation (B := B))
207       (∑ j ∈ q.support, Polynomial.monomial j (q.coeff j)))
208   rw [map_sum]
209   refine Finset.sum_induction (M := PairStage (B := B))
210     (fun j => normalFormAddHom (coordinateDerivation (B := B))
211       (Polynomial.monomial j (q.coeff j)))
212     (HasPositiveCoordinateFactor B N) ?_ ?_ ?_
213   · intro a b ha hb
214     exact hasPositiveCoordinateFactor_add (B := B) ha hb
215   · exact hasPositiveCoordinateFactor_zero B N
216   · intro j hj
217     change HasPositiveCoordinateFactor B N
218       (normalForm (coordinateDerivation (B := B))
219         (Polynomial.monomial j (q.coeff j)))
220     rw [normalForm_monomial]
221     exact coordinateCoefficientTerm_hasPositive B (q.coeff j) j N
222       (hsupport j hj)

225 lemma support_sub_X_pow_lt
226     (H : Polynomial (CoordinateStage (B := B))) (N : ℕ)
227     (hN : H.coeff N = 1)
228     (hgt : ∀ j, N < j → H.coeff j = 0) :
229     ∀ j ∈ (H - Polynomial.X ^ N).support, j < N := by
230   intro j hj
231   by_contra hnot
232   have hle : N ≤ j := Nat.le_of_not_gt hnot
233   have hcoeff : (H - Polynomial.X ^ N).coeff j = 0 := by
234     rcases hle.eq_or_lt with rfl | hlt
235     · simp [hN]
236     · rw [Polynomial.coeff_sub, hgt j hlt]
237       simp [Polynomial.coeff_X_pow, hlt.ne']
238   exact (Polynomial.mem_support_iff.mp hj) hcoeff

240 /-- Subtracting the monic outer power leaves an element to which the positive
241 factor theorem applies. -/
242 theorem normalForm_sub_X_pow_hasPositiveCoordinateFactor
243     (H : Polynomial (CoordinateStage (B := B))) (N : ℕ)
244     (hN : H.coeff N = 1)
245     (hgt : ∀ j, N < j → H.coeff j = 0) :
246     HasPositiveCoordinateFactor B N
247       (normalForm (coordinateDerivation (B := B))
248         (H - Polynomial.X ^ N)) := by
249   apply hasPositiveCoordinateFactor_of_outer_support B
250   intro j hj
251   change j ∈ ((normalFormAddEquiv (coordinateDerivation (B := B))).symm
252     ((normalFormAddEquiv (coordinateDerivation (B := B)))
253       (H - Polynomial.X ^ N))).support at hj
254   rw [(normalFormAddEquiv (coordinateDerivation (B := B))).symm_apply_apply] at hj
255   exact support_sub_X_pow_lt B H N hN hgt j hj

257 /-- The positive-factor interface turns the explicit Euler residue into the
258 exact shaped residue consumed by quotient surjectivity. -/
259 theorem positiveEulerResidue_eq_eulerSubring_mul_coordinate
260     (d sigma : PairStage (B := B)) (N : ℕ)
261     (hd : d = pairMomentum ^ N + sigma)
262     (hpositive : HasPositiveCoordinateFactor B N sigma) :
263     ∃ U : pairEulerSubring B,
264       1 + (U : PairStage (B := B)) * pairCoordinate ∈
265         canonicalRightIdeal pairCoordinate d N := by
266   obtain ⟨u, v, hmem⟩ := positiveEulerResidue_polynomial_mem
267     pairCoordinate pairMomentum d sigma N pairMomentum_mul_coordinate hd
268   obtain ⟨U, V, hright, hleft⟩ := hpositive
269   obtain ⟨su, hsu⟩ := coordinate_mul_eulerPolynomial
270     (pairOldSubring B) pairCoordinate pairMomentum
271     pairMomentum_mul_coordinate (pairOldSubring_commutes_coordinate B)
272     (rational_mem_pairOldSubring B) u
273   obtain ⟨sv, hsv⟩ := coordinate_mul_eulerPolynomial
274     (pairOldSubring B) pairCoordinate pairMomentum
275     pairMomentum_mul_coordinate (pairOldSubring_commutes_coordinate B)
276     (rational_mem_pairOldSubring B) v
277   refine ⟨U * su + V * sv, ?_⟩
278   convert hmem using 1
279   simp only [Subring.coe_add, Subring.coe_mul, add_mul]
280   rw [hright, hleft]
281   simp only [mul_assoc]
282   rw [hsu, hsv]

371 /-- The concrete remainder of a normalized presented Weyl operator satisfies
372 both positive-coordinate factorizations. -/
373 theorem presentedRemainder_hasPositiveCoordinateFactor
374     (n N : ℕ) {d : WeylAlg k (n + 1)}
375     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
376     HasPositiveCoordinateFactor (IteratedPairStage k n) N
377       (presentedToIterated k (n + 1) d - stageMomentum k n ^ N) := by
378   let H := presentedOuterPolynomial k n d
379   have hpositive := normalForm_sub_X_pow_hasPositiveCoordinateFactor
380     (IteratedPairStage k n) H N
381     (outer_coeff_eq_one_at_bound k n N hd)
382     (fun j hj => outer_coeff_eq_zero_of_exponent_gt k n N hd hj)
383   have hnormal :
384       normalForm (coordinateDerivation (B := IteratedPairStage k n))
385           (H - Polynomial.X ^ N) =
386         presentedToIterated k (n + 1) d - stageMomentum k n ^ N := by
387     change normalFormAddHom coordinateDerivation (H - Polynomial.X ^ N) = _
388     have hH : normalForm (coordinateDerivation (B := IteratedPairStage k n)) H =
389         presentedToIterated k (n + 1) d := by
390       exact (normalFormAddEquiv coordinateDerivation).apply_symm_apply _
391     have hH' : normalFormAddHom coordinateDerivation H =
392         presentedToIterated k (n + 1) d := hH
393     rw [map_sub, hH']
394     congr 1
395     rw [Polynomial.X_pow_eq_monomial]
396     change normalForm coordinateDerivation (Polynomial.monomial N 1) = _
397     rw [normalForm_monomial]
398     rw [map_one, one_mul]
399     change pairMomentum (B := IteratedPairStage k n) ^ N =
400       stageMomentum k n ^ N
401     rfl
402   rw [hnormal] at hpositive
403   exact hpositive

405 /-- The normalized presented operator has the exact positive Euler residue in
406 its literal canonical right ideal. -/
407 theorem presentedPositiveEulerResidue
408     (n N : ℕ) {d : WeylAlg k (n + 1)}
409     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
410     ∃ U : pairEulerSubring (IteratedPairStage k n),
411       1 + (U : PairStage (B := IteratedPairStage k n)) * pairCoordinate ∈
412         canonicalRightIdeal (pairCoordinate (B := IteratedPairStage k n))
413           (presentedToIterated k (n + 1) d) N := by
414   let sigma := presentedToIterated k (n + 1) d - stageMomentum k n ^ N
415   have hdecomp : presentedToIterated k (n + 1) d =
416       stageMomentum k n ^ N + sigma := by
417     simp [sigma]
418   have hpositive : HasPositiveCoordinateFactor (IteratedPairStage k n) N sigma :=
419     presentedRemainder_hasPositiveCoordinateFactor k n N hd
420   exact positiveEulerResidue_eq_eulerSubring_mul_coordinate
421     (IteratedPairStage k n) (presentedToIterated k (n + 1) d) sigma N
422       hdecomp hpositive
