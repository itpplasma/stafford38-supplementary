9 /-!
10 # Positive outer-Ore remainders
11 
12 An element whose outer momentum support is strictly below `N` acquires a
13 right coordinate factor after multiplication by `x^N` on either side.  The
14 cofactors remain in the concrete Euler subring.
15 -/

43 /-- Both products by `x^N` have a right `x` factor with cofactor in the Euler
44 subring. -/
45 def HasPositiveCoordinateFactor (N : ℕ) (z : PairStage (B := B)) : Prop :=
46   ∃ U V : pairEulerSubring B,
47     z * pairCoordinate ^ N = (U : PairStage (B := B)) * pairCoordinate ∧
48     pairCoordinate ^ N * z = (V : PairStage (B := B)) * pairCoordinate

57     (hz : HasPositiveCoordinateFactor B N z)

67 lemma pairCoordinate_pow_mul_coefficient (b : B) : ∀ N : ℕ,
68     pairCoordinate ^ N * pairCoefficient b =
69       pairCoefficient b * pairCoordinate ^ N
70   | 0 => by simp
71   | N + 1 => by
72       calc
73         pairCoordinate ^ (N + 1) * pairCoefficient b =
74             (pairCoordinate ^ N * pairCoordinate) * pairCoefficient b := by
75               rw [pow_succ]
76         _ = pairCoordinate ^ N * (pairCoordinate * pairCoefficient b) := by
77               rw [mul_assoc]
78         _ = pairCoordinate ^ N * (pairCoefficient b * pairCoordinate) := by
79               rw [pairCoordinate_mul_coefficient]
80         _ = (pairCoordinate ^ N * pairCoefficient b) * pairCoordinate := by
81               rw [mul_assoc]
82         _ = pairCoefficient b * pairCoordinate ^ (N + 1) := by
83               rw [pairCoordinate_pow_mul_coefficient b N, pow_succ, mul_assoc]

86 /-- A single ordered monomial with outer momentum exponent below `N` has the
87 two required positive factorizations. -/
88 theorem pairOrderedMonomial_hasPositiveCoordinateFactor
89     (b : B) (a j N : ℕ) (hjN : j < N) :
90     HasPositiveCoordinateFactor B N
91       (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) := by
92   obtain ⟨f, hf⟩ := exists_eulerPolynomial_d_pow_mul_x_pow
93     (A := PairStage (B := B)) pairCoordinate pairMomentum
94     pairMomentum_mul_coordinate j
95   have hpoly : eulerPolynomialEval pairCoordinate pairMomentum f ∈
96       pairEulerSubring B :=
97     eulerPolynomialEval_mem_eulerSubring (pairOldSubring B)
98       pairCoordinate pairMomentum (rational_mem_pairOldSubring B) f
99   have hb : pairCoefficient b ∈ pairEulerSubring B :=
100     coefficient_mem_eulerSubring (pairOldSubring B)
101       pairCoordinate pairMomentum ⟨pairCoefficient b, ⟨b, rfl⟩⟩
102   have hxa : pairCoordinate ^ a ∈ pairEulerSubring B :=
103     (pairEulerSubring B).pow_mem
104       (coordinate_mem_eulerSubring (pairOldSubring B)
105         pairCoordinate pairMomentum) a
106   have hxleft : pairCoordinate ^ (N - j - 1) ∈ pairEulerSubring B :=
107     (pairEulerSubring B).pow_mem
108       (coordinate_mem_eulerSubring (pairOldSubring B)
109         pairCoordinate pairMomentum) (N - j - 1)
110   let U : pairEulerSubring B :=
111     ⟨pairCoefficient b * pairCoordinate ^ a *
112         eulerPolynomialEval pairCoordinate pairMomentum f *
113           pairCoordinate ^ (N - j - 1),
114       (pairEulerSubring B).mul_mem
115         ((pairEulerSubring B).mul_mem
116           ((pairEulerSubring B).mul_mem hb hxa) hpoly) hxleft⟩
117   obtain ⟨V, hV⟩ :=
118     pairOrderedMonomial_eq_eulerSubring_mul_coordinate B b (N + a) j
119       (by omega)
120   refine ⟨U, V, ?_, ?_⟩
121   · have hpowN : (pairCoordinate (B := B)) ^ N =
122         pairCoordinate ^ j * pairCoordinate ^ (N - j) := by
123       rw [← pow_add, Nat.add_sub_of_le (Nat.le_of_lt hjN)]
124     have hpowDiff : (pairCoordinate (B := B)) ^ (N - j) =
125         pairCoordinate ^ (N - j - 1) * pairCoordinate := by
126       calc
127         pairCoordinate ^ (N - j) =
128             pairCoordinate ^ ((N - j - 1) + 1) := by congr 1; omega
129         _ = pairCoordinate ^ (N - j - 1) * pairCoordinate := by
130           rw [pow_add, pow_one]
131     change
132       (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) *
133           pairCoordinate ^ N =
134         (pairCoefficient b * pairCoordinate ^ a *
135             eulerPolynomialEval pairCoordinate pairMomentum f *
136               pairCoordinate ^ (N - j - 1)) * pairCoordinate
137     calc
138       (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) *
139           pairCoordinate ^ N =
140         pairCoefficient b * pairCoordinate ^ a *
141           (pairMomentum ^ j * pairCoordinate ^ j) *
142             pairCoordinate ^ (N - j) := by
143               rw [hpowN]
144               simp only [mul_assoc]
145       _ = pairCoefficient b * pairCoordinate ^ a *
146           eulerPolynomialEval pairCoordinate pairMomentum f *
147             pairCoordinate ^ (N - j) := by
148               rw [hf]
149       _ = (pairCoefficient b * pairCoordinate ^ a *
150             eulerPolynomialEval pairCoordinate pairMomentum f *
151               pairCoordinate ^ (N - j - 1)) * pairCoordinate := by
152               rw [hpowDiff]
153               simp only [mul_assoc]
154   · calc
155       pairCoordinate ^ N *
156           (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) =
157         pairCoefficient b * pairCoordinate ^ (N + a) * pairMomentum ^ j := by
158           calc
159             pairCoordinate ^ N *
160                 (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ j) =
161               (pairCoordinate ^ N * pairCoefficient b) *
162                 pairCoordinate ^ a * pairMomentum ^ j := by
163                   simp only [mul_assoc]
164             _ = (pairCoefficient b * pairCoordinate ^ N) *
165                 pairCoordinate ^ a * pairMomentum ^ j := by
166                   rw [pairCoordinate_pow_mul_coefficient B b N]
167             _ = pairCoefficient b * pairCoordinate ^ (N + a) *
168                 pairMomentum ^ j := by
169                   rw [pow_add]
170                   simp only [mul_assoc]
171       _ = (V : PairStage (B := B)) * pairCoordinate := hV

173 private lemma coordinateCoefficientTerm_hasPositive
174     (c : CoordinateStage (B := B)) (j N : ℕ) (hjN : j < N) :
175     HasPositiveCoordinateFactor B N
176       (normalCoefficient coordinateDerivation c * pairMomentum ^ j) := by
177   let q := (normalFormAddEquiv (zeroDerivation (B := B))).symm c
178   have hc : c = normalForm (zeroDerivation (B := B)) q := by
179     exact (normalFormAddEquiv (zeroDerivation (B := B))).apply_symm_apply c |>.symm
180   rw [hc]
181   induction q using Polynomial.induction_on' with
182   | add q r hq hr =>
183       rw [normalForm_add,
184         (normalCoefficient (coordinateDerivation (B := B))).map_add, add_mul]
185       exact hasPositiveCoordinateFactor_add (B := B) hq hr
186   | monomial a b =>
187       rw [normalForm_monomial,
188         (normalCoefficient (coordinateDerivation (B := B))).map_mul,
189         (normalCoefficient (coordinateDerivation (B := B))).map_pow]
190       exact pairOrderedMonomial_hasPositiveCoordinateFactor B b a j N hjN

192 /-- Every element whose outer momentum support lies below `N` has both
193 positive right-coordinate factorizations. -/
194 theorem hasPositiveCoordinateFactor_of_outer_support
195     (z : PairStage (B := B)) (N : ℕ)
196     (hsupport : ∀ j ∈
197       ((normalFormAddEquiv coordinateDerivation).symm z).support, j < N) :
198     HasPositiveCoordinateFactor B N z := by
199   let q := (normalFormAddEquiv coordinateDerivation).symm z
200   have hz : z = normalForm coordinateDerivation q := by
201     exact (normalFormAddEquiv coordinateDerivation).apply_symm_apply z |>.symm
202   rw [hz, ← Polynomial.sum_monomial_eq q]
203   change HasPositiveCoordinateFactor B N
204     (normalFormAddHom (coordinateDerivation (B := B))
205       (∑ j ∈ q.support, Polynomial.monomial j (q.coeff j)))
206   rw [map_sum]
207   refine Finset.sum_induction (M := PairStage (B := B))
208     (fun j => normalFormAddHom (coordinateDerivation (B := B))
209       (Polynomial.monomial j (q.coeff j)))
210     (HasPositiveCoordinateFactor B N) ?_ ?_ ?_
211   · intro a b ha hb
212     exact hasPositiveCoordinateFactor_add (B := B) ha hb
213   · exact hasPositiveCoordinateFactor_zero B N
214   · intro j hj
215     change HasPositiveCoordinateFactor B N
216       (normalForm (coordinateDerivation (B := B))
217         (Polynomial.monomial j (q.coeff j)))
218     rw [normalForm_monomial]
219     exact coordinateCoefficientTerm_hasPositive B (q.coeff j) j N
220       (hsupport j hj)

223 lemma support_sub_X_pow_lt
224     (H : Polynomial (CoordinateStage (B := B))) (N : ℕ)
225     (hN : H.coeff N = 1)
226     (hgt : ∀ j, N < j → H.coeff j = 0) :
227     ∀ j ∈ (H - Polynomial.X ^ N).support, j < N := by
228   intro j hj
229   by_contra hnot
230   have hle : N ≤ j := Nat.le_of_not_gt hnot
231   have hcoeff : (H - Polynomial.X ^ N).coeff j = 0 := by
232     rcases hle.eq_or_lt with rfl | hlt
233     · simp [hN]
234     · rw [Polynomial.coeff_sub, hgt j hlt]
235       simp [Polynomial.coeff_X_pow, hlt.ne']
236   exact (Polynomial.mem_support_iff.mp hj) hcoeff

238 /-- Subtracting the monic outer power leaves an element to which the positive
239 factor theorem applies. -/
240 theorem normalForm_sub_X_pow_hasPositiveCoordinateFactor
241     (H : Polynomial (CoordinateStage (B := B))) (N : ℕ)
242     (hN : H.coeff N = 1)
243     (hgt : ∀ j, N < j → H.coeff j = 0) :
244     HasPositiveCoordinateFactor B N
245       (normalForm (coordinateDerivation (B := B))
246         (H - Polynomial.X ^ N)) := by
247   apply hasPositiveCoordinateFactor_of_outer_support B
248   intro j hj
249   change j ∈ ((normalFormAddEquiv (coordinateDerivation (B := B))).symm
250     ((normalFormAddEquiv (coordinateDerivation (B := B)))
251       (H - Polynomial.X ^ N))).support at hj
252   rw [(normalFormAddEquiv (coordinateDerivation (B := B))).symm_apply_apply] at hj
253   exact support_sub_X_pow_lt B H N hN hgt j hj

255 /-- The positive-factor interface turns the explicit Euler residue into the
256 exact shaped residue consumed by quotient surjectivity. -/
257 theorem positiveEulerResidue_eq_eulerSubring_mul_coordinate
258     (d sigma : PairStage (B := B)) (N : ℕ)
259     (hd : d = pairMomentum ^ N + sigma)
260     (hpositive : HasPositiveCoordinateFactor B N sigma) :
261     ∃ U : pairEulerSubring B,
262       1 + (U : PairStage (B := B)) * pairCoordinate ∈
263         canonicalRightIdeal pairCoordinate d N := by
264   obtain ⟨u, v, hmem⟩ := positiveEulerResidue_polynomial_mem
265     pairCoordinate pairMomentum d sigma N pairMomentum_mul_coordinate hd
266   obtain ⟨U, V, hright, hleft⟩ := hpositive
267   obtain ⟨su, hsu⟩ := coordinate_mul_eulerPolynomial
268     (pairOldSubring B) pairCoordinate pairMomentum
269     pairMomentum_mul_coordinate (pairOldSubring_commutes_coordinate B)
270     (rational_mem_pairOldSubring B) u
271   obtain ⟨sv, hsv⟩ := coordinate_mul_eulerPolynomial
272     (pairOldSubring B) pairCoordinate pairMomentum
273     pairMomentum_mul_coordinate (pairOldSubring_commutes_coordinate B)
274     (rational_mem_pairOldSubring B) v
275   refine ⟨U * su + V * sv, ?_⟩
276   convert hmem using 1
277   simp only [Subring.coe_add, Subring.coe_mul, add_mul]
278   rw [hright, hleft]
279   simp only [mul_assoc]
280   rw [hsu, hsv]

369 /-- The concrete remainder of a normalized presented Weyl operator satisfies
370 both positive-coordinate factorizations. -/
371 theorem presentedRemainder_hasPositiveCoordinateFactor
372     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
373     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
374     HasPositiveCoordinateFactor (IteratedPairStage k n) N
375       (presentedToIterated k (n + 1) d - stageMomentum k n ^ N) := by
376   let H := presentedOuterPolynomial k n d
377   have hpositive := normalForm_sub_X_pow_hasPositiveCoordinateFactor
378     (IteratedPairStage k n) H N
379     (outer_coeff_eq_one_at_bound k n N hd)
380     (fun j hj => outer_coeff_eq_zero_of_exponent_gt k n N hd hj)
381   have hnormal :
382       normalForm (coordinateDerivation (B := IteratedPairStage k n))
383           (H - Polynomial.X ^ N) =
384         presentedToIterated k (n + 1) d - stageMomentum k n ^ N := by
385     change normalFormAddHom coordinateDerivation (H - Polynomial.X ^ N) = _
386     have hH : normalForm (coordinateDerivation (B := IteratedPairStage k n)) H =
387         presentedToIterated k (n + 1) d := by
388       exact (normalFormAddEquiv coordinateDerivation).apply_symm_apply _
389     have hH' : normalFormAddHom coordinateDerivation H =
390         presentedToIterated k (n + 1) d := hH
391     rw [map_sub, hH']
392     congr 1
393     rw [Polynomial.X_pow_eq_monomial]
394     change normalForm coordinateDerivation (Polynomial.monomial N 1) = _
395     rw [normalForm_monomial]
396     rw [map_one, one_mul]
397     change pairMomentum (B := IteratedPairStage k n) ^ N =
398       stageMomentum k n ^ N
399     rfl
400   rw [hnormal] at hpositive
401   exact hpositive

403 /-- The normalized presented operator has the exact positive Euler residue in
404 its literal canonical right ideal. -/
405 theorem presentedPositiveEulerResidue
406     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
407     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
408     ∃ U : pairEulerSubring (IteratedPairStage k n),
409       1 + (U : PairStage (B := IteratedPairStage k n)) * pairCoordinate ∈
410         canonicalRightIdeal (pairCoordinate (B := IteratedPairStage k n))
411           (presentedToIterated k (n + 1) d) N := by
412   let sigma := presentedToIterated k (n + 1) d - stageMomentum k n ^ N
413   have hdecomp : presentedToIterated k (n + 1) d =
414       stageMomentum k n ^ N + sigma := by
415     simp [sigma]
416   have hpositive : HasPositiveCoordinateFactor (IteratedPairStage k n) N sigma :=
417     presentedRemainder_hasPositiveCoordinateFactor k n N hd
418   exact positiveEulerResidue_eq_eulerSubring_mul_coordinate
419     (IteratedPairStage k n) (presentedToIterated k (n + 1) d) sigma N
420       hdecomp hpositive
