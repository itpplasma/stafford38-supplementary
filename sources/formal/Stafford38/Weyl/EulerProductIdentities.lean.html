20 /-- Public names specialize the shared coefficient-ring factorial polynomials. -/
21 abbrev fallingEulerPolynomial : ℕ → Polynomial ℚ :=
22   Stafford38.Evolution.fallingPoly ℚ
23 
24 abbrev risingEulerPolynomial : ℕ → Polynomial ℚ :=
25   Stafford38.Evolution.risingPoly ℚ
26 
27 /-- The falling polynomial is the product of the Euler factors `X-i`. -/
28 theorem fallingEulerPolynomial_eq_prod (n : ℕ) :
29     fallingEulerPolynomial n = ∏ i ∈ Finset.range n,
30       (Polynomial.X - Polynomial.C (i : ℚ)) := by
31   change Stafford38.Evolution.fallingPoly ℚ n = _
32   induction n with
33   | zero => simp [Stafford38.Evolution.fallingPoly]
34   | succ n ih =>
35       rw [Stafford38.Evolution.fallingPoly_succ, ih, Finset.prod_range_succ]
36 
37 /-- The rising polynomial is the product of the Euler factors `X+(i+1)`. -/
38 theorem risingEulerPolynomial_eq_prod (n : ℕ) :
39     risingEulerPolynomial n = ∏ i ∈ Finset.range n,
40       (Polynomial.X + Polynomial.C ((i + 1 : ℕ) : ℚ)) := by
41   change Stafford38.Evolution.risingPoly ℚ n = _
42   induction n with
43   | zero => simp [Stafford38.Evolution.risingPoly]
44   | succ n ih =>
45       rw [Stafford38.Evolution.risingPoly_succ, ih, Finset.prod_range_succ]
46       push_cast
47       rfl

49 /-- Evaluation of the explicit falling Euler product gives `x^n * d^n`. -/
50 theorem eulerPolynomialEval_falling (x d : A)
51     (h : d * x = x * d + 1) (n : ℕ) :
52     Stafford.eulerPolynomialEval x d (fallingEulerPolynomial n) = x ^ n * d ^ n := by
53   change Polynomial.aeval (Stafford38.Evolution.euler x d)
54     (Stafford38.Evolution.fallingPoly ℚ n) = x ^ n * d ^ n
55   rw [Stafford38.Evolution.aeval_fallingPoly]
56   simpa [Stafford38.Evolution.euler] using
57     (Stafford38.Evolution.pow_mul_pow_eq_falling (x := x) (p := d) h n).symm
58 
59 /-- Evaluation of the explicit rising Euler product gives `d^n * x^n`. -/
60 theorem eulerPolynomialEval_rising (x d : A)
61     (h : d * x = x * d + 1) (n : ℕ) :
62     Stafford.eulerPolynomialEval x d (risingEulerPolynomial n) = d ^ n * x ^ n := by
63   change Polynomial.aeval (Stafford38.Evolution.euler x d)
64     (Stafford38.Evolution.risingPoly ℚ n) = d ^ n * x ^ n
65   rw [Stafford38.Evolution.aeval_risingPoly]
66   simpa [Stafford38.Evolution.euler] using
67     (Stafford38.Evolution.pow_mul_pow_eq_rising (x := x) (p := d) h n).symm
68 
69 /-- Evaluation of the falling product written with a finite-product sign. -/
70 theorem eval_fallingEulerProduct (x d : A)
71     (h : d * x = x * d + 1) (n : ℕ) :
72     Stafford.eulerPolynomialEval x d
73       (∏ i ∈ Finset.range n, (Polynomial.X - Polynomial.C (i : ℚ))) =
74         x ^ n * d ^ n := by
75   rw [← fallingEulerPolynomial_eq_prod n]
76   exact eulerPolynomialEval_falling x d h n
77 
78 /-- Evaluation of the rising product written with a finite-product sign. -/
79 theorem eval_risingEulerProduct (x d : A)
80     (h : d * x = x * d + 1) (n : ℕ) :
81     Stafford.eulerPolynomialEval x d
82       (∏ i ∈ Finset.range n,
83         (Polynomial.X + Polynomial.C ((i + 1 : ℕ) : ℚ))) =
84           d ^ n * x ^ n := by
85   rw [← risingEulerPolynomial_eq_prod n]
86   exact eulerPolynomialEval_rising x d h n
