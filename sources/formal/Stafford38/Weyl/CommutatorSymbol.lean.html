31 /-- Commutation with the newest coordinate differentiates the momentum power
32 in an exact ordered PBW block, with the sign dictated by `[x,p] = -1`. -/
33 theorem presentedCoordinate_commutator_coefficientOrdered
34     (n a p : ℕ) (z : WeylAlg k n) :
35     AlgebraicAnalysis.ringCommutator (presentedCoordinate k n)
36         (presentedCoefficientOrdered k n a p z) =
37       -((p : k) • presentedCoefficientOrdered k n a (p - 1) z) := by
38   let X := presentedCoordinate k n
39   let P := presentedMomentum k n
40   let Z := previousWeylEmbedding k n z
41   have hXZ : X * Z = Z * X :=
42     presentedCoordinate_commutes_previous k n z
43   have hPX : P * X = X * P + 1 := presentedMomentum_mul_coordinate k n
44   have hpowSucc : ∀ r : ℕ,
45       X * P ^ (r + 1) - P ^ (r + 1) * X = -((r + 1) • P ^ r) := by
46     intro r
47     let Xop : (WeylAlg k (n + 1))ᵐᵒᵖ := MulOpposite.op X
48     let Pop : (WeylAlg k (n + 1))ᵐᵒᵖ := MulOpposite.op P
49     have hop : Xop * Pop - Pop * Xop = 1 := by
50       have hopEq := congrArg MulOpposite.op hPX
51       change MulOpposite.op X * MulOpposite.op P -
52         MulOpposite.op P * MulOpposite.op X = 1
53       rw [show MulOpposite.op X * MulOpposite.op P =
54           MulOpposite.op P * MulOpposite.op X + 1 by simpa using hopEq]
55       abel
56     have ht := AlgebraicAnalysis.OreDivision.OreAmbient.commutator_pow_succ
57       Xop Pop hop r
58     have hu := congrArg MulOpposite.unop ht
59     change P ^ (r + 1) * X - X * P ^ (r + 1) = (r + 1) • P ^ r at hu
60     rw [← hu]
61     abel
62   have hpow : ∀ q : ℕ,
63       X * P ^ q - P ^ q * X = -((q : k) • P ^ (q - 1)) := by
64     intro q
65     cases q with
66     | zero => simp
67     | succ r =>
68         rw [Nat.cast_smul_eq_nsmul k]
69         exact hpowSucc r
70   rw [presentedCoefficientOrdered]
71   change X * (Z * X ^ a * P ^ p) - (Z * X ^ a * P ^ p) * X = _
72   let Q := Z * X ^ a
73   have hXpow : X * X ^ a = X ^ a * X := (Commute.refl X).pow_right a
74   have hQ : AlgebraicAnalysis.ringCommutator X Q = 0 := by
75     change X * (Z * X ^ a) - (Z * X ^ a) * X = 0
76     rw [show X * (Z * X ^ a) = (Z * X ^ a) * X by
77       calc
78         X * (Z * X ^ a) = (X * Z) * X ^ a := by rw [mul_assoc]
79         _ = (Z * X) * X ^ a := by rw [hXZ]
80         _ = Z * (X * X ^ a) := by rw [mul_assoc]
81         _ = Z * (X ^ a * X) := by rw [hXpow]
82         _ = (Z * X ^ a) * X := by rw [mul_assoc]]
83     exact sub_self _
84   change AlgebraicAnalysis.ringCommutator X (Q * P ^ p) = _
85   calc
86     AlgebraicAnalysis.ringCommutator X (Q * P ^ p) =
87         AlgebraicAnalysis.ringCommutator X Q * P ^ p +
88           Q * AlgebraicAnalysis.ringCommutator X (P ^ p) := by
89             simp only [AlgebraicAnalysis.ringCommutator,
90               AlgebraicAnalysis.ringCommutator, sub_mul, mul_sub, mul_assoc]
91             abel
92     _ = Q * (X * P ^ p - P ^ p * X) := by
93       have hQ' : X * Q - Q * X = 0 := by exact hQ
94       change (X * Q - Q * X) * P ^ p +
95           Q * (X * P ^ p - P ^ p * X) = _
96       rw [hQ', zero_mul, zero_add]
97     _ = Q * -((p : k) • P ^ (p - 1)) := by rw [hpow]
98     _ = -((p : k) • (Q * P ^ (p - 1))) := by
99       calc
100         Q * -((p : k) • P ^ (p - 1)) =
101             -(Q * ((p : k) • P ^ (p - 1))) := by
102               simpa only using (mul_neg Q ((p : k) • P ^ (p - 1)))
103         _ = -((p : k) • (Q * P ^ (p - 1))) :=
104           congrArg Neg.neg
105             (Algebra.mul_smul_comm (p : k) Q (P ^ (p - 1)))

150 /-- Generator/ordered-block case of the order-symbol ringCommutator formula.
151 The exact degree is `N+p-1`; positivity of `p` prevents truncated subtraction.
152 The sign records the repository convention `[x,p] = -1`. -/
153 theorem principalComponent_coordinate_commutator_eq_neg_poisson
154     (n N a p : ℕ) (z : WeylAlg k n) (hp : 0 < p) :
155     presentedPrincipalComponent k (@orderWeight (n + 1)) (N + p - 1)
156         (AlgebraicAnalysis.ringCommutator (presentedCoordinate k n)
157           (presentedCoefficientOrdered k n a p z)) =
158       -poissonBracket (MvPolynomial.X (.inl (0 : Fin (n + 1))))
159         (presentedPrincipalComponent k (@orderWeight (n + 1)) (N + p)
160           (presentedCoefficientOrdered k n a p z)) := by
161   rw [presentedCoordinate_commutator_coefficientOrdered]
162   simp only [map_neg, map_smul]
163   rw [show N + p - 1 = N + (p - 1) by omega,
164     presentedPrincipalComponent_coefficientOrdered_order,
165     presentedPrincipalComponent_coefficientOrdered_order,
166     poissonBracket_newestCoordinate,
167     MvPolynomial.pderiv_mul,
168     MvPolynomial.pderiv_monomial,
169     pderiv_rename_oldIndex_newestMomentum,
170     mul_zero, add_zero,
171     extendPhaseExponent_sub_newestMomentum n a p]
172   have hexp : (extendPhaseExponent n a p 0)
173       (.inr (0 : Fin (n + 1))) = p := by
174     simp [extendPhaseExponent, Stafford38FixedSourceChallenge.oldIndex]
175   rw [hexp]
176   congr 1
177   rw [← Algebra.smul_mul_assoc, MvPolynomial.smul_monomial]
178   simp
