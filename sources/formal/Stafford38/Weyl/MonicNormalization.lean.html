8 /-!
9 # Bernstein-top selection and scalar monic normalization
10 
11 The checked PBW normal form selects the actual top Bernstein degree of every
12 nonzero presented Weyl element. Degree zero is exactly the scalar case. In
13 positive degree, the explicit symplectic chart is applied and the transformed
14 operator is scaled by the inverse of its nonzero pure-power coefficient. The
15 normalized principal axis coefficient is then exactly one.
16 
17 This file stops at the commutative-symbol monicity interface. Identifying it
18 with a monic Ore polynomial in the selected momentum is the next dependency.
19 -/

37 theorem weightedHomogeneousComponent_weightedTotalDegree_ne_zero
38     {σ : Type*} [DecidableEq σ] (w : σ → ℕ)
39     {f : MvPolynomial σ k} (hf : f ≠ 0) :
40     MvPolynomial.weightedHomogeneousComponent w
41         (MvPolynomial.weightedTotalDegree w f) f ≠ 0 := by
42   have hs : f.support.Nonempty := MvPolynomial.support_nonempty.mpr hf
43   rcases Finset.exists_mem_eq_sup f.support hs (Finsupp.weight w) with
44     ⟨m, hm, hmax⟩
45   intro hzero
46   have hc := congrArg (fun p : MvPolynomial σ k => AddMonoidAlgebra.coeff p m) hzero
47   change AddMonoidAlgebra.coeff
48     (MvPolynomial.weightedHomogeneousComponent w
49       (MvPolynomial.weightedTotalDegree w f) f) m = 0 at hc
50   rw [MvPolynomial.coeff_weightedHomogeneousComponent] at hc
51   have hw : Finsupp.weight w m = MvPolynomial.weightedTotalDegree w f := by
52     simpa [MvPolynomial.weightedTotalDegree] using hmax.symm
53   rw [if_pos hw] at hc
54   exact (MvPolynomial.mem_support_iff.mp hm) hc

56 theorem principal_isHomogeneous {n N : ℕ} (d : PresentedWeyl k n) :
57     (presentedPrincipalComponent k (@bernsteinWeight n) N d).IsHomogeneous N := by
58   change MvPolynomial.IsWeightedHomogeneous (1 : PhaseVar n → ℕ)
59     (presentedPrincipalComponent k (@bernsteinWeight n) N d) N
60   have hweight : (@bernsteinWeight n) = (1 : PhaseVar n → ℕ) := by
61     funext i
62     rfl
63   rw [← hweight]
64   simpa [presentedPrincipalComponent] using
65     (MvPolynomial.weightedHomogeneousComponent_isWeightedHomogeneous
66       (w := @bernsteinWeight n) (n := N)
67       (φ := presentedNormalFormLinearEquiv k n d))

69 theorem exists_top_bernstein_piece {n : ℕ} {d : PresentedWeyl k n}
70     (hd : d ≠ 0) :
71     ∃ N : ℕ, d ∈ bernsteinPiece k n N ∧
72       presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0 := by
73   let f := presentedNormalFormLinearEquiv k n d
74   have hf : f ≠ 0 := by
75     intro hzero
76     apply hd
77     apply (presentedNormalFormLinearEquiv k n).injective
78     simpa [f] using hzero
79   let N := MvPolynomial.weightedTotalDegree (@bernsteinWeight n) f
80   refine ⟨N, ?_, ?_⟩
81   · change d ∈ presentedWeightPiece k (@bernsteinWeight n) N
82     rw [mem_presentedWeightPiece]
83     intro m hm
84     simpa [f, N, monomialWeight, Finsupp.weight_apply, smul_eq_mul] using
85       MvPolynomial.le_weightedTotalDegree (@bernsteinWeight n)
86         (MvPolynomial.mem_support_iff.mpr hm)
87   · exact weightedHomogeneousComponent_weightedTotalDegree_ne_zero k
88       (@bernsteinWeight n) hf

90 theorem eq_algebraMap_of_mem_bernsteinPiece_zero {n : ℕ}
91     {d : PresentedWeyl k n} (hd : d ∈ bernsteinPiece k n 0) :
92     ∃ c : k, d = algebraMap k (PresentedWeyl k n) c := by
93   let f := presentedNormalFormLinearEquiv k n d
94   let c := f.coeff 0
95   have hf : f = MvPolynomial.C c := by
96     ext m
97     by_cases hm : m = 0
98     · subst m
99       simp [c]
100     · have hcoeff : AddMonoidAlgebra.coeff f m = 0 := by
101         by_contra hne
102         have hle := (mem_presentedWeightPiece k (@bernsteinWeight n) 0 d).mp hd m hne
103         have hdeg : m.degree = 0 := by
104           apply Nat.eq_zero_of_le_zero
105           rw [Finsupp.degree_eq_weight_one]
106           simpa [f, monomialWeight, bernsteinWeight,
107             Finsupp.weight_apply, smul_eq_mul] using hle
108         exact hm ((Finsupp.degree_eq_zero_iff m).mp hdeg)
109       change AddMonoidAlgebra.coeff f m = AddMonoidAlgebra.coeff (MvPolynomial.C c) m
110       rw [hcoeff]
111       simp [Ne.symm hm]
112   refine ⟨c, ?_⟩
113   apply (presentedNormalFormLinearEquiv k n).injective
114   rw [show presentedNormalFormLinearEquiv k n d = f by rfl, hf]
115   rw [← mul_one ((algebraMap k (PresentedWeyl k n)) c),
116     ← Algebra.smul_def]
117   change MvPolynomial.C c =
118     presentedNormalFormLinearEquiv k n (c • (1 : PresentedWeyl k n))
119   rw [map_smul, presentedNormalFormLinearEquiv_one]
120   exact MvPolynomial.C_eq_smul_one

122 theorem scalar_or_positive_top_bernstein_piece {n : ℕ}
123     {d : PresentedWeyl k n} (hd : d ≠ 0) :
124     (∃ c : k, c ≠ 0 ∧ d = algebraMap k (PresentedWeyl k n) c) ∨
125       ∃ N : ℕ, 0 < N ∧ d ∈ bernsteinPiece k n N ∧
126         presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0 := by
127   rcases exists_top_bernstein_piece k hd with ⟨N, hpiece, hprincipal⟩
128   by_cases hN : N = 0
129   · subst N
130     left
131     rcases eq_algebraMap_of_mem_bernsteinPiece_zero k hpiece with ⟨c, hdc⟩
132     refine ⟨c, ?_, hdc⟩
133     intro hc
134     subst c
135     simp at hdc
136     exact hd hdc
137   · right
138     exact ⟨N, Nat.pos_of_ne_zero hN, hpiece, hprincipal⟩

140 def normalizedSymplecticImage {n : ℕ}
141     (M : Matrix (PhaseVar n) (PhaseVar n) k)
142     (hM : M * standardForm k n * Matrix.transpose M = standardForm k n)
143     (c : k) (d : PresentedWeyl k n) : PresentedWeyl k n :=
144   c⁻¹ • standardSymplecticAlgHom k M hM d

146 def HasNormalizedSymplecticChart {n : ℕ} (t : PhaseVar n) (N : ℕ)
147     (d : PresentedWeyl k n) : Prop :=
148   ∃ (M Ninv : Matrix (PhaseVar n) (PhaseVar n) k) (c : k),
149     ∃ hM : M * standardForm k n * Matrix.transpose M = standardForm k n,
150     Ninv * standardForm k n * Matrix.transpose Ninv = standardForm k n ∧
151     M * Ninv = 1 ∧ Ninv * M = 1 ∧ c ≠ 0 ∧
152     normalizedSymplecticImage k M hM c d ∈ bernsteinPiece k n N ∧
153     MvPolynomial.coeff (Finsupp.single () N)
154         (axisPolynomial k t
155           (presentedPrincipalComponent k (@bernsteinWeight n) N
156             (normalizedSymplecticImage k M hM c d))) = 1

158 theorem exists_normalized_symplectic_image [CharZero k]
159     {n N : ℕ} (t : PhaseVar n) {d : PresentedWeyl k n}
160     (hd : d ∈ bernsteinPiece k n N)
161     (hP : (presentedPrincipalComponent k (@bernsteinWeight n) N d).IsHomogeneous N)
162     (hPne : presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0)
163     (hN : 0 < N) :
164     HasNormalizedSymplecticChart k t N d := by
165   rcases exists_symplectic_chart_matrices k t hP hPne hN with
166     ⟨M, Ninv, hM, hNinv, hMN, hNM, hc⟩
167   let c : k := MvPolynomial.coeff (Finsupp.single () N)
168     (axisPolynomial k t
169       (Stafford38.CharacteristicLinearAction.symbolLinearAlgHom k M
170         (presentedPrincipalComponent k (@bernsteinWeight n) N d)))
171   have hc' : c ≠ 0 := hc
172   refine ⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc', ?_, ?_⟩
173   · exact (bernsteinPiece k n N).smul_mem _
174       (standardSymplecticAlgHom_preserves_bernsteinPiece k M hM hd)
175   · rw [normalizedSymplecticImage, map_smul]
176     rw [standardSymplecticAlgHom_principal_compatibility k M hM hd]
177     simp only [map_smul, MvPolynomial.coeff_smul]
178     change c⁻¹ * c = 1
179     exact inv_mul_cancel₀ hc'

181 theorem scalar_or_normalized_symplectic_image [CharZero k]
182     {n : ℕ} (hn : 0 < n) {d : PresentedWeyl k n} (hd : d ≠ 0) :
183     (∃ c : k, c ≠ 0 ∧ d = algebraMap k (PresentedWeyl k n) c) ∨
184       ∃ N : ℕ, 0 < N ∧
185         HasNormalizedSymplecticChart k (.inr ⟨0, hn⟩) N d := by
186   rcases scalar_or_positive_top_bernstein_piece k hd with hscalar | htop
187   · exact Or.inl hscalar
188   · rcases htop with ⟨N, hN, hpiece, hprincipal⟩
189     right
190     refine ⟨N, hN, ?_⟩
191     exact exists_normalized_symplectic_image k (.inr ⟨0, hn⟩) hpiece
192       (principal_isHomogeneous k d) hprincipal hN
