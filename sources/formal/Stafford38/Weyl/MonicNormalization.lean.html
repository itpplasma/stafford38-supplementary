12 /-!
13 # Bernstein-top selection and scalar monic normalization
14 
15 The checked PBW normal form selects the actual top Bernstein degree of every
16 nonzero presented Weyl element. Degree zero is exactly the scalar case. In
17 positive degree, the explicit symplectic chart is applied and the transformed
18 operator is scaled by the inverse of its nonzero pure-power coefficient. The
19 normalized principal axis coefficient is then exactly one.
20 
21 This file stops at the commutative-symbol monicity interface. Identifying it
22 with a monic Ore polynomial in the selected momentum is the next dependency.
23 -/

41 theorem weightedHomogeneousComponent_weightedTotalDegree_ne_zero
42     {σ : Type*} [DecidableEq σ] (w : σ → ℕ)
43     {f : MvPolynomial σ k} (hf : f ≠ 0) :
44     MvPolynomial.weightedHomogeneousComponent w
45         (MvPolynomial.weightedTotalDegree w f) f ≠ 0 := by
46   have hs : f.support.Nonempty := MvPolynomial.support_nonempty.mpr hf
47   rcases Finset.exists_mem_eq_sup f.support hs (Finsupp.weight w) with
48     ⟨m, hm, hmax⟩
49   intro hzero
50   have hc := congrArg (fun p : MvPolynomial σ k => AddMonoidAlgebra.coeff p m) hzero
51   change AddMonoidAlgebra.coeff
52     (MvPolynomial.weightedHomogeneousComponent w
53       (MvPolynomial.weightedTotalDegree w f) f) m = 0 at hc
54   rw [MvPolynomial.coeff_weightedHomogeneousComponent] at hc
55   have hw : Finsupp.weight w m = MvPolynomial.weightedTotalDegree w f := by
56     simpa [MvPolynomial.weightedTotalDegree] using hmax.symm
57   rw [if_pos hw] at hc
58   exact (MvPolynomial.mem_support_iff.mp hm) hc

60 theorem principal_isHomogeneous {n N : ℕ} (d : WeylAlg k n) :
61     (presentedPrincipalComponent k (@bernsteinWeight n) N d).IsHomogeneous N := by
62   change MvPolynomial.IsWeightedHomogeneous (1 : PhaseVar n → ℕ)
63     (presentedPrincipalComponent k (@bernsteinWeight n) N d) N
64   have hweight : (@bernsteinWeight n) = (1 : PhaseVar n → ℕ) := by
65     funext i
66     rfl
67   rw [← hweight]
68   simpa [presentedPrincipalComponent] using
69     (MvPolynomial.weightedHomogeneousComponent_isWeightedHomogeneous
70       (w := @bernsteinWeight n) (n := N)
71       (φ := presentedNormalFormLinearEquiv k n d))

73 theorem exists_top_bernstein_piece {n : ℕ} {d : WeylAlg k n}
74     (hd : d ≠ 0) :
75     ∃ N : ℕ, d ∈ bernsteinPiece k n N ∧
76       presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0 := by
77   let f := presentedNormalFormLinearEquiv k n d
78   have hf : f ≠ 0 := by
79     intro hzero
80     apply hd
81     apply (presentedNormalFormLinearEquiv k n).injective
82     simpa [f] using hzero
83   let N := MvPolynomial.weightedTotalDegree (@bernsteinWeight n) f
84   refine ⟨N, ?_, ?_⟩
85   · change d ∈ presentedWeightPiece k (@bernsteinWeight n) N
86     rw [mem_presentedWeightPiece]
87     intro m hm
88     simpa [f, N, Finsupp.weight_apply, Finsupp.weight_apply, smul_eq_mul] using
89       MvPolynomial.le_weightedTotalDegree (@bernsteinWeight n)
90         (MvPolynomial.mem_support_iff.mpr hm)
91   · exact weightedHomogeneousComponent_weightedTotalDegree_ne_zero k
92       (@bernsteinWeight n) hf

94 theorem eq_algebraMap_of_mem_bernsteinPiece_zero {n : ℕ}
95     {d : WeylAlg k n} (hd : d ∈ bernsteinPiece k n 0) :
96     ∃ c : k, d = algebraMap k (WeylAlg k n) c := by
97   let f := presentedNormalFormLinearEquiv k n d
98   let c := f.coeff 0
99   have hf : f = MvPolynomial.C c := by
100     ext m
101     by_cases hm : m = 0
102     · subst m
103       simp [c]
104     · have hcoeff : AddMonoidAlgebra.coeff f m = 0 := by
105         by_contra hne
106         have hle := (mem_presentedWeightPiece k (@bernsteinWeight n) 0 d).mp hd m hne
107         have hdeg : m.degree = 0 := by
108           apply Nat.eq_zero_of_le_zero
109           rw [Finsupp.degree_eq_weight_one]
110           simpa [f, Finsupp.weight_apply, bernsteinWeight,
111             Finsupp.weight_apply, smul_eq_mul] using hle
112         exact hm ((Finsupp.degree_eq_zero_iff m).mp hdeg)
113       change AddMonoidAlgebra.coeff f m = AddMonoidAlgebra.coeff (MvPolynomial.C c) m
114       rw [hcoeff]
115       simp [Ne.symm hm]
116   refine ⟨c, ?_⟩
117   apply (presentedNormalFormLinearEquiv k n).injective
118   rw [show presentedNormalFormLinearEquiv k n d = f by rfl, hf]
119   rw [← mul_one ((algebraMap k (WeylAlg k n)) c),
120     ← Algebra.smul_def]
121   change MvPolynomial.C c =
122     presentedNormalFormLinearEquiv k n (c • (1 : WeylAlg k n))
123   rw [map_smul, presentedNormalFormLinearEquiv_one]
124   exact MvPolynomial.C_eq_smul_one

126 theorem scalar_or_positive_top_bernstein_piece {n : ℕ}
127     {d : WeylAlg k n} (hd : d ≠ 0) :
128     (∃ c : k, c ≠ 0 ∧ d = algebraMap k (WeylAlg k n) c) ∨
129       ∃ N : ℕ, 0 < N ∧ d ∈ bernsteinPiece k n N ∧
130         presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0 := by
131   rcases exists_top_bernstein_piece k hd with ⟨N, hpiece, hprincipal⟩
132   by_cases hN : N = 0
133   · subst N
134     left
135     rcases eq_algebraMap_of_mem_bernsteinPiece_zero k hpiece with ⟨c, hdc⟩
136     refine ⟨c, ?_, hdc⟩
137     intro hc
138     subst c
139     simp at hdc
140     exact hd hdc
141   · right
142     exact ⟨N, Nat.pos_of_ne_zero hN, hpiece, hprincipal⟩

144 def normalizedSymplecticImage {n : ℕ}
145     (M : Matrix (PhaseVar n) (PhaseVar n) k)
146     (hM : M * standardForm k n * Matrix.transpose M = standardForm k n)
147     (c : k) (d : WeylAlg k n) : WeylAlg k n :=
148   c⁻¹ • standardSymplecticAlgHom k M hM d

150 def HasNormalizedSymplecticChart {n : ℕ} (t : PhaseVar n) (N : ℕ)
151     (d : WeylAlg k n) : Prop :=
152   ∃ (M Ninv : Matrix (PhaseVar n) (PhaseVar n) k) (c : k),
153     ∃ hM : M * standardForm k n * Matrix.transpose M = standardForm k n,
154     Ninv * standardForm k n * Matrix.transpose Ninv = standardForm k n ∧
155     M * Ninv = 1 ∧ Ninv * M = 1 ∧ c ≠ 0 ∧
156     normalizedSymplecticImage k M hM c d ∈ bernsteinPiece k n N ∧
157     MvPolynomial.coeff (Finsupp.single () N)
158         (axisPolynomial k t
159           (presentedPrincipalComponent k (@bernsteinWeight n) N
160             (normalizedSymplecticImage k M hM c d))) = 1

162 theorem exists_normalized_symplectic_image [CharZero k]
163     {n N : ℕ} (t : PhaseVar n) {d : WeylAlg k n}
164     (hd : d ∈ bernsteinPiece k n N)
165     (hP : (presentedPrincipalComponent k (@bernsteinWeight n) N d).IsHomogeneous N)
166     (hPne : presentedPrincipalComponent k (@bernsteinWeight n) N d ≠ 0)
167     (hN : 0 < N) :
168     HasNormalizedSymplecticChart k t N d := by
169   rcases exists_symplectic_chart_matrices k t hP hPne hN with
170     ⟨M, Ninv, hM, hNinv, hMN, hNM, hc⟩
171   let c : k := MvPolynomial.coeff (Finsupp.single () N)
172     (axisPolynomial k t
173       (Stafford38.CharacteristicLinearAction.symbolLinearAlgHom k M
174         (presentedPrincipalComponent k (@bernsteinWeight n) N d)))
175   have hc' : c ≠ 0 := hc
176   refine ⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc', ?_, ?_⟩
177   · exact (bernsteinPiece k n N).smul_mem _
178       (standardSymplecticAlgHom_preserves_bernsteinPiece k M hM hd)
179   · rw [normalizedSymplecticImage, map_smul]
180     rw [standardSymplecticAlgHom_principal_compatibility k M hM hd]
181     simp only [map_smul, MvPolynomial.coeff_smul]
182     change c⁻¹ * c = 1
183     exact inv_mul_cancel₀ hc'

185 theorem scalar_or_normalized_symplectic_image [CharZero k]
186     {n : ℕ} (hn : 0 < n) {d : WeylAlg k n} (hd : d ≠ 0) :
187     (∃ c : k, c ≠ 0 ∧ d = algebraMap k (WeylAlg k n) c) ∨
188       ∃ N : ℕ, 0 < N ∧
189         HasNormalizedSymplecticChart k (.inr ⟨0, hn⟩) N d := by
190   rcases scalar_or_positive_top_bernstein_piece k hd with hscalar | htop
191   · exact Or.inl hscalar
192   · rcases htop with ⟨N, hN, hpiece, hprincipal⟩
193     right
194     refine ⟨N, hN, ?_⟩
195     exact exists_normalized_symplectic_image k (.inr ⟨0, hn⟩) hpiece
196       (principal_isHomogeneous k d) hprincipal hN
