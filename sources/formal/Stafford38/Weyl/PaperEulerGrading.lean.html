8 /-!
9 # The paper's Euler grading and the concrete Euler subring
10 
11 The pair-stage PBW monomial `b * x^a * p^c` has Euler weight `a-c` for
12 `E = x*p`.  The nonnegative part is generated by the nonnegative PBW
13 monomials, and is exactly the subring generated by the transverse algebra,
14 `x`, and `E` used by the quotient argument.
15 -/

39   pairCoordinate * pairMomentum

41 /-- The ringCommutator operator `ad_E`, with the paper's `E = x*p`. -/
42 def eulerAdjoint (z : PairStage (B := B)) : PairStage (B := B) :=
43   eulerElement B * z - z * eulerElement B

45 omit [Algebra ℚ B] in
46 lemma eulerElement_mul_coefficient (b : B) :
47     eulerElement B * pairCoefficient b = pairCoefficient b * eulerElement B := by
48   dsimp [eulerElement]
49   calc
50     (pairCoordinate * pairMomentum) * pairCoefficient b =
51         pairCoordinate * (pairMomentum * pairCoefficient b) := by rw [mul_assoc]
52     _ = pairCoordinate * (pairCoefficient b * pairMomentum) := by
53       rw [pairMomentum_mul_coefficient]
54     _ = (pairCoordinate * pairCoefficient b) * pairMomentum := by
55       rw [← mul_assoc]
56     _ = (pairCoefficient b * pairCoordinate) * pairMomentum := by
57       rw [pairCoordinate_mul_coefficient]
58     _ = pairCoefficient b * (pairCoordinate * pairMomentum) := by
59       rw [mul_assoc]

62 lemma eulerAdjoint_coordinate :

113 theorem orderedPBWMonomial_eulerWeight (b : B) (a c : ℕ) :
114     eulerAdjoint B
115         (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ c) =
116       ((a : ℤ) - c) •
117         (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ c) := by
118   rw [eulerAdjoint_mul, eulerAdjoint_mul, eulerAdjoint_coefficient,
119     eulerAdjoint_coordinate_pow, eulerAdjoint_momentum_pow]
120   simp only [zero_mul, zero_add]
121   rw [natCast_sub_natCast_zsmul]
122   simp only [mul_smul_comm, smul_mul_assoc, mul_neg]
123   abel

125 /-! The PBW monomials with nonnegative Euler weight give a concrete generating
126 set for the existing transverse–Euler subring.  Keep the PBW description as a
127 theorem, so there is only one subring owner. -/
128 
129 def nonnegativeEulerPBWMonomials : Set (PairStage (B := B)) :=
130   {z | ∃ b a c, c ≤ a ∧
131     z = pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ c}
132 
133 lemma orderedPBWMonomial_mem_nonnegativeEulerPart
134     (b : B) (a c : ℕ) (hca : c ≤ a) :
135     pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ c ∈
136       pairEulerSubring B := by
137   exact pairOrderedMonomial_mem_eulerSubring B b a c hca
138 
139 theorem pairEulerSubring_eq_closure_nonnegativeEulerPBWMonomials :
140     pairEulerSubring B = Subring.closure (nonnegativeEulerPBWMonomials B) := by
141   apply le_antisymm
142   · change eulerSubring (pairOldSubring B) pairCoordinate pairMomentum ≤ _
143     apply Subring.closure_le.2
144     intro z hz
145     change z ∈ ((pairOldSubring B : Set (PairStage (B := B))) ∪
146       {pairCoordinate, pairCoordinate * pairMomentum}) at hz
147     simp only [Set.mem_union, Set.mem_insert_iff, Set.mem_singleton_iff] at hz
148     rcases hz with hzC | hx | hE
149     · obtain ⟨b, rfl⟩ := hzC
150       apply Subring.subset_closure
151       exact ⟨b, 0, 0, by omega, rfl⟩
152     · subst z
153       simpa [pairCoordinate, pairCoefficient] using
154         (Subring.subset_closure
155           (s := nonnegativeEulerPBWMonomials B) ⟨(1 : B), 1, 0, by omega, rfl⟩)
156     · subst z
157       simpa [pairCoordinate, pairCoefficient] using
158         (Subring.subset_closure
159           (s := nonnegativeEulerPBWMonomials B) ⟨(1 : B), 1, 1, by omega, rfl⟩)
160   · apply Subring.closure_le.2
161     rintro z ⟨b, a, c, hca, rfl⟩
162     exact pairOrderedMonomial_mem_eulerSubring B b a c hca

164 /-- Positive PBW Euler weight gives the right-coordinate factor used in the
165 paper's residue argument. -/
166 theorem orderedPBWMonomial_positive_eulerFactor
167     (b : B) (a c : ℕ) (hca : c < a) :
168     ∃ U : pairEulerSubring B,
169       pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ c =
170         (U : PairStage (B := B)) * pairCoordinate := by
171   obtain ⟨U, hU⟩ := pairOrderedMonomial_eq_eulerSubring_mul_coordinate
172     B b a c hca
173   exact ⟨U, hU⟩

178 /-- The generic quotient argument can be stated directly with the paper's
179 nonnegative PBW Euler part. -/
180 theorem rightMul_surjective_of_paperEuler_normal
181     (I : RightIdeal (PairStage (B := B)))
182     (hnormal : ∀ r : pairEulerSubring B,
183       ∃ s : pairEulerSubring B,
184         pairCoordinate * (r : PairStage (B := B)) = (s : PairStage (B := B)) * pairCoordinate)
185     (hE : pairCoordinate * pairMomentum ∈ pairEulerSubring B)
186     (hresidue : ∃ U : pairEulerSubring B,
187       1 + (U : PairStage (B := B)) * pairCoordinate ∈ I)
188     (hgenerate : Subring.closure
189       (((pairEulerSubring B : Subring (PairStage (B := B))) :
190         Set (PairStage (B := B))) ∪ {pairMomentum}) = ⊤) :
191     Function.Surjective (rightMul I pairCoordinate) := by
192   exact rightMul_surjective_of_euler_normal I (pairEulerSubring B)
193     pairCoordinate pairMomentum hnormal hE hresidue hgenerate

227 /-- The original Euler quotient-surjectivity route, now using the explicit
228 nonnegative PBW Euler part.  The normalized remainder residue is the existing
229 literal residue theorem; the Euler-subring/PBW dictionary above transports it
230 into the paper's grading language before the generic quotient argument. -/
231 theorem presentedCanonicalQuotient_rightMul_coordinate_surjective
232     (n N : ℕ) {d : WeylAlg k (n + 1)}
233     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
234     Function.Surjective
235       (rightMul
236         (canonicalRightIdeal (pairCoordinate (B := IteratedPairStage k n))
237           (presentedToIterated k (n + 1) d) N)
238         (pairCoordinate (B := IteratedPairStage k n))) := by
239   let B' := IteratedPairStage k n
240   apply rightMul_surjective_of_paperEuler_normal (B := B')
241     (I := canonicalRightIdeal (pairCoordinate (B := B'))
242       (presentedToIterated k (n + 1) d) N)
243   · intro r
244     let r' : pairEulerSubring B' := r
245     obtain ⟨s, hs⟩ := coordinate_normal_eulerSubring
246       (pairOldSubring B') pairCoordinate pairMomentum
247       (stageMomentum_mul_coordinate k n)
248       (pairOldSubring_commutes_coordinate B') r'
249     refine ⟨⟨s.1, ?_⟩, hs⟩
250     exact s.2
251   · exact euler_mem_eulerSubring (pairOldSubring B') pairCoordinate pairMomentum
252   · obtain ⟨U, hU⟩ := presentedPositiveEulerResidue k n N hd
253     refine ⟨⟨U.1, ?_⟩, hU⟩
254     change U.1 ∈ pairEulerSubring B'
255     exact U.2
256   · exact closure_eulerSubring_union_momentum_eq_top
257       (pairOldSubring B') pairCoordinate pairMomentum
258       (closure_pairOldSubring_union_coordinate_momentum_eq_top B')
259 
260 end Presented
