9 # Transposition of the presented Weyl algebra
10 
11 This file constructs the scalar-preserving Weyl transposition as an algebra
12 equivalence with the opposite algebra.  Coordinates are fixed and momenta
13 change sign.  The construction uses only the checked universal property of
14 the quotient presentation.
15 -/

39 /-- Generator images for the Weyl transposition. -/
40 def transposedGenerator (n : Nat) :
41     (Fin n ⊕ Fin n) → (PresentedWeyl k n)ᵐᵒᵖ
42   | .inl i => op (coordinate k n i)
43   | .inr i => op (-momentum k n i)

59 theorem underlying_transposedGenerator_commutator (n : Nat) :
60     ∀ i j,
61       -Stafford.commutator (unop (transposedGenerator k n i))
62           (unop (transposedGenerator k n j)) =
63         algebraMap k (PresentedWeyl k n) (Matrix.J (Fin n) k i j) := by
64   intro i j
65   cases i with
66   | inl i =>
67       cases j with
68       | inl j =>
69           change -Stafford.commutator (coordinate k n i) (coordinate k n j) = _
70           rw [show Stafford.commutator (coordinate k n i) (coordinate k n j) =
71             algebraMap k (PresentedWeyl k n)
72               (Matrix.J (Fin n) k (.inl i) (.inl j)) from
73             freeWeylGenerator_commutator
74               (k := k) (Matrix.J (Fin n) k) (.inl i) (.inl j)]
75           simp [Matrix.J]
76       | inr j =>
77           change -Stafford.commutator (coordinate k n i) (-momentum k n j) = _
78           have h := freeWeylGenerator_commutator
79             (k := k) (Matrix.J (Fin n) k) (.inl i) (.inr j)
80           rw [neg_commutator_right_neg]
81           exact h
82   | inr i =>
83       cases j with
84       | inl j =>
85           change -Stafford.commutator (-momentum k n i) (coordinate k n j) = _
86           have h := freeWeylGenerator_commutator
87             (k := k) (Matrix.J (Fin n) k) (.inr i) (.inl j)
88           rw [neg_commutator_left_neg]
89           exact h
90       | inr j =>
91           change -Stafford.commutator (-momentum k n i) (-momentum k n j) = _
92           have h := freeWeylGenerator_commutator
93             (k := k) (Matrix.J (Fin n) k) (.inr i) (.inr j)
94           have hJ : Matrix.J (Fin n) k (.inr i) (.inr j) = 0 := by
95             simp [Matrix.J]
96           rw [hJ] at h ⊢
97           rw [neg_commutator_both_neg]
98           change
99             -Stafford.commutator
100                 (freeWeylGenerator (Matrix.J (Fin n) k) (.inr i))
101                 (freeWeylGenerator (Matrix.J (Fin n) k) (.inr j)) = _
102           rw [h, map_zero, neg_zero]

104 theorem transposedGenerator_commutator (n : Nat) :
105     ∀ i j,
106       Stafford.commutator (transposedGenerator k n i) (transposedGenerator k n j) =
107         algebraMap k (PresentedWeyl k n)ᵐᵒᵖ (Matrix.J (Fin n) k i j) := by
108   intro i j
109   apply unop_injective
110   simpa [Stafford.commutator] using
111     underlying_transposedGenerator_commutator k n i j

113 /-- The scalar-preserving anti-homomorphism `x_i ↦ x_i`, `p_i ↦ -p_i`,
114 represented as an algebra homomorphism to the opposite algebra. -/
115 def transpositionHom (n : Nat) :
116     PresentedWeyl k n →ₐ[k] (PresentedWeyl k n)ᵐᵒᵖ :=
117   freeWeylLift (Matrix.J (Fin n) k) (transposedGenerator k n)
118     (transposedGenerator_commutator k n)

132 /-- Applying transposition once on each side of the opposite equivalence is
133 the identity. -/
134 theorem opComm_transpositionHom_comp_transpositionHom (n : Nat) :
135     (AlgHom.opComm (transpositionHom k n)).comp (transpositionHom k n) =
136       AlgHom.id k (PresentedWeyl k n) := by
137   apply freeWeyl_algHom_ext
138   intro i
139   cases i with
140   | inl i =>
141       simp only [AlgHom.comp_apply, AlgHom.id_apply]
142       rw [show freeWeylGenerator (Matrix.J (Fin n) k) (.inl i) =
143         coordinate k n i from rfl, transpositionHom_coordinate]
144       change unop (transpositionHom k n (coordinate k n i)) = coordinate k n i
145       rw [transpositionHom_coordinate]
146       exact unop_op _
147   | inr i =>
148       simp only [AlgHom.comp_apply, AlgHom.id_apply]
149       rw [show freeWeylGenerator (Matrix.J (Fin n) k) (.inr i) =
150         momentum k n i from rfl, transpositionHom_momentum]
151       change unop (transpositionHom k n (-momentum k n i)) = momentum k n i
152       rw [map_neg, transpositionHom_momentum]
153       simp
154 
155 theorem transpositionHom_comp_opComm_transpositionHom (n : Nat) :
156     (transpositionHom k n).comp (AlgHom.opComm (transpositionHom k n)) =
157       AlgHom.id k (PresentedWeyl k n)ᵐᵒᵖ := by
158   apply AlgHom.ext
159   intro a
160   rcases a with ⟨a⟩
161   apply unop_injective
162   have h := congrArg (fun f : PresentedWeyl k n →ₐ[k] PresentedWeyl k n => f a)
163     (opComm_transpositionHom_comp_transpositionHom k n)
164   exact h

166 /-- Transposition is a scalar-preserving equivalence with the opposite Weyl
167 algebra. -/
168 def transpositionEquiv (n : Nat) :
169     PresentedWeyl k n ≃ₐ[k] (PresentedWeyl k n)ᵐᵒᵖ :=
170   AlgEquiv.ofAlgHom (transpositionHom k n) (AlgHom.opComm (transpositionHom k n))
171     (transpositionHom_comp_opComm_transpositionHom k n)
172     (opComm_transpositionHom_comp_transpositionHom k n)

193 /-- The usual unbundled anti-automorphism underlying `transpositionEquiv`. -/
194 def transpose (n : Nat) (a : PresentedWeyl k n) : PresentedWeyl k n :=
195   unop (transpositionEquiv k n a)

214 @[simp] theorem transpose_transpose (n : Nat) (a : PresentedWeyl k n) :
215     transpose k n (transpose k n a) = a := by
216   have h := congrArg (fun f : PresentedWeyl k n →ₐ[k] PresentedWeyl k n => f a)

224 /-- A right `PresentedWeyl`-module becomes a left module by restriction of
225 scalars along transposition. -/
226 def transposedLeftModule (n : Nat) (M : Type v) [AddCommMonoid M]
227     [Module (PresentedWeyl k n)ᵐᵒᵖ M] : Module (PresentedWeyl k n) M :=
228   Module.compHom M (transpositionEquiv k n).toRingEquiv.toRingHom

233     @SMul.smul (PresentedWeyl k n) M
