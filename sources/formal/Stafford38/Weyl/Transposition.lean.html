11 # Transposition of the presented Weyl algebra
12 
13 This file constructs the scalar-preserving Weyl transposition as an algebra
14 equivalence with the opposite algebra.  Coordinates are fixed and momenta
15 change sign.  The construction uses only the checked universal property of
16 the quotient presentation.
17 -/

33 /-- The coordinate generator with index `i`. -/
34 def coordinate (n : Nat) (i : Fin n) : WeylAlg k n :=
35   freeWeylGenerator (Matrix.J (Fin n) k) (.inl i)

37 /-- The momentum generator with index `i`. -/

39   freeWeylGenerator (Matrix.J (Fin n) k) (.inr i)

41 /-- Generator images for the Weyl transposition. -/
42 def transposedGenerator (n : Nat) :
43     (Fin n ⊕ Fin n) → (WeylAlg k n)ᵐᵒᵖ
44   | .inl i => op (coordinate k n i)
45   | .inr i => op (-momentum k n i)

61 theorem underlying_transposedGenerator_commutator (n : Nat) :
62     ∀ i j,
63       -AlgebraicAnalysis.ringCommutator (unop (transposedGenerator k n i))
64           (unop (transposedGenerator k n j)) =
65         algebraMap k (WeylAlg k n) (Matrix.J (Fin n) k i j) := by
66   intro i j
67   cases i with
68   | inl i =>
69       cases j with
70       | inl j =>
71           change -AlgebraicAnalysis.ringCommutator (coordinate k n i) (coordinate k n j) = _
72           rw [show AlgebraicAnalysis.ringCommutator (coordinate k n i) (coordinate k n j) =
73             algebraMap k (WeylAlg k n)
74               (Matrix.J (Fin n) k (.inl i) (.inl j)) from
75             freeWeylGenerator_commutator
76               (k := k) (Matrix.J (Fin n) k) (.inl i) (.inl j)]
77           simp [Matrix.J]
78       | inr j =>
79           change -AlgebraicAnalysis.ringCommutator (coordinate k n i) (-momentum k n j) = _
80           have h := freeWeylGenerator_commutator
81             (k := k) (Matrix.J (Fin n) k) (.inl i) (.inr j)
82           rw [neg_commutator_right_neg]
83           exact h
84   | inr i =>
85       cases j with
86       | inl j =>
87           change -AlgebraicAnalysis.ringCommutator (-momentum k n i) (coordinate k n j) = _
88           have h := freeWeylGenerator_commutator
89             (k := k) (Matrix.J (Fin n) k) (.inr i) (.inl j)
90           rw [neg_commutator_left_neg]
91           exact h
92       | inr j =>
93           change -AlgebraicAnalysis.ringCommutator (-momentum k n i) (-momentum k n j) = _
94           have h := freeWeylGenerator_commutator
95             (k := k) (Matrix.J (Fin n) k) (.inr i) (.inr j)
96           have hJ : Matrix.J (Fin n) k (.inr i) (.inr j) = 0 := by
97             simp [Matrix.J]
98           rw [hJ] at h ⊢
99           rw [neg_commutator_both_neg]
100           change
101             -AlgebraicAnalysis.ringCommutator
102                 (freeWeylGenerator (Matrix.J (Fin n) k) (.inr i))
103                 (freeWeylGenerator (Matrix.J (Fin n) k) (.inr j)) = _
104           rw [h, map_zero, neg_zero]

106 theorem transposedGenerator_commutator (n : Nat) :
107     ∀ i j,
108       AlgebraicAnalysis.ringCommutator (transposedGenerator k n i) (transposedGenerator k n j) =
109         algebraMap k (WeylAlg k n)ᵐᵒᵖ (Matrix.J (Fin n) k i j) := by
110   intro i j
111   apply unop_injective
112   simpa [AlgebraicAnalysis.ringCommutator] using
113     underlying_transposedGenerator_commutator k n i j

115 /-- The scalar-preserving anti-homomorphism `x_i ↦ x_i`, `p_i ↦ -p_i`,
116 represented as an algebra homomorphism to the opposite algebra. -/
117 def transpositionHom (n : Nat) :
118     WeylAlg k n →ₐ[k] (WeylAlg k n)ᵐᵒᵖ :=
119   freeWeylLift (Matrix.J (Fin n) k) (transposedGenerator k n)
120     (transposedGenerator_commutator k n)

134 /-- Applying transposition once on each side of the opposite equivalence is
135 the identity. -/
136 theorem opComm_transpositionHom_comp_transpositionHom (n : Nat) :
137     (AlgHom.opComm (transpositionHom k n)).comp (transpositionHom k n) =
138       AlgHom.id k (WeylAlg k n) := by
139   apply freeWeyl_algHom_ext
140   intro i
141   cases i with
142   | inl i =>
143       simp only [AlgHom.comp_apply, AlgHom.id_apply]
144       rw [show freeWeylGenerator (Matrix.J (Fin n) k) (.inl i) =
145         coordinate k n i from rfl, transpositionHom_coordinate]
146       change unop (transpositionHom k n (coordinate k n i)) = coordinate k n i
147       rw [transpositionHom_coordinate]
148       exact unop_op _
149   | inr i =>
150       simp only [AlgHom.comp_apply, AlgHom.id_apply]
151       rw [show freeWeylGenerator (Matrix.J (Fin n) k) (.inr i) =
152         momentum k n i from rfl, transpositionHom_momentum]
153       change unop (transpositionHom k n (-momentum k n i)) = momentum k n i
154       rw [map_neg, transpositionHom_momentum]
155       simp

157 theorem transpositionHom_comp_opComm_transpositionHom (n : Nat) :
158     (transpositionHom k n).comp (AlgHom.opComm (transpositionHom k n)) =
159       AlgHom.id k (WeylAlg k n)ᵐᵒᵖ := by
160   apply AlgHom.ext
161   intro a
162   rcases a with ⟨a⟩
163   apply unop_injective
164   have h := congrArg (fun f : WeylAlg k n →ₐ[k] WeylAlg k n => f a)
165     (opComm_transpositionHom_comp_transpositionHom k n)
166   exact h

168 /-- Transposition is a scalar-preserving equivalence with the opposite Weyl
169 algebra. -/
170 def transpositionEquiv (n : Nat) :
171     WeylAlg k n ≃ₐ[k] (WeylAlg k n)ᵐᵒᵖ :=
172   AlgEquiv.ofAlgHom (transpositionHom k n) (AlgHom.opComm (transpositionHom k n))
173     (transpositionHom_comp_opComm_transpositionHom k n)
174     (opComm_transpositionHom_comp_transpositionHom k n)

195 /-- The usual unbundled anti-automorphism underlying `transpositionEquiv`. -/
196 def transpose (n : Nat) (a : WeylAlg k n) : WeylAlg k n :=
197   unop (transpositionEquiv k n a)

216 @[simp] theorem transpose_transpose (n : Nat) (a : WeylAlg k n) :
217     transpose k n (transpose k n a) = a := by

226 /-- A right `WeylAlg`-module becomes a left module by restriction of
227 scalars along transposition. -/
228 def transposedLeftModule (n : Nat) (M : Type v) [AddCommMonoid M]
229     [Module (WeylAlg k n)ᵐᵒᵖ M] : Module (WeylAlg k n) M :=
230   Module.compHom M (transpositionEquiv k n).toRingEquiv.toRingHom
231 
232 theorem transposedLeftModule_smul (n : Nat) (M : Type v) [AddCommMonoid M]
233     [Module (WeylAlg k n)ᵐᵒᵖ M]
234     (a : WeylAlg k n) (m : M) :
235     @SMul.smul (WeylAlg k n) M
236       (transposedLeftModule k n M).toSMul a m =
237       transpositionEquiv k n a • m := rfl
