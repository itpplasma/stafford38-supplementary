10 # The differential-order Rees ring
11 
12 This file constructs the literal Rees ring of the differential-order
13 filtration as the subring of the central polynomial ring `A[T]` whose
14 degree-`N` coefficient lies in `F_N A`.  No commutativity of `A`, Rees
15 specialization, or Gabber theorem is assumed.
16 -/

31 /-- The differential-order Rees ring inside the central polynomial ring over
32 the presented Weyl algebra. -/
33 def orderReesSubring : Subring (Polynomial (WeylAlg k n)) where
34   carrier := {f | ∀ N, f.coeff N ∈ orderPiece k n N}
35   zero_mem' := by simp
36   one_mem' := by
37     intro N
38     by_cases hN : N = 0
39     · subst N
40       rw [Polynomial.coeff_one, if_pos rfl]
41       exact (orderPieceOne (n := n) k).property
42     · rw [Polynomial.coeff_one, if_neg hN]
43       exact Submodule.zero_mem _
44   add_mem' := by
45     intro f g hf hg N
46     rw [Polynomial.coeff_add]
47     exact Submodule.add_mem _ (hf N) (hg N)
48   neg_mem' := by
49     intro f hf N
50     rw [Polynomial.coeff_neg]
51     exact Submodule.neg_mem _ (hf N)
52   mul_mem' := by
53     intro f g hf hg N
54     rw [Polynomial.coeff_mul]
55     apply Submodule.sum_mem
56     intro ij hij
57     have hmul := mul_mem_orderPiece k (hf ij.1) (hg ij.2)
58     have hij' : ij.1 + ij.2 = N := Finset.HasAntidiagonal.mem_antidiagonal.mp hij
59     simpa [hij'] using hmul

61 /-- The type of elements of the differential-order Rees ring. -/
62 abbrev OrderReesRing := orderReesSubring (n := n) k
63 
64 @[simp] theorem mem_orderReesSubring_iff
65     (f : Polynomial (WeylAlg k n)) :
66     f ∈ orderReesSubring (n := n) k ↔
67       ∀ N, f.coeff N ∈ orderPiece k n N :=
68   Iff.rfl

70 /-- A filtered operator inserted in its declared Rees degree. -/
71 def orderReesMonomial (N : ℕ) (z : orderPiece k n N) :
72     OrderReesRing (n := n) k :=
73   ⟨Polynomial.monomial N (z : WeylAlg k n), by
74     intro M
75     rw [Polynomial.coeff_monomial]
76     split_ifs with h
77     · subst M
78       exact z.property
79     · exact Submodule.zero_mem _⟩
80 
81 @[simp] theorem orderReesMonomial_coe (N : ℕ) (z : orderPiece k n N) :
82     ((orderReesMonomial k N z : OrderReesRing (n := n) k) :
83       Polynomial (WeylAlg k n)) =
84         Polynomial.monomial N (z : WeylAlg k n) :=
85   rfl
86 
87 @[simp] theorem orderReesMonomial_coeff_same
88     (N : ℕ) (z : orderPiece k n N) :
89     ((orderReesMonomial k N z : OrderReesRing (n := n) k) :
90       Polynomial (WeylAlg k n)).coeff N = (z : WeylAlg k n) := by
91   simp

93 /-- The central Rees parameter `T`, represented in degree one by the unit
94 operator. -/
95 def orderReesParameter : OrderReesRing (n := n) k :=
96   orderReesMonomial k 1
97     ⟨1, presentedWeightPiece_mono k orderWeight (Nat.zero_le 1)
98       (orderPieceOne (n := n) k).property⟩
99 
100 @[simp] theorem orderReesParameter_coe :
101     ((orderReesParameter (n := n) k : OrderReesRing (n := n) k) :
102       Polynomial (WeylAlg k n)) = Polynomial.X := by
103   rw [orderReesParameter, orderReesMonomial_coe]
104   exact Polynomial.monomial_one_one_eq_X
105 
106 /-- Multiplication of homogeneous Rees representatives is multiplication in
107 the Weyl algebra, with degrees added. -/
108 theorem orderReesMonomial_mul
109     {N M : ℕ} (z : orderPiece k n N) (y : orderPiece k n M) :
110     orderReesMonomial k N z * orderReesMonomial k M y =
111       orderReesMonomial k (N + M)
112         ⟨(z : WeylAlg k n) * (y : WeylAlg k n),
113           mul_mem_orderPiece k z.property y.property⟩ := by
114   apply Subtype.ext
115   simp [orderReesMonomial, Polynomial.monomial_mul_monomial]
116 
117 /-- Multiplication by the Rees parameter raises the declared degree and keeps
118 the underlying Weyl operator unchanged. -/
119 theorem orderReesParameter_mul_monomial
120     {N : ℕ} (z : orderPiece k n N) :
121     orderReesParameter (n := n) k * orderReesMonomial k N z =
122       orderReesMonomial k (N + 1)
123         ⟨(z : WeylAlg k n),
124           presentedWeightPiece_mono k orderWeight (Nat.le_succ N) z.property⟩ := by
125   apply Subtype.ext
126   simp [orderReesParameter_coe, orderReesMonomial,
127     Polynomial.X_mul_monomial, Nat.add_comm]
128 
129 /-- The polynomial Rees parameter is central, including over the
130 noncommutative Weyl coefficient ring. -/
131 theorem orderReesParameter_mul_comm
132     (r : OrderReesRing (n := n) k) :
133     orderReesParameter (n := n) k * r =
134       r * orderReesParameter (n := n) k := by
135   apply Subtype.ext
136   change Polynomial.X * (r : Polynomial (WeylAlg k n)) =
137     (r : Polynomial (WeylAlg k n)) * Polynomial.X
138   exact Polynomial.X_mul
139 
140 /-- A homogeneous class in positive Rees degree is divisible by the Rees
141 parameter exactly when its coefficient already lies one filtration step
142 lower.  This is the degreewise kernel statement behind specialization at
143 `T = 0`. -/
144 theorem exists_parameter_mul_eq_monomial_iff_mem_lower
145     {N : ℕ} (z : orderPiece k n (N + 1)) :
146     (∃ r : OrderReesRing (n := n) k,
147         orderReesParameter (n := n) k * r =
148           orderReesMonomial k (N + 1) z) ↔
149       (z : WeylAlg k n) ∈ orderPiece k n N := by
150   constructor
151   · rintro ⟨r, hr⟩
152     have hc := congrArg
153       (fun p : Polynomial (WeylAlg k n) => p.coeff (N + 1))
154       (congrArg Subtype.val hr)
155     change (Polynomial.X * (r : Polynomial (WeylAlg k n))).coeff
156         (N + 1) =
157       (Polynomial.monomial (N + 1) (z : WeylAlg k n)).coeff
158         (N + 1) at hc
159     rw [Polynomial.coeff_X_mul, Polynomial.coeff_monomial, if_pos rfl] at hc
160     rw [← hc]
161     exact r.property N
162   · intro hz
163     let zLower : orderPiece k n N := ⟨z, hz⟩
164     refine ⟨orderReesMonomial k N zLower, ?_⟩
165     rw [orderReesParameter_mul_monomial]
166 
167 /-- In Rees degree zero, a homogeneous representative is a parameter multiple
168 only when it is zero. -/
169 theorem exists_parameter_mul_eq_degreeZero_iff
170     (z : orderPiece k n 0) :
171     (∃ r : OrderReesRing (n := n) k,
172         orderReesParameter (n := n) k * r = orderReesMonomial k 0 z) ↔
173       (z : WeylAlg k n) = 0 := by
174   constructor
175   · rintro ⟨r, hr⟩
176     have hc := congrArg
177       (fun p : Polynomial (WeylAlg k n) => p.coeff 0)
178       (congrArg Subtype.val hr)
179     change (Polynomial.X * (r : Polynomial (WeylAlg k n))).coeff 0 =
180       (Polynomial.monomial 0 (z : WeylAlg k n)).coeff 0 at hc
181     simpa using hc.symm
182   · intro hz
183     have hz' : z = 0 := Subtype.ext hz
184     subst z
185     refine ⟨0, ?_⟩
186     apply Subtype.ext
187     simp [orderReesMonomial]
