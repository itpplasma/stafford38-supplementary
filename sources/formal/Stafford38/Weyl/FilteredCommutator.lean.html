9 # Filtered commutators and subprincipal symbols

13 filtered elements; it includes the degree-zero case rather than hiding natural-

32 /-- Commutators lower differential order by one.  With natural-number
33 filtration indices, the `r = s = 0` case says that two order-zero operators
34 commute and hence their ringCommutator belongs to `F₀`. -/
35 theorem commutator_mem_orderPiece_pred
36     {n r s : ℕ} {a b : WeylAlg k n}
37     (ha : a ∈ orderPiece k n r) (hb : b ∈ orderPiece k n s) :
38     AlgebraicAnalysis.ringCommutator a b ∈ orderPiece k n (r + s - 1) := by
39   have hab : a * b ∈ orderPiece k n (r + s) :=
40     mul_mem_orderPiece k ha hb
41   have hba : b * a ∈ orderPiece k n (s + r) :=
42     mul_mem_orderPiece k hb ha
43   have hba' : b * a ∈ orderPiece k n (r + s) := by
44     simpa [Nat.add_comm] using hba
45   have hcomm : AlgebraicAnalysis.ringCommutator a b ∈ orderPiece k n (r + s) := by
46     exact Submodule.sub_mem _ hab hba'
47   have hmulba :
48       presentedPrincipalComponent k (@orderWeight n) (r + s) (b * a) =
49         presentedPrincipalComponent k (@orderWeight n) s b *
50           presentedPrincipalComponent k (@orderWeight n) r a := by
51     simpa [Nat.add_comm] using
52       (presentedPrincipalComponent_mul_order k hb ha)
53   have htop :
54       presentedPrincipalComponent k (@orderWeight n) (r + s)
55           (AlgebraicAnalysis.ringCommutator a b) = 0 := by
56     change presentedPrincipalComponent k (@orderWeight n) (r + s)
57         (a * b - b * a) = 0
58     rw [map_sub,
59       presentedPrincipalComponent_mul_order k ha hb,
60       hmulba]
61     exact sub_eq_zero.mpr (mul_comm _ _)
62   have hlower :=
63     (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
64       k (@orderWeight n) _ hcomm).mp htop
65   cases hsum : r + s with
66   | zero =>
67       have hr : r = 0 := by omega
68       have hs : s = 0 := by omega
69       subst r
70       subst s
71       simpa using hcomm
72   | succ t =>
73       change a * b - b * a ∈ presentedWeightPiece k orderWeight t
74       simpa [presentedStrictLowerPiece, hsum] using hlower

76 /-- Differential operators of order zero are multiplication operators and
77 therefore commute. -/
78 theorem commutator_eq_zero_of_mem_orderPiece_zero
79     {n : ℕ} {a b : WeylAlg k n}
80     (ha : a ∈ orderPiece k n 0) (hb : b ∈ orderPiece k n 0) :
81     AlgebraicAnalysis.ringCommutator a b = 0 := by
82   have hab : a * b ∈ orderPiece k n 0 := by
83     simpa using (mul_mem_orderPiece k ha hb)
84   have hba : b * a ∈ orderPiece k n 0 := by
85     simpa using (mul_mem_orderPiece k hb ha)
86   have hcomm : AlgebraicAnalysis.ringCommutator a b ∈ orderPiece k n 0 :=
87     Submodule.sub_mem _ hab hba
88   have htop :
89       presentedPrincipalComponent k (@orderWeight n) 0
90           (AlgebraicAnalysis.ringCommutator a b) = 0 := by
91     change presentedPrincipalComponent k (@orderWeight n) 0
92         (a * b - b * a) = 0
93     rw [map_sub,
94       presentedPrincipalComponent_mul_order k ha hb,
95       presentedPrincipalComponent_mul_order k hb ha]
96     exact sub_eq_zero.mpr (mul_comm _ _)
97   have hlower := (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
98     k (@orderWeight n) _ hcomm).mp htop
99   simpa [presentedStrictLowerPiece] using hlower

147 /-- The one-contraction part of normal-ordered multiplication.  It
148 differentiates the left monomial in momentum and the right monomial in the
149 matching coordinate. -/
150 def pbwFirstContraction {n : ℕ}
151     (m q : PhaseVar n →₀ ℕ) : SymbolRing k n :=
152   ∑ i : Fin n,
153     MvPolynomial.pderiv (.inr i) (MvPolynomial.monomial m 1) *
154       MvPolynomial.pderiv (.inl i) (MvPolynomial.monomial q 1)

156 /-- Antisymmetrizing the first normal-ordering contraction is exactly the
157 negative Poisson bracket for the convention `[x,p] = -1`. -/
158 theorem pbwFirstContraction_sub_swap_eq_neg_poisson {n : ℕ}
159     (m q : PhaseVar n →₀ ℕ) :
160     pbwFirstContraction k m q - pbwFirstContraction k q m =
161       -poissonBracket (MvPolynomial.monomial m 1)
162         (MvPolynomial.monomial q 1) := by
163   rw [pbwFirstContraction, pbwFirstContraction, poissonBracket,
164     ← Finset.sum_sub_distrib, ← Finset.sum_neg_distrib]
165   apply Finset.sum_congr rfl
166   intro i hi
167   ring

169 theorem momentumExponent_eq_zero_of_orderWeight_eq_zero {n : ℕ}
170     (m : PhaseVar n →₀ ℕ)
171     (hm : Finsupp.weight (@orderWeight n) m = 0) (i : Fin n) :
172     m (.inr i) = 0 := by
173   by_contra hne
174   have himem : (.inr i : PhaseVar n) ∈ m.support :=
175     Finsupp.mem_support_iff.mpr hne
176   have hle :
177       m (.inr i) * orderWeight (.inr i) ≤
178         ∑ j ∈ m.support, m j * orderWeight j :=
179     Finset.single_le_sum
180       (f := fun j => m j * orderWeight j)
181       (fun j _ => Nat.zero_le _) himem
182   have hle' : m (.inr i) ≤ Finsupp.weight (@orderWeight n) m := by
183     simpa [Finsupp.weight_apply, Finsupp.sum, orderWeight, fibreWeight] using hle
184   omega

186 theorem pbwFirstContraction_eq_zero_of_orderWeight_eq_zero {n : ℕ}
187     (m q : PhaseVar n →₀ ℕ)
188     (hm : Finsupp.weight (@orderWeight n) m = 0) :
189     pbwFirstContraction k m q = 0 := by
190   rw [pbwFirstContraction]
191   apply Finset.sum_eq_zero
192   intro i hi
193   rw [MvPolynomial.pderiv_monomial,
194     momentumExponent_eq_zero_of_orderWeight_eq_zero m hm i]
195   simp

197 /-- A one-sided first-contraction formula for PBW products implies the exact
198 PBW ringCommutator formula.  Unlike a ringCommutator restatement, the premise is the
199 specific local normal-ordering coefficient that the recursive PBW product API
200 must expose. -/
201 theorem pbw_commutator_formula_of_firstContraction
202     {n : ℕ}
203     (hProduct : ∀ m q : PhaseVar n →₀ ℕ,
204       0 < Finsupp.weight (@orderWeight n) m +
205           Finsupp.weight (@orderWeight n) q →
206       presentedPrincipalComponent k (@orderWeight n)
207           (Finsupp.weight (@orderWeight n) m +
208             Finsupp.weight (@orderWeight n) q - 1)
209           (presentedPBWBasis k n m * presentedPBWBasis k n q) =
210         pbwFirstContraction k m q) :
211     ∀ m q : PhaseVar n →₀ ℕ,
212       presentedPrincipalComponent k (@orderWeight n)
213           (Finsupp.weight (@orderWeight n) m +
214             Finsupp.weight (@orderWeight n) q - 1)
215           (AlgebraicAnalysis.ringCommutator
216             (presentedPBWBasis k n m) (presentedPBWBasis k n q)) =
217         -poissonBracket (MvPolynomial.monomial m 1)
218           (MvPolynomial.monomial q 1) := by
219   intro m q
220   let R := Finsupp.weight (@orderWeight n) m
221   let S := Finsupp.weight (@orderWeight n) q
222   by_cases hpos : 0 < R + S
223   · change presentedPrincipalComponent k (@orderWeight n) (R + S - 1)
224       (presentedPBWBasis k n m * presentedPBWBasis k n q -
225         presentedPBWBasis k n q * presentedPBWBasis k n m) = _
226     rw [map_sub, hProduct m q hpos]
227     have hpos' : 0 < S + R := by omega
228     rw [show R + S - 1 = S + R - 1 by omega, hProduct q m hpos']
229     exact pbwFirstContraction_sub_swap_eq_neg_poisson k m q
230   · have hzero : R + S = 0 := by omega
231     have hR : R = 0 := by omega
232     have hS : S = 0 := by omega
233     have hmPiece : presentedPBWBasis k n m ∈ orderPiece k n 0 := by
234       exact (presentedPBWBasis_mem_weightPiece_iff k orderWeight 0 m).mpr
235         (by simpa [R] using hR.le)
236     have hqPiece : presentedPBWBasis k n q ∈ orderPiece k n 0 := by
237       exact (presentedPBWBasis_mem_weightPiece_iff k orderWeight 0 q).mpr
238         (by simpa [S] using hS.le)
239     rw [commutator_eq_zero_of_mem_orderPiece_zero k hmPiece hqPiece, map_zero]
240     rw [← pbwFirstContraction_sub_swap_eq_neg_poisson k m q,
241       pbwFirstContraction_eq_zero_of_orderWeight_eq_zero k m q hR,
242       pbwFirstContraction_eq_zero_of_orderWeight_eq_zero k q m hS,
243       sub_zero]

245 /-- Bilinear PBW extension principle for the subprincipal ringCommutator formula.
246 The hypothesis concerns every pair of actual PBW basis vectors at their exact
247 orders.  The conclusion handles arbitrary finite PBW sums at arbitrary declared
248 filtration bounds, including vanishing leading components. -/
249 theorem principalComponent_commutator_eq_neg_poisson_of_PBW
250     {n r s : ℕ} {a b : WeylAlg k n}
251     (hPBW : ∀ m q : PhaseVar n →₀ ℕ,
252       presentedPrincipalComponent k (@orderWeight n)
253           (Finsupp.weight (@orderWeight n) m +
254             Finsupp.weight (@orderWeight n) q - 1)
255           (AlgebraicAnalysis.ringCommutator
256             (presentedPBWBasis k n m) (presentedPBWBasis k n q)) =
257         -poissonBracket (MvPolynomial.monomial m 1)
258           (MvPolynomial.monomial q 1))
259     (ha : a ∈ orderPiece k n r) (hb : b ∈ orderPiece k n s) :
260     presentedPrincipalComponent k (@orderWeight n) (r + s - 1)
261         (AlgebraicAnalysis.ringCommutator a b) =
262       -poissonBracket
263         (presentedPrincipalComponent k (@orderWeight n) r a)
264         (presentedPrincipalComponent k (@orderWeight n) s b) := by
265   rw [orderPiece, presentedWeightPiece_eq_span] at ha hb
266   apply Submodule.span_induction₂
267       (p := fun a b _ _ =>
268         presentedPrincipalComponent k (@orderWeight n) (r + s - 1)
269             (AlgebraicAnalysis.ringCommutator a b) =
270           -poissonBracket
271             (presentedPrincipalComponent k (@orderWeight n) r a)
272             (presentedPrincipalComponent k (@orderWeight n) s b))
273       (ha := ha) (hb := hb)
274   · intro x y hx hy
275     obtain ⟨m, hm, rfl⟩ := hx
276     obtain ⟨q, hq, rfl⟩ := hy
277     let R := Finsupp.weight (@orderWeight n) m
278     let S := Finsupp.weight (@orderWeight n) q
279     have hmPiece : presentedPBWBasis k n m ∈ orderPiece k n R := by
280       exact (presentedPBWBasis_mem_weightPiece_iff k orderWeight R m).mpr le_rfl
281     have hqPiece : presentedPBWBasis k n q ∈ orderPiece k n S := by
282       exact (presentedPBWBasis_mem_weightPiece_iff k orderWeight S q).mpr le_rfl
283     by_cases hR : R = r
284     · by_cases hS : S = s
285       · subst r
286         subst s
287         rw [presentedPrincipalComponent_basis,
288           presentedPrincipalComponent_basis]
289         simp only [R, S, if_pos rfl]
290         exact hPBW m q
291       · have hSlt : S < s := lt_of_le_of_ne hq hS
292         rw [presentedPrincipalComponent_basis,
293           presentedPrincipalComponent_basis]
294         simp only [R, S, hR, if_pos, hS, if_false,
295           poissonBracket_zero_right, neg_zero]
296         by_cases hzero : R + S = 0
297         · have hR0 : R = 0 := by omega
298           have hS0 : S = 0 := by omega
299           rw [commutator_eq_zero_of_mem_orderPiece_zero k
300             (by simpa [hR0] using hmPiece) (by simpa [hS0] using hqPiece),
301             map_zero]
302         · have hmem := commutator_mem_orderPiece_pred k hmPiece hqPiece
303           apply presentedPrincipalComponent_eq_zero_of_mem_of_lt k
304             (@orderWeight n) _ hmem
305           omega
306     · have hRlt : R < r := lt_of_le_of_ne hm hR
307       rw [presentedPrincipalComponent_basis,
308         presentedPrincipalComponent_basis]
309       simp only [R, S, hR, if_false, poissonBracket_zero_left, neg_zero]
310       by_cases hzero : R + S = 0
311       · have hR0 : R = 0 := by omega
312         have hS0 : S = 0 := by omega
313         rw [commutator_eq_zero_of_mem_orderPiece_zero k
314           (by simpa [hR0] using hmPiece) (by simpa [hS0] using hqPiece),
315           map_zero]
316       · have hmem := commutator_mem_orderPiece_pred k hmPiece hqPiece
317         apply presentedPrincipalComponent_eq_zero_of_mem_of_lt k
318           (@orderWeight n) _ hmem
319         omega
320   · intro y hy
321     simp [AlgebraicAnalysis.ringCommutator]
322   · intro x hx
323     simp [AlgebraicAnalysis.ringCommutator]
324   · intro x y z hx hy hz hxy hyz
325     rw [show AlgebraicAnalysis.ringCommutator (x + y) z =
326         AlgebraicAnalysis.ringCommutator x z + AlgebraicAnalysis.ringCommutator y z by
327       change (x + y) * z - z * (x + y) =
328         (x * z - z * x) + (y * z - z * y)
329       noncomm_ring,
330       map_add, hxy, hyz, map_add, poissonBracket_add_left]
331     abel
332   · intro x y z hx hy hz hxy hxz
333     rw [show AlgebraicAnalysis.ringCommutator x (y + z) =
334         AlgebraicAnalysis.ringCommutator x y + AlgebraicAnalysis.ringCommutator x z by
335       change x * (y + z) - (y + z) * x =
336         (x * y - y * x) + (x * z - z * x)
337       noncomm_ring,
338       map_add, hxy, hxz, map_add, poissonBracket_add_right]
339     abel
340   · intro c x y hx hy hxy
341     rw [show AlgebraicAnalysis.ringCommutator (c • x) y =
342         c • AlgebraicAnalysis.ringCommutator x y by
343       change (c • x) * y - y * (c • x) = c • (x * y - y * x)
344       simp only [Algebra.smul_mul_assoc,
345         Algebra.mul_smul_comm, smul_sub],
346       map_smul, hxy, map_smul, poissonBracket_smul_left, smul_neg]
347   · intro c x y hx hy hxy
348     rw [show AlgebraicAnalysis.ringCommutator x (c • y) =
349         c • AlgebraicAnalysis.ringCommutator x y by
350       change x * (c • y) - (c • y) * x = c • (x * y - y * x)
351       simp only [Algebra.mul_smul_comm,
352         Algebra.smul_mul_assoc, smul_sub],
353       map_smul, hxy, map_smul, poissonBracket_smul_right, smul_neg]

355 /-- Full arbitrary-sum consequence of the one-sided PBW first-contraction
356 formula.  This packages both nontrivial bridges: antisymmetrization produces
357 the sign-correct Poisson bracket, and PBW bilinearity handles arbitrary
358 declared filtration bounds. -/
359 theorem principalComponent_commutator_eq_neg_poisson_of_firstContraction
360     {n r s : ℕ} {a b : WeylAlg k n}
361     (hProduct : ∀ m q : PhaseVar n →₀ ℕ,
362       0 < Finsupp.weight (@orderWeight n) m +
363           Finsupp.weight (@orderWeight n) q →
364       presentedPrincipalComponent k (@orderWeight n)
365           (Finsupp.weight (@orderWeight n) m +
366             Finsupp.weight (@orderWeight n) q - 1)
367           (presentedPBWBasis k n m * presentedPBWBasis k n q) =
368         pbwFirstContraction k m q)
369     (ha : a ∈ orderPiece k n r) (hb : b ∈ orderPiece k n s) :
370     presentedPrincipalComponent k (@orderWeight n) (r + s - 1)
371         (AlgebraicAnalysis.ringCommutator a b) =
372       -poissonBracket
373         (presentedPrincipalComponent k (@orderWeight n) r a)
374         (presentedPrincipalComponent k (@orderWeight n) s b) :=
375   principalComponent_commutator_eq_neg_poisson_of_PBW k
376     (pbw_commutator_formula_of_firstContraction k hProduct) ha hb
