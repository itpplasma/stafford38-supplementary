18 def LeftHandedStatement : Prop :=
19   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (d : WeylAlg k n),
20     d ≠ 0 → ∃ R S F : WeylAlg k n,
21       (1 : WeylAlg k n) = R * d + S * d * F
22 
23 theorem leftHanded_of_universalStatement
24     (h : UniversalStatement.{u}) : LeftHandedStatement.{u} := by
25   intro k _ _ n d hd
26   have htd : transpose k n d ≠ 0 := by
27     intro hz
28     apply hd
29     simpa only [transpose_transpose, transpose_zero] using congrArg (transpose k n) hz
30   rcases h k n (transpose k n d) htd with ⟨F, R, S, hone⟩
31   refine ⟨transpose k n R, transpose k n S, transpose k n F, ?_⟩
32   have ht := congrArg (transpose k n) hone
33   have h1 : transpose k n (1 : WeylAlg k n) = 1 := by simp [transpose]
34   rw [h1] at ht
35   simpa [transpose_add, transpose_mul, transpose_transpose, mul_assoc] using ht
36 
37 end Stafford38.LeftHandedCorollary
