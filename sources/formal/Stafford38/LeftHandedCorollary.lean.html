9 /-! The left-handed form is the formal-adjoint/opposite-ring image of the
10     written-order Stafford identity. -/
11 
12 namespace Stafford38.LeftHandedCorollary
13 
14 open Stafford38
15 open Stafford38.WeylTransposition
16 open Stafford38.WeylTranspositionFiltration
17 
18 universe u
19 
20 def LeftHandedStatement : Prop :=
21   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (d : WeylAlg k n),
22     d ≠ 0 → ∃ R S F : WeylAlg k n,
23       (1 : WeylAlg k n) = R * d + S * d * F
24 
25 theorem leftHanded_of_universalStatement
26     (h : UniversalStatement.{u}) : LeftHandedStatement.{u} := by
27   intro k _ _ n d hd
28   have htd : transpose k n d ≠ 0 := by
29     intro hz
30     apply hd
31     simpa only [transpose_transpose, transpose_zero] using congrArg (transpose k n) hz
32   rcases h k n (transpose k n d) htd with ⟨F, R, S, hone⟩
33   refine ⟨transpose k n R, transpose k n S, transpose k n F, ?_⟩
34   have ht := congrArg (transpose k n) hone
35   have h1 : transpose k n (1 : WeylAlg k n) = 1 := by simp [transpose]
36   rw [h1] at ht
37   simpa [transpose_add, transpose_mul, transpose_transpose, mul_assoc] using ht
38 
39 end Stafford38.LeftHandedCorollary
