17 /-- A fraction-clearing formulation sufficient to transport `S38`. The unit
18 condition concerns only the denominator chosen for the given fraction. -/
19 theorem s38_of_rightClearing
20     {R : Type u} {L : Type v} [Ring R] [Ring L]
21     {f : R →+* L} (hR : S38 R)
22     (hclear : ∀ q : L, q ≠ 0 →
23       ∃ a s : R, IsUnit (f s) ∧ q * f s = f a) : S38 L :=
24   AlgebraicAnalysis.TwoGeneratorIdentity.of_rightUnitClearing hR hclear

30 /-- Stafford 3.8 is preserved by genuine right Ore localization. -/
31 theorem s38_rightOreLocalization
