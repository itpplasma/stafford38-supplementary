23 /-- The paper's two displayed vectors generate the free rank-two right module. -/
24 theorem paper_matrix_pair_spans (a F : A) :
25     Submodule.span Aᵐᵒᵖ
26       ({(a, 1 - F * a), (1, -F)} : Set (A × A)) = ⊤ :=
27   Stafford38.LinearAlgebra.PairReplacement.matrix_pair_spans a F

29 /-- Two individual torsion annihilators have a common nonzero right multiple. -/
30 theorem exists_common_right_annihilator_of_torsion
31     {M : Type v} [AddCommGroup M] [Module Aᵐᵒᵖ M]
32     (htorsion : Stafford38.TorsionCyclicity.IsRightTorsion
33       (A := A) (M := M)) (x y : M)
34     (hmul : ∀ a b : A, a ≠ 0 → b ≠ 0 → a * b ≠ 0) :
35     ∃ d : A, d ≠ 0 ∧
36       (MulOpposite.op d : Aᵐᵒᵖ) • x = 0 ∧
37       (MulOpposite.op d : Aᵐᵒᵖ) • y = 0 :=
38   Stafford38.LinearAlgebra.PairReplacement.common_right_annihilator
39     htorsion x y hmul

41 /-- The matrix calculation gives pair replacement for a specified common
42 annihilator and certificate. -/
43 theorem span_adjusted_pair_eq_span_pair_via_matrix
44     {M : Type v} [AddCommGroup M] [Module Aᵐᵒᵖ M]
45     (x y : M) (d F R S : A)
46     (hxann : (MulOpposite.op d : Aᵐᵒᵖ) • x = 0)
47     (hyann : (MulOpposite.op d : Aᵐᵒᵖ) • y = 0)
48     (hcert : (1 : A) = d * R + F * d * S) :
49     Submodule.span Aᵐᵒᵖ
50         ({x - (MulOpposite.op F : Aᵐᵒᵖ) • y} : Set M) =
51       Submodule.span Aᵐᵒᵖ ({x, y} : Set M) :=
52   Stafford38.LinearAlgebra.PairReplacement.span_adjusted_pair_eq_span_pair
53     x y d F R S hxann hyann hcert

55 /-- Apply the paper's certificate and matrix calculation to two torsion
56 module elements. -/
57 theorem exists_span_adjusted_pair_via_matrix
58     {M : Type v} [AddCommGroup M] [Module Aᵐᵒᵖ M]
59     (hmul : ∀ a b : A, a ≠ 0 → b ≠ 0 → a * b ≠ 0)
60     (hone : ∀ d : A, d ≠ 0 →
61       ∃ F R S : A, (1 : A) = d * R + F * d * S)
62     (htorsion : Stafford38.TorsionCyclicity.IsRightTorsion
63       (A := A) (M := M)) (x y : M) :
64     ∃ F : A, Submodule.span Aᵐᵒᵖ
65       ({x - (MulOpposite.op F : Aᵐᵒᵖ) • y} : Set M) =
66         Submodule.span Aᵐᵒᵖ ({x, y} : Set M) :=
67   Stafford38.LinearAlgebra.PairReplacement.exists_adjusted_pair
68     hmul hone htorsion x y

70 /-- Forget the explicit coefficient and retain its generated singleton. -/
71 theorem exists_span_singleton_eq_span_pair_via_matrix
72     {M : Type v} [AddCommGroup M] [Module Aᵐᵒᵖ M]
73     (hmul : ∀ a b : A, a ≠ 0 → b ≠ 0 → a * b ≠ 0)
74     (hone : ∀ d : A, d ≠ 0 →
75       ∃ F R S : A, (1 : A) = d * R + F * d * S)
76     (htorsion : Stafford38.TorsionCyclicity.IsRightTorsion
77       (A := A) (M := M)) (x y : M) :
78     ∃ z : M, Submodule.span Aᵐᵒᵖ ({z} : Set M) =
79       Submodule.span Aᵐᵒᵖ ({x, y} : Set M) :=
80   Stafford38.LinearAlgebra.PairReplacement.exists_singleton_eq_span_pair

83 /-- Exact cyclicity statement in the Weyl-algebra presentation. -/
84 theorem paper_torsion_module_is_cyclic
85     (k : Type u) [Field k] [CharZero k] (n : ℕ)
86     (M : Type v) [AddCommGroup M]
87     [Module (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ M]
88     [Module.Finite (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ M]
89     (hM : Stafford38.TorsionCyclicity.IsRightTorsion
90       (A := Stafford38Challenge.WeylAlg k n) (M := M)) :
91     ∃ z : M,
92       Submodule.span (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ ({z} : Set M) = ⊤ :=
93   Stafford38.LinearAlgebra.PairReplacement.finite_torsion_module_is_cyclic
94     (A := Stafford38Challenge.WeylAlg k n) (M := M)
95     (fun _ _ ha hb => Stafford38.WeylDomain.mul_ne_zero ha hb)
96     (Stafford38.universalStatement (k := k) n) hM
