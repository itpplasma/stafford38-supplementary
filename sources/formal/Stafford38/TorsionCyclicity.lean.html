25 /-- The paper's generator calculation for a specified common annihilator and
26 certificate, derived from the canonical matrix proof. -/
27 theorem span_adjusted_pair_eq_span_pair
28     (x y : M) (d F R S : A)
29     (hxann : (MulOpposite.op d : Aᵐᵒᵖ) • x = 0)
30     (hyann : (MulOpposite.op d : Aᵐᵒᵖ) • y = 0)
31     (hcert : (1 : A) = d * R + F * d * S) :
32     Submodule.span Aᵐᵒᵖ
33         ({x - (MulOpposite.op F : Aᵐᵒᵖ) • y} : Set M) =
34       Submodule.span Aᵐᵒᵖ ({x, y} : Set M) :=
35   Stafford38.PaperCyclicity.span_adjusted_pair_eq_span_pair_via_matrix
36     x y d F R S hxann hyann hcert

38 /-- The paper generator collapses any torsion pair to one adjusted generator. -/
39 theorem exists_span_adjusted_pair
40     (hmul : ∀ a b : A, a ≠ 0 → b ≠ 0 → a * b ≠ 0)
41     (hone : ∀ d : A, d ≠ 0 →
42       ∃ F R S : A, (1 : A) = d * R + F * d * S)
43     (htorsion : IsRightTorsion (A := A) (M := M)) (x y : M) :
44     ∃ F : A,
45       Submodule.span Aᵐᵒᵖ
46         ({x - (MulOpposite.op F : Aᵐᵒᵖ) • y} : Set M) =
47         Submodule.span Aᵐᵒᵖ ({x, y} : Set M) :=
48   Stafford38.PaperCyclicity.exists_span_adjusted_pair_via_matrix
49     hmul hone htorsion x y

51 /-- Forget the displayed coefficient and keep the generated singleton. -/
52 theorem exists_span_singleton_eq_span_pair

54     (hone : ∀ d : A, d ≠ 0 →

63 /-- Every finitely generated torsion right module over a Weyl algebra is
64 cyclic. The main theorem supplies the coefficient identity used at each
65 pair-replacement step in the finite-generator induction. -/
66 theorem weyl_isCyclic_of_isRightTorsion
67     (k : Type u) [Field k] [CharZero k] (n : ℕ)
68     (M : Type v) [AddCommGroup M]
69     [Module (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ M]
70     [Module.Finite (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ M]
71     (hM : IsRightTorsion (A := Stafford38Challenge.WeylAlg k n) (M := M)) :
72     ∃ z : M,
73       Submodule.span (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ ({z} : Set M) = ⊤ :=
74   Stafford38.PaperCyclicity.paper_torsion_module_is_cyclic
75     (k := k) (n := n) (M := M) hM
