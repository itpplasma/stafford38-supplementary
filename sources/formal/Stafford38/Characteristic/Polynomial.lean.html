9 /-!
10 # Polynomial phase space for characteristic support
11 
12 The order symbol ring of the `n`th Weyl algebra has base variables of fibre
13 weight zero and cotangent variables of fibre weight one. This file introduces
14 that ring and its canonical polynomial Poisson bracket without assuming any
15 characteristic-variety theorem.
16 -/
17 
18 namespace Stafford38.Characteristic
19 
20 /-- Coordinate ring of affine cotangent space. -/
21 abbrev SymbolRing (k : Type*) [CommRing k] (n : ℕ) :=
22   MvPolynomial (PhaseVar n) k
23 
24 /-- Fibre degree: base variables have weight zero and covariables weight one. -/
25 def fibreWeight {n : ℕ} : PhaseVar n → ℕ
26   | Sum.inl _ => 0
27   | Sum.inr _ => 1
28 
29 /-- The standard Poisson bracket on affine cotangent space. -/
30 noncomputable def poissonBracket
31     {k : Type*} [CommRing k] {n : ℕ}
32     (f g : SymbolRing k n) : SymbolRing k n :=
33   ∑ i : Fin n,
34     (MvPolynomial.pderiv (Sum.inl i : PhaseVar n) f *
35         MvPolynomial.pderiv (Sum.inr i : PhaseVar n) g -
36       MvPolynomial.pderiv (Sum.inr i : PhaseVar n) f *
37         MvPolynomial.pderiv (Sum.inl i : PhaseVar n) g)
38 
39 /-- An ideal is Poisson when it is stable under bracketing with every symbol. -/
40 def IsPoisson
41     {k : Type*} [CommRing k] {n : ℕ}
42     (J : Ideal (SymbolRing k n)) : Prop :=
43   ∀ f ∈ J, ∀ g, poissonBracket f g ∈ J
44 
45 @[simp] theorem poissonBracket_self
46     {k : Type*} [CommRing k] {n : ℕ} (f : SymbolRing k n) :
47     poissonBracket f f = 0 := by
48   simp [poissonBracket, mul_comm]
49 
50 end Stafford38.Characteristic
