10 /-!
11 # Source matrices for the concrete equation-(3.3) bridge
12 
13 This file carries out source-level matrix algebra preceding the trace
14 argument in Singh--Kumar, Proposition 3.2. It keeps the deformation ring
15 noncommutative and preserves the written left-factor orientation of the
16 central parameter. It constructs strict parameter cofactors and literal
17 trace cofactors for first-order source matrices.
18 
19 The module records the specialized source action but does not identify it
20 with the operator induced by the bracket cofactor on the concrete localized
21 module. The canonical-support proof uses the separate
22 `PaperLocalizedInvolutivity` route.
23 -/

51 /-- A surjective reduction with an exact written-order left-principal kernel.
52 This is the elementwise form supplied by the concrete localized Rees
53 specialization. -/
54 structure LeftPrincipalParameterReduction where
55   parameter : B
56   modParameter : B →+* Abar
57   surjective : Function.Surjective modParameter
58   parameter_comm : ∀ z : B, parameter * z = z * parameter
59   kernel_left : ∀ z : B,
60     modParameter z = 0 ↔ ∃ w : B, parameter * w = z
61 
62 /-- The source's ideal-valued reduction follows from the stronger explicit
63 left-principal kernel, using centrality only to match Mathlib's left-ideal
64 generator convention. -/
65 def LeftPrincipalParameterReduction.toParameterIdealReduction
66     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar)) :
67     ParameterIdealReduction B Abar where
68   parameter := S.parameter
69   modParameter := S.modParameter
70   ker_modParameter := by
71     ext z
72     constructor
73     · intro hz
74       obtain ⟨w, hw⟩ := (S.kernel_left z).1 hz
75       rw [Ideal.mem_span_singleton']
76       exact ⟨w, (S.parameter_comm w).symm.trans hw⟩
77     · intro hz
78       rw [Ideal.mem_span_singleton'] at hz
79       obtain ⟨w, rfl⟩ := hz
80       rw [← S.parameter_comm w]
81       exact (S.kernel_left (S.parameter * w)).2 ⟨w, rfl⟩

83 /-- A zero-preserving choice of a lift through a surjective reduction. -/
84 def zeroPreservingLift
85     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
86     (z : Abar) : B := by
87   classical
88   exact if hz : z = 0 then 0 else Classical.choose (S.surjective z)

95   classical
96   by_cases hz : z = 0
97   · simp [zeroPreservingLift, hz]

107 /-- A zero-preserving written-order parameter cofactor. -/
108 def principalCofactor
109     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
110     (z : B) : B := by
111   classical
112   exact if hz : z = 0 then 0
113     else if hker : S.modParameter z = 0 then
114       Classical.choose ((S.kernel_left z).1 hker)
115     else 0

117 theorem parameter_mul_principalCofactor
118     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
119     {z : B} (hz : S.modParameter z = 0) :
120     S.parameter * principalCofactor S z = z := by
121   classical
122   by_cases hzero : z = 0
123   · simp [principalCofactor, hzero]
124   · simp only [principalCofactor, hzero, hz, ↓reduceDIte]
125     exact Classical.choose_spec ((S.kernel_left z).1 hz)

142 /-- Entrywise scalar ringCommutator, in the order used in equation (3.3). -/
143 def scalarMatrixCommutator
144     (x : B) (M : Matrix (Fin (n + 1)) (Fin (n + 1)) B) :
145     Matrix (Fin (n + 1)) (Fin (n + 1)) B :=
146   fun i j => x * M i j - M i j * x

148 /-- Written-order matrix ringCommutator over the deformation ring. -/
149 def sourceMatrixCommutator
150     (M N : Matrix (Fin (n + 1)) (Fin (n + 1)) B) :
151     Matrix (Fin (n + 1)) (Fin (n + 1)) B :=
152   M * N - N * M

239 theorem sourceMatrixCommutator_map_eq_zero
240     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
241     (M N : Matrix (Fin (n + 1)) (Fin (n + 1)) B)
242     (hcomm : M.map S.modParameter * N.map S.modParameter =
243       N.map S.modParameter * M.map S.modParameter) :
244     (sourceMatrixCommutator M N).map S.modParameter = 0 := by
245   ext i j
246   have hij := congrArg (fun Q => Q i j) hcomm
247   change S.modParameter ((M * N) i j - (N * M) i j) = 0
248   rw [S.modParameter.map_sub, Matrix.mul_apply, Matrix.mul_apply]
249   simp only [map_sum, S.modParameter.map_mul]
250   simpa [Matrix.mul_apply, Matrix.map_apply] using sub_eq_zero.mpr hij

266 /-- The five cofactors retained from the primary-source calculation: three
267 strict matrix cofactors and two scalar trace cofactors. -/
268 structure Equation33Cofactors
269     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
270     (x y : B)
271     (A Bm Gamma Theta : Matrix (Fin (n + 1)) (Fin (n + 1)) B) where
272   X : Matrix (Fin (n + 1)) (Fin (n + 1)) B
273   Y : Matrix (Fin (n + 1)) (Fin (n + 1)) B
274   Omega : Matrix (Fin (n + 1)) (Fin (n + 1)) B
275   tauThetaA : B
276   tauBGamma : B
277   xB_eq : ∀ i j, S.parameter * X i j = scalarMatrixCommutator x Bm i j
278   yA_eq : ∀ i j, S.parameter * Y i j = scalarMatrixCommutator y A i j
279   BA_eq : ∀ i j,
280     S.parameter * Omega i j = sourceMatrixCommutator Bm A i j
281   X_strict : IsStrictUpperTriangularOver X
282   Y_strict : IsStrictUpperTriangularOver Y
283   Omega_strict : IsStrictUpperTriangularOver Omega
284   trace_ThetaA_eq :
285     S.parameter * tauThetaA = Matrix.trace (sourceMatrixCommutator Theta A)
286   trace_BGamma_eq :
287     S.parameter * tauBGamma = Matrix.trace (sourceMatrixCommutator Bm Gamma)
288 
289 /-- The source cofactors are constructed, not assumed. -/
290 def equation33Cofactors
291     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
292     (x y : B)
293     (A Bm Gamma Theta : Matrix (Fin (n + 1)) (Fin (n + 1)) B)
294     (hA : IsStrictUpperTriangularOver A)
295     (hB : IsStrictUpperTriangularOver Bm)
296     (hBAcomm : Bm.map S.modParameter * A.map S.modParameter =
297       A.map S.modParameter * Bm.map S.modParameter) :
298     Equation33Cofactors S x y A Bm Gamma Theta := by
299   let X := factorMatrix S (scalarMatrixCommutator x Bm)
300   let Y := factorMatrix S (scalarMatrixCommutator y A)
301   let Omega := factorMatrix S (sourceMatrixCommutator Bm A)
302   let tauThetaA := principalCofactor S
303     (Matrix.trace (sourceMatrixCommutator Theta A))
304   let tauBGamma := principalCofactor S
305     (Matrix.trace (sourceMatrixCommutator Bm Gamma))
306   refine
307     { X := X
308       Y := Y
309       Omega := Omega
310       tauThetaA := tauThetaA
311       tauBGamma := tauBGamma
312       xB_eq := parameter_mul_factorMatrix S _
313         (scalarMatrixCommutator_map_eq_zero S x Bm)
314       yA_eq := parameter_mul_factorMatrix S _
315         (scalarMatrixCommutator_map_eq_zero S y A)
316       BA_eq := parameter_mul_factorMatrix S _
317         (sourceMatrixCommutator_map_eq_zero S Bm A hBAcomm)
318       X_strict := factorMatrix_isStrictUpperTriangular S _
319         (scalarMatrixCommutator_isStrictUpperTriangular x Bm hB)
320       Y_strict := factorMatrix_isStrictUpperTriangular S _
321         (scalarMatrixCommutator_isStrictUpperTriangular y A hA)
322       Omega_strict := factorMatrix_isStrictUpperTriangular S _
323         (sourceMatrixCommutator_isStrictUpperTriangular Bm A hB hA)
324       trace_ThetaA_eq := ?_
325       trace_BGamma_eq := ?_ }
326   · apply parameter_mul_principalCofactor S
327     calc
328       S.modParameter (Matrix.trace (sourceMatrixCommutator Theta A)) =
329           Matrix.trace ((sourceMatrixCommutator Theta A).map S.modParameter) := by
330             simpa using
331               (AddMonoidHom.map_trace S.modParameter.toAddMonoidHom
332                 (sourceMatrixCommutator Theta A))
333       _ = 0 := sourceMatrixCommutator_map_trace_eq_zero S Theta A
334   · apply parameter_mul_principalCofactor S
335     calc
336       S.modParameter (Matrix.trace (sourceMatrixCommutator Bm Gamma)) =
337           Matrix.trace ((sourceMatrixCommutator Bm Gamma).map S.modParameter) := by
338             simpa using
339               (AddMonoidHom.map_trace S.modParameter.toAddMonoidHom
340                 (sourceMatrixCommutator Bm Gamma))
341       _ = 0 := sourceMatrixCommutator_map_trace_eq_zero S Bm Gamma

448 /-- Actual zeroth-order adapted-basis action matrices in Lean's column
449 convention, with strict zero-preserving lifts to the deformation ring.  The
450 returned lifts `x` and `y` specialize to the requested elements.  Passing to
451 the source's row convention requires the explicit transpose adapter above. -/
452 theorem exists_lifted_adapted_actionMatrices
453     (S : LeftPrincipalParameterReduction (B := B) (Abar := R))
454     (hdim : Module.finrank K V = n + 1)
455     (xbar ybar : R) (hx : xbar ∈ m) (hy : ybar ∈ m) :
456     ∃ (b : Module.Basis (Fin (n + 1)) K V)
457       (level : Fin (n + 1) → ℕ)
458       (x y : B)
459       (A Bm : Matrix (Fin (n + 1)) (Fin (n + 1)) B),
460       IsMaximalIdealFiltrationAdapted m b level ∧
461       S.modParameter x = xbar ∧
462       S.modParameter y = ybar ∧
463       A.map S.modParameter =
464         (LinearMap.toMatrix b b
465           (leftMultiplicationEnd (K := K) (V := V) xbar)).map
466             (algebraMap K R) ∧
467       Bm.map S.modParameter =
468         (LinearMap.toMatrix b b
469           (leftMultiplicationEnd (K := K) (V := V) ybar)).map
470             (algebraMap K R) ∧
471       IsStrictUpperTriangularOver A ∧
472       IsStrictUpperTriangularOver Bm ∧
473       Bm.map S.modParameter * A.map S.modParameter =
474         A.map S.modParameter * Bm.map S.modParameter := by
475   obtain ⟨b, level, hadapted⟩ :=
476     exists_maximalIdealFiltrationAdaptedBasis (K := K) (R := R) (V := V) hdim
477   let Ax := leftMultiplicationEnd (K := K) (V := V) xbar
478   let By := leftMultiplicationEnd (K := K) (V := V) ybar
479   let A0K := LinearMap.toMatrix b b Ax
480   let B0K := LinearMap.toMatrix b b By
481   let A0 := A0K.map (algebraMap K R)
482   let B0 := B0K.map (algebraMap K R)
483   let A := liftMatrix S A0
484   let Bm := liftMatrix S B0
485   let x := zeroPreservingLift S xbar
486   let y := zeroPreservingLift S ybar
487   have hA0K : IsStrictUpperTriangularOver A0K :=
488     leftMultiplicationMatrix_isStrictUpperTriangular b level hadapted hx
489   have hB0K : IsStrictUpperTriangularOver B0K :=
490     leftMultiplicationMatrix_isStrictUpperTriangular b level hadapted hy
491   have hA0 : IsStrictUpperTriangularOver A0 :=
492     fun i j hji => by simp [A0, Matrix.map_apply, hA0K i j hji]
493   have hB0 : IsStrictUpperTriangularOver B0 :=
494     fun i j hji => by simp [B0, Matrix.map_apply, hB0K i j hji]
495   have hEnd : By * Ax = Ax * By := by
496     ext v
497     change ybar • xbar • v = xbar • ybar • v
498     rw [← mul_smul, ← mul_smul, mul_comm]
499   have hK : B0K * A0K = A0K * B0K := by
500     rw [← LinearMap.toMatrix_mul, ← LinearMap.toMatrix_mul, hEnd]
501   have hcomm : B0 * A0 = A0 * B0 := by
502     change B0K.map (algebraMap K R) * A0K.map (algebraMap K R) =
503       A0K.map (algebraMap K R) * B0K.map (algebraMap K R)
