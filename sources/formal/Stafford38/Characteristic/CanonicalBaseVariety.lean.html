10 # Affine base points for the visible-frame producer
11 
12 Over an algebraically closed field, nonempty order-characteristic support
13 forces the contracted reduced base ideal to have a rational point. If the
14 support avoids the distinguished coordinate hyperplane, every such point has
15 nonzero distinguished coordinate. These lemmas supply affine input to
16 `ExactDivisorialVisibleFrameExistence`, which constructs the retained place
17 and visible frame.
18 -/

37 /-- A nonempty reduced order support makes its contracted base ideal proper. -/
38 theorem reducedOrderBaseIdeal_ne_top_of_support_nonempty
39     (I : RightIdeal (PresentedWeyl k n))
40     (hsupp : (orderCharacteristicSupport k I).Nonempty) :
41     reducedOrderBaseIdeal k I ≠ ⊤ := by
42   rintro htop
43   rcases hsupp with ⟨p, hp⟩
44   have hpReduced : p ∈ PrimeSpectrum.zeroLocus
45       (reducedOrderSupportIdeal k I : Set (SymbolRing k n)) := by
46     rw [← orderCharacteristicSupport_eq_zeroLocus_reduced k I]
47     exact hp
48   rw [PrimeSpectrum.mem_zeroLocus] at hpReduced
49   have honeBase : (1 : MvPolynomial (Fin n) k) ∈
50       reducedOrderBaseIdeal k I := by rw [htop]; exact Submodule.mem_top
51   have honeReduced : (1 : SymbolRing k n) ∈
52       reducedOrderSupportIdeal k I := by
53     simpa [baseLift] using
54       (mem_reducedOrderBaseIdeal_iff k I 1).mp honeBase
55   exact p.2.ne_top
56     ((Ideal.eq_top_iff_one p.asIdeal).2 (hpReduced honeReduced))

58 /-- Over an algebraically closed field, nonempty support therefore produces a
59 ground-field point of the contracted base zero set. -/
60 theorem exists_reducedOrderBaseZero_of_support_nonempty
61     (I : RightIdeal (PresentedWeyl k n))
62     (hsupp : (orderCharacteristicSupport k I).Nonempty) :
63     ∃ y : Fin n → k,
64       ∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0 := by
65   let B := reducedOrderBaseIdeal k I
66   obtain ⟨M, hMmax, hBM⟩ :=
67     Ideal.exists_le_maximal B
68       (reducedOrderBaseIdeal_ne_top_of_support_nonempty I hsupp)
69   obtain ⟨y, hMy⟩ :=
70     MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hMmax
71   refine ⟨y, ?_⟩
72   intro f hf
73   have hfM : f ∈ M := hBM hf
74   rw [hMy, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hfM
75   exact hfM

78 /-- If support avoids the coordinate-zero prime locus, every point of the
79 contracted base variety has nonzero distinguished coordinate. -/
80 theorem coordinate_ne_zero_of_baseZero_of_support_disjoint
81     (I : RightIdeal (PresentedWeyl k n)) (i : Fin n)
82     (hdisjoint : Disjoint
83       (orderCharacteristicSupport k I)
84       (PrimeSpectrum.zeroLocus
85         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
86     (y : Fin n → k)
87     (hy : ∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0) :
88     y i ≠ 0 := by
89   intro hyi
90   let q : PhaseVar n → k := zeroSectionPoint y
91   let p : PrimeSpectrum (SymbolRing k n) := MvPolynomial.pointToPoint q
92   have hzero : ∀ P ∈ reducedOrderSupportIdeal k I,
93       MvPolynomial.eval q P = 0 := by
94     exact zeroSection_mem_of_mem_reducedOrderBaseZeroSet k I y hy
95   have hpSupport : p ∈ orderCharacteristicSupport k I := by
96     rw [orderCharacteristicSupport_eq_zeroLocus_reduced]
97     rw [PrimeSpectrum.mem_zeroLocus]
98     intro P hP
99     change P ∈ MvPolynomial.vanishingIdeal k {q}
100     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff]
101     exact hzero P hP
102   have hpCoordinate : p ∈ PrimeSpectrum.zeroLocus
103       ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n)) := by
104     rw [PrimeSpectrum.mem_zeroLocus]
105     intro P hP
106     change P ∈ MvPolynomial.vanishingIdeal k {q}
107     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff,
108       Set.mem_singleton_iff.mp hP]
109     simp [q, zeroSectionPoint, hyi]
110   exact Set.disjoint_left.mp hdisjoint hpSupport hpCoordinate

112 /-- The two affine hypotheses required by the asymptotic boundary theorem:
113 the base zero set is inhabited and lies in the principal open `X_i ≠ 0`. -/
114 theorem exists_baseZero_and_all_coordinate_ne_zero
115     (I : RightIdeal (PresentedWeyl k n)) (i : Fin n)
116     (hdisjoint : Disjoint
117       (orderCharacteristicSupport k I)
118       (PrimeSpectrum.zeroLocus
119         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
120     (hsupp : (orderCharacteristicSupport k I).Nonempty) :
121     (∃ y : Fin n → k,
122       ∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0) ∧
123     (∀ y : Fin n → k,
124       (∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0) →
125       y i ≠ 0) := by
126   refine ⟨exists_reducedOrderBaseZero_of_support_nonempty I hsupp, ?_⟩
127   intro y hy
128   exact coordinate_ne_zero_of_baseZero_of_support_disjoint I i hdisjoint y hy
