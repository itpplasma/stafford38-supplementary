11 /-!
12 # Affine base points for the visible-frame producer
13 
14 Over an algebraically closed field, nonempty order-characteristic support
15 forces the contracted reduced base ideal to have a rational point. If the
16 support avoids the distinguished coordinate hyperplane, every such point has
17 nonzero distinguished coordinate. These lemmas supply affine input to
18 `ExactDivisorialVisibleFrameExistence`, which constructs the retained place
19 and visible frame.
20 -/

39 /-- A nonempty reduced order support makes its contracted base ideal proper. -/
40 theorem reducedOrderBaseIdeal_ne_top_of_support_nonempty
41     (I : RightIdeal (WeylAlg k n))
42     (hsupp : (orderCharacteristicSupport k I).Nonempty) :
43     reducedOrderBaseIdeal k I ≠ ⊤ := by
44   rintro htop
45   rcases hsupp with ⟨p, hp⟩
46   have hpReduced : p ∈ PrimeSpectrum.zeroLocus
47       (reducedOrderSupportIdeal k I : Set (SymbolRing k n)) := by
48     rw [← orderCharacteristicSupport_eq_zeroLocus_reduced k I]
49     exact hp
50   rw [PrimeSpectrum.mem_zeroLocus] at hpReduced
51   have honeBase : (1 : MvPolynomial (Fin n) k) ∈
52       reducedOrderBaseIdeal k I := by rw [htop]; exact Submodule.mem_top
53   have honeReduced : (1 : SymbolRing k n) ∈
54       reducedOrderSupportIdeal k I := by
55     simpa [baseLift] using
56       (mem_reducedOrderBaseIdeal_iff k I 1).mp honeBase
57   exact p.2.ne_top
58     ((Ideal.eq_top_iff_one p.asIdeal).2 (hpReduced honeReduced))

60 /-- Over an algebraically closed field, nonempty support therefore produces a
61 ground-field point of the contracted base zero set. -/
62 theorem exists_reducedOrderBaseZero_of_support_nonempty
63     (I : RightIdeal (WeylAlg k n))
64     (hsupp : (orderCharacteristicSupport k I).Nonempty) :
65     ∃ y : Fin n → k,
66       ∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0 := by
67   let B := reducedOrderBaseIdeal k I
68   obtain ⟨M, hMmax, hBM⟩ :=
69     Ideal.exists_le_maximal B
70       (reducedOrderBaseIdeal_ne_top_of_support_nonempty I hsupp)
71   obtain ⟨y, hMy⟩ :=
72     MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hMmax
73   refine ⟨y, ?_⟩
74   intro f hf
75   have hfM : f ∈ M := hBM hf
76   rw [hMy, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hfM
77   exact hfM

80 /-- If support avoids the coordinate-zero prime locus, every point of the
81 contracted base variety has nonzero distinguished coordinate. -/
82 theorem coordinate_ne_zero_of_baseZero_of_support_disjoint
83     (I : RightIdeal (WeylAlg k n)) (i : Fin n)
84     (hdisjoint : Disjoint
85       (orderCharacteristicSupport k I)
86       (PrimeSpectrum.zeroLocus
87         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
88     (y : Fin n → k)
89     (hy : ∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0) :
90     y i ≠ 0 := by
91   intro hyi
92   let q : PhaseVar n → k := zeroSectionPoint y
93   let p : PrimeSpectrum (SymbolRing k n) := MvPolynomial.pointToPoint q
94   have hzero : ∀ P ∈ reducedOrderSupportIdeal k I,
95       MvPolynomial.eval q P = 0 := by
96     exact zeroSection_mem_of_mem_reducedOrderBaseZeroSet k I y hy
97   have hpSupport : p ∈ orderCharacteristicSupport k I := by
98     rw [orderCharacteristicSupport_eq_zeroLocus_reduced]
99     rw [PrimeSpectrum.mem_zeroLocus]
100     intro P hP
101     change P ∈ MvPolynomial.vanishingIdeal k {q}
102     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff]
103     exact hzero P hP
104   have hpCoordinate : p ∈ PrimeSpectrum.zeroLocus
105       ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n)) := by
106     rw [PrimeSpectrum.mem_zeroLocus]
107     intro P hP
108     change P ∈ MvPolynomial.vanishingIdeal k {q}
109     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff,
110       Set.mem_singleton_iff.mp hP]
111     simp [q, zeroSectionPoint, hyi]
112   exact Set.disjoint_left.mp hdisjoint hpSupport hpCoordinate

114 /-- The two affine hypotheses required by the asymptotic boundary theorem:
115 the base zero set is inhabited and lies in the principal open `X_i ≠ 0`. -/
116 theorem exists_baseZero_and_all_coordinate_ne_zero
117     (I : RightIdeal (WeylAlg k n)) (i : Fin n)
118     (hdisjoint : Disjoint
119       (orderCharacteristicSupport k I)
120       (PrimeSpectrum.zeroLocus
121         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
122     (hsupp : (orderCharacteristicSupport k I).Nonempty) :
123     (∃ y : Fin n → k,
124       ∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0) ∧
125     (∀ y : Fin n → k,
126       (∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0) →
127       y i ≠ 0) := by
128   refine ⟨exists_reducedOrderBaseZero_of_support_nonempty I hsupp, ?_⟩
129   intro y hy
130   exact coordinate_ne_zero_of_baseZero_of_support_disjoint I i hdisjoint y hy
