11 # Rees--Koszul criteria for canonical axis cancellation
12 
13 This file relates ordinary coordinate surjectivity to restriction on the
14 actual order-associated-graded module and to a one-element Rees--Koszul
15 complex. It does not identify ordinary quotient surjectivity with graded
16 surjectivity. The current canonical-support proof uses the separate
17 canonical-page argument in `CanonicalKoszulContradiction`.
18 -/

70 /-- The initial-ideal target is exactly surjectivity of coordinate
71 multiplication on the actual order-associated graded module.  Cyclicity is
72 used in both directions, and the annihilator is the literal order initial
73 ideal. -/
74 theorem canonical_orderInitialIdeal_sup_coordinate_eq_top_iff_graded_surjective
75     (n N : ℕ) (d : PresentedWeyl k (n + 1)) :
76     orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
77           Ideal.span {AxisCoordinate k n} = ⊤ ↔
78       Function.Surjective
79         (fun q : OrderAssociatedGradedModule k (CanonicalIdeal k n N d) ↦
80           AxisCoordinate k n • q) := by
81   classical
82   let I := CanonicalIdeal k n N d
83   let X := AxisCoordinate k n
84   let g := orderAssociatedGradedGenerator k I
85   constructor
86   · intro htop q
87     obtain ⟨P, rfl⟩ := exists_smul_orderAssociatedGradedGenerator k I q
88     have hone : (1 : SymbolRing k (n + 1)) ∈
89         orderInitialIdeal k I ⊔ Ideal.span {X} := by
90       rw [htop]
91       exact Submodule.mem_top
92     obtain ⟨j, hj, s, hs, hjs⟩ := Submodule.mem_sup.mp hone
93     obtain ⟨a, ha⟩ := Ideal.mem_span_singleton.mp hs
94     have hjg : j • g = 0 :=
95       (smul_orderAssociatedGradedGenerator_eq_zero_iff k I j).2 hj
96     have hsg : s • g = g := by
97       calc
98         s • g = (j + s) • g := by rw [add_smul, hjg, zero_add]
99         _ = 1 • g := congrArg (fun r : SymbolRing k (n + 1) => r • g) hjs
100         _ = g := one_smul _ _
101     refine ⟨(P * a) • g, ?_⟩
102     calc
103       X • ((P * a) • g) = (X * (P * a)) • g := by rw [smul_smul]
104       _ = (P * s) • g := by rw [ha]; ring
105       _ = P • (s • g) := by rw [smul_smul]
106       _ = P • g := by rw [hsg]
107   · intro hsurj
108     obtain ⟨q, hq⟩ := hsurj
109       (orderAssociatedGradedGenerator k I)
110     obtain ⟨P, hP⟩ := exists_smul_orderAssociatedGradedGenerator k I q
111     have hq' : X • (P • orderAssociatedGradedGenerator k I) =
112         orderAssociatedGradedGenerator k I := by
113       rw [hP]
114       simpa [X, I] using hq
115     have hkill : (X * P - 1) • orderAssociatedGradedGenerator k I = 0 := by
116       rw [sub_smul, mul_smul, hq', one_smul, sub_self]
117     have hrel : X * P - 1 ∈ orderInitialIdeal k I :=
118       (smul_orderAssociatedGradedGenerator_eq_zero_iff k I (X * P - 1)).1
119         hkill
120     apply (Ideal.eq_top_iff_one _).2
121     rw [show (1 : SymbolRing k (n + 1)) = -(X * P - 1) + X * P by ring]
122     apply Submodule.add_mem
123     · exact Submodule.mem_sup_left ((orderInitialIdeal k I).neg_mem hrel)
124     · apply Submodule.mem_sup_right
125       exact (Ideal.span {X}).mul_mem_right P
126         (Ideal.subset_span (Set.mem_singleton X))

131 theorem canonical_orderInitialIdeal_sup_coordinate_eq_top_iff_koszulH0_subsingleton

138     (R := SymbolRing k (n + 1))

152 /-- The filtered additive model and the literal regular-right-module quotient
153 intertwine their written right coordinate actions. -/
154 theorem filteredRightQuotientEquivRightQuotient_coordinate
155     (n N : ℕ) (d : PresentedWeyl k (n + 1))
156     (q : FilteredRightQuotient k (CanonicalIdeal k n N d)) :
157     filteredRightQuotientEquivRightQuotient k (CanonicalIdeal k n N d)
158         (filteredRightMul k (CanonicalIdeal k n N d)

160       rightMul (CanonicalIdeal k n N d) (presentedCoordinate k n)

162           (CanonicalIdeal k n N d) q) := by
163   refine Submodule.Quotient.induction_on _ q ?_

167 /-- PBW monicity makes the ordinary one-coordinate Koszul `H₀` literally
168 zero.  The requested initial-top theorem is the assertion that this remains
169 true after passing to the order-Rees special fibre. -/
170 theorem canonical_ordinary_coordinateKoszulH0_subsingleton_of_axisMonic
171     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
172     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
173     Subsingleton (CanonicalOrdinaryCoordinateKoszulH0 k n N d) := by
174   rw [Submodule.Quotient.subsingleton_iff]
175   apply top_unique
176   intro q hq
177   let e := filteredRightQuotientEquivRightQuotient k
178     (CanonicalIdeal k n N d)
179   obtain ⟨r, hr⟩ :=
180     canonical_ordinary_coordinate_surjective_of_axisMonic k n N hd (e q)
181   refine ⟨e.symm r, ?_⟩
182   apply e.injective
183   rw [filteredRightQuotientEquivRightQuotient_coordinate]
184   simpa [e] using hr
