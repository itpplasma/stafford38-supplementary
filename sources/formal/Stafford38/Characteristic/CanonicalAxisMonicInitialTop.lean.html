13 # Rees--Koszul criteria for canonical axis cancellation
14 
15 This file relates ordinary coordinate surjectivity to restriction on the
16 actual order-associated-graded module and to a one-element Rees--Koszul
17 complex. It does not identify ordinary quotient surjectivity with graded
18 surjectivity. The current canonical-support proof uses the separate
19 canonical-page argument in `CanonicalKoszulContradiction`.
20 -/

72 /-- The initial-ideal target is exactly surjectivity of coordinate
73 multiplication on the actual order-associated graded module.  Cyclicity is
74 used in both directions, and the annihilator is the literal order initial
75 ideal. -/
76 theorem canonical_orderInitialIdeal_sup_coordinate_eq_top_iff_graded_surjective
77     (n N : ℕ) (d : WeylAlg k (n + 1)) :
78     orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
79           Ideal.span {AxisCoordinate k n} = ⊤ ↔
80       Function.Surjective
81         (fun q : OrderAssociatedGradedModule k (CanonicalIdeal k n N d) ↦
82           AxisCoordinate k n • q) := by
83   classical
84   let I := CanonicalIdeal k n N d
85   let X := AxisCoordinate k n
86   let g := orderAssociatedGradedGenerator k I
87   constructor
88   · intro htop q
89     obtain ⟨P, rfl⟩ := exists_smul_orderAssociatedGradedGenerator k I q
90     have hone : (1 : SymbolRing k (n + 1)) ∈
91         orderInitialIdeal k I ⊔ Ideal.span {X} := by
92       rw [htop]
93       exact Submodule.mem_top
94     obtain ⟨j, hj, s, hs, hjs⟩ := Submodule.mem_sup.mp hone
95     obtain ⟨a, ha⟩ := Ideal.mem_span_singleton.mp hs
96     have hjg : j • g = 0 :=
97       (smul_orderAssociatedGradedGenerator_eq_zero_iff k I j).2 hj
98     have hsg : s • g = g := by
99       calc
100         s • g = (j + s) • g := by rw [add_smul, hjg, zero_add]
101         _ = 1 • g := congrArg (fun r : SymbolRing k (n + 1) => r • g) hjs
102         _ = g := one_smul _ _
103     refine ⟨(P * a) • g, ?_⟩
104     calc
105       X • ((P * a) • g) = (X * (P * a)) • g := by rw [smul_smul]
106       _ = (P * s) • g := by rw [ha]; ring
107       _ = P • (s • g) := by rw [smul_smul]
108       _ = P • g := by rw [hsg]
109   · intro hsurj
110     obtain ⟨q, hq⟩ := hsurj
111       (orderAssociatedGradedGenerator k I)
112     obtain ⟨P, hP⟩ := exists_smul_orderAssociatedGradedGenerator k I q
113     have hq' : X • (P • orderAssociatedGradedGenerator k I) =
114         orderAssociatedGradedGenerator k I := by
115       rw [hP]
116       simpa [X, I] using hq
117     have hkill : (X * P - 1) • orderAssociatedGradedGenerator k I = 0 := by
118       rw [sub_smul, mul_smul, hq', one_smul, sub_self]
119     have hrel : X * P - 1 ∈ orderInitialIdeal k I :=
120       (smul_orderAssociatedGradedGenerator_eq_zero_iff k I (X * P - 1)).1
121         hkill
122     apply (Ideal.eq_top_iff_one _).2
123     rw [show (1 : SymbolRing k (n + 1)) = -(X * P - 1) + X * P by ring]
124     apply Submodule.add_mem
125     · exact Submodule.mem_sup_left ((orderInitialIdeal k I).neg_mem hrel)
126     · apply Submodule.mem_sup_right
127       exact (Ideal.span {X}).mul_mem_right P
128         (Ideal.subset_span (Set.mem_singleton X))

130 /-- Exact Koszul formulation of the requested theorem: the initial ideal and
131 the coordinate generate one iff the specialized one-coordinate Koszul
132 `H₀` is zero. -/
133 theorem canonical_orderInitialIdeal_sup_coordinate_eq_top_iff_koszulH0_subsingleton
134     (n N : ℕ) (d : WeylAlg k (n + 1)) :
135     orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
136           Ideal.span {AxisCoordinate k n} = ⊤ ↔
137       Subsingleton (CanonicalGradedCoordinateKoszulH0 k n N d) := by
138   rw [canonical_orderInitialIdeal_sup_coordinate_eq_top_iff_graded_surjective]

140     (R := SymbolRing k (n + 1))
141     (M := OrderAssociatedGradedModule k (CanonicalIdeal k n N d))

154 /-- The filtered additive model and the literal regular-right-module quotient
155 intertwine their written right coordinate actions. -/
156 theorem filteredRightQuotientEquivRightQuotient_coordinate
157     (n N : ℕ) (d : WeylAlg k (n + 1))
158     (q : FilteredRightQuotient k (CanonicalIdeal k n N d)) :
159     filteredRightQuotientEquivRightQuotient k (CanonicalIdeal k n N d)
160         (filteredRightMul k (CanonicalIdeal k n N d)
161           (presentedCoordinate k n) q) =
162       rightMul (CanonicalIdeal k n N d) (presentedCoordinate k n)
163         (filteredRightQuotientEquivRightQuotient k
164           (CanonicalIdeal k n N d) q) := by
165   refine Submodule.Quotient.induction_on _ q ?_
166   intro z
167   rfl

169 /-- PBW monicity makes the ordinary one-coordinate Koszul `H₀` literally
170 zero.  The requested initial-top theorem is the assertion that this remains
171 true after passing to the order-Rees special fibre. -/
172 theorem canonical_ordinary_coordinateKoszulH0_subsingleton_of_axisMonic
173     (n N : ℕ) {d : WeylAlg k (n + 1)}
174     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
175     Subsingleton (CanonicalOrdinaryCoordinateKoszulH0 k n N d) := by
176   rw [Submodule.Quotient.subsingleton_iff]
177   apply top_unique
178   intro q hq
179   let e := filteredRightQuotientEquivRightQuotient k
180     (CanonicalIdeal k n N d)
181   obtain ⟨r, hr⟩ :=
182     canonical_ordinary_coordinate_surjective_of_axisMonic k n N hd (e q)
183   refine ⟨e.symm r, ?_⟩
184   apply e.injective
185   rw [filteredRightQuotientEquivRightQuotient_coordinate]
186   simpa [e] using hr
