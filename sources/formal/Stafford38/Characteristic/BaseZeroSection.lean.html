9 /-!
10 # Zero-section containment over the contracted base zero locus
11 
12 The fibre-degree-zero component of a phase-space polynomial is its
13 specialization at zero fibre coordinates, lifted back from the base ring.
14 Consequently a common zero of the contracted reduced order-support ideal
15 already determines a zero-section point of the full reduced support; no
16 chosen point in the original support fibre is needed.
17 -/

45 /-- Set all fibre variables to zero and retain the base variables. -/
46 def fibreZeroSpecialization :
47     SymbolRing k n →ₐ[k] MvPolynomial (Fin n) k :=
48   MvPolynomial.bind₁ (Sum.elim MvPolynomial.X (fun _ => 0))
49 
50 @[simp]
51 theorem fibreZeroSpecialization_X_base (i : Fin n) :
52     fibreZeroSpecialization k (MvPolynomial.X (Sum.inl i)) =
53       MvPolynomial.X i := by
54   simp [fibreZeroSpecialization]
55 
56 @[simp]
57 theorem fibreZeroSpecialization_X_fibre (i : Fin n) :
58     fibreZeroSpecialization k (MvPolynomial.X (Sum.inr i)) = 0 := by
59   simp [fibreZeroSpecialization]

61 /-- Evaluating after zero-fibre specialization is evaluation at the
62 corresponding zero-section point. -/
63 theorem eval_fibreZeroSpecialization
64     (y : Fin n → k) (P : SymbolRing k n) :
65     MvPolynomial.eval y (fibreZeroSpecialization k P) =
66       MvPolynomial.eval (zeroSectionPoint y) P := by
67   change MvPolynomial.aeval y
68       (MvPolynomial.bind₁ (Sum.elim MvPolynomial.X (fun _ => 0)) P) = _
69   rw [MvPolynomial.aeval_bind₁]
70   apply MvPolynomial.eval₂_congr
71   intro i c hi hc
72   rcases i with i | i <;> simp [zeroSectionPoint]
73 
74 /-- A base polynomial lifted to phase space evaluates independently of the
75 fibre coordinate. -/
76 theorem eval_baseLift
77     (y ξ : Fin n → k) (f : MvPolynomial (Fin n) k) :
78     MvPolynomial.eval (Sum.elim y ξ) (baseLift f) =
79       MvPolynomial.eval y f := by
80   change MvPolynomial.aeval (Sum.elim y ξ)
81       (MvPolynomial.rename Sum.inl f) = _
82   rw [MvPolynomial.aeval_rename]
83   rfl

85 /-- The order-degree-zero component is exactly zero-fibre specialization,
86 viewed again in the phase-space polynomial ring. -/
87 theorem zeroComponent_eq_baseLift_fibreZeroSpecialization
88     (P : SymbolRing k n) :
89     (DirectSum.decompose (orderDecomposition (n := n) k) P 0 :
90         SymbolRing k n) =
91       baseLift (fibreZeroSpecialization k P) := by
92   induction P using MvPolynomial.induction_on with
93   | C a =>
94       have hC : MvPolynomial.C a ∈ orderDecomposition (n := n) k 0 :=
95         MvPolynomial.isWeightedHomogeneous_C (@orderWeight n) a
96       let C0 : orderDecomposition (n := n) k 0 := ⟨MvPolynomial.C a, hC⟩
97       have hd := congrArg (fun z => z 0)
98         (DirectSum.decompose_coe (orderDecomposition (n := n) k) C0)
99       have hd' : (DirectSum.decompose (orderDecomposition (n := n) k)
100           (MvPolynomial.C a) 0 : SymbolRing k n) = MvPolynomial.C a := by
101         simpa [C0] using congrArg Subtype.val hd
102       rw [hd']
103       simp [fibreZeroSpecialization, baseLift]
104   | add P Q hP hQ =>
105       rw [DirectSum.decompose_add]
106       change
107         (DirectSum.decompose (orderDecomposition (n := n) k) P 0 :
108             SymbolRing k n) +
109           (DirectSum.decompose (orderDecomposition (n := n) k) Q 0 :
110             SymbolRing k n) = _
111       simp only [map_add, hP, hQ]
112   | mul_X P i hP =>
113       rcases i with i | i
114       · have hX : MvPolynomial.X (Sum.inl i : PhaseVar n) ∈
115             orderDecomposition (n := n) k 0 :=
116           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) (Sum.inl i)
117         rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
118           (orderDecomposition (n := n) k) hX (Nat.zero_le 0)]
119         simp only [Nat.zero_sub, map_mul, hP, fibreZeroSpecialization_X_base]
120         simp [baseLift]
121       · have hX : MvPolynomial.X (Sum.inr i : PhaseVar n) ∈
122             orderDecomposition (n := n) k 1 :=
123           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) (Sum.inr i)
124         rw [DirectSum.coe_decompose_mul_of_right_mem_of_not_le
125           (orderDecomposition (n := n) k) hX (by omega)]
126         simp

128 /-- Every base point of the contracted reduced order-support zero locus lifts
129 canonically to a zero-section point of the full reduced support. -/
130 theorem zeroSection_mem_of_mem_reducedOrderBaseZeroSet
131     (I : RightIdeal (WeylAlg k n)) (y : Fin n → k)
132     (hy : ∀ f ∈ reducedOrderBaseIdeal k I,
133       MvPolynomial.eval y f = 0) :
134     ∀ P ∈ reducedOrderSupportIdeal k I,
135       MvPolynomial.eval (zeroSectionPoint y) P = 0 := by
136   intro P hP
137   obtain ⟨m, hm⟩ := (mem_reducedOrderSupportIdeal_iff k I P).mp hP
138   let f := fibreZeroSpecialization k (P ^ m)
139   have hcomponent :
140       (DirectSum.decompose (orderDecomposition (n := n) k) (P ^ m) 0 :
141         SymbolRing k n) ∈ orderInitialIdeal k I :=
142     coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k I 0
143       (DirectSum.decompose (orderDecomposition (n := n) k) (P ^ m) 0)
144       (decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
145         k I (P ^ m) hm 0)
146   have hbaseLift : baseLift f ∈ reducedOrderSupportIdeal k I := by
147     rw [← zeroComponent_eq_baseLift_fibreZeroSpecialization]
148     exact orderInitialIdeal_le_reducedOrderSupportIdeal k I hcomponent
149   have hfy : MvPolynomial.eval y f = 0 :=
150     hy f ((mem_reducedOrderBaseIdeal_iff k I f).mpr hbaseLift)
151   have heval :
152       MvPolynomial.eval (zeroSectionPoint y) (P ^ m) = 0 := by
153     rw [← eval_fibreZeroSpecialization k y (P ^ m)]
154     exact hfy
155   rw [map_pow] at heval
156   exact (pow_eq_zero_iff'.mp heval).1

158 /-- Field-valued projection is exact: a base point annihilates the contracted
159 ideal if and only if some fibre point over it annihilates the full reduced
160 support ideal. The forward witness is canonically the zero fibre. -/
161 theorem mem_reducedOrderBaseZeroSet_iff_exists_supportFibre
162     (I : RightIdeal (WeylAlg k n)) (y : Fin n → k) :
163     (∀ f ∈ reducedOrderBaseIdeal k I, MvPolynomial.eval y f = 0) ↔
164       ∃ ξ : Fin n → k,
165         ∀ P ∈ reducedOrderSupportIdeal k I,
166           MvPolynomial.eval (Sum.elim y ξ) P = 0 := by
167   constructor
168   · intro hy
169     refine ⟨0, ?_⟩
170     have hpoint : Sum.elim y (0 : Fin n → k) = zeroSectionPoint y := by
171       funext i
172       rcases i with i | i <;> rfl
173     rw [hpoint]
174     exact zeroSection_mem_of_mem_reducedOrderBaseZeroSet k I y hy
175   · rintro ⟨ξ, hξ⟩ f hf
176     have hLift := hξ (baseLift f)
177       ((mem_reducedOrderBaseIdeal_iff k I f).mp hf)
178     rw [eval_baseLift k y ξ f] at hLift
179     exact hLift
