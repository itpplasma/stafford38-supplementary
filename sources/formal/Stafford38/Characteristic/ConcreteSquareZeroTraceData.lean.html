11 # Concrete square-zero trace data for a filtered Weyl quotient
12 
13 This file assembles the order-Rees two-jet ring and the filtered quotient
14 two-jet module into the abstract right-module interface used by the trace
15 argument.  The recorded bracket is the negative standard Poisson bracket,
16 as forced by the convention `[x,p] = -1`.
17 
18 No localization, finite-length trace theorem, or radical involutivity theorem
19 is asserted here.

44 /-- The sign-correct first-order bracket as a curried bilinear map. -/
45 def negativePoissonBracketLinear :
46     SymbolRing k n →ₗ[k] SymbolRing k n →ₗ[k] SymbolRing k n where
47   toFun f :=
48     { toFun := fun g => -poissonBracket f g
49       map_add' := by
50         intro g h
51         change -poissonBracket f (g + h) =
52           -poissonBracket f g + -poissonBracket f h
53         rw [poissonBracket_add_right, neg_add]
54       map_smul' := by
55         intro c g
56         change -poissonBracket f (c • g) =
57           c • (-poissonBracket f g)
58         rw [poissonBracket_smul_right, smul_neg] }
59   map_add' := by
60     intro f g
61     apply LinearMap.ext
62     intro h
63     change -poissonBracket (f + g) h =
64       -poissonBracket f h + -poissonBracket g h
65     rw [poissonBracket_add_left, neg_add]
66   map_smul' := by
67     intro c f
68     apply LinearMap.ext
69     intro g
70     change -poissonBracket (c • f) g =
71       c • (-poissonBracket f g)
72     rw [poissonBracket_smul_left, smul_neg]
73 
74 @[simp] theorem negativePoissonBracketLinear_apply
75     (f g : SymbolRing k n) :
76     negativePoissonBracketLinear (n := n) k f g =
77       -poissonBracket f g :=
78   rfl

80 /-- The two-jet action commutes with the original coefficient-field action. -/
81 def filteredQuotientTwoJetSMulCommClass
82     (I : RightIdeal (WeylAlg k n)) :
83     SMulCommClass k (OrderReesTwoJet (n := n) k)ᵐᵒᵖ
84       (FilteredQuotientTwoJet k I) where
85   smul_comm c b m := by
86     change c • quotientOrderReesTwoJetAction k I b m =
87       quotientOrderReesTwoJetAction k I b (c • m)
88     exact ((quotientOrderReesTwoJetAction k I b).map_smul c m).symm
