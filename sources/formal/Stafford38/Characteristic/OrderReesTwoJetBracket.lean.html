9 /-!
10 # The ringCommutator bracket in the order-Rees two-jet
11 
12 Every ringCommutator in the order-Rees ring loses one unit of differential order,
13 so it has an explicit factor of the Rees parameter.  This file constructs the
14 quotient coefficient by coefficient.  After specialization, that quotient is
15 the negative Poisson bracket, with the sign forced by `[x,p] = -1`.
16 
17 The quotient is deliberately attached to chosen Rees lifts: modulo `T²`, a
18 quotient by `T` is only determined modulo `T`.  The final theorem therefore
19 asserts existence for arbitrary two-jet classes while retaining the explicit
20 source-level witness used to prove it.
21 -/

45 /-- The quotient of a homogeneous Rees ringCommutator by one Rees parameter. -/
46 def homogeneousCommutatorQuotient
47     (r s : ℕ) (a : orderPiece k n r) (b : orderPiece k n s) :
48     OrderReesRing (n := n) k :=
49   if _h : 0 < r + s then
50     orderReesMonomial k (r + s - 1)
51       ⟨AlgebraicAnalysis.ringCommutator (a : WeylAlg k n) b,
52         commutator_mem_orderPiece_pred k a.property b.property⟩
53   else 0
54 
55 /-- Multiplying the homogeneous quotient by `T` recovers the written-order
56 ringCommutator of the two homogeneous Rees terms. -/
57 theorem parameter_mul_homogeneousCommutatorQuotient
58     (r s : ℕ) (a : orderPiece k n r) (b : orderPiece k n s) :
59     orderReesParameter (n := n) k *
60         homogeneousCommutatorQuotient k r s a b =
61       AlgebraicAnalysis.ringCommutator
62         (orderReesMonomial k r a) (orderReesMonomial k s b) := by
63   by_cases h : 0 < r + s
64   · rw [homogeneousCommutatorQuotient, dif_pos h,
65       orderReesParameter_mul_monomial]
66     apply Subtype.ext
67     change Polynomial.monomial ((r + s - 1) + 1)
68         (AlgebraicAnalysis.ringCommutator (a : WeylAlg k n) b) =
69       Polynomial.monomial r (a : WeylAlg k n) *
70           Polynomial.monomial s (b : WeylAlg k n) -
71         Polynomial.monomial s (b : WeylAlg k n) *
72           Polynomial.monomial r (a : WeylAlg k n)
73     simp only [Polynomial.monomial_mul_monomial, AlgebraicAnalysis.ringCommutator,
74       AlgebraicAnalysis.ringCommutator,
75       ← Polynomial.monomial_sub]
76     rw [show s + r = r + s by omega]
77     rw [Nat.sub_add_cancel (by omega : 1 ≤ r + s)]
78     exact Polynomial.monomial_sub (r + s)
79   · have hrs : r = 0 ∧ s = 0 := by omega
80     rcases hrs with ⟨rfl, rfl⟩
81     rw [homogeneousCommutatorQuotient, dif_neg (by omega), mul_zero]
82     have hab := commutator_eq_zero_of_mem_orderPiece_zero
83       k a.property b.property
84     apply Subtype.ext
85     change (0 : Polynomial (WeylAlg k n)) =
86       Polynomial.monomial 0 (a : WeylAlg k n) *
87           Polynomial.monomial 0 (b : WeylAlg k n) -
88         Polynomial.monomial 0 (b : WeylAlg k n) *
89           Polynomial.monomial 0 (a : WeylAlg k n)
90     simp only [Polynomial.monomial_mul_monomial, zero_add,
91       ← Polynomial.monomial_sub]
92     rw [show (a : WeylAlg k n) * b - b * a = 0 by
93       simpa [AlgebraicAnalysis.ringCommutator, AlgebraicAnalysis.ringCommutator] using hab]
94     simp

96 /-- The homogeneous quotient specializes to the sign-correct negative
97 Poisson bracket of the two homogeneous principal symbols. -/
98 theorem specialization_homogeneousCommutatorQuotient
99     (r s : ℕ) (a : orderPiece k n r) (b : orderPiece k n s) :
100     orderReesSpecialization (n := n) k
101         (homogeneousCommutatorQuotient k r s a b) =
102       -poissonBracket
103         (presentedPrincipalComponent k (@orderWeight n) r a)
104         (presentedPrincipalComponent k (@orderWeight n) s b) := by
105   by_cases h : 0 < r + s
106   · rw [homogeneousCommutatorQuotient, dif_pos h,
107       orderReesSpecialization_monomial]
108     exact principalComponent_commutator_eq_neg_poisson
109       k a.property b.property
110   · have hrs : r = 0 ∧ s = 0 := by omega
111     rcases hrs with ⟨rfl, rfl⟩
112     rw [homogeneousCommutatorQuotient, dif_neg (by omega), map_zero]
113     have hcomm := principalComponent_commutator_eq_neg_poisson
114       k a.property b.property
115     rw [commutator_eq_zero_of_mem_orderPiece_zero k a.property b.property,
116       map_zero] at hcomm
117     exact hcomm

119 /-- Explicit source-level quotient of an arbitrary Rees ringCommutator by `T`.
120 It is the finite double sum of the homogeneous subprincipal commutators. -/
121 def orderReesCommutatorQuotient
122     (a b : OrderReesRing (n := n) k) : OrderReesRing (n := n) k :=
123   ∑ r ∈ (a : Polynomial (WeylAlg k n)).support,
124     ∑ s ∈ (b : Polynomial (WeylAlg k n)).support,
125       homogeneousCommutatorQuotient k r s
126         ⟨(a : Polynomial (WeylAlg k n)).coeff r, a.property r⟩
127         ⟨(b : Polynomial (WeylAlg k n)).coeff s, b.property s⟩
128 
129 private theorem orderRees_eq_sum_monomials
130     (a : OrderReesRing (n := n) k) :
131     a = ∑ r ∈ (a : Polynomial (WeylAlg k n)).support,
132       orderReesMonomial k r
133         ⟨(a : Polynomial (WeylAlg k n)).coeff r, a.property r⟩ := by
134   apply Subtype.ext
135   ext N
136   simp [orderReesMonomial, Polynomial.coeff_monomial]

138 private theorem commutator_finset_sum
139     {R S : Type*} (U : Finset R) (V : Finset S)
140     (f : R → OrderReesRing (n := n) k)
141     (g : S → OrderReesRing (n := n) k) :
142     AlgebraicAnalysis.ringCommutator (∑ r ∈ U, f r) (∑ s ∈ V, g s) =
143       ∑ r ∈ U, ∑ s ∈ V, AlgebraicAnalysis.ringCommutator (f r) (g s) := by
144   simp only [AlgebraicAnalysis.ringCommutator, AlgebraicAnalysis.ringCommutator,
145     Finset.sum_mul, Finset.mul_sum,
146     Finset.sum_sub_distrib]
147   rw [Finset.sum_comm]

149 /-- The explicit arbitrary witness divides the source ringCommutator by `T`. -/
150 theorem parameter_mul_orderReesCommutatorQuotient
151     (a b : OrderReesRing (n := n) k) :
152     orderReesParameter (n := n) k * orderReesCommutatorQuotient k a b =
153       AlgebraicAnalysis.ringCommutator a b := by
154   rw [orderReesCommutatorQuotient, Finset.mul_sum]
155   simp_rw [Finset.mul_sum, parameter_mul_homogeneousCommutatorQuotient]
156   rw [← commutator_finset_sum]
157   rw [← orderRees_eq_sum_monomials k a,
158     ← orderRees_eq_sum_monomials k b]

160 private theorem poissonBracket_finset_sum_left {R : Type*}
161     (U : Finset R) (f : R → SymbolRing k n) (g : SymbolRing k n) :
162     poissonBracket (∑ r ∈ U, f r) g =
163       ∑ r ∈ U, poissonBracket (f r) g := by
164   classical
165   induction U using Finset.induction_on with
166   | empty => simp
167   | @insert r U hr ih =>
168       rw [Finset.sum_insert hr, Finset.sum_insert hr,
169         poissonBracket_add_left, ih]
170 
171 private theorem poissonBracket_finset_sum_right {S : Type*}
172     (V : Finset S) (f : SymbolRing k n) (g : S → SymbolRing k n) :
173     poissonBracket f (∑ s ∈ V, g s) =
174       ∑ s ∈ V, poissonBracket f (g s) := by
175   classical
176   induction V using Finset.induction_on with
177   | empty => simp
178   | @insert s V hs ih =>
179       rw [Finset.sum_insert hs, Finset.sum_insert hs,
180         poissonBracket_add_right, ih]

182 private theorem poissonBracket_finset_sum {R S : Type*}
183     (U : Finset R) (V : Finset S)
184     (f : R → SymbolRing k n) (g : S → SymbolRing k n) :
185     poissonBracket (∑ r ∈ U, f r) (∑ s ∈ V, g s) =
186       ∑ r ∈ U, ∑ s ∈ V, poissonBracket (f r) (g s) := by
187   rw [poissonBracket_finset_sum_left]
188   apply Finset.sum_congr rfl
189   intro r hr
190   exact poissonBracket_finset_sum_right k V (f r) g

200 /-- The explicit arbitrary quotient specializes to the negative Poisson
201 bracket of the two Rees specializations. -/
202 theorem specialization_orderReesCommutatorQuotient
203     (a b : OrderReesRing (n := n) k) :
204     orderReesSpecialization (n := n) k
205         (orderReesCommutatorQuotient k a b) =
206       -poissonBracket
207         (orderReesSpecialization (n := n) k a)
208         (orderReesSpecialization (n := n) k b) := by
209   rw [orderReesCommutatorQuotient, map_sum]
210   simp_rw [map_sum, specialization_homogeneousCommutatorQuotient]
211   change
212     (∑ r ∈ (a : Polynomial (WeylAlg k n)).support,
213       ∑ s ∈ (b : Polynomial (WeylAlg k n)).support,
214         -poissonBracket
215           (presentedPrincipalComponent k (@orderWeight n) r
216             ((a : Polynomial (WeylAlg k n)).coeff r))
217           (presentedPrincipalComponent k (@orderWeight n) s
218             ((b : Polynomial (WeylAlg k n)).coeff s))) = _
219   rw [orderReesSpecialization_eq_sum, orderReesSpecialization_eq_sum]
220   simp_rw [Finset.sum_neg_distrib]
221   rw [← neg_eq_iff_eq_neg]
222   simp only [neg_neg]
223   exact (poissonBracket_finset_sum k _ _ _ _).symm

225 /-- An explicit two-jet witness attached to chosen source lifts. -/
226 def orderReesTwoJetCommutatorWitness
227     (a b : OrderReesRing (n := n) k) : OrderReesTwoJet (n := n) k :=
228   orderReesTwoJetQuotient (n := n) k
229     (orderReesCommutatorQuotient k a b)

231 /-- Exact two-jet ringCommutator factorization for chosen Rees lifts. -/
232 theorem twoJet_commutator_eq_parameter_mul_witness
233     (a b : OrderReesRing (n := n) k) :
234     AlgebraicAnalysis.ringCommutator
235         (orderReesTwoJetQuotient (n := n) k a)
236         (orderReesTwoJetQuotient (n := n) k b) =
237       orderReesTwoJetParameter (n := n) k *
238         orderReesTwoJetCommutatorWitness k a b := by
239   change orderReesTwoJetQuotient (n := n) k
240       (AlgebraicAnalysis.ringCommutator a b) =
241     orderReesTwoJetQuotient (n := n) k
242       (orderReesParameter (n := n) k *
243         orderReesCommutatorQuotient k a b)
244   exact congrArg (orderReesTwoJetQuotient (n := n) k)
245     (parameter_mul_orderReesCommutatorQuotient k a b).symm

247 /-- The chosen two-jet witness retains the negative-Poisson specialization. -/
248 theorem twoJet_witness_specialization
249     (a b : OrderReesRing (n := n) k) :
250     orderReesTwoJetSpecialization (n := n) k
251         (orderReesTwoJetCommutatorWitness k a b) =
252       -poissonBracket
253         (orderReesTwoJetSpecialization (n := n) k
254           (orderReesTwoJetQuotient (n := n) k a))
255         (orderReesTwoJetSpecialization (n := n) k
256           (orderReesTwoJetQuotient (n := n) k b)) := by
257   simpa [orderReesTwoJetCommutatorWitness] using
258     specialization_orderReesCommutatorQuotient k a b

260 /-- Every arbitrary two-jet ringCommutator is a parameter multiple, and one may
261 choose a quotient whose specialization is exactly the negative Poisson bracket
262 of the specialized inputs. -/
263 theorem exists_twoJet_commutatorQuotient
264     (x y : OrderReesTwoJet (n := n) k) :
265     ∃ z : OrderReesTwoJet (n := n) k,
266       AlgebraicAnalysis.ringCommutator x y =
267           orderReesTwoJetParameter (n := n) k * z ∧
268         orderReesTwoJetSpecialization (n := n) k z =
269           -poissonBracket
270             (orderReesTwoJetSpecialization (n := n) k x)
271             (orderReesTwoJetSpecialization (n := n) k y) := by
272   obtain ⟨a, rfl⟩ := Quotient.mk''_surjective x
273   obtain ⟨b, rfl⟩ := Quotient.mk''_surjective y
274   exact ⟨orderReesTwoJetCommutatorWitness k a b,
275     twoJet_commutator_eq_parameter_mul_witness k a b,
276     twoJet_witness_specialization k a b⟩
