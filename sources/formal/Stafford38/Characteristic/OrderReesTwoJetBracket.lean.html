7 /-!
8 # The commutator bracket in the order-Rees two-jet
9 
10 Every commutator in the order-Rees ring loses one unit of differential order,
11 so it has an explicit factor of the Rees parameter.  This file constructs the
12 quotient coefficient by coefficient.  After specialization, that quotient is
13 the negative Poisson bracket, with the sign forced by `[x,p] = -1`.
14 
15 The quotient is deliberately attached to chosen Rees lifts: modulo `T²`, a
16 quotient by `T` is only determined modulo `T`.  The final theorem therefore
17 asserts existence for arbitrary two-jet classes while retaining the explicit
18 source-level witness used to prove it.
19 -/

43 /-- The quotient of a homogeneous Rees commutator by one Rees parameter. -/
44 def homogeneousCommutatorQuotient
45     (r s : ℕ) (a : orderPiece k n r) (b : orderPiece k n s) :
46     OrderReesRing (n := n) k :=
47   if _h : 0 < r + s then
48     orderReesMonomial k (r + s - 1)
49       ⟨Stafford.commutator (a : PresentedWeyl k n) b,
50         commutator_mem_orderPiece_pred k a.property b.property⟩
51   else 0
52 
53 /-- Multiplying the homogeneous quotient by `T` recovers the written-order
54 commutator of the two homogeneous Rees terms. -/
55 theorem parameter_mul_homogeneousCommutatorQuotient
56     (r s : ℕ) (a : orderPiece k n r) (b : orderPiece k n s) :
57     orderReesParameter (n := n) k *
58         homogeneousCommutatorQuotient k r s a b =
59       Stafford.commutator
60         (orderReesMonomial k r a) (orderReesMonomial k s b) := by
61   by_cases h : 0 < r + s
62   · rw [homogeneousCommutatorQuotient, dif_pos h,
63       orderReesParameter_mul_monomial]
64     apply Subtype.ext
65     change Polynomial.monomial ((r + s - 1) + 1)
66         (Stafford.commutator (a : PresentedWeyl k n) b) =
67       Polynomial.monomial r (a : PresentedWeyl k n) *
68           Polynomial.monomial s (b : PresentedWeyl k n) -
69         Polynomial.monomial s (b : PresentedWeyl k n) *
70           Polynomial.monomial r (a : PresentedWeyl k n)
71     simp only [Polynomial.monomial_mul_monomial, Stafford.commutator,
72       AlgebraicAnalysis.ringCommutator,
73       ← Polynomial.monomial_sub]
74     rw [show s + r = r + s by omega]
75     rw [Nat.sub_add_cancel (by omega : 1 ≤ r + s)]
76     exact Polynomial.monomial_sub (r + s)
77   · have hrs : r = 0 ∧ s = 0 := by omega
78     rcases hrs with ⟨rfl, rfl⟩
79     rw [homogeneousCommutatorQuotient, dif_neg (by omega), mul_zero]
80     have hab := commutator_eq_zero_of_mem_orderPiece_zero
81       k a.property b.property
82     apply Subtype.ext
83     change (0 : Polynomial (PresentedWeyl k n)) =
84       Polynomial.monomial 0 (a : PresentedWeyl k n) *
85           Polynomial.monomial 0 (b : PresentedWeyl k n) -
86         Polynomial.monomial 0 (b : PresentedWeyl k n) *
87           Polynomial.monomial 0 (a : PresentedWeyl k n)
88     simp only [Polynomial.monomial_mul_monomial, zero_add,
89       ← Polynomial.monomial_sub]
90     rw [show (a : PresentedWeyl k n) * b - b * a = 0 by
91       simpa [Stafford.commutator, AlgebraicAnalysis.ringCommutator] using hab]
92     simp

94 /-- The homogeneous quotient specializes to the sign-correct negative
95 Poisson bracket of the two homogeneous principal symbols. -/
96 theorem specialization_homogeneousCommutatorQuotient
97     (r s : ℕ) (a : orderPiece k n r) (b : orderPiece k n s) :
98     orderReesSpecialization (n := n) k
99         (homogeneousCommutatorQuotient k r s a b) =
100       -poissonBracket
101         (presentedPrincipalComponent k (@orderWeight n) r a)
102         (presentedPrincipalComponent k (@orderWeight n) s b) := by
103   by_cases h : 0 < r + s
104   · rw [homogeneousCommutatorQuotient, dif_pos h,
105       orderReesSpecialization_monomial]
106     exact principalComponent_commutator_eq_neg_poisson
107       k a.property b.property
108   · have hrs : r = 0 ∧ s = 0 := by omega
109     rcases hrs with ⟨rfl, rfl⟩
110     rw [homogeneousCommutatorQuotient, dif_neg (by omega), map_zero]
111     have hcomm := principalComponent_commutator_eq_neg_poisson
112       k a.property b.property
113     rw [commutator_eq_zero_of_mem_orderPiece_zero k a.property b.property,
114       map_zero] at hcomm
115     exact hcomm

117 /-- Explicit source-level quotient of an arbitrary Rees commutator by `T`.
118 It is the finite double sum of the homogeneous subprincipal commutators. -/
119 def orderReesCommutatorQuotient
120     (a b : OrderReesRing (n := n) k) : OrderReesRing (n := n) k :=
121   ∑ r ∈ (a : Polynomial (PresentedWeyl k n)).support,
122     ∑ s ∈ (b : Polynomial (PresentedWeyl k n)).support,
123       homogeneousCommutatorQuotient k r s
124         ⟨(a : Polynomial (PresentedWeyl k n)).coeff r, a.property r⟩
125         ⟨(b : Polynomial (PresentedWeyl k n)).coeff s, b.property s⟩

127 private theorem orderRees_eq_sum_monomials
128     (a : OrderReesRing (n := n) k) :
129     a = ∑ r ∈ (a : Polynomial (PresentedWeyl k n)).support,
130       orderReesMonomial k r
131         ⟨(a : Polynomial (PresentedWeyl k n)).coeff r, a.property r⟩ := by
132   apply Subtype.ext
133   ext N
134   simp [orderReesMonomial, Polynomial.coeff_monomial]

136 private theorem commutator_finset_sum
137     {R S : Type*} (U : Finset R) (V : Finset S)
138     (f : R → OrderReesRing (n := n) k)
139     (g : S → OrderReesRing (n := n) k) :
140     Stafford.commutator (∑ r ∈ U, f r) (∑ s ∈ V, g s) =
141       ∑ r ∈ U, ∑ s ∈ V, Stafford.commutator (f r) (g s) := by
142   simp only [Stafford.commutator, AlgebraicAnalysis.ringCommutator,
143     Finset.sum_mul, Finset.mul_sum,
144     Finset.sum_sub_distrib]
145   rw [Finset.sum_comm]

147 /-- The explicit arbitrary witness divides the source commutator by `T`. -/
148 theorem parameter_mul_orderReesCommutatorQuotient
149     (a b : OrderReesRing (n := n) k) :
150     orderReesParameter (n := n) k * orderReesCommutatorQuotient k a b =
151       Stafford.commutator a b := by
152   rw [orderReesCommutatorQuotient, Finset.mul_sum]
153   simp_rw [Finset.mul_sum, parameter_mul_homogeneousCommutatorQuotient]
154   rw [← commutator_finset_sum]
155   rw [← orderRees_eq_sum_monomials k a,
156     ← orderRees_eq_sum_monomials k b]

158 private theorem poissonBracket_finset_sum_left {R : Type*}
159     (U : Finset R) (f : R → SymbolRing k n) (g : SymbolRing k n) :
160     poissonBracket (∑ r ∈ U, f r) g =
161       ∑ r ∈ U, poissonBracket (f r) g := by
162   classical
163   induction U using Finset.induction_on with
164   | empty => simp
165   | @insert r U hr ih =>
166       rw [Finset.sum_insert hr, Finset.sum_insert hr,
167         poissonBracket_add_left, ih]
168 
169 private theorem poissonBracket_finset_sum_right {S : Type*}
170     (V : Finset S) (f : SymbolRing k n) (g : S → SymbolRing k n) :
171     poissonBracket f (∑ s ∈ V, g s) =
172       ∑ s ∈ V, poissonBracket f (g s) := by
173   classical
174   induction V using Finset.induction_on with
175   | empty => simp
176   | @insert s V hs ih =>
177       rw [Finset.sum_insert hs, Finset.sum_insert hs,
178         poissonBracket_add_right, ih]

180 private theorem poissonBracket_finset_sum {R S : Type*}
181     (U : Finset R) (V : Finset S)
182     (f : R → SymbolRing k n) (g : S → SymbolRing k n) :
183     poissonBracket (∑ r ∈ U, f r) (∑ s ∈ V, g s) =
184       ∑ r ∈ U, ∑ s ∈ V, poissonBracket (f r) (g s) := by
185   rw [poissonBracket_finset_sum_left]
186   apply Finset.sum_congr rfl
187   intro r hr
188   exact poissonBracket_finset_sum_right k V (f r) g

198 /-- The explicit arbitrary quotient specializes to the negative Poisson
199 bracket of the two Rees specializations. -/
200 theorem specialization_orderReesCommutatorQuotient
201     (a b : OrderReesRing (n := n) k) :
202     orderReesSpecialization (n := n) k
203         (orderReesCommutatorQuotient k a b) =
204       -poissonBracket
205         (orderReesSpecialization (n := n) k a)
206         (orderReesSpecialization (n := n) k b) := by
207   rw [orderReesCommutatorQuotient, map_sum]
208   simp_rw [map_sum, specialization_homogeneousCommutatorQuotient]
209   change
210     (∑ r ∈ (a : Polynomial (PresentedWeyl k n)).support,
211       ∑ s ∈ (b : Polynomial (PresentedWeyl k n)).support,
212         -poissonBracket
213           (presentedPrincipalComponent k (@orderWeight n) r
214             ((a : Polynomial (PresentedWeyl k n)).coeff r))
215           (presentedPrincipalComponent k (@orderWeight n) s
216             ((b : Polynomial (PresentedWeyl k n)).coeff s))) = _
217   rw [orderReesSpecialization_eq_sum, orderReesSpecialization_eq_sum]
218   simp_rw [Finset.sum_neg_distrib]
219   rw [← neg_eq_iff_eq_neg]
220   simp only [neg_neg]
221   exact (poissonBracket_finset_sum k _ _ _ _).symm

223 /-- An explicit two-jet witness attached to chosen source lifts. -/
224 def orderReesTwoJetCommutatorWitness
225     (a b : OrderReesRing (n := n) k) : OrderReesTwoJet (n := n) k :=
226   orderReesTwoJetQuotient (n := n) k
227     (orderReesCommutatorQuotient k a b)

229 /-- Exact two-jet commutator factorization for chosen Rees lifts. -/
230 theorem twoJet_commutator_eq_parameter_mul_witness
231     (a b : OrderReesRing (n := n) k) :
232     Stafford.commutator
233         (orderReesTwoJetQuotient (n := n) k a)
234         (orderReesTwoJetQuotient (n := n) k b) =
235       orderReesTwoJetParameter (n := n) k *
236         orderReesTwoJetCommutatorWitness k a b := by
237   change orderReesTwoJetQuotient (n := n) k
238       (Stafford.commutator a b) =
239     orderReesTwoJetQuotient (n := n) k
240       (orderReesParameter (n := n) k *
241         orderReesCommutatorQuotient k a b)
242   exact congrArg (orderReesTwoJetQuotient (n := n) k)
243     (parameter_mul_orderReesCommutatorQuotient k a b).symm

245 /-- The chosen two-jet witness retains the negative-Poisson specialization. -/
246 theorem twoJet_witness_specialization
247     (a b : OrderReesRing (n := n) k) :
248     orderReesTwoJetSpecialization (n := n) k
249         (orderReesTwoJetCommutatorWitness k a b) =
250       -poissonBracket
251         (orderReesTwoJetSpecialization (n := n) k
252           (orderReesTwoJetQuotient (n := n) k a))
253         (orderReesTwoJetSpecialization (n := n) k
254           (orderReesTwoJetQuotient (n := n) k b)) := by
255   simpa [orderReesTwoJetCommutatorWitness] using
256     specialization_orderReesCommutatorQuotient k a b

258 /-- Every arbitrary two-jet commutator is a parameter multiple, and one may
259 choose a quotient whose specialization is exactly the negative Poisson bracket
260 of the specialized inputs. -/
261 theorem exists_twoJet_commutatorQuotient
262     (x y : OrderReesTwoJet (n := n) k) :
263     ∃ z : OrderReesTwoJet (n := n) k,
264       Stafford.commutator x y =
265           orderReesTwoJetParameter (n := n) k * z ∧
266         orderReesTwoJetSpecialization (n := n) k z =
267           -poissonBracket
268             (orderReesTwoJetSpecialization (n := n) k x)
269             (orderReesTwoJetSpecialization (n := n) k y) := by
270   obtain ⟨a, rfl⟩ := Quotient.mk''_surjective x
271   obtain ⟨b, rfl⟩ := Quotient.mk''_surjective y
272   exact ⟨orderReesTwoJetCommutatorWitness k a b,
273     twoJet_commutator_eq_parameter_mul_witness k a b,
274     twoJet_witness_specialization k a b⟩
