12 /-!
13 # A concrete filtered criterion for canonical monic saturation
14 
15 For every degree, ordinary injectivity of coordinate multiplication on the
16 canonical right quotient, together with coordinate preimages that preserve
17 strict lower-order pieces, implies the exact cancellation statement on every
18 actual associated-graded piece.  This is an ambient filtered/Ore criterion:
19 it does not mention the initial ideal or assume its saturation.
20 
21 For a Bernstein-monic operator of degree one, commutation with the selected
22 coordinate is exactly `-1`.  Hence the two literal canonical generators
23 `d` and `x * d` already generate the unit right ideal.  This proves coordinate
24 saturation (and therefore strict noncharacteristic cancellation) without
25 assuming a generic D-module inverse-image theorem.
26 
27 The argument deliberately stops at degree one.  For degree at least two, the
28 same ringCommutator has positive differential order; proving saturation then needs
29 an order-preserving reduction of those higher commutators.  Existing monic Ore
30 division and the positive Euler residue provide unrestricted coordinate
31 preimages, but do not provide the strict differential-order estimate isolated
32 below.
33 -/

58 /-- The concrete filtered estimate missing from unrestricted Euler
59 surjectivity: every class represented below order `m` has a coordinate
60 predecessor represented below the same order.  This is stated in the ambient
61 Weyl algebra, not in the associated graded or the initial ideal. -/
62 def StrictLowerCoordinatePreimages (n N : ℕ)
63     (d : WeylAlg k (n + 1)) : Prop :=
64   ∀ (m : ℕ) (l : WeylAlg k (n + 1)),
65     l ∈ presentedStrictLowerPiece k orderWeight m →
66     ∃ y : WeylAlg k (n + 1),
67       y ∈ presentedStrictLowerPiece k orderWeight m ∧
68       qmk (presentedCanonicalRightIdeal (k := k) n N d)
69           (y * presentedCoordinate k n) =
70         qmk (presentedCanonicalRightIdeal (k := k) n N d) l

72 /-- A genuinely general filtered reduction.  Ordinary cancellation on the
73 canonical quotient and strict-order coordinate division together imply the
74 load-bearing cancellation theorem on every filtration degree. -/
75 theorem coordinateCancellation_of_quotient_injective_of_strict_preimages
76     (n N : ℕ) (d : WeylAlg k (n + 1))
77     (hinjective : Function.Injective
78       (rightMul (presentedCanonicalRightIdeal (k := k) n N d) (presentedCoordinate k n)))
79     (hstrict : StrictLowerCoordinatePreimages k n N d) :
80     CoordinateCancellation k n N d := by
81   intro m z hz hzx
82   obtain ⟨i, hi, l, hl, hil⟩ := Submodule.mem_sup.mp hzx
83   obtain ⟨y, hy, hyx⟩ := hstrict m l hl
84   have hzxl : qmk (presentedCanonicalRightIdeal (k := k) n N d)
85         (z * presentedCoordinate k n) =
86       qmk (presentedCanonicalRightIdeal (k := k) n N d) l := by
87     apply (Submodule.Quotient.eq (presentedCanonicalRightIdeal (k := k) n N d)).2
88     rw [← hil]
89     simpa using hi
90   have hxy : rightMul (presentedCanonicalRightIdeal (k := k) n N d)
91         (presentedCoordinate k n) (qmk (presentedCanonicalRightIdeal (k := k) n N d) z) =
92       rightMul (presentedCanonicalRightIdeal (k := k) n N d)
93         (presentedCoordinate k n) (qmk (presentedCanonicalRightIdeal (k := k) n N d) y) := by
94     rw [← qmk_right_mul, ← qmk_right_mul, hzxl, hyx]
95   have hzy : qmk (presentedCanonicalRightIdeal (k := k) n N d) z =
96       qmk (presentedCanonicalRightIdeal (k := k) n N d) y := hinjective hxy
97   have hdiff : z - y ∈ presentedCanonicalRightIdeal (k := k) n N d :=
98     (Submodule.Quotient.eq (presentedCanonicalRightIdeal (k := k) n N d)).mp hzy
99   rw [show z = (z - y) + y by abel]
100   exact Submodule.add_mem _
101     (Submodule.mem_sup_left hdiff)
102     (Submodule.mem_sup_right hy)

160     {n : ℕ} {P : SymbolRing k n} {m : PhaseVar n →₀ ℕ}
161     (t : Fin n)

181 /-- The coefficient of every monomial involving the selected momentum variable
182 vanishes after removing its linear coefficient from a fibre-only order-one
183 symbol.  Isolating this coefficient calculation keeps the support argument
184 below independent of polynomial representation reductions. -/
185 private theorem coeff_sub_X_eq_zero_of_order_one_fibreOnly
186     {n : ℕ} (t : Fin n) (P : SymbolRing k n)
187     (hhom : MvPolynomial.IsWeightedHomogeneous (@orderWeight n) P 1)
188     (hfibre : IsFibreOnly k P)
189     (hcoeff : P.coeff (Finsupp.single (.inr t) 1) = 1)
190     (m : PhaseVar n →₀ ℕ) (hmt : m (.inr t) ≠ 0) :
191     MvPolynomial.coeff m (P - MvPolynomial.X (.inr t)) = 0 := by
192   classical
193   by_cases hm : m = Finsupp.single (.inr t) 1
194   · subst m
195     have hcoeff' : AddMonoidAlgebra.coeff P
196         (Finsupp.single (.inr t) 1) = 1 := by
197       change MvPolynomial.coeff (Finsupp.single (.inr t) 1) P = 1
198       exact hcoeff
199     have hX' : AddMonoidAlgebra.coeff
200         (MvPolynomial.X (.inr t : PhaseVar n) : SymbolRing k n)
201         (Finsupp.single (.inr t) 1) = 1 := by
202       simp [MvPolynomial.X]
203     change AddMonoidAlgebra.coeff
204       (P - MvPolynomial.X (.inr t : PhaseVar n))
205       (Finsupp.single (.inr t) 1) = 0
206     rw [AddMonoidAlgebra.coeff_sub]
207     change AddMonoidAlgebra.coeff P (Finsupp.single (.inr t) 1) -
208       AddMonoidAlgebra.coeff (MvPolynomial.X (.inr t) : SymbolRing k n)
209         (Finsupp.single (.inr t) 1) = 0
210     rw [hcoeff', hX', sub_self]
211   · have hPzero : MvPolynomial.coeff m P = 0 := by
212       by_contra hP
213       exact hm (exponent_eq_selected_of_order_one_of_selected_ne_zero
214         k t hP hhom hfibre hmt)
215     have hXzero : AddMonoidAlgebra.coeff
216         (MvPolynomial.X (.inr t : PhaseVar n) : SymbolRing k n) m = 0 := by
217       simp [MvPolynomial.X, Ne.symm hm]
218     have hPzero' : AddMonoidAlgebra.coeff P m = 0 := by
219       change MvPolynomial.coeff m P = 0
220       exact hPzero
221     have hXzero' : AddMonoidAlgebra.coeff
222         (MvPolynomial.X (.inr t : PhaseVar n) : SymbolRing k n) m = 0 := by
223       change MvPolynomial.coeff m (MvPolynomial.X (.inr t)) = 0
224       exact hXzero
225     change AddMonoidAlgebra.coeff
226       (P - MvPolynomial.X (.inr t : PhaseVar n)) m = 0
227     rw [AddMonoidAlgebra.coeff_sub]
228     change AddMonoidAlgebra.coeff P m -
229       AddMonoidAlgebra.coeff (MvPolynomial.X (.inr t) : SymbolRing k n) m = 0
230     rw [hPzero', hXzero', sub_self]

261 /-- In canonical Bernstein degree one, the selected coordinate ringCommutator is
262 the scalar `-1`.  This is the strict low-order replacement for the missing
263 higher-order division estimate. -/

275         orderPiece k (n + 1) 0 := by
276     simpa using commutator_mem_orderPiece_pred k hxOrder hdOrder

285   have hpderiv :
286       MvPolynomial.pderiv (.inr (0 : Fin (n + 1))) P = 1 :=

297       (Stafford38.CharacteristicAssociatedGradedModule.orderPieceOne
298         (n := n + 1) k).property
299   have hdiffOrder :
300       AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) d - (-1) ∈
301         orderPiece k (n + 1) 0 :=
302     (orderPiece k (n + 1) 0).sub_mem hcommOrder hnegOneOrder
303   have hdiffPrincipal :
304       presentedPrincipalComponent k (@orderWeight (n + 1)) 0

313         (n := n + 1) k
314     rw [honepc]
315     simp
316   have hlower :=
317     (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
318       k (@orderWeight (n + 1)) _ hdiffOrder).mp hdiffPrincipal
319   have hdiff : AlgebraicAnalysis.ringCommutator (presentedCoordinate k n) d - (-1) = 0 := by
320     simpa [presentedStrictLowerPiece] using hlower
