10 /-!
11 # A concrete filtered criterion for canonical monic saturation
12 
13 For every degree, ordinary injectivity of coordinate multiplication on the
14 canonical right quotient, together with coordinate preimages that preserve
15 strict lower-order pieces, implies the exact cancellation statement on every
16 actual associated-graded piece.  This is an ambient filtered/Ore criterion:
17 it does not mention the initial ideal or assume its saturation.
18 
19 For a Bernstein-monic operator of degree one, commutation with the selected
20 coordinate is exactly `-1`.  Hence the two literal canonical generators
21 `d` and `x * d` already generate the unit right ideal.  This proves coordinate
22 saturation (and therefore strict noncharacteristic cancellation) without
23 assuming a generic D-module inverse-image theorem.
24 
25 The argument deliberately stops at degree one.  For degree at least two, the
26 same commutator has positive differential order; proving saturation then needs
27 an order-preserving reduction of those higher commutators.  Existing monic Ore
28 division and the positive Euler residue provide unrestricted coordinate
29 preimages, but do not provide the strict differential-order estimate isolated
30 below.
31 -/

56 /-- The concrete filtered estimate missing from unrestricted Euler
57 surjectivity: every class represented below order `m` has a coordinate
58 predecessor represented below the same order.  This is stated in the ambient
59 Weyl algebra, not in the associated graded or the initial ideal. -/
60 def StrictLowerCoordinatePreimages (n N : ℕ)
61     (d : PresentedWeyl k (n + 1)) : Prop :=
62   ∀ (m : ℕ) (l : PresentedWeyl k (n + 1)),
63     l ∈ presentedStrictLowerPiece k orderWeight m →
64     ∃ y : PresentedWeyl k (n + 1),
65       y ∈ presentedStrictLowerPiece k orderWeight m ∧
66       qmk (presentedCanonicalRightIdeal (k := k) n N d)
67           (y * presentedCoordinate k n) =
68         qmk (presentedCanonicalRightIdeal (k := k) n N d) l

70 /-- A genuinely general filtered reduction.  Ordinary cancellation on the
71 canonical quotient and strict-order coordinate division together imply the
72 load-bearing cancellation theorem on every filtration degree. -/
73 theorem coordinateCancellation_of_quotient_injective_of_strict_preimages
74     (n N : ℕ) (d : PresentedWeyl k (n + 1))
75     (hinjective : Function.Injective
76       (rightMul (presentedCanonicalRightIdeal (k := k) n N d) (presentedCoordinate k n)))
77     (hstrict : StrictLowerCoordinatePreimages k n N d) :
78     CoordinateCancellation k n N d := by
79   intro m z hz hzx
80   obtain ⟨i, hi, l, hl, hil⟩ := Submodule.mem_sup.mp hzx
81   obtain ⟨y, hy, hyx⟩ := hstrict m l hl
82   have hzxl : qmk (presentedCanonicalRightIdeal (k := k) n N d)
83         (z * presentedCoordinate k n) =
84       qmk (presentedCanonicalRightIdeal (k := k) n N d) l := by
85     apply (Submodule.Quotient.eq (presentedCanonicalRightIdeal (k := k) n N d)).2
86     rw [← hil]
87     simpa using hi
88   have hxy : rightMul (presentedCanonicalRightIdeal (k := k) n N d)
89         (presentedCoordinate k n) (qmk (presentedCanonicalRightIdeal (k := k) n N d) z) =
90       rightMul (presentedCanonicalRightIdeal (k := k) n N d)
91         (presentedCoordinate k n) (qmk (presentedCanonicalRightIdeal (k := k) n N d) y) := by
92     rw [← qmk_right_mul, ← qmk_right_mul, hzxl, hyx]
93   have hzy : qmk (presentedCanonicalRightIdeal (k := k) n N d) z =
94       qmk (presentedCanonicalRightIdeal (k := k) n N d) y := hinjective hxy
95   have hdiff : z - y ∈ presentedCanonicalRightIdeal (k := k) n N d :=
96     (Submodule.Quotient.eq (presentedCanonicalRightIdeal (k := k) n N d)).mp hzy
97   rw [show z = (z - y) + y by abel]
98   exact Submodule.add_mem _
99     (Submodule.mem_sup_left hdiff)
100     (Submodule.mem_sup_right hy)

179 /-- The coefficient of every monomial involving the selected momentum variable
180 vanishes after removing its linear coefficient from a fibre-only order-one
181 symbol.  Isolating this coefficient calculation keeps the support argument
182 below independent of polynomial representation reductions. -/
183 private theorem coeff_sub_X_eq_zero_of_order_one_fibreOnly
184     {n : ℕ} (t : Fin n) (P : SymbolRing k n)
185     (hhom : MvPolynomial.IsWeightedHomogeneous (@orderWeight n) P 1)
186     (hfibre : IsFibreOnly k P)
187     (hcoeff : P.coeff (Finsupp.single (.inr t) 1) = 1)
188     (m : PhaseVar n →₀ ℕ) (hmt : m (.inr t) ≠ 0) :
189     MvPolynomial.coeff m (P - MvPolynomial.X (.inr t)) = 0 := by
190   classical
191   by_cases hm : m = Finsupp.single (.inr t) 1
192   · subst m
193     have hcoeff' : AddMonoidAlgebra.coeff P
194         (Finsupp.single (.inr t) 1) = 1 := by
195       change MvPolynomial.coeff (Finsupp.single (.inr t) 1) P = 1
196       exact hcoeff
197     have hX' : AddMonoidAlgebra.coeff
198         (MvPolynomial.X (.inr t : PhaseVar n) : SymbolRing k n)
199         (Finsupp.single (.inr t) 1) = 1 := by
200       simp [MvPolynomial.X]
201     change AddMonoidAlgebra.coeff
202       (P - MvPolynomial.X (.inr t : PhaseVar n))
203       (Finsupp.single (.inr t) 1) = 0
204     rw [AddMonoidAlgebra.coeff_sub]
205     change AddMonoidAlgebra.coeff P (Finsupp.single (.inr t) 1) -
206       AddMonoidAlgebra.coeff (MvPolynomial.X (.inr t) : SymbolRing k n)
207         (Finsupp.single (.inr t) 1) = 0
208     rw [hcoeff', hX', sub_self]
209   · have hPzero : MvPolynomial.coeff m P = 0 := by
210       by_contra hP
211       exact hm (exponent_eq_selected_of_order_one_of_selected_ne_zero
212         k t hP hhom hfibre hmt)
213     have hXzero : AddMonoidAlgebra.coeff
214         (MvPolynomial.X (.inr t : PhaseVar n) : SymbolRing k n) m = 0 := by
215       simp [MvPolynomial.X, Ne.symm hm]
216     have hPzero' : AddMonoidAlgebra.coeff P m = 0 := by
217       change MvPolynomial.coeff m P = 0
218       exact hPzero
219     have hXzero' : AddMonoidAlgebra.coeff
220         (MvPolynomial.X (.inr t : PhaseVar n) : SymbolRing k n) m = 0 := by
221       change MvPolynomial.coeff m (MvPolynomial.X (.inr t)) = 0
222       exact hXzero
223     change AddMonoidAlgebra.coeff
224       (P - MvPolynomial.X (.inr t : PhaseVar n)) m = 0
225     rw [AddMonoidAlgebra.coeff_sub]
226     change AddMonoidAlgebra.coeff P m -
227       AddMonoidAlgebra.coeff (MvPolynomial.X (.inr t) : SymbolRing k n) m = 0
228     rw [hPzero', hXzero', sub_self]

259 /-- In canonical Bernstein degree one, the selected coordinate commutator is
260 the scalar `-1`.  This is the strict low-order replacement for the missing
261 higher-order division estimate. -/
262 theorem coordinate_commutator_eq_neg_one_degree_one
263     (n : ℕ) {d : PresentedWeyl k (n + 1)}

270     presentedCoordinate_mem_orderPiece_zero k n

272       Stafford.commutator (presentedCoordinate k n) d ∈
273         orderPiece k (n + 1) 0 := by
274     simpa using commutator_mem_orderPiece_pred k hxOrder hdOrder
275   have hhom : MvPolynomial.IsWeightedHomogeneous (@orderWeight (n + 1)) P 1 := by
276     exact MvPolynomial.weightedHomogeneousComponent_mem _ _ _

278     canonical_orderPrincipalComponent_isFibreOnly k n 1 hd
279   have hcoeff :
280       MvPolynomial.coeff
281           (Finsupp.single (.inr (0 : Fin (n + 1))) 1) P = 1 :=
282     canonical_orderPrincipalComponent_pureMomentumCoefficient k n 1 hd
283   have hpderiv :
284       MvPolynomial.pderiv (.inr (0 : Fin (n + 1))) P = 1 :=
285     pderiv_eq_one_of_order_one_fibreOnly k 0 P hhom hfibre hcoeff
286   have hprincipal :
287       presentedPrincipalComponent k (@orderWeight (n + 1)) 0
288           (Stafford.commutator (presentedCoordinate k n) d) = -1 := by
289     rw [principalComponent_coordinate_commutator_arbitrary]

295       (Stafford38.CharacteristicAssociatedGradedModule.orderPieceOne
296         (n := n + 1) k).property
297   have hdiffOrder :
298       Stafford.commutator (presentedCoordinate k n) d - (-1) ∈
299         orderPiece k (n + 1) 0 :=
300     (orderPiece k (n + 1) 0).sub_mem hcommOrder hnegOneOrder
301   have hdiffPrincipal :
302       presentedPrincipalComponent k (@orderWeight (n + 1)) 0
303           (Stafford.commutator (presentedCoordinate k n) d - (-1)) = 0 := by

310       exact Stafford38.CharacteristicAssociatedGradedModule.presentedPrincipalComponent_orderPieceOne
311         (n := n + 1) k
312     rw [honepc]
313     simp
314   have hlower :=
315     (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
316       k (@orderWeight (n + 1)) _ hdiffOrder).mp hdiffPrincipal
317   have hdiff : Stafford.commutator (presentedCoordinate k n) d - (-1) = 0 := by
318     simpa [presentedStrictLowerPiece] using hlower
319   exact sub_eq_zero.mp hdiff
