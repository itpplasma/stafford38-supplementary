14 private theorem derivation_unit_inv_apply
15     {k B : Type*} [CommRing k] [CommRing B] [Algebra k B]
16     (D : Derivation k B B) (u : Bˣ) :
17     D (↑(u⁻¹) : B) = -(↑(u⁻¹) : B) * ↑(u⁻¹) * D (u : B) := by
18   have hsum : (u : B) * D (↑(u⁻¹) : B) +
19       (↑(u⁻¹) : B) * D (u : B) = 0 := by
20     have hzero : D ((u : B) * (↑(u⁻¹) : B)) = 0 := by simp
21     have h := D.leibniz (u : B) (↑(u⁻¹) : B)
22     rw [hzero] at h
23     simpa [smul_eq_mul] using h.symm
24   have hmul : (u : B) * D (↑(u⁻¹) : B) =
25       -((↑(u⁻¹) : B) * D (u : B)) := by
26     exact eq_neg_of_add_eq_zero_left hsum
27   have hu : IsUnit (u : B) := ⟨u, rfl⟩
28   apply hu.mul_right_inj.mp
29   calc
30     (u : B) * D (↑(u⁻¹) : B) = -((↑(u⁻¹) : B) * D (u : B)) := hmul
31     _ = (u : B) * (-(↑(u⁻¹) : B) * ↑(u⁻¹) * D (u : B)) := by
32       calc
33         -((↑(u⁻¹) : B) * D (u : B)) =
34             -((1 : B) * ↑(u⁻¹) * D (u : B)) := by simp
35         _ = -(((u : B) * (↑(u⁻¹) : B)) * ↑(u⁻¹) * D (u : B)) := by
36               have heq := congrArg
37                 (fun z : B => -(z * (↑(u⁻¹) : B) * D (u : B))) u.val_inv
38               simpa using heq.symm
39         _ = (u : B) * (-(↑(u⁻¹) : B) * ↑(u⁻¹) * D (u : B)) := by ring

41 theorem extendDerivation_apply_mk'_cross
42     {k A B : Type u} [CommRing k] [CommRing A] [CommRing B]
43     [Algebra k A] [Algebra k B] [Algebra A B] [IsScalarTower k A B]
44     (S : Submonoid A) [IsLocalization S B]
45     (D : Derivation k A B) (a : A) (s : S) :
46     (extendDerivation k A B S D) (IsLocalization.mk' B a s) *
47         (algebraMap A B (s : A)) ^ 2 =
48       algebraMap A B (s : A) * D a - algebraMap A B a * D (s : A) := by
49   let E := extendDerivation k A B S D
50   let u : B := algebraMap A B (s : A)
51   let uunit : Bˣ := (IsLocalization.map_units B s).unit
52   let ui : B := ↑(uunit⁻¹)
53   let v : B := algebraMap A B a
54   have hu : IsUnit u := IsLocalization.map_units B s
55   have hui : u * ui = 1 := by
56     change u * (↑(uunit⁻¹) : B) = 1
57     rw [← hu.unit_spec]
58     exact hu.unit.val_inv
59   have hcomp := extendDerivation_compAlgebraMap k A B S D
60   have hdu : E u = D (s : A) := by
61     have h := Derivation.congr_fun hcomp (s : A)
62     change E (algebraMap A B (s : A)) = D (s : A) at h
63     change E u = D (s : A)
64     exact h
65   have hdv : E v = D a := by
66     have h := Derivation.congr_fun hcomp a
67     change E (algebraMap A B a) = D a at h
68     change E v = D a
69     exact h
70   have hrepr : IsLocalization.mk' B a s = v * ui := by
71     apply hu.mul_right_inj.mp
72     calc
73       u * IsLocalization.mk' B a s = v := by
74         calc
75           u * IsLocalization.mk' B a s = IsLocalization.mk' B a s * u := by ac_rfl
76           _ = v := IsLocalization.mk'_spec B a s
77       _ = u * (v * ui) := by
78         calc
79           v = v * 1 := by rw [mul_one]
80           _ = v * (u * ui) := by rw [hui]
81           _ = u * (v * ui) := by ac_rfl
82   have hcancel : u ^ 2 * ui = u := by
83     calc
84       u ^ 2 * ui = u * (u * ui) := by ring
85       _ = u := by rw [hui, mul_one]
86   have hinv2 : u ^ 2 * ui ^ 2 = 1 := by
87     rw [← mul_pow, hui, one_pow]
88   calc
89     E (IsLocalization.mk' B a s) * u ^ 2 =
90         u ^ 2 * (v * E ui + ui * E v) := by
91           rw [hrepr, Derivation.leibniz]
92           simp only [Algebra.smul_def, Algebra.algebraMap_self, RingHom.id_apply]
93           ring
94     _ = u * D a - v * D (s : A) := by
95           have hinv : E ui = -(ui * ui * E u) := by
96             simpa [ui, uunit, u, hu.unit_spec] using
97               (derivation_unit_inv_apply E uunit)
98           rw [hinv, hdv, hdu]
99           calc
100             u ^ 2 * (v * (-(ui * ui * D (s : A))) + ui * D a) =
101                 -(v * D (s : A) * (u ^ 2 * ui ^ 2)) +
102                   (u ^ 2 * ui) * D a := by ring
103             _ = u * D a - v * D (s : A) := by rw [hinv2, hcancel]; ring
104 
105 end
