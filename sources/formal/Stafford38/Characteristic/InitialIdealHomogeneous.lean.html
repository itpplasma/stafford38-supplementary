8 /-!
9 # Homogeneous pieces of the order initial ideal
10 
11 The generators of `orderInitialIdeal` are weighted homogeneous and are exact
12 symbol relations.  This file proves that taking the degree-`N` homogeneous
13 component of any element of their ideal span again gives an exact degree-`N`
14 relation.  The proof uses the finite direct-sum decomposition of each
15 polynomial multiplier and the homogeneous relation closure proved in
16 `FilteredQuotientGraded`.
17 -/

48 /-- Only finitely many order-homogeneous components of a symbol are nonzero. -/
49 theorem orderHomogeneousComponent_support_finite (P : SymbolRing k n) :
50     (Function.support fun N =>
51       MvPolynomial.weightedHomogeneousComponent (@orderWeight n) N P).Finite :=
52   MvPolynomial.weightedHomogeneousComponent_finsupp P

54 /-- Reconstruction from the finite family of order-homogeneous components. -/

56     (∑ᶠ N, MvPolynomial.weightedHomogeneousComponent
57       (@orderWeight n) N P) = P :=

60 /-- Multiplying a symbol whose every homogeneous component is an exact
61 relation preserves that property. -/
62 private theorem decompose_mul_mem_orderSymbolRelation
63     (I : RightIdeal (WeylAlg k n)) (a x : SymbolRing k n)
64     (hx : ∀ M, DirectSum.decompose (orderDecomposition (n := n) k) x M ∈
65       orderSymbolRelation k I M) (N : ℕ) :
66     DirectSum.decompose (orderDecomposition (n := n) k) (a * x) N ∈
67       orderSymbolRelation k I N := by
68   classical
69   letI : ∀ (M : ℕ) (Q : OrderHomogeneous (n := n) k M),
70       Decidable (Q ≠ 0) :=
71     fun _ Q => Classical.propDecidable (Q ≠ 0)
72   have hxsum :
73       (∑ M ∈ (DirectSum.decompose
74           (orderDecomposition (n := n) k) x).support,
75         (DirectSum.decompose (orderDecomposition (n := n) k) x M :
76           SymbolRing k n)) = x :=
77     DirectSum.sum_support_decompose (orderDecomposition (n := n) k) x
78   rw [← hxsum, Finset.mul_sum,
79     DirectSum.decompose_sum]
80   rw [DFinsupp.finsetSum_apply]
81   apply Submodule.sum_mem
82   intro M hM
83   by_cases hMN : M ≤ N
84   · have hmul := homogeneousRightMul_mem_orderSymbolRelation k I
85       (DirectSum.decompose (orderDecomposition k) a (N - M))
86       (DirectSum.decompose (orderDecomposition (n := n) k) x M) (hx M)
87     have hsum : M + (N - M) = N := Nat.add_sub_of_le hMN
88     rw [← hsum]
89     have hMN' : M ≤ M + (N - M) := Nat.le_add_right M (N - M)
90     have heq :
91         DirectSum.decompose (orderDecomposition (n := n) k)
92             (a * (DirectSum.decompose (orderDecomposition (n := n) k) x M :
93               SymbolRing k n)) (M + (N - M)) =
94           homogeneousRightMul k
95             (DirectSum.decompose (orderDecomposition (n := n) k) a (N - M))
96             (DirectSum.decompose (orderDecomposition (n := n) k) x M) := by
97       apply Subtype.ext
98       rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
99         (orderDecomposition k)
100         (DirectSum.decompose (orderDecomposition (n := n) k) x M).property hMN']
101       simp only [coe_homogeneousRightMul]
102       rw [Nat.add_sub_cancel_left]
103       exact mul_comm
104         (DirectSum.decompose (orderDecomposition (n := n) k) a (N - M) :
105           SymbolRing k n)
106         (DirectSum.decompose (orderDecomposition (n := n) k) x M :
107           SymbolRing k n)
108     rw [heq]
109     exact hmul
110   · have heq :
111         DirectSum.decompose (orderDecomposition (n := n) k)
112             (a * (DirectSum.decompose (orderDecomposition (n := n) k) x M :
113               SymbolRing k n)) N = 0 := by
114       apply Subtype.ext
115       exact DirectSum.coe_decompose_mul_of_right_mem_of_not_le
116         (orderDecomposition k)
117         (DirectSum.decompose (orderDecomposition (n := n) k) x M).property hMN
118     rw [heq]
119     exact Submodule.zero_mem _

121 /-- Every homogeneous component of the ideal generated by order-principal
122 symbols is an exact symbol relation in that degree. -/
123 private theorem all_decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
124     (I : RightIdeal (WeylAlg k n)) (P : SymbolRing k n)
125     (hP : P ∈ orderInitialIdeal k I) :
126     ∀ N, DirectSum.decompose (orderDecomposition (n := n) k) P N ∈
127       orderSymbolRelation k I N := by
128   classical
129   change P ∈ Ideal.span (orderInitialGeneratorSet k I) at hP
130   induction hP using Submodule.span_induction with
131   | mem P hP =>
132       intro N
133       obtain ⟨M, z, hzM, hzI, rfl⟩ := hP
134       let Q : OrderHomogeneous k M :=
135         principalComponentOnPiece k (@orderWeight n) M ⟨z, hzM⟩
136       have hQ : Q ∈ orderSymbolRelation k I M :=
137         principalComponent_mem_orderSymbolRelation k I M z hzM hzI
138       by_cases hNM : N = M
139       · subst N
140         change DirectSum.decompose (orderDecomposition (n := n) k)
141           (Q : SymbolRing k n) M ∈ orderSymbolRelation k I M
142         simpa only [DirectSum.decompose_coe, DirectSum.of_eq_same] using hQ
143       · have hzero :
144             (DirectSum.decompose (orderDecomposition k)
145               (presentedPrincipalComponent k orderWeight M z) N :
146                 SymbolRing k n) = 0 := by
147           change DirectSum.decompose (orderDecomposition k)
148             (Q : SymbolRing k n) N = (0 : SymbolRing k n)
149           exact DirectSum.decompose_of_mem_ne (orderDecomposition k) Q.property
150             (fun h => hNM h.symm)
151         have hzero' : DirectSum.decompose (orderDecomposition k)
152             (presentedPrincipalComponent k orderWeight M z) N = 0 :=
153           Subtype.ext hzero
154         rw [hzero']
155         exact Submodule.zero_mem _
156   | zero =>
157       intro N
158       rw [DirectSum.decompose_zero]
159       exact Submodule.zero_mem _
160   | add P Q _ _ hP hQ =>
161       intro N
162       simpa only [DirectSum.decompose_add, DirectSum.add_apply] using
163         (Submodule.add_mem (orderSymbolRelation k I N) (hP N) (hQ N))
164   | smul a P _ hP =>
165       intro N
166       change DirectSum.decompose (orderDecomposition k) (a * P) N ∈
167         orderSymbolRelation k I N
168       exact decompose_mul_mem_orderSymbolRelation k I a P hP N
169 
170 /-- Every homogeneous component of the ideal generated by order-principal
171 symbols is an exact symbol relation in that degree. -/
172 theorem decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
173     (I : RightIdeal (WeylAlg k n)) (P : SymbolRing k n)
174     (hP : P ∈ orderInitialIdeal k I) (N : ℕ) :
175     DirectSum.decompose (orderDecomposition (n := n) k) P N ∈
176       orderSymbolRelation k I N :=
177   all_decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal k I P hP N
178 
179 /-- Reverse homogeneous-piece inclusion: a degree-`N` homogeneous element of
180 the order initial ideal is an exact relation in degree `N`. -/
181 theorem mem_orderSymbolRelation_of_coe_mem_orderInitialIdeal
182     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
183     (P : OrderHomogeneous k N)
184     (hP : (P : SymbolRing k n) ∈ orderInitialIdeal k I) :
185     P ∈ orderSymbolRelation k I N := by
186   have hcomponent :=
187     decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal k I P hP N
188   simpa [DirectSum.decompose_coe] using hcomponent
189 
190 /-- Exact homogeneous relations are precisely the homogeneous pieces of the
191 order initial ideal. -/
192 theorem mem_orderSymbolRelation_iff_coe_mem_orderInitialIdeal
193     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
194     (P : OrderHomogeneous k N) :
195     P ∈ orderSymbolRelation k I N ↔
196       (P : SymbolRing k n) ∈ orderInitialIdeal k I :=
197   ⟨coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k I N P,
198     mem_orderSymbolRelation_of_coe_mem_orderInitialIdeal k I N P⟩
