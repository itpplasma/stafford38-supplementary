7 # Homogeneous pieces of the order initial ideal
8 
9 The generators of `orderInitialIdeal` are weighted homogeneous and are exact
10 symbol relations.  This file proves that taking the degree-`N` homogeneous
11 component of any element of their ideal span again gives an exact degree-`N`
12 relation.  The proof uses the finite direct-sum decomposition of each
13 polynomial multiplier and the homogeneous relation closure proved in
14 `FilteredQuotientGraded`.
15 -/

58 /-- Multiplying a symbol whose every homogeneous component is an exact
59 relation preserves that property. -/
60 private theorem decompose_mul_mem_orderSymbolRelation
61     (I : RightIdeal (PresentedWeyl k n)) (a x : SymbolRing k n)
62     (hx : ∀ M, DirectSum.decompose (orderDecomposition (n := n) k) x M ∈
63       orderSymbolRelation k I M) (N : ℕ) :
64     DirectSum.decompose (orderDecomposition (n := n) k) (a * x) N ∈
65       orderSymbolRelation k I N := by
66   classical
67   letI : ∀ (M : ℕ) (Q : OrderHomogeneous (n := n) k M),
68       Decidable (Q ≠ 0) :=
69     fun _ Q => Classical.propDecidable (Q ≠ 0)
70   have hxsum :
71       (∑ M ∈ (DirectSum.decompose
72           (orderDecomposition (n := n) k) x).support,
73         (DirectSum.decompose (orderDecomposition (n := n) k) x M :
74           SymbolRing k n)) = x :=
75     DirectSum.sum_support_decompose (orderDecomposition (n := n) k) x
76   rw [← hxsum, Finset.mul_sum,
77     DirectSum.decompose_sum]
78   rw [DFinsupp.finsetSum_apply]
79   apply Submodule.sum_mem
80   intro M hM
81   by_cases hMN : M ≤ N
82   · have hmul := homogeneousRightMul_mem_orderSymbolRelation k I
83       (DirectSum.decompose (orderDecomposition k) a (N - M))
84       (DirectSum.decompose (orderDecomposition (n := n) k) x M) (hx M)
85     have hsum : M + (N - M) = N := Nat.add_sub_of_le hMN
86     rw [← hsum]
87     have hMN' : M ≤ M + (N - M) := Nat.le_add_right M (N - M)
88     have heq :
89         DirectSum.decompose (orderDecomposition (n := n) k)
90             (a * (DirectSum.decompose (orderDecomposition (n := n) k) x M :
91               SymbolRing k n)) (M + (N - M)) =
92           homogeneousRightMul k
93             (DirectSum.decompose (orderDecomposition (n := n) k) a (N - M))
94             (DirectSum.decompose (orderDecomposition (n := n) k) x M) := by
95       apply Subtype.ext
96       rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
97         (orderDecomposition k)
98         (DirectSum.decompose (orderDecomposition (n := n) k) x M).property hMN']
99       simp only [coe_homogeneousRightMul]
100       rw [Nat.add_sub_cancel_left]
101       exact mul_comm
102         (DirectSum.decompose (orderDecomposition (n := n) k) a (N - M) :
103           SymbolRing k n)
104         (DirectSum.decompose (orderDecomposition (n := n) k) x M :
105           SymbolRing k n)
106     rw [heq]
107     exact hmul
108   · have heq :
109         DirectSum.decompose (orderDecomposition (n := n) k)
110             (a * (DirectSum.decompose (orderDecomposition (n := n) k) x M :
111               SymbolRing k n)) N = 0 := by
112       apply Subtype.ext
113       exact DirectSum.coe_decompose_mul_of_right_mem_of_not_le
114         (orderDecomposition k)
115         (DirectSum.decompose (orderDecomposition (n := n) k) x M).property hMN
116     rw [heq]
117     exact Submodule.zero_mem _

119 /-- Every homogeneous component of the ideal generated by order-principal
120 symbols is an exact symbol relation in that degree. -/
121 private theorem all_decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
122     (I : RightIdeal (PresentedWeyl k n)) (P : SymbolRing k n)
123     (hP : P ∈ orderInitialIdeal k I) :
124     ∀ N, DirectSum.decompose (orderDecomposition (n := n) k) P N ∈
125       orderSymbolRelation k I N := by
126   classical
127   change P ∈ Ideal.span (orderInitialGeneratorSet k I) at hP
128   induction hP using Submodule.span_induction with
129   | mem P hP =>
130       intro N
131       obtain ⟨M, z, hzM, hzI, rfl⟩ := hP
132       let Q : OrderHomogeneous k M :=
133         principalComponentOnPiece k (@orderWeight n) M ⟨z, hzM⟩
134       have hQ : Q ∈ orderSymbolRelation k I M :=
135         principalComponent_mem_orderSymbolRelation k I M z hzM hzI
136       by_cases hNM : N = M
137       · subst N
138         change DirectSum.decompose (orderDecomposition (n := n) k)
139           (Q : SymbolRing k n) M ∈ orderSymbolRelation k I M
140         simpa only [DirectSum.decompose_coe, DirectSum.of_eq_same] using hQ
141       · have hzero :
142             (DirectSum.decompose (orderDecomposition k)
143               (presentedPrincipalComponent k orderWeight M z) N :
144                 SymbolRing k n) = 0 := by
145           change DirectSum.decompose (orderDecomposition k)
146             (Q : SymbolRing k n) N = (0 : SymbolRing k n)
147           exact DirectSum.decompose_of_mem_ne (orderDecomposition k) Q.property
148             (fun h => hNM h.symm)
149         have hzero' : DirectSum.decompose (orderDecomposition k)
150             (presentedPrincipalComponent k orderWeight M z) N = 0 :=
151           Subtype.ext hzero
152         rw [hzero']
153         exact Submodule.zero_mem _
154   | zero =>
155       intro N
156       rw [DirectSum.decompose_zero]
157       exact Submodule.zero_mem _
158   | add P Q _ _ hP hQ =>
159       intro N
160       simpa only [DirectSum.decompose_add, DirectSum.add_apply] using
161         (Submodule.add_mem (orderSymbolRelation k I N) (hP N) (hQ N))
162   | smul a P _ hP =>
163       intro N
164       change DirectSum.decompose (orderDecomposition k) (a * P) N ∈
165         orderSymbolRelation k I N
166       exact decompose_mul_mem_orderSymbolRelation k I a P hP N
167 
168 /-- Every homogeneous component of the ideal generated by order-principal
169 symbols is an exact symbol relation in that degree. -/
170 theorem decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
171     (I : RightIdeal (PresentedWeyl k n)) (P : SymbolRing k n)
172     (hP : P ∈ orderInitialIdeal k I) (N : ℕ) :
173     DirectSum.decompose (orderDecomposition (n := n) k) P N ∈
174       orderSymbolRelation k I N :=
175   all_decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal k I P hP N
176 
177 /-- Reverse homogeneous-piece inclusion: a degree-`N` homogeneous element of
178 the order initial ideal is an exact relation in degree `N`. -/
179 theorem mem_orderSymbolRelation_of_coe_mem_orderInitialIdeal
180     (I : RightIdeal (PresentedWeyl k n)) (N : ℕ)
181     (P : OrderHomogeneous k N)
182     (hP : (P : SymbolRing k n) ∈ orderInitialIdeal k I) :
183     P ∈ orderSymbolRelation k I N := by
184   have hcomponent :=
185     decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal k I P hP N
186   simpa [DirectSum.decompose_coe] using hcomponent
187 
188 /-- Exact homogeneous relations are precisely the homogeneous pieces of the
189 order initial ideal. -/
190 theorem mem_orderSymbolRelation_iff_coe_mem_orderInitialIdeal
191     (I : RightIdeal (PresentedWeyl k n)) (N : ℕ)
192     (P : OrderHomogeneous k N) :
193     P ∈ orderSymbolRelation k I N ↔
194       (P : SymbolRing k n) ∈ orderInitialIdeal k I :=
195   ⟨coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k I N P,
196     mem_orderSymbolRelation_of_coe_mem_orderInitialIdeal k I N P⟩
