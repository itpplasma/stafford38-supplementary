7 # Exact specialization of the filtered quotient Rees direct sum
8 
9 This file completes the additive exactness statement for the concrete direct
10 sum constructed in `FilteredQuotientRees`.  The filtration inclusions make the
11 degree shift injective, and the kernel of the componentwise associated-graded
12 quotient is exactly its range.
13 
14 Only `k`-linear maps are used here.  In particular, this file introduces no
15 Rees-ring action and asserts no special-fibre ring or module equivalence.
16 -/

37 private theorem quotientOrderReesShift_apply_succ
38     (I : RightIdeal (PresentedWeyl k n))
39     (x : QuotientOrderReesModule k I) (N : ℕ) :
40     (quotientOrderReesShift k I x) (N + 1) =
41       quotientOrderPieceSucc k I N (x N) := by
42   induction x using DirectSum.induction_on with
43   | zero => simp
44   | of M q =>
45       rw [quotientOrderReesShift_of]
46       simp only [DirectSum.of_apply]
47       by_cases h : M = N
48       · subst M
49         simp
50       · have hs : M + 1 ≠ N + 1 := by omega
51         simp [h, hs]
52   | add x y hx hy =>
53       simpa using congrArg₂ (· + ·) hx hy

55 private theorem quotientOrderReesToAssociatedGraded_apply
56     (I : RightIdeal (PresentedWeyl k n))
57     (x : QuotientOrderReesModule k I) (N : ℕ) :
58     (quotientOrderReesToAssociatedGraded k I x) N =
59       quotientOrderPieceToGraded k I N (x N) := by
60   induction x using DirectSum.induction_on with
61   | zero => simp
62   | of M q =>
63       rw [quotientOrderReesToAssociatedGraded_of]
64       simp only [DirectSum.of_apply]
65       by_cases h : M = N
66       · subst M
67         simp
68       · simp [h]
69   | add x y hx hy =>
70       simpa using congrArg₂ (· + ·) hx hy

72 /-- The degree-raising filtration shift on the concrete quotient Rees direct
73 sum is injective. -/
74 theorem quotientOrderReesShift_injective
75     (I : RightIdeal (PresentedWeyl k n)) :
76     Function.Injective (quotientOrderReesShift k I) := by
77   intro x y hxy
78   apply DFinsupp.ext
79   intro N
80   have hcomponent := congrArg (fun z => z (N + 1)) hxy
81   change (quotientOrderReesShift k I x) (N + 1) =
82     (quotientOrderReesShift k I y) (N + 1) at hcomponent
83   rw [quotientOrderReesShift_apply_succ,
84     quotientOrderReesShift_apply_succ] at hcomponent
85   have hval :
86       ((quotientOrderPieceSucc k I N (x N) :
87         quotientOrderPiece k I (N + 1)) : FilteredRightQuotient k I) =
88       ((quotientOrderPieceSucc k I N (y N) :
89         quotientOrderPiece k I (N + 1)) : FilteredRightQuotient k I) :=
90     congrArg
91       (fun z : quotientOrderPiece k I (N + 1) =>
92         (z : FilteredRightQuotient k I)) hcomponent
93   apply Subtype.ext
94   exact hval

96 private theorem homogeneous_mem_range_of_specialization_eq_zero
97     (I : RightIdeal (PresentedWeyl k n)) (N : ℕ)
98     (q : quotientOrderPiece k I N)
99     (hq : quotientOrderPieceToGraded k I N q = 0) :
100     DirectSum.of (fun M => quotientOrderPiece k I M) N q ∈
101       LinearMap.range (quotientOrderReesShift k I) := by
102   cases N with
103   | zero =>
104       have hmem := (Submodule.Quotient.mk_eq_zero _).mp hq
105       have hqzero : q = 0 := by
106         apply Subtype.ext
107         change (q : FilteredRightQuotient k I) = 0
108         simpa [quotientOrderStrictLowerPiece,
109           Stafford38.WeylAssociatedGraded.presentedStrictLowerPiece] using hmem
110       subst q
111       rw [map_zero]
112       exact (LinearMap.range (quotientOrderReesShift k I)).zero_mem
113   | succ N =>
114       have hmem := (Submodule.Quotient.mk_eq_zero _).mp hq
115       have hprev : (q : FilteredRightQuotient k I) ∈
116           quotientOrderPiece k I N := by
117         simpa [quotientOrderStrictLowerPiece_succ_eq] using hmem
118       let qprev : quotientOrderPiece k I N := ⟨q, hprev⟩
119       refine ⟨DirectSum.of (fun M => quotientOrderPiece k I M) N qprev, ?_⟩
120       rw [quotientOrderReesShift_of]
121       congr 1

123 /-- Exactness of specialization for the concrete filtered quotient direct sum:
124 the elements killed by passage to the associated graded object are precisely
125 the images of the degree-raising filtration shift. -/
126 theorem quotientOrderReesToAssociatedGraded_ker_eq_range_shift
127     (I : RightIdeal (PresentedWeyl k n)) :
128     LinearMap.ker (quotientOrderReesToAssociatedGraded k I) =
129       LinearMap.range (quotientOrderReesShift k I) := by
130   classical
131   apply le_antisymm
132   · intro x hx
133     have hmap : quotientOrderReesToAssociatedGraded k I x = 0 :=
134       LinearMap.mem_ker.mp hx
135     rw [← DirectSum.sum_support_of x]
136     apply Submodule.sum_mem
137     intro N hN
138     apply homogeneous_mem_range_of_specialization_eq_zero k I N
139     have hcomponent := congrArg (fun z => z N) hmap
140     simpa [quotientOrderReesToAssociatedGraded_apply] using hcomponent
141   · intro x hx
142     obtain ⟨y, rfl⟩ := hx
143     apply LinearMap.mem_ker.mpr
144     have hcomp := LinearMap.congr_fun
145       (quotientOrderReesToAssociatedGraded_comp_shift k I) y
146     simpa using hcomp
