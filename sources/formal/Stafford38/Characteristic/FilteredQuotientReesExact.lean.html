8 /-!
9 # Exact specialization of the filtered quotient Rees direct sum
10 
11 This file completes the additive exactness statement for the concrete direct
12 sum constructed in `FilteredQuotientRees`.  The filtration inclusions make the
13 degree shift injective, and the kernel of the componentwise associated-graded
14 quotient is exactly its range.
15 
16 Only `k`-linear maps are used here.  In particular, this file introduces no
17 Rees-ring action and asserts no special-fibre ring or module equivalence.
18 -/

39 private theorem quotientOrderReesShift_apply_succ
40     (I : RightIdeal (WeylAlg k n))
41     (x : QuotientOrderReesModule k I) (N : ℕ) :
42     (quotientOrderReesShift k I x) (N + 1) =
43       quotientOrderPieceSucc k I N (x N) := by
44   induction x using DirectSum.induction_on with
45   | zero => simp
46   | of M q =>
47       rw [quotientOrderReesShift_of]
48       simp only [DirectSum.of_apply]
49       by_cases h : M = N
50       · subst M
51         simp
52       · have hs : M + 1 ≠ N + 1 := by omega
53         simp [h, hs]
54   | add x y hx hy =>
55       simpa using congrArg₂ (· + ·) hx hy

57 private theorem quotientOrderReesToAssociatedGraded_apply
58     (I : RightIdeal (WeylAlg k n))
59     (x : QuotientOrderReesModule k I) (N : ℕ) :
60     (quotientOrderReesToAssociatedGraded k I x) N =
61       quotientOrderPieceToGraded k I N (x N) := by
62   induction x using DirectSum.induction_on with
63   | zero => simp
64   | of M q =>
65       rw [quotientOrderReesToAssociatedGraded_of]
66       simp only [DirectSum.of_apply]
67       by_cases h : M = N
68       · subst M
69         simp
70       · simp [h]
71   | add x y hx hy =>
72       simpa using congrArg₂ (· + ·) hx hy

74 /-- The degree-raising filtration shift on the concrete quotient Rees direct
75 sum is injective. -/
76 theorem quotientOrderReesShift_injective
77     (I : RightIdeal (WeylAlg k n)) :
78     Function.Injective (quotientOrderReesShift k I) := by
79   intro x y hxy
80   apply DFinsupp.ext
81   intro N
82   have hcomponent := congrArg (fun z => z (N + 1)) hxy
83   change (quotientOrderReesShift k I x) (N + 1) =
84     (quotientOrderReesShift k I y) (N + 1) at hcomponent
85   rw [quotientOrderReesShift_apply_succ,
86     quotientOrderReesShift_apply_succ] at hcomponent
87   have hval :
88       ((quotientOrderPieceSucc k I N (x N) :
89         quotientOrderPiece k I (N + 1)) : FilteredRightQuotient k I) =
90       ((quotientOrderPieceSucc k I N (y N) :
91         quotientOrderPiece k I (N + 1)) : FilteredRightQuotient k I) :=
92     congrArg
93       (fun z : quotientOrderPiece k I (N + 1) =>
94         (z : FilteredRightQuotient k I)) hcomponent
95   apply Subtype.ext
96   exact hval

98 private theorem homogeneous_mem_range_of_specialization_eq_zero
99     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
100     (q : quotientOrderPiece k I N)
101     (hq : quotientOrderPieceToGraded k I N q = 0) :
102     DirectSum.of (fun M => quotientOrderPiece k I M) N q ∈
103       LinearMap.range (quotientOrderReesShift k I) := by
104   cases N with
105   | zero =>
106       have hmem := (Submodule.Quotient.mk_eq_zero _).mp hq
107       have hqzero : q = 0 := by
108         apply Subtype.ext
109         change (q : FilteredRightQuotient k I) = 0
110         simpa [quotientOrderStrictLowerPiece,
111           Stafford38.WeylAssociatedGraded.presentedStrictLowerPiece] using hmem
112       subst q
113       rw [map_zero]
114       exact (LinearMap.range (quotientOrderReesShift k I)).zero_mem
115   | succ N =>
116       have hmem := (Submodule.Quotient.mk_eq_zero _).mp hq
117       have hprev : (q : FilteredRightQuotient k I) ∈
118           quotientOrderPiece k I N := by
119         simpa [quotientOrderStrictLowerPiece_succ_eq] using hmem
120       let qprev : quotientOrderPiece k I N := ⟨q, hprev⟩
121       refine ⟨DirectSum.of (fun M => quotientOrderPiece k I M) N qprev, ?_⟩
122       rw [quotientOrderReesShift_of]
123       congr 1

125 /-- Exactness of specialization for the concrete filtered quotient direct sum:
126 the elements killed by passage to the associated graded object are precisely
127 the images of the degree-raising filtration shift. -/
128 theorem quotientOrderReesToAssociatedGraded_ker_eq_range_shift
129     (I : RightIdeal (WeylAlg k n)) :
130     LinearMap.ker (quotientOrderReesToAssociatedGraded k I) =
131       LinearMap.range (quotientOrderReesShift k I) := by
132   classical
133   apply le_antisymm
134   · intro x hx
135     have hmap : quotientOrderReesToAssociatedGraded k I x = 0 :=
136       LinearMap.mem_ker.mp hx
137     rw [← DirectSum.sum_support_of x]
138     apply Submodule.sum_mem
139     intro N hN
140     apply homogeneous_mem_range_of_specialization_eq_zero k I N
141     have hcomponent := congrArg (fun z => z N) hmap
142     simpa [quotientOrderReesToAssociatedGraded_apply] using hcomponent
143   · intro x hx
144     obtain ⟨y, rfl⟩ := hx
145     apply LinearMap.mem_ker.mpr
146     have hcomp := LinearMap.congr_fun
147       (quotientOrderReesToAssociatedGraded_comp_shift k I) y
148     simpa using hcomp
