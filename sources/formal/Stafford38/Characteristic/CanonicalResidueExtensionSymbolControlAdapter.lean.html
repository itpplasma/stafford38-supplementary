10 # Residue-extension symbol control from base-relative Poisson closure
11 
12 The consumer `equationConormalLocus_groundMap_subset_geometricReducedSupport`
13 states its scalar extension through `scalarPolynomialMap`, whose coefficient
14 map is `algebraMap k (LaurentSeries K)`.  The terminal interface
15 `CanonicalResidueExtensionSymbolControl` states its conormal locus through
16 `groundPolynomialMap`, now an abbreviation of `scalarPolynomialMap` with
17 the same canonical coefficient map. The equality to the explicit composite
18 `(algebraMap K (LaurentSeries K)).comp (algebraMap k K)` is retained as
19 `groundLaurentComposite_eq_algebraMap`.
20 
21 Diagnosis (`#synth`, `pp.explicit`): both `Algebra k (LaurentSeries K)` and
22 `Algebra K (LaurentSeries K)` are the instance
23 `HahnSeries.powerSeriesAlgebra ℤ K`, whose structure map is
24 `(ofPowerSeries ℤ K).comp (algebraMap _ (PowerSeries K))`, and the power-series
25 structure map is `C ∘ algebraMap`.  The two coefficient maps are therefore
26 propositionally equal ring homomorphisms; the public canonical-map bridge
27 below is now definitional. No `IsScalarTower k K (LaurentSeries K)` instance exists:
28 the `Algebra`-derived scalar action on `LaurentSeries K` is multiplication by
29 the structure map, not `HahnSeries.instSMul`.  No scalar-tower instance is
30 constructed or needed here; the bridge is an equality of ring homomorphisms.
31 
32 The mathematical content is unchanged from the existing adapter: symbol
33 control follows from base-relative Poisson closure of the geometric reduced
34 order support over the same Laurent field `LaurentSeries K`.
35 
36 ## Reduction to a field-generic Gabber fragment
37 
38 `CanonicalResidueExtensionGeometricBaseRelativePoisson` asks for base-relative
39 Poisson closure of the *geometric* reduced support: the radical of the scalar
40 extension, to `LaurentSeries K`, of the ground reduced support.  The repository
41 identifies that ideal with the ordinary reduced order support of the
42 scalar-extended canonical quotient over the field `LaurentSeries K`
43 (`geometricReducedOrderSupportIdeal_eq_scalarExtension`) and transports PBW
44 monicity along the same extension.  The second section records the exact
45 consequence: the residue-extension input is implied by a Gabber fragment
46 stated over an arbitrary characteristic-zero field, with no scalar extension,
47 algebraic closure, or Laurent series in its statement; and that fragment is
48 implied by involutivity of every minimal prime of the order initial ideal.
49 
50 Nothing here proves any Gabber-type statement; the new predicates are
51 theorem-shaped interfaces that a caller must still prove.
52 -/

109 /-- The exact arbitrary-residue-field geometric input.  For every field
110 extension `K/k`, the geometric reduced order support of the canonical quotient
111 is required to be base-relatively Poisson over `LaurentSeries K`.  This is a
112 theorem-shaped interface, not an axiom: callers must provide a proof of it.
113 It is stated verbatim as in
114 `Stafford38.Characteristic.CanonicalResidueExtensionSymbolControlAdapter`. -/
115 def CanonicalResidueExtensionGeometricBaseRelativePoisson : Prop :=
116   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
117     (K : Type u) [Field K] [Algebra k K]
118     (n N : ℕ) (d : PresentedWeyl k (n + 1)),
119     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
120       IsBaseRelativePoisson
121         (geometricReducedOrderSupportIdeal
122           (K := LaurentSeries K)
123           (canonicalRightIdeal (presentedCoordinate k n) d N))
124 
125 /-- Arbitrary residue-field symbol control follows from the corresponding
126 base-relative Poisson statement over the same completed residue field. -/
127 theorem canonicalResidueExtensionSymbolControl_of_geometric_inputs
128     (hgabber : CanonicalResidueExtensionGeometricBaseRelativePoisson.{u}) :
129     CanonicalResidueExtensionSymbolControl.{u} := by
130   intro k _ _ _ K _ _ n N d hN hd
131   have : CharZero K :=
132     charZero_of_injective_algebraMap (algebraMap k K).injective
133   have : CharZero (LaurentSeries K) :=
134     charZero_of_injective_algebraMap
135       (algebraMap K (LaurentSeries K)).injective
136   let W := canonicalRightIdeal (presentedCoordinate k n) d N
137   rcases exists_canonical_fibrePolynomial n N hd with ⟨P, hP⟩
138   refine ⟨P, hP, ?_⟩
139   intro q hq
140   have hq' : q ∈ equationConormalLocus
141       ((reducedOrderBaseIdeal k W).map
142         (scalarPolynomialMap (k := k) (K := LaurentSeries K)
143           (Fin (n + 1)))) := by
144     rw [groundEquationConormalLocus_eq] at hq
145     exact hq
146   have hqSupport : q ∈ MvPolynomial.zeroLocus (LaurentSeries K)
147       (geometricReducedOrderSupportIdeal
148         (K := LaurentSeries K) W) :=
149     equationConormalLocus_groundMap_subset_geometricReducedSupport
150       W (hgabber k K n N d hN hd) hq'
151   have hprincipal : presentedPrincipalComponent k
152         (@Stafford38.WeylFiltration.orderWeight (n + 1)) N d ∈
153       reducedOrderSupportIdeal k W :=
154     orderInitialIdeal_le_reducedOrderSupportIdeal k W
155       (canonical_orderPrincipalComponent_mem_initialIdeal k n N hd)
156   have hmapped : scalarPolynomialMap
157         (k := k) (K := LaurentSeries K) (PhaseVar (n + 1))
158         (presentedPrincipalComponent k
159           (@Stafford38.WeylFiltration.orderWeight (n + 1)) N d) ∈
160       geometricReducedOrderSupportIdeal
161         (K := LaurentSeries K) W :=
162     Ideal.le_radical (Ideal.mem_map_of_mem
163       (scalarPolynomialMap
164         (k := k) (K := LaurentSeries K) (PhaseVar (n + 1))) hprincipal)
165   have hzero := hqSupport _ hmapped
166   rw [← hP] at hzero
167   rw [groundLaurentMap_eq_algebraMap]
168   simpa only [MvPolynomial.aeval_eq_eval, eval_scalarPolynomialMap] using hzero

173 /-- The base-relative Gabber fragment for the canonical quotient over an
174 arbitrary characteristic-zero field.  No algebraic closure, scalar extension,
175 or Laurent series occurs in the statement. -/
176 def CanonicalBaseRelativePoissonOverFields : Prop :=
177   ∀ (L : Type u) [Field L] [CharZero L] (n N : ℕ)
178     (d : PresentedWeyl L (n + 1)),
179     0 < N → IsPBWMonicAt L (.inr (0 : Fin (n + 1))) N d →
180       IsBaseRelativePoisson
181         (reducedOrderSupportIdeal L
182           (canonicalRightIdeal (presentedCoordinate L n) d N))
183 
184 /-- Gabber's theorem in its usual form for the canonical quotient over an
185 arbitrary characteristic-zero field: every minimal prime of the order initial
186 ideal is involutive. -/
187 def CanonicalMinimalPrimeInvolutivityOverFields : Prop :=
188   ∀ (L : Type u) [Field L] [CharZero L] (n N : ℕ)
189     (d : PresentedWeyl L (n + 1)),
190     0 < N → IsPBWMonicAt L (.inr (0 : Fin (n + 1))) N d →
191       ∀ P ∈ (orderInitialIdeal L
192           (canonicalRightIdeal (presentedCoordinate L n) d N)).minimalPrimes,
193         IsInvolutive P
194 
195 /-- Minimal-prime involutivity gives the base-relative fragment on the
196 radical. -/
197 theorem canonicalBaseRelativePoissonOverFields_of_minimalPrimeInvolutivity
198     (h : CanonicalMinimalPrimeInvolutivityOverFields.{u}) :
199     CanonicalBaseRelativePoissonOverFields.{u} := by
200   intro L _ _ n N d hN hd
201   exact radical_isBaseRelativePoisson_of_minimalPrimes_isInvolutive
202     (orderInitialIdeal L (canonicalRightIdeal (presentedCoordinate L n) d N))
203     (h L n N d hN hd)
204 
205 /-- The residue-extension geometric input follows from the field-generic
206 fragment, applied over `LaurentSeries K` to the scalar-extended operator. -/
207 theorem canonicalResidueExtensionGeometricBaseRelativePoisson_of_overFields
208     (h : CanonicalBaseRelativePoissonOverFields.{u}) :
209     CanonicalResidueExtensionGeometricBaseRelativePoisson.{u} := by
210   intro k _ _ _ K _ _ n N d hN hd
211   have : CharZero K :=
212     charZero_of_injective_algebraMap (algebraMap k K).injective
213   have : CharZero (LaurentSeries K) :=
214     charZero_of_injective_algebraMap
215       (algebraMap K (LaurentSeries K)).injective
216   rw [geometricReducedOrderSupportIdeal_eq_scalarExtension
217     (k := k) (K := LaurentSeries K) n N d]
218   exact h (LaurentSeries K) n N
219     (presentedWeylScalarExtension (k := k) (K := LaurentSeries K) (n + 1) d)
220     hN
221     (presentedWeylScalarExtension_isPBWMonicAt
222       (k := k) (K := LaurentSeries K) n N hd)
223 
224 /-- Residue-extension symbol control from the field-generic fragment. -/
225 theorem canonicalResidueExtensionSymbolControl_of_overFields
226     (h : CanonicalBaseRelativePoissonOverFields.{u}) :
227     CanonicalResidueExtensionSymbolControl.{u} :=
228   canonicalResidueExtensionSymbolControl_of_geometric_inputs
