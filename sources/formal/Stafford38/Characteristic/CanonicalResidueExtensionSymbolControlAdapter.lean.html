12 # Residue-extension symbol control from base-relative Poisson closure
13 
14 The consumer `equationConormalLocus_groundMap_subset_geometricReducedSupport`
15 states its scalar extension through `scalarPolynomialMap`, whose coefficient
16 map is `algebraMap k (LaurentSeries K)`.  The terminal interface
17 `CanonicalResidueExtensionSymbolControl` states its conormal locus through
18 the same `scalarPolynomialMap`. The equality to the explicit composite
19 `(algebraMap K (LaurentSeries K)).comp (algebraMap k K)` is retained as
20 `groundLaurentComposite_eq_algebraMap`.
21 
22 Diagnosis (`#synth`, `pp.explicit`): both `Algebra k (LaurentSeries K)` and
23 `Algebra K (LaurentSeries K)` are the instance
24 `HahnSeries.powerSeriesAlgebra ℤ K`, whose structure map is
25 `(ofPowerSeries ℤ K).comp (algebraMap _ (PowerSeries K))`, and the power-series
26 structure map is `C ∘ algebraMap`.  The two coefficient maps are therefore
27 propositionally equal ring homomorphisms; the public canonical-map bridge
28 below is now definitional. No `IsScalarTower k K (LaurentSeries K)` instance exists:
29 the `Algebra`-derived scalar action on `LaurentSeries K` is multiplication by
30 the structure map, not `HahnSeries.instSMul`.  No scalar-tower instance is
31 constructed or needed here; the bridge is an equality of ring homomorphisms.
32 
33 The mathematical content is unchanged from the existing adapter: symbol
34 control follows from base-relative Poisson closure of the geometric reduced
35 order support over the same Laurent field `LaurentSeries K`.
36 
37 ## Reduction to a field-generic Gabber fragment
38 
39 `CanonicalResidueExtensionGeometricBaseRelativePoisson` asks for base-relative
40 Poisson closure of the *geometric* reduced support: the radical of the scalar
41 extension, to `LaurentSeries K`, of the ground reduced support.  The repository
42 identifies that ideal with the ordinary reduced order support of the
43 scalar-extended canonical quotient over the field `LaurentSeries K`
44 (`geometricReducedOrderSupportIdeal_eq_scalarExtension`) and transports PBW
45 monicity along the same extension.  The second section records the exact
46 consequence: the residue-extension input is implied by a Gabber fragment
47 stated over an arbitrary characteristic-zero field, with no scalar extension,
48 algebraic closure, or Laurent series in its statement; and that fragment is
49 implied by involutivity of every minimal prime of the order initial ideal.
50 
51 Nothing here proves any Gabber-type statement; the new predicates are
52 theorem-shaped interfaces that a caller must still prove.
53 -/

104 /-- The exact arbitrary-residue-field geometric input.  For every field
105 extension `K/k`, the geometric reduced order support of the canonical quotient
106 is required to be base-relatively Poisson over `LaurentSeries K`.  This is a
107 theorem-shaped interface, not an axiom: callers must provide a proof of it.
108 It is stated verbatim as in
109 `Stafford38.Characteristic.CanonicalResidueExtensionSymbolControlAdapter`. -/
110 def CanonicalResidueExtensionGeometricBaseRelativePoisson : Prop :=
111   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
112     (K : Type u) [Field K] [Algebra k K]
113     (n N : ℕ) (d : WeylAlg k (n + 1)),
114     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
115       IsBaseRelativePoisson
116         (geometricReducedOrderSupportIdeal
117           (K := LaurentSeries K)
118           (canonicalRightIdeal (presentedCoordinate k n) d N))

120 /-- Arbitrary residue-field symbol control follows from the corresponding
121 base-relative Poisson statement over the same completed residue field. -/
122 theorem canonicalResidueExtensionSymbolControl_of_geometric_inputs
123     (hgabber : CanonicalResidueExtensionGeometricBaseRelativePoisson.{u}) :
124     CanonicalResidueExtensionSymbolControl.{u} := by
125   intro k _ _ _ K _ _ n N d hN hd
126   have : CharZero K :=
127     charZero_of_injective_algebraMap (algebraMap k K).injective
128   have : CharZero (LaurentSeries K) :=
129     charZero_of_injective_algebraMap
130       (algebraMap K (LaurentSeries K)).injective
131   let W := canonicalRightIdeal (presentedCoordinate k n) d N
132   rcases exists_canonical_fibrePolynomial n N hd with ⟨P, hP⟩
133   refine ⟨P, hP, ?_⟩
134   intro q hq
135   have hq' : q ∈ equationConormalLocus
136       ((reducedOrderBaseIdeal k W).map
137         (scalarPolynomialMap (k := k) (K := LaurentSeries K)
138           (Fin (n + 1)))) := by
139     rw [groundEquationConormalLocus_eq] at hq
140     exact hq
141   have hqSupport : q ∈ MvPolynomial.zeroLocus (LaurentSeries K)
142       (geometricReducedOrderSupportIdeal
143         (K := LaurentSeries K) W) :=
144     equationConormalLocus_groundMap_subset_geometricReducedSupport
145       W (hgabber k K n N d hN hd) hq'
146   have hprincipal : presentedPrincipalComponent k
147         (@Stafford38.WeylFiltration.orderWeight (n + 1)) N d ∈
148       reducedOrderSupportIdeal k W :=
149     orderInitialIdeal_le_reducedOrderSupportIdeal k W
150       (canonical_orderPrincipalComponent_mem_initialIdeal k n N hd)
151   have hmapped : scalarPolynomialMap
152         (k := k) (K := LaurentSeries K) (PhaseVar (n + 1))
153         (presentedPrincipalComponent k
154           (@Stafford38.WeylFiltration.orderWeight (n + 1)) N d) ∈
155       geometricReducedOrderSupportIdeal
156         (K := LaurentSeries K) W :=
157     Ideal.le_radical (Ideal.mem_map_of_mem
158       (scalarPolynomialMap
159         (k := k) (K := LaurentSeries K) (PhaseVar (n + 1))) hprincipal)
160   have hzero := hqSupport _ hmapped
161   rw [← hP] at hzero
162   rw [groundLaurentMap_eq_algebraMap]
163   simpa only [MvPolynomial.aeval_eq_eval, eval_scalarPolynomialMap] using hzero

168 /-- The base-relative Gabber fragment for the canonical quotient over an
169 arbitrary characteristic-zero field.  No algebraic closure, scalar extension,
170 or Laurent series occurs in the statement. -/
171 def CanonicalBaseRelativePoissonOverFields : Prop :=
172   ∀ (L : Type u) [Field L] [CharZero L] (n N : ℕ)
173     (d : WeylAlg L (n + 1)),
174     0 < N → IsPBWMonicAt L (.inr (0 : Fin (n + 1))) N d →
175       IsBaseRelativePoisson
176         (reducedOrderSupportIdeal L
177           (canonicalRightIdeal (presentedCoordinate L n) d N))
178 
179 /-- Gabber's theorem in its usual form for the canonical quotient over an
180 arbitrary characteristic-zero field: every minimal prime of the order initial
181 ideal is involutive. -/
182 def CanonicalMinimalPrimeInvolutivityOverFields : Prop :=
183   ∀ (L : Type u) [Field L] [CharZero L] (n N : ℕ)
184     (d : WeylAlg L (n + 1)),
185     0 < N → IsPBWMonicAt L (.inr (0 : Fin (n + 1))) N d →
186       ∀ P ∈ (orderInitialIdeal L
187           (canonicalRightIdeal (presentedCoordinate L n) d N)).minimalPrimes,
188         IsInvolutive P
189 
190 /-- Minimal-prime involutivity gives the base-relative fragment on the
191 radical. -/
192 theorem canonicalBaseRelativePoissonOverFields_of_minimalPrimeInvolutivity
193     (h : CanonicalMinimalPrimeInvolutivityOverFields.{u}) :
194     CanonicalBaseRelativePoissonOverFields.{u} := by
195   intro L _ _ n N d hN hd
196   exact radical_isBaseRelativePoisson_of_minimalPrimes_isInvolutive
197     (orderInitialIdeal L (canonicalRightIdeal (presentedCoordinate L n) d N))
198     (h L n N d hN hd)

200 /-- The residue-extension geometric input follows from the field-generic
201 fragment, applied over `LaurentSeries K` to the scalar-extended operator. -/
202 theorem canonicalResidueExtensionGeometricBaseRelativePoisson_of_overFields
203     (h : CanonicalBaseRelativePoissonOverFields.{u}) :
204     CanonicalResidueExtensionGeometricBaseRelativePoisson.{u} := by
205   intro k _ _ _ K _ _ n N d hN hd
206   have : CharZero K :=
207     charZero_of_injective_algebraMap (algebraMap k K).injective
208   have : CharZero (LaurentSeries K) :=
209     charZero_of_injective_algebraMap
210       (algebraMap K (LaurentSeries K)).injective
211   rw [geometricReducedOrderSupportIdeal_eq_scalarExtension
212     (k := k) (K := LaurentSeries K) n N d]
213   exact h (LaurentSeries K) n N
214     (presentedWeylScalarExtension (k := k) (K := LaurentSeries K) (n + 1) d)
215     hN
216     (presentedWeylScalarExtension_isPBWMonicAt
217       (k := k) (K := LaurentSeries K) n N hd)

222     CanonicalResidueExtensionSymbolControl.{u} :=
223   canonicalResidueExtensionSymbolControl_of_geometric_inputs
