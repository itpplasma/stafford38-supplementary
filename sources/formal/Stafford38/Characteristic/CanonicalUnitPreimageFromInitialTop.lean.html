8 /-!
9 # A strict unit preimage from degree-zero initial-ideal generation
10 
11 If the canonical order initial ideal and the distinguished coordinate symbol
12 generate the unit ideal, the degree-zero homogeneous part of that identity is
13 an exact order-symbol relation.  An exact filtered representative of that
14 relation differs from `y * x - 1` by the strict lower order piece in degree
15 zero, which is zero.  This produces the literal order-zero coordinate
16 predecessor required by `StrictUnitCoordinatePreimage`.
17 
18 No noncharacteristic or characteristic-variety theorem is used here.
19 -/

60 /-- The converse needed by the terminal cancellation route: unit generation
61 in the order initial ideal yields an actual order-zero coordinate predecessor
62 of the unit in the canonical right quotient. -/
63 theorem strictUnitCoordinatePreimage_of_orderInitialIdeal_sup_coordinate_eq_top
64     (n N : ℕ) {d : WeylAlg k (n + 1)}
65     (htop :
66       orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
67         Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} = ⊤) :
68     StrictUnitCoordinatePreimage k n N d := by
69   classical
70   let I := CanonicalIdeal k n N d
71   let x := presentedCoordinate k n
72   let X : OrderHomogeneous k (n + 1) 0 :=
73     principalComponentOnPiece k (@orderWeight (n + 1)) 0
74       ⟨x, presentedCoordinate_mem_orderPiece_zero k n⟩
75   have hX : (X : SymbolRing k (n + 1)) =
76       MvPolynomial.X (.inl (0 : Fin (n + 1))) := by
77     exact coe_coordinate_order_symbol k n
78   have honeSup : (1 : SymbolRing k (n + 1)) ∈
79       orderInitialIdeal k I ⊔
80         Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} := by
81     rw [show orderInitialIdeal k I ⊔
82         Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} = ⊤ by
83       simpa [I] using htop]
84     exact Submodule.mem_top
85   obtain ⟨j, hj, s, hs, hjs⟩ := Submodule.mem_sup.mp honeSup
86   obtain ⟨a, ha⟩ := (Ideal.mem_span_singleton.mp hs)
87   have hsEq : s = a * (X : SymbolRing k (n + 1)) := by
88     calc
89       s = MvPolynomial.X (.inl (0 : Fin (n + 1))) * a := ha
90       _ = a * MvPolynomial.X (.inl (0 : Fin (n + 1))) := mul_comm _ _
91       _ = a * (X : SymbolRing k (n + 1)) := by rw [hX]
92   have hrelationInitial : s - 1 ∈ orderInitialIdeal k I := by
93     have hsj : s - 1 = -j := by
94       rw [← hjs]
95       abel
96     rw [hsj]
97     exact (orderInitialIdeal k I).neg_mem hj
98   let A₀ : OrderHomogeneous k (n + 1) 0 :=
99     DirectSum.decompose (orderDecomposition k (n + 1)) a 0
100   let oneH : OrderHomogeneous k (n + 1) 0 :=
101     ⟨1, MvPolynomial.isWeightedHomogeneous_one k orderWeight⟩
102   have hprod :
103       (A₀ : SymbolRing k (n + 1)) * (X : SymbolRing k (n + 1)) ∈
104         MvPolynomial.weightedHomogeneousSubmodule k
105           (@orderWeight (n + 1)) 0 := by
106     simpa using A₀.property.mul X.property
107   have hdiffHom :
108       (A₀ : SymbolRing k (n + 1)) * (X : SymbolRing k (n + 1)) - 1 ∈
109         MvPolynomial.weightedHomogeneousSubmodule k
110           (@orderWeight (n + 1)) 0 :=
111     (MvPolynomial.weightedHomogeneousSubmodule k
112       (@orderWeight (n + 1)) 0).sub_mem hprod oneH.property
113   have hcomponent :
114       DirectSum.decompose (orderDecomposition k (n + 1)) (s - 1) 0 =
115         ⟨(A₀ : SymbolRing k (n + 1)) * (X : SymbolRing k (n + 1)) - 1,
116           hdiffHom⟩ := by
117     apply Subtype.ext
118     rw [DirectSum.decompose_sub, hsEq]
119     have hmul := DirectSum.coe_decompose_mul_of_right_mem_of_le
120       (orderDecomposition k (n + 1)) X.property (Nat.zero_le 0)
121       (a := a) (n := 0)
122     change ((DirectSum.decompose (orderDecomposition k (n + 1))
123         (a * (X : SymbolRing k (n + 1))) 0 :
124           OrderHomogeneous k (n + 1) 0) : SymbolRing k (n + 1)) -
125         ((DirectSum.decompose (orderDecomposition k (n + 1))
126           (1 : SymbolRing k (n + 1)) 0 :
127             OrderHomogeneous k (n + 1) 0) : SymbolRing k (n + 1)) = _
128     rw [hmul]
129     have honeDecomp :
130         DirectSum.decompose (orderDecomposition k (n + 1))
131             (1 : SymbolRing k (n + 1)) 0 = oneH := by
132       change DirectSum.decompose (orderDecomposition k (n + 1))
133           (oneH : SymbolRing k (n + 1)) 0 = oneH
134       simp only [DirectSum.decompose_coe, DirectSum.of_eq_same]
135     rw [honeDecomp]
136   have hcomponentRelation :
137       DirectSum.decompose (orderDecomposition k (n + 1)) (s - 1) 0 ∈
138         orderSymbolRelation k I 0 :=
139     decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
140       k I (s - 1) hrelationInitial 0
141   rw [hcomponent] at hcomponentRelation
142   obtain ⟨yPiece, hySymbol⟩ :=
143     principalComponentOnPiece_surjective k (@orderWeight (n + 1)) A₀
144   let y : WeylAlg k (n + 1) := yPiece
145   have hy : y ∈ orderPiece k (n + 1) 0 := yPiece.property
146   have hx : x ∈ orderPiece k (n + 1) 0 :=
147     presentedCoordinate_mem_orderPiece_zero k n
148   have hyxOne : y * x - 1 ∈ orderPiece k (n + 1) 0 :=
149     (orderPiece k (n + 1) 0).sub_mem
150       (mul_mem_orderPiece k hy hx)
151       (orderPieceOne (n := n + 1) k).property
152   let relationH : OrderHomogeneous k (n + 1) 0 :=
153     ⟨(A₀ : SymbolRing k (n + 1)) * (X : SymbolRing k (n + 1)) - 1,
154       hdiffHom⟩
155   have hrelationH : relationH ∈ orderSymbolRelation k I 0 := by
156     exact hcomponentRelation
157   obtain ⟨z, hz, hzI, hzSymbol⟩ :=
158     (mem_orderSymbolRelation_iff k I 0 relationH).mp hrelationH
159   have hpcEq :
160       principalComponentOnPiece k (@orderWeight (n + 1)) 0
161           ⟨y * x - 1, hyxOne⟩ =
162         principalComponentOnPiece k (@orderWeight (n + 1)) 0 ⟨z, hz⟩ := by
163     rw [← hzSymbol]
164     apply Subtype.ext
165     change presentedPrincipalComponent k orderWeight 0 (y * x - 1) =
166       (relationH : SymbolRing k (n + 1))
167     rw [map_sub, presentedPrincipalComponent_mul_order k hy hx]
168     have honePC : presentedPrincipalComponent k orderWeight 0
169         (1 : WeylAlg k (n + 1)) = 1 := by
170       change presentedPrincipalComponent k orderWeight 0
171           ((orderPieceOne (n := n + 1) k : orderPiece k (n + 1) 0) :
172             WeylAlg k (n + 1)) = 1
173       exact presentedPrincipalComponent_orderPieceOne (n := n + 1) k
174     rw [honePC]
175     change presentedPrincipalComponent k orderWeight 0 y *
176         presentedPrincipalComponent k orderWeight 0 x - 1 = _
177     rw [show presentedPrincipalComponent k orderWeight 0 y =
178         (A₀ : SymbolRing k (n + 1)) by
179       exact congrArg Subtype.val hySymbol]
180     change (A₀ : SymbolRing k (n + 1)) *
181         presentedPrincipalComponent k orderWeight 0 x - 1 = _
182     rw [show presentedPrincipalComponent k orderWeight 0 x =
183         (X : SymbolRing k (n + 1)) by rfl]
184   have heq : y * x - 1 = z := by
185     have hdiffPiece : y * x - 1 - z ∈ orderPiece k (n + 1) 0 :=
186       (orderPiece k (n + 1) 0).sub_mem hyxOne hz
187     have hzero : presentedPrincipalComponent k orderWeight 0
188         (y * x - 1 - z) = 0 := by
189       rw [map_sub]
190       exact sub_eq_zero.mpr (congrArg
191         (fun q : OrderHomogeneous k (n + 1) 0 =>
192           (q : SymbolRing k (n + 1))) hpcEq)
193     have hlower := (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
194       k (@orderWeight (n + 1)) (y * x - 1 - z) hdiffPiece).mp hzero
195     change y * x - 1 - z ∈ (⊥ : Submodule k (WeylAlg k (n + 1))) at hlower
196     rw [Submodule.mem_bot] at hlower
197     exact sub_eq_zero.mp hlower
198   refine ⟨y, hy, ?_⟩
199   apply (Submodule.Quotient.eq I).2
200   change y * x - 1 ∈ I
201   rw [heq]
202   exact hzI
