10 /-!
11 # The localized two-block quotient
12 
13 The localized opposite deformation ring is noncommutative, so the candidate
14 truncation must be formed with a two-sided ideal rather than `Ideal.Quotient`.
15 For an annihilating power in the commutative special fibre, this file pulls
16 the power back to a two-sided ideal `H`, lets `J` be the two-sided span of the
17 products `a * b` with `a,b in H`, and proves that `J` annihilates the actual
18 localized deformation module.  Consequently the action descends to the ring
19 congruence quotient by `J`.
20 
21 No source matrices, adapted basis, or trace identity is assumed here.
22 -/

41 /-- The two-sided ideal of scalars acting by zero on every module vector. -/
42 def actionAnnihilator : TwoSidedIdeal C :=
43   TwoSidedIdeal.mk'
44     {a : C | ∀ w : W, a • w = 0}
45     (by intro w; exact zero_smul C w)
46     (by
47       intro a b ha hb w
48       rw [add_smul, ha w, hb w, add_zero])
49     (by
50       intro a ha w
51       rw [neg_smul, ha w, neg_zero])
52     (by
53       intro a b hb w
54       rw [mul_smul, hb w, smul_zero])
55     (by
56       intro a b ha w
57       rw [mul_smul, ha (b • w)])
58 
59 @[simp]
60 theorem mem_actionAnnihilator_iff {a : C} :
61     a ∈ actionAnnihilator (C := C) (W := W) ↔ ∀ w : W, a • w = 0 := by
62   simp [actionAnnihilator, TwoSidedIdeal.mem_mk']
63 
64 /-- The two-sided ideal generated by products of two elements of `H`. -/
65 def twoBlockIdeal (H : TwoSidedIdeal C) : TwoSidedIdeal C :=
66   TwoSidedIdeal.span
67     {z : C | ∃ a ∈ H, ∃ b ∈ H, a * b = z}
68 
69 /-- Killing every displayed two-block product kills its full two-sided span. -/
70 theorem twoBlockIdeal_le_actionAnnihilator
71     (H : TwoSidedIdeal C)
72     (hprod : ∀ a ∈ H, ∀ b ∈ H, ∀ w : W, (a * b) • w = 0) :
73     twoBlockIdeal H ≤ actionAnnihilator (C := C) (W := W) := by
74   intro z hz
75   change z ∈ TwoSidedIdeal.span
76       {z : C | ∃ a ∈ H, ∃ b ∈ H, a * b = z} at hz
77   rw [TwoSidedIdeal.mem_span_iff] at hz
78   exact hz (actionAnnihilator (C := C) (W := W)) (by
79     rintro z ⟨a, ha, b, hb, rfl⟩
80     exact (mem_actionAnnihilator_iff (C := C) (W := W)).2 (hprod a ha b hb))
81 
82 /-- Scalar multiplication by the quotient ring, when the quotient ideal acts
83 trivially. -/
84 def quotientSMul (J : TwoSidedIdeal C)
85     (hJ : J ≤ actionAnnihilator (C := C) (W := W)) :
86     SMul J.ringCon.Quotient W where
87   smul q w := q.liftOn (fun a : C => a • w) (by
88     intro a b hab
89     have hsub : (a - b) • w = 0 :=
90       (mem_actionAnnihilator_iff (C := C) (W := W)).1
91         (hJ ((J.rel_iff a b).1 hab)) w
92     rw [sub_smul] at hsub
93     exact sub_eq_zero.mp hsub)
94 
95 /-- The original module action descended through a noncommutative two-sided
96 quotient. -/
97 def quotientModule (J : TwoSidedIdeal C)
98     (hJ : J ≤ actionAnnihilator (C := C) (W := W)) :
99     Module J.ringCon.Quotient W := by
100   letI : SMul J.ringCon.Quotient W := quotientSMul J hJ
101   exact Function.Surjective.moduleLeft J.ringCon.mk'
102       Quotient.mk''_surjective (by
103     intro a w
104     rfl)
105 
106 @[simp]
107 theorem quotient_mk_smul (J : TwoSidedIdeal C)
108     (hJ : J ≤ actionAnnihilator (C := C) (W := W))
109     (a : C) (w : W) :
110     letI := quotientModule J hJ
111     J.ringCon.mk' a • w = a • w := by
112   change (quotientSMul J hJ).smul (J.ringCon.mk' a) w = a • w
113   rfl

117 /-- A ring homomorphism that kills a two-sided ideal descends through its
118 ring-congruence quotient. -/
119 def quotientRingHom (J : TwoSidedIdeal C) (f : C →+* T)
120     (hf : ∀ a ∈ J, f a = 0) : J.ringCon.Quotient →+* T where
121   toFun q := q.liftOn f (by
122     intro a b hab
123     have hz : f (a - b) = 0 := hf (a - b) ((J.rel_iff a b).1 hab)
124     rw [map_sub] at hz
125     exact sub_eq_zero.mp hz)
126   map_zero' := f.map_zero
127   map_one' := f.map_one
128   map_add' x y := Quotient.inductionOn₂' x y fun a b => f.map_add a b
129   map_mul' x y := Quotient.inductionOn₂' x y fun a b => f.map_mul a b
130 
131 @[simp]
132 theorem quotientRingHom_mk (J : TwoSidedIdeal C) (f : C →+* T)
133     (hf : ∀ a ∈ J, f a = 0) (a : C) :
134     quotientRingHom J f hf (J.ringCon.mk' a) = f a := by
135   unfold quotientRingHom
136   rfl

159 /-- The pullback `H` of the annihilating maximal-ideal power.  It is a
160 two-sided ideal because it is formed by comapping the corresponding
161 two-sided ideal of the commutative special fibre. -/
162 def localizedHighPowerPullback (q : ℕ) : TwoSidedIdeal Cₗ :=
163   TwoSidedIdeal.comap (localizedOppositeSpecialization D S)
164     (TwoSidedIdeal.fromIdeal (𝔪 ^ q))
165 
166 theorem mem_localizedHighPowerPullback_iff (q : ℕ) (a : Cₗ) :
167     a ∈ localizedHighPowerPullback D S q ↔
168       localizedOppositeSpecialization D S a ∈ 𝔪 ^ q := by
169   rw [localizedHighPowerPullback, TwoSidedIdeal.mem_comap]
170   have h := (TwoSidedIdeal.orderIsoIdeal (R := A₀)).right_inv (𝔪 ^ q)
171   change localizedOppositeSpecialization D S a ∈
172       TwoSidedIdeal.asIdeal (TwoSidedIdeal.fromIdeal (𝔪 ^ q)) ↔ _
173   exact SetLike.ext_iff.mp h (localizedOppositeSpecialization D S a)
174 
175 /-- The genuine noncommutative two-block ideal `J`. -/
176 def localizedTwoBlockIdeal (q : ℕ) : TwoSidedIdeal Cₗ :=
177   twoBlockIdeal (localizedHighPowerPullback D S q)

179 /-- The doubled commutative power used as the specialization target. -/
180 def localizedDoubledPower (q : ℕ) : Ideal A₀ :=
181   (𝔪 ^ q) * (𝔪 ^ q)
182 
183 /-- Specialization followed by the commutative doubled-power quotient. -/
184 def localizedDoubledPowerSpecialization (q : ℕ) :
185     Cₗ →+* A₀ ⧸ localizedDoubledPower S q :=
186   (Ideal.Quotient.mk (localizedDoubledPower S q)).comp
187     (localizedOppositeSpecialization D S)

189 /-- The two-block ideal lies in the kernel of doubled-power specialization. -/
190 theorem localizedTwoBlockIdeal_le_doubledPowerSpecializationKernel
191     (q : ℕ) :
192     ∀ z ∈ localizedTwoBlockIdeal D S q,
193       localizedDoubledPowerSpecialization D S q z = 0 := by
194   intro z hz
195   change z ∈ TwoSidedIdeal.span
196       {z : Cₗ | ∃ a ∈ localizedHighPowerPullback D S q,
197         ∃ b ∈ localizedHighPowerPullback D S q, a * b = z} at hz
198   rw [TwoSidedIdeal.mem_span_iff] at hz
199   have hker := hz
200     (TwoSidedIdeal.ker (localizedDoubledPowerSpecialization D S q)) (by
201       rintro z ⟨a, ha, b, hb, rfl⟩
202       apply (TwoSidedIdeal.mem_ker
203         (localizedDoubledPowerSpecialization D S q)).2
204       rw [map_mul]
205       change Ideal.Quotient.mk (localizedDoubledPower S q)
206           (localizedOppositeSpecialization D S a) *
207         Ideal.Quotient.mk (localizedDoubledPower S q)
208           (localizedOppositeSpecialization D S b) = 0
209       rw [← map_mul, Ideal.Quotient.eq_zero_iff_mem]
210       exact Ideal.mul_mem_mul
211         ((mem_localizedHighPowerPullback_iff D S q a).1 ha)
212         ((mem_localizedHighPowerPullback_iff D S q b).1 hb))
213   exact (TwoSidedIdeal.mem_ker
214     (localizedDoubledPowerSpecialization D S q)).1 hker

216 /-- The first load-bearing quotient statement: `J` acts trivially on the
217 localized deformation module. -/
218 theorem localizedTwoBlockIdeal_le_actionAnnihilator
219     (q : ℕ)
220     (hpow : 𝔪 ^ q ≤
221       Module.annihilator A₀ (LocalizedModule S G)) :
222     localizedTwoBlockIdeal D S q ≤
223       actionAnnihilator (C := Cₗ) (W := Wₗ) := by
224   apply twoBlockIdeal_le_actionAnnihilator
225   intro a ha b hb w
226   apply product_smul_eq_zero_of_specializations_mem_annihilator D S a b
227   · exact hpow ((mem_localizedHighPowerPullback_iff D S q a).1 ha)
228   · exact hpow ((mem_localizedHighPowerPullback_iff D S q b).1 hb)

230 /-- The localized two-block quotient ring. Artinianness is not asserted. -/
231 abbrev LocalizedTwoBlockRing (q : ℕ) :=
232   (localizedTwoBlockIdeal D S q).ringCon.Quotient

234 /-- The honestly descended ring specialization.  Its target is the
235 commutative doubled-power quotient; no identification of module special
236 fibres is claimed. -/
237 def localizedTwoBlockSpecialization (q : ℕ) :
238     LocalizedTwoBlockRing D S q →+*
239       A₀ ⧸ localizedDoubledPower S q :=
240   quotientRingHom (localizedTwoBlockIdeal D S q)
241     (localizedDoubledPowerSpecialization D S q)
242     (localizedTwoBlockIdeal_le_doubledPowerSpecializationKernel D S q)

247     localizedTwoBlockSpecialization D S q
248         ((localizedTwoBlockIdeal D S q).ringCon.mk' a) =

253 /-- The descended specialization is surjective. -/
254 theorem localizedTwoBlockSpecialization_surjective (q : ℕ) :
255     Function.Surjective (localizedTwoBlockSpecialization D S q) := by
256   intro y
257   obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective y
258   obtain ⟨a, ha⟩ := localizedOppositeSpecialization_surjective D S r
259   refine ⟨(localizedTwoBlockIdeal D S q).ringCon.mk' a, ?_⟩
260   rw [localizedTwoBlockSpecialization_mk, ha]

272 /-- The descended scalar action, exposed without installing a global instance. -/
273 def localizedTwoBlockAction
274     (q : ℕ)
275     (hpow : 𝔪 ^ q ≤
276       Module.annihilator A₀ (LocalizedModule S G))
277     (a : LocalizedTwoBlockRing D S q) (w : Wₗ) : Wₗ :=

306 /-- The parameter remains square-zero in the two-block quotient. -/
307 theorem localizedTwoBlock_parameter_sq
308     (q : ℕ) :
309     ((localizedTwoBlockIdeal D S q).ringCon.mk'
310         (OreLocalization.numeratorRingHom (MulOpposite.op D.c))) ^ 2 = 0 := by
311   rw [pow_two]
312   calc
313     (localizedTwoBlockIdeal D S q).ringCon.mk'
314           (OreLocalization.numeratorRingHom (MulOpposite.op D.c)) *
315         (localizedTwoBlockIdeal D S q).ringCon.mk'
316           (OreLocalization.numeratorRingHom (MulOpposite.op D.c)) =
317       (localizedTwoBlockIdeal D S q).ringCon.mk'
318         (OreLocalization.numeratorRingHom (MulOpposite.op D.c) *
319           OreLocalization.numeratorRingHom (MulOpposite.op D.c)) := by
320             exact ((localizedTwoBlockIdeal D S q).ringCon.mk').map_mul _ _ |>.symm
321     _ = (localizedTwoBlockIdeal D S q).ringCon.mk'
322         (OreLocalization.numeratorRingHom
323           (MulOpposite.op D.c * MulOpposite.op D.c)) := by
324             exact congrArg ((localizedTwoBlockIdeal D S q).ringCon.mk')
325               ((OreLocalization.numeratorRingHom
326                 (S := OppositeDenominators D S) (R := Bᵐᵒᵖ)).map_mul _ _).symm
327     _ = 0 := by
328       have hc : D.c * D.c = 0 := by
329         simpa [pow_two] using D.c_sq
330       rw [show MulOpposite.op D.c * MulOpposite.op D.c =
331         MulOpposite.op (D.c * D.c) by rfl, hc]
332       calc
333         (localizedTwoBlockIdeal D S q).ringCon.mk'
334             (OreLocalization.numeratorRingHom (MulOpposite.op 0)) =
335           (localizedTwoBlockIdeal D S q).ringCon.mk' 0 := by
336             exact congrArg ((localizedTwoBlockIdeal D S q).ringCon.mk')
337               ((OreLocalization.numeratorRingHom
338                 (S := OppositeDenominators D S) (R := Bᵐᵒᵖ)).map_zero)
339         _ = 0 := (localizedTwoBlockIdeal D S q).ringCon.mk'.map_zero
