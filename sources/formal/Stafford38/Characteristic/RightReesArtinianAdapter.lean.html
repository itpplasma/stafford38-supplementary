57 /-- A precise missing localized-action consequence.
58 
59 The factors are elements of the actual opposite localized deformation ring,
60 so their displayed product acts in the same order on the localized right-Rees
61 module.  This statement is deliberately elementwise: it is exactly what is
62 needed to make the finite sums defining the high-power product ideal act by
63 zero, without silently postulating a noncommutative ideal quotient.
64 -/
65 def LocalizedHighPowerTwoBlockVanishing (q : ℕ) : Prop :=
66   ∀ {iota : Type*} (s : Finset iota)
67     (a b : iota →
68       OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ))
69     (_ha : ∀ i ∈ s,
70       localizedOppositeSpecialization D S (a i) ∈
71         IsLocalRing.maximalIdeal (Localization S) ^ q)
72     (_hb : ∀ i ∈ s,
73       localizedOppositeSpecialization D S (b i) ∈
74         IsLocalRing.maximalIdeal (Localization S) ^ q)
75     (z : LocalizedDeformationModule D S),
76       (∑ i ∈ s, a i * b i) • z = 0

78 /-- All pre-Artinian inputs already produced by the localized square-zero
79 development, before the missing two-block action bridge and quotient.
80 
81 The structure is a proposition and introduces no replacement ring or module.
82 Every field names an equation for the concrete maps in the imported APIs.
83 Surjectivity and parameter vanishing do not identify the kernel of ring
84 specialization with the principal parameter ideal.
85 -/
86 structure LocalizedPreArtinianCore : Prop where
87   base_charZero : CharZero k
88   localizedFibre_charZero : CharZero (Localization S)
89   localizedFibre_noetherian : IsNoetherianRing (Localization S)
90   ringSpecialization_surjective :
91     Function.Surjective (localizedOppositeSpecialization D S)
92   ringSpecialization_numerator :
93     ∀ b : Bᵐᵒᵖ,
94       localizedOppositeSpecialization D S
95           (OreLocalization.numeratorHom b) =
96         algebraMap R (Localization S) (D.pi b.unop)
97   ringSpecialization_parameter :
98     localizedOppositeSpecialization D S
99         (OreLocalization.numeratorHom (MulOpposite.op D.c)) = 0
100   parameter_sq :
101     ∀ z : LocalizedDeformationModule D S,
102       localizedCAct D S (localizedCAct D S z) = 0
103   parameter_fraction :
104     ∀ (z : N) (s : OppositeDenominators D S),
105       localizedCAct D S (z /ₒ s) =
106         (MulOpposite.op D.c • z) /ₒ s
107   parameter_exact :
108     AddMonoidHom.ker (localizedCAct D S) =
109       AddMonoidHom.range (localizedCAct D S)
110   moduleSpecialization_surjective :
111     Function.Surjective (localizedSpecialization D S)
112   moduleSpecialization_fraction :
113     ∀ (z : N) (s : OppositeDenominators D S),
114       localizedSpecialization D S (z /ₒ s) =
115         LocalizedModule.mk (D.rho z) (denominatorMap D S s)
116   moduleSpecialization_exact :
117     AddMonoidHom.ker (localizedSpecialization D S) =
118       AddMonoidHom.range (localizedCAct D S)
119   fibre_nontrivial : Nontrivial (LocalizedModule S G)
120   fibre_finiteLength : IsFiniteLength (Localization S) (LocalizedModule S G)
121   annihilatingPower :
122     ∃ q : ℕ,
123       IsLocalRing.maximalIdeal (Localization S) ^ q ≤
124         Module.annihilator (Localization S) (LocalizedModule S G)

126 /-- Generic assembly of the localized pre-Artinian core from the existing ring,
127 module, and finite-length APIs.  This theorem does not prove
128 `LocalizedHighPowerTwoBlockVanishing`.
129 -/
130 theorem localizedPreArtinianCore_of_exact_finiteLength
131     (hnontrivial : Nontrivial (LocalizedModule S G))
132     (hlength : IsFiniteLength (Localization S) (LocalizedModule S G)) :
133     LocalizedPreArtinianCore D S := by
134   letI : Nontrivial (LocalizedModule S G) := hnontrivial
135   obtain ⟨q, hq⟩ :=
136     exists_maximalIdeal_pow_le_annihilator_of_finiteLength hlength
137   exact
138     { base_charZero := inferInstance
139       localizedFibre_charZero := inferInstance
140       localizedFibre_noetherian := inferInstance
141       ringSpecialization_surjective :=
142         localizedOppositeSpecialization_surjective D S
143       ringSpecialization_numerator :=
144         localizedOppositeSpecialization_numerator D S
145       ringSpecialization_parameter :=
146         localizedOppositeSpecialization_parameter D S
147       parameter_sq := localizedCAct_sq_eq_zero D S
148       parameter_fraction := localizedCAct_oreDiv D S
149       parameter_exact := localizedCAct_ker_eq_range D S
150       moduleSpecialization_surjective :=
151         localizedSpecialization_surjective D S
152       moduleSpecialization_fraction := by
153         intro z s
154         rw [localizedSpecialization_apply,
155           localizedSpecializationFun_oreDiv]
156       moduleSpecialization_exact :=
157         localizedSpecialization_ker_eq_range D S
158       fibre_nontrivial := hnontrivial
159       fibre_finiteLength := hlength
160       annihilatingPower := ⟨q, hq⟩ }

169 /-- At every minimal prime over the characteristic-module annihilator, the
170 concrete right order-Rees two-jet supplies the full localized pre-Artinian core.
171 
172 The existentially returned Ore-set instance is the one constructed from the
173 square-zero commutator factorization.  The theorem stops before asserting full
174 localized-action semilinearity, principal-parameter ring-kernel exactness,
175 `LocalizedHighPowerTwoBlockVanishing`, or Artinian quotient descent.
176 -/
177 theorem exists_concrete_localizedPreArtinianCore
178     (I : RightIdeal (PresentedWeyl K n))
179     (P : Ideal (SymbolRing K n)) [P.IsPrime]
180     (hP : P ∈
181       (Module.annihilator (SymbolRing K n)
182         (Stafford38.CharacteristicAssociatedGradedModule.OrderAssociatedGradedModule
183           K I)).minimalPrimes) :
184     ∃ h : OreLocalization.OreSet
185         (OppositeDenominators (filteredQuotientTwoJetTraceData K I)
186           P.primeCompl),
187       letI := h
188       LocalizedPreArtinianCore
189         (filteredQuotientTwoJetTraceData K I) P.primeCompl := by
190   obtain ⟨h, _hexact, hnontrivial, hlength⟩ :=
191     Stafford38.Characteristic.AssociatedGradedFinite.exists_minimalPrimeLocalizedExactnessAndFiniteLength
192       K I P hP
193   refine ⟨h, ?_⟩
194   letI := h
195   letI : CharZero (Localization P.primeCompl) :=
196     charZero_of_injective_algebraMap
197       (FaithfulSMul.algebraMap_injective K
198         (Localization P.primeCompl))
199   letI : IsNoetherianRing (Localization P.primeCompl) :=
200     IsLocalization.isNoetherianRing P.primeCompl
201       (Localization P.primeCompl) inferInstance
202   exact localizedPreArtinianCore_of_exact_finiteLength
203     (filteredQuotientTwoJetTraceData K I) P.primeCompl
204       hnontrivial hlength
