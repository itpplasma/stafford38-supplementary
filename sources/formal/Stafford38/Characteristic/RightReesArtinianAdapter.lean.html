59 /-- A precise missing localized-action consequence.
60 
61 The factors are elements of the actual opposite localized deformation ring,
62 so their displayed product acts in the same order on the localized right-Rees
63 module.  This statement is deliberately elementwise: it is exactly what is
64 needed to make the finite sums defining the high-power product ideal act by
65 zero, without silently postulating a noncommutative ideal quotient.
66 -/
67 def LocalizedHighPowerTwoBlockVanishing (q : ℕ) : Prop :=
68   ∀ {iota : Type*} (s : Finset iota)
69     (a b : iota →
70       OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ))
71     (_ha : ∀ i ∈ s,
72       localizedOppositeSpecialization D S (a i) ∈
73         IsLocalRing.maximalIdeal (Localization S) ^ q)
74     (_hb : ∀ i ∈ s,
75       localizedOppositeSpecialization D S (b i) ∈
76         IsLocalRing.maximalIdeal (Localization S) ^ q)
77     (z : LocalizedDeformationModule D S),
78       (∑ i ∈ s, a i * b i) • z = 0

80 /-- All pre-Artinian inputs already produced by the localized square-zero
81 development, before the missing two-block action bridge and quotient.
82 
83 The structure is a proposition and introduces no replacement ring or module.
84 Every field names an equation for the concrete maps in the imported APIs.
85 Surjectivity and parameter vanishing do not identify the kernel of ring
86 specialization with the principal parameter ideal.
87 -/
88 structure LocalizedPreArtinianCore : Prop where
89   base_charZero : CharZero k
90   localizedFibre_charZero : CharZero (Localization S)
91   localizedFibre_noetherian : IsNoetherianRing (Localization S)
92   ringSpecialization_surjective :
93     Function.Surjective (localizedOppositeSpecialization D S)
94   ringSpecialization_numerator :
95     ∀ b : Bᵐᵒᵖ,
96       localizedOppositeSpecialization D S
97           (OreLocalization.numeratorHom b) =
98         algebraMap R (Localization S) (D.pi b.unop)
99   ringSpecialization_parameter :
100     localizedOppositeSpecialization D S
101         (OreLocalization.numeratorHom (MulOpposite.op D.c)) = 0
102   parameter_sq :
103     ∀ z : LocalizedDeformationModule D S,
104       localizedCAct D S (localizedCAct D S z) = 0
105   parameter_fraction :
106     ∀ (z : N) (s : OppositeDenominators D S),
107       localizedCAct D S (z /ₒ s) =
108         (MulOpposite.op D.c • z) /ₒ s
109   parameter_exact :
110     AddMonoidHom.ker (localizedCAct D S) =
111       AddMonoidHom.range (localizedCAct D S)
112   moduleSpecialization_surjective :
113     Function.Surjective (localizedSpecialization D S)
114   moduleSpecialization_fraction :
115     ∀ (z : N) (s : OppositeDenominators D S),
116       localizedSpecialization D S (z /ₒ s) =
117         LocalizedModule.mk (D.rho z) (denominatorMap D S s)
118   moduleSpecialization_exact :
119     AddMonoidHom.ker (localizedSpecialization D S) =
120       AddMonoidHom.range (localizedCAct D S)
121   fibre_nontrivial : Nontrivial (LocalizedModule S G)
122   fibre_finiteLength : IsFiniteLength (Localization S) (LocalizedModule S G)
123   annihilatingPower :
124     ∃ q : ℕ,
125       IsLocalRing.maximalIdeal (Localization S) ^ q ≤
126         Module.annihilator (Localization S) (LocalizedModule S G)

128 /-- Generic assembly of the localized pre-Artinian core from the existing ring,
129 module, and finite-length APIs.  This theorem does not prove
130 `LocalizedHighPowerTwoBlockVanishing`.
131 -/
132 theorem localizedPreArtinianCore_of_exact_finiteLength
133     (hnontrivial : Nontrivial (LocalizedModule S G))
134     (hlength : IsFiniteLength (Localization S) (LocalizedModule S G)) :
135     LocalizedPreArtinianCore D S := by
136   letI : Nontrivial (LocalizedModule S G) := hnontrivial
137   obtain ⟨q, hq⟩ :=
138     exists_maximalIdeal_pow_le_annihilator_of_finiteLength hlength
139   exact
140     { base_charZero := inferInstance
141       localizedFibre_charZero := inferInstance
142       localizedFibre_noetherian := inferInstance
143       ringSpecialization_surjective :=
144         localizedOppositeSpecialization_surjective D S
145       ringSpecialization_numerator :=
146         localizedOppositeSpecialization_numerator D S
147       ringSpecialization_parameter :=
148         localizedOppositeSpecialization_parameter D S
149       parameter_sq := localizedCAct_sq_eq_zero D S
150       parameter_fraction := localizedCAct_oreDiv D S
151       parameter_exact := localizedCAct_ker_eq_range D S
152       moduleSpecialization_surjective :=
153         localizedSpecialization_surjective D S
154       moduleSpecialization_fraction := by
155         intro z s
156         rw [localizedSpecialization_apply,
157           localizedSpecializationFun_oreDiv]
158       moduleSpecialization_exact :=
159         localizedSpecialization_ker_eq_range D S
160       fibre_nontrivial := hnontrivial
161       fibre_finiteLength := hlength
162       annihilatingPower := ⟨q, hq⟩ }

171 /-- At every minimal prime over the characteristic-module annihilator, the
172 concrete right order-Rees two-jet supplies the full localized pre-Artinian core.
173 
174 The existentially returned Ore-set instance is the one constructed from the
175 square-zero ringCommutator factorization.  The theorem stops before asserting full
176 localized-action semilinearity, principal-parameter ring-kernel exactness,
177 `LocalizedHighPowerTwoBlockVanishing`, or Artinian quotient descent.
178 -/
179 theorem exists_concrete_localizedPreArtinianCore
180     (I : RightIdeal (WeylAlg K n))
181     (P : Ideal (SymbolRing K n)) [P.IsPrime]
182     (hP : P ∈
183       (Module.annihilator (SymbolRing K n)
184         (Stafford38.CharacteristicAssociatedGradedModule.OrderAssociatedGradedModule
185           K I)).minimalPrimes) :
186     ∃ h : OreLocalization.OreSet
187         (OppositeDenominators (filteredQuotientTwoJetTraceData K I)
188           P.primeCompl),
189       letI := h
190       LocalizedPreArtinianCore
191         (filteredQuotientTwoJetTraceData K I) P.primeCompl := by
192   obtain ⟨h, _hexact, hnontrivial, hlength⟩ :=
193     Stafford38.Characteristic.AssociatedGradedFinite.exists_minimalPrimeLocalizedExactnessAndFiniteLength
194       K I P hP
195   refine ⟨h, ?_⟩
196   letI := h
197   letI : CharZero (Localization P.primeCompl) :=
198     charZero_of_injective_algebraMap
199       (FaithfulSMul.algebraMap_injective K
200         (Localization P.primeCompl))
201   letI : IsNoetherianRing (Localization P.primeCompl) :=
202     IsLocalization.isNoetherianRing P.primeCompl
203       (Localization P.primeCompl) inferInstance
204   exact localizedPreArtinianCore_of_exact_finiteLength
205     (filteredQuotientTwoJetTraceData K I) P.primeCompl
206       hnontrivial hlength
