12 # The canonical filtered two-term complex
13 
14 This is the concrete filtered complex used for the canonical Weyl quotient.
15 The filtration is the actual order filtration on the quotient, extended by
16 zero in negative order.  No associated-graded identification is assumed here.
17 -/

37 abbrev CanonicalQuotient (n N : ℕ) (d : WeylAlg k (n + 1)) :=
38   FilteredRightQuotient k (presentedCanonicalRightIdeal (k := k) n N d)

40 def rightMulLinearMap (I : RightIdeal (WeylAlg k (n + 1)))
41     (a : WeylAlg k (n + 1)) :
42     FilteredRightQuotient k I →ₗ[k] FilteredRightQuotient k I :=
43   Submodule.mapQ (rightIdealKSubmodule k I) (rightIdealKSubmodule k I)
44     (LinearMap.mulRight k a) (by
45       intro z hz
46       exact I.smul_mem (MulOpposite.op a) hz)
47 
48 @[simp] theorem rightMulLinearMap_mk
49     (I : RightIdeal (WeylAlg k (n + 1)))
50     (a z : WeylAlg k (n + 1)) :
51     rightMulLinearMap k I a (Submodule.Quotient.mk z) =
52       Submodule.Quotient.mk (z * a) := by
53   change Submodule.Quotient.mk (LinearMap.mulRight k a z) = _
54   rfl

56 def canonicalOrderFiltration
57     (I : RightIdeal (WeylAlg k (n + 1))) (p : ℤ) :
58     Submodule k (FilteredRightQuotient k I) :=
59   if hp : p ≤ 0 then quotientOrderPiece k I (-p).toNat else ⊥
60 
61 theorem canonicalOrderFiltration_eq_of_nonpos
62     (I : RightIdeal (WeylAlg k (n + 1))) {p : ℤ} (hp : p ≤ 0) :
63     canonicalOrderFiltration k I p = quotientOrderPiece k I (-p).toNat := by
64   simp [canonicalOrderFiltration, hp]
65 
66 theorem canonicalOrderFiltration_eq_bot_of_pos
67     (I : RightIdeal (WeylAlg k (n + 1))) {p : ℤ} (hp : 0 < p) :
68     canonicalOrderFiltration k I p = ⊥ := by
69   simp [canonicalOrderFiltration, not_le.mpr hp]
70 
71 theorem canonicalOrderFiltration_antitone
72     (I : RightIdeal (WeylAlg k (n + 1))) :
73     Antitone (canonicalOrderFiltration k I) := by
74   intro p q hpq
75   by_cases hq : q ≤ 0
76   · have hp : p ≤ 0 := le_trans hpq hq
77     rw [canonicalOrderFiltration_eq_of_nonpos k I hp,
78       canonicalOrderFiltration_eq_of_nonpos k I hq]
79     apply Submodule.map_mono
80     exact presentedWeightPiece_mono k orderWeight (by omega)
81   · rw [canonicalOrderFiltration_eq_bot_of_pos k I (lt_of_not_ge hq)]
82     exact bot_le
83 
84 theorem canonicalOrderFiltration_exhaustive
85     (I : RightIdeal (WeylAlg k (n + 1))) :
86     ∀ q : FilteredRightQuotient k I, ∃ p : ℤ,
87       q ∈ canonicalOrderFiltration k I p := by
88   intro q
89   obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective
90     (rightIdealKSubmodule k I) q
91   obtain ⟨m, hm⟩ := exists_mem_orderPiece k z
92   refine ⟨-(m : ℤ), ?_⟩
93   rw [canonicalOrderFiltration_eq_of_nonpos k I (by omega)]
94   have hidx : (- - (m : ℤ)).toNat = m := by omega
95   rw [hidx]
96   change (rightIdealKSubmodule k I).mkQ z ∈ quotientOrderPiece k I m
97   exact Submodule.mem_map.mpr ⟨z, hm, rfl⟩
98 
99 private theorem quotientOrderPiece_mul_coordinate_mem
100     (I : RightIdeal (WeylAlg k (n + 1)))
101     {m : ℕ} {q : FilteredRightQuotient k I}
102     (hq : q ∈ quotientOrderPiece k I m)
103     (hx : presentedCoordinate k n ∈ orderPiece k (n + 1) 0) :
104     rightMulLinearMap k I (presentedCoordinate k n) q ∈
105       quotientOrderPiece k I m := by
106   rcases (Submodule.mem_map.mp hq) with ⟨z, hz, rfl⟩
107   change rightMulLinearMap k I (presentedCoordinate k n)
108     (Submodule.Quotient.mk z) ∈ quotientOrderPiece k I m
109   rw [rightMulLinearMap_mk]
110   apply Submodule.mem_map.mpr
111   refine ⟨z * presentedCoordinate k n, mul_mem_orderPiece k hz hx, ?_⟩
112   rfl

114 def canonicalFilteredTwoTerm (n N : ℕ)
115     (d : WeylAlg k (n + 1))
116     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
117     FilteredTwoTerm k (CanonicalQuotient k n N d) where
118   G := canonicalOrderFiltration k (presentedCanonicalRightIdeal (k := k) n N d)
119   antitone := canonicalOrderFiltration_antitone k _
120   f := rightMulLinearMap k (presentedCanonicalRightIdeal (k := k) n N d) (presentedCoordinate k n)
121   map_le := by
122     intro p
123     by_cases hp : p ≤ 0
124     · rw [canonicalOrderFiltration_eq_of_nonpos k _ hp]
125       apply Submodule.map_le_iff_le_comap.mpr
126       intro z hz
127       exact quotientOrderPiece_mul_coordinate_mem k _ hz
128         (presentedCoordinate_mem_orderPiece_zero k n)
129     · rw [canonicalOrderFiltration_eq_bot_of_pos k _ (lt_of_not_ge hp)]
130       simp
131 
132 theorem canonicalFilteredTwoTerm_f_surjective
133     (n N : ℕ) {d : WeylAlg k (n + 1)}
134     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
135     Function.Surjective
136       (canonicalFilteredTwoTerm k n N d hd).f := by
137   intro q
138   obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective
139     (rightIdealKSubmodule k (presentedCanonicalRightIdeal (k := k) n N d)) q
140   obtain ⟨y, hy⟩ :=
141     presentedCanonicalRightQuotient_rightMul_coordinate_surjective
142       (k := k) n N hd (qmk (presentedCanonicalRightIdeal (k := k) n N d) z)
143   obtain ⟨y', hy'⟩ := Submodule.Quotient.mk_surjective
144     (rightIdealKSubmodule k (presentedCanonicalRightIdeal (k := k) n N d)) y
145   rw [← hy'] at hy
146   refine ⟨Submodule.Quotient.mk y', ?_⟩
147   change rightMulLinearMap k (presentedCanonicalRightIdeal (k := k) n N d)
148       (presentedCoordinate k n) (Submodule.Quotient.mk y') =
149     Submodule.Quotient.mk z
150   rw [rightMulLinearMap_mk]
151   exact hy
