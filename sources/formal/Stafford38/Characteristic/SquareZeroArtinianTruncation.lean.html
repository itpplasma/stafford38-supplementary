6 /-!
7 # Artinian truncation in a square-zero deformation
8 
9 This file isolates the algebraic replacement for the invalid pullback-
10 filtration division step.  If a power of an ideal annihilates the special
11 fibre, then every deformation-ring element specializing into that power sends
12 the deformation module into the parameter layer.  Two such blocks act by
13 zero, because the parameter is central and square-zero.
14 
15 The result is deliberately stated without a noncommutative ideal on the
16 deformation ring.  Membership is tested after specialization in the
17 commutative fibre, which is exactly the datum available in the Rees two-jet.
18 -/

40 /-- If `m^n` annihilates the special fibre, a deformation element whose
41 specialization lies in `m^n` sends every vector into the parameter layer. -/
42 theorem rightAction_mem_parameterLayer_of_pi_mem_pow
43     (hpow : m ^ n ≤ Module.annihilator R G)
44     {a : B} (ha : D.pi a ∈ m ^ n) (z : N) :
45     MulOpposite.op a • z ∈ LinearMap.range D.cAct := by
46   exact rightAction_mem_range_cAct_of_pi_mem_annihilator D (hpow ha) z
47 
48 /-- Two blocks specializing into an annihilating ideal power act successively
49 by zero. This is a two-step parameter-layer statement; it does not say that
50 every element specializing into `m ^ (2 * n)` acts by zero. -/
51 theorem iterated_rightAction_eq_zero_of_pi_mem_pow
52     (hpow : m ^ n ≤ Module.annihilator R G)
53     {a b : B} (ha : D.pi a ∈ m ^ n) (hb : D.pi b ∈ m ^ n) (z : N) :
54     MulOpposite.op b • (MulOpposite.op a • z) = 0 := by
55   exact iterated_rightAction_eq_zero_of_pi_mem_annihilator D
56     (hpow ha) (hpow hb) z
57 
58 /-- Equivalently, the product of two deformation blocks lying over `m^n`
59 acts trivially on the deformation module. -/
60 theorem product_rightAction_eq_zero_of_pi_mem_pow
61     (hpow : m ^ n ≤ Module.annihilator R G)
62     {a b : B} (ha : D.pi a ∈ m ^ n) (hb : D.pi b ∈ m ^ n) (z : N) :
63     MulOpposite.op (a * b) • z = 0 := by
64   simpa only [MulOpposite.op_mul, mul_smul] using
65     iterated_rightAction_eq_zero_of_pi_mem_pow D m n hpow ha hb z
66 
67 /-- Every finite sum of two-block products acts trivially. This is the
68 elementwise ideal-product form needed by an Artinian quotient construction;
69 no noncommutative ideal API on `B` is required. -/
70 theorem sum_products_rightAction_eq_zero_of_pi_mem_pow
71     {ι : Type*} (s : Finset ι)
72     (hpow : m ^ n ≤ Module.annihilator R G)
73     (a b : ι → B)
74     (ha : ∀ i ∈ s, D.pi (a i) ∈ m ^ n)
75     (hb : ∀ i ∈ s, D.pi (b i) ∈ m ^ n) (z : N) :
76     MulOpposite.op (∑ i ∈ s, a i * b i) • z = 0 := by
77   rw [show MulOpposite.op (∑ i ∈ s, a i * b i) =
78       ∑ i ∈ s, MulOpposite.op (a i * b i) by
79     exact map_sum (MulOpposite.opAddEquiv : B ≃+ Bᵐᵒᵖ) _ s]
80   rw [Finset.sum_smul]
81   apply Finset.sum_eq_zero
82   intro i hi
83   exact product_rightAction_eq_zero_of_pi_mem_pow D m n hpow
84     (ha i hi) (hb i hi) z
