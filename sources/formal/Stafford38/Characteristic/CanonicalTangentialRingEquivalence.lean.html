18 /-- The old rank-`n` phase variable, regarded as a rank-`n+1` variable which
19 is neither the new momentum nor the new coordinate. -/
20 def oldTangentialVar (n : ℕ) (i : Fin n ⊕ Fin n) : TangentialVar n :=
21   ⟨⟨Stafford38FixedSourceChallenge.oldIndex i, by
22       cases i with
23       | inl i => simp [Stafford38FixedSourceChallenge.oldIndex]
24       | inr i => simp [Stafford38FixedSourceChallenge.oldIndex]⟩,
25     by
26       cases i with
27       | inl i => simp [Stafford38FixedSourceChallenge.oldIndex]
28       | inr i => simp [Stafford38FixedSourceChallenge.oldIndex]⟩
29 
30 theorem oldTangentialVar_injective (n : ℕ) :
31     Function.Injective (oldTangentialVar n) := by
32   intro i j h
33   apply Stafford38.WeylPBW.oldIndex_injective
34   exact congrArg (fun v : TangentialVar n => v.1.1) h
35 
36 theorem oldTangentialVar_surjective (n : ℕ) :
37     Function.Surjective (oldTangentialVar n) := by
38   intro v
39   rcases v with ⟨⟨w, hnormal⟩, hcoordinate⟩
40   rcases w with i | i
41   · rcases Fin.eq_zero_or_eq_succ i with rfl | ⟨j, rfl⟩
42     · exact False.elim (hcoordinate (by rfl))
43     · exact ⟨Sum.inl j, rfl⟩
44   · rcases Fin.eq_zero_or_eq_succ i with rfl | ⟨j, rfl⟩
45     · exact False.elim (hnormal (by rfl))
46     · exact ⟨Sum.inr j, rfl⟩

48 /-- The exact variable dictionary: the two retained subtype exclusions are
49 precisely the image of `Stafford38FixedSourceChallenge.oldIndex`. -/
50 def oldTangentialVarEquiv (n : ℕ) :
51     (Fin n ⊕ Fin n) ≃ TangentialVar n :=
52   Equiv.ofBijective (oldTangentialVar n)
53     ⟨oldTangentialVar_injective n, oldTangentialVar_surjective n⟩

55 @[simp] theorem oldTangentialVarEquiv_apply (n : ℕ) (i : Fin n ⊕ Fin n) :
56     (oldTangentialVarEquiv n i).1.1 = Stafford38FixedSourceChallenge.oldIndex i :=
57   rfl

59 /-- Rename old phase variables into the coefficient ring which omits both
60 new variables. -/
61 def oldSymbolTangentialAlgEquiv (n : ℕ) :
62     MvPolynomial (Fin n ⊕ Fin n) k ≃ₐ[k] tangentialCoeffRing (k := k) n :=
63   MvPolynomial.renameEquiv k (oldTangentialVarEquiv n)

67     oldSymbolTangentialAlgEquiv (k := k) n (MvPolynomial.X i) =
68       MvPolynomial.X (oldTangentialVarEquiv n i) := by

71 /-- Under the full coefficient-action homomorphism, the renamed variable is
72 the actual ambient symbol indexed by `Stafford38FixedSourceChallenge.oldIndex`. -/
73 theorem tangentialCoeffActionHom_oldSymbol_X (n : ℕ)

75     ((tangentialPolynomialActionHom (k := k) n).comp Polynomial.C)

83 /-- Consequently the named tangential coefficient module acts on every
84 symbol module by the same old-generator symbol used by the canonical total
85 polynomial action. -/
86 theorem tangentialCoeffModule_oldSymbol_X_action
87     (n : ℕ) (E : Type*) [AddCommGroup E]
88     [Module (SymbolRing k (n + 1)) E]
89     (i : Fin n ⊕ Fin n) (z : E) :

92         (oldSymbolTangentialAlgEquiv (k := k) n (MvPolynomial.X i)) z =
93       (MvPolynomial.X (Stafford38FixedSourceChallenge.oldIndex i) : SymbolRing k (n + 1)) • z := by
94   change ((tangentialPolynomialActionHom (k := k) n).comp Polynomial.C)
95       (oldSymbolTangentialAlgEquiv (k := k) n (MvPolynomial.X i)) • z = _
