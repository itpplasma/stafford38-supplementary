8 /-!
9 # Global support of the filtered right-ideal quotient
10 
11 The exact homogeneous relation theorem identifies, as an additive group and
12 as a vector space over the coefficient field, the external direct sum of the
13 actual filtered quotient pieces with the quotient of the symbol ring by the
14 order initial ideal.
15 
16 The further transport of the full symbol-ring module structure, and hence the
17 support equality, is not proved in this file.  No conclusion about vanishing
18 of the ungraded Weyl quotient is made here.
19 -/

46 /-- Decompose a symbol into homogeneous pieces and reduce each piece by the
47 exact relation submodule of the filtered quotient. -/
48 def symbolToOrderRelationGraded
49     (I : RightIdeal (WeylAlg k n)) :
50     SymbolRing k n →ₗ[k] OrderSymbolRelationGraded k I :=
51   (DFinsupp.mapRange.linearMap fun N =>
52       (orderSymbolRelation k I N).mkQ).comp
53     (DirectSum.decomposeLinearEquiv (orderDecomposition (n := n) k)).toLinearMap
54 
55 @[simp] theorem symbolToOrderRelationGraded_apply
56     (I : RightIdeal (WeylAlg k n)) (P : SymbolRing k n) (N : ℕ) :
57     symbolToOrderRelationGraded k I P N =
58       Submodule.Quotient.mk
59         (DirectSum.decompose (orderDecomposition (n := n) k) P N) := by
60   rfl
61 
62 /-- The global symbol map is onto: choose a representative in each nonzero
63 homogeneous quotient component and reassemble the resulting finite family. -/
64 theorem symbolToOrderRelationGraded_surjective
65     (I : RightIdeal (WeylAlg k n)) :
66     Function.Surjective (symbolToOrderRelationGraded k I) := by
67   classical
68   intro q
69   let lift : ∀ N,
70       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N ⧸
71         orderSymbolRelation k I N) →
72       MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N :=
73     fun N qN => if hq : qN = 0 then 0 else Classical.choose
74       (Submodule.Quotient.mk_surjective (orderSymbolRelation k I N) qN)
75   have hlift : ∀ N qN, Submodule.Quotient.mk (lift N qN) = qN := by
76     intro N qN
77     by_cases hq : qN = 0
78     · simp [lift, hq]
79     · simpa [lift, hq] using Classical.choose_spec
80         (Submodule.Quotient.mk_surjective (orderSymbolRelation k I N) qN)
81   let q' : DirectSum ℕ (fun N =>
82       MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N) :=
83     DFinsupp.mapRange lift (fun N => by simp [lift]) q
84   refine ⟨(DirectSum.decomposeLinearEquiv
85     (orderDecomposition (n := n) k)).symm q', ?_⟩
86   apply DFinsupp.ext
87   intro N
88   rw [symbolToOrderRelationGraded_apply]
89   have hd := (DirectSum.decomposeAddEquiv
90     (orderDecomposition (n := n) k)).apply_symm_apply q'
91   rw [show DirectSum.decompose (orderDecomposition (n := n) k)
92       ((DirectSum.decomposeLinearEquiv
93         (orderDecomposition (n := n) k)).symm q') N = q' N from
94     congrArg (fun z => z N) hd]
95   change Submodule.Quotient.mk (q' N) = q N
96   exact hlift N (q N)

98 /-- The kernel of the global symbol map is exactly the order initial ideal,
99 viewed as a `k`-submodule. -/
100 theorem ker_symbolToOrderRelationGraded
101     (I : RightIdeal (WeylAlg k n)) :
102     LinearMap.ker (symbolToOrderRelationGraded k I) =
103       (orderInitialIdeal k I).restrictScalars k := by
104   ext P
105   constructor
106   · intro hP
107     rw [LinearMap.mem_ker] at hP
108     have hcomponent : ∀ N,
109         DirectSum.decompose (orderDecomposition (n := n) k) P N ∈
110           orderSymbolRelation k I N := by
111       intro N
112       rw [← Submodule.Quotient.mk_eq_zero]
113       exact congrArg (fun q => q N) hP
114     change P ∈ orderInitialIdeal k I
115     classical
116     rw [← DirectSum.sum_support_decompose (orderDecomposition (n := n) k) P]
117     apply Ideal.sum_mem
118     intro N hN
119     exact (mem_orderSymbolRelation_iff_coe_mem_orderInitialIdeal k I N _).mp
120       (hcomponent N)
121   · intro hP
122     rw [LinearMap.mem_ker]
123     apply DFinsupp.ext
124     intro N
125     change Submodule.Quotient.mk
126       (DirectSum.decompose (orderDecomposition (n := n) k) P N) = 0
127     rw [Submodule.Quotient.mk_eq_zero]
128     exact decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
129       k I P hP N

131 /-- The cyclic order-characteristic module is linearly equivalent over the
132 coefficient field to the direct sum of the exact homogeneous quotients. -/
133 def orderCharacteristicModuleEquivRelationGraded
134     (I : RightIdeal (WeylAlg k n)) :
135     (SymbolRing k n ⧸ (orderInitialIdeal k I).restrictScalars k) ≃ₗ[k]
136       OrderSymbolRelationGraded k I :=
137   Submodule.quotEquivOfEq _ _ (ker_symbolToOrderRelationGraded k I).symm ≪≫ₗ
138     (symbolToOrderRelationGraded k I).quotKerEquivOfSurjective
139       (symbolToOrderRelationGraded_surjective k I)
140 
141 /-- The actual associated graded object is globally equivalent, as a
142 `k`-vector space, to the cyclic symbol quotient.  Its restriction to each
143 degree is the previously proved filtered-quotient equivalence. -/
144 def quotientOrderAssociatedGradedEquivCharacteristic
145     (I : RightIdeal (WeylAlg k n)) :
146     QuotientOrderAssociatedGraded k I ≃ₗ[k]
147       (SymbolRing k n ⧸ (orderInitialIdeal k I).restrictScalars k) :=
148   quotientOrderAssociatedGradedEquivSymbols k I ≪≫ₗ
149     (orderCharacteristicModuleEquivRelationGraded k I).symm
150 
151 /-- Restricting the scalar ring of an ideal does not change the underlying
152 additive quotient. -/
153 def restrictScalarsQuotientAddEquiv
154     (I : RightIdeal (WeylAlg k n)) :
155     (SymbolRing k n ⧸ (orderInitialIdeal k I).restrictScalars k) ≃+
156       OrderCharacteristicModule k I where
157   toEquiv := Quotient.congr (Equiv.refl _) (by
158     intro P Q
159     rfl)
160   map_add' q r := by
161     refine Quotient.inductionOn₂ q r ?_
162     intro P Q
163     rfl
164 
165 /-- Additive global equivalence from the actual associated graded pieces to
166 the cyclic symbol quotient. -/
167 def quotientOrderAssociatedGradedAddEquivCharacteristic
168     (I : RightIdeal (WeylAlg k n)) :
169     QuotientOrderAssociatedGraded k I ≃+ OrderCharacteristicModule k I :=
170   (quotientOrderAssociatedGradedEquivCharacteristic k I).toAddEquiv.trans
171     (restrictScalarsQuotientAddEquiv k I)
