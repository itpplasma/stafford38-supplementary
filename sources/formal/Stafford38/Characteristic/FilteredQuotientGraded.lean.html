9 # The graded bridge for a filtered right-ideal quotient
10 
11 The degreewise equivalences of `FilteredQuotient` assemble into an external
12 direct-sum equivalence.  The exact symbol relation in degree `N` is stable
13 under multiplication by a homogeneous symbol of degree `M`; transporting this
14 operator gives the actual quotient graded pieces homogeneous multiplication
15 maps compatible with Weyl multiplication.
16 
17 This file does not identify the direct sum with
18 `SymbolRing / orderInitialIdeal`.  That final identification requires proving
19 that the degree-`N` part of the ideal generated by all initial symbols is
20 exactly `orderSymbolRelation k I N`.
21 -/

41 /-- The external direct sum of the actual associated-graded pieces of `A/I`. -/
42 abbrev QuotientOrderAssociatedGraded (I : RightIdeal (WeylAlg k n)) :=
43   DirectSum ℕ (fun N => QuotientOrderGradedPiece k I N)
44 
45 /-- The external direct sum of homogeneous symbols modulo the exact relation
46 piece coming from the filtered quotient. -/
47 abbrev OrderSymbolRelationGraded (I : RightIdeal (WeylAlg k n)) :=
48   DirectSum ℕ (fun N =>
49     MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N ⧸
50       orderSymbolRelation k I N)
51 
52 /-- The degreewise filtered-quotient equivalences assemble without adding any
53 unproved multiplication or support assertion. -/
54 def quotientOrderAssociatedGradedEquivSymbols
55     (I : RightIdeal (WeylAlg k n)) :
56     QuotientOrderAssociatedGraded k I ≃ₗ[k] OrderSymbolRelationGraded k I :=
57   DFinsupp.mapRange.linearEquiv fun N =>
58     quotientOrderGradedPieceEquivSymbols k I N
59 
60 /-- Right multiplication by a homogeneous symbol, with the resulting degree
61 recorded in the codomain. -/
62 def homogeneousRightMul {N M : ℕ}
63     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M) :
64     MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N →ₗ[k]
65       MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) (N + M) where
66   toFun Q := ⟨(Q : SymbolRing k n) * P, Q.property.mul P.property⟩
67   map_add' Q R := by
68     apply Subtype.ext
69     exact add_mul (Q : SymbolRing k n) R P
70   map_smul' c Q := by
71     apply Subtype.ext
72     exact Algebra.smul_mul_assoc c (Q : SymbolRing k n) P
73 
74 @[simp] theorem coe_homogeneousRightMul {N M : ℕ}
75     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M)
76     (Q : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N) :
77     (homogeneousRightMul k P Q : SymbolRing k n) = Q * P :=
78   rfl

80 /-- The exact degree-`N` relation is closed under multiplication by every
81 degree-`M` homogeneous symbol.  This is the load-bearing multiplication
82 compatibility: it uses right-ideal closure before passing to principal
83 symbols. -/
84 theorem homogeneousRightMul_mem_orderSymbolRelation {N M : ℕ}
85     (I : RightIdeal (WeylAlg k n))
86     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M)
87     (Q : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N)
88     (hQ : Q ∈ orderSymbolRelation k I N) :
89     homogeneousRightMul k P Q ∈ orderSymbolRelation k I (N + M) := by
90   obtain ⟨z, hzN, hzI, hQz⟩ :=
91     (mem_orderSymbolRelation_iff k I N Q).mp hQ
92   obtain ⟨y, hy⟩ :=
93     principalComponentOnPiece_surjective k (@orderWeight n) P
94   let zy : WeylAlg k n := z * (y : WeylAlg k n)
95   have hzyPiece : zy ∈ orderPiece k n (N + M) :=
96     mul_mem_orderPiece k hzN y.property
97   have hzyI : zy ∈ I := by
98     exact I.smul_mem (MulOpposite.op (y : WeylAlg k n)) hzI
99   apply (mem_orderSymbolRelation_iff k I (N + M)
100     (homogeneousRightMul k P Q)).mpr
101   refine ⟨zy, hzyPiece, hzyI, ?_⟩
102   apply Subtype.ext
103   change (Q : SymbolRing k n) * P =
104     presentedPrincipalComponent k (@orderWeight n) (N + M) zy
105   rw [presentedPrincipalComponent_mul_order k hzN y.property]
106   change (Q : SymbolRing k n) * P =
107     presentedPrincipalComponent k (@orderWeight n) N z *
108       (principalComponentOnPiece k (@orderWeight n) M y : SymbolRing k n)
109   have hQz' : (Q : SymbolRing k n) =
110       presentedPrincipalComponent k (@orderWeight n) N z :=
111     congrArg Subtype.val hQz
112   have hy' :
113       (principalComponentOnPiece k (@orderWeight n) M y : SymbolRing k n) = P :=
114     congrArg Subtype.val hy
115   rw [hQz', hy']

117 /-- A homogeneous symbol acts on the corresponding quotient of homogeneous
118 symbols because exact relation pieces are multiplicatively stable. -/
119 def homogeneousSymbolAction {N M : ℕ}
120     (I : RightIdeal (WeylAlg k n))
121     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M) :
122     (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N ⧸
123         orderSymbolRelation k I N) →ₗ[k]
124       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) (N + M) ⧸
125         orderSymbolRelation k I (N + M)) :=
126   (orderSymbolRelation k I N).mapQ (orderSymbolRelation k I (N + M))
127     (homogeneousRightMul k P)
128     (homogeneousRightMul_mem_orderSymbolRelation k I P)
129 
130 @[simp] theorem homogeneousSymbolAction_mk {N M : ℕ}
131     (I : RightIdeal (WeylAlg k n))
132     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M)
133     (Q : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N) :
134     homogeneousSymbolAction k I P (Submodule.Quotient.mk Q) =
135       Submodule.Quotient.mk (homogeneousRightMul k P Q) :=
136   rfl

138 /-- Transport of homogeneous symbol multiplication to the actual
139 associated-graded pieces of the filtered quotient. -/
140 def quotientOrderHomogeneousAction {N M : ℕ}
141     (I : RightIdeal (WeylAlg k n))
142     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M) :
143     QuotientOrderGradedPiece k I N →ₗ[k]
144       QuotientOrderGradedPiece k I (N + M) :=
145   (quotientOrderGradedPieceEquivSymbols k I (N + M)).symm.toLinearMap.comp
146     ((homogeneousSymbolAction k I P).comp
147       (quotientOrderGradedPieceEquivSymbols k I N).toLinearMap)
148 
149 /-- The transported action commutes exactly with the degreewise symbol
150 equivalences. -/
151 theorem quotientOrderHomogeneousAction_compatibility {N M : ℕ}
152     (I : RightIdeal (WeylAlg k n))
153     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M)
154     (q : QuotientOrderGradedPiece k I N) :
155     quotientOrderGradedPieceEquivSymbols k I (N + M)
156         (quotientOrderHomogeneousAction k I P q) =
157       homogeneousSymbolAction k I P
158         (quotientOrderGradedPieceEquivSymbols k I N q) := by
159   simp [quotientOrderHomogeneousAction]
160 
161 @[simp] theorem quotientOrderGradedPieceEquivSymbols_mk {N : ℕ}
162     (I : RightIdeal (WeylAlg k n)) (z : orderPiece k n N) :
163     quotientOrderGradedPieceEquivSymbols k I N
164         (orderPieceToQuotientGraded k I N z) =
165       Submodule.Quotient.mk
166         (principalComponentOnPiece k (@orderWeight n) N z) := by
167   rw [quotientOrderGradedPieceEquivSymbols, LinearEquiv.trans_apply]
168   rw [← show quotientOrderGradedPieceEquivRelation k I N
169       (Submodule.Quotient.mk z) = orderPieceToQuotientGraded k I N z from rfl]
170   rw [LinearEquiv.symm_apply_apply]
171   rfl

173 /-- On classes of filtered representatives, the transported homogeneous
174 operator is exactly multiplication in the Weyl algebra followed by passage to
175 the associated graded quotient. -/
176 theorem quotientOrderHomogeneousAction_mk_mul {N M : ℕ}
177     (I : RightIdeal (WeylAlg k n))
178     (z : orderPiece k n N) (y : orderPiece k n M) :
179     quotientOrderHomogeneousAction k I
180         (principalComponentOnPiece k (@orderWeight n) M y)
181         (orderPieceToQuotientGraded k I N z) =
182       orderPieceToQuotientGraded k I (N + M)
183         ⟨(z : WeylAlg k n) * y,
184           mul_mem_orderPiece k z.property y.property⟩ := by
185   apply (quotientOrderGradedPieceEquivSymbols k I (N + M)).injective
186   rw [quotientOrderHomogeneousAction_compatibility]
187   rw [quotientOrderGradedPieceEquivSymbols_mk,
188     quotientOrderGradedPieceEquivSymbols_mk,
189     homogeneousSymbolAction_mk]
190   congr 1
191   apply Subtype.ext
192   exact (presentedPrincipalComponent_mul_order k z.property y.property).symm

194 /-!
195 The remaining prerequisite for a global `SymbolRing`-module equivalence is
196 the homogeneous-piece equality
197 
198 `P ∈ orderSymbolRelation k I N ↔
199   (P : SymbolRing k n) ∈ orderInitialIdeal k I`.
200 
201 The forward implication is already proved in `FilteredQuotient`.  The reverse
202 implication requires decomposing arbitrary ideal multipliers into weighted
203 homogeneous components and combining the action above degree by degree.  No
204 support or annihilator statement is made before that equality is available.
205 -/
