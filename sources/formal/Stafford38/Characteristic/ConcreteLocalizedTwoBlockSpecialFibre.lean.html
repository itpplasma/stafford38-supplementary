9 /-!
10 # The concrete localized two-block special fibre
11 
12 For the localized opposite order-Rees two-jet, the principal specialization
13 kernel theorem applies to the actual `RightReesArtinianAdapter` data.  After
14 the two-block quotient, the kernel is therefore the two-sided ideal generated
15 by the parameter class.  Quotienting by that ideal gives a ring equivalence
16 with the commutative doubled-power quotient.
17 
18 No Artinianness, module action, adapted basis, or equation-(3.3) input is used.
19 -/

57 def concreteLocalizedTwoBlockParameter (q : ℕ) :
58     LocalizedTwoBlockRing D S q :=
59   (localizedTwoBlockIdeal D S q).ringCon.mk'

68 /-- The actual order-Rees localized principal-kernel theorem, instantiated in
69 the notation consumed by the abstract two-block descent. -/
70 theorem concreteLocalizedTwoBlockSpecialization_eq_zero_iff_exists_parameter_mul
71     (q : ℕ) (z : LocalizedTwoBlockRing D S q) :
72     localizedTwoBlockSpecialization D S q z = 0 ↔
73       ∃ y : LocalizedTwoBlockRing D S q,
74         concreteLocalizedTwoBlockParameter k I S q * y = z := by
75   exact localizedTwoBlockSpecialization_eq_zero_iff_exists_parameter_mul
76     D S q
77       (localizedOrderReesTwoJetSpecialization_eq_zero_iff_exists_parameter_mul
78         k I S) z

80 /-- In the concrete two-jet, the specialization kernel is exactly the
81 two-sided ideal generated by the quotient parameter. -/
82 theorem concreteLocalizedTwoBlockParameterIdeal_eq_ker (q : ℕ) :
83     concreteLocalizedTwoBlockParameterIdeal k I S q =
84       TwoSidedIdeal.ker (localizedTwoBlockSpecialization D S q) := by
85   apply le_antisymm
86   · intro z hz
87     change z ∈ TwoSidedIdeal.span
88       {concreteLocalizedTwoBlockParameter k I S q} at hz
89     rw [TwoSidedIdeal.mem_span_iff] at hz
90     exact hz (TwoSidedIdeal.ker (localizedTwoBlockSpecialization D S q)) (by
91       intro c hc
92       rcases hc with rfl
93       apply (TwoSidedIdeal.mem_ker
94         (localizedTwoBlockSpecialization D S q)).2
95       change localizedTwoBlockSpecialization D S q
96         (concreteLocalizedTwoBlockParameter k I S q) = 0
97       exact
98         (concreteLocalizedTwoBlockSpecialization_eq_zero_iff_exists_parameter_mul
99           k I S q (concreteLocalizedTwoBlockParameter k I S q)).2
100             ⟨1, mul_one _⟩)
101   · intro z hz
102     have hz0 : localizedTwoBlockSpecialization D S q z = 0 :=
103       (TwoSidedIdeal.mem_ker
104         (localizedTwoBlockSpecialization D S q)).1 hz
105     obtain ⟨y, rfl⟩ :=
106       (concreteLocalizedTwoBlockSpecialization_eq_zero_iff_exists_parameter_mul
107         k I S q z).mp hz0
108     exact (concreteLocalizedTwoBlockParameterIdeal k I S q).mul_mem_right
109       _ _ (TwoSidedIdeal.subset_span (Set.mem_singleton _))

116 /-- Specialization descended to the quotient by the parameter class. -/
117 def concreteLocalizedTwoBlockSpecialFibreHom (q : ℕ) :
118     ConcreteLocalizedTwoBlockSpecialFibreRing k I S q →+*
119       (Localization S) ⧸ localizedDoubledPower S q :=
120   quotientRingHom (concreteLocalizedTwoBlockParameterIdeal k I S q)
121     (localizedTwoBlockSpecialization D S q) (by
122       intro z hz
123       exact (TwoSidedIdeal.mem_ker
124         (localizedTwoBlockSpecialization D S q)).1
125           ((concreteLocalizedTwoBlockParameterIdeal_eq_ker k I S q) ▸ hz))

135 theorem concreteLocalizedTwoBlockSpecialFibreHom_injective (q : ℕ) :
136     Function.Injective (concreteLocalizedTwoBlockSpecialFibreHom k I S q) := by
137   intro x y hxy
138   induction x using Quotient.inductionOn' with
139   | _ a =>
140     induction y using Quotient.inductionOn' with
141     | _ b =>
142       apply Quotient.sound'
143       apply (concreteLocalizedTwoBlockParameterIdeal k I S q).rel_iff a b |>.2
144       rw [concreteLocalizedTwoBlockParameterIdeal_eq_ker]
145       apply (TwoSidedIdeal.mem_ker
146         (localizedTwoBlockSpecialization D S q)).2
147       rw [map_sub, sub_eq_zero]
148       exact hxy
