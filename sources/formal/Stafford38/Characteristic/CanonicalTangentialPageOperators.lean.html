10 # Tangential filtered operators on the canonical quotient
11 
12 Right multiplication by every tangential Weyl generator gives an actual
13 filtered operator on the canonical quotient.  The distinguished coordinate
14 commutes with these operators, so they induce operators on every page.
15 -/

48 theorem oldGenerator_mem_orderPiece
49     (n : ℕ) (i : Fin n ⊕ Fin n) :
50     oldGenerator k n i ∈ orderPiece k (n + 1)
51       (match i with | .inl _ => 0 | .inr _ => 1) := by
52   cases i with
53   | inl i =>
54       rw [oldGenerator, orderPiece, mem_presentedWeightPiece,
55         presentedNormalFormLinearEquiv_generator]
56       intro m hm
57       simp only [MvPolynomial.coeff_X'] at hm
58       split at hm
59       · subst m
60         simp [oldIndex, monomialWeight, orderWeight, fibreWeight]
61       · contradiction
62   | inr i =>
63       rw [oldGenerator, orderPiece, mem_presentedWeightPiece,
64         presentedNormalFormLinearEquiv_generator]
65       intro m hm
66       simp only [MvPolynomial.coeff_X'] at hm
67       split at hm
68       · subst m
69         simp [oldIndex, monomialWeight, orderWeight, fibreWeight]
70       · contradiction

72 theorem rightMul_commute
73     (n N : ℕ) (d : PresentedWeyl k (n + 1))
74     (a : PresentedWeyl k (n + 1))
75     (ha : a * presentedCoordinate k n = presentedCoordinate k n * a) :
76     (rightMulLinearMap k (CI k n N d) (presentedCoordinate k n)).comp
77         (rightMulLinearMap k (CI k n N d) a) =
78       (rightMulLinearMap k (CI k n N d) a).comp
79         (rightMulLinearMap k (CI k n N d) (presentedCoordinate k n)) := by
80   apply LinearMap.ext
81   intro q
82   obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective
83     (rightIdealKSubmodule k (CI k n N d)) q
84   rw [LinearMap.comp_apply, LinearMap.comp_apply,
85     rightMulLinearMap_mk, rightMulLinearMap_mk, rightMulLinearMap_mk,
86     rightMulLinearMap_mk]
87   apply congrArg Submodule.Quotient.mk
88   simp only [mul_assoc]
89   rw [ha]

91 private theorem rightMul_shift
92     (n N : ℕ) (d : PresentedWeyl k (n + 1))
93     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
94     {e : ℕ}
95     (a : PresentedWeyl k (n + 1)) (ha : a ∈ orderPiece k (n + 1) e)
96     {p : ℤ} {q : FilteredRightQuotient k (CI k n N d)}
97     (hp : p ≤ 0) (hq : q ∈
98       canonicalOrderFiltration k (CI k n N d) p) :
99     rightMulLinearMap k (CI k n N d) a q ∈
100       canonicalOrderFiltration k (CI k n N d) (p - (e : ℤ)) := by
101   rw [canonicalOrderFiltration_eq_of_nonpos k _ hp] at hq
102   have htarget : p - (e : ℤ) ≤ 0 := by omega
103   rw [canonicalOrderFiltration_eq_of_nonpos k _ htarget]
104   have hidx : (-(p - (e : ℤ))).toNat = (-p).toNat + e := by omega
105   rw [hidx]
106   rcases Submodule.mem_map.mp hq with ⟨z, hz, rfl⟩
107   change rightMulLinearMap k (CI k n N d) a
108     (Submodule.Quotient.mk z) ∈ _
109   rw [rightMulLinearMap_mk]
110   apply Submodule.mem_map.mpr
111   refine ⟨z * a, ?_, rfl⟩
112   exact mul_mem_orderPiece k hz ha

114 def tangentialPageOperator
115     (n N : ℕ) (d : PresentedWeyl k (n + 1))
116     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
117     (a : PresentedWeyl k (n + 1)) (e : ℕ)
118     (ha : a ∈ orderPiece k (n + 1) e)
119     (hax : a * presentedCoordinate k n = presentedCoordinate k n * a) :
120     PageOperator k n N d hd (e : ℤ) where
121   g := rightMulLinearMap k (CI k n N d) a
122   commute := rightMul_commute k n N d a hax
123   shift := by
124     intro p q hq
125     by_cases hp : p ≤ 0
126     · exact rightMul_shift k n N d hd (a := a) ha hp hq
127     · change q ∈ canonicalOrderFiltration k (CI k n N d) p at hq
128       rw [canonicalOrderFiltration_eq_bot_of_pos k _ (lt_of_not_ge hp)] at hq
129       subst q
130       rw [map_zero]
131       exact Submodule.zero_mem _

133 def tangential_coordinate_pageOperator
134     (n N : ℕ) (d : PresentedWeyl k (n + 1))
135     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
136     (i : Fin n) :
137     PageOperator k n N d hd 0 := by
138   apply tangentialPageOperator k n N d hd

166   simp

168 private theorem rightMul_commutator_apply
169     (n N : ℕ) (d : PresentedWeyl k (n + 1))
170     (a b : PresentedWeyl k (n + 1)) (q : CanonicalQuotient k n N d) :
171     rightMulLinearMap k (CI k n N d) a
172         (rightMulLinearMap k (CI k n N d) b q) -
173       rightMulLinearMap k (CI k n N d) b
174         (rightMulLinearMap k (CI k n N d) a q) =
175       rightMulLinearMap k (CI k n N d) (b * a - a * b) q := by
176   obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective
177     (rightIdealKSubmodule k (CI k n N d)) q
178   change Submodule.Quotient.mk (z * b * a) -
179       Submodule.Quotient.mk (z * a * b) =
180     Submodule.Quotient.mk (z * (b * a - a * b))
181   rw [mul_sub]
182   rw [← Submodule.Quotient.mk_sub]
183   congr 1
184   noncomm_ring

186 private theorem rightMul_lower_of_orderPiece
187     (n N : ℕ) (d : PresentedWeyl k (n + 1))
188     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
189     {e : ℕ} (a : PresentedWeyl k (n + 1))
190     (ha : a ∈ orderPiece k (n + 1) e) {p : ℤ}
191     {z : CanonicalQuotient k n N d}
192     (hz : z ∈ (K k n N d hd).G p) :
193     rightMulLinearMap k (CI k n N d) a z ∈
194       (K k n N d hd).G (p - (e : ℤ)) := by
195   change z ∈ canonicalOrderFiltration k (CI k n N d) p at hz
196   change rightMulLinearMap k (CI k n N d) a z ∈
197     canonicalOrderFiltration k (CI k n N d) (p - (e : ℤ))
198   by_cases hp : p ≤ 0
199   · exact rightMul_shift k n N d hd (a := a) ha hp hz
200   · rw [canonicalOrderFiltration_eq_bot_of_pos k _ (lt_of_not_ge hp)] at hz
201     have hz0 : z = 0 := by simpa using hz
202     subst z
203     rw [map_zero]
204     exact Submodule.zero_mem _

206 theorem tangential_commutator_lower
207     (n N : ℕ) (d : PresentedWeyl k (n + 1))
208     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
209     (i j : Fin n ⊕ Fin n) (p : ℤ)
210     (z : CanonicalQuotient k n N d)
211     (hz : z ∈ (K k n N d hd).G p) :
212     (rightMulLinearMap k (CI k n N d) (oldGenerator k n i))
213         (rightMulLinearMap k (CI k n N d) (oldGenerator k n j) z) -
214       (rightMulLinearMap k (CI k n N d) (oldGenerator k n j))
215         (rightMulLinearMap k (CI k n N d) (oldGenerator k n i) z) ∈
216       (K k n N d hd).G
217         (p - (tangentialDegree i + tangentialDegree j : ℤ) + 1) := by
218   rcases i with i | i <;> rcases j with j | j
219   · have hxi := oldGenerator_mem_orderPiece k n (.inl i)
220     have hxj := oldGenerator_mem_orderPiece k n (.inl j)
221     have hc := commutator_mem_orderPiece_pred k hxi hxj
222     have hzero : oldGenerator k n (.inl j) * oldGenerator k n (.inl i) -
223         oldGenerator k n (.inl i) * oldGenerator k n (.inl j) = 0 := by
224       calc
225         _ = -(Stafford.commutator (oldGenerator k n (.inl i))
226           (oldGenerator k n (.inl j))) := by simp [Stafford.commutator]
227         _ = 0 := by rw [oldGenerator_commutator]; simp [Matrix.J]
228     rw [rightMul_commutator_apply, hzero]
229     have hr : rightMulLinearMap k (CI k n N d)
230         (0 : PresentedWeyl k (n + 1)) z = 0 := rightMul_zero k n N d z
231     rw [hr]
232     exact Submodule.zero_mem _
233   · have hxi := oldGenerator_mem_orderPiece k n (.inl i)
234     have hxj := oldGenerator_mem_orderPiece k n (.inr j)
235     have hc := commutator_mem_orderPiece_pred k hxi hxj
236     have hc' : oldGenerator k n (.inr j) * oldGenerator k n (.inl i) -
237         oldGenerator k n (.inl i) * oldGenerator k n (.inr j) ∈
238         orderPiece k (n + 1) 0 := by
239       simpa [Stafford.commutator] using (show
240         -(Stafford.commutator (oldGenerator k n (.inl i))
241           (oldGenerator k n (.inr j))) ∈ orderPiece k (n + 1) 0 from
242         (orderPiece k (n + 1) 0).neg_mem hc)
243     rw [rightMul_commutator_apply]
244     simpa [tangentialDegree] using
245       (rightMul_lower_of_orderPiece k n N d hd _ hc' hz)
246   · have hxi := oldGenerator_mem_orderPiece k n (.inr i)
247     have hxj := oldGenerator_mem_orderPiece k n (.inl j)
248     have hc := commutator_mem_orderPiece_pred k hxi hxj
249     have hc' : oldGenerator k n (.inl j) * oldGenerator k n (.inr i) -
250         oldGenerator k n (.inr i) * oldGenerator k n (.inl j) ∈
251         orderPiece k (n + 1) 0 := by
252       simpa [Stafford.commutator] using (show
253         -(Stafford.commutator (oldGenerator k n (.inr i))
254           (oldGenerator k n (.inl j))) ∈ orderPiece k (n + 1) 0 from
255         (orderPiece k (n + 1) 0).neg_mem hc)
256     rw [rightMul_commutator_apply]
257     simpa [tangentialDegree] using
258       (rightMul_lower_of_orderPiece k n N d hd _ hc' hz)
259   · have hxi := oldGenerator_mem_orderPiece k n (.inr i)
260     have hxj := oldGenerator_mem_orderPiece k n (.inr j)
261     have hc := commutator_mem_orderPiece_pred k hxi hxj
262     rw [rightMul_commutator_apply]
263     have hc' := (orderPiece k (n + 1) 1).neg_mem hc
264     have hc'' : oldGenerator k n (.inr j) * oldGenerator k n (.inr i) -
265         oldGenerator k n (.inr i) * oldGenerator k n (.inr j) ∈
266         orderPiece k (n + 1) 1 := by
267       convert hc' using 1 <;> simp [Stafford.commutator]
268     convert (rightMul_lower_of_orderPiece k n N d hd _ hc'' hz) using 1 <;>
269       norm_num [tangentialDegree] <;> ring
