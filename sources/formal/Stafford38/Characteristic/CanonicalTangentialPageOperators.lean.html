11 /-!
12 # Tangential filtered operators on the canonical quotient
13 
14 Right multiplication by every tangential Weyl generator gives an actual
15 filtered operator on the canonical quotient.  The distinguished coordinate
16 commutes with these operators, so they induce operators on every page.
17 -/

50 theorem oldGenerator_mem_orderPiece
51     (n : ℕ) (i : Fin n ⊕ Fin n) :
52     oldGenerator k n i ∈ orderPiece k (n + 1)
53       (match i with | .inl _ => 0 | .inr _ => 1) := by
54   cases i with
55   | inl i =>
56       rw [oldGenerator, orderPiece, mem_presentedWeightPiece,
57         presentedNormalFormLinearEquiv_generator]
58       intro m hm
59       simp only [MvPolynomial.coeff_X'] at hm
60       split at hm
61       · subst m
62         simp [Stafford38FixedSourceChallenge.oldIndex, Finsupp.weight_apply, orderWeight, fibreWeight]
63       · contradiction
64   | inr i =>
65       rw [oldGenerator, orderPiece, mem_presentedWeightPiece,
66         presentedNormalFormLinearEquiv_generator]
67       intro m hm
68       simp only [MvPolynomial.coeff_X'] at hm
69       split at hm
70       · subst m
71         simp [Stafford38FixedSourceChallenge.oldIndex, Finsupp.weight_apply, orderWeight, fibreWeight]
72       · contradiction

74 theorem rightMul_commute
75     (n N : ℕ) (d : WeylAlg k (n + 1))
76     (a : WeylAlg k (n + 1))
77     (ha : a * presentedCoordinate k n = presentedCoordinate k n * a) :
78     (rightMulLinearMap k (CI k n N d) (presentedCoordinate k n)).comp
79         (rightMulLinearMap k (CI k n N d) a) =
80       (rightMulLinearMap k (CI k n N d) a).comp
81         (rightMulLinearMap k (CI k n N d) (presentedCoordinate k n)) := by
82   apply LinearMap.ext
83   intro q
84   obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective
85     (rightIdealKSubmodule k (CI k n N d)) q
86   rw [LinearMap.comp_apply, LinearMap.comp_apply,
87     rightMulLinearMap_mk, rightMulLinearMap_mk, rightMulLinearMap_mk,
88     rightMulLinearMap_mk]
89   apply congrArg Submodule.Quotient.mk
90   simp only [mul_assoc]
91   rw [ha]

93 private theorem rightMul_shift
94     (n N : ℕ) (d : WeylAlg k (n + 1))
95     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
96     {e : ℕ}
97     (a : WeylAlg k (n + 1)) (ha : a ∈ orderPiece k (n + 1) e)
98     {p : ℤ} {q : FilteredRightQuotient k (CI k n N d)}
99     (hp : p ≤ 0) (hq : q ∈
100       canonicalOrderFiltration k (CI k n N d) p) :
101     rightMulLinearMap k (CI k n N d) a q ∈
102       canonicalOrderFiltration k (CI k n N d) (p - (e : ℤ)) := by
103   rw [canonicalOrderFiltration_eq_of_nonpos k _ hp] at hq
104   have htarget : p - (e : ℤ) ≤ 0 := by omega
105   rw [canonicalOrderFiltration_eq_of_nonpos k _ htarget]
106   have hidx : (-(p - (e : ℤ))).toNat = (-p).toNat + e := by omega
107   rw [hidx]
108   rcases Submodule.mem_map.mp hq with ⟨z, hz, rfl⟩
109   change rightMulLinearMap k (CI k n N d) a
110     (Submodule.Quotient.mk z) ∈ _
111   rw [rightMulLinearMap_mk]
112   apply Submodule.mem_map.mpr
113   refine ⟨z * a, ?_, rfl⟩
114   exact mul_mem_orderPiece k hz ha

116 def tangentialPageOperator
117     (n N : ℕ) (d : WeylAlg k (n + 1))
118     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
119     (a : WeylAlg k (n + 1)) (e : ℕ)
120     (ha : a ∈ orderPiece k (n + 1) e)
121     (hax : a * presentedCoordinate k n = presentedCoordinate k n * a) :
122     PageOperator k n N d hd (e : ℤ) where
123   g := rightMulLinearMap k (CI k n N d) a
124   commute := rightMul_commute k n N d a hax
125   shift := by
126     intro p q hq
127     by_cases hp : p ≤ 0
128     · exact rightMul_shift k n N d hd (a := a) ha hp hq
129     · change q ∈ canonicalOrderFiltration k (CI k n N d) p at hq
130       rw [canonicalOrderFiltration_eq_bot_of_pos k _ (lt_of_not_ge hp)] at hq
131       subst q
132       rw [map_zero]
133       exact Submodule.zero_mem _

137     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
138     (i : Fin n) :
139     PageOperator k n N d hd 0 := by

163     (z : CanonicalQuotient k n N d) :
164     rightMulLinearMap k (CI k n N d) 0 z = 0 := by

168   simp

170 private theorem rightMul_commutator_apply
171     (n N : ℕ) (d : WeylAlg k (n + 1))
172     (a b : WeylAlg k (n + 1)) (q : CanonicalQuotient k n N d) :
173     rightMulLinearMap k (CI k n N d) a
174         (rightMulLinearMap k (CI k n N d) b q) -
175       rightMulLinearMap k (CI k n N d) b
176         (rightMulLinearMap k (CI k n N d) a q) =
177       rightMulLinearMap k (CI k n N d) (b * a - a * b) q := by
178   obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective
179     (rightIdealKSubmodule k (CI k n N d)) q
180   change Submodule.Quotient.mk (z * b * a) -
181       Submodule.Quotient.mk (z * a * b) =
182     Submodule.Quotient.mk (z * (b * a - a * b))
183   rw [mul_sub]
184   rw [← Submodule.Quotient.mk_sub]
185   congr 1
186   noncomm_ring

188 private theorem rightMul_lower_of_orderPiece
189     (n N : ℕ) (d : WeylAlg k (n + 1))
190     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
191     {e : ℕ} (a : WeylAlg k (n + 1))
192     (ha : a ∈ orderPiece k (n + 1) e) {p : ℤ}
193     {z : CanonicalQuotient k n N d}
194     (hz : z ∈ (K k n N d hd).G p) :
195     rightMulLinearMap k (CI k n N d) a z ∈
196       (K k n N d hd).G (p - (e : ℤ)) := by
197   change z ∈ canonicalOrderFiltration k (CI k n N d) p at hz
198   change rightMulLinearMap k (CI k n N d) a z ∈
199     canonicalOrderFiltration k (CI k n N d) (p - (e : ℤ))
200   by_cases hp : p ≤ 0
201   · exact rightMul_shift k n N d hd (a := a) ha hp hz
202   · rw [canonicalOrderFiltration_eq_bot_of_pos k _ (lt_of_not_ge hp)] at hz
203     have hz0 : z = 0 := by simpa using hz
204     subst z
205     rw [map_zero]
206     exact Submodule.zero_mem _

208 theorem tangential_commutator_lower
209     (n N : ℕ) (d : WeylAlg k (n + 1))
210     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
211     (i j : Fin n ⊕ Fin n) (p : ℤ)
212     (z : CanonicalQuotient k n N d)
213     (hz : z ∈ (K k n N d hd).G p) :
214     (rightMulLinearMap k (CI k n N d) (oldGenerator k n i))
215         (rightMulLinearMap k (CI k n N d) (oldGenerator k n j) z) -
216       (rightMulLinearMap k (CI k n N d) (oldGenerator k n j))
217         (rightMulLinearMap k (CI k n N d) (oldGenerator k n i) z) ∈
218       (K k n N d hd).G
219         (p - (tangentialDegree i + tangentialDegree j : ℤ) + 1) := by
220   rcases i with i | i <;> rcases j with j | j
221   · have hxi := oldGenerator_mem_orderPiece k n (.inl i)
222     have hxj := oldGenerator_mem_orderPiece k n (.inl j)
223     have hc := commutator_mem_orderPiece_pred k hxi hxj
224     have hzero : oldGenerator k n (.inl j) * oldGenerator k n (.inl i) -
225         oldGenerator k n (.inl i) * oldGenerator k n (.inl j) = 0 := by
226       calc
227         _ = -(AlgebraicAnalysis.ringCommutator (oldGenerator k n (.inl i))
228           (oldGenerator k n (.inl j))) := by simp [AlgebraicAnalysis.ringCommutator]
229         _ = 0 := by rw [oldGenerator_commutator]; simp [Matrix.J]
230     rw [rightMul_commutator_apply, hzero]
231     have hr : rightMulLinearMap k (CI k n N d)
232         (0 : WeylAlg k (n + 1)) z = 0 := rightMul_zero k n N d z
233     rw [hr]
234     exact Submodule.zero_mem _
235   · have hxi := oldGenerator_mem_orderPiece k n (.inl i)
236     have hxj := oldGenerator_mem_orderPiece k n (.inr j)
237     have hc := commutator_mem_orderPiece_pred k hxi hxj
238     have hc' : oldGenerator k n (.inr j) * oldGenerator k n (.inl i) -
239         oldGenerator k n (.inl i) * oldGenerator k n (.inr j) ∈
240         orderPiece k (n + 1) 0 := by
241       simpa [AlgebraicAnalysis.ringCommutator] using (show
242         -(AlgebraicAnalysis.ringCommutator (oldGenerator k n (.inl i))
243           (oldGenerator k n (.inr j))) ∈ orderPiece k (n + 1) 0 from
244         (orderPiece k (n + 1) 0).neg_mem hc)
245     rw [rightMul_commutator_apply]
246     simpa [tangentialDegree] using
247       (rightMul_lower_of_orderPiece k n N d hd _ hc' hz)
248   · have hxi := oldGenerator_mem_orderPiece k n (.inr i)
249     have hxj := oldGenerator_mem_orderPiece k n (.inl j)
250     have hc := commutator_mem_orderPiece_pred k hxi hxj
251     have hc' : oldGenerator k n (.inl j) * oldGenerator k n (.inr i) -
252         oldGenerator k n (.inr i) * oldGenerator k n (.inl j) ∈
253         orderPiece k (n + 1) 0 := by
254       simpa [AlgebraicAnalysis.ringCommutator] using (show
255         -(AlgebraicAnalysis.ringCommutator (oldGenerator k n (.inr i))
256           (oldGenerator k n (.inl j))) ∈ orderPiece k (n + 1) 0 from
257         (orderPiece k (n + 1) 0).neg_mem hc)
258     rw [rightMul_commutator_apply]
259     simpa [tangentialDegree] using
260       (rightMul_lower_of_orderPiece k n N d hd _ hc' hz)
261   · have hxi := oldGenerator_mem_orderPiece k n (.inr i)
262     have hxj := oldGenerator_mem_orderPiece k n (.inr j)
263     have hc := commutator_mem_orderPiece_pred k hxi hxj
264     rw [rightMul_commutator_apply]
265     have hc' := (orderPiece k (n + 1) 1).neg_mem hc
266     have hc'' : oldGenerator k n (.inr j) * oldGenerator k n (.inr i) -
267         oldGenerator k n (.inr i) * oldGenerator k n (.inr j) ∈
268         orderPiece k (n + 1) 1 := by
269       convert hc' using 1 <;> simp [AlgebraicAnalysis.ringCommutator]
270     convert (rightMul_lower_of_orderPiece k n N d hd _ hc'' hz) using 1 <;>
271       norm_num [tangentialDegree] <;> ring
