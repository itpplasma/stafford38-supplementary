29 /-- Componentwise membership in every minimal prime assembles to the exact
30 base-relative Poisson condition on the radical.  This is the final elementary
31 commutative-algebra step after the genuine Gabber integrability input. -/
32 theorem radical_isBaseRelativePoisson_of_minimalPrimes
33     (J : Ideal (SymbolRing k n))
34     (h : ∀ P ∈ J.minimalPrimes,
35       ∀ f : MvPolynomial (Fin n) k, baseLift f ∈ J.radical →
36         ∀ g ∈ J.radical, poissonBracket (baseLift f) g ∈ P) :
37     IsBaseRelativePoisson J.radical := by
38   intro f hf g hg
39   rw [← J.sInf_minimalPrimes]
40   exact Ideal.mem_sInf.mpr fun P hP => h P hP f hf g hg
41 
42 /-- Componentwise involutivity is the natural sufficient hypothesis for the
43 minimal-prime assembly above.  The only commutative-algebra input is that the
44 radical is contained in every prime over the original ideal. -/
45 theorem radical_isBaseRelativePoisson_of_minimalPrimes_isInvolutive
46     (J : Ideal (SymbolRing k n))
47     (h : ∀ P ∈ J.minimalPrimes, IsInvolutive P) :
48     IsBaseRelativePoisson J.radical := by
49   apply radical_isBaseRelativePoisson_of_minimalPrimes J
50   intro P hP f hf g hg
51   have hprime : P.IsPrime := hP.1.1
52   have hradical : J.radical ≤ P := hprime.radical_le_iff.mpr hP.1.2
53   exact h P hP (baseLift f) (hradical hf) g (hradical hg)
