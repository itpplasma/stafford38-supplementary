7 /-!
8 # The specialization kernel of the order-Rees two-jet
9 
10 The kernel is computed before localization.  First, degreewise weighted
11 homogeneity shows that an order-Rees polynomial with zero principal-symbol
12 sum has every coefficient one filtration step below its declared degree.
13 Dividing the polynomial by the central variable `T` therefore stays inside
14 the order-Rees ring.  Passing through the quotient by `T²` gives the exact
15 elementwise principal-kernel statement for the two-jet.
16 
17 The factor is always written on the left.  Centrality of `T` is not used to
18 reverse any noncommutative product.
19 -/

42 /-- Weighted homogeneity prevents cancellation between distinct declared
43 Rees degrees in the principal-symbol sum. -/
44 private theorem principal_component_eq_zero_of_sum_eq_zero
45     (r : OrderReesRing (n := n) k)
46     (hr : orderReesSpecialization (n := n) k r = 0)
47     (N : ℕ) :
48     presentedPrincipalComponent k (@orderWeight n) N
49         ((r : Polynomial (PresentedWeyl k n)).coeff N) = 0 := by
50   classical
51   change orderPrincipalSum (n := n) k
52       (r : Polynomial (PresentedWeyl k n)) = 0 at hr
53   change (r : Polynomial (PresentedWeyl k n)).sum
54       (fun M z => presentedPrincipalComponent k (@orderWeight n) M z) = 0 at hr
55   have hcomponent := congrArg
56     (MvPolynomial.weightedHomogeneousComponent (@orderWeight n) N) hr
57   rw [map_zero] at hcomponent
58   simp only [Polynomial.sum, map_sum] at hcomponent
59   by_cases hN : N ∈ (r : Polynomial (PresentedWeyl k n)).support
60   · rw [Finset.sum_eq_single N] at hcomponent
61     · have hhom :
62           presentedPrincipalComponent k (@orderWeight n) N
63               ((r : Polynomial (PresentedWeyl k n)).coeff N) ∈
64             MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N :=
65         MvPolynomial.weightedHomogeneousComponent_mem
66           (@orderWeight n)
67           (presentedNormalFormLinearEquiv k n
68             ((r : Polynomial (PresentedWeyl k n)).coeff N)) N
69       rw [MvPolynomial.weightedHomogeneousComponent_of_mem hhom] at hcomponent
70       simpa using hcomponent
71     · intro M hM hMN
72       have hhom :
73           presentedPrincipalComponent k (@orderWeight n) M
74               ((r : Polynomial (PresentedWeyl k n)).coeff M) ∈
75             MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M :=
76         MvPolynomial.weightedHomogeneousComponent_mem
77           (@orderWeight n)
78           (presentedNormalFormLinearEquiv k n
79             ((r : Polynomial (PresentedWeyl k n)).coeff M)) M
80       rw [MvPolynomial.weightedHomogeneousComponent_of_mem hhom]
81       simp [Ne.symm hMN]
82     · intro hnot
83       exact (hnot hN).elim
84   · have hcoeff : (r : Polynomial (PresentedWeyl k n)).coeff N = 0 :=
85       Polynomial.notMem_support_iff.mp hN
86     rw [hcoeff, map_zero]

88 /-- The kernel of the unquotiented order-Rees specialization is the left
89 principal ideal generated by the central Rees parameter, stated
90 elementwise. -/
91 theorem orderReesSpecialization_eq_zero_iff_exists_parameter_mul
92     (r : OrderReesRing (n := n) k) :
93     orderReesSpecialization (n := n) k r = 0 ↔
94       ∃ s : OrderReesRing (n := n) k,
95         orderReesParameter (n := n) k * s = r := by
96   constructor
97   · intro hr
98     have hlower : ∀ N,
99         (r : Polynomial (PresentedWeyl k n)).coeff (N + 1) ∈
100           orderPiece k n N := by
101       intro N
102       have hprincipal := principal_component_eq_zero_of_sum_eq_zero
103         (n := n) k r hr (N + 1)
104       have hmem := r.property (N + 1)
105       exact (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
106         k (@orderWeight n) ((r : Polynomial (PresentedWeyl k n)).coeff (N + 1))
107           hmem).mp hprincipal
108     let s : OrderReesRing (n := n) k :=
109       ⟨Polynomial.divX (r : Polynomial (PresentedWeyl k n)), by
110         intro N
111         rw [Polynomial.coeff_divX]
112         exact hlower N⟩
113     refine ⟨s, ?_⟩
114     apply Subtype.ext
115     change Polynomial.X * Polynomial.divX
116         (r : Polynomial (PresentedWeyl k n)) =
117       (r : Polynomial (PresentedWeyl k n))
118     have hconstant : (r : Polynomial (PresentedWeyl k n)).coeff 0 = 0 := by
119       have hprincipal := principal_component_eq_zero_of_sum_eq_zero
120         (n := n) k r hr 0
121       have hmem := r.property 0
122       have hlower0 := (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
123         k (@orderWeight n) ((r : Polynomial (PresentedWeyl k n)).coeff 0)
124           hmem).mp hprincipal
125       simpa [presentedStrictLowerPiece, Submodule.mem_bot] using hlower0
126     simpa [hconstant] using
127       (Polynomial.X_mul_divX_add (r : Polynomial (PresentedWeyl k n)))
128   · rintro ⟨s, rfl⟩
129     rw [map_mul, orderReesSpecialization_parameter, zero_mul]

131 /-- Exact unlocalized specialization kernel for the concrete order-Rees
132 two-jet: every killed class is a left multiple of the square-zero parameter,
133 and every such multiple is killed. -/
134 theorem orderReesTwoJetSpecialization_eq_zero_iff_exists_parameter_mul
135     (q : OrderReesTwoJet (n := n) k) :
136     orderReesTwoJetSpecialization (n := n) k q = 0 ↔
137       ∃ b : OrderReesTwoJet (n := n) k,
138         orderReesTwoJetParameter (n := n) k * b = q := by
139   constructor
140   · intro hq
141     obtain ⟨r, rfl⟩ := Quotient.mk''_surjective q
142     change orderReesSpecialization (n := n) k r = 0 at hq
143     obtain ⟨s, hs⟩ :=
144       (orderReesSpecialization_eq_zero_iff_exists_parameter_mul
145         (n := n) k r).mp hq
146     refine ⟨orderReesTwoJetQuotient (n := n) k s, ?_⟩
147     rw [orderReesTwoJetParameter, ← map_mul, hs]
148     rfl
149   · rintro ⟨b, rfl⟩
150     rw [map_mul, orderReesTwoJetSpecialization_parameter, zero_mul]

152 /-- The same exactness statement in linear-kernel/range form. -/
153 theorem orderReesTwoJetSpecialization_ker_eq_range_parameterMul :
154     LinearMap.ker (orderReesTwoJetSpecialization (n := n) k).toLinearMap =
155       LinearMap.range
156         (LinearMap.mulLeft k (orderReesTwoJetParameter (n := n) k)) := by
157   ext q
158   rw [LinearMap.mem_ker, LinearMap.mem_range]
159   exact orderReesTwoJetSpecialization_eq_zero_iff_exists_parameter_mul
160     (n := n) k q
