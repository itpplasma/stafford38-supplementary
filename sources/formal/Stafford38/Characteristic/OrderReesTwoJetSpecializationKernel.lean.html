9 /-!
10 # The specialization kernel of the order-Rees two-jet
11 
12 The kernel is computed before localization.  First, degreewise weighted
13 homogeneity shows that an order-Rees polynomial with zero principal-symbol
14 sum has every coefficient one filtration step below its declared degree.
15 Dividing the polynomial by the central variable `T` therefore stays inside
16 the order-Rees ring.  Passing through the quotient by `T²` gives the exact
17 elementwise principal-kernel statement for the two-jet.
18 
19 The factor is always written on the left.  Centrality of `T` is not used to
20 reverse any noncommutative product.
21 -/

44 /-- Weighted homogeneity prevents cancellation between distinct declared
45 Rees degrees in the principal-symbol sum. -/
46 private theorem principal_component_eq_zero_of_sum_eq_zero
47     (r : OrderReesRing (n := n) k)
48     (hr : orderReesSpecialization (n := n) k r = 0)
49     (N : ℕ) :
50     presentedPrincipalComponent k (@orderWeight n) N
51         ((r : Polynomial (WeylAlg k n)).coeff N) = 0 := by
52   classical
53   change orderPrincipalSum (n := n) k
54       (r : Polynomial (WeylAlg k n)) = 0 at hr
55   change (r : Polynomial (WeylAlg k n)).sum
56       (fun M z => presentedPrincipalComponent k (@orderWeight n) M z) = 0 at hr
57   have hcomponent := congrArg
58     (MvPolynomial.weightedHomogeneousComponent (@orderWeight n) N) hr
59   rw [map_zero] at hcomponent
60   simp only [Polynomial.sum, map_sum] at hcomponent
61   by_cases hN : N ∈ (r : Polynomial (WeylAlg k n)).support
62   · rw [Finset.sum_eq_single N] at hcomponent
63     · have hhom :
64           presentedPrincipalComponent k (@orderWeight n) N
65               ((r : Polynomial (WeylAlg k n)).coeff N) ∈
66             MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N :=
67         MvPolynomial.weightedHomogeneousComponent_mem
68           (@orderWeight n)
69           (presentedNormalFormLinearEquiv k n
70             ((r : Polynomial (WeylAlg k n)).coeff N)) N
71       rw [MvPolynomial.weightedHomogeneousComponent_of_mem hhom] at hcomponent
72       simpa using hcomponent
73     · intro M hM hMN
74       have hhom :
75           presentedPrincipalComponent k (@orderWeight n) M
76               ((r : Polynomial (WeylAlg k n)).coeff M) ∈
77             MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M :=
78         MvPolynomial.weightedHomogeneousComponent_mem
79           (@orderWeight n)
80           (presentedNormalFormLinearEquiv k n
81             ((r : Polynomial (WeylAlg k n)).coeff M)) M
82       rw [MvPolynomial.weightedHomogeneousComponent_of_mem hhom]
83       simp [Ne.symm hMN]
84     · intro hnot
85       exact (hnot hN).elim
86   · have hcoeff : (r : Polynomial (WeylAlg k n)).coeff N = 0 :=
87       Polynomial.notMem_support_iff.mp hN
88     rw [hcoeff, map_zero]

90 /-- The kernel of the unquotiented order-Rees specialization is the left
91 principal ideal generated by the central Rees parameter, stated
92 elementwise. -/
93 theorem orderReesSpecialization_eq_zero_iff_exists_parameter_mul
94     (r : OrderReesRing (n := n) k) :
95     orderReesSpecialization (n := n) k r = 0 ↔
96       ∃ s : OrderReesRing (n := n) k,
97         orderReesParameter (n := n) k * s = r := by
98   constructor
99   · intro hr
100     have hlower : ∀ N,
101         (r : Polynomial (WeylAlg k n)).coeff (N + 1) ∈
102           orderPiece k n N := by
103       intro N
104       have hprincipal := principal_component_eq_zero_of_sum_eq_zero
105         (n := n) k r hr (N + 1)
106       have hmem := r.property (N + 1)
107       exact (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
108         k (@orderWeight n) ((r : Polynomial (WeylAlg k n)).coeff (N + 1))
109           hmem).mp hprincipal
110     let s : OrderReesRing (n := n) k :=
111       ⟨Polynomial.divX (r : Polynomial (WeylAlg k n)), by
112         intro N
113         rw [Polynomial.coeff_divX]
114         exact hlower N⟩
115     refine ⟨s, ?_⟩
116     apply Subtype.ext
117     change Polynomial.X * Polynomial.divX
118         (r : Polynomial (WeylAlg k n)) =
119       (r : Polynomial (WeylAlg k n))
120     have hconstant : (r : Polynomial (WeylAlg k n)).coeff 0 = 0 := by
121       have hprincipal := principal_component_eq_zero_of_sum_eq_zero
122         (n := n) k r hr 0
123       have hmem := r.property 0
124       have hlower0 := (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
125         k (@orderWeight n) ((r : Polynomial (WeylAlg k n)).coeff 0)
126           hmem).mp hprincipal
127       simpa [presentedStrictLowerPiece, Submodule.mem_bot] using hlower0
128     simpa [hconstant] using
129       (Polynomial.X_mul_divX_add (r : Polynomial (WeylAlg k n)))
130   · rintro ⟨s, rfl⟩
131     rw [map_mul, orderReesSpecialization_parameter, zero_mul]

133 /-- Exact unlocalized specialization kernel for the concrete order-Rees
134 two-jet: every killed class is a left multiple of the square-zero parameter,
135 and every such multiple is killed. -/
136 theorem orderReesTwoJetSpecialization_eq_zero_iff_exists_parameter_mul
137     (q : OrderReesTwoJet (n := n) k) :
138     orderReesTwoJetSpecialization (n := n) k q = 0 ↔
139       ∃ b : OrderReesTwoJet (n := n) k,
140         orderReesTwoJetParameter (n := n) k * b = q := by
141   constructor
142   · intro hq
143     obtain ⟨r, rfl⟩ := Quotient.mk''_surjective q
144     change orderReesSpecialization (n := n) k r = 0 at hq
145     obtain ⟨s, hs⟩ :=
146       (orderReesSpecialization_eq_zero_iff_exists_parameter_mul
147         (n := n) k r).mp hq
148     refine ⟨orderReesTwoJetQuotient (n := n) k s, ?_⟩
149     rw [orderReesTwoJetParameter, ← map_mul, hs]
150     rfl
151   · rintro ⟨b, rfl⟩
152     rw [map_mul, orderReesTwoJetSpecialization_parameter, zero_mul]

154 /-- The same exactness statement in linear-kernel/range form. -/
155 theorem orderReesTwoJetSpecialization_ker_eq_range_parameterMul :
156     LinearMap.ker (orderReesTwoJetSpecialization (n := n) k).toLinearMap =
157       LinearMap.range
158         (LinearMap.mulLeft k (orderReesTwoJetParameter (n := n) k)) := by
159   ext q
160   rw [LinearMap.mem_ker, LinearMap.mem_range]
161   exact orderReesTwoJetSpecialization_eq_zero_iff_exists_parameter_mul
162     (n := n) k q
