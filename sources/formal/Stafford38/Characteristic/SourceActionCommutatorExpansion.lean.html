6 /-!
7 # Commutator expansion for source-row action equations
8 
9 This file proves the algebraic calculation between two common-basis
10 first-order source equations and the induced ringCommutator action on the special
11 fibre. It is independent of the Weyl presentation, localization,
12 Artinianity, adapted basis, and trace arguments.
13 
14 The deformation module is a left module over the noncommutative ring; the
15 parameter is central and square-zero. The result records the specialized
16 action expansion without identifying it with a residue-mapped coefficient
17 matrix. The canonical-support proof uses the separate localized
18 involutivity development.
19 -/

41 /-- Apply a source-row matrix to a list of lifted basis vectors.  The first
42 index is the input vector and the second index is its output coefficient. -/
43 def sourceRowAction (M : Matrix ι ι B) (e : ι → W) (i : ι) : W :=
44   ∑ j, M i j • e j

46 theorem smul_sourceRowAction (a : B) (M : Matrix ι ι B) (e : ι → W) (i : ι) :
47     a • sourceRowAction M e i = ∑ j, (a * M i j) • e j := by
48   simp only [sourceRowAction]
49   rw [Finset.smul_sum]
50   apply Finset.sum_congr rfl
51   intro j hj
52   exact (mul_smul a (M i j) (e j)).symm

54 theorem sourceRowAction_mul (M N : Matrix ι ι B) (e : ι → W) (i : ι) :
55     sourceRowAction M (fun j => sourceRowAction N e j) i =
56       sourceRowAction (M * N) e i := by
57   simp only [sourceRowAction, Matrix.mul_apply]
58   calc
59     (∑ j, M i j • ∑ k, N j k • e k) =
60         ∑ j, ∑ k, M i j • (N j k • e k) := by
61           apply Finset.sum_congr rfl
62           intro j hj
63           rw [Finset.smul_sum]
64     _ = ∑ j, ∑ k, (M i j * N j k) • e k := by
65           apply Finset.sum_congr rfl
66           intro j hj
67           apply Finset.sum_congr rfl
68           intro k hk
69           exact (mul_smul (M i j) (N j k) (e k)).symm
70     _ = ∑ k, ∑ j, (M i j * N j k) • e k := by
71           exact Finset.sum_comm
72     _ = ∑ k, (∑ j, M i j * N j k) • e k := by
73           apply Finset.sum_congr rfl
74           intro k hk
75           rw [Finset.sum_smul]

103 /-- Move a noncommutative scalar through a source row.  The error is the
104 entrywise scalar ringCommutator. -/
105 theorem action_sourceRow_decomp
106     (a : B) (M : Matrix ι ι B) (e : ι → W) (i : ι) :
107     a • sourceRowAction M e i =
108       sourceRowAction M (fun j => a • e j) i +
109         sourceRowAction (scalarMatrixCommutator a M) e i := by
110   simp only [sourceRowAction, scalarMatrixCommutator]
111   rw [Finset.smul_sum, ← Finset.sum_add_distrib]
112   apply Finset.sum_congr rfl
113   intro j hj
114   simp only [sub_smul]
115   rw [← mul_smul (M i j) a (e j)]
116   rw [← sub_smul, ← add_smul]
117   rw [show M i j * a + (a * M i j - M i j * a) = a * M i j by
118     noncomm_ring]
119   exact (mul_smul a (M i j) (e j)).symm

121 theorem sourceRowAction_parameter_mul
122     (c : B) (hcentral : ∀ a : B, c * a = a * c)
123     (M N : Matrix ι ι B) (e : ι → W) (i : ι) :
124     sourceRowAction M (fun j => c • sourceRowAction N e j) i =
125       c • sourceRowAction (M * N) e i := by
126   simp only [sourceRowAction]
127   calc
128     (∑ j, M i j • (c • ∑ k, N j k • e k)) =
129         ∑ j, c • (M i j • ∑ k, N j k • e k) := by
130           apply Finset.sum_congr rfl
131           intro j hj
132           rw [← mul_smul (M i j) c,
133             ← hcentral (M i j)]
134           exact mul_smul c (M i j) _
135     _ = c • ∑ j, M i j • ∑ k, N j k • e k := by
136           rw [Finset.smul_sum]
137     _ = c • sourceRowAction (M * N) e i := by
138           congr 1
139           exact sourceRowAction_mul M N e i

153 /-- Expanding one operator acting on the first-order source equation for a
154 second operator. -/
155 theorem action_on_source_equation
156     (c x : B) (hcentral : ∀ a : B, c * a = a * c)
157     (hc2 : c * c = 0)
158     (A Bm Gamma Theta : Matrix ι ι B) (e : ι → W)
159     (hx : ∀ j, x • e j = sourceRowAction A e j +
160       c • sourceRowAction Gamma e j) (i : ι) :
161     x • (sourceRowAction Bm e i + c • sourceRowAction Theta e i) =
162       sourceRowAction (Bm * A) e i + c • sourceRowAction (Bm * Gamma) e i +
163         sourceRowAction (scalarMatrixCommutator x Bm) e i +
164         c • sourceRowAction (Theta * A) e i +
165         c • sourceRowAction (scalarMatrixCommutator x Theta) e i := by
166   have hdouble (w : W) : c • (c • w) = 0 := by
167     rw [← mul_smul c c, hc2, zero_smul]
168   have hxe : (fun j => x • e j) =
169       (fun j => sourceRowAction A e j + c • sourceRowAction Gamma e j) := by
170     funext j
171     exact hx j
172   calc
173     x • (sourceRowAction Bm e i + c • sourceRowAction Theta e i) =
174         x • sourceRowAction Bm e i +
175           x • (c • sourceRowAction Theta e i) := by
176           simp only [smul_add]
177     _ = (sourceRowAction Bm (fun j => x • e j) i +
178           sourceRowAction (scalarMatrixCommutator x Bm) e i) +
179         c • (sourceRowAction Theta (fun j => x • e j) i +
180           sourceRowAction (scalarMatrixCommutator x Theta) e i) := by
181           rw [action_sourceRow_decomp,
182             action_parameter_comm c x hcentral,
183             action_sourceRow_decomp]
184     _ = (sourceRowAction Bm (fun j => sourceRowAction A e j +
185           c • sourceRowAction Gamma e j) i +
186           sourceRowAction (scalarMatrixCommutator x Bm) e i) +
187         c • (sourceRowAction Theta (fun j => sourceRowAction A e j +
188           c • sourceRowAction Gamma e j) i +
189           sourceRowAction (scalarMatrixCommutator x Theta) e i) := by
190           rw [hxe]
191     _ = sourceRowAction (Bm * A) e i + c • sourceRowAction (Bm * Gamma) e i +
192           sourceRowAction (scalarMatrixCommutator x Bm) e i +
193           c • sourceRowAction (Theta * A) e i +
194           c • sourceRowAction (scalarMatrixCommutator x Theta) e i := by
195           rw [sourceRowAction_fun_add, sourceRowAction_fun_add,
196             sourceRowAction_mul,
197             sourceRowAction_parameter_mul c hcentral Bm Gamma e i,
198             sourceRowAction_mul,
199             sourceRowAction_parameter_mul c hcentral Theta Gamma e i]
200           simp only [smul_add, hdouble, zero_add, add_zero]
201           abel

203 /-- The matrix appearing after the parameter has been removed from the
204 ringCommutator expansion.  The two entrywise scalar commutators are omitted because
205 they vanish under every commutative ring specialization. -/
206 def sourceExpansionMatrix
207     (X Y Omega : Matrix ι ι B)
208     (A Bm Gamma Theta : Matrix ι ι B) : Matrix ι ι B :=
209   X - Y + Omega + (Bm * Gamma + Theta * A - A * Theta - Gamma * Bm)
210 
211 /-- The unspecialized source-row matrix produced by the ringCommutator expansion.
212 The two entrywise scalar commutators disappear only after applying a
213 commutative specialization. -/
214 def fullSourceExpansionMatrix
215     (x y : B) (X Y Omega : Matrix ι ι B)
216     (A Bm Gamma Theta : Matrix ι ι B) : Matrix ι ι B :=
217   sourceExpansionMatrix X Y Omega A Bm Gamma Theta +
218     scalarMatrixCommutator x Theta - scalarMatrixCommutator y Gamma

228 /-- After a commutative specialization, the source expansion has zero trace
229 when the three residue matrices are strictly upper triangular.  This is an
230 algebraic trace statement for the displayed source expansion; it does not
231 identify that expansion with the induced action on the residual module. -/
232 theorem sourceExpansionMatrix_map_trace_eq_zero
233     (f : B →+* Abar)
234     (X Y Omega A Bm Gamma Theta : Matrix ι ι B)
235     (hX : IsStrictUpperTriangularOver X)
236     (hY : IsStrictUpperTriangularOver Y)
237     (hOmega : IsStrictUpperTriangularOver Omega) :
238     Matrix.trace
239       ((sourceExpansionMatrix X Y Omega A Bm Gamma Theta).map f) = 0 := by
240   have hR : IsStrictUpperTriangularOver (X - Y + Omega) := by
241     intro i j hji
242     simp [hX i j hji, hY i j hji, hOmega i j hji]
243   have hRmap : IsStrictUpperTriangularOver
244       ((X - Y + Omega).map f) := by
245     intro i j hji
246     simp [Matrix.map_apply, hR i j hji]
247   have htraceR : Matrix.trace ((X - Y + Omega).map f) = 0 := by
248     rw [Matrix.trace]
249     apply Finset.sum_eq_zero
250     intro i _hi
251     exact hRmap i i le_rfl
252   have htraceComm (M N : Matrix ι ι B) :
253       Matrix.trace ((M * N - N * M).map f) = 0 := by
254     have hmap : (M * N - N * M).map f =
255         M.map f * N.map f - N.map f * M.map f := by
256       ext i j
257       simp [Matrix.map_apply, Matrix.mul_apply]
258     rw [hmap, Matrix.trace_sub, Matrix.trace_mul_comm, sub_self]
259   calc
260     Matrix.trace ((sourceExpansionMatrix X Y Omega A Bm Gamma Theta).map f) =
261         Matrix.trace ((X - Y + Omega).map f) +
262           Matrix.trace ((Bm * Gamma + Theta * A - A * Theta - Gamma * Bm).map f) := by
263       rw [sourceExpansionMatrix]
264       have hmap :
265           ((X - Y + Omega) +
266             (Bm * Gamma + Theta * A - A * Theta - Gamma * Bm)).map f =
267           (X - Y + Omega).map f +
268             (Bm * Gamma + Theta * A - A * Theta - Gamma * Bm).map f := by
269         ext i j
270         simp [Matrix.map_apply]
271       rw [hmap, Matrix.trace_add]
272     _ = 0 +
273           Matrix.trace ((Bm * Gamma + Theta * A - A * Theta - Gamma * Bm).map f) := by
274       rw [htraceR]
275     _ = 0 := by
276       have hdecomp : Bm * Gamma + Theta * A - A * Theta - Gamma * Bm =
277           (Bm * Gamma - Gamma * Bm) + (Theta * A - A * Theta) := by
278         abel
279       rw [hdecomp]
280       have hmap :
281           ((Bm * Gamma - Gamma * Bm) + (Theta * A - A * Theta)).map f =
282             (Bm * Gamma - Gamma * Bm).map f +
283               (Theta * A - A * Theta).map f := by
284         ext i j
285         simp [Matrix.map_apply]
286       rw [hmap, Matrix.trace_add, htraceComm, htraceComm]
287       simp

298 /-- The exact source-row ringCommutator expansion before specialization. -/
299 theorem commutator_on_source_equations
300     (c x y z : B) (hcentral : ∀ a : B, c * a = a * c)
301     (hc2 : c * c = 0)
302     (hxy : x * y - y * x = c * z)
303     (A Bm Gamma Theta X Y Omega : Matrix ι ι B) (e : ι → W)
304     (hx : ∀ i, x • e i = sourceRowAction A e i +
305       c • sourceRowAction Gamma e i)
306     (hy : ∀ i, y • e i = sourceRowAction Bm e i +
307       c • sourceRowAction Theta e i)
308     (hX : ∀ i j, c * X i j = scalarMatrixCommutator x Bm i j)
309     (hY : ∀ i j, c * Y i j = scalarMatrixCommutator y A i j)
310     (hOmega : ∀ i j, c * Omega i j =
311       sourceMatrixCommutator Bm A i j) (i : ι) :
312     c • (z • e i) = c • sourceRowAction
313       (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta) e i := by
314   have hxy_action : x • (y • e i) =
315       sourceRowAction (Bm * A) e i + c • sourceRowAction (Bm * Gamma) e i +
316         sourceRowAction (scalarMatrixCommutator x Bm) e i +
317         c • sourceRowAction (Theta * A) e i +
318         c • sourceRowAction (scalarMatrixCommutator x Theta) e i := by
319     rw [hy i]
320     exact action_on_source_equation c x hcentral hc2 A Bm Gamma Theta e hx i
321   have hyx_action : y • (x • e i) =
322       sourceRowAction (A * Bm) e i + c • sourceRowAction (A * Theta) e i +
323         sourceRowAction (scalarMatrixCommutator y A) e i +
324         c • sourceRowAction (Gamma * Bm) e i +
325         c • sourceRowAction (scalarMatrixCommutator y Gamma) e i := by
326     rw [hx i]
327     exact action_on_source_equation c y hcentral hc2 Bm A Theta Gamma e hy i
328   have hXrow : sourceRowAction (scalarMatrixCommutator x Bm) e i =
329       c • sourceRowAction X e i := by
330     calc
331       sourceRowAction (scalarMatrixCommutator x Bm) e i =
332           sourceRowAction (fun p q => c * X p q) e i := by
333             apply rowAction_congr
334             intro p q
335             exact (hX p q).symm
336       _ = c • sourceRowAction X e i := sourceRowAction_left_scalar c X e i
337   have hYrow : sourceRowAction (scalarMatrixCommutator y A) e i =
338       c • sourceRowAction Y e i := by
339     calc
340       sourceRowAction (scalarMatrixCommutator y A) e i =
341           sourceRowAction (fun p q => c * Y p q) e i := by
342             apply rowAction_congr
343             intro p q
344             exact (hY p q).symm
345       _ = c • sourceRowAction Y e i := sourceRowAction_left_scalar c Y e i
346   have hOmega_row : sourceRowAction (sourceMatrixCommutator Bm A) e i =
347       c • sourceRowAction Omega e i := by
348     calc
349       sourceRowAction (sourceMatrixCommutator Bm A) e i =
350           sourceRowAction (fun p q => c * Omega p q) e i := by
351             apply rowAction_congr
352             intro p q
353             exact (hOmega p q).symm
354       _ = c • sourceRowAction Omega e i := sourceRowAction_left_scalar c Omega e i
355   calc
356     c • (z • e i) = (c * z) • e i := (mul_smul c z _).symm
357     _ = (x * y - y * x) • e i := by rw [hxy]
358     _ = x • (y • e i) - y • (x • e i) := by
359       rw [sub_smul, mul_smul, mul_smul]
360     _ = (sourceRowAction (Bm * A) e i + c • sourceRowAction (Bm * Gamma) e i +
361           sourceRowAction (scalarMatrixCommutator x Bm) e i +
362           c • sourceRowAction (Theta * A) e i +
363           c • sourceRowAction (scalarMatrixCommutator x Theta) e i) -
364         (sourceRowAction (A * Bm) e i + c • sourceRowAction (A * Theta) e i +
365           sourceRowAction (scalarMatrixCommutator y A) e i +
366           c • sourceRowAction (Gamma * Bm) e i +
367           c • sourceRowAction (scalarMatrixCommutator y Gamma) e i) := by
368       rw [hxy_action, hyx_action]
369     _ = c • sourceRowAction
370         (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta) e i := by
371       have hBArow : sourceRowAction (Bm * A) e i -
372           sourceRowAction (A * Bm) e i = c • sourceRowAction Omega e i := by
373         calc
374           sourceRowAction (Bm * A) e i - sourceRowAction (A * Bm) e i =
375               sourceRowAction (sourceMatrixCommutator Bm A) e i := by
376                 simp [sourceMatrixCommutator, sourceRowAction_sub]
377           _ = c • sourceRowAction Omega e i := hOmega_row
378       rw [hXrow, hYrow]
379       calc
380         (sourceRowAction (Bm * A) e i + c • sourceRowAction (Bm * Gamma) e i +
381             c • sourceRowAction X e i + c • sourceRowAction (Theta * A) e i +
382             c • sourceRowAction (scalarMatrixCommutator x Theta) e i) -
383           (sourceRowAction (A * Bm) e i + c • sourceRowAction (A * Theta) e i +
384             c • sourceRowAction Y e i + c • sourceRowAction (Gamma * Bm) e i +
385             c • sourceRowAction (scalarMatrixCommutator y Gamma) e i) =
386           (sourceRowAction (Bm * A) e i - sourceRowAction (A * Bm) e i) +
387             (c • sourceRowAction (Bm * Gamma) e i - c • sourceRowAction (A * Theta) e i +
388               c • sourceRowAction X e i - c • sourceRowAction Y e i +
389               c • sourceRowAction (Theta * A) e i - c • sourceRowAction (Gamma * Bm) e i +
390               c • sourceRowAction (scalarMatrixCommutator x Theta) e i -
391               c • sourceRowAction (scalarMatrixCommutator y Gamma) e i) := by
392                 abel
393         _ = c • sourceRowAction
394             (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta) e i := by
395           rw [hBArow]
396           simp [fullSourceExpansionMatrix, sourceExpansionMatrix,
397             sourceRowAction_add, sourceRowAction_sub, neg_one_smul]
398           abel_nf
399           simp [neg_one_smul]
400           abel

413 /-- After exact parameter descent, the ringCommutator cofactor acts on the special
414 fibre by the specialized source expansion.  This is the precise action-level
415 conclusion needed before any trace comparison. -/
416 theorem rho_commutator_on_source_equations
417     (c x y z : B) (f : B →+* Abar) (rho : W →+ V)
418     (hcentral : ∀ a : B, c * a = a * c)
419     (hc2 : c * c = 0) (hfc : f c = 0)
420     (hxy : x * y - y * x = c * z)
421     (hact : ∀ a w, rho (a • w) = f a • rho w)
422     (hExact : AddMonoidHom.ker
423         (parameterAct (W := W) c) =
424       AddMonoidHom.range (parameterAct (W := W) c))
425     (A Bm Gamma Theta X Y Omega : Matrix ι ι B) (e : ι → W)
426     (hx : ∀ i, x • e i = sourceRowAction A e i +
427       c • sourceRowAction Gamma e i)
428     (hy : ∀ i, y • e i = sourceRowAction Bm e i +
429       c • sourceRowAction Theta e i)
430     (hX : ∀ i j, c * X i j = scalarMatrixCommutator x Bm i j)
431     (hY : ∀ i j, c * Y i j = scalarMatrixCommutator y A i j)
432     (hOmega : ∀ i j, c * Omega i j =
433       sourceMatrixCommutator Bm A i j) (i : ι) :
434     f z • rho (e i) =
435       ∑ j, f (sourceExpansionMatrix X Y Omega A Bm Gamma Theta i j) •
436         rho (e j) := by
437   have hkill : ∀ w : W, c • w = 0 → rho w = 0 := by
438     intro w hw
439     have hwker : w ∈ AddMonoidHom.ker
440         (parameterAct (W := W) c) := by
441       exact AddMonoidHom.mem_ker.mpr hw
442     rw [hExact] at hwker
443     obtain ⟨v, hv⟩ := hwker
444     rw [← hv]
445     change rho (c • v) = 0
446     rw [hact, hfc, zero_smul]
447   have hdouble := commutator_on_source_equations c x y z hcentral hc2 hxy
448       A Bm Gamma Theta X Y Omega e hx hy hX hY hOmega i
449   have hparameter : c •
450       (z • e i - sourceRowAction
451         (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta) e i) = 0 := by
452     rw [smul_sub]
453     exact sub_eq_zero.mpr hdouble
454   have hdesc := hkill _ hparameter
455   have hdesc' : rho (z • e i) =
456       rho (sourceRowAction
457         (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta) e i) := by
458     rw [map_sub] at hdesc
459     exact sub_eq_zero.mp hdesc
460   have hexpansion : f z • rho (e i) =
461       ∑ j, f (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta i j) •
462         rho (e j) := by
463     calc
464       f z • rho (e i) = rho (z • e i) := (hact z (e i)).symm
465       _ = rho (sourceRowAction
466           (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta) e i) := hdesc'
467       _ = _ := rho_sourceRowAction f rho hact _ _ _
468   calc
469     f z • rho (e i) =
470         ∑ j, f (fullSourceExpansionMatrix x y X Y Omega A Bm Gamma Theta i j) •
471           rho (e j) := hexpansion
472     _ = ∑ j, f (sourceExpansionMatrix X Y Omega A Bm Gamma Theta i j) •
