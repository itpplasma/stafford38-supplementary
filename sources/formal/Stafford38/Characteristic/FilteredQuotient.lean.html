9 /-!
10 # Differential-order filtration on a right-ideal quotient
11 
12 This file constructs the filtration induced on the actual additive quotient
13 by a right ideal and identifies each associated-graded piece with homogeneous
14 symbols modulo the principal components of the filtered ideal. The
15 degreewise quotient derived from `A / I` remains distinct from the cyclic
16 `SymbolRing / orderInitialIdeal` model. These constructions feed the
17 filtered-quotient development; the final support proof also has the checked
18 `PaperQuotientDescent` route.
19 -/

39 /-!
40 `WeylAlg` is a `RingQuot`, whose `AddCommMonoid` and `Ring` instances
41 are declared independently.  The two induced `AddCommMonoid` structures on a
42 submodule of it are definitionally equal but not syntactically identical, and
43 since Lean 4.33 instance arguments are matched only up to instance
44 transparency.  Mathlib's `Submodule.hasQuotient` expects the group-derived
45 shape, so without the two alignments below no quotient `↥p ⧸ q` of a
46 differential-order piece elaborates.  Both are `rfl`, so nothing about the
47 `k`-module structure changes.
48 -/
49 
50 local instance (priority := 10000) pieceAddCommMonoid {n : ℕ}
51     (p : Submodule k (WeylAlg k n)) : AddCommMonoid p :=
52   AddCommGroup.toAddCommMonoid

54 local instance (priority := 10000) pieceModule {n : ℕ}
55     (p : Submodule k (WeylAlg k n)) : Module k p :=
56   Submodule.module p

58 /-- A right ideal, regarded only as a `k`-linear subspace.  Its carrier is
59 literally unchanged. -/
60 abbrev rightIdealKSubmodule {A : Type*} [Ring A] [Algebra k A]
61     (I : RightIdeal A) : Submodule k A :=
62   I.restrictScalars k
63 
64 @[simp] theorem mem_rightIdealKSubmodule {A : Type*} [Ring A] [Algebra k A]
65     (I : RightIdeal A) (a : A) :
66     a ∈ rightIdealKSubmodule k I ↔ a ∈ I :=
67   Iff.rfl
68 
69 theorem rightIdealKSubmodule_eq_restrictScalars
70     {A : Type*} [Ring A] [Algebra k A] (I : RightIdeal A) :
71     rightIdealKSubmodule k I = I.restrictScalars k := by
72   ext a
73   rfl

75 /-- The `k`-linear quotient used for the filtration is canonically the same
76 underlying quotient as the regular right-module quotient. -/
77 def filteredRightQuotientEquivRightQuotient
78     {A : Type*} [Ring A] [Algebra k A] (I : RightIdeal A) :
79     (A ⧸ rightIdealKSubmodule k I) ≃ₗ[k] RightQuotient I :=
80   Submodule.quotEquivOfEq _ _ (rightIdealKSubmodule_eq_restrictScalars k I) ≪≫ₗ
81     Submodule.Quotient.restrictScalarsEquiv k I

89 /-- Global alignment for the actual filtered quotient.  Lean 4.33 matches
90 instance arguments only up to instance transparency, and the default
91 `Submodule.Quotient.addCommMonoid` is not the additive monoid derived from the
92 quotient's `AddCommGroup`.  Without this alignment the direct sum of the
93 pieces below inherits an additive monoid that `Submodule.hasQuotient` cannot
94 match, and `FilteredQuotientSpecialFibre` cannot state its quotient.  This is
95 the same repair already made below for `QuotientOrderGradedPiece`. -/

97     (I : RightIdeal (WeylAlg k n)) :
98     AddCommMonoid (FilteredRightQuotient k I) :=
99   AddCommGroup.toAddCommMonoid

106 /-- The image of the differential-order piece in the actual quotient. -/
107 def quotientOrderPiece (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
108     Submodule k (FilteredRightQuotient k I) :=
109   (presentedWeightPiece k (@orderWeight n) N).map
110     (rightIdealKSubmodule k I).mkQ

112 /-- The image of the strict lower differential-order piece in the actual
113 quotient. -/
114 def quotientOrderStrictLowerPiece (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
115     Submodule k (FilteredRightQuotient k I) :=
116   (presentedStrictLowerPiece k orderWeight N).map
117     (rightIdealKSubmodule k I).mkQ

121     quotientOrderStrictLowerPiece k I N ≤ quotientOrderPiece k I N :=

124 /-- The degree-`N` associated graded piece of the filtration induced on the

145 /-- Relations in the degree-`N` filtered algebra piece: an element is zero in
146 the quotient graded piece exactly when it lies in `I + F_{<N} A`. -/
147 def orderQuotientRelation
148     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
149     Submodule k (presentedWeightPiece k (@orderWeight n) N) :=
150   ((rightIdealKSubmodule k I) ⊔
151       presentedStrictLowerPiece k (@orderWeight n) N).comap
152     (presentedWeightPiece k (@orderWeight n) N).subtype

154 /-- The canonical map from a filtered algebra piece to the corresponding
155 filtered piece of the actual quotient. -/
156 def orderPieceToQuotientPiece
157     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
158     presentedWeightPiece k (@orderWeight n) N →ₗ[k] quotientOrderPiece k I N :=
159   ((rightIdealKSubmodule k I).mkQ.comp
160     (presentedWeightPiece k (@orderWeight n) N).subtype).codRestrict
161       (quotientOrderPiece k I N) (fun z => ⟨z, z.property, rfl⟩)

163 /-- The canonical map from a filtered algebra piece to the corresponding
164 graded piece of the actual quotient. -/
165 def orderPieceToQuotientGraded
166     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
167     presentedWeightPiece k (@orderWeight n) N →ₗ[k] QuotientOrderGradedPiece k I N :=
168   ((quotientOrderStrictLowerPiece k I N).comap
169       (quotientOrderPiece k I N).subtype).mkQ.comp
170     (orderPieceToQuotientPiece k I N)

172 theorem orderPieceToQuotientGraded_surjective
173     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
174     Function.Surjective (orderPieceToQuotientGraded k I N) := by
175   intro q
176   refine Submodule.Quotient.induction_on _ q ?_
177   rintro ⟨q, hq⟩
178   obtain ⟨z, hz, rfl⟩ := hq
179   exact ⟨⟨z, hz⟩, rfl⟩

181 /-- Exact kernel computation for the filtered quotient map. -/
182 theorem ker_orderPieceToQuotientGraded
183     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
184     LinearMap.ker (orderPieceToQuotientGraded k I N) =
185       orderQuotientRelation k I N := by
186   ext z
187   rw [LinearMap.mem_ker]
188   change Submodule.Quotient.mk
189       (⟨(rightIdealKSubmodule k I).mkQ z,
190         ⟨z, z.property, rfl⟩⟩ : quotientOrderPiece k I N) = 0 ↔ _
191   rw [Submodule.Quotient.mk_eq_zero]
192   change (rightIdealKSubmodule k I).mkQ z ∈
193       quotientOrderStrictLowerPiece k I N ↔ _
194   change (z : WeylAlg k n) ∈
195     (quotientOrderStrictLowerPiece k I N).comap
196       (rightIdealKSubmodule k I).mkQ ↔ _
197   rw [quotientOrderStrictLowerPiece, Submodule.comap_map_mkQ]
198   rfl

200 /-- The actual quotient graded piece is the filtered algebra piece modulo
201 `I + F_{<N} A`. -/
202 def quotientOrderGradedPieceEquivRelation
203     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
204     (presentedWeightPiece k (@orderWeight n) N ⧸
205       orderQuotientRelation k I N) ≃ₗ[k]
206       QuotientOrderGradedPiece k I N := by
207   let hEq := ker_orderPieceToQuotientGraded k I N
208   let g : (presentedWeightPiece k (@orderWeight n) N ⧸
209       orderQuotientRelation k I N) →ₗ[k]
210       QuotientOrderGradedPiece k I N :=
211     (orderQuotientRelation k I N).liftQ
212       (orderPieceToQuotientGraded k I N) (le_of_eq hEq.symm)
213   apply LinearEquiv.ofBijective g
214   constructor
215   · intro q₁ q₂ h
216     obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective _ q₁
217     obtain ⟨w, rfl⟩ := Submodule.Quotient.mk_surjective _ q₂
218     apply (Submodule.Quotient.eq _).2
219     have hzw : orderPieceToQuotientGraded k I N z =
220         orderPieceToQuotientGraded k I N w := by
221       simpa [g] using h
222     have hzwdiff : z - w ∈
223         LinearMap.ker (orderPieceToQuotientGraded k I N) := by
224       have e := LinearMap.map_sub (orderPieceToQuotientGraded k I N) z w
225       rw [LinearMap.mem_ker, e, hzw, sub_self]
226     rw [ker_orderPieceToQuotientGraded k I N] at hzwdiff
227     exact hzwdiff
228   · intro q
229     obtain ⟨z, hz⟩ := orderPieceToQuotientGraded_surjective k I N q
230     exact ⟨Submodule.Quotient.mk z, by simpa [g] using hz⟩

232 /-- Homogeneous symbol relations arising from `I + F_{<N} A`.  The strict
233 lower summand maps to zero, so these are precisely the degree-`N` principal
234 components contributed by filtered elements of `I`. -/
235 def orderSymbolRelation
236     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
237     Submodule k
238       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N) :=
239   (orderQuotientRelation k I N).map
240     (principalComponentOnPiece k (@orderWeight n) N)

242 /-- Align the additive monoid structure on a degree-`N` symbol quotient with
243 its additive group structure.  `Submodule.Quotient` declares the two
244 independently, so `DirectSum`'s `AddCommGroup` instance, and hence
245 `Submodule.liftQ` into the graded relation module, does not apply without
246 this `rfl` alignment.  Unlike the alignments above this one is global,
247 because `OrderSymbolRelationGraded` is assembled from these quotients in a
248 later module. -/

250     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
251     AddCommMonoid
252       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N ⧸
253         orderSymbolRelation k I N) :=
254   AddCommGroup.toAddCommMonoid

263 /-- Principal component, descended to the quotient by the exact source and
264 target relation submodules. -/
265 def quotientPrincipalComponentMap
266     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
267     (presentedWeightPiece k (@orderWeight n) N ⧸
268       orderQuotientRelation k I N) →ₗ[k]
269       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N ⧸
270         orderSymbolRelation k I N) :=
271   let f : presentedWeightPiece k (@orderWeight n) N →ₗ[k]
272       MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N :=
273     principalComponentOnPiece k (@orderWeight n) N
274   let h : orderQuotientRelation k I N ≤
275       (orderSymbolRelation k I N).comap f := by
276     intro z hz
277     change f z ∈ orderSymbolRelation k I N
278     exact ⟨z, hz, rfl⟩
279   Submodule.mapQ (p := orderQuotientRelation k I N)
280     (q := orderSymbolRelation k I N) f h

282 theorem quotientPrincipalComponentMap_surjective
283     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
284     Function.Surjective (quotientPrincipalComponentMap k I N) := by
285   intro q
286   refine Submodule.Quotient.induction_on _ q ?_
287   intro P
288   obtain ⟨z, rfl⟩ :=
289     principalComponentOnPiece_surjective k (@orderWeight n) P
290   let z' : presentedWeightPiece k (@orderWeight n) N := ⟨z, z.property⟩
291   refine ⟨Submodule.Quotient.mk z', ?_⟩
292   rw [quotientPrincipalComponentMap, Submodule.mapQ_apply]
293 
294 theorem quotientPrincipalComponentMap_injective
295     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
296     Function.Injective (quotientPrincipalComponentMap k I N) := by
297   intro q₁ q₂
298   revert q₂
299   refine Submodule.Quotient.induction_on _ q₁ ?_
300   intro z q₂
301   refine Submodule.Quotient.induction_on _ q₂ ?_
302   intro w h
303   rw [quotientPrincipalComponentMap, Submodule.mapQ_apply,
304     Submodule.mapQ_apply] at h
305   apply (Submodule.Quotient.eq _).2
306   have hpc :
307       principalComponentOnPiece k (@orderWeight n) N z -
308           principalComponentOnPiece k (@orderWeight n) N w ∈
309         orderSymbolRelation k I N :=
310     (Submodule.Quotient.eq _).mp h
311   obtain ⟨r, hr, hpr⟩ := hpc
312   have hdiffLower :
313       (((z - w) - r : presentedWeightPiece k (@orderWeight n) N) : WeylAlg k n) ∈
314         presentedStrictLowerPiece k orderWeight N := by
315     apply (mem_ker_principalComponentOnPiece_iff k (@orderWeight n)
316       ((z - w) - r)).mp
317     have e₁ := LinearMap.map_sub
318       (principalComponentOnPiece k (@orderWeight n) N) (z - w) r
319     have e₂ := LinearMap.map_sub
320       (principalComponentOnPiece k (@orderWeight n) N) z w
321     rw [LinearMap.mem_ker, e₁, e₂, hpr, sub_self]
322   have hdiffRelation : (z - w) - r ∈ orderQuotientRelation k I N := by
323     change ((((z - w) - r : presentedWeightPiece k (@orderWeight n) N) : WeylAlg k n)) ∈
324       (rightIdealKSubmodule k I) ⊔
325         presentedStrictLowerPiece k orderWeight N
326     exact Submodule.mem_sup_right hdiffLower
327   have hsum := (orderQuotientRelation k I N).add_mem hr hdiffRelation
328   have hrw : r + ((z - w) - r) = z - w := by abel
329   rwa [hrw] at hsum

331 /-- Degreewise filtered-quotient bridge: the actual associated graded piece
332 of `A / I` is canonically equivalent to homogeneous symbols modulo the
333 principal components of `I + F_{<N} A`. -/
334 def quotientOrderGradedPieceEquivSymbols
335     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
336     QuotientOrderGradedPiece k I N ≃ₗ[k]
337       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N ⧸
338         orderSymbolRelation k I N) :=
339   (quotientOrderGradedPieceEquivRelation k I N).symm ≪≫ₗ
340     LinearEquiv.ofBijective (quotientPrincipalComponentMap k I N)
341       ⟨quotientPrincipalComponentMap_injective k I N,
342         quotientPrincipalComponentMap_surjective k I N⟩

344 /-- Every filtered element of the right ideal contributes its principal
345 component to the exact symbol relation in the actual quotient graded piece. -/
346 theorem principalComponent_mem_orderSymbolRelation
347     (I : RightIdeal (WeylAlg k n)) (N : ℕ) (z : WeylAlg k n)
348     (hzN : z ∈ presentedWeightPiece k (@orderWeight n) N) (hzI : z ∈ I) :
349     principalComponentOnPiece k (@orderWeight n) N ⟨z, hzN⟩ ∈
350       orderSymbolRelation k I N := by
351   refine ⟨⟨z, hzN⟩, ?_, rfl⟩
352   change z ∈ (rightIdealKSubmodule k I) ⊔
353     presentedStrictLowerPiece k orderWeight N
354   exact Submodule.mem_sup_left hzI

356 /-- Exact description of the degree-`N` symbol relations: they are precisely
357 the principal components of elements of `I` lying in `F_N A`. -/
358 theorem mem_orderSymbolRelation_iff
359     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
360     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N) :
361     P ∈ orderSymbolRelation k I N ↔
362       ∃ (z : WeylAlg k n)
363         (hzN : z ∈ presentedWeightPiece k (@orderWeight n) N),
364         z ∈ I ∧ P = principalComponentOnPiece k (@orderWeight n) N ⟨z, hzN⟩ := by
365   constructor
366   · rintro ⟨r, hr, hPr⟩
367     change (r : WeylAlg k n) ∈
368       (rightIdealKSubmodule k I) ⊔
369         presentedStrictLowerPiece k orderWeight N at hr
370     obtain ⟨i, hi, l, hl, hil⟩ := Submodule.mem_sup.mp hr
371     have hlN : l ∈ presentedWeightPiece k (@orderWeight n) N :=
372       presentedStrictLowerPiece_le k (@orderWeight n) hl
373     have hiN : i ∈ presentedWeightPiece k (@orderWeight n) N := by
374       have hieq : i = (r : WeylAlg k n) - l := by
375         rw [← hil]
376         simp
377       rw [hieq]
378       exact Submodule.sub_mem _ r.property hlN
379     refine ⟨i, hiN, hi, ?_⟩
380     let ii : presentedWeightPiece k (@orderWeight n) N := ⟨i, hiN⟩
381     let ll : presentedWeightPiece k (@orderWeight n) N := ⟨l, hlN⟩
382     have hrl : r = ii + ll := by
383       apply Subtype.ext
384       exact hil.symm
385     have hpl :
386         principalComponentOnPiece k (@orderWeight n) N ll = 0 := by
387       apply Subtype.ext
388       exact (presentedPrincipalComponent_eq_zero_iff_mem_strictLower
389         k (@orderWeight n) ll ll.property).mpr hl
390     rw [← hPr, hrl, map_add]
391     calc
392       principalComponentOnPiece k (@orderWeight n) N ii +
393           principalComponentOnPiece k (@orderWeight n) N ll =
394           principalComponentOnPiece k (@orderWeight n) N ii + 0 :=
395         congrArg (fun Q =>
396           principalComponentOnPiece k (@orderWeight n) N ii + Q) hpl
397       _ = principalComponentOnPiece k (@orderWeight n) N ii := add_zero _
398       _ = principalComponentOnPiece k (@orderWeight n) N ⟨i, hiN⟩ := by
399         congr 1
400   · rintro ⟨z, hzN, hzI, rfl⟩
401     exact principalComponent_mem_orderSymbolRelation k I N z hzN hzI

403 /-- Every exact degree-`N` relation of the actual quotient is one of the
404 generators used by the existing order initial ideal. -/
405 theorem coe_mem_orderInitialGeneratorSet_of_mem_orderSymbolRelation
406     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
407     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) N)
408     (hP : P ∈ orderSymbolRelation k I N) :
409     (P : SymbolRing k n) ∈ orderInitialGeneratorSet k I := by
410   obtain ⟨z, hzN, hzI, hPz⟩ :=
411     (mem_orderSymbolRelation_iff k I N P).mp hP
412   refine ⟨N, z, hzN, hzI, ?_⟩
413   rw [hPz]
414   rfl

416 theorem coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation
417     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
