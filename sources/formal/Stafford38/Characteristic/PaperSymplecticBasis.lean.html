11 /-!
12 # Paper-facing symplectic basis completion
13 
14 Inductively split off a normalized symplectic plane and complete its
15 nondegenerate orthogonal complement.  The result is for an arbitrary finite
16 dimensional alternating nondegenerate space, not only the standard model.
17 -/

29 private def splitTwo : Fin 2 ≃ Fin 1 ⊕ Fin 1 :=
30   (finSumFinEquiv (m := 1) (n := 1)).symm

32 private def onePlusEquiv (n : ℕ) : Fin 1 ⊕ Fin n ≃ Fin (n + 1) where
33   toFun
34     | .inl _ => 0
35     | .inr i => i.succ
36   invFun := Fin.cases (Sum.inl 0) (fun i => Sum.inr i)
37   left_inv x := by
38     cases x with
39     | inl i => have : i = 0 := Subsingleton.elim _ _; subst i; rfl
40     | inr i => rfl
41   right_inv i := by
42     refine Fin.cases ?_ ?_ i <;> simp

44 @[simp] private theorem onePlusEquiv_symm_zero (n : ℕ) :
45     (onePlusEquiv n).symm (0 : Fin (n + 1)) = Sum.inl 0 := by
46   simp [onePlusEquiv, finSumFinEquiv]

52 private def rearrangePairIndex (n : ℕ) :
53     (Fin 1 ⊕ Fin 1) ⊕ (Fin n ⊕ Fin n) ≃
54       (Fin 1 ⊕ Fin n) ⊕ (Fin 1 ⊕ Fin n) where
55   toFun
56     | .inl (.inl i) => .inl (.inl i)
57     | .inl (.inr i) => .inr (.inl i)
58     | .inr (.inl i) => .inl (.inr i)
59     | .inr (.inr i) => .inr (.inr i)
60   invFun
61     | .inl (.inl i) => .inl (.inl i)
62     | .inl (.inr i) => .inr (.inl i)
63     | .inr (.inl i) => .inl (.inr i)
64     | .inr (.inr i) => .inr (.inr i)
65   left_inv x := by cases x with
66     | inl x => cases x <;> rfl
67     | inr x => cases x <;> rfl
68   right_inv x := by cases x with
69     | inl x => cases x <;> rfl
70     | inr x => cases x <;> rfl

72 private def symplecticPairIndex (n : ℕ) :
73     Fin 2 ⊕ (Fin n ⊕ Fin n) ≃ Fin (n + 1) ⊕ Fin (n + 1) :=
74   (Equiv.sumCongr splitTwo (Equiv.refl _)).trans
75     ((rearrangePairIndex n).trans
76       (Equiv.sumCongr (onePlusEquiv n) (onePlusEquiv n)))

78 private theorem pairVectors_range (v w : V) :
79     Set.range (pairVectors v w) = ({v, w} : Set V) := by
80   ext x
81   simp only [Set.mem_range, Set.mem_insert_iff, Set.mem_singleton_iff]
82   constructor
83   · rintro ⟨i, rfl⟩
84     fin_cases i <;> simp [pairVectors]
85   · rintro (rfl | rfl)
86     · exact ⟨0, by simp [pairVectors]⟩
87     · exact ⟨1, by simp [pairVectors]⟩

89 private theorem normalized_pair_linearIndependent (B : BilinForm k V)
90     (hAlt : ∀ x, B x x = 0) (v w : V) (hvw : B v w = 1) :
91     LinearIndependent k (pairVectors v w) := by
92   change LinearIndependent k ![v, w]
93   rw [LinearIndependent.pair_iff]
94   intro a b hab
95   have hwv : B w v = -1 := by
96     have h := BilinForm.IsAlt.neg_eq (show B.IsAlt from hAlt) v w
97     simpa [hvw] using h.symm
98   have hvsum : B (a • v + b • w) w = 0 := by
99     rw [hab, BilinForm.zero_left]
100   have hwsum : B (a • v + b • w) v = 0 := by
101     rw [hab, BilinForm.zero_left]
102   rw [BilinForm.add_left, BilinForm.smul_left, BilinForm.smul_left,
103     hvw, hAlt w] at hvsum
104   rw [BilinForm.add_left, BilinForm.smul_left, BilinForm.smul_left,
105     hAlt v, hwv] at hwsum
106   constructor
107   · simpa using hvsum
108   · simpa [mul_comm] using hwsum

134 theorem exists_symplectic_basis (n : ℕ) (B : BilinForm k V)
135     [Module.Finite k V] (hAlt : ∀ x, B x x = 0) (hB : B.Nondegenerate)
136     (hdim : Module.finrank k V = 2 * n) :
137     ∃ b : Basis (Fin n ⊕ Fin n) k V,
138       (∀ i j, B (b (Sum.inl i)) (b (Sum.inr j)) = if i = j then 1 else 0) ∧
139       (∀ i j, B (b (Sum.inl i)) (b (Sum.inl j)) = 0) ∧
140       (∀ i j, B (b (Sum.inr i)) (b (Sum.inr j)) = 0) := by
141   induction n generalizing V B with
142   | zero =>
143       have hdim0 : Module.finrank k V = 0 := by simpa using hdim
144       letI : Subsingleton V := Module.finrank_zero_iff.mp hdim0
145       letI : IsEmpty (Fin 0 ⊕ Fin 0) := inferInstance
146       refine ⟨Module.Basis.empty V, ?_, ?_, ?_⟩
147       · intro i j
148         exact isEmptyElim i
149       · intro i j
150         exact isEmptyElim i
151       · intro i j
152         exact isEmptyElim i
153   | succ n ih =>
154       have hdimV : Module.finrank k V = 2 * (n + 1) := hdim
155       have hpos : 0 < Module.finrank k V := by rw [hdimV]; omega
156       obtain ⟨v, hvne⟩ :=
157         (Module.finrank_pos_iff_exists_ne_zero (R := k) (M := V)).mp hpos
158       have hpair : ∃ w, B v w ≠ 0 := by
159         by_contra h
160         push_neg at h
161         exact hvne (hB.1 v h)
162       obtain ⟨w₀, hw₀⟩ := hpair
163       let w : V := (B v w₀)⁻¹ • w₀
164       have hvw : B v w = 1 := by
165         dsimp [w]
166         rw [BilinForm.smul_right]
167         exact inv_mul_cancel₀ hw₀
168       let U : Submodule k V := Submodule.span k {v, w}
169       let W : Submodule k V := B.orthogonal U
170       have hdata :=
171         Stafford38.LinearAlgebra.SymplecticComplement.pairComplementData B hAlt hB v w hvw
172       have hcompl : IsCompl U W := hdata.2.1
173       have hWnondeg : (B.restrict W).Nondegenerate := hdata.2.2.1
174       have hWdim : Module.finrank k W = 2 * n := by
175         have hdrop : Module.finrank k W + 2 = Module.finrank k V := hdata.2.2.2
176         rw [hdimV] at hdrop
177         omega
178       have hAltW : ∀ x : W, (B.restrict W) x x = 0 := by
179         intro x
180         exact hAlt x
181       obtain ⟨bW, hbWcross, hbWleft, hbWright⟩ :=
182         ih (V := W) (B := B.restrict W) hAltW hWnondeg hWdim
183       have hUmemV : v ∈ U := Submodule.subset_span (by simp [U])
184       have hUmemW : w ∈ U := Submodule.subset_span (by simp [U])
185       let hli := normalized_pair_linearIndependent B hAlt v w hvw
186       have hspan : Submodule.span k (Set.range (pairVectors v w)) = U := by
187         rw [pairVectors_range]
188       let bUraw : Basis (Fin 2) k (Submodule.span k (Set.range (pairVectors v w))) :=
189         Basis.span hli
190       let eU : (Submodule.span k (Set.range (pairVectors v w))) ≃ₗ[k] U :=
191         LinearEquiv.ofEq _ _ hspan
192       let bU : Basis (Fin 2) k U := bUraw.map eU
193       have hbU0 : bU 0 = ⟨v, hUmemV⟩ := by
194         apply Subtype.ext
195         dsimp [bU, bUraw, eU]
196         rw [Module.Basis.coe_span_apply]
197         simp [pairVectors]
198       have hbU1 : bU 1 = ⟨w, hUmemW⟩ := by
199         apply Subtype.ext
200         dsimp [bU, bUraw, eU]
201         rw [Module.Basis.coe_span_apply]
202         simp [pairVectors]
203       let eCompl := Submodule.prodEquivOfIsCompl U W hcompl
204       let bProd : Basis (Fin 2 ⊕ (Fin n ⊕ Fin n)) k (U × W) := bU.prod bW
205       let bBase := bProd.map eCompl
206       let b : Basis (Fin (n + 1) ⊕ Fin (n + 1)) k V :=
207         bBase.reindex (symplecticPairIndex n)
208       have hbE0 : b (Sum.inl (0 : Fin (n + 1))) = v := by
209         change (bBase.reindex (symplecticPairIndex n)) (Sum.inl 0) = v
210         rw [Module.Basis.reindex_apply, pairIndex_left0, Module.Basis.map_apply]
211         change (Submodule.prodEquivOfIsCompl U W hcompl)
212           ((bU.prod bW) (Sum.inl (0 : Fin 2))) = v
213         simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl', hbU0]
214       have hbF0 : b (Sum.inr (0 : Fin (n + 1))) = w := by
215         change (bBase.reindex (symplecticPairIndex n)) (Sum.inr 0) = w
216         rw [Module.Basis.reindex_apply, pairIndex_right0, Module.Basis.map_apply]
217         change (Submodule.prodEquivOfIsCompl U W hcompl)
218           ((bU.prod bW) (Sum.inl (1 : Fin 2))) = w
219         simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl', hbU1]
220       have hbEs (i : Fin n) : b (Sum.inl i.succ) = (bW (Sum.inl i) : V) := by
221         change (bBase.reindex (symplecticPairIndex n)) (Sum.inl i.succ) = _
222         rw [Module.Basis.reindex_apply, pairIndex_left_succ, Module.Basis.map_apply]
223         change (Submodule.prodEquivOfIsCompl U W hcompl)
224           ((bU.prod bW) (Sum.inr (Sum.inl i))) = _
225         simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl']
226       have hbFs (i : Fin n) : b (Sum.inr i.succ) = (bW (Sum.inr i) : V) := by
227         change (bBase.reindex (symplecticPairIndex n)) (Sum.inr i.succ) = _
228         rw [Module.Basis.reindex_apply, pairIndex_right_succ, Module.Basis.map_apply]
229         change (Submodule.prodEquivOfIsCompl U W hcompl)
230           ((bU.prod bW) (Sum.inr (Sum.inr i))) = _
231         simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl']
232       have hAlt' : B.IsAlt := hAlt
233       have hskew (x y : V) : B x y = -B y x := (hAlt'.neg_eq y x).symm
234       refine ⟨b, ?_, ?_, ?_⟩
235       · intro i j
236         refine Fin.cases ?_ (fun i => ?_) i
237         · refine Fin.cases ?_ (fun j => ?_) j
238           · simpa [hbE0, hbF0] using hvw
239           ·
240             simp only [ne_eq, eq_comm, Fin.succ_ne_zero, if_false]
241             rw [hbE0, hbFs]
242             exact ((bW (Sum.inr j)).property v hUmemV).symm
243         ·
244           refine Fin.cases ?_ (fun j => ?_) j
245           · simp only [Fin.succ_ne_zero, if_false]
246             rw [hbEs, hbF0, hskew]
247             simp [(bW (Sum.inl i)).property w hUmemW]
248           ·
249             have h := hbWcross i j
250             simpa [hbEs, hbFs] using h
251       · intro i j
252         refine Fin.cases ?_ (fun i => ?_) i
253         · refine Fin.cases ?_ (fun j => ?_) j
254           · rw [hbE0]
255             exact hAlt v
256           ·
257             rw [hbE0, hbEs]
258             exact (bW (Sum.inl j)).property v hUmemV
259         ·
260           refine Fin.cases ?_ (fun j => ?_) j
261           · rw [hbEs, hbE0, hskew]
262             simp [(bW (Sum.inl i)).property v hUmemV]
263           ·
264             have h := hbWleft i j
265             simpa [hbEs] using h
266       · intro i j
267         refine Fin.cases ?_ (fun i => ?_) i
268         · refine Fin.cases ?_ (fun j => ?_) j
269           · rw [hbF0]
270             exact hAlt w
271           ·
272             rw [hbF0, hbFs]
273             simp [(bW (Sum.inr j)).property w hUmemW]
274         ·
275           refine Fin.cases ?_ (fun j => ?_) j
276           · rw [hbFs, hbF0, hskew]
277             simp [(bW (Sum.inr i)).property w hUmemW]
278           ·
279             have h := hbWright i j
280             simpa [hbFs] using h

397 /-- The linear equivalence matching two symplectic bases preserves their forms. -/
398 theorem symplectic_basis_equiv (n : ℕ) (B₁ B₂ : BilinForm k V)
399     (b₁ b₂ : Basis (Fin n ⊕ Fin n) k V)
400     (hAlt₁ : ∀ x, B₁ x x = 0) (hAlt₂ : ∀ x, B₂ x x = 0)
401     (hcross₁ : ∀ i j, B₁ (b₁ (Sum.inl i)) (b₁ (Sum.inr j)) =
402       if i = j then 1 else 0)
403     (hleft₁ : ∀ i j, B₁ (b₁ (Sum.inl i)) (b₁ (Sum.inl j)) = 0)
404     (hright₁ : ∀ i j, B₁ (b₁ (Sum.inr i)) (b₁ (Sum.inr j)) = 0)
405     (hcross₂ : ∀ i j, B₂ (b₂ (Sum.inl i)) (b₂ (Sum.inr j)) =
406       if i = j then 1 else 0)
407     (hleft₂ : ∀ i j, B₂ (b₂ (Sum.inl i)) (b₂ (Sum.inl j)) = 0)
408     (hright₂ : ∀ i j, B₂ (b₂ (Sum.inr i)) (b₂ (Sum.inr j)) = 0) :
409     B₂.comp (b₁.equiv b₂ (Equiv.refl _)).toLinearMap
410       (b₁.equiv b₂ (Equiv.refl _)).toLinearMap = B₁ := by
411   let e := b₁.equiv b₂ (Equiv.refl _)
412   have hs₁ : ∀ x y, B₁ x y = -B₁ y x := by
413     intro x y
414     exact ((show B₁.IsAlt from hAlt₁).neg_eq y x).symm
415   have hs₂ : ∀ x y, B₂ x y = -B₂ y x := by
416     intro x y

437 /-- Given one nonzero vector in each space of the same symplectic dimension,
438 there is a form-preserving linear automorphism taking the first to the second. -/
439 theorem exists_symplectic_equiv_sending_nonzero (n : ℕ) (B : BilinForm k V)
440     [Module.Finite k V] (hAlt : ∀ x, B x x = 0) (hB : B.Nondegenerate)
441     (hdim : Module.finrank k V = 2 * (n + 1)) (v e₀ : V)
442     (hv : v ≠ 0) (he : e₀ ≠ 0) :
443     ∃ g : V ≃ₗ[k] V,
444       (∀ x y, B (g x) (g y) = B x y) ∧ g v = e₀ := by
445   have findPartner (x : V) (hx : x ≠ 0) : ∃ y, B x y = 1 := by
446     have hpair : ∃ y, B x y ≠ 0 := by
447       by_contra h
448       push_neg at h
449       exact hx (hB.1 x h)
450     obtain ⟨y₀, hy₀⟩ := hpair
451     refine ⟨(B x y₀)⁻¹ • y₀, ?_⟩
452     rw [BilinForm.smul_right]
453     exact inv_mul_cancel₀ hy₀
454   obtain ⟨vPartner, hvPair⟩ := findPartner v hv
455   obtain ⟨ePartner, hePair⟩ := findPartner e₀ he
456   obtain ⟨b₁, hc₁, hl₁, hr₁, hfirst₁, hsecond₁⟩ :=
457     exists_symplectic_basis_with_pair n B hAlt hB hdim v vPartner hvPair
458   obtain ⟨b₂, hc₂, hl₂, hr₂, hfirst₂, hsecond₂⟩ :=
459     exists_symplectic_basis_with_pair n B hAlt hB hdim e₀ ePartner hePair
460   let g : V ≃ₗ[k] V := b₁.equiv b₂ (Equiv.refl _)
461   have hg := symplectic_basis_equiv (n + 1) B B b₁ b₂ hAlt hAlt
462     hc₁ hl₁ hr₁ hc₂ hl₂ hr₂
463   change B.comp g.toLinearMap g.toLinearMap = B at hg
464   refine ⟨g, ?_, ?_⟩
465   · intro x y
466     have h := congrArg (fun f : BilinForm k V => f x y) hg
467     simpa [LinearMap.BilinForm.comp_apply] using h
468   · calc
469       g v = g (b₁ (Sum.inl (0 : Fin (n + 1)))) := by rw [hfirst₁]
470       _ = b₂ (Sum.inl 0) :=
471         Basis.equiv_apply b₁ (Sum.inl (0 : Fin (n + 1))) b₂ (Equiv.refl _)
472       _ = e₀ := hfirst₂
