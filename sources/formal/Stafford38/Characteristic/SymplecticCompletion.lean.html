14 /-!
15 # Symplectic completion of a phase vector
16 
17 An explicit symplectic transvection preserves the standard phase pairing.
18 One transvection sends a nonzero vector to another when they pair nontrivially;
19 an elementary bridge vector reduces the remaining case to two transvections.
20 Consequently any nonzero phase vector can occupy a prescribed matrix column.
21 Combined with homogeneous nonvanishing, this makes the transformed pure-power
22 coefficient nonzero over the original characteristic-zero field.
23 -/

35 def phasePairing {n : ℕ} (u v : PhaseVar n → k) : k :=
36   ∑ i, (u (.inr i) * v (.inl i) - u (.inl i) * v (.inr i))
37 
38 def phaseBasis {n : ℕ} (j : PhaseVar n) : PhaseVar n → k :=
39   fun i => if i = j then 1 else 0
40 
41 def symplecticTransvection {n : ℕ} (a : k) (u : PhaseVar n → k) :
42     Matrix (PhaseVar n) (PhaseVar n) k :=
43   fun i j => (if i = j then 1 else 0) +
44     a * phasePairing k (phaseBasis k j) u * u i

46 theorem phasePairing_basis_left {n : ℕ} (j : PhaseVar n)
47     (u : PhaseVar n → k) :
48     phasePairing k (phaseBasis k j) u =
49       match j with
50       | .inl i => -u (.inr i)
51       | .inr i => u (.inl i) := by
52   cases j with
53   | inl j =>
54       simp [phasePairing, phaseBasis]
55   | inr j =>
56       simp [phasePairing, phaseBasis]

58 theorem transvection_mulVec {n : ℕ} (a : k) (u z : PhaseVar n → k)
59     (i : PhaseVar n) :
60     Matrix.mulVec (symplecticTransvection k a u) z i =
61       z i + a * phasePairing k z u * u i := by
62   simp only [Matrix.mulVec, dotProduct, symplecticTransvection]
63   simp_rw [add_mul]
64   rw [Finset.sum_add_distrib]
65   rw [show (∑ x, (if i = x then 1 else 0) * z x) = z i by simp]
66   congr 1
67   rw [phasePairing]
68   rw [Fintype.sum_sum_type]
69   simp only [phasePairing_basis_left]
70   simp only [Finset.sum_sub_distrib]
71   rw [mul_sub, sub_mul]
72   rw [Finset.mul_sum, Finset.mul_sum, Finset.sum_mul, Finset.sum_mul]
73   rw [← Finset.sum_sub_distrib, ← Finset.sum_add_distrib]
74   apply Finset.sum_congr rfl
75   intro x hx
76   ring

78 theorem phasePairing_add_left {n : ℕ} (u v w : PhaseVar n → k) :
79     phasePairing k (u + v) w = phasePairing k u w + phasePairing k v w := by
80   simp [phasePairing, Finset.sum_add_distrib]

92 theorem phasePairing_smul_left {n : ℕ} (c : k) (u v : PhaseVar n → k) :

95   rw [mul_sub]

106 theorem phasePairing_skew {n : ℕ} (u v : PhaseVar n → k) :
107     phasePairing k u v = -phasePairing k v u := by
108   simp [phasePairing, Finset.sum_neg_distrib]

111 theorem phasePairing_self {n : ℕ} (u : PhaseVar n → k) :
112     phasePairing k u u = 0 := by
113   simp [phasePairing]
114   have hsum : (∑ x, u (.inl x) * u (.inr x)) =
115       ∑ x, u (.inr x) * u (.inl x) := by
116     apply Finset.sum_congr rfl
117     intro x hx
118     ring
119   rw [hsum, sub_self]

121 theorem transvection_preserves_pairing {n : ℕ} (a : k)
122     (u x y : PhaseVar n → k) :
123     phasePairing k (Matrix.mulVec (symplecticTransvection k a u) x)
124         (Matrix.mulVec (symplecticTransvection k a u) y) =
125       phasePairing k x y := by
126   have hx : Matrix.mulVec (symplecticTransvection k a u) x =
127       x + (a * phasePairing k x u) • u := by
128     funext i
129     rw [transvection_mulVec]
130     rfl
131   have hy : Matrix.mulVec (symplecticTransvection k a u) y =
132       y + (a * phasePairing k y u) • u := by
133     funext i
134     rw [transvection_mulVec]
135     rfl
136   rw [hx, hy, phasePairing_add_left]
137   simp_rw [phasePairing_add_right, phasePairing_smul_left,
138     phasePairing_smul_right, phasePairing_self]
139   rw [phasePairing_skew k u y]
140   ring

142 theorem transpose_standardForm_mul_apply {n : ℕ}
143     (A : Matrix (PhaseVar n) (PhaseVar n) k) (i j : PhaseVar n) :
144     (Matrix.transpose A * standardForm k n * A) i j =
145       phasePairing k (fun l => A l i) (fun l => A l j) := by

150 theorem phasePairing_basis {n : ℕ} (i j : PhaseVar n) :
151     phasePairing k (phaseBasis k i) (phaseBasis k j) =
152       standardForm k n i j := by
153   cases i <;> cases j <;>
154     simp [phasePairing, phaseBasis, standardForm, Matrix.J, Matrix.one_apply,
155       eq_comm]

158     (A : Matrix (PhaseVar n) (PhaseVar n) k) (j : PhaseVar n) :
159     Matrix.mulVec A (phaseBasis k j) = fun i => A i j := by

163 theorem symplecticTransvection_mem {n : ℕ} (a : k)
164     (u : PhaseVar n → k) :
165     symplecticTransvection k a u ∈ Matrix.symplecticGroup (Fin n) k := by
166   rw [SymplecticGroup.mem_iff']
167   ext i j
168   rw [transpose_standardForm_mul_apply]
169   change phasePairing k (fun l => symplecticTransvection k a u l i)
170       (fun l => symplecticTransvection k a u l j) = standardForm k n i j
171   rw [← phasePairing_basis k i j]
172   rw [← mulVec_phaseBasis k (symplecticTransvection k a u) i,
173     ← mulVec_phaseBasis k (symplecticTransvection k a u) j]
174   exact transvection_preserves_pairing k a u (phaseBasis k i) (phaseBasis k j)

176 theorem phasePairing_sub_right {n : ℕ} (u v w : PhaseVar n → k) :

183 theorem exists_phasePairing_right_ne_zero {n : ℕ}
184     {u : PhaseVar n → k} (hu : u ≠ 0) :
185     ∃ z : PhaseVar n → k, phasePairing k u z ≠ 0 := by
186   have hex : ∃ i, u i ≠ 0 := by
187     by_contra h
188     push_neg at h
189     apply hu
190     funext i
191     exact h i
192   rcases hex with ⟨i, hi⟩
193   cases i with
194   | inl i =>
195       refine ⟨phaseBasis k (.inr i), ?_⟩
196       simpa [phasePairing, phaseBasis] using neg_ne_zero.mpr hi
197   | inr i =>
198       refine ⟨phaseBasis k (.inl i), ?_⟩
199       simpa [phasePairing, phaseBasis] using hi

201 theorem exists_phasePairing_left_ne_zero {n : ℕ}
202     {u : PhaseVar n → k} (hu : u ≠ 0) :
203     ∃ z : PhaseVar n → k, phasePairing k z u ≠ 0 := by
204   rcases exists_phasePairing_right_ne_zero k hu with ⟨z, hz⟩
205   refine ⟨z, ?_⟩
206   rw [phasePairing_skew k z u]
207   exact neg_ne_zero.mpr hz

209 theorem exists_phasePairing_bridge {n : ℕ}
210     {e v : PhaseVar n → k} (he : e ≠ 0) (hv : v ≠ 0) :
211     ∃ w : PhaseVar n → k,
212       phasePairing k e w ≠ 0 ∧ phasePairing k w v ≠ 0 := by
213   rcases exists_phasePairing_right_ne_zero k he with ⟨r, her⟩
214   rcases exists_phasePairing_left_ne_zero k hv with ⟨s, hsv⟩
215   by_cases hrv : phasePairing k r v ≠ 0
216   · exact ⟨r, her, hrv⟩
217   by_cases hes : phasePairing k e s ≠ 0
218   · exact ⟨s, hes, hsv⟩
219   refine ⟨r + s, ?_, ?_⟩
220   · rw [phasePairing_add_right]
221     simpa [not_not.mp hes] using her
222   · rw [phasePairing_add_left]
223     simpa [not_not.mp hrv] using hsv

225 def transvectionSending {n : ℕ} (x y : PhaseVar n → k) :
226     Matrix (PhaseVar n) (PhaseVar n) k :=
227   symplecticTransvection k (phasePairing k x y)⁻¹ (y - x)

229 theorem transvectionSending_mem {n : ℕ} (x y : PhaseVar n → k) :
230     transvectionSending k x y ∈ Matrix.symplecticGroup (Fin n) k :=
231   symplecticTransvection_mem k _ _

242 theorem exists_symplectic_mulVec_eq {n : ℕ}
243     {e v : PhaseVar n → k} (he : e ≠ 0) (hv : v ≠ 0) :
244     ∃ M : Matrix (PhaseVar n) (PhaseVar n) k,
245       M ∈ Matrix.symplecticGroup (Fin n) k ∧ Matrix.mulVec M e = v := by
246   by_cases hn : n = 0
247   · subst n
248     letI : IsEmpty (PhaseVar 0) := inferInstance
249     have he0 : e = 0 := by funext i; exact isEmptyElim i
250     exact (he he0).elim
251   · let m := n - 1
252     have hn' : n = m + 1 := by dsimp [m]; omega
253     let B : LinearMap.BilinForm k (PhaseVar n → k) :=
254       { toFun := fun x =>
255           { toFun := fun y => phasePairing k x y
256             map_add' := phasePairing_add_right k x
257             map_smul' := by
258               intro c y
259               simpa [smul_eq_mul] using phasePairing_smul_right k c x y }
260         map_add' := by
261           intro x y
262           apply LinearMap.ext
263           intro z
264           exact phasePairing_add_left k x y z
265         map_smul' := by
266           intro c x
267           apply LinearMap.ext
268           intro z
269           simpa [smul_eq_mul] using phasePairing_smul_left k c x z }
270     have hAlt : ∀ x, B x x = 0 := by
271       intro x
272       exact phasePairing_self k x
273     have hB : B.Nondegenerate := by
274       constructor
275       · intro x hx
276         by_contra hxne
277         obtain ⟨z, hz⟩ := exists_phasePairing_right_ne_zero k hxne
278         exact hz (hx z)
279       · intro y hy
280         by_contra hyne
281         obtain ⟨z, hz⟩ := exists_phasePairing_left_ne_zero k hyne
282         exact hz (hy z)
283     have hdim : Module.finrank k (PhaseVar n → k) = 2 * (m + 1) := by
284       rw [Module.finrank_eq_card_basis (Pi.basisFun k (PhaseVar n))]
285       simp [hn']
286       ring
287     obtain ⟨g, hg, hge⟩ :=
288       Stafford38.CharacteristicPaperSymplecticBasis.exists_symplectic_equiv_sending_nonzero
289         m B hAlt hB hdim e v he hv
290     let f : (PhaseVar n → k) →ₗ[k] PhaseVar n → k := g.toLinearMap
291     let M : Matrix (PhaseVar n) (PhaseVar n) k := LinearMap.toMatrix' f
292     have hcomp : B.comp f f = B := by
293       apply (LinearMap.BilinForm.ext_iff_basis (Pi.basisFun k (PhaseVar n))).2
294       intro i j
295       exact hg _ _
296     have hBmatrix : LinearMap.BilinForm.toMatrix' B = standardForm k n := by
297       ext i j
298       rw [LinearMap.BilinForm.toMatrix'_apply]
299       change phasePairing k (Pi.single i 1) (Pi.single j 1) = _
300       have hsingle (i : PhaseVar n) : Pi.single i (1 : k) = phaseBasis k i := by
301         funext l
302         simp [Pi.single_apply, phaseBasis]
303       rw [hsingle, hsingle, phasePairing_basis]
304     have hmatrix := LinearMap.BilinForm.toMatrix'_comp B f f
305     have hM : Matrix.transpose M * standardForm k n * M = standardForm k n := by
306       calc
307         Matrix.transpose M * standardForm k n * M = LinearMap.BilinForm.toMatrix' (B.comp f f) := by
308           simpa [M, hBmatrix] using hmatrix.symm
309         _ = standardForm k n := by rw [hcomp, hBmatrix]
310     refine ⟨M, SymplecticGroup.mem_iff'.2 hM, ?_⟩
311     calc
312       Matrix.mulVec M e = f e := by simpa [M] using LinearMap.toMatrix'_mulVec f e
313       _ = v := hge

315 theorem exists_symplectic_column_eq {n : ℕ} (t : PhaseVar n)
316     {v : PhaseVar n → k} (hv : v ≠ 0) :
317     ∃ M : Matrix (PhaseVar n) (PhaseVar n) k,
318       M ∈ Matrix.symplecticGroup (Fin n) k ∧ (fun i => M i t) = v := by
319   have hb : phaseBasis k t ≠ 0 := by
320     intro h
321     have := congrFun h t
322     simp [phaseBasis] at this
323   rcases exists_symplectic_mulVec_eq k hb hv with ⟨M, hM, hMv⟩
324   refine ⟨M, hM, ?_⟩
325   rw [← mulVec_phaseBasis]
326   exact hMv

329     {n N : ℕ} (t : PhaseVar n) {P : SymbolRing k n}
330     (hP : P.IsHomogeneous N) (hne : P ≠ 0) (hN : 0 < N) :
331     ∃ M : Matrix (PhaseVar n) (PhaseVar n) k,

336   rcases exists_eval_ne_zero_of_homogeneous k hP hne with ⟨v, hv⟩

338     intro hvzero

340     apply hv
341     have hc : MvPolynomial.coeff 0 P = 0 :=
342       hP.coeff_eq_zero (by simpa using Nat.ne_of_lt hN)

350 theorem exists_symplectic_chart_matrices [CharZero k]
351     {n N : ℕ} (t : PhaseVar n) {P : SymbolRing k n}
352     (hP : P.IsHomogeneous N) (hne : P ≠ 0) (hN : 0 < N) :
353     ∃ M Ninv : Matrix (PhaseVar n) (PhaseVar n) k,
354       M * standardForm k n * Matrix.transpose M = standardForm k n ∧
355       Ninv * standardForm k n * Matrix.transpose Ninv = standardForm k n ∧
356       M * Ninv = 1 ∧ Ninv * M = 1 ∧
357       MvPolynomial.coeff (Finsupp.single () N)
358           (axisPolynomial k t
359             (Stafford38.CharacteristicLinearAction.symbolLinearAlgHom k M P)) ≠ 0 := by
360   rcases exists_symplectic_pureCoefficient_ne_zero k t hP hne hN with
361     ⟨M, hM, hcoeff⟩
362   let A : Matrix.symplecticGroup (Fin n) k := ⟨M, hM⟩
363   let Ninv : Matrix (PhaseVar n) (PhaseVar n) k := ↑(A⁻¹)
364   refine ⟨M, Ninv, SymplecticGroup.mem_iff.mp hM,
365     SymplecticGroup.mem_iff.mp (A⁻¹).property, ?_, ?_, hcoeff⟩
366   · dsimp only [Ninv]
367     calc
368       M * (↑(A⁻¹) : Matrix (PhaseVar n) (PhaseVar n) k) = ↑(A * A⁻¹) := rfl
369       _ = ↑(1 : Matrix.symplecticGroup (Fin n) k) := by simp
370       _ = 1 := rfl
371   · dsimp only [Ninv]
372     calc
373       (↑(A⁻¹) : Matrix (PhaseVar n) (PhaseVar n) k) * M = ↑(A⁻¹ * A) := rfl
374       _ = ↑(1 : Matrix.symplecticGroup (Fin n) k) := by simp
375       _ = 1 := rfl
