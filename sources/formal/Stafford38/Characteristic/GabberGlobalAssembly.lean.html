10 # Gabber involutivity for Weyl quotients
11 
12 Apply the Artinian-local ringCommutator theorem to the localized two-block Rees
13 deformation at each minimal prime. Contract bracket membership, then intersect
14 the minimal primes. The final theorem is field-generic and has no literature
15 or project assumption. The opposite-ring lifts are swapped explicitly to
16 preserve the right-module bracket orientation.
17 
18 
19 ## References and proof context
20 
21 [Gab81] Ofer Gabber, *The Integrability of the Characteristic Variety*, American Journal of Mathematics 103 (1981), no. 3, 445–468.
22 https://doi.org/10.2307/2374101
23 
24 This is the classical theorem implemented for cyclic Weyl quotients. Its exact radical-involutivity interface is proved here; see docs/literature.md and docs/proof-guide.md.
25 -/

51 theorem quotient_power_artinian
52     (R : Type u) [CommRing R] [IsLocalRing R] [IsNoetherianRing R]
53     (q : ℕ) (hq : 0 < q) :
54     IsArtinianRing (R ⧸ (IsLocalRing.maximalIdeal R) ^ q) := by
55   let m := IsLocalRing.maximalIdeal R
56   have hp : m ^ q ≠ ⊤ := ne_top_of_le_ne_top (IsLocalRing.maximalIdeal.isMaximal R).ne_top
57     (Ideal.pow_le_self hq.ne')
58   letI : Nontrivial (R ⧸ m ^ q) := Ideal.Quotient.nontrivial_iff.mpr hp
59   letI : IsLocalRing (R ⧸ m ^ q) :=
60     IsLocalRing.of_surjective' (Ideal.Quotient.mk (m ^ q)) Ideal.Quotient.mk_surjective
61   apply (isArtinianRing_iff_isNilpotent_maximalIdeal (R ⧸ m ^ q)).mpr
62   refine ⟨q, ?_⟩
63   rw [← IsLocalRing.map_maximalIdeal_of_surjective
64     (Ideal.Quotient.mk (m ^ q)) Ideal.Quotient.mk_surjective, ← Ideal.map_pow]
65   exact Ideal.map_quotient_self (m ^ q)

82 theorem localized_cofactor_mem
83     (q : ℕ) (hq : 0 < q)
84     (hpow : IsLocalRing.maximalIdeal R ^ q ≤ Module.annihilator R G)
85     [Module.Finite R G] [Nontrivial G]
86     (x y z : C q)
87     (hx : localizedTwoBlockSpecialization D S q x ∈
88       (IsLocalRing.maximalIdeal R).map (Ideal.Quotient.mk (localizedDoubledPower S q)))
89     (hy : localizedTwoBlockSpecialization D S q y ∈
90       (IsLocalRing.maximalIdeal R).map (Ideal.Quotient.mk (localizedDoubledPower S q)))
91     (hxy : x * y - y * x =
92       (concreteLeftPrincipalParameterReduction k I S q).parameter * z) :
93     localizedTwoBlockSpecialization D S q z ∈
94       (IsLocalRing.maximalIdeal R).map (Ideal.Quotient.mk (localizedDoubledPower S q)) := by
95   let H := localizedDoubledPower S q
96   let Rq := R ⧸ H
97   have hH : H = (IsLocalRing.maximalIdeal R) ^ (q + q) := by
98     exact (pow_add (IsLocalRing.maximalIdeal R) q q).symm
99   have hproper : H ≠ ⊤ := by
100     rw [hH]
101     exact ne_top_of_le_ne_top (IsLocalRing.maximalIdeal.isMaximal R).ne_top
102       (Ideal.pow_le_self (by omega))
103   letI : Nontrivial Rq := Ideal.Quotient.nontrivial_iff.mpr hproper
104   letI : IsLocalRing Rq := IsLocalRing.of_surjective'
105     (Ideal.Quotient.mk H) Ideal.Quotient.mk_surjective
106   letI : IsArtinianRing Rq := by
107     dsimp [Rq]
108     rw [hH]
109     exact quotient_power_artinian R (q + q) (by omega)
110   letI : Module Rq G := localizedDoubledPowerQuotientModule S q hpow
111   letI : IsScalarTower R Rq G := by
112     constructor
113     intro a b v
114     induction b using Quotient.inductionOn' with
115     | _ b => exact mul_smul a b v
116   letI : Module.Finite Rq G := Module.Finite.of_restrictScalars_finite R Rq G
117   letI : Module (C q) Wₗ := localizedTwoBlockModule D S q hpow
118   let E := concreteLeftPrincipalParameterReduction k I S q
119   let rho := localizedDoubledPowerModuleSpecialization D S q hpow
120   have hparam : parameterAct (W := Wₗ) E.parameter = localizedCAct D S := by
121     ext w
122     exact localizedTwoBlock_parameter_smul D S q hpow w
123   have hact : ∀ a w, rho (a • w) = E.modParameter a • rho w :=
124     localizedDoubledPowerModuleSpecialization_action D S q hpow
125   have hker : AddMonoidHom.ker rho = AddMonoidHom.range (parameterAct (W := Wₗ) E.parameter) := by
126     rw [hparam]
127     exact localizedDoubledPowerModuleSpecialization_ker D S q hpow
128   have hexact : AddMonoidHom.ker (parameterAct (W := Wₗ) E.parameter) =
129       AddMonoidHom.range (parameterAct (W := Wₗ) E.parameter) := by
130     rw [hparam]
131     exact localizedCAct_ker_eq_range D S
132   have hc2 : E.parameter * E.parameter = 0 := by
133     rw [← pow_two]
134     change ((localizedTwoBlockIdeal D S q).ringCon.mk'
135       (OreLocalization.numeratorRingHom (MulOpposite.op (orderReesTwoJetParameter (n := n) k)))) ^ 2 = 0
136     exact localizedTwoBlock_parameter_sq D S q
137   have hmax : (IsLocalRing.maximalIdeal R).map (Ideal.Quotient.mk H) =
138       IsLocalRing.maximalIdeal Rq :=
139     IsLocalRing.map_maximalIdeal_of_surjective _ Ideal.Quotient.mk_surjective
140   rw [hmax] at hx hy ⊢
141   exact artinian_local_cofactor_mem_maximalIdeal k E rho
142     (localizedDoubledPowerModuleSpecialization_surjective D S q hpow)
143     hact hker hexact hc2 x y z hx hy hxy

146 theorem minimalPrime_isInvolutive
147     (hP : P ∈ (Module.annihilator (SymbolRing k n)
148       (OrderAssociatedGradedModule k I)).minimalPrimes) : IsInvolutive P := by
149   obtain ⟨h, core⟩ := exists_concrete_localizedPreArtinianCore k I P hP
150   letI := h
151   letI : Nontrivial G := core.fibre_nontrivial
152   letI : IsNoetherian R G :=
153     (isFiniteLength_iff_isNoetherian_isArtinian.mp core.fibre_finiteLength).1
154   obtain ⟨q, hq⟩ := core.annihilatingPower
155   have hpow : IsLocalRing.maximalIdeal R ^ (q + 1) ≤ Module.annihilator R G :=
156     (Ideal.pow_le_pow_right (Nat.le_succ q)).trans hq
157   intro f hf g hg
158   obtain ⟨a, b, z, ha, hb, hab, hz⟩ :=
159     RightSquareZeroTraceData.exists_lifts_commutator_factor k _ _ D f g
160   let φ : (OrderReesTwoJet (n := n) k)ᵐᵒᵖ →+* C (q + 1) :=
161     ((localizedTwoBlockIdeal D S (q + 1)).ringCon.mk').comp
162       OreLocalization.numeratorRingHom
163   let x := φ (MulOpposite.op b)
164   let y := φ (MulOpposite.op a)
165   let zq := φ (MulOpposite.op z)
166   let H := localizedDoubledPower S (q + 1)
167   have hspec (v : OrderReesTwoJet (n := n) k) :
168       localizedTwoBlockSpecialization D S (q + 1) (φ (MulOpposite.op v)) =
169         Ideal.Quotient.mk H (algebraMap (SymbolRing k n) R ((D).pi v)) := by
170     rw [show φ (MulOpposite.op v) =
171       (localizedTwoBlockIdeal D S (q + 1)).ringCon.mk'
172         (OreLocalization.numeratorRingHom (MulOpposite.op v)) from rfl,
173       localizedTwoBlockSpecialization_mk]
174     change Ideal.Quotient.mk H (localizedOppositeSpecialization D S
175       (OreLocalization.numeratorRingHom (MulOpposite.op v))) = _
176     exact congrArg (Ideal.Quotient.mk H)
177       (localizedOppositeSpecialization_numerator D S (MulOpposite.op v))
178   have hx : localizedTwoBlockSpecialization D S (q + 1) x ∈
179       (IsLocalRing.maximalIdeal R).map (Ideal.Quotient.mk H) := by
180     rw [show x = φ (MulOpposite.op b) from rfl, hspec, hb]
181     exact Ideal.mem_map_of_mem _ ((IsLocalization.AtPrime.to_map_mem_maximal_iff R P g).mpr hg)
182   have hy : localizedTwoBlockSpecialization D S (q + 1) y ∈
183       (IsLocalRing.maximalIdeal R).map (Ideal.Quotient.mk H) := by
184     rw [show y = φ (MulOpposite.op a) from rfl, hspec, ha]
185     exact Ideal.mem_map_of_mem _ ((IsLocalization.AtPrime.to_map_mem_maximal_iff R P f).mpr hf)
186   have hop : MulOpposite.op b * MulOpposite.op a - MulOpposite.op a * MulOpposite.op b =
187       MulOpposite.op (D).c * MulOpposite.op z := by
188     apply MulOpposite.unop_injective
189     change a * b - b * a = z * (D).c
190     change a * b - b * a = (D).c * z at hab
191     exact hab.trans ((D).c_center.comm z).eq
192   have hxy : x * y - y * x =
193       (concreteLeftPrincipalParameterReduction k I S (q + 1)).parameter * zq := by
194     have hparam : (concreteLeftPrincipalParameterReduction k I S (q + 1)).parameter =
195         φ (MulOpposite.op (D).c) := rfl
196     rw [hparam]
197     have hh := congrArg φ hop
198     simpa only [map_sub, map_mul] using hh
199   have hlocal := localized_cofactor_mem k I P (q + 1) (by omega) hpow x y zq hx hy hxy
200   rw [show zq = φ (MulOpposite.op z) from rfl, hspec, hz,
201     filteredQuotientTwoJetTraceData_bracket] at hlocal
202   have hHle : H ≤ IsLocalRing.maximalIdeal R :=
203     Ideal.mul_le_left.trans (Ideal.pow_le_self (by omega))
204   have hneg : algebraMap (SymbolRing k n) R (-poissonBracket f g) ∈
205       IsLocalRing.maximalIdeal R := by
206     have hc := (Ideal.mem_comap).mpr hlocal
207     rw [Ideal.comap_map_of_surjective (Ideal.Quotient.mk H)
208       Ideal.Quotient.mk_surjective, ← RingHom.ker_eq_comap_bot,
209       Ideal.mk_ker, sup_eq_left.mpr hHle] at hc
210     exact hc
211   exact (P.neg_mem_iff).mp ((IsLocalization.AtPrime.to_map_mem_maximal_iff R P _).mp hneg)

214 theorem associatedGraded_radical_isInvolutive :
215     IsInvolutive (Module.annihilator (SymbolRing k n)
216       (OrderAssociatedGradedModule k I)).radical := by
217   let J := Module.annihilator (SymbolRing k n) (OrderAssociatedGradedModule k I)
218   intro f hf g hg
219   rw [← J.sInf_minimalPrimes]
220   apply Ideal.mem_sInf.mpr
221   intro P hP
222   letI : P.IsPrime := hP.1.1
223   have hle : J.radical ≤ P := hP.1.1.radical_le_iff.mpr hP.1.2
224   exact minimalPrime_isInvolutive k I P hP f (hle hf) g (hle hg)
