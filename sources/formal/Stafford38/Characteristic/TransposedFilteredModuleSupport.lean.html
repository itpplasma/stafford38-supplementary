10 /-!
11 # Transposition of the filtered right quotient and its support
12 
13 This file turns the concrete order filtration on a right Weyl quotient into
14 the corresponding filtration of the transposed left module.  On associated
15 graded modules the transported action is momentum-sign substitution.  The
16 resulting support is therefore the inverse image of the original support
17 under `symbolTranspositionEquiv`.
18 
19 No noncharacteristic restriction theorem or D-module comparison theorem is
20 used here.
21 -/

42 /-! ## The transported filtered quotient -/

44 /-- The actual right quotient, regarded as a distinct type so that it can
45 carry the left Weyl action obtained by transposition. -/
46 structure TransposedFilteredRightQuotient
47     (I : RightIdeal (WeylAlg k n)) where
48   /-- The underlying class in the original right quotient. -/
49   toRightQuotient : RightQuotient I
50 
51 /-- Forget the transported structure. -/
52 def transposedFilteredRightQuotientEquiv
53     (I : RightIdeal (WeylAlg k n)) :
54     TransposedFilteredRightQuotient k I ≃ RightQuotient I where
55   toFun := TransposedFilteredRightQuotient.toRightQuotient
56   invFun := TransposedFilteredRightQuotient.mk
57   left_inv q := by cases q; rfl
58   right_inv _ := rfl
59 
60 instance (I : RightIdeal (WeylAlg k n)) : AddCommGroup
61     (TransposedFilteredRightQuotient k I) :=
62   Equiv.addCommGroup (transposedFilteredRightQuotientEquiv k I)
63 
64 /-- Additive form of the forgetful equivalence. -/
65 def transposedFilteredRightQuotientAddEquiv
66     (I : RightIdeal (WeylAlg k n)) :
67     TransposedFilteredRightQuotient k I ≃+ RightQuotient I :=
68   Equiv.addEquiv (transposedFilteredRightQuotientEquiv k I)
69 
70 instance (I : RightIdeal (WeylAlg k n)) : Module k
71     (TransposedFilteredRightQuotient k I) :=
72   (transposedFilteredRightQuotientAddEquiv k I).module k
73 
74 /-- The quotient's right action restricted along Weyl transposition. -/
75 noncomputable instance (I : RightIdeal (WeylAlg k n)) :
76     Module (WeylAlg k n) (TransposedFilteredRightQuotient k I) :=
77   letI : Module (WeylAlg k n) (RightQuotient I) :=
78     transposedLeftModule k n (RightQuotient I)
79   (transposedFilteredRightQuotientAddEquiv k I).module
80     (WeylAlg k n)

82 /-- The canonical identification with the filtered `k`-linear quotient. -/
83 def filteredQuotientToTransposedLinearEquiv
84     (I : RightIdeal (WeylAlg k n)) :
85     FilteredRightQuotient k I ≃ₗ[k] TransposedFilteredRightQuotient k I :=
86   (filteredRightQuotientEquivRightQuotient k I).trans
87     { (transposedFilteredRightQuotientAddEquiv k I).symm with
88       map_smul' := by intro c q; rfl }

90 /-- The transported degree-`N` filtration piece. -/
91 def transposedQuotientOrderPiece
92     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
93     Submodule k (TransposedFilteredRightQuotient k I) :=
94   (quotientOrderPiece k I N).map
95     (filteredQuotientToTransposedLinearEquiv k I).toLinearMap
96 
97 /-- The transported quotient filtration is increasing. -/
98 theorem transposedQuotientOrderPiece_mono
99     (I : RightIdeal (WeylAlg k n)) {N M : ℕ} (hNM : N ≤ M) :
100     transposedQuotientOrderPiece k I N ≤
101       transposedQuotientOrderPiece k I M :=
102   Submodule.map_mono (Submodule.map_mono
103     (presentedWeightPiece_mono k orderWeight hNM))
104 
105 @[simp] theorem filteredQuotientToTransposedLinearEquiv_mk
106     (I : RightIdeal (WeylAlg k n)) (a : WeylAlg k n) :
107     filteredQuotientToTransposedLinearEquiv k I
108         (Submodule.Quotient.mk a) =
109       TransposedFilteredRightQuotient.mk (qmk I a) :=
110   rfl
111 
112 @[simp] theorem transposed_smul_qmk
113     (I : RightIdeal (WeylAlg k n))
114     (a b : WeylAlg k n) :
115     a • TransposedFilteredRightQuotient.mk (qmk I b) =
116       TransposedFilteredRightQuotient.mk
117         (qmk I (b * transpose k n a)) := by
118   apply (transposedFilteredRightQuotientEquiv k I).injective
119   change rightMul I (transpose k n a) (qmk I b) =
120     qmk I (b * transpose k n a)
121   exact (qmk_right_mul I b (transpose k n a)).symm

123 /-- The transported filtration is compatible with the transposed left
124 action: an operator of order at most `M` sends the degree-`N` piece into the
125 degree-`N+M` piece. -/
126 theorem smul_mem_transposedQuotientOrderPiece
127     (I : RightIdeal (WeylAlg k n)) {N M : ℕ}
128     {a : WeylAlg k n} (ha : a ∈ orderPiece k n M)
129     {q : TransposedFilteredRightQuotient k I}
130     (hq : q ∈ transposedQuotientOrderPiece k I N) :
131     a • q ∈ transposedQuotientOrderPiece k I (N + M) := by
132   rcases hq with ⟨q₀, hq₀, rfl⟩
133   rcases hq₀ with ⟨z, hz, rfl⟩
134   refine ⟨(Submodule.Quotient.mk
135       (z * transpose k n a) : FilteredRightQuotient k I), ?_, ?_⟩
136   · exact ⟨z * transpose k n a,
137         mul_mem_orderPiece k hz (transpose_mem_orderPiece k ha), rfl⟩
138   · exact (transposed_smul_qmk k I a z).symm
139 
140 /-- Every quotient class occurs in a finite transported order piece. -/
141 theorem exists_mem_transposedQuotientOrderPiece
142     (I : RightIdeal (WeylAlg k n))
143     (q : TransposedFilteredRightQuotient k I) :
144     ∃ N, q ∈ transposedQuotientOrderPiece k I N := by
145   rcases q with ⟨q⟩
146   refine Submodule.Quotient.induction_on I q ?_
147   intro z
148   obtain ⟨N, hz⟩ := exists_mem_orderPiece k z
149   refine ⟨N, Submodule.mem_map.mpr ⟨Submodule.Quotient.mk z, ?_, rfl⟩⟩
150   exact Submodule.mem_map.mpr ⟨z, hz, rfl⟩
151 
152 /-- Cyclic good-filtration form: every class in degree `N` is obtained by an
153 operator in `F_N A` acting on the class of one. -/
154 theorem exists_filtered_smul_one_of_mem
155     (I : RightIdeal (WeylAlg k n)) {N : ℕ}
156     (q : TransposedFilteredRightQuotient k I)
157     (hq : q ∈ transposedQuotientOrderPiece k I N) :
158     ∃ a : WeylAlg k n, a ∈ orderPiece k n N ∧
159       q = a • TransposedFilteredRightQuotient.mk (qmk I 1) := by
160   rcases hq with ⟨q₀, hq₀, rfl⟩
161   rcases hq₀ with ⟨z, hz, rfl⟩
162   refine ⟨transpose k n z, transpose_mem_orderPiece k hz, ?_⟩
163   change filteredQuotientToTransposedLinearEquiv k I
164       (Submodule.Quotient.mk z) =
165     transpose k n z • TransposedFilteredRightQuotient.mk (qmk I 1)
166   rw [filteredQuotientToTransposedLinearEquiv_mk,
167     transposed_smul_qmk, transpose_transpose, one_mul]

214 /-- The associated graded action is the one induced by the filtered
215 transposed action: on representatives it multiplies on the right by the
216 transpose of the acting Weyl operator. -/
217 theorem transposedSymbol_smul_of_mk_eq_transpose_mul
218     (I : RightIdeal (WeylAlg k n)) {N M : ℕ}
219     (z : orderPiece k n N) (a : orderPiece k n M) :
220     (presentedPrincipalComponent k orderWeight M a : SymbolRing k n) •
221         TransposedOrderAssociatedGradedModule.mk
222           (orderAssociatedGradedOf k I N
223             (orderPieceToQuotientGraded k I N z)) =
224       TransposedOrderAssociatedGradedModule.mk
225         (orderAssociatedGradedOf k I (N + M)
226           (orderPieceToQuotientGraded k I (N + M)
227             ⟨(z : WeylAlg k n) * transpose k n a,
228               mul_mem_orderPiece k z.property
229                 (transpose_mem_orderPiece k a.property)⟩)) := by
230   apply (transposedOrderAssociatedGradedEquiv k I).injective
231   change symbolTransposition k
232       (presentedPrincipalComponent k (@orderWeight n) M a) •
233         orderAssociatedGradedOf k I N
234           (orderPieceToQuotientGraded k I N z) = _
235   rw [← principal_transpose_order k a.property]
236   exact smul_orderAssociatedGradedOf_mk_eq_of_mul k I z
237     ⟨transpose k n a, transpose_mem_orderPiece k a.property⟩

239 /-- The annihilator of the transported associated graded module is the
240 inverse image of the original annihilator under symbol transposition. -/
241 theorem annihilator_transposedOrderAssociatedGradedModule
242     (I : RightIdeal (WeylAlg k n)) :
243     Module.annihilator (SymbolRing k n)
244         (TransposedOrderAssociatedGradedModule k I) =
245       (Module.annihilator (SymbolRing k n)
246         (OrderAssociatedGradedModule k I)).comap
247           (symbolTranspositionEquiv k).toRingEquiv.toRingHom := by
248   ext P
249   simp only [Module.mem_annihilator, Ideal.mem_comap]
250   constructor <;> intro h q
251   · have hq := congrArg
252         TransposedOrderAssociatedGradedModule.toOrderAssociatedGradedModule
253         (h (TransposedOrderAssociatedGradedModule.mk q))
254     exact hq
255   · apply (transposedOrderAssociatedGradedEquiv k I).injective
256     exact h q.toOrderAssociatedGradedModule

258 /-- The original actual associated graded quotient is cyclic and therefore
259 finite over the symbol ring. -/
260 noncomputable instance (I : RightIdeal (WeylAlg k n)) :
261     Module.Finite (SymbolRing k n) (OrderAssociatedGradedModule k I) := by
262   let g := orderAssociatedGradedGenerator k I
263   let f : SymbolRing k n →ₗ[SymbolRing k n]
264       OrderAssociatedGradedModule k I :=
265     { toFun := fun P => P • g
266       map_add' := by intro P Q; exact add_smul P Q g
267       map_smul' := by intro P Q; exact mul_smul P Q g }
268   apply Module.Finite.of_surjective f
269   intro q
270   exact exists_smul_orderAssociatedGradedGenerator k I q
271 
272 /-- The transported associated graded module remains cyclic, hence finite. -/
273 noncomputable instance (I : RightIdeal (WeylAlg k n)) :
274     Module.Finite (SymbolRing k n)
275       (TransposedOrderAssociatedGradedModule k I) := by
276   let g : TransposedOrderAssociatedGradedModule k I :=
277     TransposedOrderAssociatedGradedModule.mk
278       (orderAssociatedGradedGenerator k I)
279   let f : SymbolRing k n →ₗ[SymbolRing k n]
280       TransposedOrderAssociatedGradedModule k I :=
281     { toFun := fun P => P • g
282       map_add' := by intro P Q; exact add_smul P Q g
283       map_smul' := by intro P Q; exact mul_smul P Q g }
284   apply Module.Finite.of_surjective f
285   intro q
286   obtain ⟨P, hP⟩ := exists_smul_orderAssociatedGradedGenerator k I
287     q.toOrderAssociatedGradedModule
288   refine ⟨symbolTransposition k P, ?_⟩
289   apply (transposedOrderAssociatedGradedEquiv k I).injective
290   change symbolTransposition k (symbolTransposition k P) •
291       orderAssociatedGradedGenerator k I = q.toOrderAssociatedGradedModule
292   rw [show symbolTransposition k (symbolTransposition k P) = P by
293     exact AlgHom.congr_fun (symbolTransposition_comp_self k) P]
294   exact hP

303 /-- Transposition carries associated-graded support by the expected
304 momentum-sign automorphism of the symbol spectrum. -/
305 theorem transposedOrderAssociatedGradedSupport_eq_preimage
306     (I : RightIdeal (WeylAlg k n)) :
307     transposedOrderAssociatedGradedSupport k I =
308       PrimeSpectrum.comap
309         (symbolTranspositionEquiv k).toRingEquiv.toRingHom ⁻¹'
310         Module.support (SymbolRing k n)
311           (OrderAssociatedGradedModule k I) := by
312   rw [transposedOrderAssociatedGradedSupport,
313     Module.support_eq_zeroLocus, Module.support_eq_zeroLocus,
314     annihilator_transposedOrderAssociatedGradedModule]
315   ext p
316   simp only [PrimeSpectrum.mem_zeroLocus, Set.mem_preimage,
317     PrimeSpectrum.comap_asIdeal, SetLike.coe_subset_coe]
318   constructor
319   · intro h P hP
320     have hP' : symbolTransposition k P ∈
321         (Module.annihilator (SymbolRing k n)
322           (OrderAssociatedGradedModule k I)).comap
323             (symbolTranspositionEquiv k).toRingEquiv.toRingHom := by
324       change symbolTransposition k (symbolTransposition k P) ∈
325         Module.annihilator (SymbolRing k n)
326           (OrderAssociatedGradedModule k I)
327       rwa [show symbolTransposition k (symbolTransposition k P) = P by
328         exact AlgHom.congr_fun (symbolTransposition_comp_self k) P]
329     exact h hP'
330   · intro h P hP
331     have hP' : symbolTransposition k P ∈
332         Module.annihilator (SymbolRing k n)
333           (OrderAssociatedGradedModule k I) := hP
334     have h' := h hP'
335     change symbolTransposition k (symbolTransposition k P) ∈ p.asIdeal at h'
336     rwa [show symbolTransposition k (symbolTransposition k P) = P by
337       exact AlgHom.congr_fun (symbolTransposition_comp_self k) P] at h'
