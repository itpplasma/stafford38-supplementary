8 /-!
9 # Transposition of the filtered right quotient and its support
10 
11 This file turns the concrete order filtration on a right Weyl quotient into
12 the corresponding filtration of the transposed left module.  On associated
13 graded modules the transported action is momentum-sign substitution.  The
14 resulting support is therefore the inverse image of the original support
15 under `symbolTranspositionEquiv`.
16 
17 No noncharacteristic restriction theorem or D-module comparison theorem is
18 used here.
19 -/

42 /-- The actual right quotient, regarded as a distinct type so that it can
43 carry the left Weyl action obtained by transposition. -/
44 structure TransposedFilteredRightQuotient
45     (I : RightIdeal (PresentedWeyl k n)) where
46   /-- The underlying class in the original right quotient. -/
47   toRightQuotient : RightQuotient I
48 
49 /-- Forget the transported structure. -/
50 def transposedFilteredRightQuotientEquiv
51     (I : RightIdeal (PresentedWeyl k n)) :
52     TransposedFilteredRightQuotient k I ≃ RightQuotient I where
53   toFun := TransposedFilteredRightQuotient.toRightQuotient
54   invFun := TransposedFilteredRightQuotient.mk
55   left_inv q := by cases q; rfl
56   right_inv _ := rfl
57 
58 instance (I : RightIdeal (PresentedWeyl k n)) : AddCommGroup
59     (TransposedFilteredRightQuotient k I) :=
60   Equiv.addCommGroup (transposedFilteredRightQuotientEquiv k I)
61 
62 /-- Additive form of the forgetful equivalence. -/
63 def transposedFilteredRightQuotientAddEquiv
64     (I : RightIdeal (PresentedWeyl k n)) :
65     TransposedFilteredRightQuotient k I ≃+ RightQuotient I :=
66   Equiv.addEquiv (transposedFilteredRightQuotientEquiv k I)
67 
68 instance (I : RightIdeal (PresentedWeyl k n)) : Module k
69     (TransposedFilteredRightQuotient k I) :=
70   (transposedFilteredRightQuotientAddEquiv k I).module k
71 
72 /-- The quotient's right action restricted along Weyl transposition. -/
73 noncomputable instance (I : RightIdeal (PresentedWeyl k n)) :
74     Module (PresentedWeyl k n) (TransposedFilteredRightQuotient k I) :=
75   letI : Module (PresentedWeyl k n) (RightQuotient I) :=
76     transposedLeftModule k n (RightQuotient I)
77   (transposedFilteredRightQuotientAddEquiv k I).module
78     (PresentedWeyl k n)

80 /-- The canonical identification with the filtered `k`-linear quotient. -/
81 def filteredQuotientToTransposedLinearEquiv
82     (I : RightIdeal (PresentedWeyl k n)) :
83     FilteredRightQuotient k I ≃ₗ[k] TransposedFilteredRightQuotient k I :=
84   (filteredRightQuotientEquivRightQuotient k I).trans
85     { (transposedFilteredRightQuotientAddEquiv k I).symm with
86       map_smul' := by intro c q; rfl }

88 /-- The transported degree-`N` filtration piece. -/
89 def transposedQuotientOrderPiece
90     (I : RightIdeal (PresentedWeyl k n)) (N : ℕ) :
91     Submodule k (TransposedFilteredRightQuotient k I) :=
92   (quotientOrderPiece k I N).map
93     (filteredQuotientToTransposedLinearEquiv k I).toLinearMap
94 
95 /-- The transported quotient filtration is increasing. -/
96 theorem transposedQuotientOrderPiece_mono
97     (I : RightIdeal (PresentedWeyl k n)) {N M : ℕ} (hNM : N ≤ M) :
98     transposedQuotientOrderPiece k I N ≤
99       transposedQuotientOrderPiece k I M :=
100   Submodule.map_mono (Submodule.map_mono
101     (presentedWeightPiece_mono k orderWeight hNM))

103 @[simp] theorem filteredQuotientToTransposedLinearEquiv_mk
104     (I : RightIdeal (PresentedWeyl k n)) (a : PresentedWeyl k n) :
105     filteredQuotientToTransposedLinearEquiv k I
106         (Submodule.Quotient.mk a) =
107       TransposedFilteredRightQuotient.mk (qmk I a) :=
108   rfl

110 @[simp] theorem transposed_smul_qmk
111     (I : RightIdeal (PresentedWeyl k n))
112     (a b : PresentedWeyl k n) :
113     a • TransposedFilteredRightQuotient.mk (qmk I b) =
114       TransposedFilteredRightQuotient.mk
115         (qmk I (b * transpose k n a)) := by
116   apply (transposedFilteredRightQuotientEquiv k I).injective
117   change rightMul I (transpose k n a) (qmk I b) =
118     qmk I (b * transpose k n a)
119   exact (qmk_right_mul I b (transpose k n a)).symm

121 /-- The transported filtration is compatible with the transposed left
122 action: an operator of order at most `M` sends the degree-`N` piece into the
123 degree-`N+M` piece. -/
124 theorem smul_mem_transposedQuotientOrderPiece
125     (I : RightIdeal (PresentedWeyl k n)) {N M : ℕ}
126     {a : PresentedWeyl k n} (ha : a ∈ orderPiece k n M)
127     {q : TransposedFilteredRightQuotient k I}
128     (hq : q ∈ transposedQuotientOrderPiece k I N) :
129     a • q ∈ transposedQuotientOrderPiece k I (N + M) := by
130   rcases hq with ⟨q₀, hq₀, rfl⟩
131   rcases hq₀ with ⟨z, hz, rfl⟩
132   refine ⟨(Submodule.Quotient.mk
133       (z * transpose k n a) : FilteredRightQuotient k I), ?_, ?_⟩
134   · exact ⟨z * transpose k n a,
135         mul_mem_orderPiece k hz (transpose_mem_orderPiece k ha), rfl⟩
136   · exact (transposed_smul_qmk k I a z).symm
137 
138 /-- Every quotient class occurs in a finite transported order piece. -/
139 theorem exists_mem_transposedQuotientOrderPiece
140     (I : RightIdeal (PresentedWeyl k n))
141     (q : TransposedFilteredRightQuotient k I) :
142     ∃ N, q ∈ transposedQuotientOrderPiece k I N := by
143   rcases q with ⟨q⟩
144   refine Submodule.Quotient.induction_on I q ?_
145   intro z
146   obtain ⟨N, hz⟩ := exists_mem_orderPiece k z
147   refine ⟨N, Submodule.mem_map.mpr ⟨Submodule.Quotient.mk z, ?_, rfl⟩⟩
148   exact Submodule.mem_map.mpr ⟨z, hz, rfl⟩
149 
150 /-- Cyclic good-filtration form: every class in degree `N` is obtained by an
151 operator in `F_N A` acting on the class of one. -/
152 theorem exists_filtered_smul_one_of_mem
153     (I : RightIdeal (PresentedWeyl k n)) {N : ℕ}
154     (q : TransposedFilteredRightQuotient k I)
155     (hq : q ∈ transposedQuotientOrderPiece k I N) :
156     ∃ a : PresentedWeyl k n, a ∈ orderPiece k n N ∧
157       q = a • TransposedFilteredRightQuotient.mk (qmk I 1) := by
158   rcases hq with ⟨q₀, hq₀, rfl⟩
159   rcases hq₀ with ⟨z, hz, rfl⟩
160   refine ⟨transpose k n z, transpose_mem_orderPiece k hz, ?_⟩
161   change filteredQuotientToTransposedLinearEquiv k I
162       (Submodule.Quotient.mk z) =
163     transpose k n z • TransposedFilteredRightQuotient.mk (qmk I 1)
164   rw [filteredQuotientToTransposedLinearEquiv_mk,
165     transposed_smul_qmk, transpose_transpose, one_mul]

212 /-- The associated graded action is the one induced by the filtered
213 transposed action: on representatives it multiplies on the right by the
214 transpose of the acting Weyl operator. -/
215 theorem transposedSymbol_smul_of_mk_eq_transpose_mul
216     (I : RightIdeal (PresentedWeyl k n)) {N M : ℕ}
217     (z : orderPiece k n N) (a : orderPiece k n M) :
218     (presentedPrincipalComponent k orderWeight M a : SymbolRing k n) •
219         TransposedOrderAssociatedGradedModule.mk
220           (orderAssociatedGradedOf k I N
221             (orderPieceToQuotientGraded k I N z)) =
222       TransposedOrderAssociatedGradedModule.mk
223         (orderAssociatedGradedOf k I (N + M)
224           (orderPieceToQuotientGraded k I (N + M)
225             ⟨(z : PresentedWeyl k n) * transpose k n a,
226               mul_mem_orderPiece k z.property
227                 (transpose_mem_orderPiece k a.property)⟩)) := by
228   apply (transposedOrderAssociatedGradedEquiv k I).injective
229   change symbolTransposition k
230       (presentedPrincipalComponent k (@orderWeight n) M a) •
231         orderAssociatedGradedOf k I N
232           (orderPieceToQuotientGraded k I N z) = _
233   rw [← principal_transpose_order k a.property]
234   exact smul_orderAssociatedGradedOf_mk_eq_of_mul k I z
235     ⟨transpose k n a, transpose_mem_orderPiece k a.property⟩

237 /-- The annihilator of the transported associated graded module is the
238 inverse image of the original annihilator under symbol transposition. -/
239 theorem annihilator_transposedOrderAssociatedGradedModule
240     (I : RightIdeal (PresentedWeyl k n)) :
241     Module.annihilator (SymbolRing k n)
242         (TransposedOrderAssociatedGradedModule k I) =
243       (Module.annihilator (SymbolRing k n)
244         (OrderAssociatedGradedModule k I)).comap
245           (symbolTranspositionEquiv k).toRingEquiv.toRingHom := by
246   ext P
247   simp only [Module.mem_annihilator, Ideal.mem_comap]
248   constructor <;> intro h q
249   · have hq := congrArg
250         TransposedOrderAssociatedGradedModule.toOrderAssociatedGradedModule
251         (h (TransposedOrderAssociatedGradedModule.mk q))
252     exact hq
253   · apply (transposedOrderAssociatedGradedEquiv k I).injective
254     exact h q.toOrderAssociatedGradedModule

256 /-- The original actual associated graded quotient is cyclic and therefore
257 finite over the symbol ring. -/
258 noncomputable instance (I : RightIdeal (PresentedWeyl k n)) :
259     Module.Finite (SymbolRing k n) (OrderAssociatedGradedModule k I) := by
260   let g := orderAssociatedGradedGenerator k I
261   let f : SymbolRing k n →ₗ[SymbolRing k n]
262       OrderAssociatedGradedModule k I :=
263     { toFun := fun P => P • g
264       map_add' := by intro P Q; exact add_smul P Q g
265       map_smul' := by intro P Q; exact mul_smul P Q g }
266   apply Module.Finite.of_surjective f
267   intro q
268   exact exists_smul_orderAssociatedGradedGenerator k I q
269 
270 /-- The transported associated graded module remains cyclic, hence finite. -/
271 noncomputable instance (I : RightIdeal (PresentedWeyl k n)) :
272     Module.Finite (SymbolRing k n)
273       (TransposedOrderAssociatedGradedModule k I) := by
274   let g : TransposedOrderAssociatedGradedModule k I :=
275     TransposedOrderAssociatedGradedModule.mk
276       (orderAssociatedGradedGenerator k I)
277   let f : SymbolRing k n →ₗ[SymbolRing k n]
278       TransposedOrderAssociatedGradedModule k I :=
279     { toFun := fun P => P • g
280       map_add' := by intro P Q; exact add_smul P Q g
281       map_smul' := by intro P Q; exact mul_smul P Q g }
282   apply Module.Finite.of_surjective f
283   intro q
284   obtain ⟨P, hP⟩ := exists_smul_orderAssociatedGradedGenerator k I
285     q.toOrderAssociatedGradedModule
286   refine ⟨symbolTransposition k P, ?_⟩
287   apply (transposedOrderAssociatedGradedEquiv k I).injective
288   change symbolTransposition k (symbolTransposition k P) •
289       orderAssociatedGradedGenerator k I = q.toOrderAssociatedGradedModule
290   rw [show symbolTransposition k (symbolTransposition k P) = P by
291     exact AlgHom.congr_fun (symbolTransposition_comp_self k) P]
292   exact hP

301 /-- Transposition carries associated-graded support by the expected
302 momentum-sign automorphism of the symbol spectrum. -/
303 theorem transposedOrderAssociatedGradedSupport_eq_preimage
304     (I : RightIdeal (PresentedWeyl k n)) :
305     transposedOrderAssociatedGradedSupport k I =
306       PrimeSpectrum.comap
307         (symbolTranspositionEquiv k).toRingEquiv.toRingHom ⁻¹'
308         Module.support (SymbolRing k n)
309           (OrderAssociatedGradedModule k I) := by
310   rw [transposedOrderAssociatedGradedSupport,
311     Module.support_eq_zeroLocus, Module.support_eq_zeroLocus,
312     annihilator_transposedOrderAssociatedGradedModule]
313   ext p
314   simp only [PrimeSpectrum.mem_zeroLocus, Set.mem_preimage,
315     PrimeSpectrum.comap_asIdeal, SetLike.coe_subset_coe]
316   constructor
317   · intro h P hP
318     have hP' : symbolTransposition k P ∈
319         (Module.annihilator (SymbolRing k n)
320           (OrderAssociatedGradedModule k I)).comap
321             (symbolTranspositionEquiv k).toRingEquiv.toRingHom := by
322       change symbolTransposition k (symbolTransposition k P) ∈
323         Module.annihilator (SymbolRing k n)
324           (OrderAssociatedGradedModule k I)
325       rwa [show symbolTransposition k (symbolTransposition k P) = P by
326         exact AlgHom.congr_fun (symbolTransposition_comp_self k) P]
327     exact h hP'
328   · intro h P hP
329     have hP' : symbolTransposition k P ∈
330         Module.annihilator (SymbolRing k n)
331           (OrderAssociatedGradedModule k I) := hP
332     have h' := h hP'
333     change symbolTransposition k (symbolTransposition k P) ∈ p.asIdeal at h'
334     rwa [show symbolTransposition k (symbolTransposition k P) = P by
335       exact AlgHom.congr_fun (symbolTransposition_comp_self k) P] at h'
