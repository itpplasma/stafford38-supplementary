10 /-!
11 # Localization proof of componentwise involutivity
12 
13 This file formalizes the localization route in the paper's proof of
14 componentwise involutivity.  First extend each polynomial partial derivative
15 to fractions by the quotient rule, and use these derivations to define the
16 localized Poisson bracket.  Then localize the involutive radical, identify it
17 with the maximal ideal at a minimal prime, and contract bracket membership
18 using primeness.
19 -/

31 def localizedPoissonBracket (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
32     (f g : AtPrime P) : AtPrime P :=
33   ∑ i : Fin n,
34     (localizedPDeriv P (Sum.inl i) f * localizedPDeriv P (Sum.inr i) g -
35       localizedPDeriv P (Sum.inr i) f * localizedPDeriv P (Sum.inl i) g)

37 theorem localizedPoissonBracket_add_left
38     (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
39     (f₁ f₂ g : AtPrime P) :
40     localizedPoissonBracket P (f₁ + f₂) g =
41       localizedPoissonBracket P f₁ g + localizedPoissonBracket P f₂ g := by
42   unfold localizedPoissonBracket
43   simp only [localizedPDeriv_add, add_mul]
44   rw [← Finset.sum_add_distrib]
45   apply Finset.sum_congr rfl
46   intro i hi

111 /-- The involutive condition survives localization.  This is proved by
112 induction on the localized ideal's generators; the product identities are
113 the biderivation rules for the fraction-extended bracket above. -/
114 theorem localizedMap_isInvolutive
115     (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
116     (I : Ideal (R (k := k) (n := n))) (hI : IsInvolutive I) :
117     ∀ f ∈ I.map (algebraMap (R (k := k) (n := n)) (AtPrime P)),
118       ∀ g ∈ I.map (algebraMap (R (k := k) (n := n)) (AtPrime P)),
119         localizedPoissonBracket P f g ∈
120           I.map (algebraMap (R (k := k) (n := n)) (AtPrime P)) := by
121   let M : Ideal (AtPrime P) :=
122     I.map (algebraMap (R (k := k) (n := n)) (AtPrime P))
123   intro f hf
124   change f ∈ I.map (algebraMap (R (k := k) (n := n)) (AtPrime P)) at hf
125   rw [Ideal.map] at hf
126   induction hf using Submodule.span_induction with
127   | mem f hf =>
128       rcases hf with ⟨f₀, hf₀, rfl⟩
129       intro g hg
130       change g ∈ I.map (algebraMap (R (k := k) (n := n)) (AtPrime P)) at hg
131       rw [Ideal.map] at hg
132       induction hg using Submodule.span_induction with
133       | mem g hg =>
134           rcases hg with ⟨g₀, hg₀, rfl⟩
135           change localizedPoissonBracket P
136             (algebraMap (R (k := k) (n := n)) (AtPrime P) f₀)
137             (algebraMap (R (k := k) (n := n)) (AtPrime P) g₀) ∈ M
138           rw [localizedPoissonBracket_algebraMap]
139           exact Ideal.mem_map_of_mem _ (hI f₀ hf₀ g₀ hg₀)
140       | zero =>
141           rw [localizedPoissonBracket_zero_right]
142           exact M.zero_mem
143       | add g₁ g₂ _ _ hg₁ hg₂ =>
144           rw [localizedPoissonBracket_add_right]
145           exact M.add_mem hg₁ hg₂
146       | smul a g hgm hg =>
147           rw [smul_eq_mul, localizedPoissonBracket_mul_right]
148           exact M.add_mem (M.mul_mem_left _ hg) (M.mul_mem_right _ hgm)
149   | zero =>
150       intro g hg
151       rw [localizedPoissonBracket_zero_left]
152       exact M.zero_mem
153   | add f₁ f₂ _ _ hf₁ hf₂ =>
154       intro g hg
155       rw [localizedPoissonBracket_add_left]
156       exact M.add_mem (hf₁ g hg) (hf₂ g hg)
157   | smul a f hfm hf =>
158       intro g hg
159       rw [smul_eq_mul, localizedPoissonBracket_mul_left]
160       exact M.add_mem (M.mul_mem_left _ (hf g hg)) (M.mul_mem_right _ hfm)

162 /-- At a prime minimal over `J`, the radical of the localized ideal is the
163 localized prime.  The first equality is the general localization-radical
164 identity; the second is the minimal-prime theorem for localization. -/
165 theorem radical_map_eq_prime_map_of_minimal
166     (J P : Ideal (R (k := k) (n := n))) [P.IsPrime]
167     (hP : P ∈ J.minimalPrimes) :
168     J.radical.map (algebraMap (R (k := k) (n := n)) (AtPrime P)) =
169       P.map (algebraMap (R (k := k) (n := n)) (AtPrime P)) := by
170   rw [IsLocalization.map_radical (P.primeCompl) (AtPrime P)]
171   exact IsLocalization.AtPrime.radical_map_of_mem_minimalPrimes
172     (A := AtPrime P) P J hP

184 /-- Localization proof of the componentwise statement: involutivity of
185 `√J` implies involutivity of each minimal prime over `J`.  Unlike the
186 saturation proof in `RadicalMinimalPrimeInvolutivity`, this follows the paper's
187 argument through the localized Poisson structure, its maximal ideal, and
188 contraction by primeness. -/
189 theorem minimalPrimes_isInvolutive_of_radical_isInvolutive_viaLocalization
190     (J : Ideal (R (k := k) (n := n))) (hJ : IsInvolutive J.radical) :
191     ∀ P ∈ J.minimalPrimes, IsInvolutive P := by
192   intro P hP
193   letI : P.IsPrime := hP.1.1
194   intro f hf g hg
195   let φ := algebraMap (R (k := k) (n := n)) (AtPrime P)
196   have hrad := radical_map_eq_prime_map_of_minimal J P hP
197   have hfLoc : φ f ∈ P.map φ := Ideal.mem_map_of_mem φ hf
198   have hgLoc : φ g ∈ P.map φ := Ideal.mem_map_of_mem φ hg
199   rw [← hrad] at hfLoc hgLoc
200   have hbrLoc := localizedMap_isInvolutive P J.radical hJ (φ f) hfLoc (φ g) hgLoc
201   rw [localizedPoissonBracket_algebraMap] at hbrLoc
202   rw [hrad] at hbrLoc
203   obtain ⟨s, hs, hmul⟩ :=
204     (IsLocalization.algebraMap_mem_map_algebraMap_iff P.primeCompl
205       (AtPrime P) P (poissonBracket f g)).mp hbrLoc
206   have hprime : P.IsPrime := hP.1.1
207   rcases hprime.mem_or_mem hmul with hsP | hbr
208   · exact (hs hsP).elim
209   · exact hbr
