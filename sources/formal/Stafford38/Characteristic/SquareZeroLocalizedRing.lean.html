8 # The localized deformation-ring specialization
9 
10 The module localization used by the right Rees two-jet is over the opposite
11 deformation ring.  This file constructs the corresponding ring map from that
12 Ore localization to the ordinary commutative localization of the special
13 fibre.  It is the first ring-level input in the Artinian-local Gabber route.
14 
15 No assertion is made that the kernel is generated by the square-zero
16 parameter, that the target is an Artinian quotient, or that a coefficient
17 field exists.
18 -/

40 /-- Specialization is a ring homomorphism on the opposite deformation ring
41 because its target is commutative. -/
42 def oppositeSpecializationRingHom : Bᵐᵒᵖ →+* R where
43   toFun b := D.pi b.unop
44   map_one' := by simp
45   map_zero' := by simp
46   map_add' a b := by simp
47   map_mul' a b := by
48     change D.pi (b.unop * a.unop) = D.pi a.unop * D.pi b.unop
49     rw [map_mul, mul_comm]
50 
51 @[simp] theorem oppositeSpecializationRingHom_apply (b : Bᵐᵒᵖ) :
52     oppositeSpecializationRingHom D b = D.pi b.unop :=
53   rfl

57 variable (S : Submonoid R)
58 variable [OreLocalization.OreSet (OppositeDenominators D S)]

60 /-- The canonical numerator units in a commutative localization. -/
61 def localizationNumeratorUnits : S →* Units (Localization S) where
62   toFun := OreLocalization.numeratorUnit
63   map_one' := by
64     apply Units.ext
65     change (OreLocalization.numeratorHom : R →* Localization S) 1 = 1
66     exact map_one _
67   map_mul' a b := by
68     apply Units.ext
69     change (OreLocalization.numeratorHom : R →* Localization S)
70         ((a : R) * (b : R)) =
71       (OreLocalization.numeratorHom : R →* Localization S) a *
72         (OreLocalization.numeratorHom : R →* Localization S) b
73     exact map_mul _ _ _
74 
75 /-- Every pulled-back Ore denominator maps to its canonical numerator unit in
76 the commutative localization. -/
77 def localizedDenominatorUnits :
78     OppositeDenominators D S →* Units (Localization S) :=
79   (localizationNumeratorUnits S).comp (denominatorMap D S)
80 
81 @[simp] theorem localizedDenominatorUnits_val
82     (s : OppositeDenominators D S) :
83     (localizedDenominatorUnits D S s : Localization S) =
84       algebraMap R (Localization S) (denominatorMap D S s) := by
85   rfl
86 
87 /-- The specialization of the opposite deformation ring extends through its
88 Ore localization to the ordinary localization of the commutative fibre. -/
89 def localizedOppositeSpecialization :
90     OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ) →+* Localization S :=
91   OreLocalization.universalHom
92     ((algebraMap R (Localization S)).comp
93       (oppositeSpecializationRingHom D))
94     (localizedDenominatorUnits D S)
95     (fun s => by
96       rw [localizedDenominatorUnits_val]
97       rfl)
98 
99 @[simp] theorem localizedOppositeSpecialization_oreDiv
100     (b : Bᵐᵒᵖ) (s : OppositeDenominators D S) :
101     localizedOppositeSpecialization D S (b /ₒ s) =
102       ((localizedDenominatorUnits D S s)⁻¹ : Units (Localization S)) *
103         algebraMap R (Localization S) (D.pi b.unop) :=
104   rfl
105 
106 @[simp] theorem localizedOppositeSpecialization_numerator (b : Bᵐᵒᵖ) :
107     localizedOppositeSpecialization D S
108         (OreLocalization.numeratorHom b) =
109       algebraMap R (Localization S) (D.pi b.unop) := by
110   exact OreLocalization.universalHom_commutes _ _ _
111 
112 @[simp] theorem localizedOppositeSpecialization_parameter :
113     localizedOppositeSpecialization D S
114         (OreLocalization.numeratorHom (MulOpposite.op D.c)) = 0 := by
115   simp [D.pi_c]
116 
117 /-- Surjectivity of the original specialization survives localization. This
118 is a ring-level statement; it does not identify the kernel. -/
119 theorem localizedOppositeSpecialization_surjective :
120     Function.Surjective (localizedOppositeSpecialization D S) := by
121   intro z
122   refine Localization.induction_on z ?_
123   rintro ⟨r, s⟩
124   obtain ⟨a, ha⟩ := D.pi_surjective r
125   obtain ⟨b, hb⟩ := D.pi_surjective (s : R)
126   let sb : OppositeDenominators D S :=
127     ⟨MulOpposite.op b, by
128       change D.pi b ∈ S
129       rw [hb]
130       exact s.property⟩
131   refine ⟨(MulOpposite.op a) /ₒ sb, ?_⟩
132   rw [localizedOppositeSpecialization_oreDiv]
133   change
134     ((OreLocalization.numeratorUnit (denominatorMap D S sb))⁻¹ :
135         Units (Localization S)) *
136         algebraMap R (Localization S) (D.pi a) =
137       Localization.mk r s
138   rw [ha]
139   change (1 /ₒ denominatorMap D S sb) * (r /ₒ 1) = r /ₒ s
140   rw [OreLocalization.mul_div_one]
141   have hs : denominatorMap D S sb = s := Subtype.ext hb
142   rw [hs]
143   simp
144 
145 end Ore
