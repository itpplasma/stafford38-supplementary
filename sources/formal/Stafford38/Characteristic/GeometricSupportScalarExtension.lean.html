9 /-!
10 # Reduced order support after scalar extension
11 
12 The terminal Gabber consumer is formulated over a Laurent-series coefficient
13 field.  This file identifies its geometric reduced support with the ordinary
14 reduced order support of the scalar-extended canonical Weyl quotient.  Thus a
15 universal Gabber theorem can be run directly over the extension field; no
16 unrecorded transport of Poisson closure is needed.
17 -/

38 form needs no surjectivity of the coefficient map. -/

40     {σ : Type*} (I : Ideal (MvPolynomial σ k)) :
41     ((I.radical.map
42       (scalarPolynomialMap (k := k) (K := K) σ)).radical) =
43       ((I.map
44         (scalarPolynomialMap (k := k) (K := K) σ)).radical) := by
45   apply le_antisymm
46   · calc
47       (I.radical.map
48           (scalarPolynomialMap (k := k) (K := K) σ)).radical ≤
49           ((I.map
50             (scalarPolynomialMap (k := k) (K := K) σ)).radical).radical :=
51         Ideal.radical_mono
52           (Ideal.map_radical_le
53             (f := scalarPolynomialMap (k := k) (K := K) σ) (I := I))
54       _ = (I.map
55           (scalarPolynomialMap (k := k) (K := K) σ)).radical :=
56         Ideal.radical_idem _

58     exact Ideal.map_mono Ideal.le_radical
