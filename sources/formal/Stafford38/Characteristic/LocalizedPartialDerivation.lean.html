9 /-!
10 # Polynomial partial derivatives on a prime localization
11 
12 The partial derivative on the symbol ring is extended by the generic
13 localization operation from AlgebraicAnalysis.  The quotient-rule formula
14 used by the localization argument is recorded below as a proved
15 specialization of the generic Leibniz/inverse formula.
16 -/

33 /-- The `i`th polynomial partial derivative, extended through the localization
34 at `P`.  This is the generic AlgebraicAnalysis extension specialized to the
35 phase-space partial derivative. -/
36 noncomputable def localizedPDeriv
37     (P : Ideal (R (k := k) (n := n))) [P.IsPrime] (i : PhaseVar n) :
38     Derivation k (AtPrime (k := k) (n := n) P) (AtPrime P) :=
39   extendDerivation k (R (k := k) (n := n)) (AtPrime P) P.primeCompl
40     ((Algebra.linearMap (R (k := k) (n := n)) (AtPrime P)).compDer
41       (MvPolynomial.pderiv i))

46       (Algebra.linearMap (R (k := k) (n := n)) (AtPrime P)).compDer

55 /-- Exact quotient-rule formula for the canonical localization extension.
56 The denominator square is a localization unit, and the formula follows by
57 specializing the generic cross-multiplied Leibniz rule. -/
58 theorem localizedPDeriv_mk
59     (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
60     (i : PhaseVar n) (a : R) (s : P.primeCompl) :
61     extendDerivation k (R (k := k) (n := n)) (AtPrime P) P.primeCompl
62       ((Algebra.linearMap (R (k := k) (n := n)) (AtPrime P)).compDer
63         (MvPolynomial.pderiv i))
64       (IsLocalization.mk' (AtPrime P) a s) =
65       IsLocalization.mk' (AtPrime P)
66         (s.1 * MvPolynomial.pderiv i a - a * MvPolynomial.pderiv i s.1)
67         ⟨s.1 ^ 2, P.primeCompl.pow_mem s.2 2⟩ := by
68   let B := AtPrime (k := k) (n := n) P
69   let N : R := s.1 * MvPolynomial.pderiv i a - a * MvPolynomial.pderiv i s.1
70   have hnum : algebraMap R B N =
71       algebraMap R B s.1 * algebraMap R B (MvPolynomial.pderiv i a) -
72         algebraMap R B a * algebraMap R B (MvPolynomial.pderiv i s.1) := by
73     simp [N, map_sub, map_mul]
74   have h := Stafford38.Characteristic.LocalizedDerivationQuotient.extendDerivation_apply_mk'_cross
75     (k := k) (A := R (k := k) (n := n)) (B := B) (S := P.primeCompl)
76     ((Algebra.linearMap (R (k := k) (n := n)) B).compDer (MvPolynomial.pderiv i)) a s
77   apply (IsLocalization.map_units B
78     ⟨s.1 ^ 2, P.primeCompl.pow_mem s.2 2⟩).mul_right_inj.mp
79   calc
80     algebraMap R B (s.1 ^ 2) *
81         extendDerivation k R B P.primeCompl
82           ((Algebra.linearMap R B).compDer (MvPolynomial.pderiv i))
83           (IsLocalization.mk' B a s) = algebraMap R B N := by
84         rw [map_pow, mul_comm]
85         exact h.trans hnum.symm
86     _ = algebraMap R B (s.1 ^ 2) *
87         IsLocalization.mk' B N ⟨s.1 ^ 2, P.primeCompl.pow_mem s.2 2⟩ := by
88           rw [mul_comm]
89           exact (IsLocalization.mk'_spec B N
90             ⟨s.1 ^ 2, P.primeCompl.pow_mem s.2 2⟩).symm

92 theorem localizedPDeriv_apply_mk
93     (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
94     (i : PhaseVar n) (a : R (k := k) (n := n)) (s : P.primeCompl) :
95     localizedPDeriv P i (IsLocalization.mk' (AtPrime P) a s) =
96       IsLocalization.mk' (AtPrime P)
97         (s.1 * MvPolynomial.pderiv i a - a * MvPolynomial.pderiv i s.1)
98         ⟨s.1 ^ 2, P.primeCompl.pow_mem s.2 2⟩ := by
99   change extendDerivation k (R (k := k) (n := n)) (AtPrime P) P.primeCompl
100       ((Algebra.linearMap (R (k := k) (n := n)) (AtPrime P)).compDer
101         (MvPolynomial.pderiv i))
102       (IsLocalization.mk' (AtPrime P) a s) = _
103   exact localizedPDeriv_mk P i a s

105 theorem localizedPDeriv_add
106     (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
107     (i : PhaseVar n) (f g : AtPrime P) :
108     localizedPDeriv P i (f + g) = localizedPDeriv P i f + localizedPDeriv P i g :=
109   Derivation.map_add (localizedPDeriv P i) f g

111 theorem localizedPDeriv_mul
112     (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
113     (i : PhaseVar n) (f g : AtPrime P) :
114     localizedPDeriv P i (f * g) =
115       localizedPDeriv P i f * g + f * localizedPDeriv P i g := by
116   rw [Derivation.leibniz]

119 theorem localizedPDeriv_algebraMap
120     (P : Ideal (R (k := k) (n := n))) [P.IsPrime]
121     (i : PhaseVar n) (a : R (k := k) (n := n)) :
122     localizedPDeriv P i
123         (algebraMap (R (k := k) (n := n)) (AtPrime P) a) =
124       algebraMap (R (k := k) (n := n)) (AtPrime P) (MvPolynomial.pderiv i a) := by
125   have h := Derivation.congr_fun (localizedPDeriv_compAlgebraMap P i) a
126   change localizedPDeriv P i
127       (algebraMap (R (k := k) (n := n)) (AtPrime P) a) =
128     algebraMap (R (k := k) (n := n)) (AtPrime P) (MvPolynomial.pderiv i a) at h
129   exact h
