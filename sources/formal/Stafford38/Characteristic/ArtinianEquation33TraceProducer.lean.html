8 # Matrix trace reductions for equation (3.3)
9 
10 This file isolates the matrix implication actually used after Singh--Kumar,
11 Proposition 3.2, equation (3.3).  The calculation has three coefficient
12 layers, which must not be conflated.
13 
14 Over the deformation ring, equation (3.3) produces a strict
15 upper-triangular remainder and two correction matrices.  The corrections need
16 not have zero trace there: the source proves separately that each trace lies in
17 the parameter ideal. Reduction modulo the parameter kills those traces, and a
18 second map to the coefficient field preserves their vanishing.
19 
20 In particular, no same-size coefficient-field ringCommutator decomposition is
21 assumed.  This file does not construct the matrices from a deformation module,
22 instantiate the localized right-Rees ring, identify the reduced matrix with
23 the induced `z`-action, or prove the high-power Artinian reduction.
24 -/

38 /-- A parameter reduction whose kernel is its principal two-sided ideal. -/
39 structure ParameterIdealReduction
40     (B Abar : Type u) [Ring B] [CommRing Abar] where
41   parameter : B
42   modParameter : B →+* Abar
43   ker_modParameter : RingHom.ker modParameter = Ideal.span {parameter}
44 
45 /-- Strict upper triangularity over an arbitrary coefficient ring. -/
46 def IsStrictUpperTriangularOver {S : Type u} [Zero S]
47     (M : Matrix (Fin (n + 1)) (Fin (n + 1)) S) : Prop :=
48   ∀ i j, j ≤ i → M i j = 0
49 
50 /-- A correction trace has precisely the source-level property used after
51 equation (3.3): it lies in the parameter ideal before reduction. -/
52 def CorrectionTraceInParameter
53     (S : ParameterIdealReduction B Abar)
54     (Q : Matrix (Fin (n + 1)) (Fin (n + 1)) B) : Prop :=
55   Matrix.trace Q ∈ Ideal.span {S.parameter}
56 
57 /-- Coefficientwise reduction of the three terms retained from equation
58 (3.3), first modulo the parameter and then to the coefficient field.  The
59 correction matrices are not replaced by coefficient-field commutators. -/
60 def reducedEquation33Matrix
61     (S : ParameterIdealReduction B Abar)
62     (residue : Abar →+* K)
63     (R Q₁ Q₂ : Matrix (Fin (n + 1)) (Fin (n + 1)) B) :
64     Matrix (Fin (n + 1)) (Fin (n + 1)) K :=
65   (R.map S.modParameter).map residue +
66     (Q₁.map S.modParameter).map residue +
67     (Q₂.map S.modParameter).map residue
68 
69 /-- A strict upper-triangular deformation-ring remainder stays strict upper
70 triangular in the Artinian quotient. -/
71 theorem modParameter_isStrictUpperTriangular
72     (S : ParameterIdealReduction B Abar)
73     (R : Matrix (Fin (n + 1)) (Fin (n + 1)) B)
74     (hR : IsStrictUpperTriangularOver R) :
75     IsStrictUpperTriangularOver (R.map S.modParameter) := by
76   intro i j hji
77   simp [Matrix.map_apply, hR i j hji]
78 
79 /-- A strict upper-triangular matrix over the Artinian quotient has zero
80 trace, before passing to the coefficient field. -/
81 theorem modParameter_remainder_trace_eq_zero
82     (S : ParameterIdealReduction B Abar)
83     (R : Matrix (Fin (n + 1)) (Fin (n + 1)) B)
84     (hR : IsStrictUpperTriangularOver R) :
85     Matrix.trace (R.map S.modParameter) = 0 := by
86   rw [Matrix.trace]
87   apply Finset.sum_eq_zero
88   intro i _hi
89   exact modParameter_isStrictUpperTriangular S R hR i i le_rfl
90 
91 /-- A correction can have nonzero trace before reduction.  Membership of that
92 trace in the parameter ideal is exactly what makes its trace zero in the
93 Artinian quotient. -/
94 theorem modParameter_correction_trace_eq_zero
95     (S : ParameterIdealReduction B Abar)
96     (Q : Matrix (Fin (n + 1)) (Fin (n + 1)) B)
97     (hQ : CorrectionTraceInParameter S Q) :
98     Matrix.trace (Q.map S.modParameter) = 0 := by
99   have hker : Matrix.trace Q ∈ RingHom.ker S.modParameter := by
100     rw [S.ker_modParameter]
101     exact hQ
102   have hreduce : S.modParameter (Matrix.trace Q) = 0 := hker
103   simpa using
104     (AddMonoidHom.map_trace S.modParameter.toAddMonoidHom Q).symm.trans hreduce
105 
106 /-- Zero trace in the Artinian quotient remains zero after the separate
107 residue/coefficient-field projection. -/
108 theorem residue_trace_eq_zero
109     (residue : Abar →+* K)
110     (M : Matrix (Fin (n + 1)) (Fin (n + 1)) Abar)
111     (hM : Matrix.trace M = 0) :
112     Matrix.trace (M.map residue) = 0 := by
113   calc
114     Matrix.trace (M.map residue) = residue (Matrix.trace M) := by
115       simpa using (AddMonoidHom.map_trace residue.toAddMonoidHom M).symm
116     _ = 0 := by rw [hM, map_zero]
117 
118 /-- The separate source reductions imply trace zero for the coefficient-field
119 matrix represented by equation (3.3). -/
120 theorem reducedEquation33Matrix_trace_eq_zero
121     (S : ParameterIdealReduction B Abar)
122     (residue : Abar →+* K)
123     (R Q₁ Q₂ : Matrix (Fin (n + 1)) (Fin (n + 1)) B)
124     (hR : IsStrictUpperTriangularOver R)
125     (hQ₁ : CorrectionTraceInParameter S Q₁)
126     (hQ₂ : CorrectionTraceInParameter S Q₂) :
127     Matrix.trace (reducedEquation33Matrix S residue R Q₁ Q₂) = 0 := by
128   rw [reducedEquation33Matrix, Matrix.trace_add, Matrix.trace_add,
129     residue_trace_eq_zero residue (R.map S.modParameter)
130       (modParameter_remainder_trace_eq_zero S R hR),
131     residue_trace_eq_zero residue (Q₁.map S.modParameter)
132       (modParameter_correction_trace_eq_zero S Q₁ hQ₁),
133     residue_trace_eq_zero residue (Q₂.map S.modParameter)
134       (modParameter_correction_trace_eq_zero S Q₂ hQ₂)]
135   simp
