12 # The order-Rees two-jet ring
13 
14 This file constructs only the ring half of the order-Rees two-jet.  The
15 specialization map is defined directly as the finite sum of the order-principal
16 components of the Rees coefficients.  Its multiplication proof retains the
17 written coefficient order in the noncommutative Weyl algebra.
18 
19 No Rees-module action, quotient module, trace package, or Gabber theorem is
20 constructed here.
21 -/

44 /-- The scalar embedding into Rees degree zero. -/
45 def orderReesScalarHom : k →+* OrderReesRing (n := n) k where
46   toFun a := ⟨Polynomial.C (algebraMap k (WeylAlg k n) a), by
47     intro N
48     by_cases hN : N = 0
49     · subst N
50       rw [Polynomial.coeff_C_zero]
51       rw [Algebra.algebraMap_eq_smul_one]
52       exact Submodule.smul_mem _ a (orderPieceOne (n := n) k).property
53     · rw [Polynomial.coeff_C, if_neg hN]
54       exact Submodule.zero_mem _⟩
55   map_zero' := by ext; simp
56   map_one' := by ext; simp
57   map_add' a b := by ext; simp
58   map_mul' a b := by ext; simp

60 /-- The inherited `k`-algebra structure on the order-Rees subring, with
61 scalars placed in Rees degree zero. -/
62 noncomputable instance orderReesRingAlgebra :
63     Algebra k (OrderReesRing (n := n) k) where
64   smul a f := orderReesScalarHom (n := n) k a * f
65   algebraMap := orderReesScalarHom (n := n) k
66   commutes' a f := by
67     apply Subtype.ext
68     change Polynomial.C (algebraMap k (WeylAlg k n) a) *
69         (f : Polynomial (WeylAlg k n)) =
70       (f : Polynomial (WeylAlg k n)) *
71         Polynomial.C (algebraMap k (WeylAlg k n) a)
72     ext N
73     rw [Polynomial.coeff_C_mul, Polynomial.coeff_mul_C]
74     exact Algebra.commutes _ _
75   smul_def' _ _ := rfl

77 /-- The additive finite sum of degreewise order-principal components. -/
78 def orderPrincipalSum :
79     Polynomial (WeylAlg k n) →+ SymbolRing k n where
80   toFun f := f.sum fun N z =>
81     presentedPrincipalComponent k (@orderWeight n) N z
82   map_zero' := Polynomial.sum_zero_index _
83   map_add' f g := Polynomial.sum_add_index f g _
84     (fun N => by simp)
85     (fun N x y => by simp)

87 @[simp] theorem orderPrincipalSum_monomial
88     (N : ℕ) (z : WeylAlg k n) :
89     orderPrincipalSum (n := n) k (Polynomial.monomial N z) =
90       presentedPrincipalComponent k (@orderWeight n) N z := by
91   exact Polynomial.sum_monomial_index z _ (by simp)

93 /-- On Rees polynomials, the degreewise principal-component sum preserves
94 multiplication.  In the double sum, coefficients occur as `x * y`, in the
95 same order as in the input product. -/
96 theorem orderPrincipalSum_mul
97     (f g : OrderReesRing (n := n) k) :
98     orderPrincipalSum (n := n) k
99         ((f * g : OrderReesRing (n := n) k) :
100           Polynomial (WeylAlg k n)) =
101       orderPrincipalSum (n := n) k f *
102         orderPrincipalSum (n := n) k g := by
103   change orderPrincipalSum (n := n) k
104       ((f : Polynomial (WeylAlg k n)) *
105         (g : Polynomial (WeylAlg k n))) = _
106   rw [Polynomial.mul_eq_sum_sum, map_sum]
107   simp only [Polynomial.sum, map_sum, orderPrincipalSum_monomial]
108   change
109     (∑ i ∈ (f : Polynomial (WeylAlg k n)).support,
110       ∑ j ∈ (g : Polynomial (WeylAlg k n)).support,
111         presentedPrincipalComponent k (@orderWeight n) (i + j)
112           ((f : Polynomial (WeylAlg k n)).coeff i *
113             (g : Polynomial (WeylAlg k n)).coeff j)) =
114       (∑ i ∈ (f : Polynomial (WeylAlg k n)).support,
115         presentedPrincipalComponent k (@orderWeight n) i
116           ((f : Polynomial (WeylAlg k n)).coeff i)) *
117       (∑ j ∈ (g : Polynomial (WeylAlg k n)).support,
118         presentedPrincipalComponent k (@orderWeight n) j
119           ((g : Polynomial (WeylAlg k n)).coeff j))
120   rw [Finset.sum_mul]
121   apply Finset.sum_congr rfl
122   intro i hi
123   rw [Finset.mul_sum]
124   apply Finset.sum_congr rfl
125   intro j hj
126   exact presentedPrincipalComponent_mul_order k (f.property i) (g.property j)

128 /-- Global specialization of the order-Rees ring to its commutative symbol
129 ring. -/
130 def orderReesSpecialization :
131     OrderReesRing (n := n) k →+* SymbolRing k n where
132   toFun f := orderPrincipalSum (n := n) k f
133   map_zero' := by
134     change orderPrincipalSum (n := n) k
135       (0 : Polynomial (WeylAlg k n)) = 0
136     exact map_zero (orderPrincipalSum (n := n) k)
137   map_add' f g := by
138     change orderPrincipalSum (n := n) k
139         ((f : Polynomial (WeylAlg k n)) + g) = _
140     exact map_add (orderPrincipalSum (n := n) k)
141       (f : Polynomial (WeylAlg k n))
142       (g : Polynomial (WeylAlg k n))
143   map_one' := by
144     change orderPrincipalSum (n := n) k
145       (1 : Polynomial (WeylAlg k n)) = 1
146     rw [show (1 : Polynomial (WeylAlg k n)) = Polynomial.monomial 0 1 by
147       simp]
148     rw [orderPrincipalSum_monomial]
149     rw [presentedPrincipalComponent, LinearMap.comp_apply]
150     change MvPolynomial.weightedHomogeneousComponent (@orderWeight n) 0
151       (presentedNormalFormLinearEquiv k n 1) = 1
152     rw [Stafford38.WeylPBW.presentedNormalFormLinearEquiv_one]
153     exact (MvPolynomial.isWeightedHomogeneous_one k (@orderWeight n)).weightedHomogeneousComponent_same
154   map_mul' := orderPrincipalSum_mul k

172 /-- Every symbol has a Rees lift. -/
173 theorem orderReesSpecialization_surjective :
174     Function.Surjective (orderReesSpecialization (n := n) k) := by
175   intro P
176   induction P using MvPolynomial.induction_on' with
177   | monomial m a =>
178       let N := Finsupp.weight (@orderWeight n) m
179       let z : orderPiece k n N :=
180         ⟨a • Stafford38.WeylPBW.presentedPBWBasis k n m, Submodule.smul_mem _ a
181           ((presentedPBWBasis_mem_weightPiece_iff k (@orderWeight n) N m).mpr
182             (le_refl N))⟩
183       refine ⟨orderReesMonomial k N z, ?_⟩
184       rw [orderReesSpecialization_monomial]
185       change presentedPrincipalComponent k (@orderWeight n) N
186           (a • Stafford38.WeylPBW.presentedPBWBasis k n m) =
187         MvPolynomial.monomial m a
188       rw [map_smul, presentedPrincipalComponent_basis, if_pos rfl]
189       rw [MvPolynomial.smul_monomial]
190       simp
191   | add P Q hP hQ =>
192       obtain ⟨p, hp⟩ := hP
193       obtain ⟨q, hq⟩ := hQ
194       refine ⟨p + q, ?_⟩
195       change orderReesSpecialization (n := n) k (p + q) = P + Q
196       rw [(orderReesSpecialization (n := n) k).map_add, hp, hq]

198 @[simp] theorem orderReesSpecialization_scalar (a : k) :
199     orderReesSpecialization (n := n) k
200         (orderReesScalarHom (n := n) k a) =
201       algebraMap k (SymbolRing k n) a := by
202   change (Polynomial.C (algebraMap k (WeylAlg k n) a)).sum
203       (fun N z => presentedPrincipalComponent k (@orderWeight n) N z) = _
204   rw [Polynomial.sum_C_index]
205   · change MvPolynomial.weightedHomogeneousComponent (@orderWeight n) 0
206       (presentedNormalFormLinearEquiv k n
207         (algebraMap k (WeylAlg k n) a)) = _

209       Stafford38.WeylPBW.presentedNormalFormLinearEquiv_one]

212       (@orderWeight n)).weightedHomogeneousComponent_same]
213     exact MvPolynomial.C_eq_smul_one.symm

225 /-- The two-sided ideal generated by the square of the Rees parameter. -/
226 def orderReesTwoJetIdeal : TwoSidedIdeal (OrderReesRing (n := n) k) :=

231 /-- The order-Rees ring modulo `T²`. -/

235 /-- The canonical algebra quotient map to the two-jet. -/
236 def orderReesTwoJetQuotient :
237     OrderReesRing (n := n) k →ₐ[k] OrderReesTwoJet (n := n) k :=

239     change (orderReesTwoJetIdeal (n := n) k).ringCon.mk'

244 /-- The image of the Rees parameter in the two-jet. -/

248 /-- The two-jet parameter is central. -/
249 theorem orderReesTwoJetParameter_mem_center :
250     orderReesTwoJetParameter (n := n) k ∈
251       Set.center (OrderReesTwoJet (n := n) k) := by
252   rw [Semigroup.mem_center_iff]
253   intro q
254   obtain ⟨r, rfl⟩ := Quotient.mk''_surjective q
255   change orderReesTwoJetQuotient (n := n) k r *
256       orderReesTwoJetQuotient (n := n) k (orderReesParameter (n := n) k) =
257     orderReesTwoJetQuotient (n := n) k (orderReesParameter (n := n) k) *
258       orderReesTwoJetQuotient (n := n) k r
259   rw [← map_mul, ← map_mul, orderReesParameter_mul_comm]

261 /-- The two-jet parameter is square-zero. -/
262 theorem orderReesTwoJetParameter_sq :
263     orderReesTwoJetParameter (n := n) k ^ 2 = 0 := by
264   change orderReesTwoJetQuotient (n := n) k
265       (orderReesParameter (n := n) k) ^ 2 = 0
266   rw [← map_pow]
267   change (orderReesTwoJetIdeal (n := n) k).ringCon.mk'
268       (orderReesParameter (n := n) k ^ 2) =
269     (orderReesTwoJetIdeal (n := n) k).ringCon.mk' 0
270   apply Quotient.sound
271   apply ((orderReesTwoJetIdeal (n := n) k).rel_iff _ _).mpr
272   have hmem : orderReesParameter (n := n) k ^ 2 ∈
273       orderReesTwoJetIdeal (n := n) k :=
274     TwoSidedIdeal.subset_span (Set.mem_singleton _)
275   simpa using hmem

277 theorem orderReesTwoJetIdeal_le_specializationKer :
278     orderReesTwoJetIdeal (n := n) k ≤
279       TwoSidedIdeal.ker (orderReesSpecialization (n := n) k) := by
280   intro x hx
281   change x ∈ TwoSidedIdeal.span
282     ({orderReesParameter (n := n) k ^ 2} :
283       Set (OrderReesRing (n := n) k)) at hx
284   rw [TwoSidedIdeal.mem_span_iff] at hx
285   apply hx
286   intro z hz
287   simp only [Set.mem_singleton_iff] at hz
288   subst z
289   change orderReesSpecialization (n := n) k
290     (orderReesParameter (n := n) k ^ 2) = 0
291   rw [map_pow, orderReesSpecialization_parameter,
292     zero_pow (by omega)]

294 /-- Specialization factors through the order-Rees two-jet. -/
295 def orderReesTwoJetSpecialization :
296     OrderReesTwoJet (n := n) k →ₐ[k] SymbolRing k n :=
297   let f := orderReesSpecializationAlg (n := n) k
298   let hrel : ∀ ⦃x y : OrderReesRing (n := n) k⦄,
299       (orderReesTwoJetIdeal (n := n) k).ringCon x y → f x = f y := by
300     intro x y hxy
301     have hmem : x - y ∈ orderReesTwoJetIdeal (n := n) k :=
302       ((orderReesTwoJetIdeal (n := n) k).rel_iff x y).mp hxy
303     have hker := orderReesTwoJetIdeal_le_specializationKer (n := n) k hmem
304     rw [TwoSidedIdeal.mem_ker] at hker
305     have hker' : f (x - y) = 0 := hker
306     rw [map_sub] at hker'
307     exact sub_eq_zero.mp hker'
308   { toFun := Quotient.lift f hrel
309     map_zero' := f.map_zero
310     map_one' := f.map_one
311     map_add' := by
312       rintro ⟨x⟩ ⟨y⟩
313       exact f.map_add x y
314     map_mul' := by
315       rintro ⟨x⟩ ⟨y⟩
316       exact f.map_mul x y
317     commutes' := f.commutes }
