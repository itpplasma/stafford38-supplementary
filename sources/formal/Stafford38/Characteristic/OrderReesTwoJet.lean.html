10 # The order-Rees two-jet ring
11 
12 This file constructs only the ring half of the order-Rees two-jet.  The
13 specialization map is defined directly as the finite sum of the order-principal
14 components of the Rees coefficients.  Its multiplication proof retains the
15 written coefficient order in the noncommutative Weyl algebra.
16 
17 No Rees-module action, quotient module, trace package, or Gabber theorem is
18 constructed here.
19 -/

42 /-- The scalar embedding into Rees degree zero. -/
43 def orderReesScalarHom : k →+* OrderReesRing (n := n) k where
44   toFun a := ⟨Polynomial.C (algebraMap k (PresentedWeyl k n) a), by
45     intro N
46     by_cases hN : N = 0
47     · subst N
48       rw [Polynomial.coeff_C_zero]
49       rw [Algebra.algebraMap_eq_smul_one]
50       exact Submodule.smul_mem _ a (orderPieceOne (n := n) k).property
51     · rw [Polynomial.coeff_C, if_neg hN]
52       exact Submodule.zero_mem _⟩
53   map_zero' := by ext; simp
54   map_one' := by ext; simp
55   map_add' a b := by ext; simp
56   map_mul' a b := by ext; simp

58 /-- The inherited `k`-algebra structure on the order-Rees subring, with
59 scalars placed in Rees degree zero. -/
60 noncomputable instance orderReesRingAlgebra :
61     Algebra k (OrderReesRing (n := n) k) where
62   smul a f := orderReesScalarHom (n := n) k a * f
63   algebraMap := orderReesScalarHom (n := n) k
64   commutes' a f := by
65     apply Subtype.ext
66     change Polynomial.C (algebraMap k (PresentedWeyl k n) a) *
67         (f : Polynomial (PresentedWeyl k n)) =
68       (f : Polynomial (PresentedWeyl k n)) *
69         Polynomial.C (algebraMap k (PresentedWeyl k n) a)
70     ext N
71     rw [Polynomial.coeff_C_mul, Polynomial.coeff_mul_C]
72     exact Algebra.commutes _ _
73   smul_def' _ _ := rfl

75 /-- The additive finite sum of degreewise order-principal components. -/
76 def orderPrincipalSum :
77     Polynomial (PresentedWeyl k n) →+ SymbolRing k n where
78   toFun f := f.sum fun N z =>
79     presentedPrincipalComponent k (@orderWeight n) N z
80   map_zero' := Polynomial.sum_zero_index _
81   map_add' f g := Polynomial.sum_add_index f g _
82     (fun N => by simp)
83     (fun N x y => by simp)

85 @[simp] theorem orderPrincipalSum_monomial
86     (N : ℕ) (z : PresentedWeyl k n) :
87     orderPrincipalSum (n := n) k (Polynomial.monomial N z) =
88       presentedPrincipalComponent k (@orderWeight n) N z := by
89   exact Polynomial.sum_monomial_index z _ (by simp)

91 /-- On Rees polynomials, the degreewise principal-component sum preserves
92 multiplication.  In the double sum, coefficients occur as `x * y`, in the
93 same order as in the input product. -/
94 theorem orderPrincipalSum_mul
95     (f g : OrderReesRing (n := n) k) :
96     orderPrincipalSum (n := n) k
97         ((f * g : OrderReesRing (n := n) k) :
98           Polynomial (PresentedWeyl k n)) =
99       orderPrincipalSum (n := n) k f *
100         orderPrincipalSum (n := n) k g := by
101   change orderPrincipalSum (n := n) k
102       ((f : Polynomial (PresentedWeyl k n)) *
103         (g : Polynomial (PresentedWeyl k n))) = _
104   rw [Polynomial.mul_eq_sum_sum, map_sum]
105   simp only [Polynomial.sum, map_sum, orderPrincipalSum_monomial]
106   change
107     (∑ i ∈ (f : Polynomial (PresentedWeyl k n)).support,
108       ∑ j ∈ (g : Polynomial (PresentedWeyl k n)).support,
109         presentedPrincipalComponent k (@orderWeight n) (i + j)
110           ((f : Polynomial (PresentedWeyl k n)).coeff i *
111             (g : Polynomial (PresentedWeyl k n)).coeff j)) =
112       (∑ i ∈ (f : Polynomial (PresentedWeyl k n)).support,
113         presentedPrincipalComponent k (@orderWeight n) i
114           ((f : Polynomial (PresentedWeyl k n)).coeff i)) *
115       (∑ j ∈ (g : Polynomial (PresentedWeyl k n)).support,
116         presentedPrincipalComponent k (@orderWeight n) j
117           ((g : Polynomial (PresentedWeyl k n)).coeff j))
118   rw [Finset.sum_mul]
119   apply Finset.sum_congr rfl
120   intro i hi
121   rw [Finset.mul_sum]
122   apply Finset.sum_congr rfl
123   intro j hj
124   exact presentedPrincipalComponent_mul_order k (f.property i) (g.property j)

126 /-- Global specialization of the order-Rees ring to its commutative symbol
127 ring. -/
128 def orderReesSpecialization :
129     OrderReesRing (n := n) k →+* SymbolRing k n where
130   toFun f := orderPrincipalSum (n := n) k f
131   map_zero' := by
132     change orderPrincipalSum (n := n) k
133       (0 : Polynomial (PresentedWeyl k n)) = 0
134     exact map_zero (orderPrincipalSum (n := n) k)
135   map_add' f g := by
136     change orderPrincipalSum (n := n) k
137         ((f : Polynomial (PresentedWeyl k n)) + g) = _
138     exact map_add (orderPrincipalSum (n := n) k)
139       (f : Polynomial (PresentedWeyl k n))
140       (g : Polynomial (PresentedWeyl k n))
141   map_one' := by
142     change orderPrincipalSum (n := n) k
143       (1 : Polynomial (PresentedWeyl k n)) = 1
144     rw [show (1 : Polynomial (PresentedWeyl k n)) = Polynomial.monomial 0 1 by
145       simp]
146     rw [orderPrincipalSum_monomial]
147     rw [presentedPrincipalComponent, LinearMap.comp_apply]
148     change MvPolynomial.weightedHomogeneousComponent (@orderWeight n) 0
149       (presentedNormalFormLinearEquiv k n 1) = 1
150     rw [Stafford38.WeylPBW.presentedNormalFormLinearEquiv_one]
151     exact (MvPolynomial.isWeightedHomogeneous_one k (@orderWeight n)).weightedHomogeneousComponent_same
152   map_mul' := orderPrincipalSum_mul k

170 /-- Every symbol has a Rees lift. -/
171 theorem orderReesSpecialization_surjective :
172     Function.Surjective (orderReesSpecialization (n := n) k) := by
173   intro P
174   induction P using MvPolynomial.induction_on' with
175   | monomial m a =>
176       let N := monomialWeight (@orderWeight n) m
177       let z : orderPiece k n N :=
178         ⟨a • Stafford38.WeylPBW.presentedPBWBasis k n m, Submodule.smul_mem _ a
179           ((presentedPBWBasis_mem_weightPiece_iff k (@orderWeight n) N m).mpr
180             (le_refl N))⟩
181       refine ⟨orderReesMonomial k N z, ?_⟩
182       rw [orderReesSpecialization_monomial]
183       change presentedPrincipalComponent k (@orderWeight n) N
184           (a • Stafford38.WeylPBW.presentedPBWBasis k n m) =
185         MvPolynomial.monomial m a
186       rw [map_smul, presentedPrincipalComponent_basis, if_pos rfl]
187       rw [MvPolynomial.smul_monomial]
188       simp
189   | add P Q hP hQ =>
190       obtain ⟨p, hp⟩ := hP
191       obtain ⟨q, hq⟩ := hQ
192       refine ⟨p + q, ?_⟩
193       change orderReesSpecialization (n := n) k (p + q) = P + Q
194       rw [(orderReesSpecialization (n := n) k).map_add, hp, hq]

223 /-- The two-sided ideal generated by the square of the Rees parameter. -/

246 /-- The two-jet parameter is central. -/
247 theorem orderReesTwoJetParameter_mem_center :
248     orderReesTwoJetParameter (n := n) k ∈
249       Set.center (OrderReesTwoJet (n := n) k) := by
250   rw [Semigroup.mem_center_iff]
251   intro q
252   obtain ⟨r, rfl⟩ := Quotient.mk''_surjective q
253   change orderReesTwoJetQuotient (n := n) k r *
254       orderReesTwoJetQuotient (n := n) k (orderReesParameter (n := n) k) =
255     orderReesTwoJetQuotient (n := n) k (orderReesParameter (n := n) k) *
256       orderReesTwoJetQuotient (n := n) k r
257   rw [← map_mul, ← map_mul, orderReesParameter_mul_comm]

259 /-- The two-jet parameter is square-zero. -/
260 theorem orderReesTwoJetParameter_sq :
261     orderReesTwoJetParameter (n := n) k ^ 2 = 0 := by
262   change orderReesTwoJetQuotient (n := n) k
263       (orderReesParameter (n := n) k) ^ 2 = 0
264   rw [← map_pow]
265   change (orderReesTwoJetIdeal (n := n) k).ringCon.mk'
266       (orderReesParameter (n := n) k ^ 2) =
267     (orderReesTwoJetIdeal (n := n) k).ringCon.mk' 0
268   apply Quotient.sound
269   apply ((orderReesTwoJetIdeal (n := n) k).rel_iff _ _).mpr
270   have hmem : orderReesParameter (n := n) k ^ 2 ∈
271       orderReesTwoJetIdeal (n := n) k :=
272     TwoSidedIdeal.subset_span (Set.mem_singleton _)
273   simpa using hmem

275 theorem orderReesTwoJetIdeal_le_specializationKer :
276     orderReesTwoJetIdeal (n := n) k ≤
277       TwoSidedIdeal.ker (orderReesSpecialization (n := n) k) := by
278   intro x hx
279   change x ∈ TwoSidedIdeal.span
280     ({orderReesParameter (n := n) k ^ 2} :
281       Set (OrderReesRing (n := n) k)) at hx
282   rw [TwoSidedIdeal.mem_span_iff] at hx
283   apply hx
284   intro z hz
285   simp only [Set.mem_singleton_iff] at hz
286   subst z
287   change orderReesSpecialization (n := n) k
288     (orderReesParameter (n := n) k ^ 2) = 0
289   rw [map_pow, orderReesSpecialization_parameter,

292 /-- Specialization factors through the order-Rees two-jet. -/
293 def orderReesTwoJetSpecialization :
294     OrderReesTwoJet (n := n) k →ₐ[k] SymbolRing k n :=
295   let f := orderReesSpecializationAlg (n := n) k
296   let hrel : ∀ ⦃x y : OrderReesRing (n := n) k⦄,
297       (orderReesTwoJetIdeal (n := n) k).ringCon x y → f x = f y := by
298     intro x y hxy
299     have hmem : x - y ∈ orderReesTwoJetIdeal (n := n) k :=
300       ((orderReesTwoJetIdeal (n := n) k).rel_iff x y).mp hxy
301     have hker := orderReesTwoJetIdeal_le_specializationKer (n := n) k hmem
302     rw [TwoSidedIdeal.mem_ker] at hker
303     have hker' : f (x - y) = 0 := hker
304     rw [map_sub] at hker'
305     exact sub_eq_zero.mp hker'
306   { toFun := Quotient.lift f hrel
307     map_zero' := f.map_zero
308     map_one' := f.map_one
309     map_add' := by
310       rintro ⟨x⟩ ⟨y⟩
311       exact f.map_add x y
312     map_mul' := by
313       rintro ⟨x⟩ ⟨y⟩
314       exact f.map_mul x y
315     commutes' := f.commutes }
