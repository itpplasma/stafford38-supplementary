9 /-!
10 # Tangential symbol action on the canonical total pages
11 
12 The commuting polynomial action is constructed from the actual right Weyl
13 generators. Their commutators have lower order and hence vanish on the pages.
14 -/

32 abbrev complex := canonicalFilteredTwoTerm k n N d hd

34 def generator (i : Fin n ⊕ Fin n) :
35     (complex k n N d hd).PageOperator (tangentialDegree i : ℤ) :=
36   match i with
37   | .inl j => tangential_coordinate_pageOperator k n N d hd j
38   | .inr j => tangential_momentum_pageOperator k n N d hd j
39 
40 theorem generator_commutator_lowers (i j : Fin n ⊕ Fin n)
41     (p : ℤ) (z : CanonicalQuotient k n N d)
42     (hz : z ∈ (complex k n N d hd).G p) :
43     (generator k n N d hd i).g ((generator k n N d hd j).g z) -
44       (generator k n N d hd j).g ((generator k n N d hd i).g z) ∈
45         (complex k n N d hd).G
46           (p - (tangentialDegree i : ℤ) - (tangentialDegree j : ℤ) + 1) := by
47   have h := tangential_commutator_lower k n N d hd i j p z hz
48   cases i <;> cases j <;>
49     simpa [generator, tangential_coordinate_pageOperator,
50       tangential_momentum_pageOperator, tangentialPageOperator,
51       tangentialDegree, show p - 1 - 1 + 1 = p - 2 + 1 by omega] using h

53 def sourceGenerator (r : ℕ) (i : Fin n ⊕ Fin n) :
54     Module.End k ((complex k n N d hd).SourceTotal r) :=
55   (generator k n N d hd i).sourceTotalMap r
56 
57 def targetGenerator (r : ℕ) (i : Fin n ⊕ Fin n) :
58     Module.End k ((complex k n N d hd).TargetTotal r) :=
59   (generator k n N d hd i).targetTotalMap r

61 theorem sourceGenerator_commute (r : ℕ) (i j : Fin n ⊕ Fin n) :
62     Commute (sourceGenerator k n N d hd r i) (sourceGenerator k n N d hd r j) :=
63   (generator k n N d hd i).sourceTotalMap_commute_of_commutator_lowers
64     (generator k n N d hd j) (generator_commutator_lowers k n N d hd i j) r

66 theorem targetGenerator_commute (r : ℕ) (i j : Fin n ⊕ Fin n) :
67     Commute (targetGenerator k n N d hd r i) (targetGenerator k n N d hd r j) :=
68   (generator k n N d hd i).targetTotalMap_commute_of_commutator_lowers
69     (generator k n N d hd j) (generator_commutator_lowers k n N d hd i j) r

71 def sourceAction (r : ℕ) :
72     MvPolynomial (Fin n ⊕ Fin n) k →ₐ[k]
73       Module.End k ((complex k n N d hd).SourceTotal r) :=
74   commutingPolynomialAction (sourceGenerator k n N d hd r)
75     (sourceGenerator_commute k n N d hd r)
76 
77 def targetAction (r : ℕ) :
78     MvPolynomial (Fin n ⊕ Fin n) k →ₐ[k]
79       Module.End k ((complex k n N d hd).TargetTotal r) :=
80   commutingPolynomialAction (targetGenerator k n N d hd r)
81     (targetGenerator_commute k n N d hd r)

83 theorem totalDrop_intertwines_polynomials (r : ℕ)
84     (P : MvPolynomial (Fin n ⊕ Fin n) k) :
85     (complex k n N d hd).totalDrop r ∘ₗ sourceAction k n N d hd r P =
86       targetAction k n N d hd r P ∘ₗ (complex k n N d hd).totalDrop r :=
87   commutingPolynomialAction_intertwines _ _ _ _ _
88     (fun i => (generator k n N d hd i).totalDrop_intertwines r) P

102 /-- The concrete page differential is linear over the tangential symbol ring. -/
103 def tangentialDrop (r : ℕ) :
104     (complex k n N d hd).SourceTotal r →ₗ[MvPolynomial (Fin n ⊕ Fin n) k]
105       (complex k n N d hd).TargetTotal r where
106   toFun := (complex k n N d hd).totalDrop r
107   map_add' := map_add _
108   map_smul' P z := by
109     exact congrArg (fun f => f z)
110       (totalDrop_intertwines_polynomials k n N d hd r P)
111

114     tangentialDrop k n N d hd r z = (complex k n N d hd).totalDrop r z := rfl
