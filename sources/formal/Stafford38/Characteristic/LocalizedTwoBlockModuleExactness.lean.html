6 /-!
7 # Module exactness after the localized two-block ring quotient
8 
9 The two-block ideal annihilates the localized deformation module, so passage
10 to the quotient changes only its scalar ring.  This file names the action of
11 the quotient parameter and identifies it exactly with the already constructed
12 localized parameter action.  Parameter exactness and the additive
13 special-fibre equivalence therefore transfer without quotienting the module.
14 
15 No identification of the quotient ring modulo its parameter with the
16 commutative doubled-power quotient is asserted.
17 -/

47 /-- The additive endomorphism induced by the parameter class in the localized
48 two-block quotient ring. -/
49 def localizedTwoBlockParameterAct
50     (q : ℕ)
51     (hpow : 𝔪 ^ q ≤
52       Module.annihilator A₀ (LocalizedModule S G)) :
53     Wₗ →+ Wₗ where
54   toFun w :=
55     localizedTwoBlockAction D S q hpow
56       ((localizedTwoBlockIdeal D S q).ringCon.mk'
57         (OreLocalization.numeratorRingHom (MulOpposite.op D.c))) w
58   map_zero' := by
59     calc
60       localizedTwoBlockAction D S q hpow
61           ((localizedTwoBlockIdeal D S q).ringCon.mk'
62             (OreLocalization.numeratorRingHom (MulOpposite.op D.c))) 0 =
63         localizedCAct D S 0 :=
64           localizedTwoBlock_parameter_smul D S q hpow 0
65       _ = 0 := (localizedCAct D S).map_zero
66   map_add' x y := by
67     calc
68       localizedTwoBlockAction D S q hpow
69           ((localizedTwoBlockIdeal D S q).ringCon.mk'
70             (OreLocalization.numeratorRingHom (MulOpposite.op D.c))) (x + y) =
71         localizedCAct D S (x + y) :=
72           localizedTwoBlock_parameter_smul D S q hpow (x + y)
73       _ = localizedCAct D S x + localizedCAct D S y :=
74         (localizedCAct D S).map_add x y
75       _ = localizedTwoBlockAction D S q hpow
76             ((localizedTwoBlockIdeal D S q).ringCon.mk'
77               (OreLocalization.numeratorRingHom (MulOpposite.op D.c))) x +
78           localizedTwoBlockAction D S q hpow
79             ((localizedTwoBlockIdeal D S q).ringCon.mk'
80               (OreLocalization.numeratorRingHom (MulOpposite.op D.c))) y := by
81         rw [localizedTwoBlock_parameter_smul,
82           localizedTwoBlock_parameter_smul]

84 /-- Quotienting the scalar ring does not change the parameter endomorphism of
85 the localized deformation module. -/
86 theorem localizedTwoBlockParameterAct_eq_localizedCAct
87     (q : ℕ)
88     (hpow : 𝔪 ^ q ≤
89       Module.annihilator A₀ (LocalizedModule S G)) :
90     localizedTwoBlockParameterAct D S q hpow =
91       localizedCAct D S := by
92   ext w
93   exact localizedTwoBlock_parameter_smul D S q hpow w
94 
95 /-- The square-zero parameter sequence remains exact after descent of the
96 scalar action to the localized two-block quotient ring. -/
97 theorem localizedTwoBlockParameterAct_ker_eq_range
98     (q : ℕ)
99     (hpow : 𝔪 ^ q ≤
100       Module.annihilator A₀ (LocalizedModule S G)) :
101     AddMonoidHom.ker (localizedTwoBlockParameterAct D S q hpow) =
102       AddMonoidHom.range (localizedTwoBlockParameterAct D S q hpow) := by
103   rw [localizedTwoBlockParameterAct_eq_localizedCAct]
104   exact localizedCAct_ker_eq_range D S
105 
106 /-- The additive special fibre of the unchanged quotient-ring module is the
107 ordinary localized special fibre. -/
108 noncomputable def localizedTwoBlockSpecialFibreAddEquiv
109     (q : ℕ)
110     (hpow : 𝔪 ^ q ≤
111       Module.annihilator A₀ (LocalizedModule S G)) :
112     (Wₗ ⧸ AddMonoidHom.range
113         (localizedTwoBlockParameterAct D S q hpow)) ≃+
114       LocalizedModule S G := by
115   rw [localizedTwoBlockParameterAct_eq_localizedCAct]
116   exact localizedSpecialFibreAddEquiv D S
