12 /-!
13 # Homogeneous symbols and nonzero chart columns
14 
15 Over a characteristic-zero field, a nonzero homogeneous phase-space symbol
16 has a nonvanishing point over the original field. Linear substitution
17 identifies evaluation along a chosen matrix column with the corresponding
18 pure-power coefficient. `SymplecticCompletion` constructs a symplectic matrix
19 with the required nonzero vector in that column.
20 -/

31 theorem exists_eval_ne_zero_of_homogeneous [CharZero k]
32     {n N : ℕ} {P : SymbolRing k n}
33     (hP : P.IsHomogeneous N) (hne : P ≠ 0) :
34     ∃ v : PhaseVar n → k, MvPolynomial.eval v P ≠ 0 := by
35   by_contra h
36   push_neg at h
37   exact hne (hP.eq_zero_of_forall_eval_eq_zero h)

39 theorem eval_symbolLinearAlgHom {n : ℕ}
40     (M : Matrix (PhaseVar n) (PhaseVar n) k)
41     (v : PhaseVar n → k) (P : SymbolRing k n) :
42     MvPolynomial.eval v (symbolLinearAlgHom k M P) =
43       MvPolynomial.eval (fun i => ∑ j, M i j * v j) P := by
44   rw [symbolLinearAlgHom]
45   change MvPolynomial.aeval v
46       (MvPolynomial.aeval (symbolLinearCombination k M) P) =
47     MvPolynomial.aeval (fun i => ∑ j, M i j * v j) P
48   rw [MvPolynomial.comp_aeval_apply]
49   congr 2
50   funext i
51   simp [symbolLinearCombination]

53 theorem homogeneous_unit_eq_monomial {N : ℕ}
54     (P : MvPolynomial Unit k) (hP : P.IsHomogeneous N) :
55     P = MvPolynomial.monomial (Finsupp.single () N)
56       (MvPolynomial.coeff (Finsupp.single () N) P) := by
57   ext d
58   by_cases hd : d = Finsupp.single () N
59   · subst d
60     simp
61   · have hdegree : d.degree ≠ N := by
62       intro hdeg
63       apply hd
64       have hdform : d = Finsupp.single () (d ()) := by
65         apply Finsupp.ext
66         intro i
67         rcases i with ⟨⟩
68         simp
69       rw [hdform, Finsupp.degree_single] at hdeg
70       rw [hdform, hdeg]
71     rw [hP.coeff_eq_zero hdegree]
72     rw [MvPolynomial.coeff_monomial, if_neg (Ne.symm hd)]

74 def axisPoint {n : ℕ} (t : PhaseVar n) : PhaseVar n → k :=
75   fun i => if i = t then 1 else 0

77 def axisPolynomial {n : ℕ} (t : PhaseVar n) :
78     SymbolRing k n →ₐ[k] MvPolynomial Unit k :=
79   MvPolynomial.aeval (fun i => if i = t then MvPolynomial.X () else 0)

81 theorem axisPolynomial_isHomogeneous {n N : ℕ}
82     (t : PhaseVar n) {P : SymbolRing k n} (hP : P.IsHomogeneous N) :
83     (axisPolynomial k t P).IsHomogeneous N := by
84   simpa [axisPolynomial, ← MvPolynomial.aeval_eq_bind₁] using hP.aeval (n := 1)
85     (fun i => if i = t then MvPolynomial.X () else 0)
86     (fun i => by
87       by_cases hi : i = t
88       · simp [hi, MvPolynomial.isHomogeneous_X]
89       · simp [hi, MvPolynomial.isHomogeneous_zero])

91 theorem eval_axis_eq_eval_axisPolynomial_one {n : ℕ}
92     (t : PhaseVar n) (P : SymbolRing k n) :
93     MvPolynomial.eval (axisPoint k t) P =
94       MvPolynomial.eval (fun _ : Unit => (1 : k)) (axisPolynomial k t P) := by
95   change MvPolynomial.aeval (axisPoint k t) P =
96     MvPolynomial.aeval (fun _ : Unit => (1 : k))
97       (MvPolynomial.aeval
98         (fun i => if i = t then MvPolynomial.X () else 0) P)
99   rw [MvPolynomial.comp_aeval_apply]
100   congr 2
101   funext i
102   by_cases hi : i = t
103   · simp [axisPoint, hi]
104   · simp [axisPoint, hi]

106 theorem eval_axis_eq_pureCoefficient {n N : ℕ}
107     (t : PhaseVar n) {P : SymbolRing k n} (hP : P.IsHomogeneous N) :
108     MvPolynomial.eval (axisPoint k t) P =
109       MvPolynomial.coeff (Finsupp.single () N) (axisPolynomial k t P) := by
110   rw [eval_axis_eq_eval_axisPolynomial_one]
111   rw [homogeneous_unit_eq_monomial k (axisPolynomial k t P)
112     (axisPolynomial_isHomogeneous k t hP)]
113   simp [MvPolynomial.eval_monomial]

115 theorem symbolLinearCombination_isHomogeneous {n : ℕ}
116     (M : Matrix (PhaseVar n) (PhaseVar n) k) (i : PhaseVar n) :
117     (symbolLinearCombination k M i).IsHomogeneous 1 := by
118   rw [symbolLinearCombination]
119   apply MvPolynomial.IsHomogeneous.sum Finset.univ _ 1
120   intro j hj
121   simpa using (MvPolynomial.isHomogeneous_C (PhaseVar n) (M i j)).mul
122     (MvPolynomial.isHomogeneous_X (R := k) j)

134     (t : PhaseVar n) (P : SymbolRing k n) :

142 theorem pureCoefficient_symbolLinearAlgHom {n N : ℕ}
143     (M : Matrix (PhaseVar n) (PhaseVar n) k)
144     (t : PhaseVar n) {P : SymbolRing k n} (hP : P.IsHomogeneous N) :
145     MvPolynomial.coeff (Finsupp.single () N)
146         (axisPolynomial k t (symbolLinearAlgHom k M P)) =
