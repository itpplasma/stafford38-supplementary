10 /-!
11 # Homogeneous symbols and nonzero chart columns
12 
13 Over a characteristic-zero field, a nonzero homogeneous phase-space symbol
14 has a nonvanishing point over the original field. Linear substitution
15 identifies evaluation along a chosen matrix column with the corresponding
16 pure-power coefficient. `SymplecticCompletion` constructs a symplectic matrix
17 with the required nonzero vector in that column.
18 -/

29 theorem exists_eval_ne_zero_of_homogeneous [CharZero k]
30     {n N : ℕ} {P : SymbolRing k n}
31     (hP : P.IsHomogeneous N) (hne : P ≠ 0) :
32     ∃ v : PhaseVar n → k, MvPolynomial.eval v P ≠ 0 := by
33   by_contra h
34   push_neg at h
35   exact hne (hP.eq_zero_of_forall_eval_eq_zero h)

37 theorem eval_symbolLinearAlgHom {n : ℕ}
38     (M : Matrix (PhaseVar n) (PhaseVar n) k)
39     (v : PhaseVar n → k) (P : SymbolRing k n) :
40     MvPolynomial.eval v (symbolLinearAlgHom k M P) =
41       MvPolynomial.eval (fun i => ∑ j, M i j * v j) P := by
42   rw [symbolLinearAlgHom]
43   change MvPolynomial.aeval v
44       (MvPolynomial.aeval (symbolLinearCombination k M) P) =
45     MvPolynomial.aeval (fun i => ∑ j, M i j * v j) P
46   rw [MvPolynomial.comp_aeval_apply]
47   congr 2
48   funext i
49   simp [symbolLinearCombination]

51 theorem homogeneous_unit_eq_monomial {N : ℕ}
52     (P : MvPolynomial Unit k) (hP : P.IsHomogeneous N) :
53     P = MvPolynomial.monomial (Finsupp.single () N)
54       (MvPolynomial.coeff (Finsupp.single () N) P) := by
55   ext d
56   by_cases hd : d = Finsupp.single () N
57   · subst d
58     simp
59   · have hdegree : d.degree ≠ N := by
60       intro hdeg
61       apply hd
62       have hdform : d = Finsupp.single () (d ()) := by
63         apply Finsupp.ext
64         intro i
65         rcases i with ⟨⟩
66         simp
67       rw [hdform, Finsupp.degree_single] at hdeg
68       rw [hdform, hdeg]
69     rw [hP.coeff_eq_zero hdegree]
70     rw [MvPolynomial.coeff_monomial, if_neg (Ne.symm hd)]

72 def axisPoint {n : ℕ} (t : PhaseVar n) : PhaseVar n → k :=
73   fun i => if i = t then 1 else 0

75 def axisPolynomial {n : ℕ} (t : PhaseVar n) :
76     SymbolRing k n →ₐ[k] MvPolynomial Unit k :=
77   MvPolynomial.aeval (fun i => if i = t then MvPolynomial.X () else 0)

79 theorem axisPolynomial_isHomogeneous {n N : ℕ}
80     (t : PhaseVar n) {P : SymbolRing k n} (hP : P.IsHomogeneous N) :
81     (axisPolynomial k t P).IsHomogeneous N := by
82   simpa [axisPolynomial, ← MvPolynomial.aeval_eq_bind₁] using hP.aeval (n := 1)
83     (fun i => if i = t then MvPolynomial.X () else 0)
84     (fun i => by
85       by_cases hi : i = t
86       · simp [hi, MvPolynomial.isHomogeneous_X]
87       · simp [hi, MvPolynomial.isHomogeneous_zero])

89 theorem eval_axis_eq_eval_axisPolynomial_one {n : ℕ}
90     (t : PhaseVar n) (P : SymbolRing k n) :
91     MvPolynomial.eval (axisPoint k t) P =
92       MvPolynomial.eval (fun _ : Unit => (1 : k)) (axisPolynomial k t P) := by
93   change MvPolynomial.aeval (axisPoint k t) P =
94     MvPolynomial.aeval (fun _ : Unit => (1 : k))
95       (MvPolynomial.aeval
96         (fun i => if i = t then MvPolynomial.X () else 0) P)
97   rw [MvPolynomial.comp_aeval_apply]
98   congr 2
99   funext i
100   by_cases hi : i = t
101   · simp [axisPoint, hi]
102   · simp [axisPoint, hi]

104 theorem eval_axis_eq_pureCoefficient {n N : ℕ}
105     (t : PhaseVar n) {P : SymbolRing k n} (hP : P.IsHomogeneous N) :
106     MvPolynomial.eval (axisPoint k t) P =
107       MvPolynomial.coeff (Finsupp.single () N) (axisPolynomial k t P) := by
108   rw [eval_axis_eq_eval_axisPolynomial_one]
109   rw [homogeneous_unit_eq_monomial k (axisPolynomial k t P)
110     (axisPolynomial_isHomogeneous k t hP)]
111   simp [MvPolynomial.eval_monomial]

113 theorem symbolLinearCombination_isHomogeneous {n : ℕ}
114     (M : Matrix (PhaseVar n) (PhaseVar n) k) (i : PhaseVar n) :
115     (symbolLinearCombination k M i).IsHomogeneous 1 := by
116   rw [symbolLinearCombination]
117   apply MvPolynomial.IsHomogeneous.sum Finset.univ _ 1
118   intro j hj
119   simpa using (MvPolynomial.isHomogeneous_C (PhaseVar n) (M i j)).mul
120     (MvPolynomial.isHomogeneous_X (R := k) j)

123     (M : Matrix (PhaseVar n) (PhaseVar n) k)
124     {P : SymbolRing k n} (hP : P.IsHomogeneous N) :
125     (symbolLinearAlgHom k M P).IsHomogeneous N := by

131     (M : Matrix (PhaseVar n) (PhaseVar n) k)
132     (t : PhaseVar n) (P : SymbolRing k n) :
133     MvPolynomial.eval (axisPoint k t) (symbolLinearAlgHom k M P) =

140 theorem pureCoefficient_symbolLinearAlgHom {n N : ℕ}
141     (M : Matrix (PhaseVar n) (PhaseVar n) k)
142     (t : PhaseVar n) {P : SymbolRing k n} (hP : P.IsHomogeneous N) :
143     MvPolynomial.coeff (Finsupp.single () N)
144         (axisPolynomial k t (symbolLinearAlgHom k M P)) =
145       MvPolynomial.eval (fun i => M i t) P := by

148   exact eval_symbolLinearAlgHom_axis k M t P
