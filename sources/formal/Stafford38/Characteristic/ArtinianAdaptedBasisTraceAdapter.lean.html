12 # Adapted bases and square-zero quotient descent
13 
14 This file formalizes elementary inputs used near Singh--Kumar Proposition 3.2.
15 The special fibre is a finite-dimensional module over a commutative Artinian
16 local algebra.  A basis adapted to the powers of the maximal ideal makes left
17 multiplication by a maximal-ideal element strictly upper triangular.
18 
19 The square-zero exactness hypothesis is used at the precise point where the
20 paper uses `cM = Ann_M(c)`: an equality after multiplication by `c` descends
21 to equality of the induced operators on `M / cM`.
22 
23 The scalar-plus-maximal-ideal decomposition is converted to a matrix identity.
24 The final theorem is only a conditional assembler: its matrix-decomposition
25 hypothesis already implies trace zero. It is not a formalization of equation
26 (3.3), and neither `q = 0` nor the concrete deformation-ring/right-Rees
27 adapter is proved here.
28 -/

49 /-- Left multiplication on the special-fibre module, viewed as a
50 `K`-linear endomorphism. -/
51 def leftMultiplicationEnd (r : R) : Module.End K V where
52   toFun v := r • v
53   map_add' := smul_add r
54   map_smul' a v := by exact smul_comm r a v

57 @[simp] theorem leftMultiplicationEnd_apply (r : R) (v : V) :
58     leftMultiplicationEnd (K := K) (V := V) r v = r • v := rfl

60 /-- A basis genuinely adapted to the maximal-ideal-power filtration.
61 
62 `level i` records the last filtration layer containing `b i`.  The ordering
63 condition says deeper vectors occur earlier, as in the basis construction in
64 Proposition 3.2.  The second field identifies every power-filtration layer
65 with the span of the corresponding basis vectors. -/
66 structure IsMaximalIdealFiltrationAdapted
67     (m : Ideal R)
68     (b : Module.Basis (Fin (n + 1)) K V) (level : Fin (n + 1) → ℕ) : Prop where
69   antitone_level : Antitone level
70   power_span : ∀ t : ℕ,
71     ((m ^ t • (⊤ : Submodule R V)).restrictScalars K) =
72       Submodule.span K (b '' {i | t ≤ level i})
73 
74 /-- Multiplication by a maximal-ideal element raises the power filtration by
75 one. -/
76 lemma maximalIdeal_smul_mem_succ_power
77     {r : R} (hr : r ∈ 𝔪) {t : ℕ} {v : V}
78     (hv : v ∈ 𝔪 ^ t • (⊤ : Submodule R V)) :
79     r • v ∈ 𝔪 ^ (t + 1) • (⊤ : Submodule R V) := by
80   refine Submodule.smul_induction_on
81     (p := fun v => r • v ∈ 𝔪 ^ (t + 1) • (⊤ : Submodule R V)) hv ?_ ?_
82   · intro (a : R) ha w hw
83     rw [smul_smul]
84     apply Submodule.smul_mem_smul
85     · rw [pow_succ']
86       exact Ideal.mul_mem_mul hr ha
87     · exact Submodule.mem_top
88   · intro x y hx hy
89     rw [smul_add]
90     exact Submodule.add_mem _ hx hy

92 /-- The left multiplication matrix of a maximal-ideal element is strictly
93 upper triangular in a maximal-ideal-filtration adapted basis. -/
94 theorem leftMultiplicationMatrix_isStrictUpperTriangular
95     (b : Module.Basis (Fin (n + 1)) K V) (level : Fin (n + 1) → ℕ)
96     (hadapted : IsMaximalIdealFiltrationAdapted 𝔪 b level)
97     {r : R} (hr : r ∈ 𝔪) :
98     IsStrictUpperTriangular
99       (LinearMap.toMatrix b b (leftMultiplicationEnd (K := K) (V := V) r)) := by
100   intro i j hji
101   rw [LinearMap.toMatrix_apply]
102   have hbjSpan :
103       b j ∈ Submodule.span K (b '' {i | level j ≤ level i}) := by
104     exact (b.self_mem_span_image).mpr (show level j ≤ level j from le_rfl)
105   have hbjPower : b j ∈ 𝔪 ^ level j • (⊤ : Submodule R V) := by
106     have := hbjSpan
107     rw [← hadapted.power_span (level j)] at this
108     exact this
109   have hraised :
110       r • b j ∈ 𝔪 ^ (level j + 1) • (⊤ : Submodule R V) :=
111     maximalIdeal_smul_mem_succ_power hr hbjPower
112   have hspan :
113       r • b j ∈ Submodule.span K (b '' {i | level j + 1 ≤ level i}) := by
114     rw [← hadapted.power_span (level j + 1)]
115     exact hraised
116   have hsupp := b.repr_support_subset_of_mem_span
117     {i | level j + 1 ≤ level i} hspan
118   by_contra hne
119   have hiSupport : i ∈ (b.repr (r • b j)).support := Finsupp.mem_support_iff.mpr hne
120   have hlevel : level j + 1 ≤ level i := hsupp hiSupport
121   have hreverse : level i ≤ level j := hadapted.antitone_level hji
122   omega

124 /-- In the Artinian-local case, the matrix of multiplication by a
125 maximal-ideal element is nilpotent. -/
126 theorem leftMultiplicationMatrix_isNilpotent
127     [IsArtinianRing R]
128     (b : Module.Basis (Fin (n + 1)) K V) {r : R} (hr : r ∈ 𝔪) :
129     IsNilpotent
130       (LinearMap.toMatrix b b (leftMultiplicationEnd (K := K) (V := V) r)) := by
131   have hrnil : IsNilpotent r := by
132     obtain ⟨N, hN⟩ :=
133       Stafford38.Characteristic.ArtinianCoefficientField.maximalIdeal_isNilpotent
134         (R := R)
135     refine ⟨N, ?_⟩
136     have hrpow : r ^ N ∈ 𝔪 ^ N := Ideal.pow_mem_pow hr N
137     rw [hN] at hrpow
138     exact hrpow
139   have hend : IsNilpotent (leftMultiplicationEnd (K := K) (V := V) r) := by
140     obtain ⟨N, hN⟩ := hrnil
141     refine ⟨N, ?_⟩
142     ext v
143     have hp : ∀ (j : ℕ) (w : V),
144         ((leftMultiplicationEnd (K := K) (V := V) r) ^ j) w = r ^ j • w := by
145       intro j
146       induction j with
147       | zero => intro w; simp
148       | succ j ih =>
149           intro w
150           rw [pow_succ, Module.End.mul_apply, ih, leftMultiplicationEnd_apply]
151           exact (SemigroupAction.mul_smul (r ^ j) r w).symm.trans (by rw [pow_succ])
152     rw [hp N v, hN, zero_smul, LinearMap.zero_apply]
153   exact hend.map (LinearMap.toMatrixAlgEquiv b)

155 /-- Matrix form of equation (3.1): the coefficient-field scalar contributes
156 `q I`, and the maximal-ideal remainder contributes its multiplication
157 matrix. -/
158 theorem leftMultiplicationMatrix_decomposition
159     (b : Module.Basis (Fin (n + 1)) K V) (q : K) (r z : R)
160     (hz : z = algebraMap K R q + r) :
161     LinearMap.toMatrix b b (leftMultiplicationEnd (K := K) (V := V) z) =
162       q • (1 : Matrix (Fin (n + 1)) (Fin (n + 1)) K) +
163         LinearMap.toMatrix b b
164           (leftMultiplicationEnd (K := K) (V := V) r) := by
165   subst z
166   have hend :
167       leftMultiplicationEnd (K := K) (V := V) (algebraMap K R q + r) =
168         q • LinearMap.id + leftMultiplicationEnd (K := K) (V := V) r := by
169     ext v
170     simp [leftMultiplicationEnd, add_smul, IsScalarTower.algebraMap_smul]
171   rw [hend, (LinearMap.toMatrix b b).map_add,
172     (LinearMap.toMatrix b b).map_smul]
173   simp

183 /-- Square-zero exactness descends an equality after multiplication by the
184 parameter to equality of the induced special-fibre operators.
185 
186 This is the exact square-zero quotient bridge: from
187 `C ∘ Z = C ∘ T` and `ker C = range C`, the actions of `Z` and `T` agree on
188 `W / range C`. -/
189 theorem quotientEnd_eq_of_parameter_comp_eq
190     (C Z T : Module.End K W)
191     (hExact : LinearMap.ker C = LinearMap.range C)
192     (hZC : Z.comp C = C.comp Z)
193     (hTC : T.comp C = C.comp T)
194     (hparameter : C.comp Z = C.comp T) :
195     quotientEnd C Z hZC = quotientEnd C T hTC := by
196   apply LinearMap.ext
197   intro q
198   obtain ⟨w, rfl⟩ := Submodule.mkQ_surjective (LinearMap.range C) q
199   apply (Submodule.Quotient.eq (LinearMap.range C)).mpr
200   rw [← hExact, LinearMap.mem_ker]
201   change C (Z w - T w) = 0
202   rw [map_sub, sub_eq_zero]
203   exact LinearMap.congr_fun hparameter w

205 /-- A signed sum of three matrices that will be used only in the conditional
206 matrix assembler below. -/
207 def strictTriangularSignedSum
208     (X Y Γ : Matrix (Fin (n + 1)) (Fin (n + 1)) K) :
209     Matrix (Fin (n + 1)) (Fin (n + 1)) K :=
210   X - Y + Γ
211 
212 /-- A signed sum of strictly upper-triangular matrices is strictly upper
213 triangular. -/
214 theorem strictTriangularSignedSum_isStrictUpperTriangular
215     (X Y Γ : Matrix (Fin (n + 1)) (Fin (n + 1)) K)
216     (hX : IsStrictUpperTriangular X)
217     (hY : IsStrictUpperTriangular Y)
218     (hΓ : IsStrictUpperTriangular Γ) :
219     IsStrictUpperTriangular (strictTriangularSignedSum X Y Γ) := by
220   intro i j hji
221   simp [strictTriangularSignedSum, hX i j hji, hY i j hji, hΓ i j hji]

223 /-- Conditional trace assembler for a supplied matrix decomposition.
224 
225 The actual basis is on the special fibre `W / range C`.  Source-specific
226 upstream data has already produced the strict upper-triangular remainder
227 `R₀`; `hdecomposition` assumes that the corrected quotient operator is a
228 strictly triangular matrix plus commutators. This hypothesis already carries
229 the trace-zero content and is not claimed to follow directly from equation
230 (3.3). Square-zero exactness only replaces `T` by the quotient action of `Z`.
231 -/
232 theorem adaptedBasis_conditional_trace_assembler
233     (C Z T : Module.End K W)
234     (hExact : LinearMap.ker C = LinearMap.range C)
235     (hZC : Z.comp C = C.comp Z)
236     (hTC : T.comp C = C.comp T)
237     (hparameter : C.comp Z = C.comp T)
238     (b : Module.Basis (Fin (n + 1)) K (W ⧸ LinearMap.range C))
239     (R₀ Θ Φ Ψ H : Matrix (Fin (n + 1)) (Fin (n + 1)) K)
240     (hR₀ : IsStrictUpperTriangular R₀)
241     (hdecomposition :
242       LinearMap.toMatrix b b (quotientEnd C T hTC) =
243         adaptedFirstOrderMatrix R₀ Θ Φ Ψ H) :
244     LinearMap.toMatrix b b (quotientEnd C Z hZC) =
245         adaptedFirstOrderMatrix R₀ Θ Φ Ψ H ∧
246       Matrix.trace (LinearMap.toMatrix b b (quotientEnd C Z hZC)) = 0 := by
247   have hdesc := quotientEnd_eq_of_parameter_comp_eq
248     C Z T hExact hZC hTC hparameter
249   constructor
250   · rw [hdesc]
251     exact hdecomposition
252   · rw [hdesc, hdecomposition]
253     exact trace_adaptedFirstOrderMatrix_eq_zero R₀ Θ Φ Ψ H hR₀

275 /-- Conditional assembly of adapted-basis data with a supplied trace-zero
276 matrix decomposition.
277 
278 The basis `b`, indexed by the nonempty finite type `Fin (n+1)`, exhibits the
279 special fibre `V` as nonzero and finite-dimensional over `K`; `V` is also a
280 module over the commutative Artinian local algebra `R`.  Its adaptedness makes
281 the multiplication matrix `U` of the maximal-ideal remainder in (3.1)
282 strictly upper triangular, while Artinianness makes `U` nilpotent.  This is
283 kept distinct from the strict triangular correction remainder `Δ` in (3.3).
284 
285 The equivalence `e` identifies `V` with the square-zero quotient of the
286 deformation module. Exactness descends equality of the supplied operators to
287 the special fibre. The hypothesis `hdecomposition` already presents the
288 corrected operator as a strictly triangular matrix plus commutators.
289 
290 The conclusion packages strictness and nilpotence of the (3.1) remainder,
291 trace zero conditional on the supplied decomposition, and the
292 scalar-plus-remainder matrix identity.  It does not assume or prove the scalar
293 conclusion `q = 0`. It does not derive the supplied matrix decomposition from
294 Singh--Kumar equation (3.3). -/
295 theorem adaptedBasis_conditional_terminal_inputs
296     (b : Module.Basis (Fin (n + 1)) K V) (level : Fin (n + 1) → ℕ)
297     (hadapted : IsMaximalIdealFiltrationAdapted 𝔪 b level)
298     (q : K) (r z : R) (hr : r ∈ 𝔪)
299     (hz : z = algebraMap K R q + r)
300     (C Z T : Module.End K W)
301     (hExact : LinearMap.ker C = LinearMap.range C)
302     (hZC : Z.comp C = C.comp Z)
303     (hTC : T.comp C = C.comp T)
304     (hparameter : C.comp Z = C.comp T)
305     (e : (W ⧸ LinearMap.range C) ≃ₗ[K] V)
306     (hZaction :
307       LinearMap.toMatrix (b.map e.symm) (b.map e.symm)
308           (quotientEnd C Z hZC) =
309         LinearMap.toMatrix b b
310           (leftMultiplicationEnd (K := K) (V := V) z))
311     (Xcorr Ycorr Γ : Matrix (Fin (n + 1)) (Fin (n + 1)) K)
312     (Θ Φ Ψ H : Matrix (Fin (n + 1)) (Fin (n + 1)) K)
313     (hXcorr : IsStrictUpperTriangular Xcorr)
314     (hYcorr : IsStrictUpperTriangular Ycorr)
315     (hΓ : IsStrictUpperTriangular Γ)
316     (hdecomposition :
317       LinearMap.toMatrix (b.map e.symm) (b.map e.symm)
318           (quotientEnd C T hTC) =
319         adaptedFirstOrderMatrix
320           (strictTriangularSignedSum Xcorr Ycorr Γ) Θ Φ Ψ H) :
321     IsStrictUpperTriangular
322         (LinearMap.toMatrix b b
323           (leftMultiplicationEnd (K := K) (V := V) r)) ∧
324       IsNilpotent
325         (LinearMap.toMatrix b b
326           (leftMultiplicationEnd (K := K) (V := V) r)) ∧
327       Matrix.trace
328         (LinearMap.toMatrix (b.map e.symm) (b.map e.symm)
329           (quotientEnd C Z hZC)) = 0 ∧
330       q • (1 : Matrix (Fin (n + 1)) (Fin (n + 1)) K) +
331           LinearMap.toMatrix b b
332             (leftMultiplicationEnd (K := K) (V := V) r) =
333         LinearMap.toMatrix (b.map e.symm) (b.map e.symm)
334           (quotientEnd C Z hZC) := by
335   have hadaptedR : IsStrictUpperTriangular
336       (LinearMap.toMatrix b b
337         (leftMultiplicationEnd (K := K) (V := V) r)) :=
338     leftMultiplicationMatrix_isStrictUpperTriangular b level hadapted hr
339   have hmatrix := adaptedBasis_conditional_trace_assembler
340     C Z T hExact hZC hTC hparameter (b.map e.symm)
341       (strictTriangularSignedSum Xcorr Ycorr Γ) Θ Φ Ψ H
342       (strictTriangularSignedSum_isStrictUpperTriangular
343         Xcorr Ycorr Γ hXcorr hYcorr hΓ) hdecomposition
344   refine ⟨hadaptedR, leftMultiplicationMatrix_isNilpotent b hr, hmatrix.2, ?_⟩
345   rw [hZaction, ← leftMultiplicationMatrix_decomposition b q r z hz]
