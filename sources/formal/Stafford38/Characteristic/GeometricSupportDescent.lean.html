11 /-!
12 # Geometric points and support after algebraic closure
13 
14 For finitely many polynomial variables over an arbitrary field, a proper
15 ideal has a common zero in the algebraic closure.  This file records the
16 result with the coefficient map and its map/comap consequences explicit.
17 
18 The proof chooses a maximal ideal above the given ideal, uses Zariski's lemma
19 to make its residue field algebraic over the ground field, and embeds that
20 residue field into the algebraic closure.  It does not use characteristic
21 varieties, Gabber's theorem, or asymptotic geometry.
22 -/

37 /-- A proper ideal in a finite-variable polynomial ring has a common zero in
38 the algebraic closure of the coefficient field. -/
39 theorem exists_algebraicClosure_zero_of_ne_top
40     (J : Ideal (MvPolynomial σ k)) (hJ : J ≠ ⊤) :
41     ∃ q : σ → AlgebraicClosure k,
42       q ∈ MvPolynomial.zeroLocus (AlgebraicClosure k)
43         (J.map (scalarPolynomialMap
44           (k := k) (K := AlgebraicClosure k) σ)) := by
45   obtain ⟨M, hMmax, hJM⟩ := Ideal.exists_le_maximal J hJ
46   letI : M.IsMaximal := hMmax
47   let Q := MvPolynomial σ k ⧸ M
48   letI : Field Q := Ideal.Quotient.field M
49   letI : Module.IsTorsionFree k Q :=
50     Module.isTorsionFree_iff_algebraMap_injective.mpr
51       (MvPolynomial.quotient_mk_comp_C_injective σ k M hMmax.ne_top)
52   have hIntegral :
53       RingHom.IsIntegral
54         ((Ideal.Quotient.mk M).comp (MvPolynomial.C : k →+* MvPolynomial σ k)) :=
55     MvPolynomial.comp_C_integral_of_surjective_of_isJacobsonRing
56       (Ideal.Quotient.mk M) Ideal.Quotient.mk_surjective
57   letI : Algebra.IsIntegral k Q := ⟨hIntegral⟩
58   letI : Algebra.IsAlgebraic k Q := Algebra.IsIntegral.isAlgebraic
59   let φ : Q →ₐ[k] AlgebraicClosure k := IsAlgClosed.lift
60   let q : σ → AlgebraicClosure k := fun i => φ (Ideal.Quotient.mk M (MvPolynomial.X i))
61   refine ⟨q, (mem_zeroLocus_map_iff J q).2 ?_⟩
62   intro f hf
63   let ψ : MvPolynomial σ k →+* AlgebraicClosure k :=
64     φ.toRingHom.comp (Ideal.Quotient.mk M)
65   have hψ : ψ f = 0 := by
66     change φ (Ideal.Quotient.mk M f) = 0
67     rw [Ideal.Quotient.eq_zero_iff_mem.mpr (hJM hf), map_zero]
68   have hcoeff : ψ.comp MvPolynomial.C =
69       algebraMap k (AlgebraicClosure k) := by
70     ext a
71     change φ (Ideal.Quotient.mk M (MvPolynomial.C a)) =
72       algebraMap k (AlgebraicClosure k) a
73     rw [show Ideal.Quotient.mk M (MvPolynomial.C a) = algebraMap k Q a from rfl]
74     exact φ.commutes a
75   rw [MvPolynomial.map_mvPolynomial_eq_eval₂ ψ] at hψ
76   rw [hcoeff] at hψ
77   simpa [ψ, q] using hψ

79 /-- Over the algebraic closure, geometric emptiness is exactly the unit-ideal
80 condition on the original ground-field ideal. -/
81 theorem algebraicClosure_zeroLocus_eq_empty_iff
82     (J : Ideal (MvPolynomial σ k)) :
83     MvPolynomial.zeroLocus (AlgebraicClosure k)
84         (J.map (scalarPolynomialMap
85           (k := k) (K := AlgebraicClosure k) σ)) = ∅ ↔
86       J = ⊤ := by
87   constructor
88   · intro hzero
89     by_contra hJ
90     obtain ⟨q, hq⟩ := exists_algebraicClosure_zero_of_ne_top J hJ
91     rw [hzero] at hq
92     exact hq
93   · rintro rfl
94     change MvPolynomial.zeroLocus (AlgebraicClosure k)
95       ((⊤ : Ideal (MvPolynomial σ k)).map
96         (scalarPolynomialMap
97           (k := k) (K := AlgebraicClosure k) σ)) =
98         (⊥ : Set (σ → AlgebraicClosure k))
99     rw [Ideal.map_top, MvPolynomial.zeroLocus_top]
100 
101 /-- Scalar extension to the algebraic closure reflects and preserves the
102 unit ideal for finite-variable polynomial rings. -/
103 theorem map_scalarPolynomialMap_eq_top_iff
104     (J : Ideal (MvPolynomial σ k)) :
105     J.map (scalarPolynomialMap
106         (k := k) (K := AlgebraicClosure k) σ) = ⊤ ↔
107       J = ⊤ := by
108   constructor
109   · intro hmap
110     apply (algebraicClosure_zeroLocus_eq_empty_iff J).mp
111     rw [hmap]
112     exact MvPolynomial.zeroLocus_top
113   · rintro rfl
114     exact Ideal.map_top _

116 /-- A nonempty ground-field prime-spectrum zero locus yields a geometric
117 point after passing to the algebraic closure. -/
118 theorem exists_algebraicClosure_zero_of_mem_primeZeroLocus
119     (J : Ideal (MvPolynomial σ k))
120     (p : PrimeSpectrum (MvPolynomial σ k))
121     (hp : p ∈ PrimeSpectrum.zeroLocus (J : Set (MvPolynomial σ k))) :
122     ∃ q : σ → AlgebraicClosure k,
123       q ∈ MvPolynomial.zeroLocus (AlgebraicClosure k)
124         (J.map (scalarPolynomialMap
125           (k := k) (K := AlgebraicClosure k) σ)) := by
126   apply exists_algebraicClosure_zero_of_ne_top J
127   intro hJ
128   subst J
129   exact p.2.ne_top (by simpa using hp)

131 /-- A geometric point gives a prime of the original polynomial ring by
132 contracting its evaluation kernel.  The displayed equality records the exact
133 map/comap orientation. -/
134 def groundPrimeOfAlgebraicClosurePoint (q : σ → AlgebraicClosure k) :
135     PrimeSpectrum (MvPolynomial σ k) :=
136   PrimeSpectrum.comap (scalarPolynomialMap
137     (k := k) (K := AlgebraicClosure k) σ)
138     (MvPolynomial.pointToPoint q)
139 
140 theorem groundPrimeOfAlgebraicClosurePoint_asIdeal
141     (q : σ → AlgebraicClosure k) :
142     (groundPrimeOfAlgebraicClosurePoint (k := k) q).asIdeal =
143       (MvPolynomial.vanishingIdeal (AlgebraicClosure k) {q}).comap
144         (scalarPolynomialMap
145           (k := k) (K := AlgebraicClosure k) σ) :=
146   rfl

148 /-- Conversely, a geometric zero of the extended ideal contracts to a prime
149 of the ground polynomial ring containing the original ideal. -/
150 theorem groundPrimeOfAlgebraicClosurePoint_mem_zeroLocus
151     (J : Ideal (MvPolynomial σ k)) (q : σ → AlgebraicClosure k)
152     (hq : q ∈ MvPolynomial.zeroLocus (AlgebraicClosure k)
153       (J.map (scalarPolynomialMap
154         (k := k) (K := AlgebraicClosure k) σ))) :
155     groundPrimeOfAlgebraicClosurePoint (k := k) q ∈
156       PrimeSpectrum.zeroLocus (J : Set (MvPolynomial σ k)) := by
157   rw [PrimeSpectrum.mem_zeroLocus]
158   intro f hf
159   change scalarPolynomialMap
160       (k := k) (K := AlgebraicClosure k) σ f ∈
161     MvPolynomial.vanishingIdeal (AlgebraicClosure k) {q}
162   rw [MvPolynomial.mem_vanishingIdeal_singleton_iff]
163   exact hq _ (Ideal.mem_map_of_mem _ hf)
164 
165 /-- Thus nonemptiness of the prime-spectrum support and nonemptiness of the
166 algebraic-closure geometric zero locus are equivalent. -/
167 theorem primeZeroLocus_nonempty_iff_algebraicClosure_zeroLocus_nonempty
168     (J : Ideal (MvPolynomial σ k)) :
169     Set.Nonempty (PrimeSpectrum.zeroLocus (J : Set (MvPolynomial σ k))) ↔
170       Set.Nonempty (MvPolynomial.zeroLocus (AlgebraicClosure k)
171         (J.map (scalarPolynomialMap
172           (k := k) (K := AlgebraicClosure k) σ))) := by
173   constructor
174   · rintro ⟨p, hp⟩
175     exact exists_algebraicClosure_zero_of_mem_primeZeroLocus J p hp
176   · rintro ⟨q, hq⟩
177     exact ⟨groundPrimeOfAlgebraicClosurePoint (k := k) q,
178       groundPrimeOfAlgebraicClosurePoint_mem_zeroLocus J q hq⟩

180 /-- Project-facing form: a proper order initial ideal has a geometric point
181 after coefficient extension to the algebraic closure. -/
182 theorem exists_algebraicClosure_orderInitialZero_of_ne_top
183     {n : ℕ} (I : RightIdeal (WeylAlg k n))
184     (hI : orderInitialIdeal k I ≠ ⊤) :
185     ∃ q : PhaseVar n → AlgebraicClosure k,
186       q ∈ MvPolynomial.zeroLocus (AlgebraicClosure k)
187         ((orderInitialIdeal k I).map
188           (scalarPolynomialMap
189             (k := k) (K := AlgebraicClosure k) (PhaseVar n))) :=
190   exists_algebraicClosure_zero_of_ne_top (orderInitialIdeal k I) hI

192 /-- The literal prime-spectrum order support is nonempty exactly when the
193 extended order initial ideal has an algebraic-closure-valued point. -/
194 theorem orderCharacteristicSupport_nonempty_iff_geometric_nonempty
195     {n : ℕ} (I : RightIdeal (WeylAlg k n)) :
196     Set.Nonempty (orderCharacteristicSupport k I) ↔
197       Set.Nonempty (MvPolynomial.zeroLocus (AlgebraicClosure k)
198         ((orderInitialIdeal k I).map
199           (scalarPolynomialMap
200             (k := k) (K := AlgebraicClosure k) (PhaseVar n)))) := by
201   rw [orderCharacteristicSupport_eq_zeroLocus]
202   exact primeZeroLocus_nonempty_iff_algebraicClosure_zeroLocus_nonempty
203     (orderInitialIdeal k I)

205 /-- Geometric emptiness over the algebraic closure descends all the way to
206 the unit order initial ideal and hence to empty prime-spectrum support. -/
207 theorem geometric_orderInitialZeroLocus_empty_iff_support_empty
208     {n : ℕ} (I : RightIdeal (WeylAlg k n)) :
209     MvPolynomial.zeroLocus (AlgebraicClosure k)
210         ((orderInitialIdeal k I).map
211           (scalarPolynomialMap
212             (k := k) (K := AlgebraicClosure k) (PhaseVar n))) = ∅ ↔
213       orderCharacteristicSupport k I = ∅ := by
214   rw [algebraicClosure_zeroLocus_eq_empty_iff,
215     orderCharacteristicSupport_eq_empty_iff]
