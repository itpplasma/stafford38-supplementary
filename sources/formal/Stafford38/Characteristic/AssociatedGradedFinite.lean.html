10 # Finiteness of the actual order-associated graded module
11 
12 The actual associated graded quotient is cyclic over the commutative symbol
13 ring.  This file records the resulting `Module.Finite` instance and feeds it
14 to the already checked minimal-prime localization theorem.  No Noetherian,
15 trace, or involutivity conclusion is hidden in the construction.
16 -/

34 /-- The cyclic presentation map from the symbol ring to the literal external
35 direct sum associated graded module. -/
36 def generatorMap (I : RightIdeal (WeylAlg k n)) :
37     SymbolRing k n →ₗ[SymbolRing k n] OrderAssociatedGradedModule k I :=
38   LinearMap.toSpanSingleton (SymbolRing k n)
39     (OrderAssociatedGradedModule k I)
40     (orderAssociatedGradedGenerator k I)
41 
42 /-- Cyclicity of the actual associated graded object is surjectivity of its
43 literal one-generator presentation map. -/
44 theorem generatorMap_surjective (I : RightIdeal (WeylAlg k n)) :
45     Function.Surjective (generatorMap k I) := by
46   intro q
47   obtain ⟨P, hP⟩ := exists_smul_orderAssociatedGradedGenerator k I q
48   exact ⟨P, by simpa [generatorMap, LinearMap.toSpanSingleton_apply] using hP⟩
49 
50 /-- The literal order-associated graded quotient is finite over the symbol
51 ring, with one generator. -/
52 noncomputable instance orderAssociatedGradedModule_finite
53     (I : RightIdeal (WeylAlg k n)) :
54     Module.Finite (SymbolRing k n) (OrderAssociatedGradedModule k I) :=
55   Module.Finite.of_surjective (generatorMap k I) (generatorMap_surjective k I)

57 /-- Concrete minimal-prime localization package with no unproduced finiteness
58 hypothesis: exact two-jet specialization, nontrivial localized fibre, and
59 finite length all hold at a minimal prime over the actual annihilator. -/
60 theorem exists_minimalPrimeLocalizedExactnessAndFiniteLength
61     [IsNoetherianRing (SymbolRing k n)]
62     (I : RightIdeal (WeylAlg k n))
63     (P : Ideal (SymbolRing k n)) [P.IsPrime]
64     (hP : P ∈
65       (Module.annihilator (SymbolRing k n)
66         (OrderAssociatedGradedModule k I)).minimalPrimes) :
67     ∃ h : OreLocalization.OreSet
68         (OppositeDenominators (filteredQuotientTwoJetTraceData k I)
69           P.primeCompl),
70       LocalizedExactnessFor (filteredQuotientTwoJetTraceData k I)
71           P.primeCompl h ∧
72         Nontrivial
73           (LocalizedModule P.primeCompl (OrderAssociatedGradedModule k I)) ∧
74         IsFiniteLength (Localization P.primeCompl)
75           (LocalizedModule P.primeCompl
76             (OrderAssociatedGradedModule k I)) :=
77   SquareZeroLocalizedExactness.exists_minimalPrimeLocalizedExactnessAndFiniteLength
78     (filteredQuotientTwoJetTraceData k I) P hP
