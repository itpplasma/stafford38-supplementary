67 noncomputable def nonpositiveDirectSumLinearEquiv
68     (e : ∀ m : ℕ, A (negIndex m) ≃ₗ[k] B m)
69     (hpos : ∀ p, 0 < p → Subsingleton (A p)) :
70     DirectSum ℤ A ≃ₗ[k] DirectSum ℕ B := by
71   refine LinearEquiv.ofLinearMap (nonpositiveDirectSumToNat k e)
72     (natDirectSumToNonpositive k e) ?_ ?_
73   · apply DirectSum.linearMap_ext k
74     intro m
75     apply LinearMap.ext
76     intro x
77     change nonpositiveDirectSumToNat k e
78         (natDirectSumToNonpositive k e (DirectSum.lof k ℕ B m x)) =
79       DirectSum.lof k ℕ B m x
80     rw [natDirectSumToNonpositive_lof]
81     cases m with
82     | zero =>
83         rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
84         change DirectSum.lof k ℕ B 0 (e 0 ((e 0).symm x)) = _
85         rw [(e 0).apply_symm_apply]
86     | succ m =>
87         rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
88         change DirectSum.lof k ℕ B (m + 1)
89             (e (m + 1) ((e (m + 1)).symm x)) = _
90         rw [(e (m + 1)).apply_symm_apply]
91   · apply DirectSum.linearMap_ext k
92     intro p
93     apply LinearMap.ext
94     intro x
95     change natDirectSumToNonpositive k e
96         (nonpositiveDirectSumToNat k e (DirectSum.lof k ℤ A p x)) =
97       DirectSum.lof k ℤ A p x
98     cases p with
99     | ofNat n =>
100         cases n with
101         | zero =>
102             rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
103             change natDirectSumToNonpositive k e
104                 (DirectSum.lof k ℕ B 0 (e 0 x)) = DirectSum.lof k ℤ A 0 x
105             rw [natDirectSumToNonpositive_lof]
106             congr 1
107             exact (e 0).symm_apply_apply (show A (negIndex 0) from x)
108         | succ n =>
109             haveI := hpos (Int.ofNat (n + 1)) (by simp)
110             simp [nonpositiveDirectSumToNat, Subsingleton.elim x 0]
111     | negSucc n =>
112         rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
113         change natDirectSumToNonpositive k e
114             (DirectSum.lof k ℕ B (n + 1) (e (n + 1) x)) =
115           DirectSum.lof k ℤ A (Int.negSucc n) x
116         rw [natDirectSumToNonpositive_lof]
117         congr 1
118         exact (e (n + 1)).symm_apply_apply
119           (show A (negIndex (n + 1)) from x)
