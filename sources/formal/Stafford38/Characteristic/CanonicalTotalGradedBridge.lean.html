46 noncomputable def nonpositiveDirectSumToNat
47     (e : ∀ m : ℕ, A (negIndex m) ≃ₗ[k] B m) :
48     DirectSum ℤ A →ₗ[k] DirectSum ℕ B :=
49   DirectSum.toModule k ℤ (DirectSum ℕ B) (fun p =>
50     match p with
51     | Int.ofNat 0 => (DirectSum.lof k ℕ B 0).comp (e 0).toLinearMap

53     | Int.negSucc n =>
54         (DirectSum.lof k ℕ B (n + 1)).comp (e (n + 1)).toLinearMap)

69 noncomputable def nonpositiveDirectSumLinearEquiv
70     (e : ∀ m : ℕ, A (negIndex m) ≃ₗ[k] B m)
71     (hpos : ∀ p, 0 < p → Subsingleton (A p)) :
72     DirectSum ℤ A ≃ₗ[k] DirectSum ℕ B := by
73   refine LinearEquiv.ofLinearMap (nonpositiveDirectSumToNat k e)
74     (natDirectSumToNonpositive k e) ?_ ?_
75   · apply DirectSum.linearMap_ext k
76     intro m
77     apply LinearMap.ext
78     intro x
79     change nonpositiveDirectSumToNat k e
80         (natDirectSumToNonpositive k e (DirectSum.lof k ℕ B m x)) =
81       DirectSum.lof k ℕ B m x
82     rw [natDirectSumToNonpositive_lof]
83     cases m with
84     | zero =>
85         rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
86         change DirectSum.lof k ℕ B 0 (e 0 ((e 0).symm x)) = _
87         rw [(e 0).apply_symm_apply]
88     | succ m =>
89         rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
90         change DirectSum.lof k ℕ B (m + 1)
91             (e (m + 1) ((e (m + 1)).symm x)) = _
92         rw [(e (m + 1)).apply_symm_apply]
93   · apply DirectSum.linearMap_ext k
94     intro p
95     apply LinearMap.ext
96     intro x
97     change natDirectSumToNonpositive k e
98         (nonpositiveDirectSumToNat k e (DirectSum.lof k ℤ A p x)) =
99       DirectSum.lof k ℤ A p x
100     cases p with
101     | ofNat n =>
102         cases n with
103         | zero =>
104             rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
105             change natDirectSumToNonpositive k e
106                 (DirectSum.lof k ℕ B 0 (e 0 x)) = DirectSum.lof k ℤ A 0 x
107             rw [natDirectSumToNonpositive_lof]
108             congr 1
109             exact (e 0).symm_apply_apply (show A (negIndex 0) from x)
110         | succ n =>
111             haveI := hpos (Int.ofNat (n + 1)) (by simp)
112             simp [nonpositiveDirectSumToNat, Subsingleton.elim x 0]
113     | negSucc n =>
114         rw [nonpositiveDirectSumToNat, DirectSum.toModule_lof]
115         change natDirectSumToNonpositive k e
116             (DirectSum.lof k ℕ B (n + 1) (e (n + 1) x)) =
117           DirectSum.lof k ℤ A (Int.negSucc n) x
118         rw [natDirectSumToNonpositive_lof]
119         congr 1
120         exact (e (n + 1)).symm_apply_apply
121           (show A (negIndex (n + 1)) from x)

124     (e : ∀ m : ℕ, A (negIndex m) ≃ₗ[k] B m)
125     (hpos : ∀ p, 0 < p → Subsingleton (A p)) (m : ℕ) (x : A (negIndex m)) :
126     nonpositiveDirectSumLinearEquiv k e hpos
127         (DirectSum.lof k ℤ A (negIndex m) x) =
128       DirectSum.lof k ℕ B m (e m x) := by
