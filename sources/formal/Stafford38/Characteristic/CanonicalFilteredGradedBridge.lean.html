37 theorem G_at_neg
38     (n N m : ℕ) (d : PresentedWeyl k (n + 1))
39     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
40     (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ)) =
41       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
42   change (if -(m : ℤ) ≤ 0 then
43       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) (- - (m : ℤ)).toNat else ⊥) = _
44   rw [if_pos (by omega)]
45   have hidx : (- - (m : ℤ)).toNat = m := by omega
46   rw [hidx]

48 theorem G_succ_neg
49     (n N m : ℕ) (d : PresentedWeyl k (n + 1))
50     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
51     (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ) + 1) =
52       quotientOrderStrictLowerPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
53   by_cases hm : m = 0
54   · subst m
55     norm_num
56     change (if (1 : ℤ) ≤ 0 then
57       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) 1 else
58       (⊥ : Submodule k (CanonicalQuotient k n N d))) = _
59     rw [if_neg (by norm_num)]
60     rw [quotientOrderStrictLowerPiece]
61     change (⊥ : Submodule k (CanonicalQuotient k n N d)) =
62       (presentedStrictLowerPiece k orderWeight 0).map _
63     rw [show presentedStrictLowerPiece k orderWeight 0 = ⊥ by rfl]
64     simp
65   · obtain ⟨m', rfl⟩ := Nat.exists_eq_succ_of_ne_zero hm
66     rw [show (-(↑(m' + 1) : ℤ) + 1) = -(m' : ℤ) by omega,
67       G_at_neg]
68     rw [quotientOrderStrictLowerPiece_succ_eq]

70 theorem zeroPage_source_cycles_eq_orderPiece
71     (n N m : ℕ) (d : PresentedWeyl k (n + 1))
72     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
73     (canonicalFilteredTwoTerm k n N d hd).cycles 0 (-(m : ℤ)) =
74       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
75   rw [FilteredTwoTerm.cycles, G_at_neg k n N m d hd]
76   apply inf_eq_left.mpr
77   intro z hz
78   have hzK : z ∈ (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ)) := by
79     rw [G_at_neg k n N m d hd]
80     exact hz
81   change (canonicalFilteredTwoTerm k n N d hd).f z ∈ (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ) + 0)
82   norm_num
83   apply (canonicalFilteredTwoTerm k n N d hd).map_le (-(m : ℤ))
84   exact ⟨z, hzK, rfl⟩

86 theorem zeroPage_target_boundaries_eq_orderStrictLowerPiece
87     (n N m : ℕ) (d : PresentedWeyl k (n + 1))
88     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
89     (canonicalFilteredTwoTerm k n N d hd).boundaries 0 (-(m : ℤ)) =
90       quotientOrderStrictLowerPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
91   rw [FilteredTwoTerm.boundaries, G_succ_neg k n N m d hd,

116 /-- The source component identification, with its actual `k`-linear structure. -/
117 def zeroPageSourceLinearEquivOrderGradedPiece
118     (n N m : ℕ) (d : PresentedWeyl k (n + 1))
119     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
120     (canonicalFilteredTwoTerm k n N d hd).SourcePage 0 (-(m : ℤ)) ≃ₗ[k]
121       QuotientOrderGradedPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
122   let hC := zeroPage_source_cycles_eq_orderPiece k n N m d hd
123   let hL := G_succ_neg k n N m d hd
124   let e := LinearEquiv.ofEq _ _ hC
125   exact Submodule.Quotient.equiv _ _ e (by
126     ext x
127     simp [e, hL])

142 theorem zeroPage_source_subsingleton_of_pos
143     (n N : ℕ) (p : ℤ) (d : PresentedWeyl k (n + 1))
144     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) (hp : 0 < p) :
145     Subsingleton ((canonicalFilteredTwoTerm k n N d hd).SourcePage 0 p) := by
146   have hG : (canonicalFilteredTwoTerm k n N d hd).G p = ⊥ :=
147     canonicalOrderFiltration_eq_bot_of_pos k _ hp
148   have hcycles : (canonicalFilteredTwoTerm k n N d hd).cycles 0 p = ⊥ := by
149     apply le_antisymm
150     · exact le_trans inf_le_left (le_of_eq hG)
151     · exact bot_le
152   rw [show (canonicalFilteredTwoTerm k n N d hd).SourcePage 0 p =
153       ((canonicalFilteredTwoTerm k n N d hd).cycles 0 p ⧸
154         ((canonicalFilteredTwoTerm k n N d hd).G (p + 1)).comap
155           ((canonicalFilteredTwoTerm k n N d hd).cycles 0 p).subtype) by rfl, hcycles]
156   infer_instance
