39 theorem G_at_neg
40     (n N m : ℕ) (d : WeylAlg k (n + 1))
41     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
42     (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ)) =
43       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
44   change (if -(m : ℤ) ≤ 0 then
45       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) (- - (m : ℤ)).toNat else ⊥) = _
46   rw [if_pos (by omega)]
47   have hidx : (- - (m : ℤ)).toNat = m := by omega
48   rw [hidx]

50 theorem G_succ_neg
51     (n N m : ℕ) (d : WeylAlg k (n + 1))
52     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
53     (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ) + 1) =
54       quotientOrderStrictLowerPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
55   by_cases hm : m = 0
56   · subst m
57     norm_num
58     change (if (1 : ℤ) ≤ 0 then
59       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) 1 else
60       (⊥ : Submodule k (CanonicalQuotient k n N d))) = _
61     rw [if_neg (by norm_num)]
62     rw [quotientOrderStrictLowerPiece]
63     change (⊥ : Submodule k (CanonicalQuotient k n N d)) =
64       (presentedStrictLowerPiece k orderWeight 0).map _
65     rw [show presentedStrictLowerPiece k orderWeight 0 = ⊥ by rfl]
66     simp
67   · obtain ⟨m', rfl⟩ := Nat.exists_eq_succ_of_ne_zero hm
68     rw [show (-(↑(m' + 1) : ℤ) + 1) = -(m' : ℤ) by omega,
69       G_at_neg]
70     rw [quotientOrderStrictLowerPiece_succ_eq]

72 theorem zeroPage_source_cycles_eq_orderPiece
73     (n N m : ℕ) (d : WeylAlg k (n + 1))
74     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
75     (canonicalFilteredTwoTerm k n N d hd).cycles 0 (-(m : ℤ)) =
76       quotientOrderPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
77   rw [FilteredTwoTerm.cycles, G_at_neg k n N m d hd]
78   apply inf_eq_left.mpr
79   intro z hz
80   have hzK : z ∈ (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ)) := by
81     rw [G_at_neg k n N m d hd]
82     exact hz
83   change (canonicalFilteredTwoTerm k n N d hd).f z ∈ (canonicalFilteredTwoTerm k n N d hd).G (-(m : ℤ) + 0)
84   norm_num
85   apply (canonicalFilteredTwoTerm k n N d hd).map_le (-(m : ℤ))
86   exact ⟨z, hzK, rfl⟩

88 theorem zeroPage_target_boundaries_eq_orderStrictLowerPiece
89     (n N m : ℕ) (d : WeylAlg k (n + 1))
90     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
91     (canonicalFilteredTwoTerm k n N d hd).boundaries 0 (-(m : ℤ)) =
92       quotientOrderStrictLowerPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
93   rw [FilteredTwoTerm.boundaries, G_succ_neg k n N m d hd,
94     G_at_neg k n N m d hd]

98   exact (canonicalFilteredTwoTerm k n N d hd).map_le (-(m : ℤ) + 1)

118 /-- The source component identification, with its actual `k`-linear structure. -/
119 def zeroPageSourceLinearEquivOrderGradedPiece
120     (n N m : ℕ) (d : WeylAlg k (n + 1))
121     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
122     (canonicalFilteredTwoTerm k n N d hd).SourcePage 0 (-(m : ℤ)) ≃ₗ[k]
123       QuotientOrderGradedPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
124   let hC := zeroPage_source_cycles_eq_orderPiece k n N m d hd
125   let hL := G_succ_neg k n N m d hd
126   let e := LinearEquiv.ofEq _ _ hC
127   exact Submodule.Quotient.equiv _ _ e (by
128     ext x
129     simp [e, hL])

144 theorem zeroPage_source_subsingleton_of_pos
145     (n N : ℕ) (p : ℤ) (d : WeylAlg k (n + 1))
146     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) (hp : 0 < p) :
147     Subsingleton ((canonicalFilteredTwoTerm k n N d hd).SourcePage 0 p) := by
148   have hG : (canonicalFilteredTwoTerm k n N d hd).G p = ⊥ :=
149     canonicalOrderFiltration_eq_bot_of_pos k _ hp
150   have hcycles : (canonicalFilteredTwoTerm k n N d hd).cycles 0 p = ⊥ := by
151     apply le_antisymm
152     · exact le_trans inf_le_left (le_of_eq hG)
153     · exact bot_le
154   rw [show (canonicalFilteredTwoTerm k n N d hd).SourcePage 0 p =
155       ((canonicalFilteredTwoTerm k n N d hd).cycles 0 p ⧸
156         ((canonicalFilteredTwoTerm k n N d hd).G (p + 1)).comap
157           ((canonicalFilteredTwoTerm k n N d hd).cycles 0 p).subtype) by rfl, hcycles]
158   infer_instance
