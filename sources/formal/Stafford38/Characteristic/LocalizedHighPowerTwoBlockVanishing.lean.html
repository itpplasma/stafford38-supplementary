7 /-!
8 # Localized high-power two-block vanishing
9 
10 This file discharges the elementwise two-block condition in the concrete
11 right-Rees Artinian adapter.  A localized scalar whose specialization lies in
12 an annihilating ideal sends every deformation vector into the parameter
13 image.  Two such scalar actions therefore produce two parameter factors and
14 vanish by square-zero.
15 
16 The opposite localization acts on the left in the order corresponding to
17 written right multiplication.  The finite sum below consequently retains the
18 displayed product order.  No residue-field action is placed on the deformation
19 module.
20 -/

46 /-- The central deformation parameter remains central after Ore localization.
47 The multiplication is in the opposite ring, hence represents written right
48 multiplication in the original deformation ring. -/
49 theorem localizedParameter_comm
50     (a : OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ)) :
51     OreLocalization.numeratorRingHom (MulOpposite.op D.c) * a =
52       a * OreLocalization.numeratorRingHom (MulOpposite.op D.c) := by
53   induction a using OreLocalization.ind with
54   | _ b s =>
55       rw [OreLocalization.numeratorRingHom_apply]
56       rw [OreLocalization.oreDiv_mul_char
57         (MulOpposite.op D.c) b 1 s (MulOpposite.op D.c) s]
58       · rw [OreLocalization.mul_div_one]
59         congr 1
60         · apply MulOpposite.unop_injective
61           exact (D.c_center.comm b.unop).symm
62         · simp
63       · apply MulOpposite.unop_injective
64         exact D.c_center.comm s.val.unop

66 /-- A localized scalar action commutes with the concrete parameter action.
67 This is the bridge that permits the two high-power blocks to contribute two
68 successive parameter factors. -/
69 theorem smul_localizedCAct
70     (a : OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ))
71     (z : LocalizedDeformationModule D S) :
72     a • localizedCAct D S z = localizedCAct D S (a • z) := by
73   rw [localizedCAct_apply, localizedCAct_apply]
74   rw [← OreLocalization.oreDiv_one_smul]
75   rw [← mul_smul]
76   rw [← OreLocalization.numeratorRingHom_apply (MulOpposite.op D.c)]
77   rw [← localizedParameter_comm D S a, mul_smul]
78   exact OreLocalization.oreDiv_one_smul _ _

80 /-- If the specialization of a localized scalar lies in an ideal annihilating
81 the special fibre, then its action on every deformation vector is a localized
82 parameter multiple. -/
83 theorem smul_mem_parameterRange_of_specialization_mem_annihilator
84     (a : OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ))
85     (ha : localizedOppositeSpecialization D S a ∈
86       Module.annihilator (Localization S) (LocalizedModule S G))
87     (z : LocalizedDeformationModule D S) :
88     a • z ∈ AddMonoidHom.range (localizedCAct D S) := by
89   rw [← localizedSpecialization_ker_eq_range D S]
90   rw [AddMonoidHom.mem_ker, localizedSpecialization_smul D S]
91   exact Module.mem_annihilator.mp ha (localizedSpecialization D S z)

93 /-- Two localized factors whose specializations lie in one annihilating ideal
94 act successively by zero. -/
95 theorem product_smul_eq_zero_of_specializations_mem_annihilator
96     (a b : OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ))
97     (ha : localizedOppositeSpecialization D S a ∈
98       Module.annihilator (Localization S) (LocalizedModule S G))
99     (hb : localizedOppositeSpecialization D S b ∈
100       Module.annihilator (Localization S) (LocalizedModule S G))
101     (z : LocalizedDeformationModule D S) :
102     (a * b) • z = 0 := by
103   obtain ⟨w, hw⟩ :=
104     smul_mem_parameterRange_of_specialization_mem_annihilator D S b hb z
105   obtain ⟨v, hv⟩ :=
106     smul_mem_parameterRange_of_specialization_mem_annihilator D S a ha w
107   rw [mul_smul, ← hw, smul_localizedCAct D S, ← hv]
108   exact localizedCAct_sq_eq_zero D S v

110 /-- An annihilating maximal-ideal power supplies the exact two-block
111 vanishing proposition required by `RightReesArtinianAdapter`. -/
112 theorem localizedHighPowerTwoBlockVanishing_of_annihilatingPower
113     [IsLocalRing (Localization S)]
114     (q : ℕ)
115     (hpow : IsLocalRing.maximalIdeal (Localization S) ^ q ≤
116       Module.annihilator (Localization S) (LocalizedModule S G)) :
117     LocalizedHighPowerTwoBlockVanishing D S q := by
118   intro iota s a b ha hb z
119   rw [Finset.sum_smul]
120   apply Finset.sum_eq_zero
121   intro i hi
122   exact product_smul_eq_zero_of_specializations_mem_annihilator D S
123     (a i) (b i) (hpow (ha i hi)) (hpow (hb i hi)) z
