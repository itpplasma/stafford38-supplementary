8 # Involutivity descends from a radical to its minimal primes
9 
10 Gabber's theorem is stated for the *radical* of the annihilator of an
11 associated graded module: `{√J, √J} ⊆ √J`.  The project interfaces, and the
12 coisotropic consumers downstream of them, ask instead for involutivity of each
13 individual minimal prime of `J`.  This file supplies the elementary
14 commutative-algebra step between the two, so that no Gabber-shaped statement
15 has to be strengthened by hand.
16 
17 Two facts are proved.
18 
19 * `exists_notMem_mul_mem_radical`: if `P` is minimal over `J` and `f ∈ P`,
20   then some `s ∉ P` satisfies `s * f ∈ √J`.  This is the saturation statement
21   that replaces localizing at `P`; it is proved directly from the maximally
22   disjoint prime construction applied to the multiplicative set generated by
23   `P`'s complement together with `f`, so no Noetherian hypothesis and no
24   finiteness of the minimal-prime set is used.
25 
26 * `minimalPrimes_isInvolutive_of_radical_isInvolutive`: if `√J` is involutive
27   then every minimal prime of `J` is involutive.  With `u = s * t` produced by
28   the previous lemma for `f` and `g`, the Leibniz rule gives
29 
30   ```text
31   {u f, u g} = u^2 {f, g} + u g {f, u} + u f {u, g},
32   ```
33 
34   whose left-hand side lies in `√J ⊆ P` and whose last two summands lie in `P`
35   because `f, g ∈ P`.  Hence `u^2 {f, g} ∈ P`, and `u ∉ P` with `P` prime
36   gives `{f, g} ∈ P`.
37 
38 Nothing here proves involutivity of any concrete radical; this is the
39 descent step only.
40 -/

53 /-- Saturation at a minimal prime.  If `P` is minimal over `J` and `f ∈ P`,
54 some element outside `P` carries `f` into the radical of `J`.  This is the
55 localization-free form of `√(J R_P) = P R_P`. -/
56 theorem exists_notMem_mul_mem_radical
57     {J P : Ideal R} (hP : P ∈ J.minimalPrimes) {f : R} (hf : f ∈ P) :
58     ∃ s, s ∉ P ∧ s * f ∈ J.radical := by
59   classical
60   have hprime : P.IsPrime := hP.1.1
61   by_contra hcon
62   have hcon' : ∀ s : R, s ∉ P → s * f ∉ J.radical := fun s hs hmem =>
63     hcon ⟨s, hs, hmem⟩
64   set S : Submonoid R := P.primeCompl ⊔ Submonoid.powers f with hS
65   have hdisj : Disjoint (J : Set R) (S : Set R) := by
66     rw [Set.disjoint_left]
67     rintro x hxJ hxS
68     rw [SetLike.mem_coe, hS, Submonoid.mem_sup] at hxS
69     obtain ⟨s, hs, z, hz, rfl⟩ := hxS
70     obtain ⟨m, rfl⟩ := Submonoid.mem_powers_iff z f |>.mp hz
71     have hsP : s ∉ P := hs
72     cases m with
73     | zero =>
74         rw [pow_zero, mul_one] at hxJ
75         exact hsP (hP.1.2 hxJ)
76     | succ m =>
77         refine hcon' s hsP ⟨m + 1, ?_⟩
78         have hrw : (s * f) ^ (m + 1) = s ^ m * (s * f ^ (m + 1)) := by ring
79         rw [hrw]
80         exact Ideal.mul_mem_left _ _ hxJ
81   obtain ⟨Q, hQprime, hJQ, hQdisj⟩ := Ideal.exists_le_prime_disjoint J S hdisj
82   have hQP : Q ≤ P := by
83     intro x hx
84     by_contra hxP
85     exact Set.disjoint_left.mp hQdisj hx
86       (SetLike.mem_coe.mpr (le_sup_left (a := P.primeCompl) (b := Submonoid.powers f) hxP))
87   have hPQ : P ≤ Q := hP.2 ⟨hQprime, hJQ⟩ hQP
88   exact Set.disjoint_left.mp hQdisj (hPQ hf)
89     (SetLike.mem_coe.mpr
90       (le_sup_right (a := P.primeCompl) (b := Submonoid.powers f) (Submonoid.mem_powers f)))
91 
92 end Saturation

98 /-- The Leibniz expansion used by the descent, with `{u, u} = 0` already
99 cancelled. -/
100 theorem sq_mul_poissonBracket_eq
101     (u f g : SymbolRing k n) :
102     u * u * poissonBracket f g =
103       poissonBracket (u * f) (u * g)
104         - u * (g * poissonBracket f u) - f * (u * poissonBracket u g) := by
105   rw [poissonBracket_mul_left, poissonBracket_mul_right,
106     poissonBracket_mul_right u u g, poissonBracket_self]
107   ring
108 
109 /-- Gabber's radical statement implies the minimal-prime statement the
110 project's coisotropic consumers require. -/
111 theorem minimalPrimes_isInvolutive_of_radical_isInvolutive
112     (J : Ideal (SymbolRing k n)) (h : IsInvolutive J.radical) :
113     ∀ P ∈ J.minimalPrimes, IsInvolutive P := by
114   intro P hP f hf g hg
115   have hprime : P.IsPrime := hP.1.1
116   have hrad : J.radical ≤ P := hprime.radical_le_iff.mpr hP.1.2
117   obtain ⟨s, hs, hsf⟩ := exists_notMem_mul_mem_radical hP hf
118   obtain ⟨t, ht, htg⟩ := exists_notMem_mul_mem_radical hP hg
119   set u : SymbolRing k n := s * t with hu
120   have huf : u * f ∈ J.radical := by
121     have hrw : u * f = t * (s * f) := by rw [hu]; ring
122     rw [hrw]
123     exact Ideal.mul_mem_left _ _ hsf
124   have hug : u * g ∈ J.radical := by
125     have hrw : u * g = s * (t * g) := by rw [hu]; ring
126     rw [hrw]
127     exact Ideal.mul_mem_left _ _ htg
128   have hbr : poissonBracket (u * f) (u * g) ∈ J.radical := h _ huf _ hug
129   have hmem : u * u * poissonBracket f g ∈ P := by
130     rw [sq_mul_poissonBracket_eq]
131     exact sub_mem
132       (sub_mem (hrad hbr) (Ideal.mul_mem_left _ _ (Ideal.mul_mem_right _ _ hg)))
133       (Ideal.mul_mem_right _ _ hf)
134   have huP : u ∉ P := fun hmemu => (hprime.mem_or_mem hmemu).elim hs ht
135   rcases hprime.mem_or_mem hmem with hsq | hbracket
136   · exact absurd ((hprime.mem_or_mem hsq).elim id id) huP
137   · exact hbracket
138 
139 end Involutivity
