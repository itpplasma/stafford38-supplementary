8 /-!
9 # High-power reduction for a square-zero deformation
10 
11 This file isolates the finite-length reduction used in Singh--Kumar's local
12 argument.  A finite-length module over a commutative local ring is killed by
13 a power of the maximal ideal.  In an exact square-zero deformation, two
14 blocks lying over that power act successively by zero.  Finally, if such a
15 two-block sum is also a multiple of the deformation parameter, exactness
16 forces its parameter cofactor to specialize into the maximal ideal.
17 
18 The last statement is the elementwise form of the parameter-ideal
19 intersection used in the high-power quotient argument.  It avoids introducing
20 an artificial multiplication API for two-sided ideals: membership in the
21 relevant product is recorded by an explicit finite sum of products.
22 -/

43 /-- A finite-length module over a commutative local ring is annihilated by a
44 power of the maximal ideal. -/
45 theorem exists_maximalIdeal_pow_le_annihilator_of_finiteLength
46     (hG : IsFiniteLength R G) :
47     ∃ n : ℕ, 𝔪 ^ n ≤ Module.annihilator R G := by
48   induction hG with
49   | of_subsingleton =>
50       refine ⟨0, ?_⟩
51       intro r _hr
52       rw [Module.mem_annihilator]
53       intro g
54       exact Subsingleton.elim _ _
55   | @of_simple_quotient M _ _ S _ _ ih =>
56       obtain ⟨n, hn⟩ := ih
57       refine ⟨n + 1, ?_⟩
58       have hmS : 𝔪 • (⊤ : Submodule R M) ≤ S := by
59         rw [Submodule.smul_le]
60         intro r hr g _hg
61         rw [← Submodule.Quotient.mk_eq_zero S]
62         have hmax : IsLocalRing.maximalIdeal R =
63             Module.annihilator R (M ⧸ S) :=
64           (IsLocalRing.eq_maximalIdeal
65             (IsSimpleModule.annihilator_isMaximal
66               (R := R) (M := M ⧸ S))).symm
67         have hrann : r ∈ Module.annihilator R (M ⧸ S) := hmax ▸ hr
68         exact Module.mem_annihilator.mp hrann (Submodule.Quotient.mk g)
69       have hpowS : (𝔪 ^ n) • S = ⊥ := by
70         apply le_antisymm
71         · calc
72             (𝔪 ^ n) • S ≤
73                 Module.annihilator R S • S :=
74               Submodule.smul_mono hn le_rfl
75             _ = ⊥ := Submodule.annihilator_smul S
76         · exact bot_le
77       have hpowM : (𝔪 ^ (n + 1)) • (⊤ : Submodule R M) = ⊥ := by
78         apply le_antisymm
79         · rw [pow_succ, mul_smul]
80           calc
81             (𝔪 ^ n) • (𝔪 • (⊤ : Submodule R M)) ≤
82                 (𝔪 ^ n) • S := Submodule.smul_mono le_rfl hmS
83             _ = ⊥ := hpowS
84         · exact bot_le
85       intro r hr
86       rw [Module.mem_annihilator]
87       intro g
88       have hmem : r • g ∈
89           (𝔪 ^ (n + 1)) • (⊤ : Submodule R M) :=
90         Submodule.smul_mem_smul hr trivial
91       have : r • g ∈ (⊥ : Submodule R M) := by
92         rw [← hpowM]
93         exact hmem
94       simpa using this

96 /-- If a parameter multiple acts trivially on the deformation module, then
97 its cofactor specializes into the maximal ideal.  This is the exactness and
98 nonzero-special-fibre step in the parameter-ideal intersection argument. -/
99 theorem parameterFactor_mem_maximalIdeal_of_rightAction_eq_zero
100     [Nontrivial G]
101     (D : RightSquareZeroTraceData k R B N G)
102     {z : B}
103     (hz : ∀ w : N, MulOpposite.op (D.c * z) • w = 0) :
104     D.pi z ∈ 𝔪 := by
105   have hpiAnn : D.pi z ∈ Module.annihilator R G := by
106     rw [Module.mem_annihilator]
107     intro g
108     obtain ⟨w, rfl⟩ := D.rho_surjective g
109     have hcAct : D.cAct (MulOpposite.op z • w) = 0 := by
110       rw [D.cAct_apply, ← mul_smul]
111       change MulOpposite.op (z * D.c) • w = 0
112       rw [← D.c_center.comm z]
113       exact hz w
114     have hrange : MulOpposite.op z • w ∈ LinearMap.range D.cAct := by
115       rw [← D.c_exact, LinearMap.mem_ker]
116       exact hcAct
117     have hrho : D.rho (MulOpposite.op z • w) = 0 := by
118       rw [← D.rho_ker] at hrange
119       exact LinearMap.mem_ker.mp hrange
120     rw [D.rho_action] at hrho
121     exact hrho
122   by_contra hnot
123   have hunit : IsUnit (D.pi z) := IsLocalRing.notMem_maximalIdeal.mp hnot
124   obtain ⟨g, hg⟩ := exists_ne (0 : G)
125   have hzero := Module.mem_annihilator.mp hpiAnn g
126   exact hg (hunit.smul_eq_zero.mp hzero)

128 /-- Elementwise parameter-ideal intersection.  A finite sum of products of
129 two blocks over an annihilating power acts by zero; if that sum is `c*z`, then
130 the parameter cofactor `z` lies over the maximal ideal. -/
131 theorem parameterFactor_mem_maximalIdeal_of_eq_sum_products
132     [Nontrivial G]
133     (D : RightSquareZeroTraceData k R B N G)
134     (m : Ideal R) (n : ℕ)
135     {z : B} {iota : Type*} (s : Finset iota)
136     (hpow : m ^ n ≤ Module.annihilator R G)
137     (a b : iota → B)
138     (ha : ∀ i ∈ s, D.pi (a i) ∈ m ^ n)
139     (hb : ∀ i ∈ s, D.pi (b i) ∈ m ^ n)
140     (heq : D.c * z = ∑ i ∈ s, a i * b i) :
141     D.pi z ∈ 𝔪 := by
142   apply parameterFactor_mem_maximalIdeal_of_rightAction_eq_zero D
143   intro w
144   rw [heq]
145   exact sum_products_rightAction_eq_zero_of_pi_mem_pow
146     D m n s hpow a b ha hb w

148 /-- The exact high-power descent step.  If a parameter multiple is congruent,
149 modulo a two-block sum, to a parameter multiple whose cofactor is already over
150 the maximal ideal, then its own cofactor is over the maximal ideal. -/
151 theorem parameterFactor_mem_maximalIdeal_of_eq_parameter_add_sum_products
152     [Nontrivial G]
153     (D : RightSquareZeroTraceData k R B N G)
154     (m : Ideal R) (n : ℕ)
155     {z w : B} {iota : Type*} (s : Finset iota)
156     (hpow : m ^ n ≤ Module.annihilator R G)
157     (hw : D.pi w ∈ 𝔪)
158     (a b : iota → B)
159     (ha : ∀ i ∈ s, D.pi (a i) ∈ m ^ n)
160     (hb : ∀ i ∈ s, D.pi (b i) ∈ m ^ n)
161     (heq : D.c * z = D.c * w + ∑ i ∈ s, a i * b i) :
162     D.pi z ∈ 𝔪 := by
163   have hsep : D.pi (z - w) ∈ 𝔪 := by
164     apply parameterFactor_mem_maximalIdeal_of_eq_sum_products
165       D m n s hpow a b ha hb
166     calc
167       D.c * (z - w) = D.c * z - D.c * w := mul_sub _ _ _
168       _ = ∑ i ∈ s, a i * b i := by rw [heq]; abel
169   have hsum : D.pi (z - w) + D.pi w ∈ 𝔪 :=
170     (IsLocalRing.maximalIdeal R).add_mem hsep hw
171   simpa using hsum

173 /-- Finite length supplies one exponent for both the two-block vanishing and
174 the parameter-ideal separation theorem. -/
175 theorem exists_highPower_parameterIdeal_reduction
176     [Nontrivial G]
177     (D : RightSquareZeroTraceData k R B N G)
178     (hG : IsFiniteLength R G) :
179     ∃ n : ℕ,
180       𝔪 ^ n ≤ Module.annihilator R G ∧
181       ∀ {z w : B} {iota : Type*} (s : Finset iota),
182         D.pi w ∈ 𝔪 →
183         ∀ (a b : iota → B),
184           (∀ i ∈ s, D.pi (a i) ∈ 𝔪 ^ n) →
185           (∀ i ∈ s, D.pi (b i) ∈ 𝔪 ^ n) →
186           D.c * z = D.c * w + ∑ i ∈ s, a i * b i →
187           D.pi z ∈ 𝔪 := by
188   obtain ⟨n, hpow⟩ :=
189     exists_maximalIdeal_pow_le_annihilator_of_finiteLength hG
190   refine ⟨n, hpow, ?_⟩
191   intro z w iota s hw a b ha hb heq
192   exact parameterFactor_mem_maximalIdeal_of_eq_parameter_add_sum_products
193     D 𝔪 n s hpow hw a b ha hb heq
