8 /-!
9 # Linear substitutions on polynomial phase space
10 
11 A square matrix acts on the phase-space polynomial ring by substituting its
12 row-linear combination for each variable. Composition is contravariant in the
13 matrix order, matching the Weyl-generator substitution convention. Explicit
14 mutual inverse certificates produce an algebra equivalence.
15 -/

27 def symbolLinearCombination {n : ℕ}
28     (M : Matrix (PhaseVar n) (PhaseVar n) k) (i : PhaseVar n) :
29     SymbolRing k n :=
30   ∑ j, MvPolynomial.C (M i j) * MvPolynomial.X j
31 
32 def symbolLinearAlgHom {n : ℕ}
33     (M : Matrix (PhaseVar n) (PhaseVar n) k) :
34     SymbolRing k n →ₐ[k] SymbolRing k n :=
35   MvPolynomial.aeval (symbolLinearCombination k M)
36 
37 @[simp] theorem symbolLinearAlgHom_X {n : ℕ}
38     (M : Matrix (PhaseVar n) (PhaseVar n) k) (i : PhaseVar n) :
39     symbolLinearAlgHom k M (MvPolynomial.X i) =
40       symbolLinearCombination k M i := by
41   rw [symbolLinearAlgHom, MvPolynomial.aeval_X]
42 
43 @[simp] theorem symbolLinearAlgHom_C {n : ℕ}
44     (M : Matrix (PhaseVar n) (PhaseVar n) k) (c : k) :
45     symbolLinearAlgHom k M (MvPolynomial.C c) = MvPolynomial.C c := by
46   rw [symbolLinearAlgHom, MvPolynomial.aeval_C]
47   rfl
48 
49 theorem symbolLinearAlgHom_comp {n : ℕ}
50     (M N : Matrix (PhaseVar n) (PhaseVar n) k) :
51     (symbolLinearAlgHom k M).comp (symbolLinearAlgHom k N) =
52       symbolLinearAlgHom k (N * M) := by
53   apply MvPolynomial.algHom_ext
54   intro i
55   rw [AlgHom.comp_apply, symbolLinearAlgHom_X, symbolLinearAlgHom_X]
56   simp only [symbolLinearCombination, map_sum, map_mul,
57     symbolLinearAlgHom_C, symbolLinearAlgHom_X, Matrix.mul_apply]
58   simp only [Finset.mul_sum, Finset.sum_mul, mul_assoc]
59   rw [Finset.sum_comm]
60 
61 theorem symbolLinearAlgHom_one {n : ℕ} :
62     symbolLinearAlgHom k (1 : Matrix (PhaseVar n) (PhaseVar n) k) =
63       AlgHom.id k (SymbolRing k n) := by
64   apply MvPolynomial.algHom_ext
65   intro i
66   simp [symbolLinearCombination, Matrix.one_apply]

68 /-- The polynomial substitution equivalence attached to explicitly certified
69 inverse matrices. -/
70 def symbolLinearAlgEquivOfInverse {n : ℕ}
71     (M N : Matrix (PhaseVar n) (PhaseVar n) k)
72     (hMN : M * N = 1) (hNM : N * M = 1) :
73     SymbolRing k n ≃ₐ[k] SymbolRing k n :=
74   AlgEquiv.ofAlgHom (symbolLinearAlgHom k M) (symbolLinearAlgHom k N)
75     (by rw [symbolLinearAlgHom_comp, hNM, symbolLinearAlgHom_one])
76     (by rw [symbolLinearAlgHom_comp, hMN, symbolLinearAlgHom_one])
77 
78 @[simp] theorem symbolLinearAlgEquivOfInverse_X {n : ℕ}
79     (M N : Matrix (PhaseVar n) (PhaseVar n) k)
80     (hMN : M * N = 1) (hNM : N * M = 1) (i : PhaseVar n) :
81     symbolLinearAlgEquivOfInverse k M N hMN hNM (MvPolynomial.X i) =
82       symbolLinearCombination k M i :=
83   symbolLinearAlgHom_X k M i

85 @[simp] theorem symbolLinearAlgEquivOfInverse_symm_X {n : ℕ}

88     (symbolLinearAlgEquivOfInverse k M N hMN hNM).symm
89         (MvPolynomial.X i) = symbolLinearCombination k N i :=
