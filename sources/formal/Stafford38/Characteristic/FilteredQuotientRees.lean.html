11 # The filtered quotient direct sum and associated-graded map
12 
13 The filtered quotient pieces assemble into a direct sum with a successor
14 shift. Quotienting each degree by the preceding piece gives the already
15 constructed associated graded module; here we define the global map, prove
16 that it kills the shift, and prove surjectivity. This file does not construct
17 the required right action of the order-Rees ring. Kernel equality and that
18 action are deliberately left as subsequent theorems.
19 -/

40 theorem quotientOrderPiece_mono
41     (I : RightIdeal (WeylAlg k n)) {N M : ℕ} (hNM : N ≤ M) :
42     quotientOrderPiece k I N ≤ quotientOrderPiece k I M := by
43   exact Submodule.map_mono (presentedWeightPiece_mono k orderWeight hNM)

45 theorem quotientOrderStrictLowerPiece_succ_eq
46     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
47     quotientOrderStrictLowerPiece k I (N + 1) =
48       quotientOrderPiece k I N := by
49   rfl

51 /-- The filtered direct sum underlying the intended right Rees module. -/
52 abbrev QuotientOrderReesModule
53     (I : RightIdeal (WeylAlg k n)) :=
54   DirectSum ℕ (fun N => quotientOrderPiece k I N)

56 /-- Inclusion of one quotient filtration piece into the next. -/
57 def quotientOrderPieceSucc
58     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
59     quotientOrderPiece k I N →ₗ[k] quotientOrderPiece k I (N + 1) :=
60   Submodule.inclusion (quotientOrderPiece_mono k I (Nat.le_succ N))
61 
62 /-- The successor inclusion shift on the filtered quotient direct sum. -/
63 def quotientOrderReesShift
64     (I : RightIdeal (WeylAlg k n)) :
65     QuotientOrderReesModule k I →ₗ[k] QuotientOrderReesModule k I :=
66   DirectSum.toModule k ℕ (QuotientOrderReesModule k I) fun N =>
67     (DirectSum.lof k ℕ (fun M => quotientOrderPiece k I M) (N + 1)).comp
68       (quotientOrderPieceSucc k I N)

70 @[simp] theorem quotientOrderReesShift_of
71     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
72     (q : quotientOrderPiece k I N) :
73     quotientOrderReesShift k I
74         (DirectSum.of (fun M => quotientOrderPiece k I M) N q) =
75       DirectSum.of (fun M => quotientOrderPiece k I M) (N + 1)
76         (quotientOrderPieceSucc k I N q) := by
77   change (DirectSum.toModule k ℕ (QuotientOrderReesModule k I) fun N =>
78       (DirectSum.lof k ℕ (fun M => quotientOrderPiece k I M) (N + 1)).comp
79         (quotientOrderPieceSucc k I N))
80       (DirectSum.lof k ℕ (fun M => quotientOrderPiece k I M) N q) = _
81   rw [DirectSum.toModule_lof, LinearMap.comp_apply, DirectSum.lof_eq_of]

83 /-- The quotient map from one filtered piece to its associated graded piece. -/
84 def quotientOrderPieceToGraded
85     (I : RightIdeal (WeylAlg k n)) (N : ℕ) :
86     quotientOrderPiece k I N →ₗ[k] QuotientOrderGradedPiece k I N :=
87   ((quotientOrderStrictLowerPiece k I N).comap
88     (quotientOrderPiece k I N).subtype).mkQ

90 /-- Global specialization of the filtered quotient direct sum to the actual
91 associated graded quotient. -/
92 def quotientOrderReesToAssociatedGraded
93     (I : RightIdeal (WeylAlg k n)) :
94     QuotientOrderReesModule k I →ₗ[k] QuotientOrderAssociatedGraded k I :=
95   DirectSum.toModule k ℕ (QuotientOrderAssociatedGraded k I) fun N =>
96     (DirectSum.lof k ℕ (fun M => QuotientOrderGradedPiece k I M) N).comp
97       (quotientOrderPieceToGraded k I N)

102     quotientOrderReesToAssociatedGraded k I

112 theorem quotientOrderPieceToGraded_succ_inclusion_eq_zero
113     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
114     (q : quotientOrderPiece k I N) :
115     quotientOrderPieceToGraded k I (N + 1)
116         (quotientOrderPieceSucc k I N q) = 0 := by
117   change Submodule.Quotient.mk (quotientOrderPieceSucc k I N q) = 0
118   rw [Submodule.Quotient.mk_eq_zero]
119   change (q : FilteredRightQuotient k I) ∈
120     quotientOrderStrictLowerPiece k I (N + 1)
121   rw [quotientOrderStrictLowerPiece_succ_eq]
122   exact q.property

124 /-- Specialization kills the successor shift. -/
125 theorem quotientOrderReesToAssociatedGraded_comp_shift
126     (I : RightIdeal (WeylAlg k n)) :
127     (quotientOrderReesToAssociatedGraded k I).comp
128         (quotientOrderReesShift k I) = 0 := by
129   apply DirectSum.linearMap_ext
130   intro N
131   apply LinearMap.ext
132   intro q
133   change quotientOrderReesToAssociatedGraded k I
134       (quotientOrderReesShift k I
135         (DirectSum.of (fun M => quotientOrderPiece k I M) N q)) = 0
136   rw [quotientOrderReesShift_of, quotientOrderReesToAssociatedGraded_of,
137     quotientOrderPieceToGraded_succ_inclusion_eq_zero]
138   exact map_zero _

140 /-- Every associated-graded class has a filtered Rees representative. -/
141 theorem quotientOrderReesToAssociatedGraded_surjective
142     (I : RightIdeal (WeylAlg k n)) :
143     Function.Surjective (quotientOrderReesToAssociatedGraded k I) := by
144   intro g
145   induction g using DirectSum.induction_on with
146   | zero =>
147       exact ⟨0, map_zero _⟩
148   | of N q =>
149       obtain ⟨p, rfl⟩ := Submodule.Quotient.mk_surjective
150         ((quotientOrderStrictLowerPiece k I N).comap
151           (quotientOrderPiece k I N).subtype) q
152       exact ⟨DirectSum.of (fun M => quotientOrderPiece k I M) N p,
153         quotientOrderReesToAssociatedGraded_of k I N p⟩
154   | add x y hx hy =>
155       obtain ⟨a, ha⟩ := hx
156       obtain ⟨b, hb⟩ := hy
157       refine ⟨a + b, ?_⟩
158       calc
159         quotientOrderReesToAssociatedGraded k I (a + b) =
160             quotientOrderReesToAssociatedGraded k I a +
161               quotientOrderReesToAssociatedGraded k I b := map_add _ _ _
162         _ = x + y := congrArg₂ (· + ·) ha hb
