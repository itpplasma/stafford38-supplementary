9 # A unit-only strict coordinate criterion for canonical axis avoidance
10 
11 The terminal axis-avoidance argument does not require coordinate cancellation
12 on every associated-graded piece.  It only needs the degree-zero class of `1`
13 to lie in the image of the coordinate action.  Equivalently, it is enough to
14 have one representative of order zero whose coordinate multiple equals `1`
15 in the literal canonical quotient.
16 
17 This file proves that this single strict unit preimage puts a relation
18 `P * X - 1` in the order initial ideal and hence excludes the distinguished
19 coordinate hyperplane from characteristic support.  It also records that the
20 previous all-degree cancellation and strict-lower-preimage hypotheses imply
21 the unit-only condition.  No converse is asserted.
22 -/

52 /-- The weakest concrete filtered input used by the terminal axis argument:
53 the quotient class of `1` has one coordinate predecessor of order zero. -/
54 def StrictUnitCoordinatePreimage (n N : ℕ)
55     (d : WeylAlg k (n + 1)) : Prop :=
56   ∃ y : WeylAlg k (n + 1),
57     y ∈ orderPiece k (n + 1) 0 ∧
58       qmk (CanonicalIdeal k n N d) (y * presentedCoordinate k n) =
59         qmk (CanonicalIdeal k n N d) 1
60 
61 /-- The previous all-degree cancellation hypothesis implies the unit-only
62 criterion, by strictifying an unrestricted preimage only at degree zero. -/
63 theorem strictUnitCoordinatePreimage_of_coordinateCancellation
64     (n N : ℕ) {d : WeylAlg k (n + 1)}
65     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
66     (hcancel : CoordinateCancellation k n N d) :
67     StrictUnitCoordinatePreimage k n N d := by
68   obtain ⟨y, hy, hyx⟩ := exists_strict_coordinate_preimage
69     k n N 0 hd hcancel 1 (orderPieceOne (n := n + 1) k).property
70   exact ⟨y, hy, hyx⟩

73 /-- Uniform strict-lower coordinate division is also stronger than necessary:
74 only its instance at cutoff `1` and target `1` is used here. -/
75 theorem strictUnitCoordinatePreimage_of_strictLowerCoordinatePreimages
76     (n N : ℕ) {d : WeylAlg k (n + 1)}
77     (hstrict : StrictLowerCoordinatePreimages k n N d) :
78     StrictUnitCoordinatePreimage k n N d := by
79   have hone : (1 : WeylAlg k (n + 1)) ∈
80       presentedStrictLowerPiece k orderWeight 1 := by
81     change (1 : WeylAlg k (n + 1)) ∈ orderPiece k (n + 1) 0
82     exact (orderPieceOne (n := n + 1) k).property
83   obtain ⟨y, hy, hyx⟩ := hstrict 1 1 hone
84   have hy0 : y ∈ orderPiece k (n + 1) 0 := by
85     change y ∈ orderPiece k (n + 1) 0 at hy
86     exact hy
87   exact ⟨y, hy0, hyx⟩

89 /-- A single strict unit preimage gives the exact degree-zero initial relation
90 needed for scheme-level coordinate-axis avoidance. -/
91 theorem canonical_orderInitialIdeal_sup_coordinate_eq_top_of_strictUnit
92     (n N : ℕ) {d : WeylAlg k (n + 1)}
93     (hunit : StrictUnitCoordinatePreimage k n N d) :
94     orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
95         Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} = ⊤ := by
96   let I := CanonicalIdeal k n N d
97   let x := presentedCoordinate k n
98   obtain ⟨y, hy, hyx⟩ := hunit
99   have hx : x ∈ orderPiece k (n + 1) 0 :=
100     presentedCoordinate_mem_orderPiece_zero k n
101   have hyxOrder : y * x ∈ orderPiece k (n + 1) 0 :=
102     mul_mem_orderPiece k hy hx
103   have honeOrder : (1 : WeylAlg k (n + 1)) ∈
104       orderPiece k (n + 1) 0 :=
105     (orderPieceOne (n := n + 1) k).property
106   have hdiffOrder : y * x - 1 ∈ orderPiece k (n + 1) 0 :=
107     (orderPiece k (n + 1) 0).sub_mem hyxOrder honeOrder
108   have hdiffIdeal : y * x - 1 ∈ I :=
109     (Submodule.Quotient.eq I).mp hyx
110   have hrelation := principalComponent_mem_orderSymbolRelation
111     k I 0 (y * x - 1) hdiffOrder hdiffIdeal
112   have hinitial :
113       ((principalComponentOnPiece k (@orderWeight (n + 1)) 0
114           ⟨y * x - 1, hdiffOrder⟩ :
115         MvPolynomial.weightedHomogeneousSubmodule k
116           (@orderWeight (n + 1)) 0) : SymbolRing k (n + 1)) ∈
117         orderInitialIdeal k I :=
118     coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k I 0 _ hrelation
119   let Y : MvPolynomial.weightedHomogeneousSubmodule k
120       (@orderWeight (n + 1)) 0 :=
121     principalComponentOnPiece k (@orderWeight (n + 1)) 0 ⟨y, hy⟩
122   have hYX : (Y : SymbolRing k (n + 1)) *
123         MvPolynomial.X (.inl (0 : Fin (n + 1))) - 1 ∈
124       orderInitialIdeal k I := by
125     convert hinitial using 1
126     change (Y : SymbolRing k (n + 1)) *
127         MvPolynomial.X (.inl (0 : Fin (n + 1))) - 1 =
128       presentedPrincipalComponent k orderWeight 0 (y * x - 1)
129     rw [map_sub, presentedPrincipalComponent_mul_order k hy hx]
130     have honePC : presentedPrincipalComponent k orderWeight 0
131         (1 : WeylAlg k (n + 1)) = 1 := by
132       change presentedPrincipalComponent k orderWeight 0
133           ((orderPieceOne (n := n + 1) k : orderPiece k (n + 1) 0) :
134             WeylAlg k (n + 1)) = 1
135       exact presentedPrincipalComponent_orderPieceOne (n := n + 1) k
136     rw [honePC]
137     rw [show MvPolynomial.X (.inl (0 : Fin (n + 1))) =
138         presentedPrincipalComponent k orderWeight 0 x by
139       exact (coe_coordinate_order_symbol k n).symm]
140     change presentedPrincipalComponent k orderWeight 0 y *
141         presentedPrincipalComponent k orderWeight 0 x - 1 = _
142     rfl
143   apply (Ideal.eq_top_iff_one _).2
144   rw [show (1 : SymbolRing k (n + 1)) =
145       -((Y : SymbolRing k (n + 1)) *
146           MvPolynomial.X (.inl (0 : Fin (n + 1))) - 1) +
147         (Y : SymbolRing k (n + 1)) *
148           MvPolynomial.X (.inl (0 : Fin (n + 1))) by ring]
149   exact Submodule.add_mem _
150     (Submodule.mem_sup_left ((orderInitialIdeal k I).neg_mem hYX))
151     (Submodule.mem_sup_right (by
152       rw [Ideal.mem_span_singleton]
153       exact ⟨(Y : SymbolRing k (n + 1)), by rw [mul_comm]⟩))

155 /-- The terminal set-theoretic conclusion follows from the unit-only strict
156 preimage; neither all-degree cancellation nor graded injectivity is consumed. -/
157 theorem canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
158     (n N : ℕ) {d : WeylAlg k (n + 1)}
159     (hunit : StrictUnitCoordinatePreimage k n N d) :
160     Disjoint
161       (orderCharacteristicSupport k (CanonicalIdeal k n N d))
162       (PrimeSpectrum.zeroLocus
163         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
164           Set (SymbolRing k (n + 1)))) := by
165   rw [Set.disjoint_left]
166   intro p hp hpx
167   rw [orderCharacteristicSupport_eq_zeroLocus,
168     PrimeSpectrum.mem_zeroLocus] at hp
169   rw [PrimeSpectrum.mem_zeroLocus] at hpx
170   have htop :=
171     canonical_orderInitialIdeal_sup_coordinate_eq_top_of_strictUnit
172       k n N hunit
173   have hle : orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
174       Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} ≤ p.asIdeal :=
175     sup_le hp (Ideal.span_le.mpr hpx)
176   rw [htop] at hle
177   exact p.2.ne_top (top_unique hle)
