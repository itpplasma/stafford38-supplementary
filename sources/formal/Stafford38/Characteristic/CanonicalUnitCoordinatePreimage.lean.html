7 # A unit-only strict coordinate criterion for canonical axis avoidance
8 
9 The terminal axis-avoidance argument does not require coordinate cancellation
10 on every associated-graded piece.  It only needs the degree-zero class of `1`
11 to lie in the image of the coordinate action.  Equivalently, it is enough to
12 have one representative of order zero whose coordinate multiple equals `1`
13 in the literal canonical quotient.
14 
15 This file proves that this single strict unit preimage puts a relation
16 `P * X - 1` in the order initial ideal and hence excludes the distinguished
17 coordinate hyperplane from characteristic support.  It also records that the
18 previous all-degree cancellation and strict-lower-preimage hypotheses imply
19 the unit-only condition.  No converse is asserted.
20 -/

50 /-- The weakest concrete filtered input used by the terminal axis argument:
51 the quotient class of `1` has one coordinate predecessor of order zero. -/
52 def StrictUnitCoordinatePreimage (n N : ℕ)
53     (d : PresentedWeyl k (n + 1)) : Prop :=
54   ∃ y : PresentedWeyl k (n + 1),
55     y ∈ orderPiece k (n + 1) 0 ∧
56       qmk (CanonicalIdeal k n N d) (y * presentedCoordinate k n) =
57         qmk (CanonicalIdeal k n N d) 1
58 
59 /-- The previous all-degree cancellation hypothesis implies the unit-only
60 criterion, by strictifying an unrestricted preimage only at degree zero. -/
61 theorem strictUnitCoordinatePreimage_of_coordinateCancellation
62     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
63     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
64     (hcancel : CoordinateCancellation k n N d) :
65     StrictUnitCoordinatePreimage k n N d := by
66   obtain ⟨y, hy, hyx⟩ := exists_strict_coordinate_preimage
67     k n N 0 hd hcancel 1 (orderPieceOne (n := n + 1) k).property
68   exact ⟨y, hy, hyx⟩

71 /-- Uniform strict-lower coordinate division is also stronger than necessary:
72 only its instance at cutoff `1` and target `1` is used here. -/
73 theorem strictUnitCoordinatePreimage_of_strictLowerCoordinatePreimages
74     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
75     (hstrict : StrictLowerCoordinatePreimages k n N d) :
76     StrictUnitCoordinatePreimage k n N d := by
77   have hone : (1 : PresentedWeyl k (n + 1)) ∈
78       presentedStrictLowerPiece k orderWeight 1 := by
79     change (1 : PresentedWeyl k (n + 1)) ∈ orderPiece k (n + 1) 0
80     exact (orderPieceOne (n := n + 1) k).property
81   obtain ⟨y, hy, hyx⟩ := hstrict 1 1 hone
82   have hy0 : y ∈ orderPiece k (n + 1) 0 := by
83     change y ∈ orderPiece k (n + 1) 0 at hy
84     exact hy
85   exact ⟨y, hy0, hyx⟩

87 /-- A single strict unit preimage gives the exact degree-zero initial relation
88 needed for scheme-level coordinate-axis avoidance. -/
89 theorem canonical_orderInitialIdeal_sup_coordinate_eq_top_of_strictUnit
90     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
91     (hunit : StrictUnitCoordinatePreimage k n N d) :
92     orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
93         Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} = ⊤ := by
94   let I := CanonicalIdeal k n N d
95   let x := presentedCoordinate k n
96   obtain ⟨y, hy, hyx⟩ := hunit
97   have hx : x ∈ orderPiece k (n + 1) 0 :=
98     presentedCoordinate_mem_orderPiece_zero k n
99   have hyxOrder : y * x ∈ orderPiece k (n + 1) 0 :=
100     mul_mem_orderPiece k hy hx
101   have honeOrder : (1 : PresentedWeyl k (n + 1)) ∈
102       orderPiece k (n + 1) 0 :=
103     (orderPieceOne (n := n + 1) k).property
104   have hdiffOrder : y * x - 1 ∈ orderPiece k (n + 1) 0 :=
105     (orderPiece k (n + 1) 0).sub_mem hyxOrder honeOrder
106   have hdiffIdeal : y * x - 1 ∈ I :=
107     (Submodule.Quotient.eq I).mp hyx
108   have hrelation := principalComponent_mem_orderSymbolRelation
109     k I 0 (y * x - 1) hdiffOrder hdiffIdeal
110   have hinitial :
111       ((principalComponentOnPiece k (@orderWeight (n + 1)) 0
112           ⟨y * x - 1, hdiffOrder⟩ :
113         MvPolynomial.weightedHomogeneousSubmodule k
114           (@orderWeight (n + 1)) 0) : SymbolRing k (n + 1)) ∈
115         orderInitialIdeal k I :=
116     coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k I 0 _ hrelation
117   let Y : MvPolynomial.weightedHomogeneousSubmodule k
118       (@orderWeight (n + 1)) 0 :=
119     principalComponentOnPiece k (@orderWeight (n + 1)) 0 ⟨y, hy⟩
120   have hYX : (Y : SymbolRing k (n + 1)) *
121         MvPolynomial.X (.inl (0 : Fin (n + 1))) - 1 ∈
122       orderInitialIdeal k I := by
123     convert hinitial using 1
124     change (Y : SymbolRing k (n + 1)) *
125         MvPolynomial.X (.inl (0 : Fin (n + 1))) - 1 =
126       presentedPrincipalComponent k orderWeight 0 (y * x - 1)
127     rw [map_sub, presentedPrincipalComponent_mul_order k hy hx]
128     have honePC : presentedPrincipalComponent k orderWeight 0
129         (1 : PresentedWeyl k (n + 1)) = 1 := by
130       change presentedPrincipalComponent k orderWeight 0
131           ((orderPieceOne (n := n + 1) k : orderPiece k (n + 1) 0) :
132             PresentedWeyl k (n + 1)) = 1
133       exact presentedPrincipalComponent_orderPieceOne (n := n + 1) k
134     rw [honePC]
135     rw [show MvPolynomial.X (.inl (0 : Fin (n + 1))) =
136         presentedPrincipalComponent k orderWeight 0 x by
137       exact (coe_coordinate_order_symbol k n).symm]
138     change presentedPrincipalComponent k orderWeight 0 y *
139         presentedPrincipalComponent k orderWeight 0 x - 1 = _
140     rfl
141   apply (Ideal.eq_top_iff_one _).2
142   rw [show (1 : SymbolRing k (n + 1)) =
143       -((Y : SymbolRing k (n + 1)) *
144           MvPolynomial.X (.inl (0 : Fin (n + 1))) - 1) +
145         (Y : SymbolRing k (n + 1)) *
146           MvPolynomial.X (.inl (0 : Fin (n + 1))) by ring]
147   exact Submodule.add_mem _
148     (Submodule.mem_sup_left ((orderInitialIdeal k I).neg_mem hYX))
149     (Submodule.mem_sup_right (by
150       rw [Ideal.mem_span_singleton]
151       exact ⟨(Y : SymbolRing k (n + 1)), by rw [mul_comm]⟩))

153 /-- The terminal set-theoretic conclusion follows from the unit-only strict
154 preimage; neither all-degree cancellation nor graded injectivity is consumed. -/
155 theorem canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
156     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
157     (hunit : StrictUnitCoordinatePreimage k n N d) :
158     Disjoint
159       (orderCharacteristicSupport k (CanonicalIdeal k n N d))
160       (PrimeSpectrum.zeroLocus
161         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
162           Set (SymbolRing k (n + 1)))) := by
163   rw [Set.disjoint_left]
164   intro p hp hpx
165   rw [orderCharacteristicSupport_eq_zeroLocus,
166     PrimeSpectrum.mem_zeroLocus] at hp
167   rw [PrimeSpectrum.mem_zeroLocus] at hpx
168   have htop :=
169     canonical_orderInitialIdeal_sup_coordinate_eq_top_of_strictUnit
170       k n N hunit
171   have hle : orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
172       Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} ≤ p.asIdeal :=
173     sup_le hp (Ideal.span_le.mpr hpx)
174   rw [htop] at hle
175   exact p.2.ne_top (top_unique hle)
