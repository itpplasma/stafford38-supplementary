11 # Bernstein initial ideals and characteristic support
12 
13 For a right ideal in the presented Weyl algebra, its Bernstein principal
14 components generate an ordinary ideal in the commutative symbol ring.  This
15 file defines the resulting cyclic graded module and its support.  The
16 construction is concrete: no abstract D-module or characteristic-variety
17 interface is assumed.
18 -/

36 /-- Principal components of Bernstein-filtered elements of a right ideal. -/
37 def bernsteinInitialGeneratorSet {n : ℕ}
38     (I : RightIdeal (WeylAlg k n)) : Set (SymbolRing k n) :=
39   {P | ∃ (N : ℕ) (z : WeylAlg k n),
40       z ∈ bernsteinPiece k n N ∧ z ∈ I ∧
41         P = presentedPrincipalComponent k bernsteinWeight N z}
42 
43 /-- The commutative initial ideal of a right ideal for the Bernstein
44 filtration. -/
45 def bernsteinInitialIdeal {n : ℕ}
46     (I : RightIdeal (WeylAlg k n)) : Ideal (SymbolRing k n) :=
47   Ideal.span (bernsteinInitialGeneratorSet k I)
48 
49 /-- The cyclic commutative module defined by the Bernstein initial ideal. -/
50 abbrev BernsteinCharacteristicModule {n : ℕ}
51     (I : RightIdeal (WeylAlg k n)) :=
52   SymbolRing k n ⧸ bernsteinInitialIdeal k I

54 private theorem quotient_subsingleton_iff_top {R : Type u} [CommRing R]
55     (J : Ideal R) : Subsingleton (R ⧸ J) ↔ J = ⊤ := by
56   constructor
57   · intro h
58     apply le_antisymm le_top
59     intro r hr
60     apply (Ideal.Quotient.eq_zero_iff_mem).mp
61     exact Subsingleton.elim (Ideal.Quotient.mk J r) 0
62   · intro h
63     constructor
64     intro a b
65     obtain ⟨a, rfl⟩ := Ideal.Quotient.mk_surjective a
66     obtain ⟨b, rfl⟩ := Ideal.Quotient.mk_surjective b
67     have ha : Ideal.Quotient.mk J a = 0 :=
68       (Ideal.Quotient.eq_zero_iff_mem).2 (h ▸ trivial)
69     have hb : Ideal.Quotient.mk J b = 0 :=
70       (Ideal.Quotient.eq_zero_iff_mem).2 (h ▸ trivial)
71     exact ha.trans hb.symm

73 /-- Characteristic support as module support on the affine symbol spectrum. -/
74 def bernsteinCharacteristicSupport {n : ℕ}
75     (I : RightIdeal (WeylAlg k n)) : Set (PrimeSpectrum (SymbolRing k n)) :=
76   Module.support (SymbolRing k n) (BernsteinCharacteristicModule k I)
77 
78 /-- The characteristic module is cyclic, so its support is exactly the closed
79 set cut out by the initial ideal. -/
80 theorem bernsteinCharacteristicSupport_eq_zeroLocus {n : ℕ}
81     (I : RightIdeal (WeylAlg k n)) :
82     bernsteinCharacteristicSupport k I =
83       PrimeSpectrum.zeroLocus (bernsteinInitialIdeal k I) := by
84   rw [bernsteinCharacteristicSupport, Module.support_of_algebra]
85   apply congrArg PrimeSpectrum.zeroLocus
86   ext P
87   change algebraMap (SymbolRing k n) (BernsteinCharacteristicModule k I) P = 0 ↔
88     P ∈ bernsteinInitialIdeal k I
89   exact Ideal.Quotient.eq_zero_iff_mem
90 
91 /-- Empty characteristic support is equivalent to the initial ideal being
92 the unit ideal.  Reflecting this condition back to the Weyl right ideal is a
93 separate filtered degree-lowering theorem. -/
94 theorem bernsteinCharacteristicSupport_eq_empty_iff {n : ℕ}
95     (I : RightIdeal (WeylAlg k n)) :
96     bernsteinCharacteristicSupport k I = ∅ ↔
97       bernsteinInitialIdeal k I = ⊤ := by
98   rw [bernsteinCharacteristicSupport, Module.support_eq_empty_iff,
99     quotient_subsingleton_iff_top]

101 theorem principalComponent_mem_bernsteinInitialIdeal {n N : ℕ}
102     (I : RightIdeal (WeylAlg k n)) (z : WeylAlg k n)
103     (hz : z ∈ bernsteinPiece k n N) (hI : z ∈ I) :
104     presentedPrincipalComponent k bernsteinWeight N z ∈
105       bernsteinInitialIdeal k I := by
106   apply Ideal.subset_span
107   exact ⟨N, z, hz, hI, rfl⟩

109 theorem mem_annihilator_bernsteinCharacteristicModule {n : ℕ}
110     (I : RightIdeal (WeylAlg k n)) {P : SymbolRing k n}
111     (hP : P ∈ bernsteinInitialIdeal k I) :
112     P ∈ Module.annihilator (SymbolRing k n)
113       (BernsteinCharacteristicModule k I) := by
114   rw [Module.mem_annihilator]
115   intro q
116   obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective q
117   change Ideal.Quotient.mk (bernsteinInitialIdeal k I) (P * r) = 0
118   rw [Ideal.Quotient.eq_zero_iff_mem]
119   exact (bernsteinInitialIdeal k I).mul_mem_right r hP

121 /-- Every element of the initial ideal vanishes on characteristic support. -/
122 theorem bernsteinCharacteristicSupport_subset_zeroLocus {n : ℕ}
123     (I : RightIdeal (WeylAlg k n)) {P : SymbolRing k n}
124     (hP : P ∈ bernsteinInitialIdeal k I) :
125     bernsteinCharacteristicSupport k I ⊆
126       PrimeSpectrum.zeroLocus ({P} : Set (SymbolRing k n)) := by
127   intro p hp
128   rw [PrimeSpectrum.mem_zeroLocus]
129   intro f hf
130   rw [Set.mem_singleton_iff.mp hf]
131   exact Module.annihilator_le_of_mem_support hp

168 /-! ## Differential-order characteristic support
169 
170 The preceding Bernstein construction is a finite-weight scaffold. The
171 noncharacteristic theorem uses the differential-order filtration, whose base
172 variables have weight zero and fibre variables weight one. The canonical
173 theorems below deliberately retain the normalized Bernstein bound: the
174 in characteristic zero, symplectic normalization produces it for every nonzero
175 non-scalar operator
176 (`Stafford38.WeylMonicNormalization.scalar_or_normalized_symplectic_image`), and together
177 with order degree `N` it forces the order principal symbol to involve fibre
178 variables only. This implication is proved explicitly below.
179 
180 The cyclic quotient by `orderInitialIdeal` is the intended concrete model for
181 the associated graded quotient. Its identification with the filtration
182 quotients, and reflection of an empty order support back to the Weyl quotient,
183 remain separate theorems; neither is assumed here. -/
184 
185 def orderInitialGeneratorSet {n : ℕ}
186     (I : RightIdeal (WeylAlg k n)) : Set (SymbolRing k n) :=
187   {P | ∃ (N : ℕ) (z : WeylAlg k n),
188       z ∈ orderPiece k n N ∧ z ∈ I ∧
189         P = presentedPrincipalComponent k orderWeight N z}

191 def orderInitialIdeal {n : ℕ}
192     (I : RightIdeal (WeylAlg k n)) : Ideal (SymbolRing k n) :=
193   Ideal.span (orderInitialGeneratorSet k I)

220 /-- Differential order is bounded above by Bernstein degree on every PBW
221 monomial. -/
222 theorem monomialWeight_order_le_bernstein {n : ℕ}
223     (m : PhaseVar n →₀ ℕ) :
224     Finsupp.weight (@orderWeight n) m ≤
225       Finsupp.weight (@bernsteinWeight n) m := by
226   classical
227   simp only [Finsupp.weight_apply, Finsupp.weight_apply]
228   apply Finsupp.sum_le_sum
229   intro j hj
230   cases j <;> simp [orderWeight, fibreWeight, bernsteinWeight]
231 
232 /-- If order degree reaches a Bernstein bound, no coordinate exponent can
233 occur. -/
234 theorem coordinateExponent_eq_zero_of_order_eq_of_bernstein_le
235     {n N : ℕ} (m : PhaseVar n →₀ ℕ)
236     (horder : Finsupp.weight (@orderWeight n) m = N)
237     (hbernstein : Finsupp.weight (@bernsteinWeight n) m ≤ N)
238     (i : Fin n) : m (.inl i) = 0 := by
239   by_contra hi
240   let m' := m - Finsupp.single (.inl i) 1
241   have hordSub := Finsupp.weight_sub_single_add
242     (w := @orderWeight n) hi
243   have hbernSub := Finsupp.weight_sub_single_add
244     (w := @bernsteinWeight n) hi
245   have hle := monomialWeight_order_le_bernstein m'
246   have hordCoordinate : orderWeight (.inl i) = 0 := by
247     simp [orderWeight, fibreWeight]
248   have hbernCoordinate : bernsteinWeight (.inl i) = 1 := by
249     simp [bernsteinWeight]
250   change Finsupp.weight (@orderWeight n) m' + orderWeight (.inl i) =
251     Finsupp.weight (@orderWeight n) m at hordSub
252   change Finsupp.weight (@bernsteinWeight n) m' + bernsteinWeight (.inl i) =
253     Finsupp.weight (@bernsteinWeight n) m at hbernSub
254   rw [hordCoordinate, add_zero, horder] at hordSub
255   rw [hbernCoordinate] at hbernSub
256   omega

258 /-- A symbol uses only fibre variables when every monomial carrying a
259 coordinate exponent has zero coefficient. -/
260 def IsFibreOnly {n : ℕ} (P : SymbolRing k n) : Prop :=
261   ∀ m, MvPolynomial.coeff m P ≠ 0 →
262     ∀ i : Fin n, m (.inl i) = 0

304 theorem canonical_orderPrincipalComponent_pureMomentumCoefficient
305     (n N : ℕ) {d : WeylAlg k (n + 1)}
306     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
307     MvPolynomial.coeff
308         (Finsupp.single (.inr (0 : Fin (n + 1))) N)
309         (presentedPrincipalComponent k orderWeight N d) = 1 := by
310   change AddMonoidAlgebra.coeff
311       (presentedPrincipalComponent k orderWeight N d)
312       (Finsupp.single (.inr (0 : Fin (n + 1))) N) = 1
313   rw [coeff_presentedPrincipalComponent]
314   have hweight : Finsupp.weight (@orderWeight (n + 1))
315       (Finsupp.single (.inr (0 : Fin (n + 1))) N) = N := by
316     simp [Finsupp.weight_apply, orderWeight, fibreWeight]
317   rw [if_pos hweight]
318   exact hd.2

320 /-- The normalized Bernstein bound makes the degree-`N` order symbol a
321 constant-coefficient polynomial in the fibre variables. -/
322 theorem canonical_orderPrincipalComponent_isFibreOnly
323     (n N : ℕ) {d : WeylAlg k (n + 1)}
324     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
325     IsFibreOnly k (presentedPrincipalComponent k orderWeight N d) := by
326   intro m hm i
327   change AddMonoidAlgebra.coeff
328       (presentedPrincipalComponent k orderWeight N d) m ≠ 0 at hm
329   rw [coeff_presentedPrincipalComponent] at hm
330   by_cases horder : Finsupp.weight (@orderWeight (n + 1)) m = N
331   · rw [if_pos horder] at hm
332     have hbernstein :=
333       (mem_presentedWeightPiece k (@bernsteinWeight (n + 1)) N d).mp
334         hd.1 m hm
335     exact coordinateExponent_eq_zero_of_order_eq_of_bernstein_le
336       m horder hbernstein i
337   · rw [if_neg horder] at hm
338     exact (hm rfl).elim
