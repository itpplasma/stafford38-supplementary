8 /-!
9 # Bernstein initial ideals and characteristic support
10 
11 For a right ideal in the presented Weyl algebra, its Bernstein principal
12 components generate an ordinary ideal in the commutative symbol ring.  This
13 file defines the resulting cyclic graded module and its support.  The
14 construction is concrete: no abstract D-module or characteristic-variety
15 interface is assumed.
16 -/

34 /-- Principal components of Bernstein-filtered elements of a right ideal. -/
35 def bernsteinInitialGeneratorSet {n : ℕ}
36     (I : RightIdeal (PresentedWeyl k n)) : Set (SymbolRing k n) :=
37   {P | ∃ (N : ℕ) (z : PresentedWeyl k n),
38       z ∈ bernsteinPiece k n N ∧ z ∈ I ∧
39         P = presentedPrincipalComponent k bernsteinWeight N z}
40 
41 /-- The commutative initial ideal of a right ideal for the Bernstein
42 filtration. -/
43 def bernsteinInitialIdeal {n : ℕ}
44     (I : RightIdeal (PresentedWeyl k n)) : Ideal (SymbolRing k n) :=
45   Ideal.span (bernsteinInitialGeneratorSet k I)

47 /-- The cyclic commutative module defined by the Bernstein initial ideal. -/
48 abbrev BernsteinCharacteristicModule {n : ℕ}
49     (I : RightIdeal (PresentedWeyl k n)) :=
50   SymbolRing k n ⧸ bernsteinInitialIdeal k I

52 private theorem quotient_subsingleton_iff_top {R : Type u} [CommRing R]
53     (J : Ideal R) : Subsingleton (R ⧸ J) ↔ J = ⊤ := by
54   constructor
55   · intro h
56     apply le_antisymm le_top
57     intro r hr
58     apply (Ideal.Quotient.eq_zero_iff_mem).mp
59     exact Subsingleton.elim (Ideal.Quotient.mk J r) 0
60   · intro h
61     constructor
62     intro a b
63     obtain ⟨a, rfl⟩ := Ideal.Quotient.mk_surjective a
64     obtain ⟨b, rfl⟩ := Ideal.Quotient.mk_surjective b
65     have ha : Ideal.Quotient.mk J a = 0 :=
66       (Ideal.Quotient.eq_zero_iff_mem).2 (h ▸ trivial)
67     have hb : Ideal.Quotient.mk J b = 0 :=
68       (Ideal.Quotient.eq_zero_iff_mem).2 (h ▸ trivial)
69     exact ha.trans hb.symm

71 /-- Characteristic support as module support on the affine symbol spectrum. -/
72 def bernsteinCharacteristicSupport {n : ℕ}
73     (I : RightIdeal (PresentedWeyl k n)) : Set (PrimeSpectrum (SymbolRing k n)) :=
74   Module.support (SymbolRing k n) (BernsteinCharacteristicModule k I)
75 
76 /-- The characteristic module is cyclic, so its support is exactly the closed
77 set cut out by the initial ideal. -/
78 theorem bernsteinCharacteristicSupport_eq_zeroLocus {n : ℕ}
79     (I : RightIdeal (PresentedWeyl k n)) :
80     bernsteinCharacteristicSupport k I =
81       PrimeSpectrum.zeroLocus (bernsteinInitialIdeal k I) := by
82   rw [bernsteinCharacteristicSupport, Module.support_of_algebra]
83   apply congrArg PrimeSpectrum.zeroLocus
84   ext P
85   change algebraMap (SymbolRing k n) (BernsteinCharacteristicModule k I) P = 0 ↔
86     P ∈ bernsteinInitialIdeal k I
87   exact Ideal.Quotient.eq_zero_iff_mem

89 /-- Empty characteristic support is equivalent to the initial ideal being
90 the unit ideal.  Reflecting this condition back to the Weyl right ideal is a
91 separate filtered degree-lowering theorem. -/
92 theorem bernsteinCharacteristicSupport_eq_empty_iff {n : ℕ}
93     (I : RightIdeal (PresentedWeyl k n)) :
94     bernsteinCharacteristicSupport k I = ∅ ↔
95       bernsteinInitialIdeal k I = ⊤ := by
96   rw [bernsteinCharacteristicSupport, Module.support_eq_empty_iff,
97     quotient_subsingleton_iff_top]

99 theorem principalComponent_mem_bernsteinInitialIdeal {n N : ℕ}
100     (I : RightIdeal (PresentedWeyl k n)) (z : PresentedWeyl k n)
101     (hz : z ∈ bernsteinPiece k n N) (hI : z ∈ I) :
102     presentedPrincipalComponent k bernsteinWeight N z ∈
103       bernsteinInitialIdeal k I := by
104   apply Ideal.subset_span
105   exact ⟨N, z, hz, hI, rfl⟩

107 theorem mem_annihilator_bernsteinCharacteristicModule {n : ℕ}
108     (I : RightIdeal (PresentedWeyl k n)) {P : SymbolRing k n}
109     (hP : P ∈ bernsteinInitialIdeal k I) :
110     P ∈ Module.annihilator (SymbolRing k n)
111       (BernsteinCharacteristicModule k I) := by
112   rw [Module.mem_annihilator]
113   intro q
114   obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective q
115   change Ideal.Quotient.mk (bernsteinInitialIdeal k I) (P * r) = 0
116   rw [Ideal.Quotient.eq_zero_iff_mem]
117   exact (bernsteinInitialIdeal k I).mul_mem_right r hP

119 /-- Every element of the initial ideal vanishes on characteristic support. -/
120 theorem bernsteinCharacteristicSupport_subset_zeroLocus {n : ℕ}
121     (I : RightIdeal (PresentedWeyl k n)) {P : SymbolRing k n}
122     (hP : P ∈ bernsteinInitialIdeal k I) :
123     bernsteinCharacteristicSupport k I ⊆
124       PrimeSpectrum.zeroLocus ({P} : Set (SymbolRing k n)) := by
125   intro p hp
126   rw [PrimeSpectrum.mem_zeroLocus]
127   intro f hf
128   rw [Set.mem_singleton_iff.mp hf]
129   exact Module.annihilator_le_of_mem_support hp

166 /-! ## Differential-order characteristic support
167 
168 The preceding Bernstein construction is a finite-weight scaffold. The
169 noncharacteristic theorem uses the differential-order filtration, whose base
170 variables have weight zero and fibre variables weight one. The canonical
171 theorems below deliberately retain the normalized Bernstein bound: the
172 in characteristic zero, symplectic normalization produces it for every nonzero
173 non-scalar operator
174 (`Stafford38.WeylMonicNormalization.scalar_or_normalized_symplectic_image`), and together
175 with order degree `N` it forces the order principal symbol to involve fibre
176 variables only. This implication is proved explicitly below.
177 
178 The cyclic quotient by `orderInitialIdeal` is the intended concrete model for
179 the associated graded quotient. Its identification with the filtration
180 quotients, and reflection of an empty order support back to the Weyl quotient,
181 remain separate theorems; neither is assumed here. -/
182 
183 def orderInitialGeneratorSet {n : ℕ}
184     (I : RightIdeal (PresentedWeyl k n)) : Set (SymbolRing k n) :=
185   {P | ∃ (N : ℕ) (z : PresentedWeyl k n),
186       z ∈ orderPiece k n N ∧ z ∈ I ∧
187         P = presentedPrincipalComponent k orderWeight N z}
188 
189 def orderInitialIdeal {n : ℕ}
190     (I : RightIdeal (PresentedWeyl k n)) : Ideal (SymbolRing k n) :=
191   Ideal.span (orderInitialGeneratorSet k I)

218 /-- Differential order is bounded above by Bernstein degree on every PBW
219 monomial. -/
220 theorem monomialWeight_order_le_bernstein {n : ℕ}
221     (m : PhaseVar n →₀ ℕ) :
222     monomialWeight (@orderWeight n) m ≤
223       monomialWeight (@bernsteinWeight n) m := by
224   classical
225   simp only [monomialWeight, Finsupp.weight_apply]
226   apply Finsupp.sum_le_sum
227   intro j hj
228   cases j <;> simp [orderWeight, fibreWeight, bernsteinWeight]
229 
230 /-- If order degree reaches a Bernstein bound, no coordinate exponent can
231 occur. -/
232 theorem coordinateExponent_eq_zero_of_order_eq_of_bernstein_le
233     {n N : ℕ} (m : PhaseVar n →₀ ℕ)
234     (horder : monomialWeight (@orderWeight n) m = N)
235     (hbernstein : monomialWeight (@bernsteinWeight n) m ≤ N)
236     (i : Fin n) : m (.inl i) = 0 := by
237   by_contra hi
238   let m' := m - Finsupp.single (.inl i) 1
239   have hordSub := Finsupp.weight_sub_single_add
240     (w := @orderWeight n) hi
241   have hbernSub := Finsupp.weight_sub_single_add
242     (w := @bernsteinWeight n) hi
243   have hle := monomialWeight_order_le_bernstein m'
244   have hordCoordinate : orderWeight (.inl i) = 0 := by
245     simp [orderWeight, fibreWeight]
246   have hbernCoordinate : bernsteinWeight (.inl i) = 1 := by
247     simp [bernsteinWeight]
248   change monomialWeight (@orderWeight n) m' + orderWeight (.inl i) =
249     monomialWeight (@orderWeight n) m at hordSub
250   change monomialWeight (@bernsteinWeight n) m' + bernsteinWeight (.inl i) =
251     monomialWeight (@bernsteinWeight n) m at hbernSub
252   rw [hordCoordinate, add_zero, horder] at hordSub
253   rw [hbernCoordinate] at hbernSub
254   omega

256 /-- A symbol uses only fibre variables when every monomial carrying a
257 coordinate exponent has zero coefficient. -/
258 def IsFibreOnly {n : ℕ} (P : SymbolRing k n) : Prop :=
259   ∀ m, MvPolynomial.coeff m P ≠ 0 →
260     ∀ i : Fin n, m (.inl i) = 0

302 theorem canonical_orderPrincipalComponent_pureMomentumCoefficient
303     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
304     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
305     MvPolynomial.coeff
306         (Finsupp.single (.inr (0 : Fin (n + 1))) N)
307         (presentedPrincipalComponent k orderWeight N d) = 1 := by
308   change AddMonoidAlgebra.coeff
309       (presentedPrincipalComponent k orderWeight N d)
310       (Finsupp.single (.inr (0 : Fin (n + 1))) N) = 1
311   rw [coeff_presentedPrincipalComponent]
312   have hweight : monomialWeight (@orderWeight (n + 1))
313       (Finsupp.single (.inr (0 : Fin (n + 1))) N) = N := by
314     simp [monomialWeight, orderWeight, fibreWeight]
315   rw [if_pos hweight]
316   exact hd.2

318 /-- The normalized Bernstein bound makes the degree-`N` order symbol a
319 constant-coefficient polynomial in the fibre variables. -/
320 theorem canonical_orderPrincipalComponent_isFibreOnly
321     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
322     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
323     IsFibreOnly k (presentedPrincipalComponent k orderWeight N d) := by
324   intro m hm i
325   change AddMonoidAlgebra.coeff
326       (presentedPrincipalComponent k orderWeight N d) m ≠ 0 at hm
327   rw [coeff_presentedPrincipalComponent] at hm
328   by_cases horder : monomialWeight (@orderWeight (n + 1)) m = N
329   · rw [if_pos horder] at hm
330     have hbernstein :=
331       (mem_presentedWeightPiece k (@bernsteinWeight (n + 1)) N d).mp
332         hd.1 m hm
333     exact coordinateExponent_eq_zero_of_order_eq_of_bernstein_le
334       m horder hbernstein i
335   · rw [if_neg horder] at hm
336     exact (hm rfl).elim
