25 /-! A zero coordinate cokernel has empty support.  Transposition fixes the
26 coordinate variable, so the same empty-support statement excludes the
27 coordinate zero locus from the transposed support. -/
28 theorem canonical_support_avoidance_of_coordinate_cokernel_subsingleton
29     (n N : ℕ) (d : WeylAlg k (n + 1))
30     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
31     (hzero : Subsingleton (QuotSMulTop
32       (MvPolynomial.X (.inl (0 : Fin (n + 1))) : SymbolRing k (n + 1))
33       (OrderAssociatedGradedModule k
34         (canonicalRightIdeal (presentedCoordinate k n) d N)))) :
35     Disjoint
36       (transposedOrderAssociatedGradedSupport k
37         (canonicalRightIdeal (presentedCoordinate k n) d N))
38       (PrimeSpectrum.zeroLocus
39         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
40           Set (SymbolRing k (n + 1)))) := by
41   let R := SymbolRing k (n + 1)
42   let I := canonicalRightIdeal (presentedCoordinate k n) d N
43   let E := OrderAssociatedGradedModule k I
44   let x : R := MvPolynomial.X (.inl (0 : Fin (n + 1)))
45   let τ : R →+* R := (symbolTranspositionEquiv k).toRingEquiv.toRingHom
46   let qmod := QuotSMulTop x E
47   letI : Subsingleton qmod := hzero
48   have hqempty : Module.support R qmod = ∅ := Module.support_eq_empty
49   rw [transposedOrderAssociatedGradedSupport_eq_preimage]
50   apply Set.disjoint_left.2
51   intro p hp haxis
52   let q : PrimeSpectrum R := PrimeSpectrum.comap τ p
53   have hq : q ∈ Module.support R E := hp
54   have hqx : q ∈ PrimeSpectrum.zeroLocus ({x} : Set R) := by
55     rw [PrimeSpectrum.mem_zeroLocus]
56     intro y hy
57     have hyx : y = x := by simpa using hy
58     subst y
59     change τ x ∈ p.asIdeal
60     have hτ : τ x = x := by
61       change symbolTransposition k (MvPolynomial.X (.inl (0 : Fin (n + 1)))) = _
62       exact symbolTransposition_coordinate k 0
63     rw [hτ]
64     exact haxis (by simp [x])
65   have hqmod : q ∈ Module.support R qmod := by
66     rw [Module.support_quotSMulTop]
67     exact ⟨hq, hqx⟩
68   rw [hqempty] at hqmod
69   exact hqmod
70 
71 end
72 end Stafford38.Characteristic.CanonicalSupportAvoidanceFromCokernel
