9 /-!
10 # Zero-section containment of the reduced order support
11 
12 The differential-order initial ideal is homogeneous for fibre degree.  At the
13 level of field-valued points this forces every fibre-conical support fibre to
14 contain its zero: if `(y, ξ)` is a common zero of the reduced order support
15 ideal, then `(y, 0)` is a common zero as well.
16 
17 The proof does not assume conicality of the radical.  For an element of the
18 radical, it takes a power in the order initial ideal, extracts its fibre-degree
19 zero component there, and then uses reducedness of the ground field.
20 -/

48 /-- Evaluation of the fibre-degree-zero component at `(y, ξ)` is evaluation
49 of the original polynomial at `(y, 0)`. -/
50 private theorem eval_zeroComponent
51     (y ξ : Fin n → k) (P : SymbolRing k n) :
52     MvPolynomial.eval (Sum.elim y ξ)
53         (DirectSum.decompose (orderDecomposition (n := n) k) P 0 :
54           SymbolRing k n) =
55       MvPolynomial.eval (zeroSectionPoint y) P := by
56   induction P using MvPolynomial.induction_on with
57   | C a =>
58       have hC : MvPolynomial.C a ∈ orderDecomposition (n := n) k 0 :=
59         MvPolynomial.isWeightedHomogeneous_C (@orderWeight n) a
60       let C0 : orderDecomposition (n := n) k 0 := ⟨MvPolynomial.C a, hC⟩
61       have hd := congrArg (fun z => z 0)
62         (DirectSum.decompose_coe (orderDecomposition (n := n) k) C0)
63       have hd' : (DirectSum.decompose (orderDecomposition (n := n) k)
64           (MvPolynomial.C a) 0 : SymbolRing k n) = MvPolynomial.C a := by
65         simpa [C0] using congrArg Subtype.val hd
66       rw [hd']
67       simp
68   | add P Q hP hQ =>
69       rw [DirectSum.decompose_add]
70       change MvPolynomial.eval (Sum.elim y ξ)
71           ((DirectSum.decompose (orderDecomposition (n := n) k) P 0 :
72               SymbolRing k n) +
73             (DirectSum.decompose (orderDecomposition (n := n) k) Q 0 :
74               SymbolRing k n)) = _
75       simp only [MvPolynomial.eval_add, hP, hQ]
76   | mul_X P i hP =>
77       rcases i with i | i
78       · have hX : MvPolynomial.X (Sum.inl i : PhaseVar n) ∈
79             orderDecomposition (n := n) k 0 := by
80           exact MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n)
81             (Sum.inl i)
82         rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
83           (orderDecomposition (n := n) k) hX (Nat.zero_le 0)]
84         simp only [Nat.zero_sub, MvPolynomial.eval_mul, hP,
85           MvPolynomial.eval_X]
86         rfl
87       · have hX : MvPolynomial.X (Sum.inr i : PhaseVar n) ∈
88             orderDecomposition (n := n) k 1 := by
89           exact MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n)
90             (Sum.inr i)
91         rw [DirectSum.coe_decompose_mul_of_right_mem_of_not_le
92           (orderDecomposition (n := n) k) hX (by omega)]
93         simp [zeroSectionPoint]

95 /-- A field-valued common zero of the reduced order support ideal remains a
96 common zero after its fibre coordinate is set to zero. -/
97 theorem zeroSection_mem_of_mem_reducedOrderSupport_zeroSet
98     (I : RightIdeal (WeylAlg k n)) (y ξ : Fin n → k)
99     (hpoint : ∀ P ∈ reducedOrderSupportIdeal k I,
100       MvPolynomial.eval (Sum.elim y ξ) P = 0) :
101     ∀ P ∈ reducedOrderSupportIdeal k I,
102       MvPolynomial.eval (zeroSectionPoint y) P = 0 := by
103   intro P hP
104   obtain ⟨m, hm⟩ := (mem_reducedOrderSupportIdeal_iff k I P).mp hP
105   have hcomponent :
106       (DirectSum.decompose (orderDecomposition (n := n) k) (P ^ m) 0 :
107         SymbolRing k n) ∈ orderInitialIdeal k I :=
108     coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k I 0
109       (DirectSum.decompose (orderDecomposition (n := n) k) (P ^ m) 0)
110       (decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
111         k I (P ^ m) hm 0)
112   have hevalComponent := hpoint _
113     (orderInitialIdeal_le_reducedOrderSupportIdeal k I hcomponent)
114   rw [eval_zeroComponent k y ξ (P ^ m)] at hevalComponent
115   rw [map_pow] at hevalComponent
116   exact (pow_eq_zero_iff'.mp hevalComponent).1

120 end
