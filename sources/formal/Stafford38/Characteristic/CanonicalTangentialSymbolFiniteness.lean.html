76 /-- Multiplication by the polynomial variable after splitting off `x₀`.
77 The lemma below identifies this transported operator with the actual `x₀`
78 symbol action. -/
79 def tangentialCoordinateMap (n : ℕ) (E : Type*) [AddCommGroup E]
80     [Module (SymbolRing k (n + 1)) E] :
81     E →ₗ[tangentialCoeffRing (k := k) n] E := by
82   refine
83     { toFun := fun z => (Polynomial.X :
84           Polynomial (tangentialCoeffRing (k := k) n)) • z
85       map_add' := smul_add _
86       map_smul' := ?_ }
87   intro r z
88   change (Polynomial.X : Polynomial (tangentialCoeffRing (k := k) n)) •
89       (Polynomial.C r • z) =
90     Polynomial.C r •
91       ((Polynomial.X : Polynomial (tangentialCoeffRing (k := k) n)) • z)
92   rw [← mul_smul, ← mul_smul, mul_comm]

95     [AddCommGroup E] [Module (SymbolRing k (n + 1)) E] (z : E) :
96     tangentialCoordinateMap (k := k) n E z =
97       (MvPolynomial.X (Sum.inl (0 : Fin (n + 1))) :
98         SymbolRing k (n + 1)) • z := by
99   change (normalSymbolAlgEquiv (k := k) n).symm
100       (Polynomial.C ((normalCoeffTangentialAlgEquiv (k := k) n).symm
101         Polynomial.X)) • z = _
102   congr 1

104   rw [(normalSymbolAlgEquiv (k := k) n).apply_symm_apply]
105   have hx : (normalCoeffTangentialAlgEquiv (k := k) n).symm
106       Polynomial.X = MvPolynomial.X

115 /-- Kernel and cokernel of the actual distinguished coordinate symbol on the
116 canonical associated graded quotient are finite over the ring omitting both
117 that coordinate and the normal covariable. -/
118 theorem canonical_orderAssociatedGradedModule_finite_tangential_coordinate
119     {n N : ℕ} {d : WeylAlg k (n + 1)}
120     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
121     let E := OrderAssociatedGradedModule k
122       (canonicalRightIdeal (presentedCoordinate k n) d N)
123     @Module.Finite (tangentialCoeffRing (k := k) n)
124       (LinearMap.ker (tangentialCoordinateMap (k := k) n E)) _ _
125         (inferInstance) ∧
126     @Module.Finite (tangentialCoeffRing (k := k) n)
127       (E ⧸ LinearMap.range (tangentialCoordinateMap (k := k) n E)) _ _
128         (inferInstance) := by
129   let T := tangentialCoeffRing (k := k) n
130   let R := normalCoeffRing (k := k) n
131   let E := OrderAssociatedGradedModule k
132     (canonicalRightIdeal (presentedCoordinate k n) d N)
133   let e := normalCoeffTangentialAlgEquiv (k := k) n
134   letI : Module R E := normalCoeffModule (k := k) n E
135   haveI hR : Module.Finite R E :=
136     canonical_orderAssociatedGradedModule_finite_normalCoeffRing hd
137   letI : Module (Polynomial T) E := tangentialPolynomialModule (k := k) n E
138   letI : Module T E := tangentialCoeffModule (k := k) n E
139   letI : IsScalarTower T (Polynomial T) E :=
140     ⟨by
141       intro r p z
142       rw [Polynomial.smul_eq_C_mul, mul_smul]
143       change Polynomial.C r • (p • z) = Polynomial.C r • (p • z)
144       rfl⟩
145   have hscalar (r : R) (z : E) : e r • z = r • z := by
146     change (normalSymbolAlgEquiv (k := k) n).symm
147         (Polynomial.C (e.symm (e r))) • z =
148       (normalSymbolAlgEquiv (k := k) n).symm (Polynomial.C r) • z
149     rw [e.symm_apply_apply]
150   haveI : Module.Finite (Polynomial T) E := by
151     rcases hR with ⟨⟨s, hs⟩⟩
152     refine ⟨⟨s, top_unique ?_⟩⟩
153     intro z hz
154     have hzold : z ∈ Submodule.span R (s : Set E) := by
155       rw [hs]
156       exact Submodule.mem_top
157     exact Submodule.span_induction
158       (p := fun z _ => z ∈ Submodule.span (Polynomial T) (s : Set E))
159       (fun x hx => Submodule.subset_span hx)
160       (Submodule.zero_mem _)
161       (fun _ _ _ _ hx hy => Submodule.add_mem _ hx hy)
162       (fun r x _ hx => by
163         rw [← hscalar r x]
164         exact Submodule.smul_mem _ (e r) hx)
165       hzold
166   let f := LinearMap.lsmul (Polynomial T) E Polynomial.X
167   have hf : tangentialCoordinateMap (k := k) n E =
168       f.restrictScalars T := by
169     ext z
170     rfl
171   dsimp only
172   rw [hf]
173   exact finite_kernel_and_cokernel_variable (R := T) (E := E)
