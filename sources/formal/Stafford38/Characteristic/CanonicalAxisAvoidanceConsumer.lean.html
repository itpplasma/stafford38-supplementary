12 # Conditional consumer for canonical axis avoidance
13 
14 This file assumes, but does not prove, the missing degreewise cancellation
15 statement for the canonical filtered quotient.  It combines that hypothesis
16 with the proved surjectivity of right multiplication by the distinguished
17 coordinate and derives strict filtered surjectivity and its associated-graded
18 form.
19 -/

43 /-- The distinguished coordinate belongs to differential order zero. -/
44 theorem presentedCoordinate_mem_orderPiece_zero (n : ℕ) :
45     presentedCoordinate k n ∈ orderPiece k (n + 1) 0 := by
46   rw [orderPiece, mem_presentedWeightPiece,
47     presentedCoordinate, presentedNormalFormLinearEquiv_generator]
48   intro m hm
49   change AddMonoidAlgebra.coeff
50     (MvPolynomial.X (.inl (0 : Fin (n + 1)))) m ≠ 0 at hm
51   rw [MvPolynomial.coeff_X'] at hm
52   split at hm
53   · next heq =>
54       subst m
55       simp [Finsupp.weight_apply, orderWeight, fibreWeight]
56   · exact (hm rfl).elim

58 /-- The exact missing input: multiplication by the coordinate can be
59 cancelled in every degree of the actual filtered quotient. -/
60 def CoordinateCancellation (n N : ℕ)
61     (d : WeylAlg k (n + 1)) : Prop :=
62   ∀ (m : ℕ) (z : WeylAlg k (n + 1)),
63     z ∈ orderPiece k (n + 1) m →
64     z * presentedCoordinate k n ∈
65         rightIdealKSubmodule k (canonicalRightIdeal (presentedCoordinate k n) d N) ⊔
66           presentedStrictLowerPiece k orderWeight m →
67     z ∈ rightIdealKSubmodule k (canonicalRightIdeal (presentedCoordinate k n) d N) ⊔
68       presentedStrictLowerPiece k orderWeight m

70 /-- Cancellation plus unrestricted quotient surjectivity lowers an arbitrary
71 predecessor until it has no larger order than the target. -/
72 theorem exists_strict_coordinate_preimage
73     (n N m : ℕ) {d : WeylAlg k (n + 1)}
74     (hd : Stafford38.WeylPBWMonicBridge.IsPBWMonicAt k
75       (.inr (0 : Fin (n + 1))) N d)
76     (hcancel : CoordinateCancellation k n N d)
77     (z : WeylAlg k (n + 1))
78     (hz : z ∈ orderPiece k (n + 1) m) :
79     ∃ y : WeylAlg k (n + 1),
80       y ∈ orderPiece k (n + 1) m ∧
81       qmk (canonicalRightIdeal (presentedCoordinate k n) d N) (y * presentedCoordinate k n) =
82         qmk (canonicalRightIdeal (presentedCoordinate k n) d N) z := by
83   let I := canonicalRightIdeal (presentedCoordinate k n) d N
84   let x := presentedCoordinate k n
85   have hsurj : Function.Surjective (rightMul I x) := by
86     exact presentedCanonicalRightQuotient_rightMul_coordinate_surjective
87       (k := k) n N hd
88   obtain ⟨q, hq⟩ := hsurj (qmk I z)
89   obtain ⟨y, rfl⟩ := Submodule.Quotient.mk_surjective I q
90   change rightMul I x (qmk I y) = qmk I z at hq
91   rw [← qmk_right_mul] at hq
92   obtain ⟨L, hyL⟩ := exists_mem_orderPiece k y
93   have hdesc : ∀ L, ∀ y : WeylAlg k (n + 1),
94       y ∈ orderPiece k (n + 1) L → qmk I (y * x) = qmk I z →
95       ∃ y' : WeylAlg k (n + 1),
96         y' ∈ orderPiece k (n + 1) m ∧ qmk I (y' * x) = qmk I z := by
97     intro L
98     induction L using Nat.strong_induction_on with
99     | h L ih =>
100       intro y hyL hyx
101       by_cases hLm : L ≤ m
102       · exact ⟨y, presentedWeightPiece_mono k orderWeight hLm hyL, hyx⟩
103       · have hmL : m < L := Nat.lt_of_not_ge hLm
104         have hzLower : z ∈ presentedStrictLowerPiece k orderWeight L := by
105           cases L with
106           | zero => omega
107           | succ L =>
108               simp only [presentedStrictLowerPiece]
109               exact presentedWeightPiece_mono k orderWeight (Nat.le_of_lt_succ hmL) hz
110         have hyxSub : y * x - z ∈ I :=
111           (Submodule.Quotient.eq I).mp hyx
112         have hyxMem : y * x ∈
113             rightIdealKSubmodule k I ⊔
114               presentedStrictLowerPiece k orderWeight L := by
115           rw [show y * x = (y * x - z) + z by abel]
116           exact Submodule.add_mem _
117             (Submodule.mem_sup_left hyxSub)
118             (Submodule.mem_sup_right hzLower)
119         have hyMem := hcancel L y hyL hyxMem
120         obtain ⟨i, hi, l, hl, hil⟩ := Submodule.mem_sup.mp hyMem
121         cases L with
122         | zero => omega
123         | succ L =>
124             simp only [presentedStrictLowerPiece] at hl
125             have hyl : qmk I y = qmk I l := by
126               apply (Submodule.Quotient.eq I).2
127               rw [← hil]
128               simpa using hi
129             have hlx : qmk I (l * x) = qmk I z := by
130               rw [qmk_right_mul, ← hyl, ← qmk_right_mul]
131               exact hyx
132             exact ih L (Nat.lt_succ_self L) l hl hlx
133   exact hdesc L y hyL hq

135 /-- Every actual associated-graded piece has surjective multiplication by
136 the order-zero coordinate symbol. -/
137 theorem canonical_graded_coordinate_surjective
138     (n N m : ℕ) {d : WeylAlg k (n + 1)}
139     (hd : Stafford38.WeylPBWMonicBridge.IsPBWMonicAt k
140       (.inr (0 : Fin (n + 1))) N d)
141     (hcancel : CoordinateCancellation k n N d) :
142     Function.Surjective
143       (quotientOrderHomogeneousAction (N := m) (M := 0)
144         k (canonicalRightIdeal (presentedCoordinate k n) d N)
145         (principalComponentOnPiece k (@orderWeight (n + 1)) 0
146           ⟨presentedCoordinate k n,
147             presentedCoordinate_mem_orderPiece_zero k n⟩)) := by
148   intro q
149   obtain ⟨z, rfl⟩ := orderPieceToQuotientGraded_surjective
150     k (canonicalRightIdeal (presentedCoordinate k n) d N) m q
151   obtain ⟨y, hy, hyx⟩ :=
152     exists_strict_coordinate_preimage k n N m hd hcancel z z.property
153   refine ⟨orderPieceToQuotientGraded k (canonicalRightIdeal (presentedCoordinate k n) d N) m ⟨y, hy⟩, ?_⟩
154   rw [quotientOrderHomogeneousAction_mk_mul]
155   have hyx' : (rightIdealKSubmodule k (canonicalRightIdeal (presentedCoordinate k n) d N)).mkQ
156         (y * presentedCoordinate k n) =
157       (rightIdealKSubmodule k (canonicalRightIdeal (presentedCoordinate k n) d N)).mkQ z :=
158     congrArg
159       (filteredRightQuotientEquivRightQuotient k
160         (canonicalRightIdeal (presentedCoordinate k n) d N)).symm hyx
161   apply (Submodule.Quotient.eq _).2
162   change (orderPieceToQuotientPiece k (canonicalRightIdeal (presentedCoordinate k n) d N) m
163       ⟨y * presentedCoordinate k n, _⟩ :
164         FilteredRightQuotient k (canonicalRightIdeal (presentedCoordinate k n) d N)) -
165       orderPieceToQuotientPiece k (canonicalRightIdeal (presentedCoordinate k n) d N) m z ∈
166         quotientOrderStrictLowerPiece k (canonicalRightIdeal (presentedCoordinate k n) d N) m
167   rw [show (orderPieceToQuotientPiece k (canonicalRightIdeal (presentedCoordinate k n) d N) m
168       ⟨y * presentedCoordinate k n, _⟩ :
169         FilteredRightQuotient k (canonicalRightIdeal (presentedCoordinate k n) d N)) =
170       (rightIdealKSubmodule k (canonicalRightIdeal (presentedCoordinate k n) d N)).mkQ
171         (y * presentedCoordinate k n) from rfl]
172   rw [show (orderPieceToQuotientPiece k (canonicalRightIdeal (presentedCoordinate k n) d N) m z :
173       FilteredRightQuotient k (canonicalRightIdeal (presentedCoordinate k n) d N)) =
174       (rightIdealKSubmodule k (canonicalRightIdeal (presentedCoordinate k n) d N)).mkQ z from rfl]
175   rw [hyx', sub_self]
176   exact Submodule.zero_mem _

178 /-- The order-zero principal symbol of the distinguished coordinate is the
179 corresponding base variable. -/
180 theorem coe_coordinate_order_symbol (n : ℕ) :
181     ((principalComponentOnPiece k (@orderWeight (n + 1)) 0
182         ⟨presentedCoordinate k n,
183           presentedCoordinate_mem_orderPiece_zero k n⟩) :
184       SymbolRing k (n + 1)) =
185       MvPolynomial.X (.inl (0 : Fin (n + 1))) := by
186   ext m
187   change AddMonoidAlgebra.coeff
188       (presentedPrincipalComponent k orderWeight 0
189         (presentedCoordinate k n)) m =
190     AddMonoidAlgebra.coeff (MvPolynomial.X (.inl (0 : Fin (n + 1)))) m
191   rw [coeff_presentedPrincipalComponent, presentedCoordinate,
192     presentedNormalFormLinearEquiv_generator, MvPolynomial.coeff_X']
193   by_cases hm : Finsupp.single (.inl (0 : Fin (n + 1))) 1 = m
194   · subst m
195     simp [Finsupp.weight_apply, orderWeight, fibreWeight]
196   · simp [hm]

198 /-- Conditional scheme-level axis avoidance: the order initial ideal together
199 with the distinguished coordinate generates the unit ideal. -/
200 theorem canonical_orderInitialIdeal_sup_coordinate_eq_top
201     (n N : ℕ) {d : WeylAlg k (n + 1)}
202     (hd : Stafford38.WeylPBWMonicBridge.IsPBWMonicAt k
203       (.inr (0 : Fin (n + 1))) N d)
204     (hcancel : CoordinateCancellation k n N d) :
205     orderInitialIdeal k (canonicalRightIdeal (presentedCoordinate k n) d N) ⊔
206         Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} = ⊤ := by
207   let I := canonicalRightIdeal (presentedCoordinate k n) d N
208   let X : MvPolynomial.weightedHomogeneousSubmodule k
209       (@orderWeight (n + 1)) 0 :=
210     principalComponentOnPiece k (@orderWeight (n + 1)) 0
211       ⟨presentedCoordinate k n, presentedCoordinate_mem_orderPiece_zero k n⟩
212   let oneH : MvPolynomial.weightedHomogeneousSubmodule k
213       (@orderWeight (n + 1)) 0 :=
214     ⟨1, MvPolynomial.isWeightedHomogeneous_one k orderWeight⟩
215   have hsurjActual := canonical_graded_coordinate_surjective
216     k n N 0 hd hcancel
217   have hsurjSymbol : Function.Surjective
218       (homogeneousSymbolAction (N := 0) (M := 0) k I X) := by
219     intro q
220     obtain ⟨r, hr⟩ := hsurjActual
221       ((quotientOrderGradedPieceEquivSymbols k I 0).symm q)
222     refine ⟨quotientOrderGradedPieceEquivSymbols k I 0 r, ?_⟩
223     rw [← quotientOrderHomogeneousAction_compatibility]
224     simpa [I, X] using congrArg
225       (quotientOrderGradedPieceEquivSymbols k I 0) hr
226   obtain ⟨q, hq⟩ := hsurjSymbol (Submodule.Quotient.mk oneH)
227   obtain ⟨P, rfl⟩ := Submodule.Quotient.mk_surjective
228     (orderSymbolRelation k I 0) q
229   rw [homogeneousSymbolAction_mk] at hq
230   have hrel : homogeneousRightMul k X P - oneH ∈
231       orderSymbolRelation k I 0 :=
232     (Submodule.Quotient.eq _).mp hq
233   have hJ : ((homogeneousRightMul k X P - oneH :
234       MvPolynomial.weightedHomogeneousSubmodule k
235         (@orderWeight (n + 1)) 0) : SymbolRing k (n + 1)) ∈
236       orderInitialIdeal k I :=
237     (mem_orderSymbolRelation_iff_coe_mem_orderInitialIdeal k I 0 _).mp hrel
238   have hJ' : (P : SymbolRing k (n + 1)) *
239         MvPolynomial.X (.inl (0 : Fin (n + 1))) - 1 ∈
240       orderInitialIdeal k I := by
241     convert hJ using 1
242     simp only [Submodule.coe_sub, coe_homogeneousRightMul,
243       Submodule.coe_mk]
244     rw [show (X : SymbolRing k (n + 1)) =
245       MvPolynomial.X (.inl (0 : Fin (n + 1))) by
246         exact coe_coordinate_order_symbol k n]
247   apply (Ideal.eq_top_iff_one _).2
248   rw [show (1 : SymbolRing k (n + 1)) =
249       -((P : SymbolRing k (n + 1)) * MvPolynomial.X (.inl 0) - 1) +
250         (P : SymbolRing k (n + 1)) * MvPolynomial.X (.inl 0) by ring]
251   apply Submodule.add_mem
252   · apply Submodule.mem_sup_left
253     apply (orderInitialIdeal k I).neg_mem
254     exact hJ'
255   · apply Submodule.mem_sup_right
256     rw [Ideal.mem_span_singleton]
257     exact ⟨(P : SymbolRing k (n + 1)), by rw [mul_comm]⟩

259 /-- Set-theoretic consequence: the canonical order-characteristic support is
260 disjoint from the distinguished coordinate hyperplane. -/
261 theorem canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus
262     (n N : ℕ) {d : WeylAlg k (n + 1)}
263     (hd : Stafford38.WeylPBWMonicBridge.IsPBWMonicAt k
264       (.inr (0 : Fin (n + 1))) N d)
265     (hcancel : CoordinateCancellation k n N d) :
266     Disjoint
267       (orderCharacteristicSupport k (canonicalRightIdeal (presentedCoordinate k n) d N))
268       (PrimeSpectrum.zeroLocus
269         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
270           Set (SymbolRing k (n + 1)))) := by
271   rw [Set.disjoint_left]
272   intro p hp hpx
273   rw [orderCharacteristicSupport_eq_zeroLocus,
274     PrimeSpectrum.mem_zeroLocus] at hp
275   rw [PrimeSpectrum.mem_zeroLocus] at hpx
276   have htop := canonical_orderInitialIdeal_sup_coordinate_eq_top
277     k n N hd hcancel
278   have hle : orderInitialIdeal k (canonicalRightIdeal (presentedCoordinate k n) d N) ⊔
279       Ideal.span {MvPolynomial.X (.inl (0 : Fin (n + 1)))} ≤ p.asIdeal :=
280     sup_le hp (Ideal.span_le.mpr hpx)
281   rw [htop] at hle
282   exact p.2.ne_top (top_unique hle)
