10 /-!
11 # The localized order-Rees two-jet specialization kernel
12 
13 For the concrete order-Rees two-jet, localization of the opposite ring does
14 not enlarge the specialization kernel beyond the localized central parameter.
15 The proof is direct on Ore fractions.  A denominator that kills the
16 specialized numerator is lifted through the surjective two-jet
17 specialization, the numerator is enlarged in the opposite ring, and the exact
18 unlocalized kernel theorem factors it by the Rees parameter.
19 
20 Products remain in the opposite/right-module orientation throughout.  In
21 particular, the localized parameter is written on the left; centrality is used
22 explicitly before applying the unlocalized factorization.
23 -/

92 /-- The concrete localized opposite order-Rees two-jet specialization has
93 exactly the left-principal kernel generated by the localized central
94 parameter, stated elementwise in written multiplication order. -/
95 theorem localizedOrderReesTwoJetSpecialization_eq_zero_iff_exists_parameter_mul
96     (x : OreLocalization (OppositeDenominators (filteredQuotientTwoJetTraceData k I) S)
97       ((OrderReesTwoJet (n := n) k)ᵐᵒᵖ)) :
98     localizedOppositeSpecialization (filteredQuotientTwoJetTraceData k I) S x = 0 ↔
99       ∃ y : OreLocalization (OppositeDenominators (filteredQuotientTwoJetTraceData k I) S)
100           ((OrderReesTwoJet (n := n) k)ᵐᵒᵖ),
101         OreLocalization.numeratorHom
102             (MulOpposite.op (orderReesTwoJetParameter (n := n) k)) * y = x := by
103   constructor
104   · intro hx
105     induction x using OreLocalization.ind with
106     | _ b s =>
107       rw [localizedOppositeSpecialization_oreDiv] at hx
108       have hbase :
109           algebraMap (SymbolRing k n) (Localization S)
110             (orderReesTwoJetSpecialization (n := n) k b.unop) = 0 := by
111         exact (Units.isUnit
112           ((localizedDenominatorUnits (filteredQuotientTwoJetTraceData k I) S s)⁻¹)).mul_left_cancel
113             (by rw [mul_zero]; exact hx)
114       obtain ⟨u, hu⟩ :=
115         (IsLocalization.map_eq_zero_iff S (Localization S)
116           (orderReesTwoJetSpecialization (n := n) k b.unop)).mp hbase
117       obtain ⟨a, ha⟩ :=
118         orderReesTwoJetSpecialization_surjective (n := n) k (u : SymbolRing k n)
119       let v : (OrderReesTwoJet (n := n) k)ᵐᵒᵖ := MulOpposite.op a
120       have hvS : v ∈ OppositeDenominators (filteredQuotientTwoJetTraceData k I) S := by
121         change orderReesTwoJetSpecialization (n := n) k a ∈ S
122         rw [ha]
123         exact u.property
124       have hvb_zero :
125           orderReesTwoJetSpecialization (n := n) k (v * b).unop = 0 := by
126         change orderReesTwoJetSpecialization (n := n) k (b.unop * a) = 0
127         rw [map_mul, ha]
128         simpa [mul_comm] using hu
129       obtain ⟨q, hq⟩ :=
130         (orderReesTwoJetSpecialization_eq_zero_iff_exists_parameter_mul
131           (n := n) k (v * b).unop).mp hvb_zero
132       have hfactor :
133           MulOpposite.op (orderReesTwoJetParameter (n := n) k) *
134               MulOpposite.op q = v * b := by
135         apply MulOpposite.unop_injective
136         change q * orderReesTwoJetParameter (n := n) k = b.unop * a
137         rw [← (orderReesTwoJetParameter_mem_center (n := n) k).comm q]
138         exact hq
139       have hvs : v * (s : (OrderReesTwoJet (n := n) k)ᵐᵒᵖ) ∈
140           OppositeDenominators (filteredQuotientTwoJetTraceData k I) S :=
141         mul_mem hvS s.property
142       let t : OppositeDenominators (filteredQuotientTwoJetTraceData k I) S :=
143         ⟨v * s, hvs⟩
144       refine ⟨(MulOpposite.op q) /ₒ t, ?_⟩
145       rw [localized_parameter_mul_oreDiv]
146       rw [hfactor]
147       symm
148       exact OreLocalization.expand b s v hvs
149   · rintro ⟨y, rfl⟩
150     rw [map_mul]
151     change
152       localizedOppositeSpecialization (filteredQuotientTwoJetTraceData k I) S
153           (OreLocalization.numeratorHom
154             (MulOpposite.op (filteredQuotientTwoJetTraceData k I).c)) *
155         localizedOppositeSpecialization (filteredQuotientTwoJetTraceData k I) S y = 0
156     rw [localizedOppositeSpecialization_parameter, zero_mul]
