6 /-!
7 # Principal specialization kernels after the localized two-block quotient
8 
9 Suppose that the kernel of the localized opposite specialization is exactly
10 the left principal ideal generated by the localized square-zero parameter.
11 The kernel of the descended two-block specialization is then exactly the
12 left principal ideal generated by the parameter class in the quotient.
13 
14 The proof retains multiplication order.  An element of the doubled power is
15 lifted as an ordered sum of products `a * b`; those products lie in the
16 two-block ideal.  Only the remaining element of the original specialization
17 kernel is factored by the parameter.
18 
19 No Artinianness, module-action statement, adapted basis, or matrix identity is
20 used here.
21 -/

53 /-- Every element of the doubled power has an ordered lift in the two-block
54 ideal.  Surjectivity is applied separately to the left and right factors, so
55 no commutation in the noncommutative source is used. -/
56 theorem exists_mem_localizedTwoBlockIdeal_map_eq_of_mem_doubledPower
57     (q : ℕ) {r : A₀}
58     (hr : r ∈ localizedDoubledPower S q) :
59     ∃ z : Cₗ,
60       z ∈ localizedTwoBlockIdeal D S q ∧
61         localizedOppositeSpecialization D S z = r := by
62   change r ∈ (𝔪 ^ q) • (𝔪 ^ q) at hr
63   refine Submodule.smul_induction_on hr ?_ ?_
64   · intro a ha b hb
65     obtain ⟨x, hx⟩ :=
66       localizedOppositeSpecialization_surjective D S a
67     obtain ⟨y, hy⟩ :=
68       localizedOppositeSpecialization_surjective D S b
69     refine ⟨x * y, ?_, ?_⟩
70     · apply TwoSidedIdeal.subset_span
71       exact ⟨x,
72         (mem_localizedHighPowerPullback_iff D S q x).2 (hx ▸ ha),
73         y, (mem_localizedHighPowerPullback_iff D S q y).2 (hy ▸ hb), rfl⟩
74     · rw [map_mul, hx, hy]
75       rfl
76   · rintro r₁ r₂ ⟨z₁, hz₁, hmap₁⟩ ⟨z₂, hz₂, hmap₂⟩
77     refine ⟨z₁ + z₂,
78       (localizedTwoBlockIdeal D S q).add_mem hz₁ hz₂, ?_⟩
79     rw [map_add, hmap₁, hmap₂]

81 /-- Written-order left multiplication by the parameter class in the
82 localized two-block quotient. -/
83 def localizedTwoBlockParameterMul (q : ℕ) :
84     LocalizedTwoBlockRing D S q →+
85       LocalizedTwoBlockRing D S q where
86   toFun y :=
87     (localizedTwoBlockIdeal D S q).ringCon.mk'
88         (OreLocalization.numeratorRingHom (MulOpposite.op D.c)) * y
89   map_zero' := mul_zero _
90   map_add' := mul_add _

100 /-- Descent of an exact left-principal specialization kernel through the
101 two-block quotient.  The principal-kernel hypothesis is stated on `Cₗ` in
102 the same written multiplication order as the conclusion. -/
103 theorem localizedTwoBlockSpecialization_eq_zero_iff_exists_parameter_mul
104     (q : ℕ)
105     (hprincipal : ∀ x : Cₗ,
106       localizedOppositeSpecialization D S x = 0 ↔
107         ∃ y : Cₗ,
108           OreLocalization.numeratorRingHom (MulOpposite.op D.c) * y = x)
109     (z : LocalizedTwoBlockRing D S q) :
110     localizedTwoBlockSpecialization D S q z = 0 ↔
111       ∃ y : LocalizedTwoBlockRing D S q,
112         (localizedTwoBlockIdeal D S q).ringCon.mk'
113             (OreLocalization.numeratorRingHom (MulOpposite.op D.c)) * y = z := by
114   induction z using Quotient.inductionOn' with
115   | _ x =>
116     constructor
117     · intro hx
118       change Ideal.Quotient.mk (localizedDoubledPower S q)
119           (localizedOppositeSpecialization D S x) = 0 at hx
120       rw [Ideal.Quotient.eq_zero_iff_mem] at hx
121       obtain ⟨j, hj, hmapj⟩ :=
122         exists_mem_localizedTwoBlockIdeal_map_eq_of_mem_doubledPower
123           D S q hx
124       have hzero :
125           localizedOppositeSpecialization D S (x - j) = 0 := by
126         rw [map_sub, hmapj, sub_self]
127       obtain ⟨y, hy⟩ := (hprincipal (x - j)).mp hzero
128       refine ⟨(localizedTwoBlockIdeal D S q).ringCon.mk' y, ?_⟩
129       have hj0 :
130           (localizedTwoBlockIdeal D S q).ringCon.mk' j = 0 := by
131         apply (TwoSidedIdeal.mem_ker
132           ((localizedTwoBlockIdeal D S q).ringCon.mk')).1
133         simpa using hj
134       calc
135         (localizedTwoBlockIdeal D S q).ringCon.mk'
136               (OreLocalization.numeratorRingHom (MulOpposite.op D.c)) *
137             (localizedTwoBlockIdeal D S q).ringCon.mk' y =
138           (localizedTwoBlockIdeal D S q).ringCon.mk'
139             (OreLocalization.numeratorRingHom (MulOpposite.op D.c) * y) := by
140               exact ((localizedTwoBlockIdeal D S q).ringCon.mk').map_mul _ _ |>.symm
141         _ = (localizedTwoBlockIdeal D S q).ringCon.mk' (x - j) :=
142           congrArg ((localizedTwoBlockIdeal D S q).ringCon.mk') hy
143         _ = (localizedTwoBlockIdeal D S q).ringCon.mk' x -
144             (localizedTwoBlockIdeal D S q).ringCon.mk' j :=
145           ((localizedTwoBlockIdeal D S q).ringCon.mk').map_sub x j
146         _ = (localizedTwoBlockIdeal D S q).ringCon.mk' x := by
147           rw [hj0, sub_zero]
148     · rintro ⟨y, hy⟩
149       rw [← hy, map_mul]
150       have hc0 :
151           localizedOppositeSpecialization D S
152             (OreLocalization.numeratorRingHom (MulOpposite.op D.c)) = 0 :=
153         (hprincipal
154           (OreLocalization.numeratorRingHom (MulOpposite.op D.c))).2
155             ⟨1, mul_one _⟩
156       rw [localizedTwoBlockSpecialization_mk, hc0, map_zero, zero_mul]

158 /-- Kernel/range form of the descended left-principal kernel theorem. -/
159 theorem localizedTwoBlockSpecialization_ker_eq_range_parameterMul
160     (q : ℕ)
161     (hprincipal : ∀ x : Cₗ,
162       localizedOppositeSpecialization D S x = 0 ↔
163         ∃ y : Cₗ,
164           OreLocalization.numeratorRingHom (MulOpposite.op D.c) * y = x) :
165     AddMonoidHom.ker
166         (localizedTwoBlockSpecialization D S q).toAddMonoidHom =
167       AddMonoidHom.range (localizedTwoBlockParameterMul D S q) := by
168   ext z
169   rw [AddMonoidHom.mem_ker, AddMonoidHom.mem_range]
170   exact localizedTwoBlockSpecialization_eq_zero_iff_exists_parameter_mul
171     D S q hprincipal z
