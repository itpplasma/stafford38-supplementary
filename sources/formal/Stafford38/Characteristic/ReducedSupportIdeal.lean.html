9 /-!
10 # The reduced order-characteristic ideal
11 
12 This file defines the radical order-support ideal and its contraction to the
13 base polynomial ring, and proves their basic ideal/support identities. It
14 does not assert Poisson closure. The canonical support proof supplies its
15 involutivity through `GabberGlobalAssembly`.
16 -/

29 variable (k : Type*) [Field k] {n : ℕ}

31 /-- The radical of the differential-order initial ideal. -/
32 def reducedOrderSupportIdeal
33     (I : RightIdeal (WeylAlg k n)) : Ideal (SymbolRing k n) :=
34   (orderInitialIdeal k I).radical
35 
36 /-- The order initial ideal is contained in its reduced support ideal. -/
37 theorem orderInitialIdeal_le_reducedOrderSupportIdeal
38     (I : RightIdeal (WeylAlg k n)) :
39     orderInitialIdeal k I ≤ reducedOrderSupportIdeal k I :=
40   Ideal.le_radical
41 
42 /-- Membership in the reduced support ideal is witnessed by a power in the
43 order initial ideal. -/
44 theorem mem_reducedOrderSupportIdeal_iff
45     (I : RightIdeal (WeylAlg k n)) (P : SymbolRing k n) :
46     P ∈ reducedOrderSupportIdeal k I ↔
47       ∃ m : ℕ, P ^ m ∈ orderInitialIdeal k I :=
48   Iff.rfl
49 
50 /-- The reduced support ideal is radical. -/
51 theorem reducedOrderSupportIdeal_isRadical
52     (I : RightIdeal (WeylAlg k n)) :
53     (reducedOrderSupportIdeal k I).IsRadical :=
54   Ideal.radical_isRadical _
55 
56 /-- Passing to the radical does not change the order characteristic support. -/
57 theorem orderCharacteristicSupport_eq_zeroLocus_reduced
58     (I : RightIdeal (WeylAlg k n)) :
59     orderCharacteristicSupport k I =
60       PrimeSpectrum.zeroLocus (reducedOrderSupportIdeal k I) := by
61   rw [orderCharacteristicSupport_eq_zeroLocus, reducedOrderSupportIdeal,
62     PrimeSpectrum.zeroLocus_radical]
63 
64 /-- The reduced support ideal is intrinsically the vanishing ideal of the
65 order characteristic support. -/
66 theorem vanishingIdeal_orderCharacteristicSupport
67     (I : RightIdeal (WeylAlg k n)) :
68     PrimeSpectrum.vanishingIdeal (orderCharacteristicSupport k I) =
69       reducedOrderSupportIdeal k I := by
70   rw [orderCharacteristicSupport_eq_zeroLocus, reducedOrderSupportIdeal,
71     PrimeSpectrum.vanishingIdeal_zeroLocus_eq_radical]
72 
73 /-- The contraction of the reduced support ideal along the base-coordinate
74 embedding.  This is the precise ideal denoted informally by
75 `E ∩ k[x₁, …, xₙ]`. -/
76 def reducedOrderBaseIdeal
77     (I : RightIdeal (WeylAlg k n)) : Ideal (MvPolynomial (Fin n) k) :=
78   (reducedOrderSupportIdeal k I).comap (baseLift :
79     MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom
80 
81 /-- The base-coordinate lift really is an embedding, so the contraction above
82 is an honest intersection with the base polynomial subring. -/
83 theorem baseLift_injective :
84     Function.Injective
85       (baseLift : MvPolynomial (Fin n) k → SymbolRing k n) :=
86   MvPolynomial.rename_injective Sum.inl Sum.inl_injective
87 
88 /-- Base-ideal membership is exactly membership of the lifted polynomial in
89 the reduced support ideal. -/
90 theorem mem_reducedOrderBaseIdeal_iff
91     (I : RightIdeal (WeylAlg k n)) (f : MvPolynomial (Fin n) k) :
92     f ∈ reducedOrderBaseIdeal k I ↔
93       baseLift f ∈ reducedOrderSupportIdeal k I :=
94   Iff.rfl
95 
96 /-- Contracting after reduction is the same as reducing the contracted order
97 initial ideal. -/
98 theorem reducedOrderBaseIdeal_eq_radical_comap
99     (I : RightIdeal (WeylAlg k n)) :
100     reducedOrderBaseIdeal k I =
101       ((orderInitialIdeal k I).comap (baseLift :
102         MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom).radical := by
103   simpa [reducedOrderBaseIdeal, reducedOrderSupportIdeal] using
104     Ideal.comap_radical
105       ((baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom)
106       (orderInitialIdeal k I)
107 
108 /-- The contracted base ideal is radical. -/
109 theorem reducedOrderBaseIdeal_isRadical
110     (I : RightIdeal (WeylAlg k n)) :
111     (reducedOrderBaseIdeal k I).IsRadical := by
112   rw [reducedOrderBaseIdeal_eq_radical_comap]
113   exact Ideal.radical_isRadical _
114 
115 /-- The base contraction may be recovered directly from the characteristic
116 support, without reference to a chosen presentation of the radical. -/
117 theorem reducedOrderBaseIdeal_eq_comap_vanishingIdeal
118     (I : RightIdeal (WeylAlg k n)) :
119     reducedOrderBaseIdeal k I =
120       (PrimeSpectrum.vanishingIdeal (orderCharacteristicSupport k I)).comap
121         (baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom := by
122   rw [vanishingIdeal_orderCharacteristicSupport]
123   rfl
124 
125 /-- A base polynomial lies in the contracted reduced ideal exactly when its
126 lift vanishes at every prime of the order characteristic support. -/
127 theorem mem_reducedOrderBaseIdeal_iff_forall_support
128     (I : RightIdeal (WeylAlg k n)) (f : MvPolynomial (Fin n) k) :
129     f ∈ reducedOrderBaseIdeal k I ↔
130       ∀ p ∈ orderCharacteristicSupport k I, baseLift f ∈ p.asIdeal := by
131   rw [reducedOrderBaseIdeal_eq_comap_vanishingIdeal, Ideal.mem_comap,
132     PrimeSpectrum.mem_vanishingIdeal]
133   rfl
134 
135 /-- The remaining Gabber condition for the reduced characteristic ideal,
136 written entirely in terms of its base contraction.  This is a reformulation,
137 not a proof, of base-relative Poisson closure. -/
138 theorem isBaseRelativePoisson_reduced_iff
139     (I : RightIdeal (WeylAlg k n)) :
140     IsBaseRelativePoisson (reducedOrderSupportIdeal k I) ↔
141       ∀ f ∈ reducedOrderBaseIdeal k I,
142         ∀ g ∈ reducedOrderSupportIdeal k I,
143           poissonBracket (baseLift f) g ∈ reducedOrderSupportIdeal k I := by
144   rfl
