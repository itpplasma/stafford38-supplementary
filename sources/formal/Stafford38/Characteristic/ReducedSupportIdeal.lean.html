7 /-!
8 # The reduced order-characteristic ideal
9 
10 This file defines the radical order-support ideal and its contraction to the
11 base polynomial ring, and proves their basic ideal/support identities. It
12 does not assert Poisson closure. The canonical support proof supplies its
13 involutivity through `GabberGlobalAssembly`.
14 -/

27 variable (k : Type*) [Field k] {n : ℕ}

29 /-- The radical of the differential-order initial ideal. -/
30 def reducedOrderSupportIdeal
31     (I : RightIdeal (PresentedWeyl k n)) : Ideal (SymbolRing k n) :=
32   (orderInitialIdeal k I).radical
33 
34 /-- The order initial ideal is contained in its reduced support ideal. -/
35 theorem orderInitialIdeal_le_reducedOrderSupportIdeal
36     (I : RightIdeal (PresentedWeyl k n)) :
37     orderInitialIdeal k I ≤ reducedOrderSupportIdeal k I :=
38   Ideal.le_radical
39 
40 /-- Membership in the reduced support ideal is witnessed by a power in the
41 order initial ideal. -/
42 theorem mem_reducedOrderSupportIdeal_iff
43     (I : RightIdeal (PresentedWeyl k n)) (P : SymbolRing k n) :
44     P ∈ reducedOrderSupportIdeal k I ↔
45       ∃ m : ℕ, P ^ m ∈ orderInitialIdeal k I :=
46   Iff.rfl
47 
48 /-- The reduced support ideal is radical. -/
49 theorem reducedOrderSupportIdeal_isRadical
50     (I : RightIdeal (PresentedWeyl k n)) :
51     (reducedOrderSupportIdeal k I).IsRadical :=
52   Ideal.radical_isRadical _
53 
54 /-- Passing to the radical does not change the order characteristic support. -/
55 theorem orderCharacteristicSupport_eq_zeroLocus_reduced
56     (I : RightIdeal (PresentedWeyl k n)) :
57     orderCharacteristicSupport k I =
58       PrimeSpectrum.zeroLocus (reducedOrderSupportIdeal k I) := by
59   rw [orderCharacteristicSupport_eq_zeroLocus, reducedOrderSupportIdeal,
60     PrimeSpectrum.zeroLocus_radical]
61 
62 /-- The reduced support ideal is intrinsically the vanishing ideal of the
63 order characteristic support. -/
64 theorem vanishingIdeal_orderCharacteristicSupport
65     (I : RightIdeal (PresentedWeyl k n)) :
66     PrimeSpectrum.vanishingIdeal (orderCharacteristicSupport k I) =
67       reducedOrderSupportIdeal k I := by
68   rw [orderCharacteristicSupport_eq_zeroLocus, reducedOrderSupportIdeal,
69     PrimeSpectrum.vanishingIdeal_zeroLocus_eq_radical]
70 
71 /-- The contraction of the reduced support ideal along the base-coordinate
72 embedding.  This is the precise ideal denoted informally by
73 `E ∩ k[x₁, …, xₙ]`. -/
74 def reducedOrderBaseIdeal
75     (I : RightIdeal (PresentedWeyl k n)) : Ideal (MvPolynomial (Fin n) k) :=
76   (reducedOrderSupportIdeal k I).comap (baseLift :
77     MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom
78 
79 /-- The base-coordinate lift really is an embedding, so the contraction above
80 is an honest intersection with the base polynomial subring. -/
81 theorem baseLift_injective :
82     Function.Injective
83       (baseLift : MvPolynomial (Fin n) k → SymbolRing k n) :=
84   MvPolynomial.rename_injective Sum.inl Sum.inl_injective
85 
86 /-- Base-ideal membership is exactly membership of the lifted polynomial in
87 the reduced support ideal. -/
88 theorem mem_reducedOrderBaseIdeal_iff
89     (I : RightIdeal (PresentedWeyl k n)) (f : MvPolynomial (Fin n) k) :
90     f ∈ reducedOrderBaseIdeal k I ↔
91       baseLift f ∈ reducedOrderSupportIdeal k I :=
92   Iff.rfl
93 
94 /-- Contracting after reduction is the same as reducing the contracted order
95 initial ideal. -/
96 theorem reducedOrderBaseIdeal_eq_radical_comap
97     (I : RightIdeal (PresentedWeyl k n)) :
98     reducedOrderBaseIdeal k I =
99       ((orderInitialIdeal k I).comap (baseLift :
100         MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom).radical := by
101   simpa [reducedOrderBaseIdeal, reducedOrderSupportIdeal] using
102     Ideal.comap_radical
103       ((baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom)
104       (orderInitialIdeal k I)
105 
106 /-- The contracted base ideal is radical. -/
107 theorem reducedOrderBaseIdeal_isRadical
108     (I : RightIdeal (PresentedWeyl k n)) :
109     (reducedOrderBaseIdeal k I).IsRadical := by
110   rw [reducedOrderBaseIdeal_eq_radical_comap]
111   exact Ideal.radical_isRadical _
112 
113 /-- The base contraction may be recovered directly from the characteristic
114 support, without reference to a chosen presentation of the radical. -/
115 theorem reducedOrderBaseIdeal_eq_comap_vanishingIdeal
116     (I : RightIdeal (PresentedWeyl k n)) :
117     reducedOrderBaseIdeal k I =
118       (PrimeSpectrum.vanishingIdeal (orderCharacteristicSupport k I)).comap
119         (baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom := by
120   rw [vanishingIdeal_orderCharacteristicSupport]
121   rfl
122 
123 /-- A base polynomial lies in the contracted reduced ideal exactly when its
124 lift vanishes at every prime of the order characteristic support. -/
125 theorem mem_reducedOrderBaseIdeal_iff_forall_support
126     (I : RightIdeal (PresentedWeyl k n)) (f : MvPolynomial (Fin n) k) :
127     f ∈ reducedOrderBaseIdeal k I ↔
128       ∀ p ∈ orderCharacteristicSupport k I, baseLift f ∈ p.asIdeal := by
129   rw [reducedOrderBaseIdeal_eq_comap_vanishingIdeal, Ideal.mem_comap,
130     PrimeSpectrum.mem_vanishingIdeal]
131   rfl
132 
133 /-- The remaining Gabber condition for the reduced characteristic ideal,
134 written entirely in terms of its base contraction.  This is a reformulation,
135 not a proof, of base-relative Poisson closure. -/
136 theorem isBaseRelativePoisson_reduced_iff
137     (I : RightIdeal (PresentedWeyl k n)) :
138     IsBaseRelativePoisson (reducedOrderSupportIdeal k I) ↔
139       ∀ f ∈ reducedOrderBaseIdeal k I,
140         ∀ g ∈ reducedOrderSupportIdeal k I,
141           poissonBracket (baseLift f) g ∈ reducedOrderSupportIdeal k I := by
142   rfl
