13 /-!
14 # Localization preserves the square-zero deformation sequence
15 
16 Mathlib constructs the localization of a left module over a left Ore set as
17 `OreLocalization S N`.  Applied to the opposite deformation ring, this is the
18 localization of the original right module.  This file proves directly on Ore
19 fractions that the square-zero parameter remains exact.  It also constructs
20 the specialization map to the ordinary commutative localization of the
21 special fibre and proves that its kernel is the same parameter image.
22 
23 The results are not conditional localized-exactness interfaces: the localized
24 module, parameter action, and specialization map are concrete constructions.
25 The only typeclass parameter is Mathlib's `OreSet`; its existence for the
26 pulled-back denominators is proved in `SquareZeroOreLocalization`.
27 -/

48 /-- Denominators in the opposite deformation ring lying over `S`. -/
49 abbrev OppositeDenominators (S : Submonoid R) : Submonoid Bᵐᵒᵖ :=
50   S.comap (oppositeSpecializationMonoidHom D)

52 /-- The concrete localized opposite-ring module supplied by Mathlib's Ore
53 localization construction. -/
54 abbrev LocalizedDeformationModule (S : Submonoid R)
55     [OreLocalization.OreSet (OppositeDenominators D S)] :=
56   OreLocalization (OppositeDenominators D S) N

58 /-- A pulled-back Ore denominator specializes to a denominator in `S`. -/
59 def denominatorMap (S : Submonoid R) :
60     OppositeDenominators D S →* S where
61   toFun s := ⟨oppositeSpecializationMonoidHom D s, s.property⟩
62   map_one' := by ext; simp
63   map_mul' a b := by ext; simp

70 /-- Action of the square-zero parameter on the localized module.  It is the
71 original opposite-ring action, extended by the Ore localization module
72 construction. -/
73 def localizedCAct :
74     LocalizedDeformationModule D S →+ LocalizedDeformationModule D S where
75   toFun x := MulOpposite.op D.c • x
76   map_zero' := smul_zero _
77   map_add' x y := smul_add _ x y
78 
79 @[simp] theorem localizedCAct_apply
80     (x : LocalizedDeformationModule D S) :
81     localizedCAct D S x = MulOpposite.op D.c • x :=
82   rfl

84 /-- On a displayed Ore fraction, the localized parameter acts on its
85 numerator without changing its denominator. -/
86 theorem localizedCAct_oreDiv
87     (m : N) (s : OppositeDenominators D S) :
88     localizedCAct D S (m /ₒ s) =
89       (MulOpposite.op D.c • m) /ₒ s := by
90   change MulOpposite.op D.c • (m /ₒ s) = _
91   rw [← OreLocalization.oreDiv_one_smul]
92   simpa using OreLocalization.oreDiv_smul_char
93     (MulOpposite.op D.c) m 1 s (MulOpposite.op D.c) s
94       (show (s : Bᵐᵒᵖ) * MulOpposite.op D.c =
95         MulOpposite.op D.c * s from by
96           apply MulOpposite.unop_injective
97           exact D.c_center.comm s.val.unop)
98 
99 /-- The localized parameter still squares to zero. -/
100 theorem localizedCAct_sq_eq_zero
101     (x : LocalizedDeformationModule D S) :
102     localizedCAct D S (localizedCAct D S x) = 0 := by
103   change MulOpposite.op D.c • (MulOpposite.op D.c • x) = 0
104   rw [← mul_smul]
105   have hc : MulOpposite.op D.c * MulOpposite.op D.c = 0 := by
106     apply MulOpposite.unop_injective
107     simpa [pow_two] using D.c_sq
108   rw [hc, zero_smul]

110 /-- Ore localization preserves the exact square-zero parameter sequence:
111 the kernel of multiplication by `c` remains its image. -/
112 theorem localizedCAct_ker_eq_range :
113     AddMonoidHom.ker (localizedCAct D S) =
114       AddMonoidHom.range (localizedCAct D S) := by
115   apply le_antisymm
116   · intro x hx
117     induction x using OreLocalization.ind with
118     | _ m s =>
119       rw [AddMonoidHom.mem_ker, localizedCAct_oreDiv] at hx
120       have hfrac : (MulOpposite.op D.c • m) /ₒ s =
121           (0 : N) /ₒ (1 : OppositeDenominators D S) := by
122         simpa using hx
123       obtain ⟨u, v, hv, huv⟩ :=
124         OreLocalization.oreDiv_eq_iff.mp hfrac
125       have hvc : v • (MulOpposite.op D.c • m) = 0 := by
126         simpa using hv.symm
127       have hcm : MulOpposite.op D.c • (v • m) = 0 := by
128         rw [← mul_smul]
129         rw [show MulOpposite.op D.c * v = v * MulOpposite.op D.c by
130           apply MulOpposite.unop_injective
131           exact (D.c_center.comm v.unop).symm]
132         simpa [mul_smul] using hvc
133       have hker : v • m ∈ LinearMap.ker D.cAct := by
134         rw [LinearMap.mem_ker, D.cAct_apply]
135         exact hcm
136       rw [D.c_exact] at hker
137       obtain ⟨q, hq⟩ := hker
138       have hq' : MulOpposite.op D.c • q = v • m := by
139         simpa [D.cAct_apply] using hq
140       have hvs : v * (s : Bᵐᵒᵖ) ∈ OppositeDenominators D S := by
141         rw [← huv]
142         have hu1 : (u : Bᵐᵒᵖ) *
143             ((1 : OppositeDenominators D S) : Bᵐᵒᵖ) = u := by simp
144         rw [hu1]
145         exact u.property
146       let t : OppositeDenominators D S := ⟨v * s, hvs⟩
147       refine ⟨q /ₒ t, ?_⟩
148       rw [localizedCAct_oreDiv, hq']
149       calc
150         (v • m) /ₒ t = m /ₒ s := by
151           symm
152           exact OreLocalization.expand m s v hvs
153         _ = m /ₒ s := rfl
154   · intro x hx
155     obtain ⟨y, rfl⟩ := hx
156     rw [AddMonoidHom.mem_ker]
157     exact localizedCAct_sq_eq_zero D S y

159 /-- Specialization of an Ore-localized deformation fraction to the ordinary
160 commutative localization of the special fibre. -/
161 def localizedSpecializationFun :
162     LocalizedDeformationModule D S → LocalizedModule S G :=
163   OreLocalization.liftExpand
164     (fun m s => LocalizedModule.mk (D.rho m) (denominatorMap D S s))
165     (by
166       intro m t s hts
167       apply LocalizedModule.mk_eq.mpr
168       refine ⟨1, ?_⟩
169       simp only [one_smul]
170       have hrho : D.rho (t • m) = D.pi t.unop • D.rho m := by
171         simpa using D.rho_action t.unop m
172       rw [hrho]
173       change
174         (oppositeSpecializationMonoidHom D
175             (⟨t * s, hts⟩ : OppositeDenominators D S)) • D.rho m =
176           (denominatorMap D S s : R) •
177             (D.pi t.unop • D.rho m)
178       simp only [map_mul, MulOpposite.unop_mul]
179       rw [mul_smul]
180       change D.pi t.unop • D.pi s.val.unop • D.rho m =
181         D.pi s.val.unop • D.pi t.unop • D.rho m
182       rw [smul_comm])
183 
184 @[simp] theorem localizedSpecializationFun_oreDiv
185     (m : N) (s : OppositeDenominators D S) :
186     localizedSpecializationFun D S (m /ₒ s) =
187       LocalizedModule.mk (D.rho m) (denominatorMap D S s) :=
188   rfl

190 /-- The localized specialization is an additive homomorphism. -/
191 def localizedSpecialization :
192     LocalizedDeformationModule D S →+ LocalizedModule S G where
193   toFun := localizedSpecializationFun D S
194   map_zero' := by
195     change localizedSpecializationFun D S 0 = 0
196     rw [← OreLocalization.zero_oreDiv
197       (1 : OppositeDenominators D S)]
198     rw [localizedSpecializationFun_oreDiv]
199     simp
200   map_add' x y := by
201     induction x using OreLocalization.ind with
202     | _ m s =>
203       induction y using OreLocalization.ind with
204       | _ n t =>
205         rw [OreLocalization.oreDiv_add_oreDiv]
206         simp only [localizedSpecializationFun_oreDiv]
207         rw [LocalizedModule.mk_add_mk]
208         apply LocalizedModule.mk_eq.mpr
209         refine ⟨1, ?_⟩
210         simp only [one_smul]
211         rw [map_add]
212         have hm : D.rho (OreLocalization.oreDenom (s : Bᵐᵒᵖ) t • m) =
213             D.pi (OreLocalization.oreDenom (s : Bᵐᵒᵖ) t).val.unop •
214               D.rho m := by
215           have h := D.rho_action
216             (OreLocalization.oreDenom (s : Bᵐᵒᵖ) t).val.unop m
217           rwa [MulOpposite.op_unop] at h
218         have hn : D.rho (OreLocalization.oreNum (s : Bᵐᵒᵖ) t • n) =
219             D.pi (OreLocalization.oreNum (s : Bᵐᵒᵖ) t).unop •
220               D.rho n := by
221           simpa using D.rho_action
222             (OreLocalization.oreNum (s : Bᵐᵒᵖ) t).unop n
223         rw [hm, hn]
224         have hore := OreLocalization.ore_eq (s : Bᵐᵒᵖ) t
225         have hab := congrArg (oppositeSpecializationMonoidHom D) hore
226         simp only [map_mul] at hab ⊢
227         have hab' :
228             (denominatorMap D S
229                 (OreLocalization.oreDenom (s : Bᵐᵒᵖ) t) : R) *
230                 (denominatorMap D S s : R) =
231               D.pi (OreLocalization.oreNum (s : Bᵐᵒᵖ) t).unop *
232                 (denominatorMap D S t : R) := by
233           exact hab
234         simp only [smul_add, Submonoid.smul_def, Submonoid.coe_mul,
235           ← mul_smul]
236         congr 1
237         · congr 1
238           rw [show D.pi
239               (OreLocalization.oreDenom (s : Bᵐᵒᵖ) t).val.unop =
240                 (denominatorMap D S
241                   (OreLocalization.oreDenom (s : Bᵐᵒᵖ) t) : R) from rfl]
242           ring
243         · congr 1
244           calc
245             (denominatorMap D S s : R) *
246                   (denominatorMap D S t : R) *
247                   D.pi (OreLocalization.oreNum (s : Bᵐᵒᵖ) t).unop =
248                 (D.pi (OreLocalization.oreNum (s : Bᵐᵒᵖ) t).unop *
249                   (denominatorMap D S t : R)) *
250                   (denominatorMap D S s : R) := by ring
251             _ = ((denominatorMap D S
252                     (OreLocalization.oreDenom (s : Bᵐᵒᵖ) t) : R) *
253                   (denominatorMap D S s : R)) *
254                   (denominatorMap D S s : R) := by rw [← hab']

261 /-- Specialization remains surjective after localization.  A commutative
262 denominator is lifted through the surjective deformation specialization. -/
263 theorem localizedSpecialization_surjective :
264     Function.Surjective (localizedSpecialization D S) := by
265   intro y
266   induction y using LocalizedModule.induction_on with
267   | _ g s =>
268     obtain ⟨m, hm⟩ := D.rho_surjective g
269     obtain ⟨b, hb⟩ := D.pi_surjective (s : R)
270     have hbS : oppositeSpecializationMonoidHom D (MulOpposite.op b) ∈ S := by
271       change D.pi b ∈ S
272       rw [hb]
273       exact s.property
274     let t : OppositeDenominators D S := ⟨MulOpposite.op b, hbS⟩
275     have hden : denominatorMap D S t = s := by
276       apply Subtype.ext
277       exact hb
278     refine ⟨m /ₒ t, ?_⟩
279     rw [localizedSpecialization_apply,
280       localizedSpecializationFun_oreDiv, hm, hden]

282 /-- The localized specialization kills the localized parameter image. -/
283 theorem localizedSpecialization_localizedCAct_eq_zero
284     (x : LocalizedDeformationModule D S) :
285     localizedSpecialization D S (localizedCAct D S x) = 0 := by
286   induction x using OreLocalization.ind with
287   | _ m s =>
288     rw [localizedCAct_oreDiv, localizedSpecialization_apply,
289       localizedSpecializationFun_oreDiv]
290     have hrho : D.rho (MulOpposite.op D.c • m) = 0 := by
291       rw [D.rho_action, D.pi_c, zero_smul]
292     rw [hrho, LocalizedModule.zero_mk]

294 /-- The specialization kernel after Ore localization is exactly the image of
295 the localized square-zero parameter. -/
296 theorem localizedSpecialization_ker_eq_range :
297     AddMonoidHom.ker (localizedSpecialization D S) =
298       AddMonoidHom.range (localizedCAct D S) := by
299   apply le_antisymm
300   · intro x hx
301     induction x using OreLocalization.ind with
302     | _ m s =>
303       rw [AddMonoidHom.mem_ker, localizedSpecialization_apply,
304         localizedSpecializationFun_oreDiv] at hx
305       have hzero :
306           LocalizedModule.mk (D.rho m) (denominatorMap D S s) =
307             LocalizedModule.mk (0 : G) (1 : S) := by
308         simpa using hx
309       obtain ⟨u, hu⟩ := LocalizedModule.mk_eq.mp hzero
310       have hurho' : u • D.rho m = 0 := by
311         simpa using hu
312       have hurho : (u : R) • D.rho m = 0 := hurho'
313       obtain ⟨b, hb⟩ := D.pi_surjective (u : R)
314       let v : Bᵐᵒᵖ := MulOpposite.op b
315       have hvrho : D.rho (v • m) = 0 := by
316         have h := D.rho_action b m
317         change D.rho (v • m) = D.pi b • D.rho m at h
318         rw [h, hb]
319         exact hurho
320       have hvker : v • m ∈ LinearMap.ker D.rho := by
321         exact LinearMap.mem_ker.mpr hvrho
322       rw [D.rho_ker] at hvker
323       obtain ⟨q, hq⟩ := hvker
324       have hq' : MulOpposite.op D.c • q = v • m := by
325         simpa [D.cAct_apply] using hq
326       have hvS : v ∈ OppositeDenominators D S := by
327         change D.pi b ∈ S
328         rw [hb]
329         exact u.property
330       have hvs : v * (s : Bᵐᵒᵖ) ∈ OppositeDenominators D S :=
331         mul_mem hvS s.property
332       let t : OppositeDenominators D S := ⟨v * s, hvs⟩
333       refine ⟨q /ₒ t, ?_⟩
334       rw [localizedCAct_oreDiv, hq']
335       symm
336       exact OreLocalization.expand m s v hvs
337   · intro x hx
338     obtain ⟨y, rfl⟩ := hx
339     rw [AddMonoidHom.mem_ker]
340     exact localizedSpecialization_localizedCAct_eq_zero D S y

342 /-- Quotienting the localized deformation module by the parameter image gives
343 the ordinary commutative localization of the special fibre. -/
344 noncomputable def localizedSpecialFibreAddEquiv :
345     (LocalizedDeformationModule D S ⧸
346         AddMonoidHom.range (localizedCAct D S)) ≃+
347       LocalizedModule S G :=
348   (QuotientAddGroup.quotientAddEquivOfEq
349       (localizedSpecialization_ker_eq_range D S).symm).trans
350     (QuotientAddGroup.quotientKerEquivOfSurjective
351       (localizedSpecialization D S)
352       (localizedSpecialization_surjective D S))

354 /-- The complete localized exactness package needed by the local trace
355 argument: both parameter exactness and specialization exactness hold, and the
356 specialization is onto the ordinary localized special fibre. -/
357 theorem localized_squareZero_exactness :
358     AddMonoidHom.ker (localizedCAct D S) =
359         AddMonoidHom.range (localizedCAct D S) ∧
360       Function.Surjective (localizedSpecialization D S) ∧
361       AddMonoidHom.ker (localizedSpecialization D S) =
362         AddMonoidHom.range (localizedCAct D S) :=
363   ⟨localizedCAct_ker_eq_range D S,
364     localizedSpecialization_surjective D S,
365     localizedSpecialization_ker_eq_range D S⟩

376 /-- Exactness for one explicit choice of Mathlib's pulled-back Ore-set
377 structure.  This packages only already constructed maps and equations. -/
378 def LocalizedExactnessFor (S : Submonoid R)
379     (h : OreLocalization.OreSet (OppositeDenominators D S)) : Prop :=
380   letI := h
381   AddMonoidHom.ker (localizedCAct D S) =
382       AddMonoidHom.range (localizedCAct D S) ∧
383     Function.Surjective (localizedSpecialization D S) ∧
384     AddMonoidHom.ker (localizedSpecialization D S) =
385       AddMonoidHom.range (localizedCAct D S)

387 /-- The Ore-set existence theorem and the fraction-level exactness proof
388 together produce a genuine localized deformation sequence for every
389 multiplicative set in the special fibre. -/
390 theorem exists_localizedExactnessFor (S : Submonoid R) :
391     ∃ h : OreLocalization.OreSet (OppositeDenominators D S),
392       LocalizedExactnessFor D S h := by
393   obtain ⟨h⟩ := nonempty_oreSet_comap_op D S
394   refine ⟨h, ?_⟩
395   letI := h
396   exact localized_squareZero_exactness D S

398 /-- At a minimal prime over the special-fibre annihilator, the localized
399 deformation exact sequence exists and its commutative special fibre is
400 simultaneously nonzero and of finite length. -/
401 theorem exists_minimalPrimeLocalizedExactnessAndFiniteLength
402     [IsNoetherianRing R] [Module.Finite R G]
403     (P : Ideal R) [P.IsPrime]
404     (hP : P ∈ (Module.annihilator R G).minimalPrimes) :
405     ∃ h : OreLocalization.OreSet
406         (OppositeDenominators D P.primeCompl),
407       LocalizedExactnessFor D P.primeCompl h ∧
408         Nontrivial (LocalizedModule P.primeCompl G) ∧
409         IsFiniteLength (Localization P.primeCompl)
410           (LocalizedModule P.primeCompl G) := by
411   obtain ⟨h, hexact⟩ := exists_localizedExactnessFor D P.primeCompl
412   refine ⟨h, hexact, ?_⟩
413   exact Stafford38.Characteristic.MinimalPrimeFiniteLengthLocalization.localizedModule_nontrivial_and_isFiniteLength
