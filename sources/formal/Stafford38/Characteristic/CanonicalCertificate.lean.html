8 /-!
9 # Certificate extraction from empty canonical characteristic support
10 
11 This file closes the algebraic end of the characteristic-support route.  If
12 the order-characteristic support of the literal canonical quotient is empty,
13 then that quotient vanishes and membership of `1` in its two-generator right
14 ideal yields the exact fixed-source Stafford certificate.
15 
16 No theorem proving that the support is empty is assumed or supplied here.
17 -/

33 /-- Empty order-characteristic support of the canonical right ideal gives the
34 literal fixed-source certificate with source `x^N`. -/
35 theorem exists_fixedSource_certificate_of_orderCharacteristicSupport_eq_empty
36     (n N : ℕ) (d : WeylAlg k (n + 1))
37     (hsupport : orderCharacteristicSupport k
38       (canonicalRightIdeal (presentedCoordinate k n) d N) = ∅) :
39     ∃ R S : WeylAlg k (n + 1),
40       (1 : WeylAlg k (n + 1)) =
41         d * R + (presentedCoordinate k n) ^ N * d * S := by
42   let I := canonicalRightIdeal (presentedCoordinate k n) d N
43   have hquotient : Subsingleton (RightQuotient I) :=
44     rightQuotient_subsingleton_of_orderCharacteristicSupport_eq_empty
45       k I hsupport
46   have htop : I = ⊤ :=
47     Submodule.Quotient.subsingleton_iff.mp hquotient
48   have hone : (1 : WeylAlg k (n + 1)) ∈ I := by
49     rw [htop]
50     trivial
51   change (1 : WeylAlg k (n + 1)) ∈
52     Submodule.span (WeylAlg k (n + 1))ᵐᵒᵖ
53       ({d, (presentedCoordinate k n) ^ N * d} :
54         Set (WeylAlg k (n + 1))) at hone
55   rcases Submodule.mem_span_pair.mp hone with ⟨r, s, hrs⟩
56   refine ⟨r.unop, s.unop, ?_⟩
57   simpa [op_smul_eq_mul] using hrs.symm
58

61 end
