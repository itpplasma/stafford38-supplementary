27 abbrev normalCoeffRing (n : ℕ) :=
28   MvPolynomial {v : PhaseVar (n + 1) // v ≠ Sum.inr (0 : Fin (n + 1))} k

30 def normalPolynomialActionHom (n : ℕ) :
31     Polynomial (normalCoeffRing (k := k) n) →+* SymbolRing k (n + 1) :=
32   (normalSymbolAlgEquiv (k := k) n).symm.toRingHom
33 
34 /-- The polynomial-ring action obtained by transporting the symbol action
35 across the normal-variable equivalence. -/
36 @[instance_reducible] def normalPolynomialModule (n : ℕ) (E : Type*) [AddCommGroup E]
37     [Module (SymbolRing k (n + 1)) E] :
38     Module (Polynomial (normalCoeffRing (k := k) n)) E :=
39   Module.compHom E (normalPolynomialActionHom (k := k) n)
40 
41 /-- The corresponding action of the ring of all non-normal symbols. -/
42 @[instance_reducible] def normalCoeffModule (n : ℕ) (E : Type*) [AddCommGroup E]
43     [Module (SymbolRing k (n + 1)) E] : Module (normalCoeffRing (k := k) n) E :=
44   Module.compHom E
45     ((normalPolynomialActionHom (k := k) n).comp Polynomial.C)

47 /-- The actual canonical order-associated graded module is finite over the
48 polynomial ring in all symbols except the distinguished normal covariable. -/
49 theorem canonical_orderAssociatedGradedModule_finite_normalCoeffRing
50     {n N : ℕ} {d : WeylAlg k (n + 1)}
51     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
52     @Module.Finite (normalCoeffRing (k := k) n)
53       (OrderAssociatedGradedModule k
54         (canonicalRightIdeal (presentedCoordinate k n) d N))
55       _ _ (normalCoeffModule (k := k) n _) := by
56   let R := normalCoeffRing (k := k) n
57   let E := OrderAssociatedGradedModule k
58     (canonicalRightIdeal (presentedCoordinate k n) d N)
59   let e := normalSymbolAlgEquiv (k := k) n
60   let g : Polynomial R := canonicalNormalPolynomial (k := k) (n := n) (N := N) d
61   letI : Module (Polynomial R) E := normalPolynomialModule (k := k) n E
62   letI : Module R E := normalCoeffModule (k := k) n E
63   letI : IsScalarTower R (Polynomial R) E :=
64     ⟨by
65       intro r p z
66       change (e.symm (r • p)) • z =
67         e.symm (Polynomial.C r) • e.symm p • z
68       rw [Polynomial.smul_eq_C_mul, map_mul, mul_smul]⟩
69   let generator : E := orderAssociatedGradedGenerator k
70     (canonicalRightIdeal (presentedCoordinate k n) d N)
71   let generatorMap : Polynomial R →ₗ[Polynomial R] E :=
72     LinearMap.toSpanSingleton (Polynomial R) E generator
73   have hsurj : Function.Surjective generatorMap := by
74     intro z
75     obtain ⟨P, hP⟩ := exists_smul_orderAssociatedGradedGenerator k
76       (canonicalRightIdeal (presentedCoordinate k n) d N) z
77     refine ⟨e P, ?_⟩
78     change e.symm (e P) • generator = z
79     rw [e.symm_apply_apply]
80     exact hP
81   letI : Module.Finite (Polynomial R) E :=
82     Module.Finite.of_surjective generatorMap hsurj
83   apply finite_of_monic_annihilator g
84   · exact canonicalNormalPolynomial_monic hd
85   · intro z
86     change e.symm g • z = 0
87     apply Module.mem_annihilator.mp
88     rw [annihilator_orderAssociatedGradedModule]
89     change (normalSymbolAlgEquiv (k := k) n).symm
90       (canonicalNormalPolynomial (k := k) (n := n) (N := N) d) ∈ _
91     rw [canonicalNormalPolynomial,
92       (normalSymbolAlgEquiv (k := k) n).symm_apply_apply]
93     exact canonical_orderPrincipalComponent_mem_initialIdeal k n N hd
