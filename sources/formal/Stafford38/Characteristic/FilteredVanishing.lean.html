8 /-!
9 # Vanishing reflected by an exhaustive filtered quotient
10 
11 For the actual quotient `A / I` with its induced differential-order
12 filtration, vanishing of every associated graded piece forces the quotient
13 itself to vanish.  The proof is internal to the filtration: degree zero uses
14 that its strict-lower piece is zero, and the successor step lowers the degree
15 by one.  Exhaustivity of the PBW order filtration then reaches every quotient
16 class.
17 -/

34 /-- If every associated graded piece of the induced order filtration is
35 subsingleton, then every filtration piece is zero.  The induction explicitly
36 separates the degree-zero and successor cases. -/
37 theorem quotientOrderPiece_eq_bot_of_all_graded_subsingleton
38     (I : RightIdeal (WeylAlg k n))
39     (hgraded : ∀ N, Subsingleton (QuotientOrderGradedPiece k I N)) :
40     ∀ N, quotientOrderPiece k I N = ⊥ := by
41   intro N
42   induction N with
43   | zero =>
44       apply le_antisymm
45       · intro q hq
46         let qN : quotientOrderPiece k I 0 := ⟨q, hq⟩
47         have hmk :
48             (Submodule.Quotient.mk qN : QuotientOrderGradedPiece k I 0) = 0 :=
49           Subsingleton.elim _ _
50         have hlower : qN ∈
51             (quotientOrderStrictLowerPiece k I 0).comap
52               (quotientOrderPiece k I 0).subtype :=
53           (Submodule.Quotient.mk_eq_zero _).mp hmk
54         change q ∈ quotientOrderStrictLowerPiece k I 0 at hlower
55         simpa [quotientOrderStrictLowerPiece, presentedStrictLowerPiece] using hlower
56       · exact bot_le
57   | succ N ih =>
58       apply le_antisymm
59       · intro q hq
60         let qN : quotientOrderPiece k I (N + 1) := ⟨q, hq⟩
61         have hmk :
62             (Submodule.Quotient.mk qN :
63               QuotientOrderGradedPiece k I (N + 1)) = 0 :=
64           Subsingleton.elim _ _
65         have hlower : qN ∈
66             (quotientOrderStrictLowerPiece k I (N + 1)).comap
67               (quotientOrderPiece k I (N + 1)).subtype :=
68           (Submodule.Quotient.mk_eq_zero _).mp hmk
69         change q ∈ quotientOrderStrictLowerPiece k I (N + 1) at hlower
70         have hqN : q ∈ quotientOrderPiece k I N := by
71           simpa [quotientOrderStrictLowerPiece, presentedStrictLowerPiece,
72             quotientOrderPiece] using hlower
73         rw [ih] at hqN
74         exact hqN
75       · exact bot_le

77 /-- Vanishing of every actual associated graded piece reflects to the actual
78 filtered quotient because the induced order filtration is exhaustive. -/
79 theorem filteredRightQuotient_subsingleton_of_all_graded_subsingleton
80     (I : RightIdeal (WeylAlg k n))
81     (hgraded : ∀ N, Subsingleton (QuotientOrderGradedPiece k I N)) :
82     Subsingleton (FilteredRightQuotient k I) := by
83   have hzero : ∀ q : FilteredRightQuotient k I, q = 0 := by
84     intro q
85     refine Submodule.Quotient.induction_on _ q ?_
86     intro a
87     obtain ⟨N, hN⟩ := exists_mem_orderPiece k a
88     have hpiece :
89         (rightIdealKSubmodule k I).mkQ a ∈ quotientOrderPiece k I N :=
90       ⟨a, hN, rfl⟩
91     rw [quotientOrderPiece_eq_bot_of_all_graded_subsingleton k I hgraded] at hpiece
92     exact hpiece
93   exact ⟨fun q r => (hzero q).trans (hzero r).symm⟩
94 
95 /-- Consequently the existing regular-right-module quotient is zero as well,
96 via its canonical linear equivalence with the filtered quotient. -/
97 theorem rightQuotient_subsingleton_of_all_graded_subsingleton
98     (I : RightIdeal (WeylAlg k n))
99     (hgraded : ∀ N, Subsingleton (QuotientOrderGradedPiece k I N)) :
100     Subsingleton (RightQuotient I) := by
101   letI : Subsingleton (FilteredRightQuotient k I) :=
102     filteredRightQuotient_subsingleton_of_all_graded_subsingleton k I hgraded
103   constructor
104   intro q r
105   apply (filteredRightQuotientEquivRightQuotient k I).symm.injective
106   exact Subsingleton.elim _ _
