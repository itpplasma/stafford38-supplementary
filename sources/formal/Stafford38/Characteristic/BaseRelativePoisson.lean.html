6 /-!
7 # Base-relative Poisson closure
8 
9 The vertical-translation argument needs brackets whose left entry comes from
10 the base-coordinate ring. This file defines that fragment and proves its
11 consequences for Hamiltonian iteration and fibre-line translation. It does
12 not assert that a particular ideal has the property; the unconditional
13 canonical-support proof is assembled in `FoundationClosure`.
14 -/
15

25 /--
26 The exact base-relative fragment of Poisson involutivity:
27 `{J ∩ k[x], J} ⊆ J`, expressed using the canonical base embedding.
28 -/
29 def IsBaseRelativePoisson (J : Ideal (SymbolRing k n)) : Prop :=
30   ∀ f : MvPolynomial (Fin n) k, baseLift f ∈ J →
31     ∀ g ∈ J, poissonBracket (baseLift f) g ∈ J
32 
33 /-- Full Poisson closure implies the base-relative fragment. -/
34 theorem IsPoisson.isBaseRelativePoisson
35     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J) :
36     IsBaseRelativePoisson J := by
37   intro f hf g hg
38   exact hJ (baseLift f) hf g
39 
40 /-- Base-relative closure is preserved through all iterated Hamiltonian brackets. -/
41 theorem hamiltonIter_mem_of_isBaseRelativePoisson
42     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
43     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
44     (g : SymbolRing k n) (hg : g ∈ J) :
45     ∀ m, hamiltonIter f m g ∈ J := by
46   intro m
47   induction m with
48   | zero => exact hg
49   | succ m hm => exact hJ f hf (hamiltonIter f m g) hm
50 
51 /--
52 The fibre-line polynomial vanishes under the exact base-relative Gabber
53 fragment; full Poisson closure is not needed.
54 -/
55 theorem fibreLinePolynomial_eq_zero_of_isBaseRelativePoisson
56     [CharZero k]
57     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
58     (y : Fin n → k)
59     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
60     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
61     (g : SymbolRing k n) (hg : g ∈ J) :
62     fibreLinePolynomial y (differentialAt y f) g = 0 := by
63   apply polynomial_eq_zero_of_eval_iterate_derivative_zero
64   intro m
65   rw [iterate_derivative_fibreLinePolynomial, eval_zero_fibreLinePolynomial]
66   exact hzero _ (hamiltonIter_mem_of_isBaseRelativePoisson J hJ f hf g hg m)

68 /--
69 A zero-section common zero remains a common zero after translation by
70 `t df_y`, assuming only `{J ∩ k[x], J} ⊆ J`.
71 -/
72 theorem zeroSection_stable_under_differential_translation_of_isBaseRelativePoisson
73     [CharZero k]
74     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
75     (y : Fin n → k)
76     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
77     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
78     (g : SymbolRing k n) (hg : g ∈ J) (t : k) :
79     MvPolynomial.eval (differentialTranslatePoint y f t) g = 0 := by
80   have heval :
81       MvPolynomial.eval (differentialTranslatePoint y f t) g =
82         MvPolynomial.eval (Sum.elim y (fun i => t * differentialAt y f i)) g := by
83     apply MvPolynomial.eval₂_congr
84     intro i c hi hc
85     rcases i with i | i <;> simp [differentialTranslatePoint, affineFibreTranslatePoint]
86   rw [heval, ← eval_fibreLinePolynomial]
87   rw [fibreLinePolynomial_eq_zero_of_isBaseRelativePoisson J hJ y hzero f hf g hg]
88   exact Polynomial.eval_zero
