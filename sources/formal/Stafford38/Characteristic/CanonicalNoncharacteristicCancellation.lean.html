10 /-!
11 # Coordinate-cancellation criteria on the associated graded
12 
13 For the canonical filtered Weyl quotient, this file identifies
14 `CoordinateCancellation` with injectivity of the distinguished coordinate on
15 every actual order-associated-graded piece. It also proves a sufficient
16 initial-ideal saturation criterion. The unconditional support exclusion used
17 by `FoundationClosure` is proved by
18 `CanonicalKoszulContradiction.canonical_support_avoidance`.
19 -/

45 /-- The distinguished coordinate, regarded as a homogeneous order-zero
46 symbol. -/
47 def coordinateOrderSymbol (n : ℕ) :
48     MvPolynomial.weightedHomogeneousSubmodule k
49       (@orderWeight (n + 1)) 0 :=
50   principalComponentOnPiece k (@orderWeight (n + 1)) 0
51     ⟨presentedCoordinate k n,
52       presentedCoordinate_mem_orderPiece_zero k n⟩

54 /-- Multiplication by the distinguished coordinate on the actual degree-`m`
55 associated-graded piece of the canonical filtered quotient. -/
56 def canonicalGradedCoordinateAction (n N m : ℕ)
57     (d : WeylAlg k (n + 1)) :
58     QuotientOrderGradedPiece k (presentedCanonicalRightIdeal (k := k) n N d) m →ₗ[k]
59       QuotientOrderGradedPiece k (presentedCanonicalRightIdeal (k := k) n N d) m := by
60   simpa using
61     (quotientOrderHomogeneousAction (N := m) (M := 0)
62       k (presentedCanonicalRightIdeal (k := k) n N d) (coordinateOrderSymbol k n))

64 @[simp] theorem canonicalGradedCoordinateAction_mk
65     (n N m : ℕ) (d : WeylAlg k (n + 1))
66     (z : orderPiece k (n + 1) m) :
67     canonicalGradedCoordinateAction k n N m d
68         (orderPieceToQuotientGraded k (presentedCanonicalRightIdeal (k := k) n N d) m z) =
69       orderPieceToQuotientGraded k (presentedCanonicalRightIdeal (k := k) n N d) m
70         ⟨(z : WeylAlg k (n + 1)) * presentedCoordinate k n,
71           mul_mem_orderPiece k z.property
72             (presentedCoordinate_mem_orderPiece_zero k n)⟩ := by
73   change quotientOrderHomogeneousAction k (presentedCanonicalRightIdeal (k := k) n N d)
74       (coordinateOrderSymbol k n)
75         (orderPieceToQuotientGraded k (presentedCanonicalRightIdeal (k := k) n N d) m z) = _
76   simpa [coordinateOrderSymbol] using
77     quotientOrderHomogeneousAction_mk_mul k (presentedCanonicalRightIdeal (k := k) n N d) z

79         presentedCoordinate_mem_orderPiece_zero k n⟩

81 /-- Degreewise coordinate cancellation is exactly injectivity of coordinate
82 multiplication on every actual associated-graded quotient piece.  Thus the
83 previously conditional consumer's hypothesis is the strictness part of the
84 noncharacteristic inverse-image theorem, expressed without any geometric
85 terminology. -/
86 theorem coordinateCancellation_iff_forall_graded_injective
87     (n N : ℕ) (d : WeylAlg k (n + 1)) :
88     CoordinateCancellation k n N d ↔
89       ∀ m, Function.Injective
90         (canonicalGradedCoordinateAction k n N m d) := by
91   let I := presentedCanonicalRightIdeal (k := k) n N d
92   let x := presentedCoordinate k n
93   constructor
94   · intro hcancel m
95     rw [← LinearMap.ker_eq_bot]
96     apply le_antisymm
97     · intro q hq
98       obtain ⟨z, rfl⟩ :=
99         orderPieceToQuotientGraded_surjective k I m q
100       rw [LinearMap.mem_ker] at hq
101       have hproduct :
102           (z : WeylAlg k (n + 1)) * x ∈
103             rightIdealKSubmodule k I ⊔
104               presentedStrictLowerPiece k orderWeight m := by
105         have hproductKer :
106             ⟨(z : WeylAlg k (n + 1)) * x,
107               mul_mem_orderPiece k z.property
108                 (presentedCoordinate_mem_orderPiece_zero k n)⟩ ∈
109               LinearMap.ker (orderPieceToQuotientGraded k I m) := by
110           rw [LinearMap.mem_ker]
111           simpa [I, x] using hq
112         rw [ker_orderPieceToQuotientGraded] at hproductKer
113         exact hproductKer
114       have hz := hcancel m z z.property hproduct
115       rw [Submodule.mem_bot]
116       rw [← LinearMap.mem_ker, ker_orderPieceToQuotientGraded]
117       change (z : WeylAlg k (n + 1)) ∈
118         rightIdealKSubmodule k I ⊔
119           presentedStrictLowerPiece k orderWeight m
120       exact hz
121     · exact bot_le
122   · intro hinjective m z hz hzx
123     let zz : orderPiece k (n + 1) m := ⟨z, hz⟩
124     have hproductZero :
125         orderPieceToQuotientGraded k (presentedCanonicalRightIdeal (k := k) n N d) m
126             ⟨z * presentedCoordinate k n,
127               mul_mem_orderPiece k hz
128                 (presentedCoordinate_mem_orderPiece_zero k n)⟩ = 0 := by
129       rw [← LinearMap.mem_ker, ker_orderPieceToQuotientGraded]
130       exact hzx
131     have hzero :
132         orderPieceToQuotientGraded k (presentedCanonicalRightIdeal (k := k) n N d) m zz = 0 := by
133       apply hinjective m
134       rw [map_zero, canonicalGradedCoordinateAction_mk]
135       exact hproductZero
136     change z ∈ rightIdealKSubmodule k (presentedCanonicalRightIdeal (k := k) n N d) ⊔
137       presentedStrictLowerPiece k orderWeight m
138     have hzker : zz ∈ LinearMap.ker
139         (orderPieceToQuotientGraded k (presentedCanonicalRightIdeal (k := k) n N d) m) := by
140       rw [LinearMap.mem_ker]
141       exact hzero
142     rw [ker_orderPieceToQuotientGraded] at hzker
143     exact hzker

145 /-- Saturation of the concrete order initial ideal by the coordinate implies
146 the exact degreewise cancellation needed by the canonical consumer.  This is
147 a sufficient commutative-algebra formulation of strictness; it is not assumed
148 to follow from monicity in this file. -/
149 theorem coordinateCancellation_of_initialIdeal_coordinate_saturated
150     (n N : ℕ) (d : WeylAlg k (n + 1))
151     (hsaturated :
152       (orderInitialIdeal k (presentedCanonicalRightIdeal (k := k) n N d)).colon
153           ({MvPolynomial.X (Sum.inl (0 : Fin (n + 1)))} :
154             Set (SymbolRing k (n + 1))) =
155         orderInitialIdeal k (presentedCanonicalRightIdeal (k := k) n N d)) :
156     CoordinateCancellation k n N d := by
157   rw [coordinateCancellation_iff_forall_graded_injective]
158   intro m
159   let I := presentedCanonicalRightIdeal (k := k) n N d
160   let X := coordinateOrderSymbol k n
161   have hX : (X : SymbolRing k (n + 1)) =
162       MvPolynomial.X (.inl (0 : Fin (n + 1))) :=
163     coe_coordinate_order_symbol k n
164   have hsymbol : Function.Injective
165       (homogeneousSymbolAction (N := m) (M := 0) k I X) := by
166     rw [← LinearMap.ker_eq_bot]
167     apply le_antisymm
168     · intro q hq
169       rw [LinearMap.mem_ker] at hq
170       rw [Submodule.mem_bot]
171       refine Submodule.Quotient.induction_on _ q ?_ hq
172       intro P hP
173       rw [homogeneousSymbolAction_mk,
174         Submodule.Quotient.mk_eq_zero] at hP
175       have hPX : (P : SymbolRing k (n + 1)) *
176           MvPolynomial.X (.inl (0 : Fin (n + 1))) ∈
177             orderInitialIdeal k I := by
178         rw [← hX]
179         exact (mem_orderSymbolRelation_iff_coe_mem_orderInitialIdeal
180           k I m (homogeneousRightMul k X P)).mp hP
181       have hPcolon : (P : SymbolRing k (n + 1)) ∈
182           (orderInitialIdeal k I).colon
183             ({MvPolynomial.X (Sum.inl (0 : Fin (n + 1)))} :
184               Set (SymbolRing k (n + 1))) :=
185         Submodule.mem_colon_singleton.mpr hPX
186       have hPJ : (P : SymbolRing k (n + 1)) ∈ orderInitialIdeal k I := by
187         rw [hsaturated] at hPcolon
188         exact hPcolon
189       rw [Submodule.Quotient.mk_eq_zero]
190       exact (mem_orderSymbolRelation_iff_coe_mem_orderInitialIdeal
191         k I m P).mpr hPJ
192     · exact bot_le
193   intro q₁ q₂ hq
194   apply (quotientOrderGradedPieceEquivSymbols k I m).injective
195   apply hsymbol
196   rw [← quotientOrderHomogeneousAction_compatibility,
197     ← quotientOrderHomogeneousAction_compatibility]
198   exact congrArg (quotientOrderGradedPieceEquivSymbols k I m) hq

200 /-- A saturated canonical initial ideal therefore satisfies the already
201 formalized scheme-level coordinate-axis avoidance conclusion. -/
202 theorem canonical_support_disjoint_coordinate_zeroLocus_of_saturated
203     (n N : ℕ) {d : WeylAlg k (n + 1)}
204     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
205     (hsaturated :
206       (orderInitialIdeal k (presentedCanonicalRightIdeal (k := k) n N d)).colon
207           ({MvPolynomial.X (Sum.inl (0 : Fin (n + 1)))} :
208             Set (SymbolRing k (n + 1))) =
209         orderInitialIdeal k (presentedCanonicalRightIdeal (k := k) n N d)) :
210     Disjoint
211       (orderCharacteristicSupport k (presentedCanonicalRightIdeal (k := k) n N d))
212       (PrimeSpectrum.zeroLocus
213         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
214           Set (SymbolRing k (n + 1)))) := by
215   exact canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus
216     k n N hd
217       (coordinateCancellation_of_initialIdeal_coordinate_saturated
218         k n N d hsaturated)
