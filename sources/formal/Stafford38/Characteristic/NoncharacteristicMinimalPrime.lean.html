31 private def iterPderiv (i : PhaseVar (n + 1)) (r : ℕ)
32     (f : SymbolRing k (n + 1)) : SymbolRing k (n + 1) :=
33   (MvPolynomial.pderiv i)^[r] f

35 private theorem iterPderiv_mem_of_involutive

37     (j : Fin (n + 1)) (hx : MvPolynomial.X (.inl j) ∈ P)
38     (f : SymbolRing k (n + 1)) (hf : f ∈ P) :
39     ∀ r, iterPderiv (.inr j) r f ∈ P := by
40   intro r
41   induction r with
42   | zero => simpa [iterPderiv] using hf
43   | succ r ihr =>
44       have hb := hP (MvPolynomial.X (.inl j)) hx
45         (iterPderiv (.inr j) r f) ihr

53     iterPderiv i N f = MvPolynomial.C (N.factorial : k) := by

64           ((m + r).descFactorial r : k) := by

68     | succ r ihr =>
69         intro m

71           MvPolynomial.pderiv i (iterPderiv i r f) by
72             simp [iterPderiv, Function.iterate_succ_apply']]

76         change MvPolynomial.coeff

79           MvPolynomial.coeff

99         rw [hcarg]
