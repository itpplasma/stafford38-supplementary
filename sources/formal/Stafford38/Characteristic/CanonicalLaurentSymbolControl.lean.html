10 /-!
11 # Conditional Laurent symbol control
12 
13 This module extracts a fibre polynomial from the canonical order symbol and
14 proves the ground-base conormal containment used by this Laurent-direction
15 route. Its `CanonicalGeometricBaseRelativePoisson` hypothesis is specific to
16 that route. The unconditional support proof in `FoundationClosure` uses the
17 checked localized-involutivity and general coisotropic route.
18 -/

48 /-- A phase polynomial supported only on fibre variables comes from a fibre
49 polynomial by the canonical fibre inclusion. -/
50 theorem exists_fibreLift_eq_of_isFibreOnly
51     (Q : SymbolRing k n) (hQ : IsFibreOnly k Q) :
52     ∃ P : MvPolynomial (Fin n) k, fibreLift P = Q := by
53   apply MvPolynomial.exists_rename_eq_of_vars_subset_range Q Sum.inr
54     Sum.inr_injective
55   intro i hi
56   change i ∈ Q.vars at hi
57   rw [MvPolynomial.mem_vars] at hi
58   rcases hi with ⟨m, hm, him⟩
59   have hcoeff : MvPolynomial.coeff m Q ≠ 0 :=
60     MvPolynomial.mem_support_iff.mp hm
61   rcases i with i | i
62   · have hzero := hQ m hcoeff i
63     exact (Finsupp.mem_support_iff.mp him hzero).elim
64   · exact ⟨i, rfl⟩

88 /-- A geometric zero of the scalar extension of the ground base contraction
89 lifts to the zero section of the geometric reduced support.  This avoids any
90 claim that scalar extension commutes with the whole base contraction. -/
91 theorem zeroSection_mem_geometricReducedSupport_of_groundBaseZero
92     {K : Type*} [Field K] [Algebra k K]
93     (W : RightIdeal (WeylAlg k n)) (y : Fin n → K)
94     (hy : ∀ f ∈ reducedOrderBaseIdeal k W,
95       MvPolynomial.eval₂ (algebraMap k K) y f = 0) :
96     zeroSectionPoint y ∈ MvPolynomial.zeroLocus K
97       (geometricReducedOrderSupportIdeal (k := k) (K := K) W) := by
98   rw [mem_zeroLocus_geometricReducedOrderSupportIdeal_iff]
99   intro P hP
100   obtain ⟨m, hm⟩ := (mem_reducedOrderSupportIdeal_iff k W P).mp hP
101   let f := fibreZeroSpecialization k (P ^ m)
102   have hcomponent :
103       (DirectSum.decompose
104           (orderDecompositionGroundExtension (k := k) (n := n)) (P ^ m) 0 :
105         SymbolRing k n) ∈ orderInitialIdeal k W :=
106     coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k W 0
107       (DirectSum.decompose
108         (orderDecompositionGroundExtension (k := k) (n := n)) (P ^ m) 0)
109       (decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
110         k W (P ^ m) hm 0)
111   have hbaseLift : baseLift f ∈ reducedOrderSupportIdeal k W := by
112     rw [← zeroComponent_eq_baseLift_fibreZeroSpecialization]
113     exact orderInitialIdeal_le_reducedOrderSupportIdeal k W hcomponent
114   have hfbase : f ∈ reducedOrderBaseIdeal k W :=
115     (mem_reducedOrderBaseIdeal_iff k W f).mpr hbaseLift
116   have hfy := hy f hfbase
117   have heval :
118       MvPolynomial.eval (zeroSectionPoint y)
119         ((scalarPolynomialMap (k := k) (K := K) (PhaseVar n) P) ^ m) = 0 := by
120     rw [← map_pow,
121       ← eval_fibreZeroSpecialization K y
122         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) (P ^ m)),
123       ← scalarPolynomialMap_fibreZeroSpecialization]
124     simpa only [eval_scalarPolynomialMap] using hfy
125   have hp0 :
126       MvPolynomial.eval (zeroSectionPoint y)
127         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) P) = 0 := by
128     exact (pow_eq_zero_iff'.mp (by simpa only [map_pow] using heval)).1
129   simpa [eval_scalarPolynomialMap] using hp0

131 /-- Under post-extension base-relative Poisson closure, the conormal locus of
132 the scalar-extended *ground* base contraction already lies in geometric
133 support.  Equality with the post-extension base contraction is unnecessary. -/
134 theorem equationConormalLocus_groundMap_subset_geometricReducedSupport
135     {K : Type*} [Field K] [Algebra k K] [CharZero K]
136     (W : RightIdeal (WeylAlg k n))
137     (hJ : IsBaseRelativePoisson
138       (geometricReducedOrderSupportIdeal (k := k) (K := K) W)) :
139     equationConormalLocus
140         ((reducedOrderBaseIdeal k W).map
141           (scalarPolynomialMap (k := k) (K := K) (Fin n))) ⊆
142       MvPolynomial.zeroLocus K
143         (geometricReducedOrderSupportIdeal (k := k) (K := K) W) := by
144   intro q hq
145   let y : Fin n → K := fun i => q (Sum.inl i)
146   let xi : Fin n → K := fun i => q (Sum.inr i)
147   have hy : ∀ f ∈ reducedOrderBaseIdeal k W,
148       MvPolynomial.eval₂ (algebraMap k K) y f = 0 :=
149     (mem_equationConormalLocus_map_iff
150       (reducedOrderBaseIdeal k W) q).mp hq |>.1
151   have hzero :=
152     zeroSection_mem_geometricReducedSupport_of_groundBaseZero W y hy
153   have hpoint := affineConormal_coordinatePoint_isCommonZero
154     (geometricReducedOrderSupportIdeal (k := k) (K := K) W) hJ y hzero
155     ((reducedOrderBaseIdeal k W).map
156       (scalarPolynomialMap (k := k) (K := K) (Fin n)))
157     (fun f ↦ map_baseContraction_le_geometricBaseContraction
158       (reducedOrderSupportIdeal k W) f.2)
159     xi hq.2
160   have hsplit : Sum.elim y xi = q := by
161     funext i
162     rcases i with i | i <;> rfl
163   rw [← hsplit]
164   exact hpoint

166 /-- The precise post-extension base-relative Gabber input used by the Laurent
167 symbol argument. -/
168 def CanonicalGeometricBaseRelativePoisson : Prop :=
169   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
170     (d : WeylAlg k (n + 1)),
171     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
172       IsBaseRelativePoisson
173         (geometricReducedOrderSupportIdeal
174           (K := LaurentSeries k)
175           (canonicalRightIdeal (presentedCoordinate k n) d N))

177 /-- Post-extension Gabber closure produces the packaged Laurent symbol control
178 consumed by the terminal proof. -/
179 theorem canonicalLaurentSymbolControl_of_geometric_inputs
180     (hgabber : CanonicalGeometricBaseRelativePoisson.{u}) :
181     CanonicalLaurentSymbolControl.{u} := by
182   intro k _ _ _ n N d hN hd
183   letI : CharZero (LaurentSeries k) :=
184     charZero_of_injective_algebraMap
185       (algebraMap k (LaurentSeries k)).injective
186   let W := canonicalRightIdeal (presentedCoordinate k n) d N
187   rcases exists_canonical_fibrePolynomial n N hd with ⟨P, hP⟩
188   refine ⟨P, hP, ?_⟩
189   intro q hq
190   have hqSupport : q ∈ MvPolynomial.zeroLocus (LaurentSeries k)
191       (geometricReducedOrderSupportIdeal
192         (K := LaurentSeries k) W) :=
193     equationConormalLocus_groundMap_subset_geometricReducedSupport
194       W (hgabber k n N d hN hd) hq
195   have hprincipal : presentedPrincipalComponent k
196         (@Stafford38.WeylFiltration.orderWeight (n + 1)) N d ∈
197       reducedOrderSupportIdeal k W :=
198     orderInitialIdeal_le_reducedOrderSupportIdeal k W
199       (canonical_orderPrincipalComponent_mem_initialIdeal k n N hd)
200   have hmapped : scalarPolynomialMap
201         (k := k) (K := LaurentSeries k) (PhaseVar (n + 1))
202         (presentedPrincipalComponent k
203           (@Stafford38.WeylFiltration.orderWeight (n + 1)) N d) ∈
204       geometricReducedOrderSupportIdeal
205         (K := LaurentSeries k) W :=
206     Ideal.le_radical (Ideal.mem_map_of_mem
207       (scalarPolynomialMap
208         (k := k) (K := LaurentSeries k) (PhaseVar (n + 1))) hprincipal)
209   have hzero := hqSupport _ hmapped
210   rw [← hP] at hzero
211   simpa only [MvPolynomial.aeval_eq_eval, eval_scalarPolynomialMap] using hzero
