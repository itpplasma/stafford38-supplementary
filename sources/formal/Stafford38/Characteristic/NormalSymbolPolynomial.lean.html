55 theorem optionEquivLeft_monic_of_isHomogeneous
56     {S : Type*} [Finite S] (P : MvPolynomial (Option S) k) (N : ℕ)
57     (hP : P.IsHomogeneous N)
58     (hlead : MvPolynomial.coeff (Finsupp.single none N) P = 1) :
59     (MvPolynomial.optionEquivLeft k S P).Monic := by
60   classical
61   apply Polynomial.monic_of_natDegree_le_of_coeff_eq_one N
62   · rw [MvPolynomial.natDegree_optionEquivLeft]
63     exact (MvPolynomial.degreeOf_le_totalDegree P none).trans hP.totalDegree_le
64   · ext m
65     rw [MvPolynomial.optionEquivLeft_coeff_coeff]
66     by_cases hm : m = 0
67     · subst m
68       simpa using hlead
69     · have hzero : MvPolynomial.coeff (m.optionElim N) P = 0 := by
70         by_contra hcoeff
71         have hcoeffHomogeneous :
72             ((MvPolynomial.optionEquivLeft k S P).coeff N).IsHomogeneous 0 := by
73           intro e he
74           rw [MvPolynomial.optionEquivLeft_coeff_coeff] at he
75           have hdegree := hP he
76           have hzero : (Finsupp.weight (1 : S → ℕ)) e = 0 := by
77             by_contra hne
78             have hpos : 0 < (Finsupp.weight (1 : S → ℕ)) e := Nat.pos_of_ne_zero hne
79             have hall : N + (Finsupp.weight (1 : S → ℕ)) e = N := by
80               rw [Finsupp.weight_apply, Finsupp.sum_option_index_smul] at hdegree
81               simpa [Finsupp.weight_apply] using hdegree
82             omega
83           exact hzero
84         have hcoeff' : MvPolynomial.coeff m
85             ((MvPolynomial.optionEquivLeft k S P).coeff N) ≠ 0 := by
86           change AddMonoidAlgebra.coeff
87             ((MvPolynomial.optionEquivLeft k S P).coeff N) m ≠ 0
88           rw [MvPolynomial.optionEquivLeft_coeff_coeff]
89           exact hcoeff
90         have hdegree := hcoeffHomogeneous hcoeff'
91         rw [Finsupp.weight_apply] at hdegree
92         apply hm
93         ext x
94         by_cases hx : x ∈ m.support
95         · simpa using (Finset.sum_eq_zero_iff_of_nonneg
96               (fun _ _ ↦ Nat.zero_le _)).mp hdegree x hx
97         · exact Finsupp.notMem_support_iff.mp hx
98       have hzero' : AddMonoidAlgebra.coeff P (m.optionElim N) = 0 := by
99         change MvPolynomial.coeff (m.optionElim N) P = 0
100         exact hzero
101       change AddMonoidAlgebra.coeff P (m.optionElim N) =
102         AddMonoidAlgebra.coeff (1 : MvPolynomial S k) m
103       rw [hzero', MvPolynomial.coeff_one]
104       simp [Ne.symm hm]

106 theorem canonicalNormalPolynomial_monic {n N : ℕ}
107     {d : WeylAlg k (n + 1)}
108     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
109     (canonicalNormalPolynomial (k := k) (n := n) (N := N) d).Monic := by
110   apply optionEquivLeft_monic_of_isHomogeneous
111   · exact (canonical_orderPrincipalComponent_isHomogeneous n N hd).rename_isHomogeneous
112   · change AddMonoidAlgebra.coeff
113         (MvPolynomial.rename (normalVariableEquiv n).symm
114           (presentedPrincipalComponent k orderWeight N d))
115         (Finsupp.single none N) = 1
116     have hsingle : (Finsupp.single (.inr (0 : Fin (n + 1))) N).mapDomain
117         (normalVariableEquiv n).symm = Finsupp.single none N := by
118       simp [normalVariableEquiv]
119     rw [← hsingle, MvPolynomial.coeff_rename_mapDomain
120       (normalVariableEquiv n).symm (normalVariableEquiv n).symm.injective]
121     have hlead : AddMonoidAlgebra.coeff
122         (presentedPrincipalComponent k orderWeight N d)
123         (Finsupp.single (.inr (0 : Fin (n + 1))) N) = 1 := by
124       change MvPolynomial.coeff
125         (Finsupp.single (.inr (0 : Fin (n + 1))) N)
126         (presentedPrincipalComponent k orderWeight N d) = 1
127       exact canonical_orderPrincipalComponent_pureMomentumCoefficient k n N hd
128     exact hlead
