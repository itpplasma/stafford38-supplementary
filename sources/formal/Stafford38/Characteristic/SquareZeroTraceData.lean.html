7 # Generic square-zero trace data
8 
9 This file records the minimal algebraic interface consumed by a later trace
10 argument.  The deformation ring may be noncommutative, and its module acts on
11 the right, encoded as a left action of the opposite ring.  In particular,
12 right multiplication reverses composition of endomorphisms.
13 
14 No localization, trace theorem, minimal-prime statement, Gabber theorem, or
15 concrete Weyl instance is asserted here.
16 -/

29 /-- Right multiplication by `b`, regarded as a `k`-linear endomorphism.
30 
31 The scalar action is by `Bᵐᵒᵖ`; thus `rightActionEnd k B N b` sends `m` to
32 `m b` in ordinary right-module notation. -/
33 def rightActionEnd (b : B) : Module.End k N where
34   toFun m := MulOpposite.op b • m
35   map_add' _ _ := smul_add _ _ _
36   map_smul' a m := (smul_comm a (MulOpposite.op b) m).symm
37 
38 @[simp] theorem rightActionEnd_apply (b : B) (m : N) :
39     rightActionEnd k B N b m = MulOpposite.op b • m :=
40   rfl
41 
42 /-- Composition of right actions reverses the displayed endomorphism order. -/
43 theorem rightActionEnd_mul_reverse (a b : B) :
44     rightActionEnd k B N b * rightActionEnd k B N a =
45       rightActionEnd k B N (a * b) := by
46   ext m
47   change MulOpposite.op b • (MulOpposite.op a • m) =
48     MulOpposite.op (a * b) • m
49   rw [← mul_smul]
50   rfl
51 
52 /-- The ringCommutator of right-action endomorphisms has reversed inputs.
53 
54 With `R_b(m) = m b`, this is `[R_b,R_a] = R_[a,b]`. -/
55 theorem rightAction_commutator (a b : B) :
56     AlgebraicAnalysis.ringCommutator
57         (rightActionEnd k B N b) (rightActionEnd k B N a) =
58       rightActionEnd k B N (AlgebraicAnalysis.ringCommutator a b) := by
59   simp only [AlgebraicAnalysis.ringCommutator]
60   rw [rightActionEnd_mul_reverse, rightActionEnd_mul_reverse]
61   ext m
62   simp [rightActionEnd, AlgebraicAnalysis.ringCommutator, sub_smul]

64 /-- Minimal data for a right-module square-zero deformation and its
65 first-order ringCommutator bracket.
66 
67 The structure deliberately records only the interfaces required by the later
68 trace argument.  It does not assert that such data exist for the Weyl Rees
69 two-jet. -/
70 structure RightSquareZeroTraceData
71     (k : Type u_k) (R : Type u_R) (B : Type u_B)
72     (N : Type u_N) (G : Type u_G)
73     [Field k] [CommRing R] [Algebra k R]
74     [Ring B] [Algebra k B]
75     [AddCommGroup N] [Module k N] [Module Bᵐᵒᵖ N]
76     [SMulCommClass k Bᵐᵒᵖ N]
77     [AddCommGroup G] [Module k G] [Module R G] where
78   /-- The central square-zero deformation parameter. -/
79   c : B
80   c_center : c ∈ Set.center B
81   c_sq : c ^ 2 = 0
82   /-- Specialization from the deformation ring to the commutative fibre. -/
83   pi : B →ₐ[k] R
84   pi_surjective : Function.Surjective pi
85   pi_c : pi c = 0
86   /-- Action of the deformation parameter on the right module. -/
87   cAct : N →ₗ[k] N
88   cAct_apply : ∀ m, cAct m = MulOpposite.op c • m
89   c_exact : LinearMap.ker cAct = LinearMap.range cAct
90   /-- Specialization of the module to its commutative fibre. -/
91   rho : N →ₗ[k] G
92   rho_surjective : Function.Surjective rho
93   rho_ker : LinearMap.ker rho = LinearMap.range cAct
94   rho_action : ∀ b m, rho (MulOpposite.op b • m) = pi b • rho m
95   /-- The `k`-bilinear first-order bracket on the special fibre. -/
96   bracket : R →ₗ[k] R →ₗ[k] R
97   /-- Every deformation-ring ringCommutator is divisible on the left by `c`,
98   with quotient specializing to the bracket in the displayed input order. -/
99   commutator_factor : ∀ a b : B, ∃ z : B,
100     AlgebraicAnalysis.ringCommutator a b = c * z ∧
101       pi z = bracket (pi a) (pi b)

112 /-- The recorded parameter action is literally right multiplication by `c`. -/
113 theorem cAct_eq_rightActionEnd :
114     D.cAct = rightActionEnd k B N D.c := by
115   ext m
116   exact D.cAct_apply m
117 
118 /-- The parameter action squares to zero. -/
119 theorem cAct_comp_self : D.cAct.comp D.cAct = 0 := by
120   ext m
121   simp only [LinearMap.comp_apply, D.cAct_apply, LinearMap.zero_apply]
122   rw [← mul_smul]
123   rw [← MulOpposite.op_mul]
124   rw [← pow_two, D.c_sq]
125   exact zero_smul _ _
126 
127 /-- Module specialization kills the parameter action. -/
128 theorem rho_comp_cAct : D.rho.comp D.cAct = 0 := by
129   ext m
130   simp only [LinearMap.comp_apply, D.cAct_apply, LinearMap.zero_apply]
131   rw [D.rho_action, D.pi_c, zero_smul]
132 
133 /-- Special-fibre elements admit deformation lifts whose ringCommutator has the
134 recorded first-order factorization. -/
135 theorem exists_lifts_commutator_factor (x y : R) :
136     ∃ a b z : B,
137       D.pi a = x ∧ D.pi b = y ∧
138       AlgebraicAnalysis.ringCommutator a b = D.c * z ∧
139       D.pi z = D.bracket x y := by
140   obtain ⟨a, ha⟩ := D.pi_surjective x
141   obtain ⟨b, hb⟩ := D.pi_surjective y
142   obtain ⟨z, hz, hbracket⟩ := D.commutator_factor a b
143   refine ⟨a, b, z, ha, hb, hz, ?_⟩
144   simpa [ha, hb] using hbracket
145 
146 end RightSquareZeroTraceData
