41 /-- Right multiplication by a Weyl operator on the additive quotient. -/
42 def filteredRightMul (I : RightIdeal (WeylAlg k n))
43     (y : WeylAlg k n) :
44     FilteredRightQuotient k I →ₗ[k] FilteredRightQuotient k I :=
45   (rightIdealKSubmodule k I).mapQ (rightIdealKSubmodule k I)
46     { toFun := fun z => z * y
47       map_add' := fun z w => add_mul z w y
48       map_smul' := fun c z => Algebra.smul_mul_assoc c z y }
49     (by
50       intro z hz
51       change z * y ∈ I
52       exact I.smul_mem (MulOpposite.op y) hz)
53 
54 @[simp] theorem filteredRightMul_mk
55     (I : RightIdeal (WeylAlg k n))
56     (y z : WeylAlg k n) :
57     filteredRightMul k I y ((rightIdealKSubmodule k I).mkQ z) =
58       (rightIdealKSubmodule k I).mkQ (z * y) :=
59   rfl

61 @[simp] theorem filteredRightMul_one
62     (I : RightIdeal (WeylAlg k n))
63     (q : FilteredRightQuotient k I) :
64     filteredRightMul k I 1 q = q := by
65   refine Submodule.Quotient.induction_on _ q ?_
66   intro z
67   change (rightIdealKSubmodule k I).mkQ (z * 1) =
68     (rightIdealKSubmodule k I).mkQ z
69   rw [mul_one]

82 /-- The additive quotient is an honest right module over the Weyl algebra. -/
83 def filteredRightQuotientOpModule
84     (I : RightIdeal (WeylAlg k n)) :
85     Module (WeylAlg k n)ᵐᵒᵖ (FilteredRightQuotient k I) :=
86   Module.ofMinimalAxioms
87     (fun y q₁ q₂ => by
88       refine Submodule.Quotient.induction_on _ q₁ ?_
89       intro z
90       refine Submodule.Quotient.induction_on _ q₂ ?_
91       intro w
92       change (rightIdealKSubmodule k I).mkQ ((z + w) * y.unop) = _
93       rw [add_mul]
94       change (rightIdealKSubmodule k I).mkQ (z * y.unop + w * y.unop) =
95         (rightIdealKSubmodule k I).mkQ (z * y.unop) +
96           (rightIdealKSubmodule k I).mkQ (w * y.unop)
97       exact (rightIdealKSubmodule k I).mkQ.map_add _ _)
98     (fun y₁ y₂ q => by
99       refine Submodule.Quotient.induction_on _ q ?_
100       intro z
101       change (rightIdealKSubmodule k I).mkQ (z * (y₁ + y₂).unop) = _
102       rw [MulOpposite.unop_add, mul_add]
103       change (rightIdealKSubmodule k I).mkQ
104           (z * y₁.unop + z * y₂.unop) =
105         (rightIdealKSubmodule k I).mkQ (z * y₁.unop) +
106           (rightIdealKSubmodule k I).mkQ (z * y₂.unop)
107       exact (rightIdealKSubmodule k I).mkQ.map_add _ _)
108     (fun y₁ y₂ q => by
109       refine Submodule.Quotient.induction_on _ q ?_
110       intro z
111       change (rightIdealKSubmodule k I).mkQ (z * (y₁ * y₂).unop) = _
112       rw [MulOpposite.unop_mul]
113       change (rightIdealKSubmodule k I).mkQ (z * (y₂.unop * y₁.unop)) =
114         (rightIdealKSubmodule k I).mkQ ((z * y₂.unop) * y₁.unop)
115       rw [mul_assoc])
116     (fun q => by
117       refine Submodule.Quotient.induction_on _ q ?_
118       intro z
119       change (rightIdealKSubmodule k I).mkQ (z * (1 : WeylAlg k n)) = _
120       rw [mul_one]
121       rfl)

137 /-- The degree-`N` order piece, placed in the opposite Weyl algebra. -/
138 def orderPieceOp (N : ℕ) : Submodule k (WeylAlg k n)ᵐᵒᵖ :=
139   (orderPiece k n N).map (MulOpposite.opLinearEquiv k).toLinearMap

141 theorem mem_orderPieceOp_iff (N : ℕ) (y : (WeylAlg k n)ᵐᵒᵖ) :
142     y ∈ orderPieceOp (n := n) k N ↔
143       y.unop ∈ orderPiece k n N := by
144   constructor
145   · rintro ⟨z, hz, rfl⟩
146     simpa using hz
147   · intro hy
148     exact ⟨y.unop, hy, by simp⟩

152   one_mem := by

155   mul_mem := by

179 /-- Insert the coefficients of an opposite Rees polynomial in the external
180 direct sum of opposite filtered pieces. -/
181 def orderReesOpCoefficients
182     (r : (OrderReesRing (n := n) k)ᵐᵒᵖ) :
183     OppositeOrderPieceSum (n := n) k :=
184     DirectSum.mk _ r.unop.val.support fun N =>
185       ⟨MulOpposite.op (r.unop.val.coeff N),
186         (mem_orderPieceOp_iff k N _).2 (r.unop.property N)⟩

188 @[simp] theorem orderReesOpCoefficients_apply
189     (r : (OrderReesRing (n := n) k)ᵐᵒᵖ) (N : ℕ) :
190     ((orderReesOpCoefficients (n := n) k r) N :
191       (WeylAlg k n)ᵐᵒᵖ) =
192       MulOpposite.op (r.unop.val.coeff N) := by
193   by_cases hN : N ∈ r.unop.val.support
194   · rw [orderReesOpCoefficients, DirectSum.mk_apply_of_mem hN]
195   · rw [orderReesOpCoefficients, DirectSum.mk_apply_of_notMem hN]
196     rw [Polynomial.notMem_support_iff.mp hN]
197     rfl

199 /-- Coefficient decomposition as an additive homomorphism. -/
200 def orderReesOpToDirectSum :
201     (OrderReesRing (n := n) k)ᵐᵒᵖ →+ OppositeOrderPieceSum (n := n) k where
202   toFun := orderReesOpCoefficients (n := n) k
203   map_zero' := by
204     ext N
205     change ((orderReesOpCoefficients (n := n) k 0) N :
206       (WeylAlg k n)ᵐᵒᵖ) = 0
207     rw [orderReesOpCoefficients_apply]
208     simp
209   map_add' r s := by
210     ext N
211     change ((orderReesOpCoefficients (n := n) k (r + s)) N :
212       (WeylAlg k n)ᵐᵒᵖ) =
213         ((orderReesOpCoefficients (n := n) k r) N :
214           (WeylAlg k n)ᵐᵒᵖ) +
215         ((orderReesOpCoefficients (n := n) k s) N :
216           (WeylAlg k n)ᵐᵒᵖ)
217     rw [orderReesOpCoefficients_apply, orderReesOpCoefficients_apply]
218     simp

220 @[simp] theorem orderReesOpToDirectSum_apply
221     (r : (OrderReesRing (n := n) k)ᵐᵒᵖ) (N : ℕ) :
222     ((orderReesOpToDirectSum (n := n) k r) N :
223       (WeylAlg k n)ᵐᵒᵖ) =
224       MulOpposite.op (r.unop.val.coeff N) := by
225   exact orderReesOpCoefficients_apply k r N

229 /-- A homogeneous opposite filtered coefficient as an opposite Rees
230 monomial. -/
231 def orderPieceOpToReesOp (N : ℕ) :
232     orderPieceOp (n := n) k N →+
233       (OrderReesRing (n := n) k)ᵐᵒᵖ where
234   toFun y :=
235     let hy : (y : (WeylAlg k n)ᵐᵒᵖ).unop ∈ orderPiece k n N :=
236       (mem_orderPieceOp_iff (n := n) k N y).1 y.property
237     MulOpposite.op <| orderReesMonomial k N ⟨y.val.unop, hy⟩
238   map_zero' := by
239     apply MulOpposite.unop_injective
240     apply Subtype.ext
241     simp [orderReesMonomial]
242   map_add' y z := by
243     apply MulOpposite.unop_injective
244     apply Subtype.ext
245     simp [orderReesMonomial]

250           (fun N => orderPieceOp (n := n) k N) _ _) = 1 := by
251   apply MulOpposite.unop_injective
252   apply Subtype.ext

255 theorem orderPieceOpToReesOp_mul
256     {N M : ℕ} (y : orderPieceOp (n := n) k N)

416 /-- Homogeneous Rees action is represented by written right Weyl

418 theorem orderReesMonomial_op_smul_of
419     (I : RightIdeal (WeylAlg k n))

423       DirectSum.of (fun L => quotientOrderPiece k I L) N q =
424       DirectSum.of (fun L => quotientOrderPiece k I L) (M + N)

433       orderReesOpToDirectSum (n := n) k
434           (MulOpposite.op (orderReesMonomial k M y)) =
