39 /-- Right multiplication by a Weyl operator on the additive quotient. -/
40 def filteredRightMul (I : RightIdeal (PresentedWeyl k n))
41     (y : PresentedWeyl k n) :
42     FilteredRightQuotient k I →ₗ[k] FilteredRightQuotient k I :=
43   (rightIdealKSubmodule k I).mapQ (rightIdealKSubmodule k I)
44     { toFun := fun z => z * y
45       map_add' := fun z w => add_mul z w y
46       map_smul' := fun c z => Algebra.smul_mul_assoc c z y }
47     (by
48       intro z hz
49       change z * y ∈ I
50       exact I.smul_mem (MulOpposite.op y) hz)

80 /-- The additive quotient is an honest right module over the Weyl algebra. -/
81 def filteredRightQuotientOpModule
82     (I : RightIdeal (PresentedWeyl k n)) :
83     Module (PresentedWeyl k n)ᵐᵒᵖ (FilteredRightQuotient k I) :=
84   Module.ofMinimalAxioms
85     (fun y q₁ q₂ => by
86       refine Submodule.Quotient.induction_on _ q₁ ?_
87       intro z
88       refine Submodule.Quotient.induction_on _ q₂ ?_
89       intro w
90       change (rightIdealKSubmodule k I).mkQ ((z + w) * y.unop) = _
91       rw [add_mul]
92       change (rightIdealKSubmodule k I).mkQ (z * y.unop + w * y.unop) =
93         (rightIdealKSubmodule k I).mkQ (z * y.unop) +
94           (rightIdealKSubmodule k I).mkQ (w * y.unop)
95       exact (rightIdealKSubmodule k I).mkQ.map_add _ _)
96     (fun y₁ y₂ q => by
97       refine Submodule.Quotient.induction_on _ q ?_
98       intro z
99       change (rightIdealKSubmodule k I).mkQ (z * (y₁ + y₂).unop) = _
100       rw [MulOpposite.unop_add, mul_add]
101       change (rightIdealKSubmodule k I).mkQ
102           (z * y₁.unop + z * y₂.unop) =
103         (rightIdealKSubmodule k I).mkQ (z * y₁.unop) +
104           (rightIdealKSubmodule k I).mkQ (z * y₂.unop)
105       exact (rightIdealKSubmodule k I).mkQ.map_add _ _)
106     (fun y₁ y₂ q => by
107       refine Submodule.Quotient.induction_on _ q ?_
108       intro z
109       change (rightIdealKSubmodule k I).mkQ (z * (y₁ * y₂).unop) = _
110       rw [MulOpposite.unop_mul]
111       change (rightIdealKSubmodule k I).mkQ (z * (y₂.unop * y₁.unop)) =
112         (rightIdealKSubmodule k I).mkQ ((z * y₂.unop) * y₁.unop)
113       rw [mul_assoc])
114     (fun q => by
115       refine Submodule.Quotient.induction_on _ q ?_
116       intro z
117       change (rightIdealKSubmodule k I).mkQ (z * (1 : PresentedWeyl k n)) = _
118       rw [mul_one]
119       rfl)

139 theorem mem_orderPieceOp_iff (N : ℕ) (y : (PresentedWeyl k n)ᵐᵒᵖ) :
140     y ∈ orderPieceOp (n := n) k N ↔
141       y.unop ∈ orderPiece k n N := by
142   constructor

145   · intro hy

186 @[simp] theorem orderReesOpCoefficients_apply
187     (r : (OrderReesRing (n := n) k)ᵐᵒᵖ) (N : ℕ) :
188     ((orderReesOpCoefficients (n := n) k r) N :
189       (PresentedWeyl k n)ᵐᵒᵖ) =
190       MulOpposite.op (r.unop.val.coeff N) := by
191   by_cases hN : N ∈ r.unop.val.support
192   · rw [orderReesOpCoefficients, DirectSum.mk_apply_of_mem hN]

197 /-- Coefficient decomposition as an additive homomorphism. -/
198 def orderReesOpToDirectSum :
199     (OrderReesRing (n := n) k)ᵐᵒᵖ →+ OppositeOrderPieceSum (n := n) k where
200   toFun := orderReesOpCoefficients (n := n) k
201   map_zero' := by
202     ext N
203     change ((orderReesOpCoefficients (n := n) k 0) N :
204       (PresentedWeyl k n)ᵐᵒᵖ) = 0
205     rw [orderReesOpCoefficients_apply]
206     simp
207   map_add' r s := by
208     ext N
209     change ((orderReesOpCoefficients (n := n) k (r + s)) N :
210       (PresentedWeyl k n)ᵐᵒᵖ) =
211         ((orderReesOpCoefficients (n := n) k r) N :
212           (PresentedWeyl k n)ᵐᵒᵖ) +
213         ((orderReesOpCoefficients (n := n) k s) N :
214           (PresentedWeyl k n)ᵐᵒᵖ)
215     rw [orderReesOpCoefficients_apply, orderReesOpCoefficients_apply]
216     simp

227 /-- A homogeneous opposite filtered coefficient as an opposite Rees
228 monomial. -/
229 def orderPieceOpToReesOp (N : ℕ) :
230     orderPieceOp (n := n) k N →+
231       (OrderReesRing (n := n) k)ᵐᵒᵖ where
232   toFun y :=
233     let hy : (y : (PresentedWeyl k n)ᵐᵒᵖ).unop ∈ orderPiece k n N :=
234       (mem_orderPieceOp_iff (n := n) k N y).1 y.property
235     MulOpposite.op <| orderReesMonomial k N ⟨y.val.unop, hy⟩
236   map_zero' := by
237     apply MulOpposite.unop_injective
238     apply Subtype.ext
239     simp [orderReesMonomial]
240   map_add' y z := by
241     apply MulOpposite.unop_injective
242     apply Subtype.ext
243     simp [orderReesMonomial]

414 /-- Homogeneous Rees action is represented by written right Weyl

416 theorem orderReesMonomial_op_smul_of
417     (I : RightIdeal (PresentedWeyl k n))

421       DirectSum.of (fun L => quotientOrderPiece k I L) N q =
