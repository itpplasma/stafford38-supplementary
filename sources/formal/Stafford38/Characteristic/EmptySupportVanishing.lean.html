9 /-!
10 # Empty order-characteristic support forces quotient vanishing
11 
12 The global additive equivalence between the cyclic symbol quotient and the
13 external direct sum of the actual filtered quotient pieces is enough for
14 vanishing.  No action of the symbol ring on that direct sum is transported or
15 used here.
16 -/

36 /-- Empty order-characteristic support makes every actual associated-graded
37 piece of the induced quotient filtration subsingleton.  The argument uses
38 only the global additive equivalence with the cyclic symbol quotient. -/
39 theorem all_graded_subsingleton_of_orderCharacteristicSupport_eq_empty
40     (I : RightIdeal (WeylAlg k n))
41     (hsupport : orderCharacteristicSupport k I = ∅) :
42     ∀ N, Subsingleton (QuotientOrderGradedPiece k I N) := by
43   have htop : orderInitialIdeal k I = ⊤ :=
44     (orderCharacteristicSupport_eq_empty_iff k I).mp hsupport
45   letI : Subsingleton (OrderCharacteristicModule k I) := by
46     rw [Submodule.Quotient.subsingleton_iff]
47     exact htop
48   letI : Subsingleton (QuotientOrderAssociatedGraded k I) :=
49     (quotientOrderAssociatedGradedAddEquivCharacteristic k I).toEquiv.subsingleton
50   intro N
51   exact ⟨fun q r => DirectSum.of_injective N (Subsingleton.elim
52     (DirectSum.of (fun M => QuotientOrderGradedPiece k I M) N q)
53     (DirectSum.of (fun M => QuotientOrderGradedPiece k I M) N r))⟩
54 
55 /-- Empty order-characteristic support reflects through the actual associated
56 graded pieces to vanishing of the original right-module quotient. -/
57 theorem rightQuotient_subsingleton_of_orderCharacteristicSupport_eq_empty
58     (I : RightIdeal (WeylAlg k n))
59     (hsupport : orderCharacteristicSupport k I = ∅) :
60     Subsingleton (RightQuotient I) :=
61   rightQuotient_subsingleton_of_all_graded_subsingleton k I
62     (all_graded_subsingleton_of_orderCharacteristicSupport_eq_empty k I hsupport)
63 
64 /-- Vanishing of the actual quotient also forces every filtered graded piece
65 to vanish. -/
66 theorem all_graded_subsingleton_of_rightQuotient_subsingleton
67     (I : RightIdeal (WeylAlg k n))
68     (hquotient : Subsingleton (RightQuotient I)) :
69     ∀ N, Subsingleton (QuotientOrderGradedPiece k I N) := by
70   letI : Subsingleton (RightQuotient I) := hquotient
71   letI : Subsingleton (FilteredRightQuotient k I) :=
72     (filteredRightQuotientEquivRightQuotient k I).toEquiv.subsingleton
73   intro N
74   constructor
75   intro q r
76   refine Submodule.Quotient.induction_on _ q ?_
77   intro a
78   refine Submodule.Quotient.induction_on _ r ?_
79   intro b
80   exact congrArg Submodule.Quotient.mk
81     (Subtype.ext (Subsingleton.elim a.1 b.1))
82 
83 /-- For the concrete order filtration, the initial ideal is the unit ideal
84 exactly when the original right-module quotient vanishes. -/
85 theorem orderInitialIdeal_eq_top_iff_rightQuotient_subsingleton
86     (I : RightIdeal (WeylAlg k n)) :
87     orderInitialIdeal k I = ⊤ ↔ Subsingleton (RightQuotient I) := by
88   constructor
89   · intro htop
90     apply rightQuotient_subsingleton_of_orderCharacteristicSupport_eq_empty k I
91     exact (orderCharacteristicSupport_eq_empty_iff k I).mpr htop
92   · intro hquotient
93     letI : ∀ N, Subsingleton (QuotientOrderGradedPiece k I N) :=
94       all_graded_subsingleton_of_rightQuotient_subsingleton k I hquotient
95     letI : Subsingleton (QuotientOrderAssociatedGraded k I) := by
96       infer_instance
97     letI : Subsingleton (OrderCharacteristicModule k I) :=
98       (Equiv.subsingleton_congr
99         (quotientOrderAssociatedGradedAddEquivCharacteristic k I).toEquiv).mp
100           (by infer_instance)
101     exact (Submodule.Quotient.subsingleton_iff).mp (by infer_instance)
102 
103 /-- Clean support formulation of the same vanishing criterion. -/
104 theorem orderCharacteristicSupport_eq_empty_iff_rightQuotient_subsingleton
105     (I : RightIdeal (WeylAlg k n)) :
106     orderCharacteristicSupport k I = ∅ ↔ Subsingleton (RightQuotient I) := by
107   rw [orderCharacteristicSupport_eq_empty_iff,
108     orderInitialIdeal_eq_top_iff_rightQuotient_subsingleton]
