10 /-!
11 # Scheme-theoretic exclusion of the punctured normal fibre axis
12 
13 The canonical order symbol is homogeneous, fibre-only, and has unit pure
14 normal coefficient.  Hence a characteristic-support prime containing every
15 tangential momentum variable must also contain the normal momentum variable.
16 This is the exact prime-ideal form of normal-axis exclusion.  It proves no
17 filtered strictness or noncharacteristic inverse-image theorem.
18 -/

39 /-- The phase variables other than the distinguished normal momentum that may
40 occur in the canonical principal symbol. -/
41 def tangentialMomentumVariables (n : ℕ) : Set (PhaseVar (n + 1)) :=
42   {v | ∃ i : Fin (n + 1), i ≠ 0 ∧ v = .inr i}
43 
44 /-- A fibre-only polynomial uses only momentum variables. -/
45 theorem vars_subset_momentum_of_isFibreOnly
46     {P : SymbolRing k (n + 1)} (hP : IsFibreOnly k P) :
47     (P.vars : Set (PhaseVar (n + 1))) ⊆ Set.range Sum.inr := by
48   classical
49   intro v hv
50   rcases v with i | i
51   · obtain ⟨m, hm, hmi⟩ := (MvPolynomial.mem_vars _).mp hv
52     have hzero := hP m (MvPolynomial.mem_support_iff.mp hm) i
53     exact False.elim ((Finsupp.mem_support_iff.mp hmi) hzero)
54   · exact ⟨i, rfl⟩
55 
56 /-- The variables of a fibre-only polynomial split into the distinguished
57 normal momentum and the tangential momenta. -/
58 theorem vars_subset_normal_insert_tangential
59     {P : SymbolRing k (n + 1)} (hP : IsFibreOnly k P) :
60     (P.vars : Set (PhaseVar (n + 1))) ⊆
61       insert (.inr (0 : Fin (n + 1))) (tangentialMomentumVariables n) := by
62   intro v hv
63   obtain ⟨i, rfl⟩ := vars_subset_momentum_of_isFibreOnly hP hv
64   by_cases hi : i = 0
65   · left
66     rw [hi]
67   · right
68     exact ⟨i, hi, rfl⟩

70 /-- Every canonical characteristic-support prime containing all tangential
71 momentum variables contains the distinguished normal momentum variable. -/
72 theorem normalMomentum_mem_of_mem_canonicalSupport
73     (d : WeylAlg k (n + 1))
74     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
75     {p : PrimeSpectrum (SymbolRing k (n + 1))}
76     (hp : p ∈ orderCharacteristicSupport k
77       (canonicalRightIdeal (presentedCoordinate k n) d N))
78     (htan : ∀ i : Fin (n + 1), i ≠ 0 →
79       MvPolynomial.X (.inr i) ∈ p.asIdeal) :
80     MvPolynomial.X (.inr (0 : Fin (n + 1))) ∈ p.asIdeal := by
81   let P := presentedPrincipalComponent k orderWeight N d
82   have hPmem : P ∈ p.asIdeal := by
83     rw [orderCharacteristicSupport_eq_zeroLocus,
84       PrimeSpectrum.mem_zeroLocus] at hp
85     exact hp (canonical_orderPrincipalComponent_mem_initialIdeal k n N hd)
86   apply AlgebraicAnalysis.MvPolynomial.X_mem_of_homogeneous_mem_prime
87     (canonical_orderPrincipalComponent_isHomogeneous n N hd)
88     (vars_subset_normal_insert_tangential
89       (canonical_orderPrincipalComponent_isFibreOnly k n N hd))
90     (canonical_orderPrincipalComponent_pureMomentumCoefficient k n N hd)
91     isUnit_one p.isPrime hPmem
92   intro v hv
93   obtain ⟨i, hi, rfl⟩ := hv
94   exact htan i hi

96 /-- Equivalently, no canonical support prime lies on the punctured normal
97 momentum axis. -/
98 theorem no_canonicalSupport_prime_on_punctured_normal_axis
99     (d : WeylAlg k (n + 1))
100     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
101     ¬ ∃ p ∈ orderCharacteristicSupport k
102         (canonicalRightIdeal (presentedCoordinate k n) d N),
103       (∀ i : Fin (n + 1), i ≠ 0 →
104         MvPolynomial.X (.inr i) ∈ p.asIdeal) ∧
105       MvPolynomial.X (.inr (0 : Fin (n + 1))) ∉ p.asIdeal := by
106   rintro ⟨p, hp, htan, hnormal⟩
107   exact hnormal (normalMomentum_mem_of_mem_canonicalSupport d hd hp htan)

109 /-- The same punctured-axis exclusion for the transposed associated-graded
110 support used by the left-module noncharacteristic theorem. -/
111 theorem normalMomentum_mem_of_mem_transposedCanonicalSupport
112     (d : WeylAlg k (n + 1))
113     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
114     {p : PrimeSpectrum (SymbolRing k (n + 1))}
115     (hp : p ∈ transposedOrderAssociatedGradedSupport k
116       (canonicalRightIdeal (presentedCoordinate k n) d N))
117     (htan : ∀ i : Fin (n + 1), i ≠ 0 →
118       MvPolynomial.X (.inr i) ∈ p.asIdeal) :
119     MvPolynomial.X (.inr (0 : Fin (n + 1))) ∈ p.asIdeal := by
120   let τ : SymbolRing k (n + 1) →+* SymbolRing k (n + 1) :=
121     (symbolTranspositionEquiv k).toRingEquiv.toRingHom
122   let q : PrimeSpectrum (SymbolRing k (n + 1)) :=
123     PrimeSpectrum.comap τ p
124   have hqModule : q ∈ Module.support (SymbolRing k (n + 1))
125       (OrderAssociatedGradedModule k
126         (canonicalRightIdeal (presentedCoordinate k n) d N)) := by
127     rw [transposedOrderAssociatedGradedSupport_eq_preimage] at hp
128     exact hp
129   have hq : q ∈ orderCharacteristicSupport k
130       (canonicalRightIdeal (presentedCoordinate k n) d N) := by
131     rw [orderCharacteristicSupport_eq_zeroLocus]
132     rw [Module.support_eq_zeroLocus,
133       annihilator_orderAssociatedGradedModule] at hqModule
134     exact hqModule
135   have hqtan : ∀ i : Fin (n + 1), i ≠ 0 →
136       MvPolynomial.X (.inr i) ∈ q.asIdeal := by
137     intro i hi
138     change symbolTransposition k (MvPolynomial.X (.inr i)) ∈ p.asIdeal
139     rw [symbolTransposition_momentum]
140     exact p.asIdeal.neg_mem (htan i hi)
141   have hqnormal := normalMomentum_mem_of_mem_canonicalSupport d hd hq hqtan
142   change symbolTransposition k
143       (MvPolynomial.X (.inr (0 : Fin (n + 1)))) ∈ p.asIdeal at hqnormal
144   rw [symbolTransposition_momentum] at hqnormal
145   simpa using p.asIdeal.neg_mem hqnormal
