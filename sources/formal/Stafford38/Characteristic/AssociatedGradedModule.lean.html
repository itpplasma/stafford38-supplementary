9 # The symbol-ring module on the actual associated graded quotient
10 
11 The filtered quotient construction already gives the actual degree pieces of
12 `A / I`, their external direct sum, and an exact global symbol map whose kernel
13 is `orderInitialIdeal`.  This file uses that exact bridge to put the genuine
14 `SymbolRing`-module structure on a dedicated copy of the actual associated
15 graded object.
16 
17 The construction is not merely additive: the resulting module is linearly
18 equivalent over `SymbolRing` to its cyclic quotient by `orderInitialIdeal`, and
19 its annihilator is proved equal (not merely comparable) to that ideal.  No
20 Rees algebra, radical involutivity, or Gabber theorem is asserted here.
21 -/

74 ideal quotient is also `k`-linear. -/

114 /-- The actual associated graded quotient is a genuine module over the symbol
115 ring, linearly equivalent over that ring to the cyclic quotient by the order
116 initial ideal. -/
117 def orderAssociatedGradedLinearEquivCharacteristic
118     (I : RightIdeal (PresentedWeyl k n)) :
119     OrderAssociatedGradedModule k I ≃ₗ[SymbolRing k n]
120       OrderCharacteristicModule k I :=
121   { orderAssociatedGradedAddEquivCharacteristic k I with
122     map_smul' := by
123       intro P q
124       change orderAssociatedGradedAddEquivCharacteristic k I
125           ((orderAssociatedGradedAddEquivCharacteristic k I).symm
126             (P • orderAssociatedGradedAddEquivCharacteristic k I q)) =
127       P • orderAssociatedGradedAddEquivCharacteristic k I q
128       exact (orderAssociatedGradedAddEquivCharacteristic k I).apply_symm_apply _ }

144 /-- The global characteristic equivalence sends the class of a filtered
145 representative in degree `N` to the class of its degree-`N` principal symbol.
146 This is the commuting-square statement that connects the transported module
147 to the actual filtered quotient pieces. -/
148 theorem orderAssociatedGradedLinearEquivCharacteristic_of_mk
149     (I : RightIdeal (PresentedWeyl k n)) (N : ℕ)
150     (z : orderPiece k n N) :
151     orderAssociatedGradedLinearEquivCharacteristic k I
152         (orderAssociatedGradedOf k I N
153           (orderPieceToQuotientGraded k I N z)) =
154       Ideal.Quotient.mk (orderInitialIdeal k I)
155         (presentedPrincipalComponent k orderWeight N z) := by
156   change restrictScalarsQuotientLinearEquiv k I
157       (quotientOrderAssociatedGradedEquivCharacteristic k I
158         (orderAssociatedGradedOf k I N
159           (orderPieceToQuotientGraded k I N z))) = _
160   have hsource :
161       quotientOrderAssociatedGradedEquivCharacteristic k I
162           (orderAssociatedGradedOf k I N
163             (orderPieceToQuotientGraded k I N z)) =
164         Submodule.Quotient.mk
165           (presentedPrincipalComponent k orderWeight N z) := by
166     have htrans := LinearEquiv.trans_apply
167       (e₁₂ := quotientOrderAssociatedGradedEquivSymbols k I)
168       (e₂₃ := (orderCharacteristicModuleEquivRelationGraded k I).symm)
169       (orderAssociatedGradedOf k I N (orderPieceToQuotientGraded k I N z))
170     rw [quotientOrderAssociatedGradedEquivCharacteristic, htrans]
171     apply (orderCharacteristicModuleEquivRelationGraded k I).injective
172     rw [LinearEquiv.apply_symm_apply]
173     apply DFinsupp.ext
174     intro M
175     rw [orderCharacteristicModuleEquivRelationGraded_mk]
176     rw [symbolToOrderRelationGraded_apply]
177     change
178       ((DFinsupp.mapRange.linearEquiv fun L =>
179           quotientOrderGradedPieceEquivSymbols k I L)
180         (DirectSum.of (fun L => QuotientOrderGradedPiece k I L) N
181           (orderPieceToQuotientGraded k I N z))) M =
182         Submodule.Quotient.mk
183           ((DirectSum.decompose
184             (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n))
185             (principalComponentOnPiece k (@orderWeight n) N z :
186               SymbolRing k n)) M)
187     rw [DFinsupp.mapRange.linearEquiv_apply, DirectSum.decompose_coe]
188     change quotientOrderGradedPieceEquivSymbols k I M
189         ((DirectSum.of (fun L => QuotientOrderGradedPiece k I L) N
190           (orderPieceToQuotientGraded k I N z)) M) =
191       Submodule.Quotient.mk
192         ((DirectSum.of (fun L =>
193           MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) L) N
194           (principalComponentOnPiece k (@orderWeight n) N z)) M)
195     by_cases hMN : N = M
196     · subst M
197       rw [DirectSum.of_eq_same, DirectSum.of_eq_same,
198         quotientOrderGradedPieceEquivSymbols_mk]
199     · rw [DirectSum.of_eq_of_ne _ _ _ (Ne.symm hMN),
200         DirectSum.of_eq_of_ne _ _ _ (Ne.symm hMN)]
201       rw [(quotientOrderGradedPieceEquivSymbols k I M).map_zero]
202       exact (Submodule.Quotient.mk_zero _).symm
203   calc
204     _ = restrictScalarsQuotientLinearEquiv k I
205         (Submodule.Quotient.mk
206           (presentedPrincipalComponent k orderWeight N z)) :=
207       congrArg (restrictScalarsQuotientLinearEquiv k I) hsource
208     _ = _ := restrictScalarsQuotientLinearEquiv_mk k I _

223 /-- Multiplication by a homogeneous symbol on the transported global module
224 is exactly the canonical degree-shifting action on the actual quotient graded
225 piece.  The right side is the action constructed from right multiplication in
226 the filtered Weyl quotient, not a second transported definition. -/
227 theorem smul_orderAssociatedGradedOf_eq_of_homogeneousAction
228     (I : RightIdeal (PresentedWeyl k n)) {N M : ℕ}
229     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M)
230     (q : QuotientOrderGradedPiece k I N) :
231     (P : SymbolRing k n) • orderAssociatedGradedOf k I N q =
232       orderAssociatedGradedOf k I (N + M)
233         (quotientOrderHomogeneousAction k I P q) := by
234   obtain ⟨z, rfl⟩ := orderPieceToQuotientGraded_surjective k I N q
235   obtain ⟨y, hy⟩ :=
236     principalComponentOnPiece_surjective k (@orderWeight n) P
237   subst P
238   apply (orderAssociatedGradedLinearEquivCharacteristic k I).injective
239   rw [map_smul, orderAssociatedGradedLinearEquivCharacteristic_of_mk,
240     quotientOrderHomogeneousAction_mk_mul,
241     orderAssociatedGradedLinearEquivCharacteristic_of_mk,
242     symbol_smul_characteristicModule_mk]
243   apply congrArg (Ideal.Quotient.mk (orderInitialIdeal k I))
244   rw [presentedPrincipalComponent_mul_order k z.property y.property]
245   exact mul_comm _ _

247 /-- Representative form of homogeneous compatibility: the symbol of `y`
248 acting on the degree-`N` class of `z` is the degree-`N+M` class represented by
249 the written-order Weyl product `z * y`. -/
250 theorem smul_orderAssociatedGradedOf_mk_eq_of_mul
251     (I : RightIdeal (PresentedWeyl k n)) {N M : ℕ}
252     (z : orderPiece k n N) (y : orderPiece k n M) :
253     (principalComponentOnPiece k (@orderWeight n) M y : SymbolRing k n) •
254         orderAssociatedGradedOf k I N
255           (orderPieceToQuotientGraded k I N z) =
256       orderAssociatedGradedOf k I (N + M)
257         (orderPieceToQuotientGraded k I (N + M)
258           ⟨(z : PresentedWeyl k n) * y,
259             mul_mem_orderPiece k z.property y.property⟩) := by
260   rw [smul_orderAssociatedGradedOf_eq_of_homogeneousAction,
261     quotientOrderHomogeneousAction_mk_mul]

263 /-- The unit operator belongs to the degree-zero order piece. -/
264 def orderPieceOne : orderPiece k n 0 :=
265   ⟨1, by

335 /-- Every element of the actual associated graded quotient is a symbol
336 multiple of the distinguished generator. -/
337 theorem exists_smul_orderAssociatedGradedGenerator
338     (I : RightIdeal (PresentedWeyl k n))
339     (q : OrderAssociatedGradedModule k I) :
340     ∃ P : SymbolRing k n, P • orderAssociatedGradedGenerator k I = q := by
341   obtain ⟨P, hP⟩ := Ideal.Quotient.mk_surjective
342     (orderAssociatedGradedLinearEquivCharacteristic k I q)
343   refine ⟨P, ?_⟩
344   apply (orderAssociatedGradedLinearEquivCharacteristic k I).injective
345   rw [map_smul, orderAssociatedGradedLinearEquivCharacteristic_generator]
346   rw [symbol_smul_one_characteristicModule]
347   exact hP

349 /-- A symbol kills the cyclic generator exactly when it belongs to the order
350 initial ideal.  This is the elementwise form of the exact annihilator theorem. -/
351 theorem smul_orderAssociatedGradedGenerator_eq_zero_iff
352     (I : RightIdeal (PresentedWeyl k n)) (P : SymbolRing k n) :
353     P • orderAssociatedGradedGenerator k I = 0 ↔
354       P ∈ orderInitialIdeal k I := by
355   rw [← Ideal.Quotient.eq_zero_iff_mem]
356   constructor
357   · intro h
358     have h' : P • (1 : OrderCharacteristicModule k I) = 0 := by
359       simpa only [map_smul,
360         orderAssociatedGradedLinearEquivCharacteristic_generator, map_zero]
361         using congrArg (orderAssociatedGradedLinearEquivCharacteristic k I) h
362     rw [symbol_smul_one_characteristicModule] at h'
363     exact h'
364   · intro h
365     apply (orderAssociatedGradedLinearEquivCharacteristic k I).injective
366     rw [map_smul, orderAssociatedGradedLinearEquivCharacteristic_generator,
367       map_zero]
368     rw [symbol_smul_one_characteristicModule]
369     exact h
