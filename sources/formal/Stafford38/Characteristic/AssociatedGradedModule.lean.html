10 /-!
11 # The symbol-ring module on the actual associated graded quotient
12 
13 The filtered quotient construction already gives the actual degree pieces of
14 `A / I`, their external direct sum, and an exact global symbol map whose kernel
15 is `orderInitialIdeal`.  This file uses that exact bridge to put the genuine
16 `SymbolRing`-module structure on a dedicated copy of the actual associated
17 graded object.
18 
19 The construction is not merely additive: the resulting module is linearly
20 equivalent over `SymbolRing` to its cyclic quotient by `orderInitialIdeal`, and
21 its annihilator is proved equal (not merely comparable) to that ideal.  No
22 Rees algebra, radical involutivity, or Gabber theorem is asserted here.
23 -/

116 /-- The actual associated graded quotient is a genuine module over the symbol
117 ring, linearly equivalent over that ring to the cyclic quotient by the order
118 initial ideal. -/
119 def orderAssociatedGradedLinearEquivCharacteristic
120     (I : RightIdeal (WeylAlg k n)) :
121     OrderAssociatedGradedModule k I ≃ₗ[SymbolRing k n]
122       OrderCharacteristicModule k I :=
123   { orderAssociatedGradedAddEquivCharacteristic k I with
124     map_smul' := by
125       intro P q
126       change orderAssociatedGradedAddEquivCharacteristic k I
127           ((orderAssociatedGradedAddEquivCharacteristic k I).symm
128             (P • orderAssociatedGradedAddEquivCharacteristic k I q)) =
129       P • orderAssociatedGradedAddEquivCharacteristic k I q
130       exact (orderAssociatedGradedAddEquivCharacteristic k I).apply_symm_apply _ }

146 /-- The global characteristic equivalence sends the class of a filtered
147 representative in degree `N` to the class of its degree-`N` principal symbol.
148 This is the commuting-square statement that connects the transported module
149 to the actual filtered quotient pieces. -/
150 theorem orderAssociatedGradedLinearEquivCharacteristic_of_mk
151     (I : RightIdeal (WeylAlg k n)) (N : ℕ)
152     (z : orderPiece k n N) :
153     orderAssociatedGradedLinearEquivCharacteristic k I
154         (orderAssociatedGradedOf k I N
155           (orderPieceToQuotientGraded k I N z)) =
156       Ideal.Quotient.mk (orderInitialIdeal k I)
157         (presentedPrincipalComponent k orderWeight N z) := by
158   change restrictScalarsQuotientLinearEquiv k I
159       (quotientOrderAssociatedGradedEquivCharacteristic k I
160         (orderAssociatedGradedOf k I N
161           (orderPieceToQuotientGraded k I N z))) = _
162   have hsource :
163       quotientOrderAssociatedGradedEquivCharacteristic k I
164           (orderAssociatedGradedOf k I N
165             (orderPieceToQuotientGraded k I N z)) =
166         Submodule.Quotient.mk
167           (presentedPrincipalComponent k orderWeight N z) := by
168     have htrans := LinearEquiv.trans_apply
169       (e₁₂ := quotientOrderAssociatedGradedEquivSymbols k I)
170       (e₂₃ := (orderCharacteristicModuleEquivRelationGraded k I).symm)
171       (orderAssociatedGradedOf k I N (orderPieceToQuotientGraded k I N z))
172     rw [quotientOrderAssociatedGradedEquivCharacteristic, htrans]
173     apply (orderCharacteristicModuleEquivRelationGraded k I).injective
174     rw [LinearEquiv.apply_symm_apply]
175     apply DFinsupp.ext
176     intro M
177     rw [orderCharacteristicModuleEquivRelationGraded_mk]
178     rw [symbolToOrderRelationGraded_apply]
179     change
180       ((DFinsupp.mapRange.linearEquiv fun L =>
181           quotientOrderGradedPieceEquivSymbols k I L)
182         (DirectSum.of (fun L => QuotientOrderGradedPiece k I L) N
183           (orderPieceToQuotientGraded k I N z))) M =
184         Submodule.Quotient.mk
185           ((DirectSum.decompose
186             (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n))
187             (principalComponentOnPiece k (@orderWeight n) N z :
188               SymbolRing k n)) M)
189     rw [DFinsupp.mapRange.linearEquiv_apply, DirectSum.decompose_coe]
190     change quotientOrderGradedPieceEquivSymbols k I M
191         ((DirectSum.of (fun L => QuotientOrderGradedPiece k I L) N
192           (orderPieceToQuotientGraded k I N z)) M) =
193       Submodule.Quotient.mk
194         ((DirectSum.of (fun L =>
195           MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) L) N
196           (principalComponentOnPiece k (@orderWeight n) N z)) M)
197     by_cases hMN : N = M
198     · subst M
199       rw [DirectSum.of_eq_same, DirectSum.of_eq_same,
200         quotientOrderGradedPieceEquivSymbols_mk]
201     · rw [DirectSum.of_eq_of_ne _ _ _ (Ne.symm hMN),
202         DirectSum.of_eq_of_ne _ _ _ (Ne.symm hMN)]
203       rw [(quotientOrderGradedPieceEquivSymbols k I M).map_zero]
204       exact (Submodule.Quotient.mk_zero _).symm
205   calc
206     _ = restrictScalarsQuotientLinearEquiv k I
207         (Submodule.Quotient.mk
208           (presentedPrincipalComponent k orderWeight N z)) :=
209       congrArg (restrictScalarsQuotientLinearEquiv k I) hsource
210     _ = _ := restrictScalarsQuotientLinearEquiv_mk k I _

225 /-- Multiplication by a homogeneous symbol on the transported global module
226 is exactly the canonical degree-shifting action on the actual quotient graded
227 piece.  The right side is the action constructed from right multiplication in
228 the filtered Weyl quotient, not a second transported definition. -/
229 theorem smul_orderAssociatedGradedOf_eq_of_homogeneousAction
230     (I : RightIdeal (WeylAlg k n)) {N M : ℕ}
231     (P : MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n) M)
232     (q : QuotientOrderGradedPiece k I N) :
233     (P : SymbolRing k n) • orderAssociatedGradedOf k I N q =
234       orderAssociatedGradedOf k I (N + M)
235         (quotientOrderHomogeneousAction k I P q) := by
236   obtain ⟨z, rfl⟩ := orderPieceToQuotientGraded_surjective k I N q
237   obtain ⟨y, hy⟩ :=
238     principalComponentOnPiece_surjective k (@orderWeight n) P
239   subst P
240   apply (orderAssociatedGradedLinearEquivCharacteristic k I).injective
241   rw [map_smul, orderAssociatedGradedLinearEquivCharacteristic_of_mk,
242     quotientOrderHomogeneousAction_mk_mul,
243     orderAssociatedGradedLinearEquivCharacteristic_of_mk,
244     symbol_smul_characteristicModule_mk]
245   apply congrArg (Ideal.Quotient.mk (orderInitialIdeal k I))
246   rw [presentedPrincipalComponent_mul_order k z.property y.property]
247   exact mul_comm _ _

249 /-- Representative form of homogeneous compatibility: the symbol of `y`
250 acting on the degree-`N` class of `z` is the degree-`N+M` class represented by
251 the written-order Weyl product `z * y`. -/
252 theorem smul_orderAssociatedGradedOf_mk_eq_of_mul
253     (I : RightIdeal (WeylAlg k n)) {N M : ℕ}
254     (z : orderPiece k n N) (y : orderPiece k n M) :
255     (principalComponentOnPiece k (@orderWeight n) M y : SymbolRing k n) •
256         orderAssociatedGradedOf k I N
257           (orderPieceToQuotientGraded k I N z) =
258       orderAssociatedGradedOf k I (N + M)
259         (orderPieceToQuotientGraded k I (N + M)
260           ⟨(z : WeylAlg k n) * y,
261             mul_mem_orderPiece k z.property y.property⟩) := by
262   rw [smul_orderAssociatedGradedOf_eq_of_homogeneousAction,
263     quotientOrderHomogeneousAction_mk_mul]

265 /-- The unit operator belongs to the degree-zero order piece. -/
266 def orderPieceOne : orderPiece k n 0 :=
267   ⟨1, by
268     rw [orderPiece, mem_presentedWeightPiece]
269     intro m hm
270     rw [presentedNormalFormLinearEquiv_one] at hm

337 /-- Every element of the actual associated graded quotient is a symbol
338 multiple of the distinguished generator. -/
339 theorem exists_smul_orderAssociatedGradedGenerator
340     (I : RightIdeal (WeylAlg k n))
341     (q : OrderAssociatedGradedModule k I) :
342     ∃ P : SymbolRing k n, P • orderAssociatedGradedGenerator k I = q := by
343   obtain ⟨P, hP⟩ := Ideal.Quotient.mk_surjective
344     (orderAssociatedGradedLinearEquivCharacteristic k I q)
345   refine ⟨P, ?_⟩
346   apply (orderAssociatedGradedLinearEquivCharacteristic k I).injective
347   rw [map_smul, orderAssociatedGradedLinearEquivCharacteristic_generator]
348   rw [symbol_smul_one_characteristicModule]
349   exact hP

351 /-- A symbol kills the cyclic generator exactly when it belongs to the order
352 initial ideal.  This is the elementwise form of the exact annihilator theorem. -/
353 theorem smul_orderAssociatedGradedGenerator_eq_zero_iff
354     (I : RightIdeal (WeylAlg k n)) (P : SymbolRing k n) :
355     P • orderAssociatedGradedGenerator k I = 0 ↔
356       P ∈ orderInitialIdeal k I := by
357   rw [← Ideal.Quotient.eq_zero_iff_mem]
358   constructor
359   · intro h
360     have h' : P • (1 : OrderCharacteristicModule k I) = 0 := by
361       simpa only [map_smul,
362         orderAssociatedGradedLinearEquivCharacteristic_generator, map_zero]
363         using congrArg (orderAssociatedGradedLinearEquivCharacteristic k I) h
364     rw [symbol_smul_one_characteristicModule] at h'
365     exact h'
366   · intro h
367     apply (orderAssociatedGradedLinearEquivCharacteristic k I).injective
368     rw [map_smul, orderAssociatedGradedLinearEquivCharacteristic_generator,
369       map_zero]
370     rw [symbol_smul_one_characteristicModule]
371     exact h
