7 # Annihilator closure from a square-zero deformation
8 
9 The exact parameter sequence already forces the annihilator of the special
10 fibre to be closed under the recorded first-order bracket.  The proof uses
11 only right-action order, centrality and square-zero exactness; no localization,
12 trace, finiteness or commutative-algebra hypothesis beyond the existing
13 commutative special fibre is needed.
14 
15 This proves closure only when both bracket inputs lie in the annihilator
16 itself.  It does not prove bracket closure of its radical or of any prime ideal
17 for inputs that merely lie in that larger ideal.
18 -/

38 /-- If the specialization of `a` annihilates `G`, right multiplication by
39 `a` on the deformation module lands in the image of the parameter action. -/
40 theorem rightAction_mem_range_cAct_of_pi_mem_annihilator
41     {a : B} (ha : D.pi a ∈ Module.annihilator R G) (m : N) :
42     MulOpposite.op a • m ∈ LinearMap.range D.cAct := by
43   rw [← D.rho_ker, LinearMap.mem_ker]
44   rw [D.rho_action]
45   exact Module.mem_annihilator.mp ha (D.rho m)
46 
47 /-- Centrality of the parameter lets every right action pass through its
48 recorded action on `N`. -/
49 theorem rightAction_cAct_comm (a : B) (m : N) :
50     MulOpposite.op a • D.cAct m = D.cAct (MulOpposite.op a • m) := by
51   rw [D.cAct_apply, D.cAct_apply, ← mul_smul, ← mul_smul]
52   change MulOpposite.op (D.c * a) • m = MulOpposite.op (a * D.c) • m
53   rw [D.c_center.comm a]
54 
55 /-- Two successive right actions vanish when both specializations annihilate
56 the special fibre.  The displayed order is literal: first `a`, then `b`. -/
57 theorem iterated_rightAction_eq_zero_of_pi_mem_annihilator
58     {a b : B}
59     (ha : D.pi a ∈ Module.annihilator R G)
60     (hb : D.pi b ∈ Module.annihilator R G) (m : N) :
61     MulOpposite.op b • (MulOpposite.op a • m) = 0 := by
62   obtain ⟨u, hu⟩ :=
63     rightAction_mem_range_cAct_of_pi_mem_annihilator D ha m
64   rw [← hu, rightAction_cAct_comm]
65   obtain ⟨v, hv⟩ :=
66     rightAction_mem_range_cAct_of_pi_mem_annihilator D hb u
67   rw [← hv]
68   change (D.cAct.comp D.cAct) v = 0
69   rw [D.cAct_comp_self]
70   rfl
71 
72 /-- Consequently the right action of the deformation-ring ringCommutator
73 vanishes.  The reversal `[R_b,R_a] = R_[a,b]` is explicit. -/
74 theorem commutator_rightAction_eq_zero_of_pi_mem_annihilator
75     {a b : B}
76     (ha : D.pi a ∈ Module.annihilator R G)
77     (hb : D.pi b ∈ Module.annihilator R G) (m : N) :
78     MulOpposite.op (AlgebraicAnalysis.ringCommutator a b) • m = 0 := by
79   have hab :=
80     iterated_rightAction_eq_zero_of_pi_mem_annihilator D ha hb m
81   have hba :=
82     iterated_rightAction_eq_zero_of_pi_mem_annihilator D hb ha m
83   rw [← rightActionEnd_apply (k := k) (B := B) (N := N),
84     ← rightAction_commutator (k := k) (B := B) (N := N) a b]
85   change MulOpposite.op b • (MulOpposite.op a • m) -
86     MulOpposite.op a • (MulOpposite.op b • m) = 0
87   rw [hab, hba, sub_self]
88 
89 /-- If `[a,b]=c*z`, then the quotient `z` acts trivially after module
90 specialization.  The explicit left factor `c*z` is essential here: centrality
91 moves it to the right-module order needed by `cAct`. -/
92 theorem rho_rightAction_eq_zero_of_commutator_factor
93     {a b z : B}
94     (ha : D.pi a ∈ Module.annihilator R G)
95     (hb : D.pi b ∈ Module.annihilator R G)
96     (hz : AlgebraicAnalysis.ringCommutator a b = D.c * z) (m : N) :
97     D.rho (MulOpposite.op z • m) = 0 := by
98   have hcomm :=
99     commutator_rightAction_eq_zero_of_pi_mem_annihilator D ha hb m
100   have hcAct : D.cAct (MulOpposite.op z • m) = 0 := by
101     calc
102       D.cAct (MulOpposite.op z • m) =
103           MulOpposite.op D.c • (MulOpposite.op z • m) := D.cAct_apply _
104       _ = MulOpposite.op (z * D.c) • m := by rw [← mul_smul]; rfl
105       _ = MulOpposite.op (D.c * z) • m := by rw [D.c_center.comm z]
106       _ = MulOpposite.op (AlgebraicAnalysis.ringCommutator a b) • m := by rw [hz]
107       _ = 0 := hcomm
108   have hm : MulOpposite.op z • m ∈ LinearMap.range D.cAct := by
109     rw [← D.c_exact, LinearMap.mem_ker]
110     exact hcAct
111   rw [← D.rho_ker] at hm
112   exact LinearMap.mem_ker.mp hm
113 
114 /-- The recorded bracket preserves the annihilator of the special fibre in
115 both inputs. -/
116 theorem bracket_mem_annihilator
117     {x y : R}
118     (hx : x ∈ Module.annihilator R G)
119     (hy : y ∈ Module.annihilator R G) :
120     D.bracket x y ∈ Module.annihilator R G := by
121   obtain ⟨a, b, z, ha, hb, hz, hbracket⟩ :=
122     RightSquareZeroTraceData.exists_lifts_commutator_factor
123       (k := k) (B := B) (N := N) D x y
124   rw [Module.mem_annihilator]
125   intro g
126   obtain ⟨m, rfl⟩ := D.rho_surjective g
127   have hzrho := rho_rightAction_eq_zero_of_commutator_factor D
128     (ha ▸ hx) (hb ▸ hy) hz m
129   rw [D.rho_action] at hzrho
130   simpa [hbracket] using hzrho
