111     targetGradedEquiv k n N d hd (tangentialDrop k n N d hd 0 z) =
112       oldCoordinateMap (k := k) n (Graded k n N d) (sourceGradedEquiv k n N d hd z) :=

115 def zeroDropKernelEquiv :
116     (tangentialDrop k n N d hd 0).ker ≃ₗ[T k n]
117       (oldCoordinateMap (k := k) n (Graded k n N d)).ker :=
118   LinearEquiv.ofBijective
119     ((sourceGradedEquiv k n N d hd).toLinearMap.domRestrict
120       (tangentialDrop k n N d hd 0).ker |>.codRestrict
121         (oldCoordinateMap (k := k) n (Graded k n N d)).ker (fun z => by
122           exact LinearMap.mem_ker.mpr (by
123             change oldCoordinateMap (k := k) n (Graded k n N d)
124               (sourceGradedEquiv k n N d hd z.val) = 0
125             rw [← gradedEquiv_drop]
126             simp only [LinearMap.mem_ker.mp z.property, map_zero])))
127     (by
128       constructor
129       · intro a b hab
130         exact Subtype.ext ((sourceGradedEquiv k n N d hd).injective
131           (congrArg Subtype.val hab))
132       · intro z
133         refine ⟨⟨(sourceGradedEquiv k n N d hd).symm z, ?_⟩,
134           Subtype.ext ((sourceGradedEquiv k n N d hd).apply_symm_apply z)⟩
135         apply LinearMap.mem_ker.mpr
136         apply (targetGradedEquiv k n N d hd).injective
137         rw [gradedEquiv_drop, LinearEquiv.apply_symm_apply, map_zero]
138         exact LinearMap.mem_ker.mp z.property)

141     Submodule.map (targetGradedEquiv k n N d hd).toLinearMap
142         (tangentialDrop k n N d hd 0).range =
143       (oldCoordinateMap (k := k) n (Graded k n N d)).range := by
144   ext z
