10 /-!
11 # Concrete lifted source-basis action equations
12 
13 This file constructs the first-order source-row matrix attached to an actual
14 square-zero deformation module.  The construction uses only the surjective
15 module specialization, its exact parameter kernel, and a basis of the special
16 fibre.  In particular, no scalar action on the deformation module by the
17 coefficient field is introduced.
18 
19 The row convention is literal: entry `(i,j)` is the coefficient of the lifted
20 basis vector `j` in the action on lifted basis vector `i`.  This is the
21 transpose of Mathlib's `LinearMap.toMatrix` convention.  For the localized
22 two-block module the abstract construction is instantiated with the descended
23 quotient-ring action and the localized opposite-ring specialization.
24 
25 The induced `z`-operator identity is not assumed in this file.  Its proof also
26 requires expanding the ringCommutator of the two constructed action equations and
27 descending the resulting parameter-multiple equality through exactness.
28 -/

59 /-- Additive action of the deformation parameter on the module. -/
60 def parameterAct (c : B) : W →+ W where
61   toFun w := c • w
62   map_zero' := smul_zero c
63   map_add' x y := smul_add c x y

65 @[simp] theorem parameterAct_apply (c : B) (w : W) :
66     parameterAct (W := W) c w = c • w := rfl

68 /-- A fixed lift of every basis vector through a surjective specialization.
69 Calling the construction for two operators therefore uses the same lifted
70 basis, as required in the source calculation. -/
71 def liftedBasisVector
72     (rho : W →+ V) (hrho : Function.Surjective rho)
73     (b : Module.Basis (Fin (r + 1)) K V) (i : Fin (r + 1)) : W :=
74   Classical.choose (hrho (b i))

76 @[simp] theorem rho_liftedBasisVector
77     (rho : W →+ V) (hrho : Function.Surjective rho)
78     (b : Module.Basis (Fin (r + 1)) K V) (i : Fin (r + 1)) :
79     rho (liftedBasisVector rho hrho b i) = b i :=
80   Classical.choose_spec (hrho (b i))

82 /-- Source-row coefficients of an endomorphism.  The first index names the
83 input basis vector, and the second names an output coefficient. -/
84 def sourceActionCoefficients
85     (b : Module.Basis (Fin (r + 1)) K V) (T : Module.End K V) :
86     Matrix (Fin (r + 1)) (Fin (r + 1)) Abar :=
87   fun i j ↦ algebraMap K Abar (b.repr (T (b i)) j)

89 /-- The source-row coefficient matrix is the transpose of Mathlib's column
90 matrix, followed by coefficient-field inclusion. -/
91 theorem sourceActionCoefficients_eq_transpose_toMatrix_map
92     (b : Module.Basis (Fin (r + 1)) K V) (T : Module.End K V) :
93     sourceActionCoefficients (Abar := Abar) b T =
94       (LinearMap.toMatrix b b T)ᵀ.map (algebraMap K Abar) := by
95   ext i j
96   simp [sourceActionCoefficients, LinearMap.toMatrix_apply, Matrix.map_apply]

98 /-- The row coefficients reconstruct the action on the named input vector. -/
99 theorem sum_sourceActionCoefficients_smul
100     (b : Module.Basis (Fin (r + 1)) K V) (T : Module.End K V)
101     (i : Fin (r + 1)) :
102     (∑ j, sourceActionCoefficients (Abar := Abar) b T i j • b j) =

122 /-- Exact first-order source-basis action equation.
123 
124 The matrix `Gamma` is constructed from the two exactness statements.  First,
125 the zeroth-order error is lifted through `ker rho = range(c)`.  Its chosen
126 preimage is expanded in the special-fibre basis; after lifting those scalar
127 coordinates, the remaining error again lies in `range(c)`, and multiplication
128 by the square-zero parameter kills it. -/
129 theorem exists_firstOrderSourceActionMatrix_over_coefficientField
130     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
131     (rho : W →+ V) (hrho : Function.Surjective rho)
132     (hrho_action : ∀ (a : B) (w : W),
133       rho (a • w) = S.modParameter a • rho w)
134     (hrho_ker : AddMonoidHom.ker rho =
135       AddMonoidHom.range (parameterAct (W := W) S.parameter))
136     (hparameter_sq : S.parameter * S.parameter = 0)
137     (b : Module.Basis (Fin (r + 1)) K V) (a : B) :
138     ∃ (Gamma : Matrix (Fin (r + 1)) (Fin (r + 1)) B)
139       (GammaK : Matrix (Fin (r + 1)) (Fin (r + 1)) K),
140       Gamma.map S.modParameter = GammaK.map (algebraMap K Abar) ∧
141       ∀ i,
142         a • liftedBasisVector rho hrho b i =
143           (∑ j, liftedSourceActionMatrix S b
144               (leftMultiplicationEnd (K := K) (V := V)
145                 (S.modParameter a)) i j •
146               liftedBasisVector rho hrho b j) +
147             S.parameter •
148               (∑ j, Gamma i j • liftedBasisVector rho hrho b j) := by
149   classical
150   let A := liftedSourceActionMatrix S b
151     (leftMultiplicationEnd (K := K) (V := V) (S.modParameter a))
152   let error : Fin (r + 1) → W := fun i ↦
153     a • liftedBasisVector rho hrho b i -
154       ∑ j, A i j • liftedBasisVector rho hrho b j
155   have herror : ∀ i, rho (error i) = 0 := by
156     intro i
157     rw [show rho (error i) =
158         rho (a • liftedBasisVector rho hrho b i) -
159           rho (∑ j, A i j • liftedBasisVector rho hrho b j) by
160       simp [error]]
161     rw [hrho_action, rho_liftedBasisVector]
162     simp_rw [map_sum, hrho_action, rho_liftedBasisVector]
163     have hAmap := liftedSourceActionMatrix_map S b
164       (leftMultiplicationEnd (K := K) (V := V) (S.modParameter a))
165     have hentry : ∀ i j, S.modParameter (A i j) =
166         sourceActionCoefficients (Abar := Abar) b
167           (leftMultiplicationEnd (K := K) (V := V)
168             (S.modParameter a)) i j := by
169       intro p q
170       exact congrArg (fun M ↦ M p q) hAmap
171     simp_rw [hentry]
172     rw [sum_sourceActionCoefficients_smul]
173     simp
174   have herror_range : ∀ i, error i ∈
175       AddMonoidHom.range (parameterAct (W := W) S.parameter) := by
176     intro i
177     rw [← hrho_ker]
178     exact AddMonoidHom.mem_ker.mpr (herror i)
179   choose u hu using herror_range
180   let Gamma : Matrix (Fin (r + 1)) (Fin (r + 1)) B := fun i j ↦
181     zeroPreservingLift S
182       (algebraMap K Abar (b.repr (rho (u i)) j))
183   have hresidual : ∀ i,
184       rho (u i - ∑ j, Gamma i j • liftedBasisVector rho hrho b j) = 0 := by
185     intro i
186     rw [map_sub, map_sum]
187     simp_rw [hrho_action, rho_liftedBasisVector]
188     have hGamma : ∀ j, S.modParameter (Gamma i j) =
189         algebraMap K Abar (b.repr (rho (u i)) j) := by
190       intro j
191       exact modParameter_zeroPreservingLift S _
192     simp_rw [hGamma, IsScalarTower.algebraMap_smul]
193     rw [b.sum_repr, sub_self]
194   have hresidual_range : ∀ i,
195       u i - ∑ j, Gamma i j • liftedBasisVector rho hrho b j ∈
196         AddMonoidHom.range (parameterAct (W := W) S.parameter) := by
197     intro i
198     rw [← hrho_ker]
199     exact AddMonoidHom.mem_ker.mpr (hresidual i)
200   choose v hv using hresidual_range
201   refine ⟨Gamma, (fun i j ↦ b.repr (rho (u i)) j), ?_, ?_⟩
202   · ext i j
203     exact modParameter_zeroPreservingLift S _
204   intro i
205   have hu_i : S.parameter • u i = error i := hu i
206   have hv_i : S.parameter • v i =
207       u i - ∑ j, Gamma i j • liftedBasisVector rho hrho b j := hv i
208   have hkill : S.parameter •
209       (u i - ∑ j, Gamma i j • liftedBasisVector rho hrho b j) = 0 := by
210     calc
211       S.parameter •
212           (u i - ∑ j, Gamma i j • liftedBasisVector rho hrho b j) =
213         S.parameter • (S.parameter • v i) := by rw [hv_i]
214       _ = (S.parameter * S.parameter) • v i := by
215         exact (mul_smul S.parameter S.parameter (v i)).symm
216       _ = 0 := by rw [hparameter_sq, zero_smul]
217   have hcu : S.parameter • u i =
218       S.parameter •
219         (∑ j, Gamma i j • liftedBasisVector rho hrho b j) := by
220     rw [smul_sub] at hkill
221     exact sub_eq_zero.mp hkill
222   have herr : error i =
223       S.parameter •
224         (∑ j, Gamma i j • liftedBasisVector rho hrho b j) :=
225     hu_i.symm.trans hcu
226   calc
227     a • liftedBasisVector rho hrho b i =
228         (∑ j, A i j • liftedBasisVector rho hrho b j) + error i := by
229       dsimp [error]
230       abel
231     _ = (∑ j, A i j • liftedBasisVector rho hrho b j) +
232         S.parameter •
233           (∑ j, Gamma i j • liftedBasisVector rho hrho b j) := by
234       rw [herr]

236 /-- Compatibility form retaining only the source equation. The stronger
237 producer also supplies coefficient-field reductions of every correction. -/
238 theorem exists_firstOrderSourceActionMatrix
239     (S : LeftPrincipalParameterReduction (B := B) (Abar := Abar))
240     (rho : W →+ V) (hrho : Function.Surjective rho)
241     (hrho_action : ∀ (a : B) (w : W),
242       rho (a • w) = S.modParameter a • rho w)
243     (hrho_ker : AddMonoidHom.ker rho =
244       AddMonoidHom.range (parameterAct (W := W) S.parameter))
245     (hparameter_sq : S.parameter * S.parameter = 0)
246     (b : Module.Basis (Fin (r + 1)) K V) (a : B) :
247     ∃ Gamma : Matrix (Fin (r + 1)) (Fin (r + 1)) B,
248       ∀ i,
249         a • liftedBasisVector rho hrho b i =
250           (∑ j, liftedSourceActionMatrix S b
251               (leftMultiplicationEnd (K := K) (V := V)
252                 (S.modParameter a)) i j •
253               liftedBasisVector rho hrho b j) +
254             S.parameter •
255               (∑ j, Gamma i j • liftedBasisVector rho hrho b j) := by
256   obtain ⟨Gamma, _, _, heq⟩ :=
257     exists_firstOrderSourceActionMatrix_over_coefficientField
258       S rho hrho hrho_action hrho_ker hparameter_sq b a
259   exact ⟨Gamma, heq⟩

321 /-- Full action compatibility after both honest quotient descents.  The left
322 side is the descended noncommutative two-block action; the right side is the
323 canonical doubled-power quotient action on the actual localized special
324 fibre. -/
325 theorem localizedDoubledPowerModuleSpecialization_action
326     (q : ℕ)
327     (hpow : 𝔪 ^ q ≤ Module.annihilator A₀ (LocalizedModule S G))
328     (a : C₂ q) (w : Wₗ) :
329     letI : Module (A₀ ⧸ localizedDoubledPower S q)
330         (LocalizedModule S G) :=
331       localizedDoubledPowerQuotientModule S q hpow
332     localizedDoubledPowerModuleSpecialization D S q hpow
333           (localizedTwoBlockAction D S q hpow a w) =
334         localizedTwoBlockSpecialization D S q a •
335           localizedDoubledPowerModuleSpecialization D S q hpow w := by

337       (LocalizedModule S G) :=

340   | _ z =>
341     change localizedSpecialization D S (z • w) =
342       Ideal.Quotient.mk (localizedDoubledPower S q)
343           (SquareZeroLocalizedRing.localizedOppositeSpecialization D S z) •
344         localizedSpecialization D S w
345     rw [localizedSpecialization_smul]

406 /-- The concrete first-order action-equation producer.
407 
408 For an actual element of the localized two-block ring and an actual basis of
409 the doubled-power special fibre, this theorem constructs the source-row
410 matrix `Gamma` and proves the lifted basis-vector equations.  The same fixed
411 `liftedBasisVector` definition is used for every operator, so applying the
412 theorem to `x` and `y` produces `Gamma` and `Theta` on one common lifted basis.
413 -/
414 theorem exists_localizedFirstOrderSourceActionMatrix
415     {K : Type u} [Field K]
416     (q : ℕ)
417     (hpow : 𝔪 ^ q ≤ Module.annihilator A₀ (LocalizedModule S G))
418     [Algebra K (A₀ ⧸ localizedDoubledPower S q)]
419     (P : LeftPrincipalParameterReduction
420       (B := C₂ q) (Abar := A₀ ⧸ localizedDoubledPower S q))
421     (hPmod : P.modParameter = localizedTwoBlockSpecialization D S q)
422     (hPparameter : P.parameter =
423       (localizedTwoBlockIdeal D S q).ringCon.mk'
424         (OreLocalization.numeratorRingHom (MulOpposite.op D.c)))
425     (a : C₂ q) :
426     letI : Module (A₀ ⧸ localizedDoubledPower S q)
427         (LocalizedModule S G) :=
428       localizedDoubledPowerQuotientModule S q hpow
429     letI : Module K (LocalizedModule S G) :=
430       Module.compHom _ (algebraMap K (A₀ ⧸ localizedDoubledPower S q))
431     letI : IsScalarTower K (A₀ ⧸ localizedDoubledPower S q)
432         (LocalizedModule S G) :=
433       IsScalarTower.of_algebraMap_smul fun _ _ ↦ rfl
434     ∀ {r : ℕ} (b : Module.Basis (Fin (r + 1)) K (LocalizedModule S G)),
435       ∃ Gamma : Matrix (Fin (r + 1)) (Fin (r + 1)) (C₂ q),
436         ∀ i,
437           localizedTwoBlockAction D S q hpow a
438             (liftedBasisVector
439               (localizedDoubledPowerModuleSpecialization D S q hpow)
440               (localizedDoubledPowerModuleSpecialization_surjective
441                 D S q hpow) b i) =
442             localizedSourceLinearCombination D S q hpow
443               (liftedSourceActionMatrix P b
444                 (leftMultiplicationEnd (K := K)
445                   (V := LocalizedModule S G)
446                   (P.modParameter a)))
447               (liftedBasisVector
448                 (localizedDoubledPowerModuleSpecialization D S q hpow)
449                 (localizedDoubledPowerModuleSpecialization_surjective
450                   D S q hpow) b) i +
451               localizedTwoBlockAction D S q hpow P.parameter
452                 (localizedSourceLinearCombination D S q hpow Gamma
453                   (liftedBasisVector
454                   (localizedDoubledPowerModuleSpecialization D S q hpow)
455                   (localizedDoubledPowerModuleSpecialization_surjective
456                     D S q hpow) b) i) := by
457   letI : Module (A₀ ⧸ localizedDoubledPower S q)
458       (LocalizedModule S G) :=
459     localizedDoubledPowerQuotientModule S q hpow
460   letI : Module K (LocalizedModule S G) :=
461     Module.compHom _ (algebraMap K (A₀ ⧸ localizedDoubledPower S q))
462   letI : IsScalarTower K (A₀ ⧸ localizedDoubledPower S q)
463       (LocalizedModule S G) :=
464     IsScalarTower.of_algebraMap_smul fun _ _ ↦ rfl
465   letI : Module (C₂ q) Wₗ := localizedTwoBlockModule D S q hpow
466   intro r b
467   have haction : ∀ (z : C₂ q) (w : Wₗ),
468       localizedDoubledPowerModuleSpecialization D S q hpow (z • w) =
469         P.modParameter z •
470           localizedDoubledPowerModuleSpecialization D S q hpow w := by
471     intro z w
472     rw [hPmod]
473     exact localizedDoubledPowerModuleSpecialization_action D S q hpow z w
474   have hparameterAct :
475       parameterAct (W := Wₗ) P.parameter = localizedCAct D S := by
476     ext w
477     change P.parameter • w = localizedCAct D S w
478     rw [hPparameter]
479     exact localizedTwoBlock_parameter_smul D S q hpow w
480   have hker : AddMonoidHom.ker
481         (localizedDoubledPowerModuleSpecialization D S q hpow) =
482       AddMonoidHom.range (parameterAct (W := Wₗ) P.parameter) := by
483     rw [hparameterAct]
484     exact localizedDoubledPowerModuleSpecialization_ker D S q hpow
485   have hsq : P.parameter * P.parameter = 0 := by
486     rw [hPparameter]
487     have h := localizedTwoBlock_parameter_sq D S q
488     rw [pow_two] at h
489     exact h
490   have hresult := exists_firstOrderSourceActionMatrix P
491     (localizedDoubledPowerModuleSpecialization D S q hpow)
492     (localizedDoubledPowerModuleSpecialization_surjective D S q hpow)
493     haction hker hsq b a
494   simp only [localizedSourceLinearCombination]
495   exact hresult
496 
497 end LocalizedTwoBlockProducer

529 localized deformation module action. -/

541     letI : Module K

544     letI : IsScalarTower K (A₀ ⧸ localizedDoubledPower S q)
