6 /-!
7 # Localized specialization respects the localized action
8 
9 The Ore localization of the opposite deformation ring acts on the localized
10 deformation module.  This file proves directly on displayed left-Ore
11 fractions that module specialization intertwines this action with the
12 commutative special-fibre localization action.
13 
14 The use of the opposite ring is load-bearing: an Ore equation in `Bᵐᵒᵖ` is
15 sent through `oppositeSpecializationRingHom`, whose multiplicativity is
16 exactly the reversal from written right multiplication to the left action.
17 No Artinian descent, characteristic-variety, or involutivity statement is
18 made here.
19 -/

43 /-- Specialization of the concrete localized deformation module intertwines
44 the full localized opposite-ring action with specialization of the scalar.
45 The multiplication displayed on the source is the left action corresponding
46 to written right multiplication before passage to the opposite ring. -/
47 theorem localizedSpecialization_smul
48     (a : OreLocalization (OppositeDenominators D S) (Bᵐᵒᵖ))
49     (z : LocalizedDeformationModule D S) :
50     localizedSpecialization D S (a • z) =
51       localizedOppositeSpecialization D S a •
52         localizedSpecialization D S z := by
53   induction a using OreLocalization.ind with
54   | _ b s =>
55     induction z using OreLocalization.ind with
56     | _ m t =>
57       obtain ⟨r, u, hur, haction⟩ :=
58         OreLocalization.oreDivSMulChar' b m s t
59       rw [haction, localizedSpecialization_apply,
60         localizedSpecializationFun_oreDiv,
61         localizedOppositeSpecialization_oreDiv,
62         localizedSpecialization_apply,
63         localizedSpecializationFun_oreDiv]
64       have hscalar :
65           ((localizedDenominatorUnits D S s)⁻¹ :
66               Units (Localization S)) *
67               algebraMap R (Localization S) (D.pi b.unop) =
68             Localization.mk (D.pi b.unop) (denominatorMap D S s) := by
69         change
70           ((1 : R) /ₒ denominatorMap D S s) *
71               (D.pi b.unop /ₒ (1 : S)) =
72             D.pi b.unop /ₒ denominatorMap D S s
73         simpa only [one_mul] using
74           (OreLocalization.one_div_mul
75             (S := S) (r := D.pi b.unop) (s := (1 : S))
76             (t := denominatorMap D S s))
77       rw [hscalar, LocalizedModule.mk_smul_mk]
78       have hore := congrArg (oppositeSpecializationRingHom D) hur
79       simp only [map_mul, oppositeSpecializationRingHom_apply] at hore
80       apply LocalizedModule.mk_eq.mpr
81       refine ⟨1, ?_⟩
82       simp only [one_smul, Submonoid.smul_def, Submonoid.coe_mul]
83       have hrho : D.rho (r • m) = D.pi r.unop • D.rho m := by
84         simpa using D.rho_action r.unop m
85       have hcoeff :
86           (D.pi s.val.unop * D.pi t.val.unop) * D.pi r.unop =
87             (D.pi u.val.unop * D.pi s.val.unop) * D.pi b.unop := by
88         calc
89           (D.pi s.val.unop * D.pi t.val.unop) * D.pi r.unop =
90             D.pi s.val.unop * (D.pi r.unop * D.pi t.val.unop) := by
91               ring
92           _ = D.pi s.val.unop * (D.pi u.val.unop * D.pi b.unop) := by
93               rw [hore]
94           _ = (D.pi u.val.unop * D.pi s.val.unop) * D.pi b.unop := by
95               ring
96       rw [hrho, ← mul_smul, ← mul_smul]
97       have hcoeff' :
98           ((denominatorMap D S s : R) *
99               (denominatorMap D S t : R)) * D.pi r.unop =
100             (denominatorMap D S (u * s) : R) * D.pi b.unop := by
101         rw [map_mul]
102         exact hcoeff
103       exact congrArg (fun q : R => q • D.rho m) hcoeff'
