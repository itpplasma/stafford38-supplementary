50 /-- At a minimal tangential support prime, the actual page Euler inequality
51 contradicts the strict coordinate Koszul inequality. -/
52 theorem no_minimal_coordinate_cokernel_support
53     (q : PrimeSpectrum (T k n))
54     (hqmem : q ∈ Module.support (T k n)
55       (Graded k n N d ⧸ (oldCoordinateMap (k := k) n (Graded k n N d)).range))
56     (hqmin : ∀ p ∈ Module.support (T k n)
57         (Graded k n N d ⧸ (oldCoordinateMap (k := k) n (Graded k n N d)).range),
58       p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal) : False := by
59   let E := Graded k n N d
60   let R := T k n
61   let C := SymbolRing k (n + 1)
62   let x : C := MvPolynomial.X (.inl (0 : Fin (n + 1)))
63   let fC : Module.End C E := LinearMap.lsmul C E x
64   have hf : fC.restrictScalars R = oldCoordinateMap (k := k) n E := rfl
65   have hfinite := canonical_finite_old_coordinate_kernel_cokernel hd
66   haveI : Module.Finite R (fC.restrictScalars R).ker := hfinite.1
67   haveI : Module.Finite R (E ⧸ (fC.restrictScalars R).range) := hfinite.2
68   have hlength := localized_kernel_and_cokernel_isFiniteLength fC q hqmem hqmin
69   let S := q.asIdeal.primeCompl
70   let ek := localizedEquiv S (firstSourceGradedKernelEquiv k n N d hd)
71   let ec := localizedEquiv S (firstTargetGradedCokernelEquiv k n N d hd)
72   have hA : IsFiniteLength (Localization S)
73       (LocalizedModule S ((complex k n N d hd).SourceTotal 1)) :=
74     ek.symm.isFiniteLength hlength.2
75   have hB : IsFiniteLength (Localization S)
76       (LocalizedModule S ((complex k n N d hd).TargetTotal 1)) :=
77     ec.symm.isFiniteLength hlength.1
78   have hle := canonicalPage_length_target_le_source k n N d hd S hA hB
79   rw [ec.length_eq, ek.length_eq] at hle
80   have hlt := localized_length_cokernel_gt_kernel (R := R) (C := C) (E := E)
81     x q hqmem hqmin (fun p hp =>
82       canonical_minimalPrime_mem_of_normalCoordinate_false d hN hd hp)
83   exact (not_lt_of_ge hle) hlt
