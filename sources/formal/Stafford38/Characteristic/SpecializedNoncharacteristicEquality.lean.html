55 /-- In the transposed left canonical quotient, left multiplication by the
56 distinguished coordinate is the original right-coordinate action. -/
57 theorem transposedCanonical_coordinate_smul_eq_rightMul
58     (n N : ℕ) (d : WeylAlg k (n + 1))
59     (q : TransposedFilteredRightQuotient k (CanonicalIdeal k n N d)) :
60     (presentedCoordinate k n • q).toRightQuotient =
61       rightMul (CanonicalIdeal k n N d) (presentedCoordinate k n)
62         q.toRightQuotient := by
63   rcases q with ⟨q⟩
64   change rightMul (CanonicalIdeal k n N d)
65       (transpose k (n + 1) (presentedCoordinate k n)) q = _
66   rw [show presentedCoordinate k n = coordinate k (n + 1) 0 from rfl,
67     transpose_coordinate]
68 
69 /-- PBW monicity already kills ordinary degree-zero restriction of the
70 transposed left module: coordinate multiplication is onto. -/
71 theorem transposedCanonical_coordinate_smul_surjective
72     [Algebra ℚ k] (n N : ℕ) {d : WeylAlg k (n + 1)}
73     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
74     Function.Surjective
75       (fun q : TransposedFilteredRightQuotient k (CanonicalIdeal k n N d) ↦
76         presentedCoordinate k n • q) := by
77   intro q
78   obtain ⟨r, hr⟩ :=
79     presentedCanonicalRightQuotient_rightMul_coordinate_surjective
80       (k := k) n N hd q.toRightQuotient
81   refine ⟨TransposedFilteredRightQuotient.mk r, ?_⟩
82   apply (transposedFilteredRightQuotientEquiv k
83     (CanonicalIdeal k n N d)).injective
84   change (presentedCoordinate k n •
85       TransposedFilteredRightQuotient.mk r).toRightQuotient =
86     q.toRightQuotient
87   rw [transposedCanonical_coordinate_smul_eq_rightMul]
88   exact hr

96 /-- The induced involution on the prime spectrum is literally involutive. -/
97 theorem primeComap_symbolTransposition_involutive
98     {n : ℕ} (p : PrimeSpectrum (SymbolRing k n)) :
99     PrimeSpectrum.comap
100         (symbolTranspositionEquiv k).toRingEquiv.toRingHom
101         (PrimeSpectrum.comap
102           (symbolTranspositionEquiv k).toRingEquiv.toRingHom p) = p := by
103   rw [← PrimeSpectrum.comap_comp_apply]
104   have hcomp :
105       (symbolTranspositionEquiv k).toRingEquiv.toRingHom.comp
106           (symbolTranspositionEquiv k).toRingEquiv.toRingHom =
107         RingHom.id (SymbolRing k n) := by
108     apply RingHom.ext
109     intro P
110     change symbolTransposition k (symbolTransposition k P) = P
111     exact AlgHom.congr_fun (symbolTransposition_comp_self k) P
112   rw [hcomp]
113   rfl

115 /-- Momentum-sign substitution fixes the coordinate hyperplane. -/
116 theorem axisZeroLocus_preimage_symbolTransposition (n : ℕ) :
117     PrimeSpectrum.comap
118         (symbolTranspositionEquiv k).toRingEquiv.toRingHom ⁻¹'
119         AxisZeroLocus k n =
120       AxisZeroLocus k n := by
121   ext p
122   simp only [Set.mem_preimage, AxisZeroLocus,
123     PrimeSpectrum.mem_zeroLocus, Set.singleton_subset_iff]
124   change (symbolTranspositionEquiv k).toRingEquiv.toRingHom
125       (AxisCoordinate k n) ∈ p.asIdeal ↔
126     AxisCoordinate k n ∈ p.asIdeal
127   change symbolTransposition k (AxisCoordinate k n) ∈ p.asIdeal ↔
128     AxisCoordinate k n ∈ p.asIdeal
129   rw [symbolTransposition_axisCoordinate]

131 private theorem disjoint_preimage_iff_of_involution_of_invariant
132     {X : Type*} (f : X → X) (hf : ∀ x, f (f x) = x)
133     (A B : Set X) (hB : f ⁻¹' B = B) :
134     Disjoint (f ⁻¹' A) B ↔ Disjoint A B := by
135   rw [Set.disjoint_left, Set.disjoint_left]
136   constructor
137   · intro h x hxA hxB
138     have hxpre : f x ∈ f ⁻¹' A := by
139       change f (f x) ∈ A
140       rwa [hf]
141     have hxfb : f x ∈ B := by
142       have := Set.ext_iff.mp hB x
143       exact this.mpr hxB
144     exact h hxpre hxfb
145   · intro h x hxpre hxB
146     have hfxB : f x ∈ B := by
147       have := Set.ext_iff.mp hB x
148       exact this.mpr hxB
149     exact h hxpre hfxB

151 /-- Consequently, support exclusion from the coordinate hyperplane is
152 unchanged by the right-to-left symbol transposition. -/
153 theorem transposedSupport_disjoint_axis_iff
154     (n N : ℕ) (d : WeylAlg k (n + 1)) :
155     Disjoint
156         (transposedOrderAssociatedGradedSupport k (CanonicalIdeal k n N d))
157         (AxisZeroLocus k n) ↔
158       Disjoint
159         (Module.support (SymbolRing k (n + 1))
160           (OrderAssociatedGradedModule k (CanonicalIdeal k n N d)))
161         (AxisZeroLocus k n) := by
162   rw [transposedOrderAssociatedGradedSupport_eq_preimage]
163   apply disjoint_preimage_iff_of_involution_of_invariant
164   · exact primeComap_symbolTransposition_involutive k
165   · exact axisZeroLocus_preimage_symbolTransposition k n

169 private theorem disjoint_zeroLocus_singleton_iff_sup_eq_top
170     {R : Type*} [CommRing R] (I : Ideal R) (x : R) :
171     Disjoint (PrimeSpectrum.zeroLocus I)
172         (PrimeSpectrum.zeroLocus ({x} : Set R)) ↔
173       I ⊔ Ideal.span {x} = ⊤ := by
174   rw [Set.disjoint_iff_inter_eq_empty,
175     ← PrimeSpectrum.zeroLocus_span ({x} : Set R),
176     ← PrimeSpectrum.zeroLocus_sup,
177     PrimeSpectrum.zeroLocus_empty_iff_eq_top]

179 /-- Since symbol transposition fixes the base coordinate, coordinate
180 surjectivity on the transposed associated graded module is equivalent to the
181 same statement on the original associated graded module. -/
182 theorem transposedGradedCoordinate_surjective_iff
183     (n N : ℕ) (d : WeylAlg k (n + 1)) :
184     Function.Surjective
185         (fun q : TransposedOrderAssociatedGradedModule k
186             (CanonicalIdeal k n N d) ↦ AxisCoordinate k n • q) ↔
187       Function.Surjective
188         (fun q : OrderAssociatedGradedModule k (CanonicalIdeal k n N d) ↦
189           AxisCoordinate k n • q) := by
190   constructor
191   · intro h q
192     obtain ⟨r, hr⟩ := h (TransposedOrderAssociatedGradedModule.mk q)
193     refine ⟨r.toOrderAssociatedGradedModule, ?_⟩
194     have hr' := congrArg
195       TransposedOrderAssociatedGradedModule.toOrderAssociatedGradedModule hr
196     simpa only [transposedSymbol_smul, symbolTransposition_axisCoordinate]
197       using hr'
198   · intro h q
199     obtain ⟨r, hr⟩ := h q.toOrderAssociatedGradedModule
200     refine ⟨TransposedOrderAssociatedGradedModule.mk r, ?_⟩
201     apply (transposedOrderAssociatedGradedEquiv k
202       (CanonicalIdeal k n N d)).injective
203     change TransposedOrderAssociatedGradedModule.toOrderAssociatedGradedModule
204         (AxisCoordinate k n •
205           TransposedOrderAssociatedGradedModule.mk r) =
206       q.toOrderAssociatedGradedModule
207     simpa only [transposedSymbol_smul, symbolTransposition_axisCoordinate]
208       using hr

210 /-- For the canonical quotient, hyperplane support exclusion is exactly
211 surjectivity of the coordinate on the actual associated graded module. -/
212 theorem canonicalSupport_disjoint_axis_iff_gradedCoordinate_surjective
213     [Algebra ℚ k] (n N : ℕ) (d : WeylAlg k (n + 1)) :
214     Disjoint
215         (Module.support (SymbolRing k (n + 1))
216           (OrderAssociatedGradedModule k (CanonicalIdeal k n N d)))
217         (AxisZeroLocus k n) ↔
218       Function.Surjective
219         (fun q : OrderAssociatedGradedModule k (CanonicalIdeal k n N d) ↦
220           AxisCoordinate k n • q) := by
221   have hsupp :
222       Module.support (SymbolRing k (n + 1))
223           (OrderAssociatedGradedModule k (CanonicalIdeal k n N d)) =
224         PrimeSpectrum.zeroLocus
225           (orderInitialIdeal k (CanonicalIdeal k n N d)) := by
226     rw [Module.support_eq_zeroLocus,
227       annihilator_orderAssociatedGradedModule]
228   rw [hsupp]
229   exact (disjoint_zeroLocus_singleton_iff_sup_eq_top
230     (orderInitialIdeal k (CanonicalIdeal k n N d))
231     (AxisCoordinate k n)).trans
232       (canonical_orderInitialIdeal_sup_coordinate_eq_top_iff_graded_surjective
233         k n N d)

235 /-- Fully transposed form of the exact boundary.  This is the strongest
236 specialized equality obtainable from the current local and Mathlib APIs:
237 the desired support exclusion is equivalent to strict associated-graded
238 coordinate surjectivity, while ordinary coordinate surjectivity was proved
239 above from monicity. -/
240 theorem transposedSupport_disjoint_axis_iff_gradedCoordinate_surjective
241     [Algebra ℚ k] (n N : ℕ) (d : WeylAlg k (n + 1)) :
242     Disjoint
243         (transposedOrderAssociatedGradedSupport k (CanonicalIdeal k n N d))
244         (AxisZeroLocus k n) ↔
245       Function.Surjective
246         (fun q : OrderAssociatedGradedModule k (CanonicalIdeal k n N d) ↦
247           AxisCoordinate k n • q) := by
248   rw [transposedSupport_disjoint_axis_iff]
249   exact canonicalSupport_disjoint_axis_iff_gradedCoordinate_surjective
250     k n N d

265 /-- The project-specific algebraic suffix after the noncharacteristic theorem:
266 transposed support avoidance produces the canonical order-zero predecessor of
267 the quotient unit.  No D-module theorem is used in this conversion. -/
268 theorem strictUnitCoordinatePreimage_of_transposedSupport_disjoint_axis
269     [Algebra ℚ k] (n N : ℕ) (d : WeylAlg k (n + 1))
270     (havoid : Disjoint
271       (transposedOrderAssociatedGradedSupport k (CanonicalIdeal k n N d))
272       (AxisZeroLocus k n)) :
273     Stafford38.CanonicalUnitCoordinatePreimage.StrictUnitCoordinatePreimage
274       k n N d := by
275   have horiginal :=
276     (transposedSupport_disjoint_axis_iff k n N d).mp havoid
277   have hzero : Disjoint
278       (PrimeSpectrum.zeroLocus
279         (orderInitialIdeal k (CanonicalIdeal k n N d)))
280       (AxisZeroLocus k n) := by
281     rw [Module.support_eq_zeroLocus,
282       annihilator_orderAssociatedGradedModule] at horiginal
283     exact horiginal
284   have htop : orderInitialIdeal k (CanonicalIdeal k n N d) ⊔
285       Ideal.span {AxisCoordinate k n} = ⊤ :=
286     (disjoint_zeroLocus_singleton_iff_sup_eq_top
287       (orderInitialIdeal k (CanonicalIdeal k n N d))
288       (AxisCoordinate k n)).mp hzero
289   exact strictUnitCoordinatePreimage_of_orderInitialIdeal_sup_coordinate_eq_top
290     k n N htop
