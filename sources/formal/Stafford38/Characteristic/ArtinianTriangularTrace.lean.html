8 # The Artinian triangular trace calculation
9 
10 This file formalizes the terminal finite-dimensional trace core used in
11 Singh--Kumar, Proposition 3.2.
12 After choosing a coefficient field in the commutative Artinian special fibre
13 and a basis adapted to powers of its maximal ideal, the action of a lifted
14 maximal-ideal element is strictly triangular.  The first-order ringCommutator
15 calculation then represents the bracket-cofactor action as
16 
17 `R + (ΘA - AΘ) + (BΓ - ΓB)`,
18 
19 where `R` is strictly upper triangular.  Its trace is zero.  On the other
20 hand, the residue decomposition of that action is a scalar matrix plus a
21 nilpotent matrix.  In positive dimension and characteristic zero the scalar
22 must therefore vanish.
23 
24 The faithful terminal theorem assumes only that the induced operator has trace
25 zero.  This is exactly what the source proves from equation (3.3); it does not
26 identify the correction terms with commutators of same-size matrices over the
27 coefficient field.  A stronger ringCommutator-shaped corollary is retained as an
28 abstract convenience, but is not claimed to follow directly from the source.
29 
30 This file does not construct the coefficient field or adapted basis, derive
31 the trace-zero operator from a square-zero module, or connect it to the
32 localized right Rees module.
33 -/

46 /-- A square matrix is strictly upper triangular when every entry on or below
47 the diagonal is zero. -/
48 abbrev IsStrictUpperTriangular
49     (M : Matrix (Fin (n + 1)) (Fin (n + 1)) K) : Prop :=
50   Stafford38.Characteristic.ArtinianEquation33TraceProducer.IsStrictUpperTriangularOver M
51 
52 /-- A strictly upper triangular matrix has zero trace. -/
53 theorem trace_eq_zero_of_isStrictUpperTriangular
54     {M : Matrix (Fin (n + 1)) (Fin (n + 1)) K}
55     (hM : IsStrictUpperTriangular M) :
56     Matrix.trace M = 0 := by
57   rw [Matrix.trace]
58   apply Finset.sum_eq_zero
59   intro i _hi
60   exact hM i i le_rfl
61 
62 /-- Written-order matrix ringCommutator. -/
63 def matrixCommutator
64     (X Y : Matrix (Fin (n + 1)) (Fin (n + 1)) K) :
65     Matrix (Fin (n + 1)) (Fin (n + 1)) K :=
66   X * Y - Y * X
67 
68 /-- Cyclicity of matrix trace kills a matrix ringCommutator over the coefficient
69 field. -/
70 theorem trace_matrixCommutator_eq_zero
71     (X Y : Matrix (Fin (n + 1)) (Fin (n + 1)) K) :
72     Matrix.trace (matrixCommutator X Y) = 0 := by
73   rw [matrixCommutator, Matrix.trace_sub, Matrix.trace_mul_comm, sub_self]
74 
75 /-- The matrix left after the maximal-ideal-adapted first-order calculation.
76 The order of both commutators is the order occurring in the published
77 calculation. -/
78 def adaptedFirstOrderMatrix
79     (R Θ A B Γ : Matrix (Fin (n + 1)) (Fin (n + 1)) K) :
80     Matrix (Fin (n + 1)) (Fin (n + 1)) K :=
81   R + matrixCommutator Θ A + matrixCommutator B Γ
82 
83 /-- The adapted first-order matrix has trace zero.  No triangularity of
84 `A`, `B`, `Γ`, or `Θ` is needed after the calculation has isolated the
85 strictly triangular remainder `R`. -/
86 theorem trace_adaptedFirstOrderMatrix_eq_zero
87     (R Θ A B Γ : Matrix (Fin (n + 1)) (Fin (n + 1)) K)
88     (hR : IsStrictUpperTriangular R) :
89     Matrix.trace (adaptedFirstOrderMatrix R Θ A B Γ) = 0 := by
90   simp only [adaptedFirstOrderMatrix, Matrix.trace_add,
91     trace_eq_zero_of_isStrictUpperTriangular hR,
92     trace_matrixCommutator_eq_zero, zero_add]

94 /-- Faithful terminal trace calculation used by Proposition 3.2.
95 
96 The source-specific work upstream proves that the induced operator `T` has
97 trace zero.  The coefficient-field decomposition gives `q I + U = T`, where
98 maximal-ideal multiplication makes `U` nilpotent.  Positive dimension and
99 characteristic zero then force `q = 0`.
100 -/
101 theorem scalar_eq_zero_of_trace_identity
102     [CharZero K]
103     (q : K)
104     (U T : Matrix (Fin (n + 1)) (Fin (n + 1)) K)
105     (hU : IsNilpotent U)
106     (htraceT : Matrix.trace T = 0)
107     (hidentity :
108       q • (1 : Matrix (Fin (n + 1)) (Fin (n + 1)) K) + U = T) :
109     q = 0 := by
110   have htraceU : Matrix.trace U = 0 := by
111     exact isNilpotent_iff_eq_zero.mp
112       (Matrix.isNilpotent_trace_of_isNilpotent hU)
113   have htrace := congrArg Matrix.trace hidentity
114   rw [Matrix.trace_add, Matrix.trace_smul, Matrix.trace_one, htraceU,
115     add_zero, htraceT, smul_eq_mul] at htrace
116   exact (mul_eq_zero.mp htrace).resolve_right
117     (Nat.cast_ne_zero.mpr (by simp))
118 
119 /-- Stronger abstract corollary with an explicit ringCommutator decomposition.
120 
121 This implication is correct, but Proposition 3.2 does not directly provide its
122 same-size coefficient-field matrix identity.  The source-specific adapter
123 should instead establish the zero-trace hypothesis of
124 `scalar_eq_zero_of_trace_identity`.
125 -/
126 theorem scalar_eq_zero_of_adapted_triangular_matrix_identity
127     [CharZero K]
128     (q : K)
129     (U R Θ A B Γ : Matrix (Fin (n + 1)) (Fin (n + 1)) K)
130     (hU : IsNilpotent U)
131     (hR : IsStrictUpperTriangular R)
132     (hidentity :
133       q • (1 : Matrix (Fin (n + 1)) (Fin (n + 1)) K) + U =
134         adaptedFirstOrderMatrix R Θ A B Γ) :
135     q = 0 :=
136   scalar_eq_zero_of_trace_identity q U
137     (adaptedFirstOrderMatrix R Θ A B Γ) hU
138     (trace_adaptedFirstOrderMatrix_eq_zero R Θ A B Γ hR) hidentity
