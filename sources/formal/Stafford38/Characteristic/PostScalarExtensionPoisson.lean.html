8 /-!
9 # Poisson closure after scalar extension
10 
11 This file transports bracket closure through coefficient extension.  Its main
12 result uses the Gabber-shaped involutivity hypothesis `{J,J} ⊆ J`, rather than
13 the much stronger project predicate `IsPoisson J = {J,R} ⊆ J`.  A second result
14 transports that stronger predicate by extracting stability under every
15 coordinate derivation.  No claim is made that taking a radical after base
16 change preserves bracket closure.
17 -/

33 /-- The Gabber-shaped bracket condition: both bracket entries lie in the
34 ideal.  This is weaker than the project predicate `IsPoisson`. -/
35 def IsInvolutive (J : Ideal (SymbolRing k n)) : Prop :=
36   ∀ f ∈ J, ∀ g ∈ J, poissonBracket f g ∈ J

38 /-- Coefficient extension commutes with the canonical Poisson bracket. -/
39 theorem poissonBracket_scalarPolynomialMap
40     (f g : SymbolRing k n) :
41     poissonBracket
42         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) f)
43         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) g) =
44       scalarPolynomialMap (k := k) (K := K) (PhaseVar n)
45         (poissonBracket f g) := by
46   simp [poissonBracket, scalarPolynomialMap, MvPolynomial.pderiv_map]

48 theorem poissonBracket_add_left (f₁ f₂ g : SymbolRing k n) :
49     poissonBracket (f₁ + f₂) g =
50       poissonBracket f₁ g + poissonBracket f₂ g := by
51   simp [poissonBracket, add_mul, Finset.sum_add_distrib]
52   ring
53 
54 theorem poissonBracket_add_right (f g₁ g₂ : SymbolRing k n) :
55     poissonBracket f (g₁ + g₂) =
56       poissonBracket f g₁ + poissonBracket f g₂ := by
57   simp [poissonBracket, mul_add, Finset.sum_add_distrib]
58   ring
59 
60 theorem poissonBracket_mul_left (a f g : SymbolRing k n) :
61     poissonBracket (a * f) g =
62       a * poissonBracket f g + f * poissonBracket a g := by
63   simp only [poissonBracket, MvPolynomial.pderiv_mul]
64   rw [Finset.mul_sum, Finset.mul_sum, ← Finset.sum_add_distrib]
65   apply Finset.sum_congr rfl
66   intro i hi
67   ring
68 
69 theorem poissonBracket_mul_right (f a g : SymbolRing k n) :
70     poissonBracket f (a * g) =
71       a * poissonBracket f g + g * poissonBracket f a := by
72   simp only [poissonBracket, MvPolynomial.pderiv_mul]
73   rw [Finset.mul_sum, Finset.mul_sum, ← Finset.sum_add_distrib]
74   apply Finset.sum_congr rfl
75   intro i hi

78 /-- Gabber-shaped involutivity is preserved by arbitrary field-valued
79 coefficient extension. -/
80 theorem isInvolutive_map_scalarPolynomialMap
81     (J : Ideal (SymbolRing k n)) (hJ : IsInvolutive J) :
82     IsInvolutive
83       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) := by
84   let I := J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))
85   intro f hf
86   change f ∈ J.map
87     (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) at hf
88   rw [Ideal.map] at hf
89   induction hf using Submodule.span_induction with
90   | mem f hf =>
91       rcases hf with ⟨f₀, hf₀, rfl⟩
92       intro g hg
93       change g ∈ J.map
94         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) at hg
95       rw [Ideal.map] at hg
96       induction hg using Submodule.span_induction with
97       | mem g hg =>
98           rcases hg with ⟨g₀, hg₀, rfl⟩
99           rw [poissonBracket_scalarPolynomialMap]
100           exact Ideal.mem_map_of_mem _ (hJ f₀ hf₀ g₀ hg₀)
101       | zero => simp [poissonBracket]
102       | add g₁ g₂ _ _ hg₁ hg₂ =>
103           rw [poissonBracket_add_right]
104           exact I.add_mem hg₁ hg₂
105       | smul a g hgm hg =>
106           rw [smul_eq_mul, poissonBracket_mul_right]
107           exact I.add_mem (I.mul_mem_left _ hg) (I.mul_mem_right _ hgm)
108   | zero =>
109       intro g hg
110       simp [poissonBracket]
111   | add f₁ f₂ _ _ hf₁ hf₂ =>
112       intro g hg
113       rw [poissonBracket_add_left]
114       exact I.add_mem (hf₁ g hg) (hf₂ g hg)
115   | smul a f hfm hf =>
116       intro g hg
117       rw [smul_eq_mul, poissonBracket_mul_left]
118       exact I.add_mem (I.mul_mem_left _ (hf g hg)) (I.mul_mem_right _ hfm)

120 /-- Scalar extension of Gabber-shaped involutivity gives exactly the
121 base-relative condition consumed by the conormal proof. -/
122 theorem isBaseRelativePoisson_map_of_isInvolutive
123     (J : Ideal (SymbolRing k n)) (hJ : IsInvolutive J) :
124     BaseRelativePoisson.IsBaseRelativePoisson
125       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) := by
126   intro f hf g hg
127   exact isInvolutive_map_scalarPolynomialMap J hJ (baseLift f) hf g hg

129 /-- A Poisson ideal is stable under each phase-space partial derivative. -/
130 theorem pderiv_mem_of_isPoisson
131     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
132     {f : SymbolRing k n} (hf : f ∈ J) (i : PhaseVar n) :
133     MvPolynomial.pderiv i f ∈ J := by
134   rcases i with i | i
135   · have h := hJ f hf (MvPolynomial.X (Sum.inr i))
136     simpa [poissonBracket, Pi.single_apply] using h
137   · have h := hJ f hf (MvPolynomial.X (Sum.inl i))
138     have hn : -MvPolynomial.pderiv (Sum.inr i) f ∈ J := by
139       simpa [poissonBracket, Pi.single_apply] using h
140     simpa using J.neg_mem hn

142 /-- Every coordinate derivation preserves the ideal generated after scalar
143 extension from a Poisson ideal. -/
144 theorem pderiv_mem_map_scalarPolynomialMap
145     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
146     {f : SymbolRing K n}
147     (hf : f ∈ J.map
148       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)))
149     (i : PhaseVar n) :
150     MvPolynomial.pderiv i f ∈ J.map
151       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) := by
152   rw [Ideal.map] at hf
153   induction hf using Submodule.span_induction with
154   | mem f hf =>
155       rcases hf with ⟨g, hg, rfl⟩
156       change MvPolynomial.pderiv i
157           (MvPolynomial.map (algebraMap k K) g) ∈ _
158       rw [MvPolynomial.pderiv_map]
159       exact Ideal.mem_map_of_mem _ (pderiv_mem_of_isPoisson J hJ hg i)
160   | zero => simp
161   | add f g _ _ hf hg =>
162       simpa using (J.map
163         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).add_mem hf hg
164   | smul a f hfm hf =>
165       rw [smul_eq_mul, MvPolynomial.pderiv_mul]
166       exact (J.map
167         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).add_mem
168           ((J.map
169             (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_left _ hfm)
170           ((J.map
171             (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_left _ hf)

173 /-- Full Poisson closure is preserved by arbitrary field-valued coefficient
174 extension.  The target is the actual extended ideal `map`, not its radical. -/
175 theorem isPoisson_map_scalarPolynomialMap
176     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J) :
177     IsPoisson
178       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) := by
179   intro f hf g
180   apply (J.map
181     (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).sum_mem
182   intro i hi
183   apply (J.map
184     (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).sub_mem
185   · exact (J.map
186       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_right _
187         (pderiv_mem_map_scalarPolynomialMap J hJ hf (Sum.inl i))
188   · exact (J.map
189       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_right _
190         (pderiv_mem_map_scalarPolynomialMap J hJ hf (Sum.inr i))

192 /-- The transported full Poisson theorem supplies the exact base-relative
193 fragment used by the conormal argument. -/
194 theorem isBaseRelativePoisson_map_scalarPolynomialMap
195     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J) :
196     BaseRelativePoisson.IsBaseRelativePoisson
197       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) :=
198   BaseRelativePoisson.IsPoisson.isBaseRelativePoisson _
199     (isPoisson_map_scalarPolynomialMap J hJ)
