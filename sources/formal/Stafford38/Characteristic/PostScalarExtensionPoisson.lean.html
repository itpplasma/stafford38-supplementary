6 /-!
7 # Poisson closure after scalar extension
8 
9 This file transports bracket closure through coefficient extension.  Its main
10 result uses the Gabber-shaped involutivity hypothesis `{J,J} ⊆ J`, rather than
11 the much stronger project predicate `IsPoisson J = {J,R} ⊆ J`.  A second result
12 transports that stronger predicate by extracting stability under every
13 coordinate derivation.  No claim is made that taking a radical after base
14 change preserves bracket closure.
15 -/

31 /-- The Gabber-shaped bracket condition: both bracket entries lie in the
32 ideal.  This is weaker than the project predicate `IsPoisson`. -/
33 def IsInvolutive (J : Ideal (SymbolRing k n)) : Prop :=
34   ∀ f ∈ J, ∀ g ∈ J, poissonBracket f g ∈ J

36 /-- Coefficient extension commutes with the canonical Poisson bracket. -/
37 theorem poissonBracket_scalarPolynomialMap
38     (f g : SymbolRing k n) :
39     poissonBracket
40         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) f)
41         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) g) =
42       scalarPolynomialMap (k := k) (K := K) (PhaseVar n)
43         (poissonBracket f g) := by
44   simp [poissonBracket, scalarPolynomialMap, MvPolynomial.pderiv_map]

46 theorem poissonBracket_add_left (f₁ f₂ g : SymbolRing k n) :
47     poissonBracket (f₁ + f₂) g =
48       poissonBracket f₁ g + poissonBracket f₂ g := by
49   simp [poissonBracket, add_mul, Finset.sum_add_distrib]
50   ring
51 
52 theorem poissonBracket_add_right (f g₁ g₂ : SymbolRing k n) :
53     poissonBracket f (g₁ + g₂) =
54       poissonBracket f g₁ + poissonBracket f g₂ := by
55   simp [poissonBracket, mul_add, Finset.sum_add_distrib]
56   ring
57 
58 theorem poissonBracket_mul_left (a f g : SymbolRing k n) :
59     poissonBracket (a * f) g =
60       a * poissonBracket f g + f * poissonBracket a g := by
61   simp only [poissonBracket, MvPolynomial.pderiv_mul]
62   rw [Finset.mul_sum, Finset.mul_sum, ← Finset.sum_add_distrib]
63   apply Finset.sum_congr rfl
64   intro i hi
65   ring
66 
67 theorem poissonBracket_mul_right (f a g : SymbolRing k n) :
68     poissonBracket f (a * g) =
69       a * poissonBracket f g + g * poissonBracket f a := by
70   simp only [poissonBracket, MvPolynomial.pderiv_mul]
71   rw [Finset.mul_sum, Finset.mul_sum, ← Finset.sum_add_distrib]
72   apply Finset.sum_congr rfl
73   intro i hi

76 /-- Gabber-shaped involutivity is preserved by arbitrary field-valued
77 coefficient extension. -/
78 theorem isInvolutive_map_scalarPolynomialMap
79     (J : Ideal (SymbolRing k n)) (hJ : IsInvolutive J) :
80     IsInvolutive
81       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) := by
82   let I := J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))
83   intro f hf
84   change f ∈ J.map
85     (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) at hf
86   rw [Ideal.map] at hf
87   induction hf using Submodule.span_induction with
88   | mem f hf =>
89       rcases hf with ⟨f₀, hf₀, rfl⟩
90       intro g hg
91       change g ∈ J.map
92         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) at hg
93       rw [Ideal.map] at hg
94       induction hg using Submodule.span_induction with
95       | mem g hg =>
96           rcases hg with ⟨g₀, hg₀, rfl⟩
97           rw [poissonBracket_scalarPolynomialMap]
98           exact Ideal.mem_map_of_mem _ (hJ f₀ hf₀ g₀ hg₀)
99       | zero => simp [poissonBracket]
100       | add g₁ g₂ _ _ hg₁ hg₂ =>
101           rw [poissonBracket_add_right]
102           exact I.add_mem hg₁ hg₂
103       | smul a g hgm hg =>
104           rw [smul_eq_mul, poissonBracket_mul_right]
105           exact I.add_mem (I.mul_mem_left _ hg) (I.mul_mem_right _ hgm)
106   | zero =>
107       intro g hg
108       simp [poissonBracket]
109   | add f₁ f₂ _ _ hf₁ hf₂ =>
110       intro g hg
111       rw [poissonBracket_add_left]
112       exact I.add_mem (hf₁ g hg) (hf₂ g hg)
113   | smul a f hfm hf =>
114       intro g hg
115       rw [smul_eq_mul, poissonBracket_mul_left]
116       exact I.add_mem (I.mul_mem_left _ (hf g hg)) (I.mul_mem_right _ hfm)

118 /-- Scalar extension of Gabber-shaped involutivity gives exactly the
119 base-relative condition consumed by the conormal proof. -/
120 theorem isBaseRelativePoisson_map_of_isInvolutive
121     (J : Ideal (SymbolRing k n)) (hJ : IsInvolutive J) :
122     BaseRelativePoisson.IsBaseRelativePoisson
123       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) := by
124   intro f hf g hg
125   exact isInvolutive_map_scalarPolynomialMap J hJ (baseLift f) hf g hg

127 /-- A Poisson ideal is stable under each phase-space partial derivative. -/
128 theorem pderiv_mem_of_isPoisson
129     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
130     {f : SymbolRing k n} (hf : f ∈ J) (i : PhaseVar n) :
131     MvPolynomial.pderiv i f ∈ J := by
132   rcases i with i | i
133   · have h := hJ f hf (MvPolynomial.X (Sum.inr i))
134     simpa [poissonBracket, Pi.single_apply] using h
135   · have h := hJ f hf (MvPolynomial.X (Sum.inl i))
136     have hn : -MvPolynomial.pderiv (Sum.inr i) f ∈ J := by
137       simpa [poissonBracket, Pi.single_apply] using h
138     simpa using J.neg_mem hn

140 /-- Every coordinate derivation preserves the ideal generated after scalar
141 extension from a Poisson ideal. -/
142 theorem pderiv_mem_map_scalarPolynomialMap
143     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
144     {f : SymbolRing K n}
145     (hf : f ∈ J.map
146       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)))
147     (i : PhaseVar n) :
148     MvPolynomial.pderiv i f ∈ J.map
149       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) := by
150   rw [Ideal.map] at hf
151   induction hf using Submodule.span_induction with
152   | mem f hf =>
153       rcases hf with ⟨g, hg, rfl⟩
154       change MvPolynomial.pderiv i
155           (MvPolynomial.map (algebraMap k K) g) ∈ _
156       rw [MvPolynomial.pderiv_map]
157       exact Ideal.mem_map_of_mem _ (pderiv_mem_of_isPoisson J hJ hg i)
158   | zero => simp
159   | add f g _ _ hf hg =>
160       simpa using (J.map
161         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).add_mem hf hg
162   | smul a f hfm hf =>
163       rw [smul_eq_mul, MvPolynomial.pderiv_mul]
164       exact (J.map
165         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).add_mem
166           ((J.map
167             (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_left _ hfm)
168           ((J.map
169             (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_left _ hf)

171 /-- Full Poisson closure is preserved by arbitrary field-valued coefficient
172 extension.  The target is the actual extended ideal `map`, not its radical. -/
173 theorem isPoisson_map_scalarPolynomialMap
174     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J) :
175     IsPoisson
176       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) := by
177   intro f hf g
178   apply (J.map
179     (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).sum_mem
180   intro i hi
181   apply (J.map
182     (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).sub_mem
183   · exact (J.map
184       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_right _
185         (pderiv_mem_map_scalarPolynomialMap J hJ hf (Sum.inl i))
186   · exact (J.map
187       (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).mul_mem_right _
188         (pderiv_mem_map_scalarPolynomialMap J hJ hf (Sum.inr i))

190 /-- The transported full Poisson theorem supplies the exact base-relative
191 fragment used by the conormal argument. -/
192 theorem isBaseRelativePoisson_map_scalarPolynomialMap
193     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J) :
194     BaseRelativePoisson.IsBaseRelativePoisson
195       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))) :=
196   BaseRelativePoisson.IsPoisson.isBaseRelativePoisson _
