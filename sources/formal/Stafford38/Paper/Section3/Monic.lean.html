8 # Section 3: the scalar case and Proposition 3.2
9 
10 The opening paragraph of Section 3 settles Bernstein degree `0`.  The proof of
11 Proposition 3.2 is split into its five paragraphs, `step_3_2_*`, and
12 `proposition_3_2` chains them; the second paragraph calls Lemma 3.1.
13 
14 Notation: `e_p = phaseBasis k (.inr 0)` is the basis vector of `𝕍` for
15 `p = ∂₁`; evaluation of a symbol at `e_p` is evaluation at
16 `axisPoint k (.inr 0)`, the same function.
17 -/

39 /-- **Section 3, opening paragraph.** If `N = deg_B d = 0`, then
40 `d ∈ B_0 = k` is a nonzero scalar `c`, hence a unit, and `1 = dR + FdS` with
41 `F = 1`, `R = d⁻¹ = c⁻¹`, `S = 0`.  Otherwise `N ≥ 1`. -/
42 theorem step_scalar_case {n : ℕ} {d : PresentedWeyl k n} (hd : d ≠ 0) (hN : bernsteinDegree k d = 0) :
43     ∃ c : k, c ≠ 0 ∧ d = algebraMap k (PresentedWeyl k n) c ∧
44       (1 : PresentedWeyl k n) = d * algebraMap k _ c⁻¹ + 1 * d * 0 := by
45   have hpiece : d ∈ bernsteinPiece k n 0 := (bernsteinDegree_le_iff k d).mp hN.le
46   obtain ⟨c, rfl⟩ := eq_algebraMap_of_mem_bernsteinPiece_zero k hpiece
47   have hc : c ≠ 0 := by
48     rintro rfl
49     exact hd (map_zero _)
50   refine ⟨c, hc, rfl, ?_⟩
51   rw [mul_zero, add_zero, ← map_mul, mul_inv_cancel₀ hc, map_one]

53 /-- A nonzero `d` lies in `B_N` for `N = deg_B d` (Definition 2.3), and its
54 Bernstein symbol `σ_N^B(d) ∈ B_N/B_{N-1}` (`Section2.symbol`), read in
55 `gr A_n(k) = k[x, ξ]` (`grEquivSymbolRing`), is the degree-`N` principal component,
56 nonzero and homogeneous of degree `N`. -/
57 theorem bernsteinSymbol_ne_zero {n : ℕ} {d : PresentedWeyl k n} (hd : d ≠ 0) :
58     d ∈ bernsteinPiece k n (bernsteinDegree k d) ∧
59       presentedPrincipalComponent k (@bernsteinWeight n) (bernsteinDegree k d) d ≠ 0 ∧
60       (presentedPrincipalComponent k (@bernsteinWeight n) (bernsteinDegree k d) d).IsHomogeneous
61         (bernsteinDegree k d) ∧
62       symbol (isFiltration_bernstein k n) d ≠ 0 ∧
63       grEquivSymbolRing k (@bernsteinWeight n)
64           (DirectSum.of _ (bernsteinDegree k d) (symbol (isFiltration_bernstein k n) d)) =
65         presentedPrincipalComponent k (@bernsteinWeight n) (bernsteinDegree k d) d := by
66   have hmem : d ∈ bernsteinPiece k n (bernsteinDegree k d) :=
67     mem_filtrationDegree (isFiltration_bernstein k n) d
68   obtain ⟨N, hpiece, hne⟩ := exists_top_bernstein_piece k hd
69   obtain rfl : bernsteinDegree k d = N := bernsteinDegree_eq_of_principal_ne_zero k hpiece hne
70   refine ⟨hmem, hne, principal_isHomogeneous k d, ?_, ?_⟩
71   swap
72   · let _ := MvPolynomial.weightedDecomposition k (@bernsteinWeight n)
73     simp only [grEquivSymbolRing, LinearEquiv.trans_apply]
74     erw [DFinsupp.mapRange.linearEquiv_apply, DFinsupp.mapRange_single]
75     erw [DirectSum.decomposeLinearEquiv_symm_apply, DirectSum.decompose_symm_of]
76     rfl
77   intro h0
78   exact hne ((grPiece_mk_eq_zero_iff k (@bernsteinWeight n) ⟨d, hmem⟩).mp h0)

80 /-- **Proposition 3.2, paragraph 1.** The Bernstein symbol `s` of `d` is a
81 nonzero homogeneous polynomial of degree `N ≥ 1` on `𝕍`.  A field of
82 characteristic zero is infinite, so `s(v) ≠ 0` for some `v`; as `s(0) = 0`,
83 `v ≠ 0`. -/
84 theorem step_3_2_nonvanishing [CharZero k] {n N : ℕ} {s : SymbolRing k n}
85     (hs : s.IsHomogeneous N) (hne : s ≠ 0) (hN : 1 ≤ N) :
86     ∃ v : PhaseVar n → k, MvPolynomial.eval v s ≠ 0 ∧ v ≠ 0 := by
87   obtain ⟨v, hv⟩ := exists_eval_ne_zero_of_homogeneous k hs hne
88   refine ⟨v, hv, ?_⟩
89   rintro rfl
90   apply hv
91   have hc : MvPolynomial.coeff 0 s = 0 := hs.coeff_eq_zero (by simp; omega)
92   simpa [MvPolynomial.eval_zero', MvPolynomial.constantCoeff] using hc

94 /-- **Proposition 3.2, paragraph 2 (first part).** By Lemma 3.1 there is an
95 `ω`-preserving automorphism `h` of `𝕍` with `h(e_p) = v`. -/
96 theorem step_3_2_h [CharZero k] {n : ℕ} {v : PhaseVar n → k} (hv : v ≠ 0)
97     (t : PhaseVar n) :
98     ∃ h : (PhaseVar n → k) ≃ₗ[k] (PhaseVar n → k),
99       (∀ x y, omega k n (h x) (h y) = omega k n x y) ∧ h (phaseBasis k t) = v := by
100   have he : phaseBasis k t ≠ 0 := by
101     intro h
102     have := congrFun h t
103     simp [phaseBasis] at this
104   exact (lemma_3_1 n (omega k n) (omega_isSymplectic k n).1 (omega_isSymplectic k n).2
105     (finrank_phaseSpace k n) hv).2 _ he

107 /-- **Proposition 3.2, paragraph 2 (second part).** If `h` preserves `ω`,
108 then `hᵀ J h = J`; inverting and using `J⁻¹ = -J` gives `h J hᵀ = J`, so the
109 matrix `M = h` (whose rows are the images `g(u_i)` of the generators for
110 `g = hᵀ`) is symplectic, and `M e_p = h(e_p)`. -/
111 theorem step_3_2_transpose {n : ℕ}
112     (h : (PhaseVar n → k) ≃ₗ[k] (PhaseVar n → k))
113     (hpres : ∀ x y, omega k n (h x) (h y) = omega k n x y) :
114     ∃ g : Matrix.symplecticGroup (Fin n) k,
115       g.1 = LinearMap.toMatrix' h.toLinearMap ∧
116         ∀ e, Matrix.mulVec g.1 e = h e :=
117   ⟨⟨_, toMatrix_mem_symplecticGroup k h.toLinearMap hpres⟩, rfl,
118     fun e => LinearMap.toMatrix'_mulVec h.toLinearMap e⟩
119 
120 /-- **Proposition 3.2, paragraph 3.** `Φ_g` maps `B_m` into `B_m`, and modulo
121 `B_{N-1}` it substitutes `u_i ↦ ∑_j M_ij u_j` in the commutative symbol, so
122 the Bernstein symbol of `Φ_g(d)` is `s_g : a ↦ s(Ma)`; at `e_p` its value is
123 `s(M e_p)`.  If this value is nonzero, `deg_B Φ_g(d) = N`. -/
124 theorem step_3_2_symbol {n N : ℕ} (g : Matrix.symplecticGroup (Fin n) k)
125     {d : PresentedWeyl k n} (hd : d ∈ bernsteinPiece k n N) (t : PhaseVar n) :
126     phiG k g d ∈ bernsteinPiece k n N ∧
127       MvPolynomial.eval (axisPoint k t)
128           (presentedPrincipalComponent k (@bernsteinWeight n) N (phiG k g d)) =
129         MvPolynomial.eval (Matrix.mulVec g.1 (phaseBasis k t))
130           (presentedPrincipalComponent k (@bernsteinWeight n) N d) ∧
131       (MvPolynomial.eval (Matrix.mulVec g.1 (phaseBasis k t))
132           (presentedPrincipalComponent k (@bernsteinWeight n) N d) ≠ 0 →
133         bernsteinDegree k (phiG k g d) = N) := by
134   have hpiece := phiG_mem_bernsteinPiece k g hd
135   have heval : MvPolynomial.eval (axisPoint k t)
136       (presentedPrincipalComponent k (@bernsteinWeight n) N (phiG k g d)) =
137       MvPolynomial.eval (Matrix.mulVec g.1 (phaseBasis k t))
138         (presentedPrincipalComponent k (@bernsteinWeight n) N d) := by
139     rw [phiG_principal k g hd, eval_symbolLinearAlgHom_axis, mulVec_phaseBasis]
140   refine ⟨hpiece, heval, fun hne => bernsteinDegree_eq_of_principal_ne_zero k hpiece ?_⟩
141   intro hzero
142   rw [hzero, map_zero] at heval
143   exact hne heval.symm

145 /-- **Proposition 3.2, paragraph 4.** With `c := s(v) ≠ 0`, a central unit,
146 `d̃ := c⁻¹ Φ_g(d)` satisfies `deg_B d̃ = N` and `σ_N^B(d̃)(e_p) = 1`. -/
147 theorem step_3_2_normalize {n N : ℕ} {e : PresentedWeyl k n} (he : e ∈ bernsteinPiece k n N)
148     (hdeg : bernsteinDegree k e = N) (t : PhaseVar n) {c : k} (hc : c ≠ 0)
149     (hval : MvPolynomial.eval (axisPoint k t)
150       (presentedPrincipalComponent k (@bernsteinWeight n) N e) = c) :
151     c⁻¹ • e ∈ bernsteinPiece k n N ∧ bernsteinDegree k (c⁻¹ • e) = N ∧
152       MvPolynomial.eval (axisPoint k t)
153         (presentedPrincipalComponent k (@bernsteinWeight n) N (c⁻¹ • e)) = 1 := by
154   have hpiece := (bernsteinPiece k n N).smul_mem c⁻¹ he
155   have heval : MvPolynomial.eval (axisPoint k t)
156       (presentedPrincipalComponent k (@bernsteinWeight n) N (c⁻¹ • e)) = 1 := by
157     rw [map_smul, MvPolynomial.smul_eval, hval, inv_mul_cancel₀ hc]
158   refine ⟨hpiece, ?_, heval⟩
159   · subst hdeg
160     refine bernsteinDegree_eq_of_principal_ne_zero k hpiece ?_
161     intro hzero
162     rw [hzero, map_zero] at heval
163     exact zero_ne_one heval

165 /-- At `e_p`, a degree-`L` Bernstein symbol takes the value of the PBW
166 coefficient of `∂₁^L`: every other monomial of the homogeneous symbol contains
167 another coordinate and vanishes at `e_p`. -/
168 theorem eval_ep_principal {n L : ℕ} (a : PresentedWeyl k (n + 1)) :
169     MvPolynomial.eval (axisPoint k (.inr (0 : Fin (n + 1))))
170         (presentedPrincipalComponent k (@bernsteinWeight (n + 1)) L a) =
171       MvPolynomial.coeff (Finsupp.single (.inr (0 : Fin (n + 1))) L)
172         (presentedNormalFormLinearEquiv k (n + 1) a) := by
173   rw [eval_axis_eq_pureCoefficient k _ (principal_isHomogeneous k a), coeff_axisPolynomial,
174     coeff_principal_p_pow k _ rfl]

176 /-- **Proposition 3.2, paragraph 5 (decomposition).** Using the normal form
177 `A_n(k) = ⊕_j p^j C` (Convention 2.2), `d̃ = ∑_{j=0}^N p^j a_j` with
178 `a_j ∈ C`; the Bernstein filtration is compatible with this decomposition, so
179 `deg_B a_j ≤ N - j`, in particular `a_N ∈ C ∩ B_0 = k`. -/
180 theorem step_3_2_decomposition {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
181     (hdt : dt ∈ bernsteinPiece k (n + 1) N) :
182     ∃ a : ℕ → PresentedWeyl k (n + 1),
183       (∀ j, a j ∈ C k n ∧ bernsteinDegree k (a j) ≤ N - j) ∧
184       dt = ∑ j ∈ Finset.range (N + 1), p k n ^ j * a j ∧
185       ∃ c : k, a N = algebraMap k _ c := by
186   obtain ⟨a, ha, hdec⟩ := exists_bounded_decomposition k n N hdt
187   refine ⟨a, ha, hdec, eq_algebraMap_of_mem_bernsteinPiece_zero k ?_⟩
188   rw [← bernsteinDegree_le_iff]
189   simpa using (ha N).2
190 
191 /-- **Proposition 3.2, paragraph 5 (leading coefficient).** The symbol is
192 `σ_N^B(d̃) = ∑_{j + deg_B a_j = N} σ(p)^j σ(a_j)`.  For `j < N` the factor
193 `σ_{N-j}^B(a_j)` has positive degree and contains no `ξ₁` (`a_j ∈ C`), so it
194 vanishes at `e_p`; evaluating at `e_p` gives `a_N = σ_N^B(d̃)(e_p) = 1`. -/
195 theorem step_3_2_leading {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
196     (a : ℕ → PresentedWeyl k (n + 1)) (ha : ∀ j, a j ∈ C k n ∧ bernsteinDegree k (a j) ≤ N - j)
197     (hdec : dt = ∑ j ∈ Finset.range (N + 1), p k n ^ j * a j) (c : k)
198     (hc : a N = algebraMap k _ c)
199     (hval : MvPolynomial.eval (axisPoint k (.inr (0 : Fin (n + 1))))
200       (presentedPrincipalComponent k (@bernsteinWeight (n + 1)) N dt) = 1) :
201     a N = 1 := by
202   have hsum : MvPolynomial.eval (axisPoint k (.inr (0 : Fin (n + 1))))
203       (presentedPrincipalComponent k (@bernsteinWeight (n + 1)) N dt) =
204       ∑ j ∈ Finset.range (N + 1), MvPolynomial.coeff
205         (Finsupp.single (.inr (0 : Fin (n + 1))) (N - j))
206         (presentedNormalFormLinearEquiv k (n + 1) (a j)) := by
207     rw [hdec, map_sum, map_sum]
208     apply Finset.sum_congr rfl
209     intro j hj
210     have hjN : j ≤ N := Nat.lt_succ_iff.mp (Finset.mem_range.mp hj)
211     rw [(principal_bernstein_p_pow_mul k hjN ((bernsteinDegree_le_iff k (a j)).mp (ha j).2)).2,
212       map_mul, map_pow, MvPolynomial.eval_X, eval_ep_principal]
213     simp [axisPoint]
214   rw [hsum, Finset.sum_range_succ, Finset.sum_eq_zero, zero_add, Nat.sub_self,
215     Finsupp.single_zero, hc, Algebra.algebraMap_eq_smul_one, map_smul,
216     presentedNormalFormLinearEquiv_one] at hval
217   · have hc1 : c = 1 := by simpa using hval
218     rw [hc, hc1, map_one]
219   · intro j hj
220     have hjN := Finset.mem_range.mp hj
221     by_contra hne
222     have := normalForm_C_no_p k (ha j).1 _ hne
223     simp at this
224     omega

226 /-- **Proposition 3.2, paragraph 5.** Hence
227 `d̃ = p^N + ∑_{j<N} p^j a_j` with `a_j ∈ C` and `deg_B a_j ≤ N - j`. -/
228 theorem step_3_2_monic {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
229     (hdt : dt ∈ bernsteinPiece k (n + 1) N)
230     (hval : MvPolynomial.eval (axisPoint k (.inr (0 : Fin (n + 1))))
231       (presentedPrincipalComponent k (@bernsteinWeight (n + 1)) N dt) = 1) :
232     IsMonicNormalForm k N dt := by
233   obtain ⟨a, ha, hdec, c, hc⟩ := step_3_2_decomposition k hdt
234   have hN := step_3_2_leading k a ha hdec c hc hval
235   refine ⟨a, fun j _ => ha j, ?_⟩
236   rw [hdec, Finset.sum_range_succ, hN, mul_one]
237   exact add_comm _ _

239 /-- **Proposition 3.2.** Let `0 ≠ d ∈ A_n(k)` with `N := deg_B d ≥ 1`.  There
240 are an automorphism `Φ_g` of `A_n(k)` and `c ∈ k^×` with
241 `d̃ := c⁻¹ Φ_g(d) = p^N + ∑_{j<N} p^j a_j`, `a_j ∈ C`, `deg_B a_j ≤ N - j`
242 (eq. (3.1)), and `deg_B d̃ = N`. -/
243 theorem proposition_3_2 [CharZero k] {n : ℕ} {d : PresentedWeyl k (n + 1)} (hd : d ≠ 0)
244     (hN : 1 ≤ bernsteinDegree k d) :
245     ∃ (g : Matrix.symplecticGroup (Fin (n + 1)) k) (c : k), c ≠ 0 ∧
246       IsMonicNormalForm k (bernsteinDegree k d) (c⁻¹ • phiG k g d) ∧
247       bernsteinDegree k (c⁻¹ • phiG k g d) = bernsteinDegree k d := by
248   obtain ⟨hpiece, hne, hhom, -, -⟩ := bernsteinSymbol_ne_zero k hd
249   obtain ⟨v, hv, hv0⟩ := step_3_2_nonvanishing k hhom hne hN
250   obtain ⟨h, hpres, hhe⟩ := step_3_2_h k hv0 (.inr (0 : Fin (n + 1)))
251   obtain ⟨g, -, hmul⟩ := step_3_2_transpose k h hpres
252   obtain ⟨hpiece', heval, hdeg'⟩ := step_3_2_symbol k g hpiece (.inr (0 : Fin (n + 1)))
253   rw [hmul, hhe] at heval hdeg'
254   obtain ⟨hpiece'', hdeg'', hval⟩ := step_3_2_normalize k hpiece' (hdeg' hv) _ hv heval
255   exact ⟨g, _, hv, step_3_2_monic k hpiece'' hval, hdeg''⟩
