7 /-!
8 # The decomposition `d = ∑_j p^j a_j` is compatible with `B_•`
9 
10 Proof of Proposition 3.2, paragraph 5, first sentences: in the normal form
11 `A_n(k) = ⊕_j p^j C` of Convention 2.2, an element `d ∈ B_N` has
12 `d = ∑_{j ≤ N} p^j a_j` with `a_j ∈ C` and `deg_B a_j ≤ N - j`.
13 
14 The bounds are read through the coefficient-left normal form of the iterated
15 Ore presentation and its change to right coefficients.  The argument is that of
16 `WeylPaperRightMonic.presentedOuterRightBernsteinBound`, which is stated only
17 for PBW-monic elements although it uses only `d ∈ B_N`; its private helpers are
18 repeated here with that weaker hypothesis.
19 -/

104 private theorem rightPBWBasis_repr_monomial' {B : Type*} [Ring B] [Nontrivial B]
105     (D : OreDivisionDerivation B) (l : ℕ) (b : B) :
106     (rightOrePBWBasis D).repr (normalForm D (Polynomial.monomial l b)) =
107       ∑ ij ∈ Finset.HasAntidiagonal.antidiagonal l,
108         l.choose ij.1 •
109           (if Even ij.1 then Finsupp.single ij.2 (MulOpposite.op ((D^[ij.1]) b))
110           else -Finsupp.single ij.2 (MulOpposite.op ((D^[ij.1]) b))) := by
111   rw [normalForm_monomial_reverse]
112   simp only [map_sum, map_nsmul]
113   apply Finset.sum_congr rfl
114   intro ij _
115   by_cases he : Even ij.1
116   · rw [he.neg_one_pow, one_mul]
117     change l.choose ij.1 •
118       (rightOrePBWBasis D).repr
119         (MulOpposite.op ((D^[ij.1]) b) • rightPBWMonomial D ij.2) = _
120     rw [rightPBWBasis_repr_single', ite_eq_left he]
121   · have ho : Odd ij.1 := Nat.not_even_iff_odd.mp he
122     rw [ho.neg_one_pow, neg_one_mul, map_neg]
123     change l.choose ij.1 • -((rightOrePBWBasis D).repr
124       (MulOpposite.op ((D^[ij.1]) b) • rightPBWMonomial D ij.2)) = _
125     rw [rightPBWBasis_repr_single', ite_eq_right he]

128     [Nontrivial B] (D : OreDivisionDerivation B) (l j : ℕ) (b : B) (hlj : l < j) :
129     ((rightOrePBWBasis D).repr (normalForm D (Polynomial.monomial l b))) j = 0 := by

135   by_cases he : Even ij.1
136   · rw [ite_eq_left he, Finsupp.smul_single]
137     simp [hsecond]
138   · rw [ite_eq_right he]
139     simp only [smul_neg]

143 private theorem pairRightBasis_repr_monomial_bound' (n N l : ℕ)
144     (b : CoordinateStage (B := IteratedPairStage k n)) (hl : l ≤ N)
145     (hb : b ∈ coordinateBernsteinPiece k n (N - l)) :
146     rightBernsteinBound k n N
147       ((pairStage_rightOrePBWBasis (B := IteratedPairStage k n)).repr
148         (normalForm (coordinateDerivation :
149           OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)))
150           (Polynomial.monomial l b))) := by
151   change rightBernsteinBound k n N
152     ((rightOrePBWBasis (coordinateDerivation :
153       OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)))).repr
154       (normalForm (coordinateDerivation :
155         OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)))
156         (Polynomial.monomial l b)))
157   rw [rightPBWBasis_repr_monomial']
158   apply rightBernsteinBound_sum' k n N
159   intro ij hij
160   have hadd : ij.1 + ij.2 = l := Finset.HasAntidiagonal.mem_antidiagonal.mp hij
161   have hderiv : (((coordinateDerivation :
162       OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)))^[ij.1]) b) ∈
163       coordinateBernsteinPiece k n (N - ij.2) :=
164     coordinateBernsteinPiece_mono k n (N - l) (N - ij.2) (by omega)
165       (coordinateDerivation_iterate_mem_coordinateBernsteinPiece k hb)
166   have hsingle := rightBernsteinBound_single k n N ij.2 (b := MulOpposite.op
167     (((coordinateDerivation :
168       OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)))^[ij.1]) b))
169     (by simpa using hderiv)
170   have hsign : rightBernsteinBound k n N
171       (if Even ij.1 then Finsupp.single ij.2 (MulOpposite.op
172           (((coordinateDerivation :
173             OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)))^[ij.1]) b))
174        else -Finsupp.single ij.2 (MulOpposite.op
175           (((coordinateDerivation :
176             OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)))^[ij.1]) b))) := by
177     split_ifs
178     · exact hsingle
179     · exact rightBernsteinBound_neg k n N hsingle
180   exact rightBernsteinBound_nsmul k n N (Nat.choose l ij.1) hsign

183 above `N`. -/
184 private theorem right_repr_bounds (n N : ℕ) {d : PresentedWeyl k (n + 1)}
185     (hd : d ∈ bernsteinPiece k (n + 1) N) :
186     rightBernsteinBound k n N
187         ((pairStage_rightOrePBWBasis (B := IteratedPairStage k n)).repr
188           (presentedToIterated k (n + 1) d)) ∧
189       ∀ q, N < q → ((pairStage_rightOrePBWBasis (B := IteratedPairStage k n)).repr
190           (presentedToIterated k (n + 1) d)) q = 0 := by
191   let H := presentedOuterPolynomial k n d
192   let D : OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)) :=
193     coordinateDerivation
194   have hnormal : normalForm D H = presentedToIterated k (n + 1) d := by
195     dsimp [D, H, presentedOuterPolynomial]
196     exact (normalFormAddEquiv coordinateDerivation).apply_symm_apply _
197   have hsum : H = ∑ l ∈ H.support, Polynomial.monomial l (H.coeff l) := by
198     simpa [Polynomial.sum_def] using (Polynomial.sum_monomial_eq H).symm
199   have hrepr : (rightOrePBWBasis D).repr (normalForm D H) =
200       ∑ l ∈ H.support, (rightOrePBWBasis D).repr
201         (normalForm D (Polynomial.monomial l (H.coeff l))) := by
202     conv_lhs => rw [hsum]
203     change (rightOrePBWBasis D).repr
204         ((normalFormAddEquiv D) (∑ l ∈ H.support, Polynomial.monomial l (H.coeff l))) = _
205     rw [map_sum, map_sum]
206     rfl
207   have hlN : ∀ l ∈ H.support, l ≤ N := by
208     intro l hl
209     by_contra h
210     exact (Polynomial.mem_support_iff.mp hl) (outer_coeff_zero_of_mem k n N hd (by omega))
211   rw [← hnormal]
212   refine ⟨?_, fun q hq => ?_⟩
213   · change rightBernsteinBound k n N ((rightOrePBWBasis D).repr (normalForm D H))
214     rw [hrepr]
215     apply rightBernsteinBound_sum' k n N
216     intro l hl
217     exact pairRightBasis_repr_monomial_bound' k n N l (H.coeff l) (hlN l hl)
218       (presentedOuterCoefficient_mem_coordinateBernsteinPiece k n N l d hd)
219   · change ((rightOrePBWBasis D).repr (normalForm D H)) q = 0
220     rw [hrepr, Finset.sum_apply']
221     exact Finset.sum_eq_zero fun l hl =>
222       rightPBWBasis_repr_monomial_coeff_zero_of_lt' D l q (H.coeff l)
223         (lt_of_le_of_lt (hlN l hl) hq)

225 /-- The PBW coefficients of `d ∈ B_N` in `A_n(k) = ⊕_j p^j C` satisfy
226 `deg_B a_j ≤ N - j`. -/
227 theorem step_bounded_coefficients (n N : ℕ) {d : PresentedWeyl k (n + 1)}
228     (hd : d ∈ bernsteinPiece k (n + 1) N) :
229     ∃ a : ℕ → PresentedWeyl k (n + 1),
230       (∀ j, a j ∈ C k n ∧ bernsteinDegree k (a j) ≤ N - j) ∧
231         d = ∑ j ∈ Finset.range (N + 1), p k n ^ j * a j := by
232   let b := pairStage_rightOrePBWBasis (B := IteratedPairStage k n)
233   let c := b.repr (presentedToIterated k (n + 1) d)
234   obtain ⟨hbound, hzero⟩ := right_repr_bounds k n N hd
235   have hsub : c.support ⊆ Finset.range (N + 1) := by
236     intro j hj
237     by_contra h
238     exact (Finsupp.mem_support_iff.mp hj) (hzero j (by simp at h; omega))
239   refine ⟨fun j => coordinateStageToPresented k n (c j).unop, fun j => ⟨?_, ?_⟩, ?_⟩
240   · rw [C_eq_range]
241     exact ⟨_, rfl⟩
242   · rw [bernsteinDegree_le_iff]
243     exact hbound j
244   · have hd' : presentedToIterated k (n + 1) d =
245         ∑ j ∈ Finset.range (N + 1), c j • rightPBWMonomial (coordinateDerivation :
246           OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n))) j := by
247       conv_lhs => rw [← b.linearCombination_repr (presentedToIterated k (n + 1) d)]
248       rw [Finsupp.linearCombination_apply, Finsupp.sum_of_support_subset _ hsub _ (by simp)]
249       apply Finset.sum_congr rfl
250       intro j _
251       simp [b, pairStage_rightOrePBWBasis]
252       rfl
253     have hback : d = stageToPresented k n (presentedToIterated k (n + 1) d) :=
254       (DFunLike.congr_fun (iteratedToPresented_comp_presentedToIterated k (n + 1)) d).symm
255     rw [hback, hd', map_sum]
256     apply Finset.sum_congr rfl
257     intro j _
258     exact stageToPresented_rightTerm k n j (c j).unop

263     ∑ j ∈ Finset.range M, x ^ j * (if j < L then a j else 0) =
264       ∑ j ∈ Finset.range L, x ^ j * a j := by

273 /-- **Proposition 3.2, paragraph 5 (the decomposition).** Using
274 `A_n(k) = ⊕_j p^j C` (Convention 2.2: existence and uniqueness of the normal
275 form), `d ∈ B_N` is `d = ∑_{j=0}^N p^j a_j` with `a_j ∈ C`; the Bernstein
276 filtration is compatible with the decomposition, so `deg_B a_j ≤ N - j`. -/
277 theorem exists_bounded_decomposition (n N : ℕ) {d : PresentedWeyl k (n + 1)}
278     (hd : d ∈ bernsteinPiece k (n + 1) N) :
279     ∃ a : ℕ → PresentedWeyl k (n + 1),
280       (∀ j, a j ∈ C k n ∧ bernsteinDegree k (a j) ≤ N - j) ∧
281         d = ∑ j ∈ Finset.range (N + 1), p k n ^ j * a j := by
282   -- The normal form of Convention 2.2.
283   obtain ⟨L, a, haC, hda⟩ := convention_2_2_exists k n d
284   -- Its coefficients are those bounded by the Bernstein filtration.
285   obtain ⟨b, hb, hdb⟩ := step_bounded_coefficients k n N hd
286   let M := max L (N + 1)
287   let a' : ℕ → PresentedWeyl k (n + 1) := fun j => if j < L then a j else 0
288   let b' : ℕ → PresentedWeyl k (n + 1) := fun j => if j < N + 1 then b j else 0
289   have ha'C : ∀ j, a' j ∈ C k n := fun j => by
290     by_cases hj : j < L
291     · simp only [a', hj, ↓reduceIte]; exact haC j
292     · simp only [a', hj, ↓reduceIte]; exact Subalgebra.zero_mem _
293   have hb'C : ∀ j, b' j ∈ C k n := fun j => by
294     by_cases hj : j < N + 1
295     · simp only [b', hj, ↓reduceIte]; exact (hb j).1
296     · simp only [b', hj, ↓reduceIte]; exact Subalgebra.zero_mem _
297   have hsum : ∑ j ∈ Finset.range M, p k n ^ j * (a' j - b' j) = 0 := by
298     simp only [mul_sub, Finset.sum_sub_distrib]
299     rw [sum_range_pad _ a (le_max_left L (N + 1)), sum_range_pad _ b (le_max_right L (N + 1)),
300       ← hda, ← hdb, sub_self]
301   have huniq := convention_2_2_unique k n M (fun j => a' j - b' j)
302     (fun j => Subalgebra.sub_mem _ (ha'C j) (hb'C j)) hsum
303   have hab : ∀ j, a' j = b' j := fun j => by
304     by_cases hj : j < M
305     · exact sub_eq_zero.mp (huniq j hj)
306     · have h1 : ¬ j < L := fun h => hj (lt_of_lt_of_le h (le_max_left _ _))
307       have h2 : ¬ j < N + 1 := fun h => hj (lt_of_lt_of_le h (le_max_right _ _))
308       simp only [a', b', h1, h2, ↓reduceIte]
309   refine ⟨a', fun j => ⟨ha'C j, ?_⟩, ?_⟩
310   · rw [hab j]
311     by_cases hj : j < N + 1
312     · simp only [b', hj, ↓reduceIte]; exact (hb j).2
313     · simp only [b', hj, ↓reduceIte]
314       exact (bernsteinDegree_le_iff k 0).mpr (Submodule.zero_mem _)
315   · rw [hdb]
316     apply Finset.sum_congr rfl
317     intro j hj
318     rw [hab j]
319     simp only [b', Finset.mem_range.mp hj, ↓reduceIte]
