10 # Definition 3.4, Lemma 3.5 and Proposition 3.6
11 
12 For the normal form `d̃` of (3.1) with `N = deg_B d̃`, the order principal
13 symbol `σ_N^ord(d̃)` is `presentedPrincipalComponent k orderWeight N d̃` in
14 `k[x, ξ] = SymbolRing k (n + 1)`, where `Sum.inl i` indexes `x_{i+1}` and
15 `Sum.inr i` indexes `ξ_{i+1}`.  A polynomial `P ∈ k[ξ_1, …, ξ_{n+1}]` is an
16 element of `MvPolynomial (Fin (n + 1)) k`, regarded in `k[x, ξ]` through
17 `rename Sum.inr`.
18 
19 `Q = A/𝓘` carries the order filtration `Q_m = q F_m` induced by `q = [1]`;
20 its associated graded module is `grQ`, the external direct sum of the
21 quotients `Q_m / Q_{m-1}` with its `k[x, ξ]`-module structure
22 (Definition 2.4, `Stafford38.Paper.Section2.grCyclic`).
23 -/

48 /-- **Definition 3.4.** `P = ∑_{i=0}^N c_i ξ₁^i ∈ k[ξ_1, …, ξ_n]`, with
49 `c_i ∈ k[ξ_2, …, ξ_n]`, is monic of degree `N` in `ξ₁` if `c_N = 1`.  Here
50 `finSuccEquiv` writes `P` as a polynomial in `ξ₁ = X 0` with coefficients in
51 the remaining variables. -/
52 def IsMonicInXi1 {n : ℕ} (N : ℕ) (P : MvPolynomial (Fin (n + 1)) k) : Prop :=
53   (MvPolynomial.finSuccEquiv k n P).natDegree ≤ N ∧
54     (MvPolynomial.finSuccEquiv k n P).coeff N = 1

56 /-- The order principal symbol `σ_N^ord(d̃)`. -/
57 abbrev orderSymbol {n : ℕ} (N : ℕ) (dt : PresentedWeyl k n) : SymbolRing k n :=
58   presentedPrincipalComponent k (@orderWeight n) N dt

60 /-- **Lemma 3.5, paragraph 1.** Since `deg_B d̃ = N`, every PBW monomial
61 `x^α ∂^β` of `d̃` has `|α| + |β| ≤ N`; a monomial contributing to the order-`N`
62 part has `|β| = N`, hence `|α| = 0`.  So `d̃ ∈ F_N` and its order-`N` part is a
63 `k`-linear combination of `∂^β` with `|β| = N`: `σ_N^ord(d̃)` involves no `x_i`
64 and is homogeneous of degree `N`. -/
65 theorem step_3_5_fibre {n N : ℕ} {dt : PresentedWeyl k n} (hdeg : bernsteinDegree k dt = N) :
66     dt ∈ orderPiece k n N ∧ IsFibreOnly k (orderSymbol k N dt) ∧
67       (orderSymbol k N dt).IsHomogeneous N := by
68   have hB : dt ∈ bernsteinPiece k n N := (bernsteinDegree_le_iff k dt).mp hdeg.le
69   have hfibre : IsFibreOnly k (orderSymbol k N dt) := by
70     intro m hm i
71     change AddMonoidAlgebra.coeff (orderSymbol k N dt) m ≠ 0 at hm
72     rw [coeff_presentedPrincipalComponent] at hm
73     by_cases horder : monomialWeight (@orderWeight n) m = N
74     · rw [ite_eq_left horder] at hm
75       exact coordinateExponent_eq_zero_of_order_eq_of_bernstein_le m horder
76         ((mem_presentedWeightPiece k (@bernsteinWeight n) N dt).mp hB m hm) i
77     · rw [ite_eq_right horder] at hm
78       exact (hm rfl).elim
79   refine ⟨bernsteinPiece_le_orderPiece k n N hB, hfibre, ?_⟩
80   apply isHomogeneous_of_orderHomogeneous_of_isFibreOnly _ hfibre
81   exact MvPolynomial.weightedHomogeneousComponent_isWeightedHomogeneous
82     (w := @orderWeight n) (n := N) (presentedNormalFormLinearEquiv k n dt)

84 /-- **Lemma 3.5, paragraph 1 (conclusion).** A symbol without `x_i` is a
85 polynomial `P(ξ)` in the fibre variables alone, homogeneous of the same
86 degree. -/
87 theorem step_3_5_fibre_polynomial {n N : ℕ} {s : SymbolRing k n} (hfibre : IsFibreOnly k s)
88     (hs : s.IsHomogeneous N) :
89     ∃ P : MvPolynomial (Fin n) k, s = MvPolynomial.rename Sum.inr P ∧ P.IsHomogeneous N := by
90   obtain ⟨P, hP⟩ := MvPolynomial.exists_rename_eq_of_vars_subset_range s Sum.inr
91     Sum.inr_injective (by
92       intro i hi
93       rw [Finset.mem_coe, MvPolynomial.mem_vars_iff_mem_support] at hi
94       obtain ⟨m, hm, hmi⟩ := hi
95       rcases i with i | i
96       · exact absurd (hfibre m (MvPolynomial.mem_support_iff.mp hm) i)
97           (Finsupp.mem_support_iff.mp hmi)
98       · exact ⟨i, rfl⟩)
99   refine ⟨P, hP.symm, ?_⟩
100   rw [← MvPolynomial.IsHomogeneous.rename_isHomogeneous_iff Sum.inr_injective, hP]
101   exact hs

103 /-- **Lemma 3.5, paragraph 2 (the terms `p^j a_j`, `j < N`).** Since
104 `a_j ∈ C` involves no further factor of `p = ∂₁`, every order-`N` monomial of
105 `p^j a_j` has `∂₁`-exponent exactly `j < N` (and so carries a positive power of
106 some transverse `∂_i`); it contributes only to the coefficient `c_j` of
107 `ξ₁^j`, never to `c_N` (tex lines 494–500).  The monomials are read in the
108 order symbol, `σ_N^ord(p^j a_j) = ξ₁^j σ_{N-j}^ord(a_j)`, whose monomials are
109 the order-`N` PBW monomials of `p^j a_j`. -/
110 theorem step_3_5_lower_term {n j N : ℕ} (hj : j < N) {a : PresentedWeyl k (n + 1)}
111     (haC : a ∈ C k n) (ha : bernsteinDegree k a ≤ N - j) (m : PhaseVar (n + 1) →₀ ℕ)
112     (hm : MvPolynomial.coeff m (orderSymbol k N (p k n ^ j * a)) ≠ 0) :
113     m (.inr 0) = j := by
114   have haF : a ∈ orderPiece k (n + 1) (N - j) :=
115     bernsteinPiece_le_orderPiece k (n + 1) _ ((bernsteinDegree_le_iff k a).mp ha)
116   rw [orderSymbol, principal_order_p_pow_mul k hj.le haF] at hm
117   simp only [MvPolynomial.coeff] at hm
118   rw [MvPolynomial.X_pow_eq_monomial, MvPolynomial.coeff_monomial_mul'] at hm
119   split_ifs at hm with hle
120   · rw [one_mul, coeff_presentedPrincipalComponent] at hm
121     split_ifs at hm
122     swap
123     · exact absurd rfl hm
124     have h0 := normalForm_C_no_p k haC _ hm
125     have h1 := Finsupp.le_def.mp hle (.inr 0)
126     simp only [Finsupp.coe_tsub, Pi.sub_apply, Finsupp.single_eq_same] at h0 h1
127     omega
128   · exact absurd rfl hm

130 /-- **Lemma 3.5, paragraph 2 (the coefficient `c_N`).** The monomial
131 `p^N = ∂₁^N` occurs with coefficient `1`, and by `step_3_5_lower_term` no other
132 term contributes to `c_N`: the coefficient of `ξ₁^N` in `σ_N^ord(d̃)` is `1`
133 (tex lines 494–503). -/
134 theorem step_3_5_pure_coefficient {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
135     (hform : IsMonicNormalForm k N dt) :
136     MvPolynomial.coeff (Finsupp.single (.inr (0 : Fin (n + 1))) N) (orderSymbol k N dt) = 1 := by
137   obtain ⟨a, ha, rfl⟩ := hform
138   obtain ⟨-, hσN⟩ := p_pow_piece_principal k (@orderWeight (n + 1)) rfl N
139   rw [orderSymbol, map_add, map_sum, hσN]
140   simp only [MvPolynomial.coeff]
141   rw [AddMonoidAlgebra.coeff_add, Finsupp.add_apply, MvPolynomial.coeff_sum,
142     Finset.sum_eq_zero, add_zero, MvPolynomial.X_pow_eq_monomial, MvPolynomial.coeff_monomial,
143     ite_eq_left rfl]
144   intro j hj
145   have hjN := Finset.mem_range.mp hj
146   by_contra hne
147   have := step_3_5_lower_term k hjN (ha j hjN).1 (ha j hjN).2 _ hne
148   simp at this
149   omega

163 /-- **Lemma 3.5, paragraph 2 (conclusion).** If `P` is homogeneous of degree
164 `N` and its coefficient of `ξ₁^N` is `1`, then `c_N = 1` identically, so `P`
165 is monic of degree `N` in `ξ₁` (Definition 3.4), `deg P = N`, and evaluating
166 at `(1, 0, …, 0)` gives `P(1, 0, …, 0) = 1`. -/
167 theorem step_3_5_monic {n N : ℕ} {P : MvPolynomial (Fin (n + 1)) k} (hP : P.IsHomogeneous N)
168     (hcoeff : MvPolynomial.coeff (Finsupp.single 0 N) P = 1) :
169     IsMonicInXi1 k N P ∧ P.totalDegree = N ∧
170       MvPolynomial.eval (fun i => if i = 0 then 1 else 0) P = 1 := by
171   have hne : P ≠ 0 := by
172     rintro rfl
173     simp at hcoeff
174   refine ⟨⟨?_, ?_⟩, hP.totalDegree hne, ?_⟩
175   · rw [MvPolynomial.natDegree_finSuccEquiv]
176     exact (MvPolynomial.degreeOf_le_totalDegree P 0).trans hP.totalDegree_le
177   · ext m
178     rw [MvPolynomial.finSuccEquiv_coeff_coeff]
179     by_cases hm : m = 0
180     · subst m
181       have h0 : (Finsupp.cons N (0 : Fin n →₀ ℕ)) = Finsupp.single 0 N := by
182         ext i
183         refine Fin.cases ?_ (fun i => ?_) i <;> simp
184       rw [h0]
185       simpa using hcoeff
186     · rw [MvPolynomial.coeff_one, ite_eq_right (Ne.symm hm)]
187       apply hP.coeff_eq_zero
188       rw [degree_cons]
189       have : m.degree ≠ 0 := fun h => hm ((Finsupp.degree_eq_zero_iff m).mp h)
190       omega
191   · have hsym := eval_axis_eq_one_of_pureCoefficient_one (k := k) (n := n + 1) (N := N)
192       (.inr (0 : Fin (n + 1))) (P := MvPolynomial.rename Sum.inr P)
193       ((MvPolynomial.IsHomogeneous.rename_isHomogeneous_iff Sum.inr_injective).mpr hP) (by
194         have h := MvPolynomial.coeff_rename_mapDomain
195           (Sum.inr : Fin (n + 1) → Fin (n + 1) ⊕ Fin (n + 1)) Sum.inr_injective P
196           (Finsupp.single 0 N)
197         rw [Finsupp.mapDomain_single] at h
198         simpa [MvPolynomial.coeff] using h.trans hcoeff)
199     have hpt : (axisPoint k (.inr (0 : Fin (n + 1))) ∘ Sum.inr : Fin (n + 1) → k) =
200         fun i => if i = 0 then 1 else 0 := by
201       funext i
202       simp [axisPoint]
203     rwa [MvPolynomial.eval_rename, hpt] at hsym

205 /-- **Lemma 3.5.** Let `d̃ = p^N + ∑_{j<N} p^j a_j` with `a_j ∈ C`,
206 `deg_B a_j ≤ N - j` as in (3.1), and `N = deg_B d̃`.  Then `d̃` has order `N`,
207 and its order principal symbol is a polynomial in `ξ` alone,
208 `σ_N^ord(d̃) = P(ξ)` with `P ∈ k[ξ_1, …, ξ_n]`, `deg P = N` (eq. (3.3)); `P` is
209 homogeneous, monic of degree `N` in `ξ₁`, and `P(1, 0, …, 0) = 1`
210 (eq. (3.4)). -/
211 theorem lemma_3_5 {n N : ℕ} {dt : PresentedWeyl k (n + 1)} (hform : IsMonicNormalForm k N dt)
212     (hdeg : bernsteinDegree k dt = N) :
213     (dt ∈ orderPiece k (n + 1) N ∧ ∀ j < N, dt ∉ orderPiece k (n + 1) j) ∧
214       ∃ P : MvPolynomial (Fin (n + 1)) k,
215         orderSymbol k N dt = MvPolynomial.rename Sum.inr P ∧
216           P.IsHomogeneous N ∧ P.totalDegree = N ∧ IsMonicInXi1 k N P ∧
217           MvPolynomial.eval (fun i => if i = 0 then 1 else 0) P = 1 := by
218   obtain ⟨hF, hfibre, hhom⟩ := step_3_5_fibre k hdeg
219   obtain ⟨P, hsP, hP⟩ := step_3_5_fibre_polynomial k hfibre hhom
220   have hpure := step_3_5_pure_coefficient k hform
221   have hne : orderSymbol k N dt ≠ 0 := by
222     intro h
223     rw [h] at hpure
224     simp at hpure
225   have hcoeff : MvPolynomial.coeff (Finsupp.single 0 N) P = 1 := by
226     have h := MvPolynomial.coeff_rename_mapDomain
227       (Sum.inr : Fin (n + 1) → Fin (n + 1) ⊕ Fin (n + 1)) Sum.inr_injective P
228       (Finsupp.single 0 N)
229     rw [Finsupp.mapDomain_single, ← hsP] at h
230     simpa [MvPolynomial.coeff] using h.symm.trans hpure
231   obtain ⟨hmonic, htot, heval⟩ := step_3_5_monic k hP hcoeff
232   exact ⟨step_3_5_order k hF hne, P, hsP, hP, htot, hmonic, heval⟩

234 /-- Equation (3.4) read in `k[x, ξ]`: `σ_N^ord(d̃)` takes the value `1` at the
235 covector `ξ = (1, 0, …, 0)` over every base point `x = 0`, i.e. at
236 `axisPoint (.inr 0)`. -/
237 theorem lemma_3_5_eval_axis {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
238     (hform : IsMonicNormalForm k N dt) (hdeg : bernsteinDegree k dt = N) :
239     MvPolynomial.eval (axisPoint k (.inr (0 : Fin (n + 1)))) (orderSymbol k N dt) = 1 := by
240   obtain ⟨-, P, hsP, -, -, -, heval⟩ := lemma_3_5 k hform hdeg
241   have hpt : (axisPoint k (.inr (0 : Fin (n + 1))) ∘ Sum.inr : Fin (n + 1) → k) =
242       fun i => if i = 0 then 1 else 0 := by
243     funext i
244     simp [axisPoint]
245   rw [hsP, MvPolynomial.eval_rename, hpt]
246   exact heval

248 /-- The associated graded module `gr Q` of `Q = A/𝓘` for the order filtration

250 abbrev grQ {n : ℕ} (N : ℕ) (dt : PresentedWeyl k (n + 1)) :=

253 /-- **Proposition 3.6, proof, first sentence.** `Q_•` is good because `Q` is
254 cyclic: `Q_m = qF_m` is a compatible filtration of `Q` (Definition 2.4), and
255 `gr Q` is finite over `k[x, ξ]`, generated by `q̄`. -/
256 theorem step_3_6_generated {n N : ℕ} (dt : PresentedWeyl k (n + 1)) :
257     IsCompatibleFiltration (orderFiltration k (n + 1))
258         (cyclicFiltration (orderFiltration k (n + 1)) (qClass k (idealI k N dt))) ∧
259       Module.Finite (SymbolRing k (n + 1)) (grQ k N dt) ∧
260       ∀ m : grQ k N dt, ∃ f : SymbolRing k (n + 1), f • qBar k (idealI k N dt) = m :=
261   ⟨cyclicFiltration_order_isCompatible k _, grCyclic_good k _, grCyclic_generated k _⟩
262 
263 /-- **Proposition 3.6, proof, second sentence.** `q d̃ = [d̃] = 0` in `Q`
264 because `d̃ ∈ 𝓘`, and `d̃ ∈ F_N`, so `q d̃ ∈ Q_N`; its image `q̄ · P` in
265 `Q_N/Q_{N-1}` is zero. -/
266 theorem step_3_6_kills {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
267     (hF : dt ∈ orderPiece k (n + 1) N) :
268     orderSymbol k N dt • qBar k (idealI k N dt) = 0 := by
269   have h := smul_orderAssociatedGradedOf_mk_eq_of_mul k (idealI k N dt)
270     (orderPieceOne (n := n + 1) k) (⟨dt, hF⟩ : orderPiece k (n + 1) N)
271   change orderSymbol k N dt • qBar k (idealI k N dt) = _ at h
272   rw [h]
273   have hzero : orderPieceToQuotientGraded k (idealI k N dt) (0 + N)
274       ⟨(orderPieceOne (n := n + 1) k : PresentedWeyl k (n + 1)) * dt,
275         mul_mem_orderPiece k (orderPieceOne (n := n + 1) k).property hF⟩ = 0 := by
276     have hpiece : orderPieceToQuotientPiece k (idealI k N dt) (0 + N)
277         ⟨(orderPieceOne (n := n + 1) k : PresentedWeyl k (n + 1)) * dt,
278           mul_mem_orderPiece k (orderPieceOne (n := n + 1) k).property hF⟩ = 0 := by
279       apply Subtype.ext
280       change Submodule.Quotient.mk ((1 : PresentedWeyl k (n + 1)) * dt) = 0
281       rw [one_mul, Submodule.Quotient.mk_eq_zero]
282       exact firstGenerator_mem (x₁ k n) dt N
283     rw [orderPieceToQuotientGraded, LinearMap.comp_apply, hpiece, map_zero]
284   rw [hzero]
285   exact map_zero _

287 /-- **Proposition 3.6.** Equip `Q` with the order filtration `Q_m = qF_m`
288 induced by the cyclic generator `q = [1]`, and write `P = σ_N^ord(d̃)`.  Then
289 `P ∈ Ann_{k[x,ξ]} gr Q` (eq. (3.5)). -/
290 theorem proposition_3_6 {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
291     (hform : IsMonicNormalForm k N dt) (hdeg : bernsteinDegree k dt = N) :
292     orderSymbol k N dt ∈ Module.annihilator (SymbolRing k (n + 1)) (grQ k N dt) := by
293   obtain ⟨⟨hF, -⟩, -⟩ := lemma_3_5 k hform hdeg
294   rw [Module.mem_annihilator]
295   intro m
296   obtain ⟨f, rfl⟩ := (step_3_6_generated k dt).2.2 m
297   rw [smul_comm, step_3_6_kills k hF, smul_zero]
