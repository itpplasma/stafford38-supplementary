8 # Lemma 3.1: symplectic completion
9 
10 Symplectic bases and the standard form `ω` are those of Definition 2.8
11 (`Stafford38.Paper.Section2.IsSymplecticBasis`, `Section2.omega`).  Lemma 3.1 completes a nonzero vector
12 `v` to a symplectic basis and produces an `ω`-preserving automorphism `g` with
13 `g(e) = v`.  Its proof is split into the five paragraphs of the paper.
14 
15 Indexing: a basis of `𝕍` is indexed by `Fin n ⊕ Fin n`; `Sum.inl i` is `e_{i+1}`
16 and `Sum.inr i` is `f_{i+1}`.  On `𝕍 = PhaseVar n → k`, `Sum.inl i` is the
17 coefficient of `x_{i+1}` and `Sum.inr i` that of `∂_{i+1}`.
18 -/

37 /-- **Lemma 3.1, paragraph 1.** As `ω` is nondegenerate and `v ≠ 0`, some `w'`
38 has `ω(v, w') = c ≠ 0`; `w := w'/c` has `ω(v, w) = 1`. -/
39 theorem step_lemma_3_1_partner (ω : BilinForm k V) (hnd : ω.Nondegenerate)
40     {v : V} (hv : v ≠ 0) : ∃ w : V, ω v w = 1 := by
41   have hpair : ∃ w', ω v w' ≠ 0 := by
42     by_contra h
43     exact hv (hnd.1 v fun y => by_contra fun hy => h ⟨y, hy⟩)
44   obtain ⟨w', hw'⟩ := hpair
45   refine ⟨(ω v w')⁻¹ • w', ?_⟩
46   rw [BilinForm.smul_right]
47   exact inv_mul_cancel₀ hw'

49 /-- **Lemma 3.1, paragraph 2.** On `U = span(v, w)`: for `av + bw ≠ 0`, if
50 `a ≠ 0` then `ω(av + bw, w) = a ≠ 0`, and if `b ≠ 0` then
51 `ω(av + bw, v) = -b ≠ 0`; so `ω|_U` is nondegenerate. -/
52 theorem step_lemma_3_1_plane (ω : BilinForm k V) (hAlt : ∀ x, ω x x = 0)
53     {v w : V} (hvw : ω v w = 1) (a b : k) :
54     ω (a • v + b • w) w = a ∧ ω (a • v + b • w) v = -b := by
55   have hwv : ω w v = -1 := by
56     have h := BilinForm.IsAlt.neg_eq (show ω.IsAlt from hAlt) v w
57     simpa [hvw] using h.symm
58   constructor
59   · rw [BilinForm.add_left, BilinForm.smul_left, BilinForm.smul_left, hvw, hAlt w]
60     ring
61   · rw [BilinForm.add_left, BilinForm.smul_left, BilinForm.smul_left, hAlt v, hwv]
62     ring

64 /-- **Lemma 3.1, paragraph 3.** `U ∩ U^⊥ = 0` because `ω|_U` is nondegenerate
65 (paragraph 2); with `dim U + dim U^⊥ = 2n` this gives `𝕍 = U ⊕ U^⊥`, and
66 `ω|_{U^⊥}` is nondegenerate.  The intersection is computed from paragraph 2;
67 the dimension count and the nondegeneracy of `ω|_{U^⊥}` are
68 `SymplecticComplement.pairComplementData`. -/
69 theorem step_lemma_3_1_complement [Module.Finite k V] (ω : BilinForm k V)
70     (hAlt : ∀ x, ω x x = 0) (hnd : ω.Nondegenerate) {v w : V} (hvw : ω v w = 1) :
71     let U : Submodule k V := Submodule.span k {v, w}
72     Disjoint U (ω.orthogonal U) ∧ IsCompl U (ω.orthogonal U) ∧
73       (ω.restrict (ω.orthogonal U)).Nondegenerate ∧
74       Module.finrank k (ω.orthogonal U) + 2 = Module.finrank k V := by
75   intro U
76   have hdata :=
77     Stafford38.LinearAlgebra.SymplecticComplement.pairComplementData ω hAlt hnd v w hvw
78   refine ⟨?_, hdata.2.1, hdata.2.2.1, hdata.2.2.2⟩
79   rw [Submodule.disjoint_def]
80   intro x hxU hxO
81   obtain ⟨a, b, rfl⟩ := Submodule.mem_span_pair.1 hxU
82   have hvU : v ∈ U := Submodule.subset_span (by simp)
83   have hwU : w ∈ U := Submodule.subset_span (by simp)
84   have hskew (x y : V) : ω x y = -ω y x :=
85     ((show ω.IsAlt from hAlt).neg_eq y x).symm
86   have hw0 : ω (a • v + b • w) w = 0 := by rw [hskew]; simp [hxO w hwU]
87   have hv0 : ω (a • v + b • w) v = 0 := by rw [hskew]; simp [hxO v hvU]
88   obtain ⟨ha, hb⟩ := step_lemma_3_1_plane ω hAlt hvw a b
89   have ha0 : a = 0 := by rw [← ha, hw0]
90   have hb0 : b = 0 := by rw [← neg_neg b, ← hb, hv0, neg_zero]
91   simp [ha0, hb0]

95 private def onePlusEquiv (n : ℕ) : Fin 1 ⊕ Fin n ≃ Fin (n + 1) where
96   toFun
97     | .inl _ => 0
98     | .inr i => i.succ
99   invFun := Fin.cases (Sum.inl 0) (fun i => Sum.inr i)
100   left_inv x := by
101     cases x with
102     | inl i => have : i = 0 := Subsingleton.elim _ _; subst i; rfl
103     | inr i => rfl
104   right_inv i := by
105     refine Fin.cases ?_ ?_ i <;> simp

107 private def rearrangePairIndex (n : ℕ) :
108     (Fin 1 ⊕ Fin 1) ⊕ (Fin n ⊕ Fin n) ≃
109       (Fin 1 ⊕ Fin n) ⊕ (Fin 1 ⊕ Fin n) where
110   toFun
111     | .inl (.inl i) => .inl (.inl i)
112     | .inl (.inr i) => .inr (.inl i)
113     | .inr (.inl i) => .inl (.inr i)
114     | .inr (.inr i) => .inr (.inr i)
115   invFun
116     | .inl (.inl i) => .inl (.inl i)
117     | .inl (.inr i) => .inr (.inl i)
118     | .inr (.inl i) => .inl (.inr i)
119     | .inr (.inr i) => .inr (.inr i)
120   left_inv x := by cases x with
121     | inl x => cases x <;> rfl
122     | inr x => cases x <;> rfl
123   right_inv x := by cases x with
124     | inl x => cases x <;> rfl
125     | inr x => cases x <;> rfl

127 private def symplecticPairIndex (n : ℕ) :
128     Fin 2 ⊕ (Fin n ⊕ Fin n) ≃ Fin (n + 1) ⊕ Fin (n + 1) :=
129   (Equiv.sumCongr (finSumFinEquiv (m := 1) (n := 1)).symm (Equiv.refl _)).trans
130     ((rearrangePairIndex n).trans
131       (Equiv.sumCongr (onePlusEquiv n) (onePlusEquiv n)))

153 /-- **Lemma 3.1, paragraph 4.** By induction on `n`, `U^⊥` has a symplectic
154 basis `e_2, …, f_n`; with `e_1 := v`, `f_1 := w` this is a symplectic basis of
155 `𝕍 = U ⊕ U^⊥`, since cross pairings between `U` and `U^⊥` vanish.  The
156 induction is `CharacteristicPaperSymplecticBasis.exists_symplectic_basis`
157 applied to `U^⊥` with the data of paragraph 3. -/
158 theorem step_lemma_3_1_basis [Module.Finite k V] (n : ℕ) (ω : BilinForm k V)
159     (hAlt : ∀ x, ω x x = 0) {v w : V} (hvw : ω v w = 1)
160     (hcompl : IsCompl (Submodule.span k {v, w}) (ω.orthogonal (Submodule.span k {v, w})))
161     (hWnd : (ω.restrict (ω.orthogonal (Submodule.span k {v, w}))).Nondegenerate)
162     (hWdim : Module.finrank k (ω.orthogonal (Submodule.span k {v, w})) = 2 * n) :
163     ∃ b : Basis (Fin (n + 1) ⊕ Fin (n + 1)) k V,
164       IsSymplecticBasis ω b ∧ b (Sum.inl 0) = v ∧ b (Sum.inr 0) = w := by
165   let U : Submodule k V := Submodule.span k {v, w}
166   let W : Submodule k V := ω.orthogonal U
167   have hAltW : ∀ x : W, (ω.restrict W) x x = 0 := fun x => hAlt x
168   obtain ⟨bW, hbWcross, hbWleft, hbWright⟩ :=
169     Stafford38.CharacteristicPaperSymplecticBasis.exists_symplectic_basis n
170       (ω.restrict W) hAltW hWnd hWdim
171   have hUmemV : v ∈ U := Submodule.subset_span (by simp)
172   have hUmemW : w ∈ U := Submodule.subset_span (by simp)
173   have hli : LinearIndependent k (pairVectors v w) := by
174     change LinearIndependent k ![v, w]
175     rw [LinearIndependent.pair_iff]
176     intro a b hab
177     obtain ⟨ha, hb⟩ := step_lemma_3_1_plane ω hAlt hvw a b
178     rw [hab, BilinForm.zero_left] at ha hb
179     exact ⟨ha.symm, by simpa using hb.symm⟩
180   have hspan : Submodule.span k (Set.range (pairVectors v w)) = U := by
181     congr 1
182     ext x
183     simp only [Set.mem_range, Set.mem_insert_iff, Set.mem_singleton_iff]
184     constructor
185     · rintro ⟨i, rfl⟩
186       fin_cases i <;> simp [pairVectors]
187     · rintro (rfl | rfl)
188       · exact ⟨0, by simp [pairVectors]⟩
189       · exact ⟨1, by simp [pairVectors]⟩
190   let bU : Basis (Fin 2) k U := (Basis.span hli).map (LinearEquiv.ofEq _ _ hspan)
191   have hbU0 : bU 0 = ⟨v, hUmemV⟩ := by
192     apply Subtype.ext
193     dsimp [bU]
194     rw [Module.Basis.coe_span_apply]
195     simp [pairVectors]
196   have hbU1 : bU 1 = ⟨w, hUmemW⟩ := by
197     apply Subtype.ext
198     dsimp [bU]
199     rw [Module.Basis.coe_span_apply]
200     simp [pairVectors]
201   let b : Basis (Fin (n + 1) ⊕ Fin (n + 1)) k V :=
202     ((bU.prod bW).map (Submodule.prodEquivOfIsCompl U W hcompl)).reindex
203       (symplecticPairIndex n)
204   have hbE0 : b (Sum.inl (0 : Fin (n + 1))) = v := by
205     simp only [b, Module.Basis.reindex_apply, pairIndex_left0, Module.Basis.map_apply]
206     simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl', hbU0]
207   have hbF0 : b (Sum.inr (0 : Fin (n + 1))) = w := by
208     simp only [b, Module.Basis.reindex_apply, pairIndex_right0, Module.Basis.map_apply]
209     simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl', hbU1]
210   have hbEs (i : Fin n) : b (Sum.inl i.succ) = (bW (Sum.inl i) : V) := by
211     simp only [b, Module.Basis.reindex_apply, pairIndex_left_succ, Module.Basis.map_apply]
212     simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl']
213   have hbFs (i : Fin n) : b (Sum.inr i.succ) = (bW (Sum.inr i) : V) := by
214     simp only [b, Module.Basis.reindex_apply, pairIndex_right_succ, Module.Basis.map_apply]
215     simp [Module.Basis.prod_apply, Submodule.coe_prodEquivOfIsCompl']
216   have hskew (x y : V) : ω x y = -ω y x :=
217     ((show ω.IsAlt from hAlt).neg_eq y x).symm
218   refine ⟨b, ⟨?_, ?_, ?_⟩, hbE0, hbF0⟩
219   · intro i j
220     refine Fin.cases ?_ (fun i => ?_) i
221     · refine Fin.cases ?_ (fun j => ?_) j
222       · simpa [hbE0, hbF0] using hvw
223       · simp only [eq_comm, Fin.succ_ne_zero, ite_false]
224         rw [hbE0, hbFs]
225         exact ((bW (Sum.inr j)).property v hUmemV).symm
226     · refine Fin.cases ?_ (fun j => ?_) j
227       · simp only [Fin.succ_ne_zero, ite_false]
228         rw [hbEs, hbF0, hskew]
229         simp [(bW (Sum.inl i)).property w hUmemW]
230       · simpa [hbEs, hbFs] using hbWcross i j
231   · intro i j
232     refine Fin.cases ?_ (fun i => ?_) i
233     · refine Fin.cases ?_ (fun j => ?_) j
234       · rw [hbE0]; exact hAlt v
235       · rw [hbE0, hbEs]; exact (bW (Sum.inl j)).property v hUmemV
236     · refine Fin.cases ?_ (fun j => ?_) j
237       · rw [hbEs, hbE0, hskew]
238         simp [(bW (Sum.inl i)).property v hUmemV]
239       · simpa [hbEs] using hbWleft i j
240   · intro i j
241     refine Fin.cases ?_ (fun i => ?_) i
242     · refine Fin.cases ?_ (fun j => ?_) j
243       · rw [hbF0]; exact hAlt w
244       · rw [hbF0, hbFs]; simp [(bW (Sum.inr j)).property w hUmemW]
245     · refine Fin.cases ?_ (fun j => ?_) j
246       · rw [hbFs, hbF0, hskew]
247         simp [(bW (Sum.inr i)).property w hUmemW]
248       · simpa [hbFs] using hbWright i j

250 /-- **Lemma 3.1, paragraph 5.** The linear map sending one symplectic basis to
251 another, vector by vector, preserves all pairwise values of `ω` on basis
252 vectors, hence by bilinearity preserves `ω`. -/
253 theorem step_lemma_3_1_transport {m : ℕ} (ω : BilinForm k V) (hAlt : ∀ x, ω x x = 0)
254     (b₁ b₂ : Basis (Fin m ⊕ Fin m) k V) (h₁ : IsSymplecticBasis ω b₁)
255     (h₂ : IsSymplecticBasis ω b₂) :
256     (∀ x y, ω (b₁.equiv b₂ (Equiv.refl _) x) (b₁.equiv b₂ (Equiv.refl _) y) = ω x y) ∧
257       ∀ i, b₁.equiv b₂ (Equiv.refl _) (b₁ i) = b₂ i := by
258   have hg := Stafford38.CharacteristicPaperSymplecticBasis.symplectic_basis_equiv m ω ω
259     b₁ b₂ hAlt hAlt h₁.1 h₁.2.1 h₁.2.2 h₂.1 h₂.2.1 h₂.2.2
260   refine ⟨fun x y => ?_, fun i => Basis.equiv_apply b₁ i b₂ (Equiv.refl _)⟩
261   have h := congrArg (fun f : BilinForm k V => f x y) hg
262   simpa [LinearMap.BilinForm.comp_apply] using h

264 /-- **Lemma 3.1.** Let `(𝕍, ω)` be a symplectic `k`-space of dimension `2n`
265 (`ω` antisymmetric and nondegenerate) and `0 ≠ v ∈ 𝕍`.  Then `v` is the first
266 vector `e_1` of a symplectic basis of `𝕍`; consequently, for every nonzero
267 `e ∈ 𝕍` there is a `k`-linear automorphism `g` preserving `ω` with
268 `g(e) = v`. -/
269 theorem lemma_3_1 [CharZero k] [Module.Finite k V] (n : ℕ) (ω : BilinForm k V)
270     (hanti : ∀ x y, ω x y = -ω y x) (hnd : ω.Nondegenerate)
271     (hdim : Module.finrank k V = 2 * n) {v : V} (hv : v ≠ 0) :
272     (∃ (hn : 0 < n) (b : Basis (Fin n ⊕ Fin n) k V),
273         IsSymplecticBasis ω b ∧ b (Sum.inl ⟨0, hn⟩) = v) ∧
274       ∀ e : V, e ≠ 0 →
275         ∃ g : V ≃ₗ[k] V, (∀ x y, ω (g x) (g y) = ω x y) ∧ g e = v := by
276   have hAlt := alternating_of_antisymmetric ω hanti
277   -- every nonzero vector is the first vector of a symplectic basis
278   have hcomplete : ∀ u : V, u ≠ 0 → ∃ m : ℕ, n = m + 1 ∧
279       ∃ b : Basis (Fin (m + 1) ⊕ Fin (m + 1)) k V,
280         IsSymplecticBasis ω b ∧ b (Sum.inl 0) = u := by
281     intro u hu
282     obtain ⟨w, huw⟩ := step_lemma_3_1_partner ω hnd hu
283     obtain ⟨-, hcompl, hWnd, hWdim⟩ := step_lemma_3_1_complement ω hAlt hnd huw
284     obtain ⟨m, rfl⟩ : ∃ m, n = m + 1 := ⟨n - 1, by omega⟩
285     refine ⟨m, rfl, ?_⟩
286     obtain ⟨b, hb, hb0, -⟩ :=
287       step_lemma_3_1_basis m ω hAlt huw hcompl hWnd (by omega)
288     exact ⟨b, hb, hb0⟩
289   obtain ⟨m, rfl, b, hb, hb0⟩ := hcomplete v hv
290   refine ⟨⟨Nat.succ_pos m, b, hb, hb0⟩, fun e he => ?_⟩
291   obtain ⟨m', hm', b', hb', hb'0⟩ := hcomplete e he
292   obtain rfl : m' = m := by omega
293   obtain ⟨hpres, happ⟩ := step_lemma_3_1_transport ω hAlt b' b hb' hb
294   exact ⟨b'.equiv b (Equiv.refl _), hpres, by rw [← hb'0, happ, hb0]⟩
