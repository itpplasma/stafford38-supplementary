10 # The monic normal form (3.1) and symbols of `p^j a` with `a ∈ C`
11 
12 With the notation of Convention 2.2 (`x₁`, `p`, `C`; `Stafford38.Paper.Section2`)
13 and the Bernstein degree of Definition 2.3 (`Section2.bernsteinDegree`),
14 `IsMonicNormalForm N d̃` is the displayed form (3.1):
15 `d̃ = p^N + ∑_{j<N} p^j a_j` with `a_j ∈ C` and `deg_B a_j ≤ N - j`.
16 
17 The lemmas below compute the symbols of the terms `p^j a_j` for both
18 filtrations; they are the PBW-monomial bookkeeping of the proofs of
19 Proposition 3.2 (paragraph 5) and Lemma 3.5 (second paragraph).
20 -/

42 /-- **Equation (3.1).** `d̃ = p^N + ∑_{j<N} p^j a_j` with `a_j ∈ C` and
43 `deg_B a_j ≤ N - j`. -/
44 def IsMonicNormalForm {n : ℕ} (N : ℕ) (dt : PresentedWeyl k (n + 1)) : Prop :=
45   ∃ a : ℕ → PresentedWeyl k (n + 1),
46     (∀ j < N, a j ∈ C k n ∧ bernsteinDegree k (a j) ≤ N - j) ∧
47       dt = p k n ^ N + ∑ j ∈ Finset.range N, p k n ^ j * a j
48 
49 /-- A PBW monomial of an element of `C` contains no `∂₁`: `C = A_{n-1}(k)[x₁]`. -/
50 theorem normalForm_C_no_p {n : ℕ} {a : PresentedWeyl k (n + 1)} (ha : a ∈ C k n)
51     (m : PhaseVar (n + 1) →₀ ℕ)
52     (hm : MvPolynomial.coeff m (presentedNormalFormLinearEquiv k (n + 1) a) ≠ 0) :
53     m (.inr 0) = 0 := by
54   rw [C_eq_range] at ha
55   obtain ⟨z, rfl⟩ := ha
56   by_contra hL
57   apply hm
58   let aq : Fin (n + 1) → ℕ := fun i => m (.inl i)
59   let pq : Fin (n + 1) → ℕ := fun i => m (.inr i)
60   have hsplit : m = pairExponent n (aq 0) (pq 0)
61       (phaseExponent (fun i => aq i.succ) (fun i => pq i.succ)) := by
62     have hphase : phaseExponent aq pq = m := by
63       ext i
64       cases i <;> simp [phaseExponent, aq, pq]
65     rw [← hphase, phaseExponent_succ]
66     rfl
67   change AddMonoidAlgebra.coeff (presentedNormalFormLinearEquiv k (n + 1)
68     (coordinateStageToPresented k n z)) m = 0
69   rw [coordinateStageToPresented_normalForm, hsplit, coeff_flattenPairSymbols,
70     Polynomial.coeff_C, ite_eq_right (show pq 0 ≠ 0 from hL)]
71   simp

73 /-- The normal form of `p^j` is `ξ₁^j`. -/
74 theorem normalForm_p_pow (n j : ℕ) :
75     presentedNormalFormLinearEquiv k (n + 1) (p k n ^ j) =
76       MvPolynomial.X (.inr (0 : Fin (n + 1))) ^ j := by
77   have h := presentedNormalFormLinearEquiv_previous_ordered k n 0 j 1
78   have hz : presentedCoefficientOrdered k n 0 j 1 = p k n ^ j := by
79     simp [presentedCoefficientOrdered]
80   rw [hz] at h
81   rw [h, presentedNormalFormLinearEquiv_one]
82   simp

84 private theorem weight_single_p (w : PhaseVar (n + 1) → ℕ) (hw : w (.inr 0) = 1) (j : ℕ) :
85     monomialWeight w (Finsupp.single (.inr (0 : Fin (n + 1))) j) = j := by
86   simp [monomialWeight, hw]

88 /-- `p^j` lies in the `j`-th piece and has symbol `ξ₁^j`, for any weight giving
89 `∂₁` degree `1` (Bernstein and order). -/
90 theorem p_pow_piece_principal {n : ℕ} (w : PhaseVar (n + 1) → ℕ) (hw : w (.inr 0) = 1)
91     (j : ℕ) :
92     p k n ^ j ∈ presentedWeightPiece k w j ∧
93       presentedPrincipalComponent k w j (p k n ^ j) =
94         MvPolynomial.X (.inr (0 : Fin (n + 1))) ^ j := by
95   constructor
96   · rw [mem_presentedWeightPiece, normalForm_p_pow]
97     intro m hm
98     simp only [MvPolynomial.coeff] at hm
99     rw [MvPolynomial.X_pow_eq_monomial, MvPolynomial.coeff_monomial] at hm
100     split_ifs at hm with hmj
101     · subst m
102       rw [weight_single_p w hw]
103     · exact absurd rfl hm
104   · rw [presentedPrincipalComponent, LinearMap.comp_apply, LinearEquiv.coe_toLinearMap,
105       normalForm_p_pow, MvPolynomial.X_pow_eq_monomial,
106       weightedHomogeneousComponent_monomial, weight_single_p w hw]
107     simp

109 /-- The coefficient of `ξ₁^L` in a degree-`L` symbol is the PBW coefficient of
110 `∂₁^L` (for any weight giving `∂₁` degree `1`). -/
111 theorem coeff_principal_p_pow {n L : ℕ} (w : PhaseVar (n + 1) → ℕ) (hw : w (.inr 0) = 1)
112     (a : PresentedWeyl k (n + 1)) :
113     MvPolynomial.coeff (Finsupp.single (.inr (0 : Fin (n + 1))) L)
114         (presentedPrincipalComponent k w L a) =
115       MvPolynomial.coeff (Finsupp.single (.inr (0 : Fin (n + 1))) L)
116         (presentedNormalFormLinearEquiv k (n + 1) a) := by
117   change AddMonoidAlgebra.coeff (presentedPrincipalComponent k w L a) _ = _
118   rw [coeff_presentedPrincipalComponent, ite_eq_left (weight_single_p w hw L)]

120 /-- The coefficient of `ξ₁^N` in `ξ₁^j f` is the coefficient of `ξ₁^{N-j}` in
121 `f`, for `j ≤ N`. -/
122 theorem coeff_X_pow_mul_single {n j N : ℕ} (hj : j ≤ N) (f : SymbolRing k (n + 1)) :
123     MvPolynomial.coeff (Finsupp.single (.inr (0 : Fin (n + 1))) N)
124         (MvPolynomial.X (.inr 0) ^ j * f) =
125       MvPolynomial.coeff (Finsupp.single (.inr (0 : Fin (n + 1))) (N - j)) f := by
126   simp only [MvPolynomial.coeff]
127   rw [MvPolynomial.X_pow_eq_monomial, MvPolynomial.coeff_monomial_mul',
128     ite_eq_left (Finsupp.single_le_single.mpr hj), one_mul, ← Finsupp.single_tsub]

130 /-- **Bernstein symbol of `p^j a`.** For `a ∈ B_{N-j}`,
131 `σ_N^B(p^j a) = ξ₁^j σ_{N-j}^B(a)`. -/
132 theorem principal_bernstein_p_pow_mul {n j N : ℕ} (hj : j ≤ N) {a : PresentedWeyl k (n + 1)}
133     (ha : a ∈ bernsteinPiece k (n + 1) (N - j)) :
134     p k n ^ j * a ∈ bernsteinPiece k (n + 1) N ∧
135       presentedPrincipalComponent k (@bernsteinWeight (n + 1)) N (p k n ^ j * a) =
136         MvPolynomial.X (.inr 0) ^ j *
137           presentedPrincipalComponent k (@bernsteinWeight (n + 1)) (N - j) a := by
138   obtain ⟨hp, hσ⟩ := p_pow_piece_principal k (@bernsteinWeight (n + 1)) rfl j
139   have hmem := mul_mem_bernsteinPiece k hp ha
140   have hmul := presentedPrincipalComponent_mul_bernstein k hp ha
141   rw [Nat.add_sub_cancel' hj] at hmem hmul
142   rw [hmul, hσ]
143   exact ⟨hmem, rfl⟩
144 
145 /-- **Order symbol of `p^j a`.** For `a ∈ F_{N-j}`,
146 `σ_N^ord(p^j a) = ξ₁^j σ_{N-j}^ord(a)`. -/
147 theorem principal_order_p_pow_mul {n j N : ℕ} (hj : j ≤ N) {a : PresentedWeyl k (n + 1)}
148     (ha : a ∈ orderPiece k (n + 1) (N - j)) :
149     presentedPrincipalComponent k (@orderWeight (n + 1)) N (p k n ^ j * a) =
150       MvPolynomial.X (.inr 0) ^ j *
151         presentedPrincipalComponent k (@orderWeight (n + 1)) (N - j) a := by
152   obtain ⟨hp, hσ⟩ := p_pow_piece_principal k (@orderWeight (n + 1)) rfl j
153   have hmul := presentedPrincipalComponent_mul_order k hp ha
154   rw [Nat.add_sub_cancel' hj] at hmul
155   rw [hmul, hσ]

157 /-- An element of the form (3.1) is PBW-monic: it lies in `B_N` and the PBW
158 coefficient of `∂₁^N` is `1`.  This connects (3.1) to the repository's
159 `WeylPBWMonicBridge.IsPBWMonicAt`, the hypothesis of
160 `UniversalAssembly.CanonicalSupportVanishing`. -/
161 theorem isPBWMonicAt_of_isMonicNormalForm {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
162     (h : IsMonicNormalForm k N dt) :
163     IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N dt := by
164   obtain ⟨a, ha, rfl⟩ := h
165   have hterm (j : ℕ) (hj : j ∈ Finset.range N) :=
166     principal_bernstein_p_pow_mul k (Finset.mem_range.mp hj).le
167       ((bernsteinDegree_le_iff k (a j)).mp (ha j (Finset.mem_range.mp hj)).2)
168   obtain ⟨hpN, hσN⟩ := p_pow_piece_principal k (@bernsteinWeight (n + 1)) rfl N
169   refine ⟨add_mem hpN (Submodule.sum_mem _ fun j hj => (hterm j hj).1), ?_⟩
170   rw [← coeff_principal_p_pow k (@bernsteinWeight (n + 1)) rfl, map_add, map_sum, hσN]
171   simp only [MvPolynomial.coeff]
172   rw [AddMonoidAlgebra.coeff_add, Finsupp.add_apply, MvPolynomial.coeff_sum,
173     Finset.sum_eq_zero, add_zero, MvPolynomial.X_pow_eq_monomial, MvPolynomial.coeff_monomial,
174     ite_eq_left rfl]
175   intro j hj
176   have hjN := Finset.mem_range.mp hj
177   rw [(hterm j hj).2]
178   have h := coeff_X_pow_mul_single k hjN.le
179     (presentedPrincipalComponent k (@bernsteinWeight (n + 1)) (N - j) (a j))
180   simp only [MvPolynomial.coeff] at h
181   rw [h]
182   have h2 := coeff_principal_p_pow k (L := N - j) (@bernsteinWeight (n + 1)) rfl (a j)
183   simp only [MvPolynomial.coeff] at h2
184   rw [h2]
185   by_contra hne
186   have := normalForm_C_no_p k (ha j hjN).1 _ hne
187   simp at this
188   omega

190 /-- In (3.1), `N = deg_B d̃`: `d̃ ∈ B_N` and its Bernstein symbol has the
191 coefficient `1` at `ξ₁^N`. -/
192 theorem bernsteinDegree_eq_of_isMonicNormalForm {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
193     (h : IsMonicNormalForm k N dt) : bernsteinDegree k dt = N := by
194   have hd := isPBWMonicAt_of_isMonicNormalForm k h
195   refine bernsteinDegree_eq_of_principal_ne_zero k hd.1 fun h0 => ?_
196   have hc : AddMonoidAlgebra.coeff
197       (presentedPrincipalComponent k (@bernsteinWeight (n + 1)) N dt)
198       (Finsupp.single (.inr (0 : Fin (n + 1))) N) = 1 := by
199     rw [coeff_presentedPrincipalComponent, ite_eq_left (by simp [monomialWeight, bernsteinWeight])]
200     exact hd.2
201   rw [h0] at hc
202   exact zero_ne_one hc
