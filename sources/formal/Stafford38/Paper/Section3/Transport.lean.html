10 # Remark 3.3 and the ideal `𝓘` and module `Q` of (3.2)
11 
12 Remark 3.3 transports a solution `1 = d̃R + x₁^N d̃S` for the normal form
13 `d̃ = c⁻¹Φ_g(d)` back to `d`.  Equation (3.2) defines
14 `𝓘 = d̃A + x₁^N d̃A` and `Q = A/𝓘`; `Q = 0` is exactly such a solution.
15 `section_3_reduction` packages the opening paragraph, Proposition 3.2 and
16 Remark 3.3 for the final assembly.
17 -/

34 /-- **Equation (3.2), the ideal.** `𝓘 = d̃A_n + x₁^N d̃A_n`, a right ideal. -/
35 abbrev idealI {n : ℕ} (N : ℕ) (dt : PresentedWeyl k (n + 1)) :
36     RightIdeal (PresentedWeyl k (n + 1)) :=
37   canonicalRightIdeal (x₁ k n) dt N
38 
39 /-- **Equation (3.2), the module.** `Q = A_n/𝓘`, a right `A_n`-module. -/
40 abbrev Q {n : ℕ} (N : ℕ) (dt : PresentedWeyl k (n + 1)) :=
41   RightQuotient (idealI k N dt)
42 
43 /-- `Q = 0` gives `1 ∈ 𝓘`, i.e. `1 = d̃R + x₁^N d̃S` for some `R, S`. -/
44 theorem exists_eq_of_Q_eq_zero {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
45     (hQ : Subsingleton (Q k N dt)) :
46     ∃ R S : PresentedWeyl k (n + 1), (1 : PresentedWeyl k (n + 1)) =
47       dt * R + x₁ k n ^ N * dt * S := by
48   have hmem : (1 : PresentedWeyl k (n + 1)) ∈ idealI k N dt := by
49     apply (Submodule.Quotient.mk_eq_zero (idealI k N dt)).mp
50     exact Subsingleton.elim _ _
51   obtain ⟨a, b, hab⟩ := Submodule.mem_span_pair.mp hmem
52   refine ⟨a.unop, b.unop, ?_⟩
53   rw [← hab]
54   rfl

56 /-- `Φ_g^{-1}(x₁) = Φ_{g^{-1}}(x₁) = ∑_j (g^{-1})_{1j} u_j` is a linear
57 symplectic coordinate. -/
58 theorem phiG_symm_x₁_isLinearWeylCoordinate {n : ℕ}
59     (g : Matrix.symplecticGroup (Fin (n + 1)) k) :
60     Stafford38FixedSourceChallenge.IsLinearWeylCoordinate k n ((phiG k g).symm (x₁ k n)) := by
61   obtain ⟨-, -, hsymm⟩ := step_phi_g k g
62   obtain ⟨hgen, -, -⟩ := step_phi_g k g⁻¹
63   refine ⟨g.1, (g⁻¹).1, SymplecticGroup.mem_iff.mp g.2, SymplecticGroup.mem_iff.mp (g⁻¹).2,
64     ?_, ?_, ?_⟩
65   · calc g.1 * (g⁻¹).1 = ((g * g⁻¹ : Matrix.symplecticGroup (Fin (n + 1)) k) :
66           Matrix _ _ k) := rfl
67       _ = 1 := by rw [mul_inv_cancel]; rfl
68   · calc (g⁻¹).1 * g.1 = ((g⁻¹ * g : Matrix.symplecticGroup (Fin (n + 1)) k) :
69           Matrix _ _ k) := rfl
70       _ = 1 := by rw [inv_mul_cancel]; rfl
71   · rw [hsymm]
72     exact hgen (.inl 0)

74 /-- **Remark 3.3.** Since `Φ_g` is a ring automorphism and `c` a central unit,
75 `1 = d̃R + x₁^N d̃S` for `d̃ = c⁻¹Φ_g(d)` transports back to
76 `1 = d (c⁻¹Φ_g^{-1}(R)) + F d (c⁻¹Φ_g^{-1}(S))` with `F = Φ_g^{-1}(x₁)^N`. -/
77 theorem remark_3_3 {n N : ℕ} (g : Matrix.symplecticGroup (Fin (n + 1)) k) (c : k)
78     (d R S : PresentedWeyl k (n + 1))
79     (h : (1 : PresentedWeyl k (n + 1)) =
80       (c⁻¹ • phiG k g d) * R + x₁ k n ^ N * (c⁻¹ • phiG k g d) * S) :
81     (1 : PresentedWeyl k (n + 1)) =
82       d * (c⁻¹ • (phiG k g).symm R) +
83         (phiG k g).symm (x₁ k n) ^ N * d * (c⁻¹ • (phiG k g).symm S) := by
84   have ht := congrArg (phiG k g).symm h
85   rw [map_one] at ht
86   rw [ht, map_add, map_mul, map_mul, map_mul, map_pow, map_smul, AlgEquiv.symm_apply_apply,
87     smul_mul_assoc, mul_smul_comm, mul_smul_comm, smul_mul_assoc, mul_smul_comm]
88 
89 /-- **Remark 3.3 with (3.2).** If `Q = A/𝓘 = 0` for `d̃ = c⁻¹Φ_g(d)`, then
90 `1 = dR' + F d S'` with `F = ℓ^N` for the linear symplectic coordinate
91 `ℓ = Φ_g^{-1}(x₁)`. -/
92 theorem remark_3_3_of_Q_eq_zero {n N : ℕ} (g : Matrix.symplecticGroup (Fin (n + 1)) k)
93     (c : k) (d : PresentedWeyl k (n + 1)) (hQ : Subsingleton (Q k N (c⁻¹ • phiG k g d))) :
94     ∃ ell R S : PresentedWeyl k (n + 1),
95       Stafford38FixedSourceChallenge.IsLinearWeylCoordinate k n ell ∧
96         (1 : PresentedWeyl k (n + 1)) = d * R + ell ^ N * d * S := by
97   obtain ⟨R, S, hRS⟩ := exists_eq_of_Q_eq_zero k hQ
98   exact ⟨_, _, _, phiG_symm_x₁_isLinearWeylCoordinate k g, remark_3_3 k g c d R S hRS⟩

100 /-- **Section 3 for the final assembly.**  For `0 ≠ d ∈ A_n(k)`, either
101 `deg_B d = 0` and `d` is a unit with `1 = dR + FdS` (opening paragraph), or
102 `N = deg_B d ≥ 1` and Proposition 3.2 gives the normal form `d̃ = c⁻¹Φ_g(d)`
103 of (3.1) with `deg_B d̃ = N`, for which `Q = A/𝓘 = 0` yields
104 `1 = dR + ℓ^N dS` with a linear symplectic coordinate `ℓ` (Remark 3.3). -/
105 theorem section_3_reduction [CharZero k] {n : ℕ} {d : PresentedWeyl k (n + 1)} (hd : d ≠ 0) :
106     (bernsteinDegree k d = 0 ∧ Stafford.Reduction.Stafford38 d) ∨
107       (1 ≤ bernsteinDegree k d ∧
108         ∃ (g : Matrix.symplecticGroup (Fin (n + 1)) k) (c : k), c ≠ 0 ∧
109           IsMonicNormalForm k (bernsteinDegree k d) (c⁻¹ • phiG k g d) ∧
110           bernsteinDegree k (c⁻¹ • phiG k g d) = bernsteinDegree k d ∧
111           (Subsingleton (Q k (bernsteinDegree k d) (c⁻¹ • phiG k g d)) →
112             ∃ ell R S : PresentedWeyl k (n + 1),
113               Stafford38FixedSourceChallenge.IsLinearWeylCoordinate k n ell ∧
114                 (1 : PresentedWeyl k (n + 1)) = d * R + ell ^ bernsteinDegree k d * d * S)) := by
115   by_cases hN : bernsteinDegree k d = 0
116   · obtain ⟨c, -, -, hcert⟩ := step_scalar_case k hd hN
117     exact Or.inl ⟨hN, 1, _, 0, hcert⟩
118   · obtain ⟨g, c, hc, hform, hdeg⟩ := proposition_3_2 k hd (Nat.one_le_iff_ne_zero.mpr hN)
119     exact Or.inr ⟨Nat.one_le_iff_ne_zero.mpr hN, g, c, hc, hform, hdeg,
120       remark_3_3_of_Q_eq_zero k g c d⟩
