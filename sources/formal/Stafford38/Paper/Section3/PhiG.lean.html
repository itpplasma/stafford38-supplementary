10 # The standard symplectic space `𝕍` and the automorphism `Φ_g`
11 
12 `𝕍 = k^{2n}` is identified with the Bernstein-degree-one part of `A_n(k)`:
13 the coordinate `Sum.inl i` of a vector is the coefficient of `x_{i+1}`, the
14 coordinate `Sum.inr i` that of `∂_{i+1}`.  It carries the standard symplectic
15 form `ω` of Definition 2.8 (`Stafford38.Paper.Section2.omega`).
16 
17 For `g` preserving `ω`, the paper's `Φ_g` is the algebra automorphism of
18 `A_n(k)` extending `u ↦ g(u)` on generators.  We record `g` by the matrix
19 `M` whose rows are the images of the generators, `g(u_i) = ∑_j M_ij u_j`
20 (as in the proof of Proposition 3.2); `ω`-preservation of `g` is
21 `M ∈ Sp(2n, k)`.
22 -/

41 /-- An `ω`-preserving linear map has symplectic matrix: from
42 `hᵀ J h = J` the inversion argument of the proof of Proposition 3.2 gives
43 `h J hᵀ = J`, i.e. `h ∈ Sp(2n, k)`. -/
44 theorem toMatrix_mem_symplecticGroup {n : ℕ}
45     (h : (PhaseVar n → k) →ₗ[k] PhaseVar n → k)
46     (hpres : ∀ x y, omega k n (h x) (h y) = omega k n x y) :
47     LinearMap.toMatrix' h ∈ Matrix.symplecticGroup (Fin n) k := by
48   have hcomp : (omega k n).comp h h = omega k n := by
49     apply (LinearMap.BilinForm.ext_iff_basis (Pi.basisFun k (PhaseVar n))).2
50     intro i j
51     exact hpres _ _
52   have hmatrix := LinearMap.BilinForm.toMatrix'_comp (omega k n) h h
53   have hgram : LinearMap.BilinForm.toMatrix' (omega k n) = -standardForm k n := by
54     rw [← gramMatrix_omega]
55     ext i j
56     rw [LinearMap.BilinForm.toMatrix'_apply]
57     simp [gramMatrix]
58   rw [hcomp, hgram] at hmatrix
59   have hT : Matrix.transpose (LinearMap.toMatrix' h) * standardForm k n *
60       LinearMap.toMatrix' h = standardForm k n := by
61     have := congrArg Neg.neg hmatrix
62     simpa [Matrix.mul_neg, Matrix.neg_mul] using this.symm
63   exact SymplecticGroup.mem_iff'.2 hT

65 /-- **The automorphism `Φ_g`.**  For `g ∈ Sp(2n, k)`, given by the matrix of
66 its rows `g(u_i) = ∑_j g_ij u_j`, `Φ_g` is the algebra automorphism of
67 `A_n(k)` extending `u_i ↦ g(u_i)`; its inverse is built from `g⁻¹`. -/
68 def phiG {n : ℕ} (g : Matrix.symplecticGroup (Fin n) k) :
69     PresentedWeyl k n ≃ₐ[k] PresentedWeyl k n :=
70   standardSymplecticAlgEquivOfInverse k g.1 (g⁻¹).1
71     (SymplecticGroup.mem_iff.mp g.2) (SymplecticGroup.mem_iff.mp (g⁻¹).2)
72     (by
73       calc g.1 * (g⁻¹).1 = ((g * g⁻¹ : Matrix.symplecticGroup (Fin n) k) : Matrix _ _ k) :=
74             rfl
75         _ = 1 := by rw [mul_inv_cancel]; rfl)
76     (by
77       calc (g⁻¹).1 * g.1 = ((g⁻¹ * g : Matrix.symplecticGroup (Fin n) k) : Matrix _ _ k) :=
78             rfl
79         _ = 1 := by rw [inv_mul_cancel]; rfl)

81 theorem phiG_apply {n : ℕ} (g : Matrix.symplecticGroup (Fin n) k) (z : PresentedWeyl k n) :
82     phiG k g z = standardSymplecticAlgHom k g.1 (SymplecticGroup.mem_iff.mp g.2) z :=
83   rfl

85 theorem phiG_symm_apply {n : ℕ} (g : Matrix.symplecticGroup (Fin n) k)

91 /-- **Construction of `Φ_g` (paragraph after Lemma 3.1).**  Because the
92 `g(u_i)` satisfy the Weyl relations, `u ↦ g(u)` extends to a unique
93 `k`-algebra endomorphism `Φ_g`; it is an automorphism with
94 `Φ_g⁻¹ = Φ_{g⁻¹}`. -/
95 theorem step_phi_g {n : ℕ} (g : Matrix.symplecticGroup (Fin n) k) :
96     (∀ i, phiG k g (freeWeylGenerator (standardForm k n) i) =
97       freeWeylLinearCombination g.1 (freeWeylGenerator (standardForm k n)) i) ∧
98     (∀ f : PresentedWeyl k n →ₐ[k] PresentedWeyl k n,
99       (∀ i, f (freeWeylGenerator (standardForm k n) i) =
100         freeWeylLinearCombination g.1 (freeWeylGenerator (standardForm k n)) i) →
101       f = (phiG k g).toAlgHom) ∧
102     (phiG k g).symm = phiG k g⁻¹ := by
103   refine ⟨fun i => by rw [phiG_apply, standardSymplecticAlgHom_generator], ?_, ?_⟩
104   · intro f hf
105     apply Stafford38.WeylUniversal.freeWeyl_algHom_ext
106     intro i
107     rw [hf i]
108     change _ = phiG k g _
109     rw [phiG_apply, standardSymplecticAlgHom_generator]
110   · apply AlgEquiv.ext
111     intro z
112     rw [phiG_symm_apply, phiG_apply]

114 /-- `Φ_g` maps Bernstein pieces into themselves (proof of Proposition 3.2). -/
115 theorem phiG_mem_bernsteinPiece {n N : ℕ} (g : Matrix.symplecticGroup (Fin n) k)
116     {z : PresentedWeyl k n} (hz : z ∈ Stafford38.WeylFiltration.bernsteinPiece k n N) :
117     phiG k g z ∈ Stafford38.WeylFiltration.bernsteinPiece k n N := by
118   rw [phiG_apply]
119   exact Stafford38.WeylSymbolCompatibility.standardSymplecticAlgHom_preserves_bernsteinPiece
120     k g.1 _ hz

124 theorem phiG_principal {n N : ℕ} (g : Matrix.symplecticGroup (Fin n) k)
