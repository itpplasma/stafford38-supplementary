13 # Theorem 5.8 and its proof (Subsection 5.3), paragraph by paragraph
14 
15 Notation as in the paper.  `X = K^n` with `n = m`, a point `(y, ξ)` of
16 `T*X = K^n × (K^n)*` is `Sum.elim y ξ : PhaseVar m → K`, closed subsets of
17 `T*X` are zero loci `V_K(L)` of ideals of `K[x, ξ] = SymbolRing K m`, and the
18 paper's coordinate `x_1` is index `⟨0, hm⟩`.  In the proof `J = I(W)`,
19 `B = J ∩ K[x]` is `exclusionBaseIdeal W` (the contraction along
20 `baseLift : K[x] → K[x, ξ]`), `Y = V_K(B)`, `df_y = differentialAt y f` and
21 `φ(t) = g(y, t df_y)` is `fibreLinePolynomial y (differentialAt y f) g`.
22 Involutivity of `W` is Definition 2.10 (`{I(W), I(W)} ⊆ I(W)`), the predicate
23 `PostScalarExtensionPoisson.IsInvolutive` of `I(W)`.
24 
25 The steps `step_*` state the paragraphs of the printed proof in order;
26 `theorem_5_8` chains them and closes with `theorem_5_7_i_printed`.
27 -/

54 /-- `d` is `K`-linear: every element of `span_K {df_y : f ∈ I}` is `dh_y` for
55 one `h ∈ I`. -/
56 theorem exists_mem_differentialAt_eq_of_mem_span
57     (I : Ideal (MvPolynomial (Fin m) k)) (y ξ : Fin m → k)
58     (hξ : ξ ∈ Submodule.span k (Set.range fun f : I => differentialAt y f.1)) :
59     ∃ h ∈ I, differentialAt y h = ξ := by
60   induction hξ using Submodule.span_induction with
61   | mem x hx =>
62       obtain ⟨f, rfl⟩ := hx
63       exact ⟨f.1, f.2, rfl⟩
64   | zero =>
65       exact ⟨0, I.zero_mem, by funext i; simp [differentialAt]⟩
66   | add x z _ _ hx hz =>
67       obtain ⟨f, hf, rfl⟩ := hx
68       obtain ⟨g, hg, rfl⟩ := hz
69       exact ⟨f + g, I.add_mem hf hg, by funext i; simp [differentialAt]⟩
70   | smul c x _ hx =>
71       obtain ⟨f, hf, rfl⟩ := hx
72       exact ⟨MvPolynomial.C c * f, I.mul_mem_left _ hf,
73         by funext i; simp [differentialAt]⟩

443 /-- **Theorem 5.8.** Let `K` be algebraically closed of characteristic zero and
444 `W ⊆ T*X` Zariski closed, fibre-conical and involutive.  If
445 `P ∈ K[ξ_1, …, ξ_n]` with `P(1, 0, …, 0) ≠ 0` vanishes on `W` and
446 `W ∩ π⁻¹(H) = ∅`, then `W = ∅`.  Proved along Subsection 5.3. -/
447 theorem theorem_5_8
448     (hm : 0 < m) (W : Set (PhaseVar m → k))
449     (hclosed : ∃ L : Ideal (SymbolRing k m), W = MvPolynomial.zeroLocus k L)
450     (hcone : IsFibreConical W)
451     (hinv : IsInvolutive (MvPolynomial.vanishingIdeal k W))
452     (P : MvPolynomial (Fin m) k)
453     (haxis : MvPolynomial.eval
454       (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) P ≠ 0)
455     (hPW : ∀ q ∈ W, MvPolynomial.eval (fun i => q (.inr i)) P = 0)
456     (hH : W ∩ {q | q (.inl ⟨0, hm⟩) = 0} = ∅) :
457     W = ∅ := by
458   -- Suppose `W ≠ ∅`.
459   by_contra hne
460   have hW : W.Nonempty := Set.nonempty_iff_ne_empty.2 hne
461   -- `B` is radical with a minimal prime `𝔮`; `Y_0 = V_K(𝔮)`.
462   obtain ⟨hrad, -, -, 𝔮, h𝔮⟩ := step_minimal_primes W hclosed hcone hW
463   obtain ⟨hprime, -, -, -, havoid⟩ := step_component hm W hclosed hcone hH 𝔮 h𝔮
464   -- The separator `s`, then `P` vanishes on `N(Y_0)`.
465   obtain ⟨s, hs𝔮, hs⟩ := step_separator W 𝔮 hrad h𝔮
466   have hPN := step_P_vanishes_on_conormal_cone W hclosed hcone hinv 𝔮 h𝔮 s hs𝔮 hs P hPW
467   -- Theorem 5.7(i) contradicts `P(1, 0, …, 0) ≠ 0`.
468   exact haxis (step_theorem_5_7_applies hm 𝔮 hprime havoid P
469     (step_P_vanishes_on_N 𝔮 hprime P hPN))
