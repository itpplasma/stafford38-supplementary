52 /-- **Theorem 5.7(i), along the paper's proof.**  If `Y = V(𝔭)` is nonempty,
53 irreducible and avoids `H = {x_1 = 0}`, every polynomial vanishing on the
54 conormal cone `N(Y)` over the smooth locus vanishes at `(1, 0, …, 0)`. -/
55 theorem theorem_5_7_i_paper
56     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
57     {m : ℕ} (hm : 0 < m)
58     (I : PrimeSpectrum (MvPolynomial (Fin m) k))
59     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) :
60     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
61       MvPolynomial.zeroLocus k
62         (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I.asIdeal)) := by
63   classical
64   intro Pol hPol
65   by_cases halg : IsAlgebraic k (componentCoordinate I ⟨0, hm⟩)
66   · -- First paragraph: `x̄_1` algebraic, so `x_1 − c ∈ 𝔭` and
67     -- `(1, 0, …, 0) = dx_1` is conormal at points of `Y`.
68     obtain ⟨c, y, -, -, hmem⟩ := exists_constant_coordinate_equation_and_pure_axis hm I halg
69     have h := fibreLift_mem_vanishingIdeal_equationConormal I.asIdeal I.isPrime Pol hPol _ hmem
70     simpa [fibreLift, MvPolynomial.eval_rename, Function.comp_def] using h
71   -- `x̄_1` transcendental.  `Y ∩ H = ∅` makes `x̄_1` a unit of `R`.
72   have hunit := exists_coordinate_inverse_of_avoidance I.asIdeal ⟨0, hm⟩ havoid
73   -- Lemmas 5.9–5.10: the discrete valuation ring `V` of `F` at infinity, the
74   -- column `q_i = x̄_i / x̄_j` and `κ = V/𝔪_V`, read in `V̂ ≅ κ[[t]]`.
75   obtain ⟨w⟩ := generalDivisorialVisibleFrameWithResidueAlgebraicity hm I hunit halg
76   let C := w.column
77   let W := C.W
78   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
79     (ComponentFractionField I) := W.ambientAlgebra
80   let V := W.place.valuation.toSubring
81   letI : IsDiscreteValuationRing V := W.place.isDiscrete
82   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
83   letI : Algebra W.coefficientField V :=
84     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
85   let κ := ResidueField V
86   letI : Algebra k κ := retainedResidueGroundAlgebra I ⟨0, hm⟩ W
87   letI : CharZero κ :=
88     charZero_of_injective_algebraMap (FaithfulSMul.algebraMap_injective k κ)
89   let q : Fin (m + 1) → PowerSeries κ := fun a ↦ retainedToCompletedPowerSeries W (C.q a)
90   let axis : Fin (m + 1) := Fin.succ ⟨0, hm⟩
91   -- Paragraph "`a = ord q_0`, `b = ord q_1`": `0 < a < b`, `j ∉ {0, 1}`.
92   obtain ⟨a, b, u₀, u₁, ha, hab, hq0, hu₀, hq1, -, hqj, -, -⟩ := step_orders hm I w hunit
93   have hq0ne : q 0 ≠ 0 := by
94     intro h
95     apply hu₀
96     have h' : (PowerSeries.X : PowerSeries κ) ^ a * u₀ = 0 := hq0.symm.trans h
97     rw [mul_eq_zero] at h'
98     rcases h' with h' | h'
99     · exact absurd h' (pow_ne_zero a PowerSeries.X_ne_zero)
100     · rw [h', map_zero]
101   -- Paragraph defining `y`, equation (generic): `f(y) = 0` for `f ∈ 𝔭`.
102   have hbase : ∀ f ∈ I.asIdeal, MvPolynomial.eval₂ (algebraMap k (LaurentSeries κ))
103       (dehomogenizedPoint (laurentColumn q)) f = 0 :=
104     fun f hf ↦ (step_generic_point hm I w f).2 hf
105   -- Lemma 5.11: `d(q_{i_l}(0))`, `i_l ∉ {0, 1, j}`, form a basis of `Ω_{κ/K}`.
106   obtain ⟨-, -, -, d, rows, hinj, -, hli, hspan, hd⟩ := witness_lemma_5_11 hm I w hunit
107   have hres := constantCoeff_retainedToCompleted_eq_residue (k := k) W
108   have hresRows : (fun l ↦ KaehlerDifferential.D k κ (PowerSeries.constantCoeff (q (rows l)))) =
109       fun l ↦ KaehlerDifferential.D k κ (residue V (C.q (rows l))) := by
110     funext l
111     rw [hres]
112   rw [← hresRows] at hli hspan
113   let rowsE : Fin d ↪ Fin (m + 1) := ⟨rows, hinj⟩
114   -- Paragraph after Lemma 5.11: dual derivations `D_k` and the matrix `Z(t)`.
115   obtain ⟨D, -, -, hminor, hZj, hZ0, hZ1⟩ :=
116     step_dual_derivations (k := k) hm q rowsE hli hspan C.chart hqj a b u₀ u₁ hq0 hq1
117   let Z := coefficientwiseTangentMatrix q D
118   -- Paragraph defining `λ`, `w = q' − Zλ`, `c ≤ a − 1` and `τ = t^{-c} w`.
119   obtain ⟨lambda, c, tau, -, -, hfactor, hprim, htaurows, htauj, htau1⟩ :=
120     step_lambda_tau hm q Z rowsE C.chart a b u₀ u₁ hqj hZj ha hab hq0 hu₀ hZ0 hq1 hZ1 hminor
121   -- Paragraph "`M(0)` has full column rank" for `M = (q, Z, τ)`.
122   obtain ⟨Cinv, hCM, hliM, haxis⟩ :=
123     step_M_full_rank hm q Z tau rowsE C.chart b u₁ hqj hZj htauj htaurows hminor hprim
124       (ha.trans hab) hq1 hZ1 htau1
125   -- Lemma 5.12(a): `dim Θ_y = dim_F Ω_{F/K} ≤ d' + 1`.
126   obtain ⟨⟨hdim, -, hfieldRank, -⟩, -, -⟩ := witness_lemma_5_12_5_13 hm I w
127   have hbound : Module.finrank (LaurentSeries κ)
128       (tangentSpaceAt (laurentColumn q) I.asIdeal) ≤ d + 1 := by
129     rw [hd]
130     exact hdim.trans_le hfieldRank
131   -- Paragraph "the columns of `M` span `𝒞`" (Lemmas 5.12(a), 5.12(b)).
132   obtain ⟨-, hspanC⟩ :=
133     step_M_spans_cone I.asIdeal q hq0ne hbase D Z rfl lambda c tau hfactor hliM hbound
134   -- Paragraph lifting `u = (0, 1, 0, …, 0)` to `ℓ(t)` with `ℓM = 0`.
135   obtain ⟨ell, hell, hellres⟩ := step_lift_ell _ Cinv hCM axis haxis
136   -- Paragraph "`ℓ` annihilates `𝒞`" (Lemma 5.13): `ℓ̃ ∈ span {df_y}`.
137   obtain ⟨hconormal, htail⟩ :=
138     step_ell_conormal hm I.asIdeal q hq0ne _ hspanC ell hell hellres
139   -- Final paragraph: `P(ℓ̃(0)) = 0` with `ℓ̃(0) = (1, 0, …, 0)`.
140   have hzero := step_polynomial_limit I Pol hPol _ hbase (fun i ↦ ell i.succ) hconormal
141   rw [htail, eval₂_extensionAxis_eq_algebraMap_eval_axis (K := κ) hm Pol] at hzero
142   exact (FaithfulSMul.algebraMap_injective k κ) (hzero.trans (map_zero _).symm)
