12 # Theorem 5.7(i), proved paragraph by paragraph
13 
14 The proof below follows Section 5.2 of the paper.  Each block names the
15 paragraph it formalizes and calls the declaration that states it:
16 the case `x̄_1` algebraic (`step_coordinate_algebraic`), Lemmas 5.9–5.10 with `Ω_{V/K}` finitely generated
17 (`GeneralDivisorialVisibleFrame.generalDivisorialVisibleFrameWithKaehlerFinite`), `V̂ ≅ κ[[t]]` (`step_completion`), the orders
18 `a`, `b` (`step_orders`), the generic point (`step_generic_point`), Lemma 5.11
19 (`witness_lemma_5_11`, via `lemma_5_11`), the dual derivations and `Z`
20 (`step_dual_derivations`), `λ`, `w`, `τ` (`step_lambda_w_tau`), the unit minor
21 of `M` (`step_M_minor`), Lemma 5.12(a) with the bound `m ≤ d' + 1`
22 (`witness_lemma_5_12_5_13_paper`, via `witness_lemma_5_12_a_bound` and
23 `lemma_5_12_a_bound`), the columns of `M` spanning `𝒞` (`step_M_spans_cone`,
24 via `lemma_5_12_a_cone`, `lemma_5_12_b`), the lift `ℓ` of `u`
25 (`step_lift_ell_minor`), Lemma 5.13 (`step_ell_conormal`, via `lemma_5_13`) and
26 the final paragraph (`step_polynomial_limit`).
27 -/

63 /-- **Theorem 5.7(i), along the paper's proof.**  If `Y = V(𝔭)` is nonempty,
64 irreducible and avoids `H = {x_1 = 0}`, every polynomial vanishing on the
65 conormal cone `N(Y)` over the smooth locus vanishes at `(1, 0, …, 0)`. -/
66 theorem theorem_5_7_i_paper
67     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
68     {m : ℕ} (hm : 0 < m)
69     (I : PrimeSpectrum (MvPolynomial (Fin m) k))
70     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) :
71     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
72       MvPolynomial.zeroLocus k
73         (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I.asIdeal)) := by
74   classical
75   intro Pol hPol
76   by_cases halg : IsAlgebraic k (componentCoordinate I ⟨0, hm⟩)
77   · -- First paragraph: `x̄_1` algebraic, so `x_1 − c ∈ 𝔭` with `c ≠ 0`, and
78     -- `(1, 0, …, 0) = dx_1` is conormal at a smooth point of `Y`, so lies in `N(Y)`.
79     obtain ⟨-, -, -, hN⟩ := step_coordinate_algebraic hm I havoid halg
80     exact hPol _ hN
81   -- `x̄_1` transcendental.  `Y ∩ H = ∅` makes `x̄_1` a unit of `R`.
82   have hunit := exists_coordinate_inverse_of_avoidance I.asIdeal ⟨0, hm⟩ havoid
83   -- Lemmas 5.9–5.10: the discrete valuation ring `V` of `F` at infinity, the
84   -- column `q_i = x̄_i / x̄_j` and `κ = V/𝔪_V`, read in `V̂ ≅ κ[[t]]`.
85   obtain ⟨w, hfin⟩ :=
86     Stafford38.Geometry.GeneralDivisorialVisibleFrame.generalDivisorialVisibleFrameWithKaehlerFinite
87       hm I hunit halg
88   let C := w.column
89   let W := C.W
90   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
91     (ComponentFractionField I) := W.ambientAlgebra
92   let V := W.place.valuation.toSubring
93   letI : IsDiscreteValuationRing V := W.place.isDiscrete
94   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
95   letI : Algebra W.coefficientField V :=
96     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
97   let κ := ResidueField V
98   letI : Algebra k κ := retainedResidueGroundAlgebra I ⟨0, hm⟩ W
99   letI : CharZero κ :=
100     charZero_of_injective_algebraMap (FaithfulSMul.algebraMap_injective k κ)
101   -- Paragraph "`V̂ ≅ κ[[t]]`": `F ↪ κ((t))`, and the residue is the constant term.
102   obtain ⟨-, -, -, -, hres, -, -, -⟩ := step_completion hm I w
103   let q : Fin (m + 1) → PowerSeries κ := fun a ↦ retainedToCompletedPowerSeries W (C.q a)
104   let axis : Fin (m + 1) := Fin.succ ⟨0, hm⟩
105   -- Paragraph "`a = ord q_0`, `b = ord q_1`": `0 < a < b`, `j ∉ {0, 1}`.
106   obtain ⟨a, b, u₀, u₁, ha, hab, hq0, hu₀, hq1, -, hqj, -, -⟩ := step_orders hm I w hunit
107   have hq0ne : q 0 ≠ 0 := by
108     intro h
109     apply hu₀
110     have h' : (PowerSeries.X : PowerSeries κ) ^ a * u₀ = 0 := hq0.symm.trans h
111     rw [mul_eq_zero] at h'
112     rcases h' with h' | h'
113     · exact absurd h' (pow_ne_zero a PowerSeries.X_ne_zero)
114     · rw [h', map_zero]
115   -- Paragraph defining `y`, equation (generic): `f(y) = 0` for `f ∈ 𝔭`.
116   have hbase : ∀ f ∈ I.asIdeal, MvPolynomial.eval₂ (algebraMap k (LaurentSeries κ))
117       (dehomogenizedPoint (laurentColumn q)) f = 0 :=
118     fun f hf ↦ (step_generic_point hm I w f).2 hf
119   -- Lemma 5.11: `d(q_{i_l}(0))`, `i_l ∉ {0, 1, j}`, form a basis of `Ω_{κ/K}`.
120   obtain ⟨-, -, -, d, rows, hinj, -, hli, hspan, hd⟩ := witness_lemma_5_11 hm I w hunit
121   have hresRows : (fun l ↦ KaehlerDifferential.D k κ (PowerSeries.constantCoeff (q (rows l)))) =
122       fun l ↦ KaehlerDifferential.D k κ (residue V (C.q (rows l))) := by
123     funext l
124     rw [hres]
125   rw [← hresRows] at hli hspan
126   let rowsE : Fin d ↪ Fin (m + 1) := ⟨rows, hinj⟩
127   -- Paragraph after Lemma 5.11: dual derivations `D_k` and the matrix `Z(t)`.
128   obtain ⟨D, -, -, hminor, hZj, hZ0, hZ1⟩ :=
129     step_dual_derivations (k := k) hm q rowsE hli hspan C.chart hqj a b u₀ u₁ hq0 hq1
130   let Z := coefficientwiseTangentMatrix q D
131   -- Paragraph defining `λ` (unique), `w = q' − Zλ`, `c ≤ a − 1` and `τ = t^{-c} w`.
132   obtain ⟨lambda, c, tau, -, -, -, -, -, -, -, hfactor, hprim, htaurows, -, htau1, htauj⟩ :=
133     step_lambda_w_tau hm q Z rowsE C.chart a b u₀ u₁ hqj hZj ha hab hq0 hu₀ hZ0 hq1 hZ1 hminor
134   -- Paragraph "`M(0)` has full column rank" for `M = (q, Z, τ)`: a unit minor
135   -- `M_ι`, independent columns over `κ((t))`, vanishing `1`-coordinates of `M(0)`.
136   obtain ⟨hinjM, -, hliM, haxis⟩ :=
137     step_M_minor hm q Z tau rowsE C.chart b u₁ hqj hZj htauj htaurows hminor hprim
138       (ha.trans hab) hq1 hZ1 htau1
139   -- Lemma 5.12(a): `dim Θ_y = dim_F Ω_{F/K} ≤ d' + 1`, the bound by Lemma 5.10(c),
140   -- the second fundamental exact sequence for `V → κ` and Nakayama.
141   obtain ⟨⟨hdim, -, hfieldRank, -⟩, -, -⟩ := witness_lemma_5_12_5_13_paper hm I w hfin
142   have hbound : Module.finrank (LaurentSeries κ)
143       (tangentSpaceAt (laurentColumn q) I.asIdeal) ≤ d + 1 := by
144     rw [hd]
145     exact hdim.trans_le hfieldRank
146   -- Paragraph "the columns of `M` span `𝒞`" (Lemmas 5.12(a), 5.12(b)).
147   obtain ⟨-, hspanC⟩ :=
148     step_M_spans_cone I.asIdeal q hq0ne hbase D Z rfl lambda c tau hfactor hliM hbound
149   -- Paragraph lifting `u = (0, 1, 0, …, 0)`: `uM(0) = 0`, an invertible `M_ι`,
150   -- and `ℓ_υ = u_υ`, `ℓ_ι = −u_υ M_υ M_ι⁻¹` with `ℓM = 0`, `ℓ(0) = u`.
151   have hu : rowMul (axisRow (k := κ) axis) (residueMatrix (formalTangentMatrix q Z tau)) = 0 := by
152     funext col
153     simp only [rowMul, axisRow, residueMatrix, ite_mul, one_mul, zero_mul,
154       Finset.sum_ite_eq', Finset.mem_univ, ite_true, Pi.zero_apply]
155     exact haxis col
156   obtain ⟨rowsι, -, -, -, -, hell, hellres⟩ :=
157     step_lift_ell_minor (formalTangentMatrix q Z tau) hinjM (axisRow (k := κ) axis) hu
158   -- Paragraph "`ℓ` annihilates `𝒞`" (Lemma 5.13): `ℓ̃ ∈ span {df_y}`.
159   obtain ⟨hconormal, htail⟩ :=
160     step_ell_conormal hm I.asIdeal q hq0ne _ hspanC _ hell hellres
161   -- Final paragraph: `P(ℓ̃(0)) = 0` with `ℓ̃(0) = (1, 0, …, 0)`.
162   have hzero := step_polynomial_limit I Pol hPol _ hbase _ hconormal
163   rw [htail, eval₂_extensionAxis_eq_algebraMap_eval_axis (K := κ) hm Pol] at hzero
164   exact (FaithfulSMul.algebraMap_injective k κ) (hzero.trans (map_zero _).symm)
