10 # Lemma 5.12(a): the generic tangent space has dimension `dim Ω_{F/k}`
11 
12 Let `F` be the function field of the component cut out by the prime `I`, with
13 generic point `y ∈ F^m`, and let `E ⊇ F` be a field extension (in the paper
14 `E = κ((t))`).  `Θ_y` is the tangent space at `y` of the equations of `I` with
15 coefficients extended to `E`.  The paper's argument is that the Jacobian has
16 rank `m - dim Ω_{F/k}` over `F` and that rank does not change under `F ⊆ E`.
17 
18 The upper bound `dim Θ_y ≤ dim Ω_{F/k}` is
19 `generic_tangent_finrank_le_component_kaehler_finrank`.  The lower bound below
20 uses the derivations of `F/k`: every `F`-linear form on `Ω_{F/k}` gives a
21 derivation `D`, and `(D y_i)_i` is tangent at `y`.  Such vectors for a dual
22 basis are linearly independent over `F`, hence over `E`.  In characteristic
23 zero `dim Ω_{F/k}` is the transcendence degree of `F/k`.
24 -/

36 /-- A vector is tangent to the scalar extension of `I` at `y` once `y` is a
37 zero of `I` and the vector is killed by the differentials of the equations of
38 `I` themselves: `E`-combinations of equations vanishing at `y` add nothing. -/
39 theorem mem_zariskiTangentSpace_map_of_forall_mem
40     {k E : Type*} [Field k] [Field E] [Algebra k E]
41     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) k)) (y w : Fin m → E)
42     (hy : ∀ g ∈ I, MvPolynomial.eval₂ (algebraMap k E) y g = 0)
43     (hw : ∀ g ∈ I, ∑ i, MvPolynomial.eval₂ (algebraMap k E) y
44       (MvPolynomial.pderiv i g) * w i = 0) :
45     w ∈ zariskiTangentSpace y (I.map (MvPolynomial.map (algebraMap k E))) := by
46   have hmap : ∀ h ∈ I.map (MvPolynomial.map (algebraMap k E)),
47       MvPolynomial.eval y h = 0 ∧ differentialCovector y h w = 0 := by
48     intro h hh
49     rw [Ideal.map] at hh
50     induction hh using Submodule.span_induction with
51     | mem h hh =>
52         rcases hh with ⟨g, hg, rfl⟩
53         refine ⟨by rw [MvPolynomial.eval_map]; exact hy g hg, ?_⟩
54         simpa [differentialCovector, differentialAt, MvPolynomial.pderiv_map,
55           MvPolynomial.eval_map] using hw g hg
56     | zero => simp [differentialCovector, differentialAt]
57     | add h₁ h₂ _ _ ih₁ ih₂ =>
58         refine ⟨by rw [MvPolynomial.eval_add, ih₁.1, ih₂.1, add_zero], ?_⟩
59         have hadd : differentialCovector y (h₁ + h₂) w =
60             differentialCovector y h₁ w + differentialCovector y h₂ w := by
61           simp only [differentialCovector_apply, differentialAt, map_add,
62             add_mul, Finset.sum_add_distrib]
63         rw [hadd, ih₁.2, ih₂.2, add_zero]
64     | smul a h _ ih =>
65         rw [show (a • h : MvPolynomial (Fin m) E) = a * h by rfl]
66         refine ⟨by rw [MvPolynomial.eval_mul, ih.1, mul_zero], ?_⟩
67         rw [differentialCovector_mul]
68         simp [ih.1, ih.2]
69   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator]
70   intro φ hφ
71   induction hφ using Submodule.span_induction with
72   | mem f hf =>
73       rcases hf with ⟨g, rfl⟩
74       exact (hmap g.1 g.2).2
75   | zero => simp
76   | add φ ψ _ _ ihφ ihψ => simp [ihφ, ihψ]
77   | smul a φ _ ihφ => simp [ihφ]

79 /-- The generic point of the component is a zero of every equation in `I`. -/
80 theorem eval₂_genericPoint_eq_zero
81     {k F : Type*} [Field k] [Field F] [Algebra k F]
82     {m : ℕ} {I : Ideal (MvPolynomial (Fin m) k)}
83     [Algebra (MvPolynomial (Fin m) k ⧸ I) F]
84     [IsScalarTower k (MvPolynomial (Fin m) k ⧸ I) F]
85     {g : MvPolynomial (Fin m) k} (hg : g ∈ I) :
86     MvPolynomial.eval₂ (algebraMap k F)
87       (GenericPointKaehlerConormal.genericPoint I F) g = 0 := by
88   have hhom : MvPolynomial.aeval (GenericPointKaehlerConormal.genericPoint I F) =
89       (IsScalarTower.toAlgHom k (MvPolynomial (Fin m) k ⧸ I) F).comp
90         (Ideal.Quotient.mkₐ k I) := by
91     apply MvPolynomial.algHom_ext
92     intro i
93     simp [GenericPointKaehlerConormal.genericPoint]
94   have h := congrArg (fun φ ↦ φ g) hhom
95   simp only [MvPolynomial.aeval_def, AlgHom.comp_apply,
96     Ideal.Quotient.mkₐ_eq_mk, (Ideal.Quotient.eq_zero_iff_mem).2 hg,
97     map_zero] at h
98   exact h

100 /-- **Lemma 5.12(a), lower bound.**  The tangent space `Θ_y` at the generic
101 point, with coefficients extended to `E`, has dimension at least
102 `dim_F Ω_{F/k}`. -/
103 theorem component_kaehler_finrank_le_generic_tangent_finrank
104     {k F E : Type u} [Field k] [Field F] [Field E]
105     [Algebra k F] [Algebra k E] [Algebra F E] [IsScalarTower k F E]
106     {m : ℕ} {I : Ideal (MvPolynomial (Fin m) k)} [I.IsPrime]
107     [Algebra (MvPolynomial (Fin m) k ⧸ I) F]
108     [IsScalarTower k (MvPolynomial (Fin m) k ⧸ I) F]
109     [IsFractionRing (MvPolynomial (Fin m) k ⧸ I) F]
110     (hgen : IntermediateField.adjoin k
111       (Set.range (GenericPointKaehlerConormal.genericPoint I F)) = ⊤) :
112     Module.finrank F (Ω[F⁄k]) ≤
113       Module.finrank E
114         (zariskiTangentSpace
115           (fun i ↦ algebraMap F E
116             (GenericPointKaehlerConormal.genericPoint I F i))
117           (I.map (MvPolynomial.map (algebraMap k E)))) := by
118   classical
119   let yF := GenericPointKaehlerConormal.genericPoint I F
120   let yE : Fin m → E := fun i ↦ algebraMap F E (yF i)
121   let IE : Ideal (MvPolynomial (Fin m) E) :=
122     I.map (MvPolynomial.map (algebraMap k E))
123   have hspan : Submodule.span F
124       (Set.range fun i ↦ KaehlerDifferential.D k F (yF i)) = ⊤ :=
125     Stafford38.Geometry.KaehlerSpanSeparableAdjoin.kaehler_span_of_intermediateField_adjoin_eq_top
126       (k := k) (E := F) yF hgen
127   have : FiniteDimensional F (Ω[F⁄k]) := by
128     have : FiniteDimensional F
129         (Submodule.span F
130           (Set.range fun i ↦ KaehlerDifferential.D k F (yF i))) :=
131       FiniteDimensional.span_of_finite F (Set.finite_range _)
132     have : FiniteDimensional F (⊤ : Submodule F (Ω[F⁄k])) :=
133       (LinearEquiv.ofEq _ _ hspan).finiteDimensional
134     exact Submodule.topEquiv.finiteDimensional
135   -- `Φ ψ = (ψ (d y_i))_i`: the Jacobian-dual coordinates of a form on `Ω`.
136   let Φ : Module.Dual F (Ω[F⁄k]) →ₗ[F] (Fin m → F) :=
137     { toFun := fun ψ i ↦ ψ (KaehlerDifferential.D k F (yF i))
138       map_add' := by intro ψ χ; rfl
139       map_smul' := by intro a ψ; rfl }
140   have hΦ : LinearMap.ker Φ = ⊥ := by
141     rw [LinearMap.ker_eq_bot']
142     intro ψ hψ
143     apply LinearMap.ext_on_range hspan
144     intro i
145     simpa [Φ] using congrFun hψ i
146   let b := (Module.finBasis F (Ω[F⁄k])).dualBasis
147   let z : Fin (Module.finrank F (Ω[F⁄k])) → Fin m → F := fun j ↦ Φ (b j)
148   have hliF : LinearIndependent F z := b.linearIndependent.map' Φ hΦ
149   have hliE : LinearIndependent E (fun j i ↦ algebraMap F E (z j i)) :=
150     (linearIndependent_algebraMap_comp_iff (R := F) (S := E)
151       (ι' := Fin m)).2 hliF
152   have htan (j : Fin (Module.finrank F (Ω[F⁄k]))) :
153       (fun i ↦ algebraMap F E (z j i)) ∈ zariskiTangentSpace yE IE := by
154     let D : Derivation k F F := (b j).compDer (KaehlerDifferential.D k F)
155     have hzD : ∀ i, z j i = D (yF i) := fun i ↦ rfl
156     have hcomp : algebraMap k E = (algebraMap F E).comp (algebraMap k F) :=
157       IsScalarTower.algebraMap_eq k F E
158     have hevalE : ∀ p : MvPolynomial (Fin m) k,
159         MvPolynomial.eval₂ (algebraMap k E) yE p =
160           algebraMap F E (MvPolynomial.eval₂ (algebraMap k F) yF p) := by
161       intro p
162       rw [MvPolynomial.eval₂_comp_left, ← hcomp]
163       rfl
164     apply mem_zariskiTangentSpace_map_of_forall_mem
165     · intro g hg
166       rw [hevalE, eval₂_genericPoint_eq_zero hg, map_zero]
167     · intro g hg
168       have hchain :=
169         Stafford38.Geometry.PowerSeriesArcTangency.derivation_eval₂ D g yF
170       rw [show yF = GenericPointKaehlerConormal.genericPoint I F from rfl,
171         eval₂_genericPoint_eq_zero hg, map_zero] at hchain
172       simp_rw [hevalE, hzD, ← map_mul, ← map_sum]
173       rw [← hchain, map_zero]
174   have hle : Submodule.span E
175       (Set.range fun j i ↦ algebraMap F E (z j i)) ≤ zariskiTangentSpace yE IE := by
176     rw [Submodule.span_le]
177     rintro _ ⟨j, rfl⟩
178     exact htan j
179   have hrank := Submodule.finrank_mono hle
180   rw [finrank_span_eq_card hliE, Fintype.card_fin] at hrank
181   simpa [yE, IE] using hrank

183 /-- **Lemma 5.12(a), `dim Θ_y = dim Ω_{F/k}`.**  The tangent space at the
184 generic point, with coefficients extended to `E ⊇ F`, has the Kähler rank of
185 the component function field as its dimension. -/
186 theorem generic_tangent_finrank_eq_component_kaehler_finrank
187     {k F E : Type u} [Field k] [Field F] [Field E]
188     [Algebra k F] [Algebra k E] [Algebra F E] [IsScalarTower k F E]
189     {m : ℕ} {I : Ideal (MvPolynomial (Fin m) k)} [I.IsPrime]
190     [Algebra (MvPolynomial (Fin m) k ⧸ I) F]
191     [IsScalarTower k (MvPolynomial (Fin m) k ⧸ I) F]
192     [IsFractionRing (MvPolynomial (Fin m) k ⧸ I) F]
193     (hgen : IntermediateField.adjoin k
194       (Set.range (GenericPointKaehlerConormal.genericPoint I F)) = ⊤) :
195     Module.finrank E
196         (zariskiTangentSpace
197           (fun i ↦ algebraMap F E
198             (GenericPointKaehlerConormal.genericPoint I F i))
199           (I.map (MvPolynomial.map (algebraMap k E)))) =
200       Module.finrank F (Ω[F⁄k]) :=
201   le_antisymm
202     (PaperGenericTangentRank.generic_tangent_finrank_le_component_kaehler_finrank
203       (E := E) hgen)
204     (component_kaehler_finrank_le_generic_tangent_finrank (E := E) hgen)

206 /-- **`dim Ω_{F/k} = trdeg_k F`.**  For a field `F` generated over `k` by
207 finitely many elements, in characteristic zero, the Kähler rank is the
208 transcendence degree (Matsumura, Thm. 26.5), which is finite. -/
209 theorem component_kaehler_finrank_eq_trdeg
210     {k F : Type u} [Field k] [CharZero k] [Field F] [Algebra k F]
211     {m : ℕ} (y : Fin m → F)
212     (hgen : IntermediateField.adjoin k (Set.range y) = ⊤) :
213     (Module.finrank F (Ω[F⁄k]) : Cardinal) = Algebra.trdeg k F ∧
214       Module.finrank F (Ω[F⁄k]) = (Algebra.trdeg k F).toNat := by
215   classical
216   have : Algebra.IsAlgebraic (Algebra.adjoin k (Set.range y)) F := by
217     rw [← IntermediateField.isAlgebraic_adjoin_iff_top, hgen,
218       Algebra.isAlgebraic_iff_isIntegral]
219     exact Algebra.isIntegral_of_surjective IntermediateField.topEquiv.surjective
220   obtain ⟨t, hts, ht⟩ := exists_isTranscendenceBasis_subset (R := k) (Set.range y)
221   have htfin : t.Finite := (Set.finite_range y).subset hts
222   let : Fintype t := htfin.fintype
223   have hrank : Module.finrank F (Ω[F⁄k]) = Fintype.card t :=
224     kaehlerFinrankOfTranscendenceBasis k F t (↑) ht
225   have hcard : (Fintype.card t : Cardinal) = Algebra.trdeg k F := by
226     rw [← Cardinal.mk_fintype]
227     exact ht.cardinalMk_eq_trdeg
228   refine ⟨by rw [hrank, hcard], ?_⟩
229   rw [← hcard, Cardinal.toNat_natCast, hrank]
230 
231 end
