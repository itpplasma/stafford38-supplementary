10 # Lemmas 5.12(b), 5.13 and the cone part of Lemma 5.12(a)
11 
12 Section 5.2 works at the generic point `y = (q_1/q_0, …, q_n/q_0)` with
13 coordinates in a field `L` (there `L = κ((t))`).  `Θ_y` is the tangent space
14 cut out by the differentials of the equations, the map
15 `ρ(v)_i = (v_i q_0 - q_i v_0) / q_0²` sends a homogeneous vector to the affine
16 direction it induces at `q`, and `𝒞 = ρ⁻¹(Θ_y)` is the affine cone over the
17 projective tangent plane.
18 
19 In Lean `Θ_y` is `zariskiTangentSpace y (𝔭 ⊗ L)`, the common kernel of the
20 differentials of all equations with coefficients extended to `L`; this is the
21 paper's `⋂_{f ∈ 𝔭} ker df_y`, because `L`-combinations of equations that vanish
22 at `y` have the same differentials as the corresponding combinations of the
23 equations themselves.
24 -/

37 /-- The paper's `ρ : L^{n+1} → L^n`, `ρ(v)_i = (v_i q_0 - q_i v_0) / q_0²`. -/
38 def rho (q : Fin (n + 1) → L) : (Fin (n + 1) → L) →ₗ[L] (Fin n → L) where
39   toFun v i := (v i.succ * q 0 - q i.succ * v 0) / q 0 ^ 2
40   map_add' v w := by
41     funext i
42     simp only [Pi.add_apply]
43     ring
44   map_smul' c v := by
45     funext i
46     simp only [Pi.smul_apply, smul_eq_mul, RingHom.id_apply]
47     ring
48 
49 @[simp] theorem rho_apply (q v : Fin (n + 1) → L) (i : Fin n) :
50     rho q v i = (v i.succ * q 0 - q i.succ * v 0) / q 0 ^ 2 := rfl

52 /-- The tangent space `Θ_y` at `y = q_i / q_0`. -/
53 def tangentSpaceAt (q : Fin (n + 1) → L) (I : Ideal (MvPolynomial (Fin n) k)) :
54     Submodule L (Fin n → L) :=
55   zariskiTangentSpace (dehomogenizedPoint q) (I.map (MvPolynomial.map (algebraMap k L)))
56 
57 /-- The affine cone `𝒞 = ρ⁻¹(Θ_y)` over the projective tangent plane. -/
58 def tangentCone (q : Fin (n + 1) → L) (I : Ideal (MvPolynomial (Fin n) k)) :
59     Submodule L (Fin (n + 1) → L) :=
60   (tangentSpaceAt q I).comap (rho q)
61 
62 /-- `ρ` is onto: `(0, q_0 w)` is sent to `w`. -/
63 theorem rho_surjective (q : Fin (n + 1) → L) (hq0 : q 0 ≠ 0) :
64     Function.Surjective (rho q) := by
65   intro w
66   refine ⟨Fin.cons 0 (fun i ↦ q 0 * w i), ?_⟩
67   funext i
68   simp only [rho_apply, Fin.cons_succ, Fin.cons_zero, mul_zero, sub_zero]
69   field_simp
70 
71 /-- The kernel of `ρ` is the line `L q`. -/
72 theorem ker_rho (q : Fin (n + 1) → L) (hq0 : q 0 ≠ 0) :
73     LinearMap.ker (rho q) = Submodule.span L {q} := by
74   apply le_antisymm
75   · intro v hv
76     rw [LinearMap.mem_ker] at hv
77     rw [Submodule.mem_span_singleton]
78     refine ⟨v 0 / q 0, ?_⟩
79     funext a
80     refine Fin.cases ?_ (fun i ↦ ?_) a
81     · simp only [Pi.smul_apply, smul_eq_mul]
82       field_simp
83     · have hi := congrFun hv i
84       simp only [rho_apply, Pi.zero_apply, div_eq_zero_iff, pow_eq_zero_iff, ne_eq,
85         OfNat.ofNat_ne_zero, not_false_eq_true, hq0, or_false, sub_eq_zero] at hi
86       simp only [Pi.smul_apply, smul_eq_mul]
87       rw [div_mul_eq_mul_div, div_eq_iff hq0]
88       linear_combination -hi
89   · rw [Submodule.span_le, Set.singleton_subset_iff, SetLike.mem_coe, LinearMap.mem_ker]
90     funext i
91     simp

93 /-- `ρ` maps `𝒞` onto `Θ_y`. -/
94 theorem rho_map_tangentCone (q : Fin (n + 1) → L) (hq0 : q 0 ≠ 0)
95     (I : Ideal (MvPolynomial (Fin n) k)) :
96     (tangentCone q I).map (rho q) = tangentSpaceAt q I :=
97   Submodule.map_comap_eq_of_surjective (rho_surjective q hq0) _

99 /-- **Lemma 5.12(a), cone part.** `dim 𝒞 = dim Θ_y + 1`. -/
100 theorem lemma_5_12_a_cone (q : Fin (n + 1) → L) (hq0 : q 0 ≠ 0)
101     (I : Ideal (MvPolynomial (Fin n) k)) :
102     Module.finrank L (tangentCone q I) = Module.finrank L (tangentSpaceAt q I) + 1 := by
103   classical
104   let f : tangentCone q I →ₗ[L] tangentSpaceAt q I :=
105     (rho q).restrict (p := tangentCone q I) (q := tangentSpaceAt q I) (fun v hv ↦ hv)
106   have hf : Function.Surjective f := by
107     rintro ⟨w, hw⟩
108     obtain ⟨v, rfl⟩ := rho_surjective q hq0 w
109     exact ⟨⟨v, hw⟩, rfl⟩
110   have hq : q ∈ tangentCone q I := by
111     change rho q q ∈ tangentSpaceAt q I
112     have : rho q q = 0 := by
113       rw [← LinearMap.mem_ker, ker_rho q hq0]
114       exact Submodule.mem_span_singleton_self q
115     rw [this]
116     exact zero_mem _
117   have hker : Module.finrank L (LinearMap.ker f) = 1 := by
118     have hq' : (⟨q, hq⟩ : tangentCone q I) ≠ 0 := by
119       intro h
120       exact hq0 (congrFun (congrArg Subtype.val h) 0)
121     rw [finrank_eq_one_iff_of_nonzero' (⟨⟨q, hq⟩, ?_⟩ : LinearMap.ker f) ?_]
122     · rintro ⟨⟨v, hv⟩, hvk⟩
123       have hv0 : v ∈ LinearMap.ker (rho q) := by
124         rw [LinearMap.mem_ker]
125         exact congrArg Subtype.val (LinearMap.mem_ker.mp hvk)
126       rw [ker_rho q hq0, Submodule.mem_span_singleton] at hv0
127       obtain ⟨c, rfl⟩ := hv0
128       exact ⟨c, rfl⟩
129     · rw [LinearMap.mem_ker]
130       apply Subtype.ext
131       change rho q q = 0
132       rw [← LinearMap.mem_ker, ker_rho q hq0]
133       exact Submodule.mem_span_singleton_self q
134     · intro h
135       exact hq' (congrArg Subtype.val h)
136   have := LinearMap.finrank_range_add_finrank_ker f
137   rw [LinearMap.range_eq_top.mpr hf, finrank_top, hker] at this
138   omega

140 /-- **Lemma 5.12(b).** If the equations vanish at `y`, then `δq ∈ 𝒞` for every
141 `K`-derivation `δ` of `L`, and `q ∈ 𝒞`. -/
142 theorem lemma_5_12_b (q : Fin (n + 1) → L) (hq0 : q 0 ≠ 0)
143     (I : Ideal (MvPolynomial (Fin n) k))
144     (hbase : ∀ f ∈ I,
145       MvPolynomial.eval₂ (algebraMap k L) (dehomogenizedPoint q) f = 0) :
146     (∀ δ : Derivation k L L, (fun a ↦ δ (q a)) ∈ tangentCone q I) ∧
147       q ∈ tangentCone q I := by
148   constructor
149   · intro δ
150     -- `δ(f(y)) = df_y · δy`, so `δy ∈ Θ_y`.
151     have htan := Stafford38.Geometry.PowerSeriesArcTangency.derivationVector_mem_zariskiTangentSpace_of_eval₂_eq_zero
152       δ I (dehomogenizedPoint q) hbase
153     -- Quotient rule: `ρ(δq) = δy`.
154     have hquot : rho q (fun a ↦ δ (q a)) = fun i ↦ δ (dehomogenizedPoint q i) := by
155       funext i
156       rw [rho_apply, dehomogenizedPoint, Derivation.leibniz_div]
157       simp only [smul_eq_mul]
158       field_simp
159     change rho q (fun a ↦ δ (q a)) ∈ tangentSpaceAt q I
160     rw [hquot]
161     exact htan
162   · change rho q q ∈ tangentSpaceAt q I
163     have : rho q q = 0 := by
164       rw [← LinearMap.mem_ker, ker_rho q hq0]
165       exact Submodule.mem_span_singleton_self q
166     rw [this]
167     exact zero_mem _

169 /-- **Lemma 5.13.** A row vector annihilating `𝒞` has its affine part
170 `(ℓ_1, …, ℓ_n)` in `span {df_y : f ∈ 𝔭}`. -/
171 theorem lemma_5_13 (q : Fin (n + 1) → L) (hq0 : q 0 ≠ 0)
172     (I : Ideal (MvPolynomial (Fin n) k)) (ell : Fin (n + 1) → L)
173     (hell : ∀ v ∈ tangentCone q I, ∑ a, ell a * v a = 0) :
174     coordinateCovector (fun i : Fin n ↦ ell i.succ) ∈
175       equationCovectorSpan (dehomogenizedPoint q)
176         (I.map (MvPolynomial.map (algebraMap k L))) := by
177   rw [← affineConormalSpace_eq_equationCovectorSpan, affineConormalSpace,
178     Submodule.mem_dualAnnihilator]
179   intro w hw
180   -- `ρ(𝒞) = Θ_y`.
181   have hw' : w ∈ (tangentCone q I).map (rho q) := by
182     rw [rho_map_tangentCone q hq0]
183     exact hw
184   obtain ⟨v, hv, rfl⟩ := hw'
185   have hq : q ∈ tangentCone q I := by
186     change rho q q ∈ tangentSpaceAt q I
187     have : rho q q = 0 := by
188       rw [← LinearMap.mem_ker, ker_rho q hq0]
189       exact Submodule.mem_span_singleton_self q
190     rw [this]
191     exact zero_mem _
192   -- `ℓ(q) = 0` and `ℓ(v) = 0` give `Σ ℓ_i ρ(v)_i = 0`.
193   have hellq := hell q hq
194   have hellv := hell v hv
195   rw [Fin.sum_univ_succ] at hellq hellv
196   rw [coordinateCovector_apply]
197   simp only [rho_apply]
198   have hsum : ∑ i : Fin n, ell i.succ * ((v i.succ * q 0 - q i.succ * v 0) / q 0 ^ 2) =
199       (q 0 * ∑ i : Fin n, ell i.succ * v i.succ -
200         v 0 * ∑ i : Fin n, ell i.succ * q i.succ) / q 0 ^ 2 := by
201     rw [Finset.mul_sum, Finset.mul_sum, ← Finset.sum_sub_distrib, Finset.sum_div]
202     refine Finset.sum_congr rfl fun i _ ↦ ?_
203     ring
204   rw [hsum]
205   have h1 : ∑ i : Fin n, ell i.succ * v i.succ = -(ell 0 * v 0) := by linear_combination hellv
206   have h2 : ∑ i : Fin n, ell i.succ * q i.succ = -(ell 0 * q 0) := by linear_combination hellq
207   rw [h1, h2]
208   ring_nf
