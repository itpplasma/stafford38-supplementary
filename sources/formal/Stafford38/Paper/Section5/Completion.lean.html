13 # The completion `V̂ ≅ κ[[t]]` and the bound of Lemma 5.12(a) at the place `V`
14 
15 Two paragraphs of the proof of Theorem 5.7(i) at the discrete valuation ring
16 `V` of Lemmas 5.9–5.10 (the witness `w`):
17 
18 * `step_completion`: a coefficient field of `V̂` containing `K` identifies
19   `V̂ ≅ κ[[t]]` with `ϖ ↦ t`; `V → κ[[t]]` is injective and extends to
20   `F ↪ κ((t))`, and the residue is the constant term.
21 * `witness_lemma_5_12_a_bound`: Lemma 5.12(a)'s bound `dim_F Ω_{F/K} ≤ d' + 1`
22   for this `V`, by the printed argument of `lemma_5_12_a_bound`.
23 -/

53 /-- **Paragraph "`V̂ ≅ κ[[t]]`".**  For the discrete valuation ring `V` of the
54 proof, with uniformizer `ϖ` (`𝔪_V = (ϖ)`), the map `V → V̂ ≅ κ[[t]]` given by
55 a coefficient field containing `K`: is injective; sends `c ∈ K` to the
56 constant series `c`; sends `ϖ` to `t`; has constant term equal to the residue
57 `res : V → κ`; and extends to an embedding `F ↪ κ((t))` which is the identity
58 on `K`. -/
59 theorem step_completion {k : Type u} [Field k] [CharZero k] {m : ℕ} (hm : 0 < m)
60     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
61     (w : GeneralDivisorialVisibleFrameWitness hm P) :
62     let C := w.column
63     let W := C.W
64     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
65       (ComponentFractionField P) := W.ambientAlgebra
66     let V := W.place.valuation.toSubring
67     letI : IsDiscreteValuationRing V := W.place.isDiscrete
68     letI : Algebra W.coefficientField V :=
69       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
70     let κ := ResidueField V
71     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
72     maximalIdeal V = Ideal.span {chosenUniformizer V} ∧
73       Function.Injective (retainedToCompletedPowerSeries W) ∧
74       (∀ c : k, retainedToCompletedPowerSeries W (C.groundCoeff c) =
75         PowerSeries.C (algebraMap k κ c)) ∧
76       retainedToCompletedPowerSeries W (chosenUniformizer V) = PowerSeries.X ∧
77       (∀ v : V, PowerSeries.constantCoeff (retainedToCompletedPowerSeries W v) =
78         residue V v) ∧
79       Function.Injective (retainedLaurentLift P ⟨0, hm⟩ W) ∧
80       (∀ v : V, retainedLaurentLift P ⟨0, hm⟩ W (algebraMap V (ComponentFractionField P) v) =
81         algebraMap (PowerSeries κ) (LaurentSeries κ) (retainedToCompletedPowerSeries W v)) ∧
82       (retainedLaurentLift P ⟨0, hm⟩ W).comp (algebraMap k (ComponentFractionField P)) =
83         groundLaurentMap (k := k) (K := κ) := by
84   intro C W
85   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
86     (ComponentFractionField P) := W.ambientAlgebra
87   intro V
88   letI : IsDiscreteValuationRing V := W.place.isDiscrete
89   letI : Algebra W.coefficientField V :=
90     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
91   intro κ
92   letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
93   have hground : ∀ c : k, retainedToCompletedPowerSeries W (C.groundCoeff c) =
94       PowerSeries.C (algebraMap k κ c) := by
95     intro c
96     rw [C.groundCoeff_eq_retained]
97     exact retainedToCompletedPowerSeries_ground P ⟨0, hm⟩ W c
98   exact ⟨maximalIdeal_eq_span_chosenUniformizer V, retainedToCompletedPowerSeries_injective W,
99     hground, retainedToCompletedPowerSeries_chosenUniformizer (k := k) W,
100     constantCoeff_retainedToCompleted_eq_residue (k := k) W, RingHom.injective _,
101     retainedLaurentLift_algebraMap P ⟨0, hm⟩ W, retainedLaurentLift_comp_algebraMap P ⟨0, hm⟩ W⟩

103 /-- **Lemma 5.12(a), `m ≤ d' + 1`, at the place of the proof.**  For the
104 discrete valuation ring `V` of the witness, with `Ω_{V/K}` finitely generated
105 (Lemma 5.10(c)), `dim_F Ω_{F/K} ≤ dim_κ Ω_{κ/K} + 1`; this is
106 `lemma_5_12_a_bound` (second fundamental exact sequence for `V → κ`,
107 Nakayama, `F ⊗_V Ω_{V/K} = Ω_{F/K}`), read with the witness's `K`-algebra
108 structure on `κ`. -/
109 theorem witness_lemma_5_12_a_bound {k : Type u} [Field k] [CharZero k] {m : ℕ} (hm : 0 < m)
110     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
111     (w : GeneralDivisorialVisibleFrameWitness hm P)
112     (hfin :
113       letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
114         (ComponentFractionField P) := w.column.W.ambientAlgebra
115       let V := w.column.W.place.valuation.toSubring
116       letI : Algebra k V := w.column.groundCoeff.toAlgebra
117       Module.Finite V (Ω[V⁄k])) :
118     let C := w.column
119     let W := C.W
120     let F := ComponentFractionField P
121     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
122     let V := W.place.valuation.toSubring
123     letI : IsDiscreteValuationRing V := W.place.isDiscrete
124     letI : Algebra W.coefficientField V :=
125       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
126     let κ := ResidueField V
127     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
128     Module.finrank F (Ω[F⁄k]) ≤ Module.finrank κ (Ω[κ⁄k]) + 1 := by
129   intro C W F
130   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
131   intro V
132   letI : IsDiscreteValuationRing V := W.place.isDiscrete
133   letI : Algebra W.coefficientField V :=
134     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
135   intro κ
136   have hfrac : IsFractionRing (↥W.place.valuation) (ComponentFractionField P) := inferInstance
137   letI : Algebra k V := C.groundCoeff.toAlgebra
138   letI : Algebra V F := V.subtype.toAlgebra
139   letI : IsScalarTower k V F := C.groundTower
140   letI : IsFractionRing V F := hfrac
141   letI : Module.Finite V (Ω[V⁄k]) := hfin
142   -- The bound with `κ` a `K`-algebra through `V`, i.e. `c ↦ res (c)`.
143   have hbound := lemma_5_12_a_bound (k := k) (V := V) (F := F)
144   -- That structure is the witness's retained one.
145   have hres : (retainedResidueGroundAlgebra P ⟨0, hm⟩ W : Algebra k κ) =
146       (inferInstance : Algebra k κ) := by
147     apply Algebra.algebra_ext
148     intro c
149     have h := RingHom.congr_fun w.retainedGroundMap c
150     change (@algebraMap k κ _ _ (retainedResidueGroundAlgebra P ⟨0, hm⟩ W)) c =
151       residue V (retainedComponentCoefficientMap P ⟨0, hm⟩ W c) at h
152     rw [h, ← C.groundCoeff_eq_retained]
153     rfl
154   letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
155   change Module.finrank F (Ω[F⁄k]) ≤
156     (letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W;
157       Module.finrank κ (Ω[κ⁄k])) + 1
158   rw [kaehler_finrank_eq_of_algebra_eq _ _ hres]
159   exact hbound
160 
161 end
