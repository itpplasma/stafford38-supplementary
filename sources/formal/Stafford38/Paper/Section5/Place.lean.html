9 # Lemmas 5.9 and 5.10: a valuation ring and a discrete valuation ring at infinity
10 
11 Section 5.2 of the paper works in the function field `F = Frac(K[x]/𝔭)` with
12 coordinate images `x̄_1, …, x̄_n` and `x̄_0 := 1`.  Here `F` is any field over
13 `k` generated by `x : Fin n → F`; the paper's `x̄_1` is `x i`.
14 
15 Both proofs use the checked stages of the formal construction
16 (`Stafford38.Geometry.LaneC`): Stage 1 is the dominating valuation ring, Stage 2
17 the normalization `A` of `C_0`, Stage 3 the height-one prime `𝔭_D` and Stage 4
18 the residue algebraicity.  The discrete valuation ring of Lemma 5.10 is the
19 same localization `A_{𝔭_D}` as in that construction.
20 -/

46 /-- **Lemma 5.9.** If `x̄_1` is transcendental over `K`, there are a valuation
47 ring `K ⊆ W ⊊ F` with `x̄_1 ∈ 𝔪_W` and an index `j` with `x̄_i / x̄_j ∈ W` for
48 all `0 ≤ i ≤ n`. -/
49 theorem lemma_5_9 {k F : Type*} [Field k] [Field F] [Algebra k F] {n : ℕ}
50     (x : Fin n → F) (i : Fin n) (htrans : Transcendental k (x i)) :
51     ∃ (W : ValuationSubring F) (j : Fin (n + 1)),
52       (∀ c : k, algebraMap k F c ∈ W) ∧ W ≠ ⊤ ∧
53       (∃ hx : x i ∈ W, (⟨x i, hx⟩ : W.toSubring) ∈ maximalIdeal W.toSubring) ∧
54       homogeneousCoordinates x j ≠ 0 ∧
55       ∀ a, homogeneousCoordinates x a / homogeneousCoordinates x j ∈ W := by
56   classical
57   -- `W` dominates `K[x̄_1]` localized at `x̄_1` (Stacks, Tag 00IA).
58   obtain ⟨W, hkW, hxW, hxinvW⟩ := stage1_exists_valuationSubring_of_transcendental (x i) htrans
59   have hx0 : x i ≠ 0 := fun h ↦ htrans (h ▸ isAlgebraic_zero)
60   have hxmax : (⟨x i, hxW⟩ : W.toSubring) ∈ maximalIdeal W.toSubring := by
61     rw [mem_maximalIdeal, mem_nonunits_iff]
62     rintro ⟨u, hu⟩
63     apply hxinvW
64     have hmul : x i * ((u⁻¹ : W.toSubringˣ) : W.toSubring) = 1 := by
65       have := congrArg Subtype.val (Units.mul_inv u)
66       simpa [hu] using this
67     have : (x i)⁻¹ = ((u⁻¹ : W.toSubringˣ) : W.toSubring) :=
68       (eq_inv_of_mul_eq_one_right hmul).symm
69     rw [this]
70     exact ((u⁻¹ : W.toSubringˣ) : W.toSubring).2
71   have hWtop : W ≠ ⊤ := by
72     rintro rfl
73     exact hxinvW (ValuationSubring.mem_top _)
74   -- Among the nonzero `x̄_i` choose one that is maximal for `a ≼ b ↔ x̄_a / x̄_b ∈ W`.
75   let Y := homogeneousCoordinates x
76   let S : Set (Fin (n + 1)) := {a | Y a ≠ 0}
77   have hS : S.Nonempty := ⟨0, by simp [S, Y]⟩
78   letI : LE (Fin (n + 1)) := ⟨fun a b ↦ Y b ≠ 0 ∧ Y a / Y b ∈ W⟩
79   have htr : IsTrans (Fin (n + 1)) LE.le := by
80     constructor
81     intro a b c hab hbc
82     rcases hab with ⟨hb, hab⟩
83     rcases hbc with ⟨hc, hbc⟩
84     refine ⟨hc, ?_⟩
85     have := W.toSubring.mul_mem hab hbc
86     rwa [div_mul_div_cancel₀ hb] at this
87   letI : IsTrans (Fin (n + 1)) LE.le := htr
88   obtain ⟨j, hjS, hjmax⟩ := S.toFinite.exists_maximalFor id S hS
89   refine ⟨W, j, hkW, hWtop, ⟨hxW, hxmax⟩, hjS, fun a ↦ ?_⟩
90   by_cases ha : Y a = 0
91   · change Y a / Y j ∈ W
92     rw [ha, zero_div]
93     exact W.zero_mem
94   rcases W.mem_or_inv_mem (Y a / Y j) with h | h
95   · exact h
96   · rw [inv_div] at h
97     exact (hjmax (show a ∈ S from ha) ⟨ha, h⟩).2

99 /-- The discrete valuation ring of Lemma 5.10 together with the construction

144 /-- **Lemma 5.10**, with the construction of its proof.  For `q_i = x̄_i / x̄_j`

343 /-- `R ⊄ V` in Lemma 5.10(a): `x̄_1 ∈ R^×` since `1 = ḡ x̄_1` with `ḡ ∈ R`, while

346 theorem Lemma510Place.not_R_le {k F : Type u} [Field k] [Field F] [Algebra k F] {n : ℕ}

353     rw [← eq_inv_of_mul_eq_one_right hxg]
354     exact hgR
