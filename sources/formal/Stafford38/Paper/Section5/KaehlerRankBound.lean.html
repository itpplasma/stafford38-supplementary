13 # Lemma 5.12(a), the bound `m ≤ d' + 1`
14 
15 The second half of the proof of Lemma 5.12(a): for a discrete valuation ring
16 `V ⊇ K` with fraction field `F`, residue field `κ` and `Ω_{V/K}` finitely
17 generated (Lemma 5.10(c)),
18 
19 * the second fundamental exact sequence `𝔪_V/𝔪_V² → κ ⊗_V Ω_{V/K} → Ω_{κ/K} → 0`
20   gives `dim_κ (κ ⊗_V Ω_{V/K}) ≤ d' + 1` with `d' = dim_κ Ω_{κ/K}`;
21 * Nakayama's lemma gives `Ω_{V/K}` generated by `dim_κ (κ ⊗_V Ω_{V/K})`
22   elements;
23 * `F ⊗_V Ω_{V/K} = Ω_{F/K}` gives `dim_F Ω_{F/K} ≤ d' + 1`.
24 -/

37 /-- **Second fundamental exact sequence for `V → κ`.**  Since
38 `𝔪_V/𝔪_V² → κ ⊗_V Ω_{V/K} → Ω_{κ/K} → 0` is exact and `𝔪_V = (ϖ)`, the
39 kernel of `κ ⊗_V Ω_{V/K} → Ω_{κ/K}` is spanned by `1 ⊗ dϖ`, so
40 `dim_κ (κ ⊗_V Ω_{V/K}) ≤ dim_κ Ω_{κ/K} + 1`. -/
41 theorem finrank_residue_tensor_kaehler_le [Module.Finite V (Ω[V⁄k])] :
42     Module.finrank (ResidueField V) (ResidueField V ⊗[V] Ω[V⁄k]) ≤
43       Module.finrank (ResidueField V) (Ω[ResidueField V⁄k]) + 1 := by
44   classical
45   have hsurj : Function.Surjective (algebraMap V (ResidueField V)) := residue_surjective
46   let φ := KaehlerDifferential.mapBaseChange k V (ResidueField V)
47   obtain ⟨ϖ, hϖ⟩ := IsDiscreteValuationRing.exists_irreducible V
48   -- Exactness at `κ ⊗_V Ω_{V/K}`: the kernel is the image of `𝔪_V/𝔪_V²`.
49   have hker : LinearMap.ker φ ≤ Submodule.span (ResidueField V)
50       {(1 : ResidueField V) ⊗ₜ KaehlerDifferential.D k V ϖ} := by
51     intro x hx
52     have hx' : x ∈ LinearMap.range
53         (KaehlerDifferential.kerCotangentToTensor k V (ResidueField V)) := by
54       rw [KaehlerDifferential.range_kerCotangentToTensor k V (ResidueField V) hsurj]
55       exact hx
56     obtain ⟨c, rfl⟩ := hx'
57     obtain ⟨z, rfl⟩ := Ideal.toCotangent_surjective _ c
58     rw [KaehlerDifferential.kerCotangentToTensor_toCotangent]
59     have hz : (z : V) ∈ maximalIdeal V := by
60       rw [← ker_residue]
61       exact z.2
62     rw [hϖ.maximalIdeal_eq, Ideal.mem_span_singleton'] at hz
63     obtain ⟨v, hv⟩ := hz
64     have hres : (algebraMap V (ResidueField V) ϖ) = 0 := by
65       rw [← RingHom.mem_ker]
66       change ϖ ∈ RingHom.ker (residue V)
67       rw [ker_residue]
68       exact hϖ.not_isUnit
69     rw [← hv, Derivation.leibniz, tmul_add, tmul_smul, tmul_smul, smul_tmul', smul_tmul',
70       Algebra.smul_def, Algebra.smul_def, mul_one, mul_one, hres, zero_tmul, add_zero]
71     have : algebraMap V (ResidueField V) v ⊗ₜ[V] KaehlerDifferential.D k V ϖ =
72         algebraMap V (ResidueField V) v •
73           ((1 : ResidueField V) ⊗ₜ[V] KaehlerDifferential.D k V ϖ) := by
74       rw [smul_tmul', smul_eq_mul, mul_one]
75     rw [this]
76     exact Submodule.smul_mem _ _ (Submodule.subset_span rfl)
77   have hrange : LinearMap.range φ = ⊤ :=
78     LinearMap.range_eq_top.mpr
79       (KaehlerDifferential.mapBaseChange_surjective k V (ResidueField V) hsurj)
80   have hrn := LinearMap.finrank_range_add_finrank_ker φ
81   rw [hrange, finrank_top] at hrn
82   have hk : Module.finrank (ResidueField V) (LinearMap.ker φ) ≤ 1 := by
83     refine (Submodule.finrank_mono hker).trans ?_
84     simpa using finrank_span_le_card (R := ResidueField V)
85       ({(1 : ResidueField V) ⊗ₜ[V] KaehlerDifferential.D k V ϖ} :
86         Set (ResidueField V ⊗[V] Ω[V⁄k]))
87   omega

89 /-- **Nakayama's lemma for `Ω_{V/K}`.**  A finitely generated `Ω_{V/K}` is
90 generated by `dim_κ (κ ⊗_V Ω_{V/K})` elements. -/
91 theorem exists_kaehler_generators [Module.Finite V (Ω[V⁄k])] :
92     ∃ f : Fin (Module.finrank (ResidueField V) (ResidueField V ⊗[V] Ω[V⁄k])) → Ω[V⁄k],
93       Submodule.span V (Set.range f) = ⊤ := by
94   let b := Module.finBasis (ResidueField V) (ResidueField V ⊗[V] Ω[V⁄k])
95   have hmk := TensorProduct.mk_surjective V (Ω[V⁄k]) (ResidueField V)
96     (residue_surjective (R := V))
97   choose f hf using fun i ↦ hmk (b i)
98   exact ⟨f, IsLocalRing.span_eq_top_of_tmul_eq_basis f b hf⟩

101 /-- **`F ⊗_V Ω_{V/K} = Ω_{F/K}`.**  Generators of `Ω_{V/K}` span `Ω_{F/K}`
102 over the fraction field `F`, so `dim_F Ω_{F/K}` is at most their number. -/
103 theorem finrank_fraction_kaehler_le_of_span
104     {F : Type u} [Field F] [Algebra V F] [IsFractionRing V F] [Algebra k F]
105     [IsScalarTower k V F] {ι : Type*} [Fintype ι] (f : ι → Ω[V⁄k])
106     (hf : Submodule.span V (Set.range f) = ⊤) :
107     Module.finrank F (Ω[F⁄k]) ≤ Fintype.card ι := by
108   have : Algebra.FormallyEtale V F := Algebra.FormallyEtale.of_isLocalization (nonZeroDivisors V)
109   let e := KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale k V F
110   let g : ι → Ω[F⁄k] := fun i ↦ KaehlerDifferential.map k k V F (f i)
111   have hmap : ∀ ω : Ω[V⁄k], KaehlerDifferential.map k k V F ω ∈
112       (Submodule.span F (Set.range g)).restrictScalars V := by
113     intro ω
114     have hω : ω ∈ Submodule.span V (Set.range f) := by rw [hf]; trivial
115     have h := Submodule.mem_map_of_mem (f := KaehlerDifferential.map k k V F) hω
116     rw [Submodule.map_span, ← Set.range_comp] at h
117     exact Submodule.span_le_restrictScalars V F _ h
118   have hspan : Submodule.span F (Set.range g) = ⊤ := by
119     suffices h : ∀ x, e x ∈ Submodule.span F (Set.range g) by
120       rw [eq_top_iff]
121       intro y _
122       obtain ⟨x, rfl⟩ := e.surjective y
123       exact h x
124     intro x
125     induction x using TensorProduct.inductionOn with
126     | tmul a ω =>
127         rw [KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale_apply,
128           KaehlerDifferential.mapBaseChange_tmul]
129         exact Submodule.smul_mem _ a (hmap ω)
130     | add x y hx hy => rw [map_add]; exact Submodule.add_mem _ hx hy
131   have h := finrank_range_le_card (R := F) g
132   rwa [Set.finrank, hspan, finrank_top] at h

134 /-- **Lemma 5.12(a), `dim_F Ω_{F/K} ≤ d' + 1`.**  For a discrete valuation ring
135 `V ⊇ K` with fraction field `F`, residue field `κ` and `Ω_{V/K}` finitely
136 generated: the second fundamental exact sequence for `V → κ` gives
137 `dim_κ (κ ⊗_V Ω_{V/K}) ≤ d' + 1`, Nakayama's lemma gives that many generators
138 of `Ω_{V/K}`, and `F ⊗_V Ω_{V/K} = Ω_{F/K}` bounds `dim_F Ω_{F/K}`. -/
139 theorem lemma_5_12_a_bound
140     {F : Type u} [Field F] [Algebra V F] [IsFractionRing V F] [Algebra k F]
141     [IsScalarTower k V F] [Module.Finite V (Ω[V⁄k])] :
142     Module.finrank F (Ω[F⁄k]) ≤ Module.finrank (ResidueField V) (Ω[ResidueField V⁄k]) + 1 := by
143   obtain ⟨f, hf⟩ := exists_kaehler_generators (k := k) (V := V)
144   have h := finrank_fraction_kaehler_le_of_span (F := F) f hf
145   rw [Fintype.card_fin] at h
146   exact h.trans finrank_residue_tensor_kaehler_le
147 
148 end
