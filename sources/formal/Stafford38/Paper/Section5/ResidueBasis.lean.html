10 # Lemma 5.11: residue field and differential basis
11 
12 Paper statement (Section 5.2): with `q_0(0) = q_1(0) = 0`, `q_j(0) = 1` and
13 `κ` algebraic over `L = K(q_0(0), …, q_n(0))`, some residues `q_i(0)` with
14 `i ∉ {0, 1, j}` have differentials forming a basis of `Ω_{κ/K}`; in
15 particular `d' = dim_κ Ω_{κ/K} < ∞`.
16 
17 The proof follows the paper: a transcendence basis of `L/K` is chosen inside
18 the residues with `i ∉ {0, 1, j}` (Lang, Thm. VIII.1.1, the closing sentence of
19 Definition 5.1), it is one of `κ/K` because `κ/L` is algebraic, and in characteristic
20 zero its differentials form a basis (Matsumura, Thm. 26.5).  The paper's index
21 `1` is `Fin.succ ⟨0, hm⟩`, the first affine coordinate.
22 -/

32 /-- **Lemma 5.11.** -/
33 theorem lemma_5_11
34     {k κ : Type u} [Field k] [CharZero k] [Field κ] [Algebra k κ]
35     {m : ℕ} (hm : 0 < m) (r : Fin (m + 1) → κ) (j : Fin (m + 1))
36     (hr0 : r 0 = 0) (hr1 : r (Fin.succ ⟨0, hm⟩) = 0) (hrj : r j = 1)
37     (halg : Algebra.IsAlgebraic
38       (IntermediateField.adjoin k (Set.range r) : IntermediateField k κ) κ) :
39     ∃ (d' : ℕ) (rows : Fin d' → Fin (m + 1)),
40       Function.Injective rows ∧
41       (∀ l, rows l ≠ 0 ∧ rows l ≠ Fin.succ ⟨0, hm⟩ ∧ rows l ≠ j) ∧
42       LinearIndependent κ (fun l ↦ KaehlerDifferential.D k κ (r (rows l))) ∧
43       Submodule.span κ
44           (Set.range fun l ↦ KaehlerDifferential.D k κ (r (rows l))) = ⊤ ∧
45       d' = Module.finrank κ (Ω[κ⁄k]) := by
46   classical
47   let axis : Fin (m + 1) := Fin.succ ⟨0, hm⟩
48   let good : Set (Fin (m + 1)) := {a | a ≠ 0 ∧ a ≠ axis ∧ a ≠ j}
49   let S : Set κ := r '' good
50   -- The excluded residues are `0` and `1`, so `L` is generated by `S`.
51   have hL : IntermediateField.adjoin k (Set.range r) = IntermediateField.adjoin k S := by
52     apply le_antisymm
53     · rw [IntermediateField.adjoin_le_iff]
54       rintro _ ⟨a, rfl⟩
55       by_cases ha : a ∈ good
56       · exact IntermediateField.subset_adjoin k S ⟨a, ha, rfl⟩
57       · simp only [good, Set.mem_ofPred_eq, not_and_or, not_not] at ha
58         rcases ha with rfl | rfl | rfl
59         · rw [hr0]; exact zero_mem _
60         · rw [hr1]; exact zero_mem _
61         · rw [hrj]; exact one_mem _
62     · apply IntermediateField.adjoin.mono
63       rintro _ ⟨a, -, rfl⟩
64       exact ⟨a, rfl⟩
65   -- `L` is generated by the finite set `S`, which therefore contains a transcendence
66   -- basis `B` of `L/K` (Definition 5.1, closing sentence; Lang, Thm. VIII.1.1).
67   let SL : Set (IntermediateField.adjoin k S) := Subtype.val ⁻¹' S
68   have hSL : IntermediateField.adjoin k SL = ⊤ := by
69     apply IntermediateField.lift_injective (IntermediateField.adjoin k S)
70     rw [IntermediateField.lift_adjoin, IntermediateField.lift_top]
71     congr 1
72     change Subtype.val '' (Subtype.val ⁻¹' S) = S
73     ext z
74     refine ⟨fun ⟨w, hw, hwz⟩ => hwz ▸ hw, fun hz => ?_⟩
75     exact ⟨⟨z, IntermediateField.subset_adjoin k S hz⟩, hz, rfl⟩
76   have hSLfin : SL.Finite :=
77     ((Set.toFinite good).image r).preimage Subtype.val_injective.injOn
78   obtain ⟨BL, hBLS, hBL⟩ := (definition_5_1_closing (K := k)
79     (L := IntermediateField.adjoin k S)).2 SL hSLfin hSL
80   -- `κ/L` is algebraic (Lemma 5.10(b)), so `B` is a transcendence basis of `κ/K`.
81   have : Algebra.IsAlgebraic (IntermediateField.adjoin k S) κ := hL ▸ halg
82   let t : Set κ := Set.range (algebraMap (IntermediateField.adjoin k S) κ ∘ ((↑) : BL → _))
83   have ht : IsTranscendenceBasis k ((↑) : t → κ) := hBL.algebraMap_comp.to_subtype_range
84   have htS : t ⊆ S := by
85     rintro _ ⟨b, rfl⟩
86     exact hBLS b.2
87   have htfin : t.Finite := (Set.toFinite good |>.image r).subset htS
88   let : Fintype t := htfin.fintype
89   -- Matsumura, Thm. 26.5: the transcendence basis is a differential basis.
90   have hrank : Module.finrank κ (Ω[κ⁄k]) = Fintype.card t :=
91     Stafford38.Geometry.kaehlerFinrankOfTranscendenceBasis k κ t (↑) ht
92   have hspanT : Submodule.span κ
93       (Set.range fun x : t ↦ KaehlerDifferential.D k κ (x : κ)) = ⊤ :=
94     Stafford38.Geometry.KaehlerDVRVisibility.kaehler_span_of_isAlgebraic_adjoin
95       (fun x : t ↦ (x : κ)) (by simpa using (definition_5_1_g_h ht).1)
96   -- Index the basis elements by residues `q_i(0)` with `i ∉ {0, 1, j}`.
97   have hpre : ∀ x : t, ∃ a ∈ good, r a = x := fun x ↦ htS x.2
98   choose idx hidx hridx using hpre
99   let e := Fintype.equivFin t
100   let rows : Fin (Fintype.card t) → Fin (m + 1) := fun l ↦ idx (e.symm l)
101   have hrows : ∀ l, r (rows l) = (e.symm l : κ) := fun l ↦ hridx _
102   have hfam : (fun l ↦ KaehlerDifferential.D k κ (r (rows l))) =
103       (fun x : t ↦ KaehlerDifferential.D k κ (x : κ)) ∘ e.symm := by
104     funext l
105     simp [hrows]
106   have hspan : Submodule.span κ
107       (Set.range fun l ↦ KaehlerDifferential.D k κ (r (rows l))) = ⊤ := by
108     rw [hfam, Set.range_comp, e.symm.surjective.range_eq, Set.image_univ, hspanT]
109   have hli : LinearIndependent κ (fun l ↦ KaehlerDifferential.D k κ (r (rows l))) :=
110     linearIndependent_of_top_le_span_of_card_eq_finrank hspan.ge
111       (by rw [Fintype.card_fin, hrank])
112   refine ⟨Fintype.card t, rows, ?_, fun l ↦ hidx _, hli, hspan, hrank.symm⟩
113   intro l l' h
114   apply e.symm.injective
115   apply Subtype.ext
116   rw [← hrows, ← hrows, h]
