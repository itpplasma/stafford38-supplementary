13 # The Jacobian criterion: Jacobian smooth points are Mathlib smooth points
14 
15 Let `𝔭 ⊂ K[x] = K[x_1, …, x_n]` be prime, `y ∈ V(𝔭)`, and suppose that
16 `span {df_y : f ∈ 𝔭}` has dimension `c = n - trdeg_K Frac(K[x]/𝔭)`.  Choose
17 `f_1, …, f_c ∈ 𝔭` with independent differentials at `y` and let `P'` be the
18 localization of `K[x]` at `y`.
19 
20 1. `B = P'/(f_1, …, f_c)` is a regular local ring: `dim B ≥ n - c` by Krull's height
21    theorem (`le_height_ker_aeval`), and `𝔪_B` is generated by `n - c` linear forms
22    (Taylor expansion and Nakayama).  Hence `B` is a domain.
23 2. The prime `𝔯 = ker(K[x] → B)` lies between `(f)` and `𝔭`.  At the generic point of
24    `𝔯` the Jacobian of `f` still has rank `c` (a `c × c` minor is a unit at `y`), so
25    `trdeg K[x]/𝔯 = dim Θ ≤ n - c = trdeg K[x]/𝔭` (Lemma 5.12(a)); a nonzero prime lowers
26    the transcendence degree (`trdeg_quotient_lt`), so `𝔯 = 𝔭` and `𝔭 P' = (f) P'`.
27 3. Mathlib's local Jacobian criterion
28    (`Algebra.FormallySmooth.iff_injective_cotangentComplexBaseChange`) for
29    `P' → (K[x]/𝔭)_𝔪` then reduces smoothness to the independence of the `df_i(y)`.
30 -/

59 /-- The linear form `Σ_j v_j (x_j - y_j)`. -/
60 def linearPart (y : Fin n → k) : (Fin n → k) →ₗ[k] MvPolynomial (Fin n) k where
61   toFun v := ∑ j, C (v j) * (X j - C (y j))
62   map_add' v w := by simp [add_mul, Finset.sum_add_distrib]
63   map_smul' a v := by
64     simp only [Pi.smul_apply, smul_eq_mul, map_mul, RingHom.id_apply]
65     rw [Finset.smul_sum]
66     refine Finset.sum_congr rfl fun j _ ↦ ?_
67     rw [MvPolynomial.smul_eq_C_mul]
68     ring

85 /-- **Taylor expansion to first order.** `g - g(y) - Σ_j ∂_j g(y) (x_j - y_j) ∈ 𝔪_y²`. -/
86 theorem taylor_mem_pointIdeal_sq (y : Fin n → k) (g : MvPolynomial (Fin n) k) :
87     g - C (MvPolynomial.eval y g) - linearPart y (differentialAt y g) ∈ pointIdeal y ^ 2 := by
88   classical
89   induction g using MvPolynomial.induction_on with
90   | C a =>
91     have : differentialAt y (C a) = 0 := by funext i; simp [differentialAt]
92     simp [this]
93   | add p q hp hq =>
94     rw [differentialAt_add, map_add]
95     convert add_mem hp hq using 1
96     simp only [map_add]
97     ring
98   | mul_X p j hp =>
99     have hd : differentialAt y (p * X j) =
100         y j • differentialAt y p + MvPolynomial.eval y p • Pi.single j 1 := by
101       funext i
102       simp only [differentialAt, MvPolynomial.pderiv_mul, MvPolynomial.pderiv_X, map_add,
103         map_mul, MvPolynomial.eval_X, Pi.add_apply, Pi.smul_apply, smul_eq_mul]
104       by_cases h : i = j
105       · subst h; simp; ring
106       · simp [h, Ne.symm h]; ring
107     have hid : p * X j - C (MvPolynomial.eval y (p * X j)) -
108         linearPart y (differentialAt y (p * X j)) =
109         linearPart y (differentialAt y p) * (X j - C (y j)) +
110           (p - C (MvPolynomial.eval y p) - linearPart y (differentialAt y p)) * X j := by
111       rw [hd, map_add, map_smul, map_smul, linearPart_single, MvPolynomial.eval_mul,
112         MvPolynomial.eval_X, MvPolynomial.smul_eq_C_mul, MvPolynomial.smul_eq_C_mul, map_mul]
113       ring
114     rw [hid, pow_two]
115     exact add_mem (Ideal.mul_mem_mul (linearPart_mem _ _) (by rw [mem_pointIdeal]; simp))
116       (Ideal.mul_mem_right _ _ (pow_two (pointIdeal y) ▸ hp))
117 
118 end Taylor

306 /-- **The Jacobian rank does not drop at the generic point of a prime `𝔯 ⊆ 𝔪_y`.** If
307 `f_1, …, f_c ∈ 𝔯` have independent differentials at `y ∈ V(𝔯)`, the tangent space at the
308 generic point of `𝔯` has dimension at most `n - c`: a `c × c` minor of the Jacobian is `1`
309 at `y`, hence not in `𝔯`. -/
310 theorem finrank_genericTangent_le {ι : Type*} [Fintype ι] (𝔯 : Ideal (MvPolynomial (Fin n) k))
311     [𝔯.IsPrime] (y : Fin n → k) (h𝔯 : 𝔯 ≤ pointIdeal y)
312     (f : ι → MvPolynomial (Fin n) k) (hf : ∀ i, f i ∈ 𝔯)
313     (hli : LinearIndependent k fun i ↦ differentialAt y (f i)) :
314     Module.finrank (FractionRing (MvPolynomial (Fin n) k ⧸ 𝔯))
315         (zariskiTangentSpace
316           (GenericPointKaehlerConormal.genericPoint 𝔯 (FractionRing (MvPolynomial (Fin n) k ⧸ 𝔯)))
317           (𝔯.map (MvPolynomial.map (algebraMap k (FractionRing (MvPolynomial (Fin n) k ⧸ 𝔯)))))) ≤
318       n - Fintype.card ι := by
319   classical
320   let L := FractionRing (MvPolynomial (Fin n) k ⧸ 𝔯)
321   let yb := GenericPointKaehlerConormal.genericPoint 𝔯 L
322   let ψ : MvPolynomial (Fin n) k →+* L :=
323     (algebraMap (MvPolynomial (Fin n) k ⧸ 𝔯) L).comp (Ideal.Quotient.mk 𝔯)
324   have hψ : ∀ p, ψ p = MvPolynomial.eval₂ (algebraMap k L) yb p :=
325     fun p ↦ algebraMap_quotient_mk_eq_eval₂ (I := 𝔯) (L := L) p
326   have hψC : ∀ c : k, ψ (C c) = algebraMap k L c := by
327     intro c; rw [hψ]; simp
328   -- the Jacobian at `y` and at the generic point
329   let A0 : Matrix ι (Fin n) k := Matrix.of fun i j ↦ differentialAt y (f i) j
330   let AL : Matrix ι (Fin n) L := Matrix.of fun i j ↦ ψ (pderiv j (f i))
331   have hA0surj : Function.Surjective A0.mulVecLin := by
332     rw [← LinearMap.range_eq_top]
333     apply Submodule.eq_top_of_finrank_eq
334     have h1 : Module.finrank k (LinearMap.range A0.mulVecLin) = A0.rank := rfl
335     rw [h1, Matrix.rank_eq_finrank_span_row]
336     have hrows : Set.range A0.row = Set.range fun i ↦ differentialAt y (f i) := rfl
337     rw [hrows, finrank_span_eq_card hli, Module.finrank_fintype_fun_eq_card]
338   choose G hG using fun i' : ι ↦ hA0surj (Pi.single i' 1)
339   let Mk : Matrix (Fin n) ι k := Matrix.of fun j i' ↦ G i' j
340   have hA0Mk : A0 * Mk = 1 := by
341     ext i i'
342     have := congrFun (hG i') i
343     simp only [Matrix.mulVecLin_apply, Matrix.mulVec, dotProduct] at this
344     simp only [Matrix.mul_apply, Mk, Matrix.of_apply, Matrix.one_apply]
345     rw [this, Pi.single_apply]
346   let N : Matrix ι ι (MvPolynomial (Fin n) k) :=
347     Matrix.of fun i i' ↦ ∑ j, pderiv j (f i) * C (Mk j i')
348   have hNy : (MvPolynomial.eval y).mapMatrix N = A0 * Mk := by
349     ext i i'
350     simp [N, A0, Matrix.mul_apply, differentialAt]
351   have hdetN : N.det ∉ 𝔯 := by
352     intro h
353     have h1 := h𝔯 h
354     rw [mem_pointIdeal, RingHom.map_det, hNy, hA0Mk, Matrix.det_one] at h1
355     exact one_ne_zero h1
356   have hNL : ψ.mapMatrix N = AL * Mk.map (algebraMap k L) := by
357     ext i i'
