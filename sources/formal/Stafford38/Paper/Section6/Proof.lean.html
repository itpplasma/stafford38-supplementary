12 # Section 6: the proof of Theorem 1.1, paragraph by paragraph (tex 1531–1597)
13 
14 For `d̃` in the normal form (3.1) with `N = deg_B d̃ ≥ 1` the paper shows
15 `Q = A_n(k)/𝓘 = 0`.  It passes to an algebraically closed `K ⊇ k`
16 (`K = k̄`), sets `W = V_K(Ann_{K[x,ξ]} gr Q_K)`, the closed points of
17 `Char(Q_K)`, checks the six hypotheses of Theorem 5.8 for `W` and
18 `P = σ_N^ord(d̃)` (one `step_W_*` per bullet), and concludes `W = ∅`, hence
19 `gr Q_K = 0`, `Q_K = 0` and `Q = 0`.
20 
21 The steps are stated for any algebraically closed `K` of characteristic zero
22 over `k`; `theorem_6_Q_eq_zero` takes `K = AlgebraicClosure k`.  A point
23 `(y, ξ) ∈ K^n × (K^n)^*` is a function `q : PhaseVar (n+1) → K`; `y₁` is
24 `q (Sum.inl 0)`, and `π⁻¹(H) = {y₁ = 0}`.

73 /-- "The module `gr Q_K` is graded by the order filtration, which assigns
74 degree `0` to the `x_i` and degree `1` to the `ξ_i`; so `Ann gr Q_K` is
75 homogeneous in `ξ`": a homogeneous `g_d` of degree `d` maps the degree-`0`
76 generator `q̄` into degree `d`, so `g q̄ = 0` forces `g_d q̄ = 0` for every
77 component, and `gr Q_K` is generated by `q̄`. -/
78 theorem step_annihilator_homogeneous {n N : ℕ} (dt : PresentedWeyl k (n + 1)) :
79     (Module.annihilator (SymbolRing K (n + 1)) (grQ K N (dtK K dt))).IsHomogeneous
80       (MvPolynomial.weightedHomogeneousSubmodule K (@orderWeight (n + 1))) := by
81   classical
82   let I := idealI K N (dtK K dt)
83   let 𝒜 := MvPolynomial.weightedHomogeneousSubmodule K (@orderWeight (n + 1))
84   let x0 := orderPieceToQuotientGraded K I 0 (orderPieceOne (n := n + 1) K)
85   have hq : qBar K I = orderAssociatedGradedOf K I 0 x0 := rfl
86   -- `π_e`, the degree-`e` component of `gr Q_K = ⊕_e Q_{K,e}/Q_{K,e-1}`.
87   let π : ∀ e : ℕ, OrderAssociatedGradedModule K I →+ QuotientOrderGradedPiece K I e :=
88     fun e => { toFun := fun z =>
89                 (show Stafford38.CharacteristicFilteredQuotientGraded.QuotientOrderAssociatedGraded
90                   K I from z) e
91                map_zero' := rfl
92                map_add' := fun _ _ => rfl }
93   have hπ : ∀ e j (x : QuotientOrderGradedPiece K I j),
94       π e (orderAssociatedGradedOf K I j x) = if h : j = e then h ▸ x else 0 := by
95     intro e j x
96     change (DirectSum.of _ j x : DirectSum ℕ _) e = _
97     rw [DirectSum.of_apply]
98   -- The degree-`(0 + d)` component of `g q̄` is `g_d q̄`.
99   have hcomp : ∀ (g : SymbolRing K (n + 1)) (d : ℕ),
100       ((DirectSum.decompose 𝒜 g d : SymbolRing K (n + 1))) • qBar K I =
101         orderAssociatedGradedOf K I (0 + d) (π (0 + d) (g • qBar K I)) := by
102     intro g d
103     rw [hq, smul_orderAssociatedGradedOf_eq_of_homogeneousAction]
104     congr 1
105     conv_rhs => rw [← DirectSum.sum_support_decompose 𝒜 g]
106     rw [Finset.sum_smul, map_sum]
107     simp_rw [smul_orderAssociatedGradedOf_eq_of_homogeneousAction, hπ]
108     by_cases hd : d ∈ (DirectSum.decompose 𝒜 g).support
109     · rw [Finset.sum_eq_single d]
110       · simp
111       · intro i _ hid
112         exact dite_eq_right_iff.mpr fun h => absurd h (by omega)
113       · exact fun h => absurd hd h
114     · have hact : CharacteristicFilteredQuotientGraded.quotientOrderHomogeneousAction K I
115           (0 : 𝒜 d) x0 = 0 := by
116         have h0 := smul_orderAssociatedGradedOf_eq_of_homogeneousAction K I (0 : 𝒜 d) x0
117         rw [Submodule.coe_zero, zero_smul] at h0
118         exact DirectSum.of_injective (β := QuotientOrderGradedPiece K I) (0 + d)
119           (h0.symm.trans (map_zero (DirectSum.of (QuotientOrderGradedPiece K I) (0 + d))).symm)
120       rw [DFinsupp.notMem_support_iff.mp hd, hact, Finset.sum_eq_zero]
121       intro i hi
122       exact dite_eq_right_iff.mpr fun h => absurd (show i = d by omega ▸ hi) hd
123   intro d g hg
124   rw [Module.mem_annihilator] at hg ⊢
125   intro m

129     map_zero _

326 /-- `n = 0`: `A_0(k) = k`, and every nonzero `d` is a unit. -/
327 theorem step_rank_zero (d : PresentedWeyl k 0) (hd : d ≠ 0) :
328     ∃ F R S : PresentedWeyl k 0, (1 : PresentedWeyl k 0) = d * R + F * d * S := by
329   let e := presentedIteratedEquiv k 0
330   let c : k := e d
331   have hc : c ≠ 0 := by
332     intro hczero
333     change e d = 0 at hczero

335     calc
336       d = e.symm (e d) := (e.symm_apply_apply d).symm
337       _ = e.symm 0 := congrArg e.symm hczero
338       _ = 0 := map_zero e.symm

340   have hmul : c * c⁻¹ = (1 : k) := mul_inv_cancel₀ hc
341   have h : d * e.symm (c⁻¹ : k) = 1 := by

344           congr 1
345           exact (e.symm_apply_apply d).symm
346       _ = e.symm (c * c⁻¹) := (map_mul e.symm _ _).symm
347       _ = e.symm 1 := congrArg e.symm hmul
348       _ = 1 := map_one e.symm
