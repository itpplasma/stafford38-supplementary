10 # Proposition 6.1: `Char(Q) ∩ V(x₁) = ∅`
11 
12 Notation of Sections 3 and 4: `d̃` is in the normal form (3.1) with `N ≥ 1`,
13 `𝓘 = d̃A_n + x₁^N d̃A_n` is `Section3.idealI k N d̃`, `Q = A_n/𝓘`,
14 `M = gr Q` is `Section3.grQ k N d̃` (the same type as Section 4's `GrQ`), and
15 `Char(Q) = V(Ann_{k[x,ξ]} M)` is `Section2.Char`.  Lean indexes the
16 coordinates from `0`, so paper `n` is Lean `n + 1` and `x₁` is
17 `X (Sum.inl 0)`.
18 -/

33 /-- The symbol `x₁ ∈ k[x, ξ]`. -/
34 abbrev x₁Symbol (k : Type u) [Field k] (n : ℕ) : SymbolRing k (n + 1) :=
35   MvPolynomial.X (.inl (0 : Fin (n + 1)))
36 
37 /-- `V(x₁) ⊆ Spec k[x, ξ]`, the primes containing `x₁`. -/
38 abbrev Vx₁ (k : Type u) [Field k] (n : ℕ) : Set (CotangentBundle k (n + 1)) :=
39   V (Ideal.span {x₁Symbol k n})
40 
41 /-- `M = gr Q` is finite over `k[x, ξ]`: the filtration `Q_m = qF_m` is good
42 (Definition 2.4; Proposition 3.6). -/
43 theorem step_6_1_finite {n N : ℕ} (dt : PresentedWeyl k (n + 1)) :
44     Module.Finite (SymbolRing k (n + 1)) (grQ k N dt) :=
45   grCyclic_finite_of_good k (idealI k N dt)
46 
47 /-- Since `M` is finite, `Supp(M/x₁M) = Supp(M) ∩ V(x₁)` (Stacks 00L3). -/
48 theorem step_6_1_support_quotient {n N : ℕ} (dt : PresentedWeyl k (n + 1)) :
49     Module.support (SymbolRing k (n + 1)) (QuotSMulTop (x₁Symbol k n) (grQ k N dt)) =
50       Module.support (SymbolRing k (n + 1)) (grQ k N dt) ∩ Vx₁ k n := by
51   have := step_6_1_finite (N := N) dt
52   rw [Module.support_quotSMulTop, Vx₁, V, Ideal.span, PrimeSpectrum.zeroLocus_span]
53 
54 /-- By Corollary 4.11, `M/x₁M = 0`, so `Supp(M) ∩ V(x₁) = ∅`. -/
55 theorem step_6_1_support_avoids [CharZero k] {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
56     (hN : 0 < N) (hform : IsMonicNormalForm k N dt) :
57     Module.support (SymbolRing k (n + 1)) (grQ k N dt) ∩ Vx₁ k n = ∅ := by
58   have hU := Stafford38.Paper.Section4.corollary_4_11_quotSMulTop (k := k) hN hform
59   rw [← step_6_1_support_quotient]
60   exact Module.support_eq_empty
61 
62 /-- Since `M` is finitely generated, `Char(Q) = V(Ann M) = Supp(M)`
63 (Stacks 00L2). -/
64 theorem step_6_1_char_eq_support {n N : ℕ} (dt : PresentedWeyl k (n + 1)) :
65     Section2.Char k (idealI k N dt) = Module.support (SymbolRing k (n + 1)) (grQ k N dt) := by
66   have := step_6_1_finite (N := N) dt
67   rw [Section2.Char_eq, Module.support_eq_zeroLocus]
68 
69 /-- **Proposition 6.1.** `Char(Q) ∩ V(x₁) = ∅`, where `V(x₁) ⊆ Spec k[x, ξ]`
70 is the set of primes containing `x₁`. -/
71 theorem proposition_6_1 [CharZero k] {n N : ℕ} {dt : PresentedWeyl k (n + 1)}
72     (hN : 0 < N) (hform : IsMonicNormalForm k N dt) :
73     Section2.Char k (idealI k N dt) ∩ Vx₁ k n = ∅ := by
74   rw [step_6_1_char_eq_support]
75   exact step_6_1_support_avoids hN hform
