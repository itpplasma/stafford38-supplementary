10 /-!
11 # Lemma 6.2: base change of `Q` to a field extension `K/k`
12 
13 `A_n(k) → A_n(K)` is `presentedWeylScalarExtension` (written `ι`) and
14 `k[x, ξ] → K[x, ξ]` is `symbolScalarExtension` (written `σ`, the coefficient
15 map `MvPolynomial.map (algebraMap k K)`).  For `d̃ ∈ A_n(k)`,
16 `𝓘_K = ι(d̃)A_n(K) + x₁^N ι(d̃)A_n(K)` is `Section3.idealI K N (ι d̃)` and
17 `Q_K = A_n(K)/𝓘_K` is `Section3.Q K N (ι d̃)`.
18 
19 The proof follows the printed order: `𝓘_K = 𝓘 ⊗_k K` (`step_6_2_ideal`) gives
20 `θ : K ⊗_k Q ≅ Q_K` (`theta62`) and `Q_{K,i} = Q_i ⊗_k K` (`step_6_2_pieces`);
21 flatness gives `Q_{K,i}/Q_{K,i-1} = (Q_i/Q_{i-1}) ⊗_k K`
22 (`step_6_2_graded_pieces`), and summing over `i`, `gr Q_K = gr Q ⊗_k K`
23 (`step_6_2_sum`, `step_6_2_graded`), compatibly with `k[x,ξ] → K[x,ξ]`
24 (`step_6_2_graded_one_tmul_smul`); then the annihilator commutes with the base
25 change (`step_6_2_annihilator`), and `Q = 0 ⇔ Q_K = 0` (`step_6_2_vanishing`).
26 -/

184 /-- `K ⊗_k M ≅ M ⊗ K ⊆ K ⊗_k X`: `K` is flat over `k`. -/
185 def baseChangeSubmoduleEquiv (M : Submodule k X) :

191 /-- **Flatness of `K/k` on a pair `M' ⊆ M`.**  If `θ : K ⊗_k X ≅ Y` carries
192 `K ⊗_k M` onto `P` and `K ⊗_k M'` onto `P'`, then
193 `K ⊗_k (M/M') ≅ P/P'`. -/
194 def tensorPieceEquiv {Y : Type*} [AddCommGroup Y] [Module K Y] (θ : K ⊗[k] X ≃ₗ[K] Y)
195     (M M' : Submodule k X) (h : M' ≤ M) (P P' : Submodule K Y)
196     (hP : (M.baseChange K).map θ.toLinearMap = P)
197     (hP' : (M'.baseChange K).map θ.toLinearMap = P') :
198     K ⊗[k] (M ⧸ M'.comap M.subtype) ≃ₗ[K] (P ⧸ P'.comap P.subtype) :=
199   let e : K ⊗[k] M ≃ₗ[K] P := (baseChangeSubmoduleEquiv M).trans
200     ((θ.submoduleMap _).trans (LinearEquiv.ofEq _ _ hP))
201   have he : ∀ t, (e t : Y) = θ (M.subtype.baseChange K t) := fun _ => rfl
202   (TensorProduct.AlgebraTensorModule.tensorQuotientEquiv K k K (M'.comap M.subtype)).trans
203     (Submodule.Quotient.equiv _ _ e (by
204       apply le_antisymm
205       · rintro _ ⟨t, ⟨s, rfl⟩, rfl⟩
206         rw [Submodule.mem_comap, Submodule.coe_subtype, LinearEquiv.coe_coe, he, ← hP']
207         refine ⟨_, ?_, rfl⟩
208         induction s using TensorProduct.induction_on with
209         | zero => simp
210         | tmul c x => exact Submodule.tmul_mem_baseChange_of_mem c x.2
211         | add a b ha hb =>
212             rw [map_add, map_add]
213             exact Submodule.add_mem _ ha hb
214       · have hR : ∀ w ∈ M'.baseChange K, ∃ t ∈ LinearMap.range
215             (TensorProduct.AlgebraTensorModule.lTensor K K
216               ((M'.comap M.subtype).subtype.restrictScalars k)),
217             M.subtype.baseChange K t = w := by
218           intro w hw
219           rw [Submodule.baseChange_eq_span] at hw
220           induction hw using Submodule.span_induction with
221           | mem x hx =>
222               obtain ⟨m', hm', rfl⟩ := hx
223               exact ⟨(1 : K) ⊗ₜ[k] (⟨m', h hm'⟩ : M),
224                 ⟨(1 : K) ⊗ₜ[k] ⟨⟨m', h hm'⟩, hm'⟩, rfl⟩, rfl⟩
225           | zero => exact ⟨0, Submodule.zero_mem _, map_zero _⟩
226           | add x y _ _ hx hy =>
227               obtain ⟨tx, htx, rfl⟩ := hx
228               obtain ⟨ty, hty, rfl⟩ := hy
229               exact ⟨tx + ty, Submodule.add_mem _ htx hty, map_add _ _ _⟩
230           | smul c x _ hx =>
231               obtain ⟨tx, htx, rfl⟩ := hx
232               exact ⟨c • tx, Submodule.smul_mem _ c htx, map_smul _ _ _⟩
233         rintro z hz
234         rw [Submodule.mem_comap, Submodule.coe_subtype, ← hP'] at hz
235         obtain ⟨w, hw, hwz⟩ := hz
236         obtain ⟨t, ht, htw⟩ := hR w hw
237         refine ⟨t, ht, Subtype.ext ?_⟩
238         rw [LinearEquiv.coe_coe, he, htw]
239         exact hwz))
