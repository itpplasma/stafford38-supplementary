10 /-!
11 # `A_n(k)` is a right Ore domain
12 
13 The proof of Corollary 6.3 cites [Björk, Prop. 1.8.4]: for nonzero `d₁, d₂`
14 there are `r₁, r₂` with `d₁r₁ = d₂r₂ ≠ 0`.  We prove it by the growth argument:
15 if `d₁A ∩ d₂A = 0`, then `(x, y) ↦ d₁x + d₂y` embeds `B_m ⊕ B_m` into `B_{m+N}`
16 (`N` bounding the Bernstein degrees of `d₁, d₂`), so `dim B_{kN} ≥ 2^k`, while
17 `dim B_m ≤ (m + 1)^{2n}` grows polynomially.
18 -/

34 theorem card_bounded_exponents_le {σ : Type*} [Fintype σ] (m : ℕ) (S : Set (σ →₀ ℕ))
35     (hS : ∀ s ∈ S, ∀ i, s i ≤ m) :
36     Finite S ∧ Nat.card S ≤ (m + 1) ^ Fintype.card σ := by
37   classical
38   let f : S → (σ → Fin (m + 1)) := fun s i => ⟨s.1 i, Nat.lt_succ_of_le (hS s.1 s.2 i)⟩
39   have hf : Function.Injective f := by
40     intro s t hst
41     apply Subtype.ext
42     ext i
43     exact congrArg Fin.val (congrFun hst i)
44   refine ⟨Finite.of_injective f hf, ?_⟩
45   calc Nat.card S ≤ Nat.card (σ → Fin (m + 1)) := Nat.card_le_card_of_injective f hf
46     _ = (m + 1) ^ Fintype.card σ := by
47       rw [Nat.card_eq_fintype_card, Fintype.card_fun, Fintype.card_fin]

49 /-- `B_m` is finite-dimensional with `dim B_m ≤ (m + 1)^{2n}`. -/
50 theorem bernsteinPiece_finite_finrank_le (n m : ℕ) :
51     Module.Finite k (bernsteinPiece k n m) ∧
52       Module.finrank k (bernsteinPiece k n m) ≤ (m + 1) ^ Fintype.card (PhaseVar n) := by
53   classical
54   let S : Set (PhaseVar n →₀ ℕ) := {s | ∀ i, s i ≤ m}
55   obtain ⟨hSfin, hScard⟩ := card_bounded_exponents_le m S fun s hs => hs
56   let P := MvPolynomial.restrictSupport k S
57   let b := MvPolynomial.basisRestrictSupport k S
58   have hPfin : Module.Finite k P := Module.Finite.of_basis b
59   let e := presentedNormalFormLinearEquiv k n
60   have hmem : ∀ x : bernsteinPiece k n m, e x ∈ P := by
61     intro x
62     rw [MvPolynomial.mem_restrictSupport_iff]
63     intro s hs i
64     have h := x.2 s (MvPolynomial.mem_support_iff.mp hs)
65     simp only [monomialWeight, bernsteinWeight, mul_one, Finsupp.sum] at h
66     by_cases hi : i ∈ s.support
67     · exact (Finset.single_le_sum (fun j _ => Nat.zero_le (s j)) hi).trans h
68     · rw [Finsupp.notMem_support_iff.mp hi]
69       exact Nat.zero_le m
70   let φ : bernsteinPiece k n m →ₗ[k] P :=
71     LinearMap.codRestrict P (e.toLinearMap.comp (bernsteinPiece k n m).subtype) hmem
72   have hφ : Function.Injective φ := by
73     intro x y hxy
74     apply Subtype.ext
75     exact e.injective (congrArg Subtype.val hxy)
76   have hfin : Module.Finite k (bernsteinPiece k n m) := Module.Finite.of_injective φ hφ
77   refine ⟨hfin, (LinearMap.finrank_le_finrank_of_injective hφ).trans ?_⟩
78   rw [Module.finrank_eq_nat_card_basis b]
79   exact hScard

81 /-- If `aA ∩ bA = 0` for nonzero `a, b ∈ B_N`, then `aB_m ⊕ bB_m ⊆ B_{N+m}`,
82 so `2 dim B_m ≤ dim B_{N+m}`. -/
83 theorem two_mul_finrank_le {n : ℕ} {a b : PresentedWeyl k n} (ha : a ≠ 0) (hb : b ≠ 0)
84     (hab : ∀ x y : PresentedWeyl k n, a * x = b * y → a * x = 0) {N : ℕ}
85     (haN : a ∈ bernsteinPiece k n N) (hbN : b ∈ bernsteinPiece k n N) (m : ℕ) :
86     2 * Module.finrank k (bernsteinPiece k n m) ≤
87       Module.finrank k (bernsteinPiece k n (N + m)) := by
88   have hBm := (bernsteinPiece_finite_finrank_le (k := k) n m).1
89   have hBNm := (bernsteinPiece_finite_finrank_le (k := k) n (N + m)).1
90   let La := LinearMap.mulLeft k a
91   let Lb := LinearMap.mulLeft k b
92   have hLa : Function.Injective La := fun x y hxy => by
93     by_contra hne
94     exact Stafford38.WeylDomain.mul_ne_zero ha (sub_ne_zero.mpr hne)
95       (by rw [mul_sub]; exact sub_eq_zero.mpr hxy)
96   have hLb : Function.Injective Lb := fun x y hxy => by
97     by_contra hne
98     exact Stafford38.WeylDomain.mul_ne_zero hb (sub_ne_zero.mpr hne)
99       (by rw [mul_sub]; exact sub_eq_zero.mpr hxy)
100   let U := (bernsteinPiece k n m).map La
101   let V := (bernsteinPiece k n m).map Lb
102   have hU : Module.finrank k U = Module.finrank k (bernsteinPiece k n m) :=
103     (LinearEquiv.finrank_eq (Submodule.equivMapOfInjective La hLa _)).symm
104   have hV : Module.finrank k V = Module.finrank k (bernsteinPiece k n m) :=
105     (LinearEquiv.finrank_eq (Submodule.equivMapOfInjective Lb hLb _)).symm
106   have hUfin : FiniteDimensional k U := Module.Finite.map _ _
107   have hVfin : FiniteDimensional k V := Module.Finite.map _ _
108   have hinf : U ⊓ V = ⊥ := by
109     rw [eq_bot_iff]
110     rintro z ⟨⟨x, -, rfl⟩, ⟨y, -, hy⟩⟩
111     exact hab x y hy.symm
112   have hsup : U ⊔ V ≤ bernsteinPiece k n (N + m) := sup_le
113     (by rintro _ ⟨x, hx, rfl⟩; exact mul_mem_bernsteinPiece k haN hx)
114     (by rintro _ ⟨y, hy, rfl⟩; exact mul_mem_bernsteinPiece k hbN hy)
115   have h1 := Submodule.finrank_sup_add_finrank_inf_eq U V
116   rw [hinf, finrank_bot, add_zero, hU, hV] at h1
117   have h2 := Submodule.finrank_mono hsup
118   omega

121 theorem exists_pow_lt_two_pow (N r : ℕ) : ∃ j : ℕ, (j * N + 1) ^ r < 2 ^ j := by
122   have h := (tendsto_pow_const_div_const_pow_of_one_lt r (by norm_num : (1 : ℝ) < 2)).const_mul
123     (((N : ℝ) + 1) ^ r)
124   rw [mul_zero] at h
125   obtain ⟨j, hj⟩ := ((h.eventually (gt_mem_nhds zero_lt_one)).and
126     (Filter.eventually_ge_atTop 1)).exists
127   obtain ⟨hlt, hj1⟩ := hj
128   refine ⟨j, ?_⟩
129   have hpos : (0 : ℝ) < 2 ^ j := by positivity
130   have hle : ((j * N + 1 : ℕ) : ℝ) ≤ ((N : ℝ) + 1) * j := by
131     have : (1 : ℝ) ≤ j := by exact_mod_cast hj1
132     push_cast
133     nlinarith
134   have hlt' : ((N : ℝ) + 1) ^ r * (j : ℝ) ^ r < 2 ^ j := by
135     rw [← mul_div_assoc, div_lt_one hpos] at hlt
136     exact hlt
137   have hpow : ((j * N + 1 : ℕ) : ℝ) ^ r ≤ ((N : ℝ) + 1) ^ r * (j : ℝ) ^ r := by
138     rw [← mul_pow]
139     exact pow_le_pow_left₀ (by positivity) hle r
140   exact_mod_cast hpow.trans_lt hlt'

153 /-- **`A_n(k)` is a right Ore domain** [Björk, Prop. 1.8.4]: nonzero `d₁, d₂`
154 have `r₁, r₂` with `d₁r₁ = d₂r₂ ≠ 0`. -/
155 theorem weyl_rightOreCondition (n : ℕ) :
156     AlgebraicAnalysis.OreRightIntersection.RightOreCondition (PresentedWeyl k n) := by
157   intro a b ha hb
158   by_contra hcon
159   have hab : ∀ x y : PresentedWeyl k n, a * x = b * y → a * x = 0 := by
160     intro x y hxy
161     by_contra hne
162     exact hcon ⟨x, y, hxy, hne⟩
163   obtain ⟨Na, hNa⟩ := exists_mem_bernsteinPiece k a
164   obtain ⟨Nb, hNb⟩ := exists_mem_bernsteinPiece k b
165   let N := max Na Nb
166   have haN : a ∈ bernsteinPiece k n N :=
167     presentedWeightPiece_mono k bernsteinWeight (le_max_left Na Nb) hNa
168   have hbN : b ∈ bernsteinPiece k n N :=
169     presentedWeightPiece_mono k bernsteinWeight (le_max_right Na Nb) hNb
170   have h10 : (1 : PresentedWeyl k n) ≠ 0 := fun h => ha (by rw [← mul_one a, h, mul_zero])
171   have hgrow : ∀ j : ℕ, 2 ^ j ≤ Module.finrank k (bernsteinPiece k n (j * N)) := by
172     intro j
173     induction j with
174     | zero =>
175         have hfin := (bernsteinPiece_finite_finrank_le (k := k) n 0).1
176         rw [pow_zero, zero_mul, Nat.one_le_iff_ne_zero]
177         intro h0
178         rw [Submodule.finrank_eq_zero] at h0
179         have h1 := one_mem_bernsteinPiece_zero (k := k) n
180         rw [h0, Submodule.mem_bot] at h1
181         exact h10 h1
182     | succ j ih =>
183         have h := two_mul_finrank_le ha hb hab haN hbN (j * N)
184         rw [show (j + 1) * N = N + j * N by ring, pow_succ]
185         omega
186   obtain ⟨j, hj⟩ := exists_pow_lt_two_pow N (Fintype.card (PhaseVar n))
187   have hle := (bernsteinPiece_finite_finrank_le (k := k) n (j * N)).2
188   have := hgrow j
189   omega
