39 /-- **[Sta78, Thm. 3.7]** (right-module version), as cited in the proof of
40 Corollary 6.3: every finitely generated torsion right `A_n(k)`-module `T` is
41 generated by two elements, and is stably two-generated (for `l ≥ 2` and
42 generators `a_1, …, a_{l+1}` there are `f_i` with `a_i + a_{l+1}f_i`,
43 `i ≤ l`, generating `T`).  Proved here from Theorem 1.1, which does not use
44 Corollary 6.3: `two_generators_of_identity` and
45 `stably_generated_of_identity`. -/
46 theorem stafford_1978_theorem_3_7 (k : Type u) [Field k] [CharZero k] (n : ℕ)
47     (T : Type v) [AddCommGroup T] [Module (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ T]
48     [Module.Finite (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ T]
49     (hT : Section2.IsTorsion (Stafford38Challenge.WeylAlg k n) T) :
50     (∃ t₁ t₂ : T, Submodule.span (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ ({t₁, t₂} : Set T) = ⊤) ∧
51       IsStablyGenerated (Stafford38Challenge.WeylAlg k n) T 2 :=
52   ⟨two_generators_of_identity (weyl_rightOreCondition (k := k) n)
53       (fun d hd => Stafford38.Paper.theorem_1_1 k n d hd) hT,
54     fun l hl => stably_generated_of_identity (weyl_rightOreCondition (k := k) n)
55       (fun d hd => Stafford38.Paper.theorem_1_1 k n d hd) hT l (by omega)⟩
