16 # Section 5.2 of the paper: Theorem 5.7 and its lemmas
17 
18 The paper's lemmas are stated and proved in
19 `Stafford38.Paper.Section5.Place` (Lemmas 5.9, 5.10),
20 `Stafford38.Paper.Section5.ResidueBasis` (Lemma 5.11),
21 `Stafford38.Paper.Section5.TangentCone` and
22 `Stafford38.Paper.Section5.GenericTangentRank` (Lemmas 5.12, 5.13).
23 
24 The paragraphs of the proof of Theorem 5.7(i) between those lemmas are stated
25 and proved in `Stafford38.Paper.Section5.Steps`; the theorems
26 `witness_lemma_*` there apply Lemmas 5.11–5.13 to the discrete valuation ring
27 and the Laurent column `q(t)` of the proof.
28 `GeneralDivisorialVisibleFrame.generalDivisorialVisibleFrameWithKaehlerFinite`
29 builds the place `V` of Lemmas 5.9–5.10 with `Ω_{V/K}` finitely generated, `Stafford38.Paper.Section5.Completion`
30 states the paragraph `V̂ ≅ κ[[t]]` and the bound of Lemma 5.12(a) at `V`, and
31 `Stafford38.Paper.Section5.KaehlerRankBound` proves that bound for any discrete
32 valuation ring.  `Stafford38.Paper.Section5.Proof` chains the paragraphs into
33 `theorem_5_7_i_paper`, which `theorem_5_7_i` below restates; `theorem_5_7_ii`
34 follows from it by `step_dir_of_fibre`.  The terminal
35 theorem in `GeneralAsymptoticConormal` proves the same statement by a second,
36 chart-based route and is kept unchanged.

61 /-- `G(a ξ) = a^{deg G} G(ξ)` for a homogeneous `G`. -/
62 theorem eval_smul_of_isHomogeneous {k : Type*} [Field k] {n : ℕ}
63     (G : MvPolynomial (Fin n) k) (d : ℕ) (hG : G.IsHomogeneous d)
64     (a : k) (v : Fin n → k) :
65     MvPolynomial.eval (a • v) G = a ^ d * MvPolynomial.eval v G := by
66   simp only [MvPolynomial.eval_eq']
67   rw [Finset.mul_sum]
68   apply Finset.sum_congr rfl
69   intro s hs
70   have hd : ∑ i, s i = d := by
71     rw [← s.sum_fintype (fun _ c => c) (fun _ => rfl)]
72     simpa [MvPolynomial.IsHomogeneous, Finsupp.weight_apply] using
73       hG (MvPolynomial.mem_support_iff.mp hs)
74   simp only [Pi.smul_apply, smul_eq_mul, mul_pow, Finset.prod_mul_distrib]
75   rw [Finset.prod_pow_eq_pow_sum, hd]
76   ring

78 /-- **Proof of Theorem 5.7(ii) from (i).**  Let every polynomial vanishing on
79 `N(Y)` vanish at `ξ ≠ 0`, and let `G` be homogeneous of degree `d` and vanish on
80 all classes `[ξ']` with `0 ≠ ξ' ∈ N(Y)`.  By homogeneity `G` vanishes on every
81 nonzero `ξ' ∈ N(Y)`, and at `0` if `d > 0`, hence on `N(Y)`; so `G(ξ) = 0`, and
82 `G` vanishes at every representative `a ξ` of `[ξ]`.  If `d = 0`, `G` is a
83 constant vanishing at a class of `N(Y)`, so `G = 0`; that such a class exists
84 follows from the hypothesis itself, applied to a coordinate `ξ_i ≠ 0`. -/
85 theorem step_dir_of_fibre {k : Type*} [Field k] [IsAlgClosed k] {n : ℕ}
86     (I : Ideal (MvPolynomial (Fin n) k)) (ξ : Fin n → k) (hξ : ξ ≠ 0)
87     (hvan : ξ ∈ MvPolynomial.zeroLocus k
88       (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I))) :
89     Projectivization.mk k ξ hξ ∈
90       projectiveHomogeneousClosure
91         (projectivizedDirectionSet (smoothConormalDirectionSet I)) := by
92   intro G d hG hT
93   -- `G` vanishes on every nonzero `ξ' ∈ N(Y)`.
94   have hnonzero : ∀ v ∈ smoothConormalFibreProjection I, v ≠ 0 →
95       MvPolynomial.eval v G = 0 := by
96     intro v hv hv0
97     rcases hv with ⟨y, hy, hc⟩
98     exact hT ⟨v, ⟨y, hy, hv0, hc⟩, hv0, rfl⟩ v hv0 rfl
99   have hξG : MvPolynomial.eval ξ G = 0 := by
100     rcases Nat.eq_zero_or_pos d with hd | hd
101     · -- `d = 0`: `G` is constant, and some nonzero `ξ' ∈ N(Y)` exists.
102       subst hd
103       have hconst : ∀ x : Fin n → k, MvPolynomial.eval x G = MvPolynomial.eval 0 G := by
104         intro x
105         have h := eval_smul_of_isHomogeneous G 0 hG 0 x
106         rw [zero_smul, pow_zero, one_mul] at h
107         exact h.symm
108       obtain ⟨i, hi⟩ : ∃ i, ξ i ≠ 0 := by simpa [funext_iff] using hξ
109       obtain ⟨v, hv, hv0⟩ : ∃ v ∈ smoothConormalFibreProjection I, v ≠ 0 := by
110         by_contra hnone
111         push Not at hnone
112         apply hi
113         have hXi : MvPolynomial.X i ∈ MvPolynomial.vanishingIdeal k
114             (smoothConormalFibreProjection I) := by
115           rw [MvPolynomial.mem_vanishingIdeal_iff]
116           intro v hv
117           rw [hnone v hv]
118           simp
119         simpa using (MvPolynomial.mem_zeroLocus_iff.mp hvan) _ hXi
120       rw [hconst, ← hconst v]
121       exact hnonzero v hv hv0
122     · -- `d > 0`: `G(0) = 0` too, so `G` vanishes on `N(Y)`.
123       have hGvan : G ∈ MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I) := by
124         rw [MvPolynomial.mem_vanishingIdeal_iff]
125         intro v hv
126         by_cases hv0 : v = 0
127         · subst hv0
128           have h := eval_smul_of_isHomogeneous G d hG 0 0
129           rw [smul_zero, zero_pow hd.ne', zero_mul] at h
130           exact h
131         · exact hnonzero v hv hv0
132       exact (MvPolynomial.mem_zeroLocus_iff.mp hvan) _ hGvan
133   -- Every representative `a ξ` of `[ξ]`.
134   intro v hv hz
135   obtain ⟨a, ha⟩ := (Projectivization.mk_eq_mk_iff k ξ v hξ hv).mp hz
136   have hscaled := eval_smul_of_isHomogeneous G d hG (a : k) v
137   have ha' : (a : k) • v = ξ := by
138     rw [← Units.smul_def]
139     exact ha
140   rw [ha', hξG] at hscaled
141   exact (mul_eq_zero.mp hscaled.symm).resolve_left (pow_ne_zero d (Units.ne_zero a))

143 /-- **Theorem 5.7(ii).** `[dx_1] ∈ Dir(Y)`: by `step_dir_of_fibre`, every
144 homogeneous `G` vanishing on the classes `[ξ]`, `0 ≠ ξ ∈ N(Y)`, vanishes on
145 `N(Y)`, hence at `(1, 0, …, 0)` by (i). -/
146 theorem theorem_5_7_ii
147     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
148     {m : ℕ} (hm : 0 < m)
149     (I : PrimeSpectrum (MvPolynomial (Fin m) k))
150     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) :
151     Projectivization.mk k
152         (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0)
153         (by intro h; have := congrFun h ⟨0, hm⟩; simp at this) ∈
154       projectiveHomogeneousClosure
155         (projectivizedDirectionSet (smoothConormalDirectionSet I.asIdeal)) :=
156   step_dir_of_fibre I.asIdeal _ _ (theorem_5_7_i hm I havoid)
