16 # Section 5.2 of the paper: Theorem 5.7 and its lemmas
17 
18 The paper's lemmas are stated and proved in
19 `Stafford38.Paper.Section5.Place` (Lemmas 5.9, 5.10),
20 `Stafford38.Paper.Section5.ResidueBasis` (Lemma 5.11),
21 `Stafford38.Paper.Section5.TangentCone` and
22 `Stafford38.Paper.Section5.GenericTangentRank` (Lemmas 5.12, 5.13).
23 
24 Theorem 5.7 is proved along the paper's route by
25 `Stafford38.Geometry.AlternativeAsymptoticConormal`: the algebraic case, the
26 discrete valuation ring at infinity, the Laurent generic point `y`, the
27 matrix `M(t)` with a unit minor and the lifted row `ℓ(t)` with `ℓ(0) = u`.  The
28 theorems `witness_lemma_*` below show that the discrete valuation ring and the
29 Laurent column `q(t)` used in that proof satisfy Lemmas 5.11–5.13 and both parts
30 of Lemma 5.12(a).  The terminal theorem in `GeneralAsymptoticConormal` proves
31 the same statement by a second, chart-based route and is kept unchanged.
32 -/

95 /-- **Lemma 5.11 for the valuation ring of the formal proof.**  The residues
96 `res q_i` of its normalized column satisfy `res q_0 = res q_1 = 0`,
97 `res q_j = 1`, and some `res q_i` with `i ∉ {0, 1, j}` have differentials

116       residue V (C.q C.chart) = 1 ∧
117       ∃ (d' : ℕ) (rows : Fin d' → Fin (m + 1)),
118         Function.Injective rows ∧
119         (∀ l, rows l ≠ 0 ∧ rows l ≠ Fin.succ ⟨0, hm⟩ ∧ rows l ≠ C.chart) ∧
120         LinearIndependent κ
121           (fun l ↦ KaehlerDifferential.D k κ (residue V (C.q (rows l)))) ∧
122         Submodule.span κ
123           (Set.range fun l ↦ KaehlerDifferential.D k κ (residue V (C.q (rows l)))) = ⊤ ∧
124         d' = Module.finrank κ (Ω[κ⁄k]) := by

160 /-- **Lemmas 5.12(a), 5.12(b) and 5.13 at the Laurent generic point of the
161 formal proof.**  For the column `q(t)` of `witness_lemma_5_11`, with
162 `y = q/q_0 ∈ κ((t))^n`: `dim Θ_y = dim_F Ω_{F/K}`, this equals
163 `trdeg_K F = m`, and `m ≤ d' + 1`; `dim 𝒞 = dim Θ_y + 1`; `δq ∈ 𝒞` for every
164 `K`-derivation `δ` of `κ((t))` and `q ∈ 𝒞`; and every row annihilating `𝒞` has
165 conormal affine part. -/

177       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
178     let κ := ResidueField V

182     letI : SMul k E := (inferInstance : Algebra k E).toSMul
