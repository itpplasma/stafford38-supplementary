14 # Section 5.2 of the paper: Theorem 5.7 and its lemmas
15 
16 The paper's lemmas are stated and proved in
17 `Stafford38.Paper.Section5.Place` (Lemmas 5.9, 5.10),
18 `Stafford38.Paper.Section5.ResidueBasis` (Lemma 5.11),
19 `Stafford38.Paper.Section5.TangentCone` and
20 `Stafford38.Paper.Section5.GenericTangentRank` (Lemmas 5.12, 5.13).
21 
22 The paragraphs of the proof of Theorem 5.7(i) between those lemmas are stated
23 and proved in `Stafford38.Paper.Section5.Steps`; the theorems
24 `witness_lemma_*` there apply Lemmas 5.11–5.13 to the discrete valuation ring
25 and the Laurent column `q(t)` of the proof.
26 `Stafford38.Paper.Section5.Proof` chains the paragraphs into
27 `theorem_5_7_i_paper`, which `theorem_5_7_i` below restates.  The terminal
28 theorem in `GeneralAsymptoticConormal` proves the same statement by a second,
29 chart-based route and is kept unchanged.
30 -/

40 /-- **Theorem 5.7(i).** If `Y = V(𝔭)` is nonempty, irreducible and avoids
41 `H = {x_1 = 0}`, every polynomial vanishing on the conormal cone `N(Y)` over the
42 smooth locus vanishes at `(1, 0, …, 0)`.  Proved paragraph by paragraph in
43 `theorem_5_7_i_paper`. -/
44 theorem theorem_5_7_i
45     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
46     {m : ℕ} (hm : 0 < m)
47     (I : PrimeSpectrum (MvPolynomial (Fin m) k))
48     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) :
49     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
50       MvPolynomial.zeroLocus k
51         (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I.asIdeal)) :=
52   theorem_5_7_i_paper hm I havoid

54 /-- **Theorem 5.7(ii).** `[dx_1] ∈ Dir(Y)`. -/
55 theorem theorem_5_7_ii
56     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
57     {m : ℕ} (hm : 0 < m)
58     (I : PrimeSpectrum (MvPolynomial (Fin m) k))
59     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) :
60     Projectivization.mk k
61         (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0)
62         (by intro h; have := congrFun h ⟨0, hm⟩; simp at this) ∈
63       projectiveHomogeneousClosure
64         (projectivizedDirectionSet (smoothConormalDirectionSet I.asIdeal)) :=
65   mk_mem_projectiveHomogeneousClosure_of_fibre_zeroLocus I.asIdeal _ _
66     (theorem_5_7_i hm I havoid)
