38 /-- **Definition 2.8 (symplectic form).** `ω` is antisymmetric and
39 nondegenerate (`ω(u, ·) = 0` forces `u = 0`). -/
40 def IsSymplectic (ω : BilinForm k V) : Prop :=
41   (∀ x y, ω x y = -ω y x) ∧ ω.Nondegenerate
42 
43 /-- **Definition 2.8 (Gram matrix).** `J = (ω(b_i, b_j))_{ij}`. -/
44 def gramMatrix {ι : Type*} (ω : BilinForm k V) (b : ι → V) : Matrix ι ι k :=
45   fun i j => ω (b i) (b j)
46 
47 /-- **Definition 2.8 (symplectic basis).** `e_1, …, e_n, f_1, …, f_n` with
48 `ω(e_i, f_j) = δ_ij` and `ω(e_i, e_j) = ω(f_i, f_j) = 0`. -/
49 def IsSymplecticBasis {n : ℕ} (ω : BilinForm k V) (b : Basis (Fin n ⊕ Fin n) k V) :
50     Prop :=
51   (∀ i j, ω (b (Sum.inl i)) (b (Sum.inr j)) = if i = j then 1 else 0) ∧
52     (∀ i j, ω (b (Sum.inl i)) (b (Sum.inl j)) = 0) ∧
53     (∀ i j, ω (b (Sum.inr i)) (b (Sum.inr j)) = 0)
54 
55 /-- In characteristic zero an antisymmetric form is alternating. -/
56 theorem alternating_of_antisymmetric [CharZero k] (ω : BilinForm k V)
57     (hanti : ∀ x y, ω x y = -ω y x) (x : V) : ω x x = 0 := by
58   have h := hanti x x
59   have h2 : (2 : k) * ω x x = 0 := by
60     rw [two_mul]
61     nth_rewrite 1 [h]
62     exact neg_add_cancel _
63   exact (mul_eq_zero.mp h2).resolve_left two_ne_zero
64 
65 end Abstract

69 /-- **Definition 2.8 (standard symplectic form).** On `𝕍 = k^{2n}` with basis
70 `x_1, …, x_n, ∂_1, …, ∂_n`, `ω(u, v) = ∑_i (u_{x_i} v_{∂_i} - u_{∂_i} v_{x_i})`,
71 so `ω(x_i, ∂_j) = δ_ij` and `ω(x_i, x_j) = ω(∂_i, ∂_j) = 0`.  It is the
72 negative of the repository's `phasePairing`, which uses the opposite sign
73 convention. -/
74 def omega (n : ℕ) : BilinForm k (PhaseVar n → k) :=
75   LinearMap.mk₂ k (fun u v => -phasePairing k u v)
76     (fun u v w => by rw [phasePairing_add_left, neg_add])
77     (fun c u v => by rw [phasePairing_smul_left, smul_eq_mul, mul_neg])
78     (fun u v w => by rw [phasePairing_add_right, neg_add])
79     (fun c u v => by rw [phasePairing_smul_right, smul_eq_mul, mul_neg])
80 
81 theorem omega_apply {n : ℕ} (u v : PhaseVar n → k) :
82     omega k n u v = -phasePairing k u v := rfl
83 
84 theorem omega_antisymm {n : ℕ} (u v : PhaseVar n → k) :
85     omega k n u v = -omega k n v u := by
86   simp only [omega_apply, phasePairing_skew k u v, neg_neg]
87 
88 theorem omega_nondegenerate (n : ℕ) : (omega k n).Nondegenerate := by
89   constructor
90   · intro x hx
91     by_contra hxne
92     obtain ⟨z, hz⟩ := exists_phasePairing_right_ne_zero k hxne
93     exact hz (neg_eq_zero.mp (hx z))
94   · intro y hy
95     by_contra hyne
96     obtain ⟨z, hz⟩ := exists_phasePairing_left_ne_zero k hyne
97     exact hz (neg_eq_zero.mp (hy z))
98 
99 theorem finrank_phaseSpace (n : ℕ) :
100     Module.finrank k (PhaseVar n → k) = 2 * n := by
101   rw [Module.finrank_eq_card_basis (Pi.basisFun k (PhaseVar n))]
102   simp [two_mul]

104 /-- The Gram matrix of `ω` in the generator basis is `-J`, where `J` is the
105 matrix of the Weyl commutators `[u_i, u_j]`. -/
106 theorem omega_toMatrix (n : ℕ) :
107     LinearMap.BilinForm.toMatrix' (omega k n) = -standardForm k n := by
108   ext i j
109   rw [LinearMap.BilinForm.toMatrix'_apply]
110   have hsingle (i : PhaseVar n) : Pi.single i (1 : k) = phaseBasis k i := by
111     funext l
112     simp [Pi.single_apply, phaseBasis]
113   rw [hsingle, hsingle, omega_apply, phasePairing_basis]
114   rfl
115 
116 /-- The Gram matrix of `ω` in the generator basis `x_1, …, ∂_n` is `-J`, where
117 `J` is the matrix of Weyl commutators `[u_i, u_j]` (`standardForm`). -/
118 theorem gramMatrix_omega (n : ℕ) :
119     gramMatrix (omega k n) (Pi.basisFun k (PhaseVar n)) = -standardForm k n := by
120   rw [← omega_toMatrix]
121   ext i j
122   rw [LinearMap.BilinForm.toMatrix'_apply]
123   simp [gramMatrix]

125 theorem omega_isSymplectic (n : ℕ) : IsSymplectic (omega k n) :=
126   ⟨omega_antisymm k, omega_nondegenerate k n⟩

128 /-- **Definition 2.9 (Poisson bracket).**
129 `{f, g} = ∑_i (∂f/∂x_i ∂g/∂ξ_i - ∂f/∂ξ_i ∂g/∂x_i)` on `k[x, ξ]`. -/
130 def poisson {n : ℕ} (f g : SymbolRing k n) : SymbolRing k n :=
131   ∑ i : Fin n,
132     (MvPolynomial.pderiv (Sum.inl i) f * MvPolynomial.pderiv (Sum.inr i) g -
133       MvPolynomial.pderiv (Sum.inr i) f * MvPolynomial.pderiv (Sum.inl i) g)
134 
135 /-- The printed bracket is the repository's `Characteristic.poissonBracket`,
136 with the same sign. -/
137 theorem poisson_eq {n : ℕ} (f g : SymbolRing k n) : poisson k f g = poissonBracket f g :=
138   rfl

140 /-- **Definition 2.10 (involutive ideal).** `{J, J} ⊆ J`. -/
141 def IsInvolutive {n : ℕ} (J : Ideal (SymbolRing k n)) : Prop :=
142   ∀ f ∈ J, ∀ g ∈ J, poisson k f g ∈ J
143 
144 /-- The printed notion is the repository's
145 `PostScalarExtensionPoisson.IsInvolutive`. -/
146 theorem isInvolutive_iff {n : ℕ} (J : Ideal (SymbolRing k n)) :
147     IsInvolutive k J ↔ Stafford38.Characteristic.PostScalarExtensionPoisson.IsInvolutive J :=
148   Iff.rfl

150 /-- **Definition 2.10 (involutive closed set).** A closed set `W ⊆ K^{2n}` of
151 closed points is involutive if `I(W)` is. -/
152 def IsInvolutiveSet {n : ℕ} (W : Set (PhaseVar n → k)) : Prop :=
153   IsInvolutive k (vanishingIdealOf k W)

155 /-- **Theorem 2.11 (Gabber).** Let `K` be a field of characteristic zero and `M`
156 a finitely generated right `A_n(K)`-module with a good order filtration `M_•`
157 (Definition 2.4, `IsGoodFiltration`: compatible and exhaustive, `gr M` finitely
158 generated over `K[x, ξ]`).  Then `J = √(Ann_{K[x,ξ]} gr M)` is involutive.  The cited result
159 [Gab81] is proved in `Stafford38.Paper.Support.Gabber`, by reduction to cyclic
160 modules and the repository's proof of the cyclic case. -/
161 theorem theorem_2_11 (K : Type u) [Field K] [CharZero K] (n : ℕ)
162     {M : Type w} [AddCommGroup M] [Module (Stafford38Challenge.WeylAlg K n)ᵐᵒᵖ M] [Module K M]
163     [IsScalarTower K (Stafford38Challenge.WeylAlg K n)ᵐᵒᵖ M]
164     [Module.Finite (Stafford38Challenge.WeylAlg K n)ᵐᵒᵖ M]
165     (Γ : ℕ → Submodule K M) (hgood : IsGoodFiltration K n Γ) :
166     IsInvolutive K (Module.annihilator (SymbolRing K n)
167       (Stafford38.Paper.Support.Gabber.GrModule Γ)).radical :=
168   (isInvolutive_iff K _).2 (Stafford38.Paper.Support.Gabber.theorem_2_11_general K n Γ
169     hgood.compatible hgood.finite)
