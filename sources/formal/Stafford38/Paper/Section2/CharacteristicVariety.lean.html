9 # Definitions 2.4 and 2.7 for finitely generated modules
10 
11 For a right `A_n(k)`-module `M` (a `(A_n(k))ᵐᵒᵖ`-module with compatible `k`-structure)
12 and a filtration `M_•` compatible with the order filtration, `gr M` is
13 `Support.Gabber.GrModule M_•`, a graded module over `gr A_n = k[x, ξ]`.
14 
15 * `IsGoodFiltration`: Definition 2.4, good means `gr M` finitely generated over
16   `k[x, ξ]`.
17 * `CharOf`: Definition 2.7, `Char(M) = V(Ann_{k[x,ξ]} gr M)`.
18 * `definition_2_7_independent`: `Char(M)` does not depend on the good filtration
19   [HTT, Lemma D.3.1].  Two good filtrations satisfy `F_j ⊆ G_{j+c}` and
20   `G_j ⊆ F_{j+c'}` (`Support.Gabber.exists_shift_le`), hence the radicals of the
21   annihilators agree (`Support.Gabber.radical_annihilator_eq_of_shift`), and
22   `V(J) = V(√J)`.
23 * `Char`: the special case `M = A_n(k)/I` with `M_m = qF_m` used in Sections 3–6;
24   `Char_eq` is the repository's description `V(Ann gr Q)` with `gr Q = grCyclic`.
25 -/

44 /-- **Definition 2.4 (good filtration).** A filtration `M_•` of a right
45 `A_n(k)`-module `M`, compatible with the order filtration `F_•`, is good if
46 `gr M = ⊕_i M_i/M_{i-1}` is finitely generated over `gr^F A_n = k[x, ξ]`. -/
47 structure IsGoodFiltration (Γ : ℕ → Submodule k M) : Prop where
48   compatible : IsCompatibleFiltration (orderFiltration k n) Γ
49   finite : Module.Finite (SymbolRing k n) (GrModule Γ)
50 
51 /-- **Definition 2.7 (characteristic variety).** For a finitely generated
52 `A_n(k)`-module `M` with a good order filtration `M_•`,
53 `Char(M) = V(Ann_{k[x,ξ]} gr M) ⊆ T^*X = Spec k[x, ξ]`. -/
54 def CharOf (Γ : ℕ → Submodule k M) : Set (CotangentBundle k n) :=
55   V (Module.annihilator (SymbolRing k n) (GrModule Γ))
56 
57 variable {k n}
58 
59 /-- **Definition 2.7, last sentence.** `Char(M)` is independent of the choice of
60 the good filtration [HTT, Lemma D.3.1]. -/
61 theorem definition_2_7_independent {Γ Γ' : ℕ → Submodule k M}
62     (hΓ : IsGoodFiltration k n Γ) (hΓ' : IsGoodFiltration k n Γ') :
63     CharOf k n Γ = CharOf k n Γ' := by
64   rw [CharOf, CharOf, V, V, ← PrimeSpectrum.zeroLocus_radical,
65     radical_annihilator_eq_of_good hΓ.compatible hΓ'.compatible hΓ.finite hΓ'.finite,
66     PrimeSpectrum.zeroLocus_radical]
67 
68 end General

72 variable (k : Type u) [Field k] {n : ℕ}

74 /-- **Definition 2.4, example.** For `Q = A_n(k)/I`, the filtration `Q_m = qF_m`
75 induced by the generator `q = [1]` is good. -/
76 theorem qFilt_isGoodFiltration (I : RightIdeal (PresentedWeyl k n)) :
77     IsGoodFiltration k n (qFilt k I) :=
78   ⟨cyclicFiltration_order_isCompatible k I, qFilt_good k I⟩
79 
80 /-- `Char(A_n(k)/I)`, Definition 2.7 for `M = A_n(k)/I` with its good filtration
81 `Q_m = qF_m`; by `definition_2_7_independent` any good filtration gives the same set
82 (`Char_eq_charOf`). -/
83 def Char (I : RightIdeal (PresentedWeyl k n)) : Set (CotangentBundle k n) :=
84   CharOf k n (qFilt k I)
85 
86 /-- `Char(A_n/I) = V(Ann_{k[x,ξ]} gr Q)` with `gr Q` the repository's `grCyclic`. -/
87 theorem Char_eq (I : RightIdeal (PresentedWeyl k n)) :
88     Char k I = V (Module.annihilator (SymbolRing k n) (grCyclic k I)) := by
89   rw [Char, CharOf, annihilator_qFilt]
90 
91 /-- `Char(A_n/I)` computed from any good filtration of `A_n/I`. -/
92 theorem Char_eq_charOf (I : RightIdeal (PresentedWeyl k n))
93     {Γ : ℕ → Submodule k (RightQuotient I)} (hΓ : IsGoodFiltration k n Γ) :
94     Char k I = CharOf k n Γ :=
95   definition_2_7_independent (qFilt_isGoodFiltration k I) hΓ
96 
97 /-- `gr Q` is finitely generated over `k[x, ξ]` because `Q_m = qF_m` is good. -/
98 theorem grCyclic_finite_of_good (I : RightIdeal (PresentedWeyl k n)) :
99     Module.Finite (SymbolRing k n) (grCyclic k I) :=
100   have := (qFilt_isGoodFiltration k I).finite
101   Module.Finite.equiv (grCyclicEquivS k I).symm
102 
103 end Cyclic
