36 /-- **Definition 2.6 (vanishing locus).** `V(J) = {𝔭 ∈ Spec S : 𝔭 ⊇ J}`, the
37 Mathlib `PrimeSpectrum.zeroLocus`. -/
38 abbrev V (J : Ideal S) : Set (PrimeSpectrum S) :=
39   PrimeSpectrum.zeroLocus (J : Set S)

41 theorem mem_V_iff (J : Ideal S) (P : PrimeSpectrum S) : P ∈ V J ↔ J ≤ P.asIdeal :=
42   PrimeSpectrum.mem_zeroLocus P J

50 /-- **Definition 2.6 (classical zero set).**
51 `V_K(J) = {a ∈ K^N : f(a) = 0 ∀ f ∈ J}`. -/
52 def VK (J : Ideal (MvPolynomial σ K)) : Set (σ → K) :=
53   {a | ∀ f ∈ J, MvPolynomial.eval a f = 0}
54 
55 theorem VK_eq_zeroLocus (J : Ideal (MvPolynomial σ K)) :
56     VK K J = MvPolynomial.zeroLocus K J := by
57   ext a
58   simp [VK, MvPolynomial.aeval_eq_eval]
59 
60 /-- **Definition 2.6 (vanishing ideal).** `I(S) = {f : f(a) = 0 ∀ a ∈ S}`. -/
61 def vanishingIdealOf (T : Set (σ → K)) : Ideal (MvPolynomial σ K) :=
62   MvPolynomial.vanishingIdeal K T
63 
64 theorem mem_vanishingIdealOf (T : Set (σ → K)) (f : MvPolynomial σ K) :
65     f ∈ vanishingIdealOf K T ↔ ∀ a ∈ T, MvPolynomial.eval a f = 0 := by
66   simp [vanishingIdealOf, MvPolynomial.mem_vanishingIdeal_iff, MvPolynomial.aeval_eq_eval]
67 
68 /-- **Definition 2.6.** `𝔪_a = (z_1 - a_1, …, z_N - a_N)`. -/
69 def maxIdealAt (a : σ → K) : Ideal (MvPolynomial σ K) :=
70   Ideal.span (Set.range fun i => MvPolynomial.X i - MvPolynomial.C (a i))
71 
72 /-- `f ∈ 𝔪_a` iff `f(a) = 0`. -/
73 theorem mem_maxIdealAt_iff (a : σ → K) (f : MvPolynomial σ K) :
74     f ∈ maxIdealAt K a ↔ MvPolynomial.eval a f = 0 := by
75   constructor
76   · intro hf
77     refine Submodule.span_induction ?_ (by simp) (fun x y _ _ hx hy => by simp [hx, hy])
78       (fun r x _ hx => by simp [hx]) hf
79     rintro _ ⟨i, rfl⟩
80     simp
81   · intro hf
82     let q := Ideal.Quotient.mk (maxIdealAt K a)
83     have hred : ∀ g : MvPolynomial σ K, q g = q (MvPolynomial.C (MvPolynomial.eval a g)) := by
84       intro g
85       induction g using MvPolynomial.induction_on with
86       | C c => simp
87       | add g h hg hh => rw [map_add, hg, hh, map_add, map_add, map_add]
88       | mul_X g i hg =>
89           have hX : q (MvPolynomial.X i) = q (MvPolynomial.C (a i)) := by
90             rw [Ideal.Quotient.eq]
91             exact Ideal.subset_span ⟨i, rfl⟩
92           rw [map_mul, hg, hX, ← map_mul, ← map_mul]
93           simp
94     have h0 := hred f
95     rw [hf, map_zero, map_zero] at h0
96     exact Ideal.Quotient.eq_zero_iff_mem.mp h0
97 
98 theorem maxIdealAt_eq_vanishingIdeal (a : σ → K) :
99     maxIdealAt K a = MvPolynomial.vanishingIdeal K {a} := by
100   ext f
101   rw [mem_maxIdealAt_iff, MvPolynomial.mem_vanishingIdeal_singleton_iff,
102     MvPolynomial.aeval_eq_eval]
103 
104 instance (a : σ → K) : (maxIdealAt K a).IsMaximal := by
105   rw [maxIdealAt_eq_vanishingIdeal]
106   infer_instance
107 
108 /-- `𝔪_a` as a point of `Spec K[z]`. -/
109 def closedPoint (a : σ → K) : PrimeSpectrum (MvPolynomial σ K) :=
110   ⟨maxIdealAt K a, Ideal.IsMaximal.isPrime inferInstance⟩
111 
112 /-- **Definition 2.6, last sentence.** `a ∈ V_K(J)` iff `𝔪_a ∈ V(J)`. -/
113 theorem mem_VK_iff_closedPoint_mem_V (J : Ideal (MvPolynomial σ K)) (a : σ → K) :
114     a ∈ VK K J ↔ closedPoint K a ∈ V J := by
115   rw [mem_V_iff]
116   change (∀ f ∈ J, MvPolynomial.eval a f = 0) ↔ J ≤ maxIdealAt K a
117   exact ⟨fun h f hf => (mem_maxIdealAt_iff K a f).mpr (h f hf),
118     fun h f hf => (mem_maxIdealAt_iff K a f).mp (h hf)⟩
119 
120 /-- **Theorem 2.12 (Hilbert's Nullstellensatz).** Let `K` be algebraically
121 closed and `J ⊆ K[z_1, …, z_N]` an ideal (the variables indexed by any
122 finite type `σ`, e.g. `PhaseVar n` for `K[x, ξ]`).
123 (a) The maximal ideals of `K[z]` are exactly the `𝔪_a`, `a ↦ 𝔪_a` is
124 injective, and `V_K(J)` corresponds to the closed points of `V(J)`;
125 (b) `V_K(J) = ∅` iff `1 ∈ J`;
126 (c) `I(V_K(J)) = √J`. -/
127 theorem theorem_2_12 [IsAlgClosed K] [Finite σ] (J : Ideal (MvPolynomial σ K)) :
128     ((∀ M : Ideal (MvPolynomial σ K), M.IsMaximal ↔ ∃ a, M = maxIdealAt K a) ∧
129       Function.Injective (maxIdealAt K (σ := σ)) ∧
130       ∀ a, a ∈ VK K J ↔ closedPoint K a ∈ V J) ∧
131     (VK K J = ∅ ↔ (1 : MvPolynomial σ K) ∈ J) ∧
132     vanishingIdealOf K (VK K J) = J.radical := by
133   have hc : vanishingIdealOf K (VK K J) = J.radical := by
134     rw [vanishingIdealOf, VK_eq_zeroLocus, MvPolynomial.vanishingIdeal_zeroLocus_eq_radical]
135   refine ⟨⟨fun M => ?_, ?_, mem_VK_iff_closedPoint_mem_V K J⟩, ⟨fun h => ?_, fun h => ?_⟩, hc⟩
136   · simp only [maxIdealAt_eq_vanishingIdeal]
137     exact MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton
138   · intro a b hab
139     funext i
140     have hmem : MvPolynomial.X i - MvPolynomial.C (a i) ∈ maxIdealAt K b :=
141       hab ▸ Ideal.subset_span ⟨i, rfl⟩
142     rw [mem_maxIdealAt_iff] at hmem
143     simpa [sub_eq_zero, eq_comm] using hmem
144   · have h1 : (1 : MvPolynomial σ K) ∈ J.radical := by
145       rw [← hc, h, mem_vanishingIdealOf]
146       simp
147     exact (Ideal.radical_eq_top.mp ((Ideal.eq_top_iff_one _).mpr h1)).symm ▸ Submodule.mem_top
148   · ext a
149     simp only [Set.mem_empty_iff_false, iff_false]
150     intro ha
151     simpa using ha 1 h
152

157 /-- **Definition 2.5 (cotangent bundle).** `T^*X = Spec k[x, ξ]` for
158 `X = 𝔸^n_k`. -/
159 abbrev CotangentBundle (n : ℕ) := PrimeSpectrum (SymbolRing k n)
160 
161 /-- **Definition 2.5 (projection on closed points).** `π(y, ξ) = y`. -/
162 def proj {K : Type v} {n : ℕ} (z : PhaseVar n → K) : Fin n → K :=
163   fun i => z (Sum.inl i)
164 
165 /-- **Definition 2.5 (fibre).** `T^*_y X = π^{-1}(y) ≅ (K^n)^*`. -/
166 def fibre {K : Type v} {n : ℕ} (y : Fin n → K) : Set (PhaseVar n → K) :=
167   proj ⁻¹' {y}
168 
169 end
