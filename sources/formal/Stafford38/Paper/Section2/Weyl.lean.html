9 # Section 2 of the paper: Definition 2.1 and Convention 2.2
10 
11 Definition 2.1 (torsion and cyclic right modules) and Convention 2.2 (the first
12 Weyl pair `(x₁, p = ∂₁)`, the transverse algebra and `C = A_{n-1}(k)[x₁]`, and
13 the normal form `A_n(k) = ⊕_j p^j C`).
14 
15 Indexing: the paper's `A_n(k)` with `n ≥ 1` is Lean's `PresentedWeyl k (n + 1)`;
16 the paper's transverse algebra `A_{n-1}(k) = k⟨x_i, ∂_i : i ≥ 2⟩` is the image
17 of `PresentedWeyl k n` under `previousWeylEmbedding`.
18 -/

39 /-- **Definition 2.1 (torsion).** Every `m ∈ M` has `md = 0` for some
40 `0 ≠ d ∈ A`.  This is the repository's `TorsionCyclicity.IsRightTorsion`. -/
41 abbrev IsTorsion : Prop :=
42   Stafford38.TorsionCyclicity.IsRightTorsion (A := A) (M := M)
43 
44 /-- **Definition 2.1 (cyclic).** `M = mA` for some `m ∈ M`. -/
45 def IsCyclic : Prop :=
46   ∃ m : M, ∀ x : M, ∃ a : A, x = MulOpposite.op a • m
47 
48 /-- Cyclicity is the Mathlib form `span {m} = ⊤` used in the corollary. -/
49 theorem isCyclic_iff_span : IsCyclic A M ↔ ∃ m : M, Submodule.span Aᵐᵒᵖ {m} = ⊤ := by
50   constructor
51   · rintro ⟨m, hm⟩
52     refine ⟨m, eq_top_iff.mpr fun x _ => ?_⟩
53     obtain ⟨a, rfl⟩ := hm x
54     exact Submodule.smul_mem _ _ (Submodule.mem_span_singleton_self m)
55   · rintro ⟨m, hm⟩
56     refine ⟨m, fun x => ?_⟩
57     have hx : x ∈ Submodule.span Aᵐᵒᵖ {m} := hm ▸ Submodule.mem_top
58     obtain ⟨a, rfl⟩ := Submodule.mem_span_singleton.mp hx
59     exact ⟨a.unop, rfl⟩

91 /-- **Convention 2.2.** The coordinate `x₁` of the first Weyl pair. -/
92 abbrev x₁ (n : ℕ) : PresentedWeyl k (n + 1) := presentedCoordinate k n
93 
94 /-- **Convention 2.2.** The momentum `p = ∂₁` of the first Weyl pair. -/
95 abbrev p (n : ℕ) : PresentedWeyl k (n + 1) := presentedMomentum k n
96 
97 /-- **Convention 2.2.** The transverse algebra `A_{n-1}(k) = k⟨x_i, ∂_i : i ≥ 2⟩`. -/
98 abbrev transverse (n : ℕ) : Subalgebra k (PresentedWeyl k (n + 1)) :=
99   (previousWeylEmbedding k n).range
100 
101 /-- **Convention 2.2.** `C = A_{n-1}(k)[x₁]`, the subalgebra generated by the
102 transverse algebra and `x₁`. -/
103 def C (n : ℕ) : Subalgebra k (PresentedWeyl k (n + 1)) :=
104   Algebra.adjoin k ((transverse k n : Set (PresentedWeyl k (n + 1))) ∪ {x₁ k n})
105 
106 /-- `C` is the coefficient ring of the last Ore stage of the iterated Ore
107 presentation `A_n(k) = C[p; δ]`, embedded by `coordinateStageToPresented`. -/
108 theorem C_eq_range (n : ℕ) : C k n = (coordinateStageToPresented k n).range := by
109   apply le_antisymm
110   · apply Algebra.adjoin_le
111     rintro _ (⟨w, rfl⟩ | hx)
112     · refine ⟨normalCoefficient zeroDerivation (presentedToIterated k n w), ?_⟩
113       change iteratedToPresented k (n + 1) (stageEmbedding k n (presentedToIterated k n w)) = _
114       rw [iteratedToPresented_coefficient]
115       exact congrArg (previousWeylEmbedding k n)
116         (DFunLike.congr_fun (iteratedToPresented_comp_presentedToIterated k n) w)
117     · rw [Set.mem_singleton_iff.mp hx]
118       exact ⟨normalVariable zeroDerivation, iteratedToPresented_coordinate k n⟩
119   · rintro _ ⟨z, rfl⟩
120     obtain ⟨q, rfl⟩ := normalForm_surjective
121       (zeroDerivation : OreDivisionDerivation (IteratedPairStage k n)) z
122     induction q using Polynomial.induction_on' with
123     | add q₁ q₂ h₁ h₂ =>
124         rw [normalForm_add]
125         change coordinateStageToPresented k n (normalForm _ q₁ + normalForm _ q₂) ∈ C k n
126         rw [map_add]
127         exact add_mem h₁ h₂
128     | monomial i b =>
129         rw [normalForm_monomial]
130         change coordinateStageToPresented k n
131           (normalCoefficient zeroDerivation b * normalVariable zeroDerivation ^ i) ∈ C k n
132         rw [map_mul, map_pow]
133         have hx : coordinateStageToPresented k n (normalVariable zeroDerivation) = x₁ k n :=
134           iteratedToPresented_coordinate k n
135         rw [hx]
136         refine mul_mem (Algebra.subset_adjoin (Set.mem_union_left _ ?_))
137           (pow_mem (Algebra.subset_adjoin (Set.mem_union_right _ (Set.mem_singleton _))) i)
138         refine ⟨iteratedToPresented k n b, ?_⟩
139         change _ = iteratedToPresented k (n + 1) (stageEmbedding k n b)
140         rw [iteratedToPresented_coefficient]
141         rfl

143 /-- The last Ore stage, mapped back to `A_{n+1}(k)`. -/
144 def stageToPresented (n : ℕ) :
145     PairStage (B := IteratedPairStage k n) →ₐ[k] PresentedWeyl k (n + 1) :=
146   iteratedToPresented k (n + 1)

148 theorem presentedToIterated_stageToPresented (n : ℕ)
149     (x : PairStage (B := IteratedPairStage k n)) :
150     presentedToIterated k (n + 1) (stageToPresented k n x) = x :=
151   DFunLike.congr_fun (presentedToIterated_comp_iteratedToPresented k (n + 1)) x
152 
153 theorem stageToPresented_rightPBWMonomial (n j : ℕ) :
154     stageToPresented k n
155         (rightPBWMonomial (coordinateDerivation :
156           OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n))) j) =
157       p k n ^ j := by
158   have hmono : rightPBWMonomial (coordinateDerivation :
159       OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n))) j =
160       pairMomentum ^ j := by
161     rw [rightPBWMonomial_apply, Polynomial.X_pow_eq_monomial, normalForm_monomial,
162       map_one, one_mul]
163     rfl
164   rw [hmono, map_pow]
165   exact congrArg (· ^ j) (iteratedToPresented_momentum k n)

167 theorem stageToPresented_rightTerm (n j : ℕ)
168     (a : CoordinateStage (B := IteratedPairStage k n)) :
169     stageToPresented k n
170         (MulOpposite.op a • rightPBWMonomial (coordinateDerivation :
171           OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n))) j) =
172       p k n ^ j * coordinateStageToPresented k n a := by
173   rw [rightPBWMonomial_op_smul, map_mul, ← rightPBWMonomial_apply,

175   rfl

177 /-- **Convention 2.2, relations.** `x₁` is central in `C`, `[p, x₁] = 1`, and
178 the transverse algebra commutes with `x₁` and `p`. -/
179 theorem convention_2_2_relations (n : ℕ) :
180     p k n * x₁ k n = x₁ k n * p k n + 1 ∧
181       (∀ c ∈ C k n, x₁ k n * c = c * x₁ k n) ∧
182       ∀ t ∈ transverse k n, t * x₁ k n = x₁ k n * t ∧ t * p k n = p k n * t := by
183   refine ⟨presentedMomentum_mul_coordinate k n, ?_, ?_⟩
184   · intro c hc
185     induction hc using Algebra.adjoin_induction with
186     | mem x hx =>
187         rcases hx with ⟨w, rfl⟩ | hx
188         · exact presentedCoordinate_commutes_previous k n w
189         · rw [Set.mem_singleton_iff.mp hx]
190     | algebraMap r => exact (Algebra.commutes r _).symm
191     | add x y _ _ hx hy => rw [mul_add, add_mul, hx, hy]
192     | mul x y _ _ hx hy => rw [← mul_assoc, hx, mul_assoc, hy, mul_assoc]
193   · rintro _ ⟨w, rfl⟩
194     exact ⟨(presentedCoordinate_commutes_previous k n w).symm,
195       (presentedMomentum_commutes_previous k n w).symm⟩

197 /-- **Convention 2.2, normal form (existence).** Every element of `A_n(k)` can
198 be written as `∑_j p^j c_j` with `c_j ∈ C`. -/
199 theorem convention_2_2_exists (n : ℕ) (d : PresentedWeyl k (n + 1)) :
200     ∃ (L : ℕ) (a : ℕ → PresentedWeyl k (n + 1)), (∀ j, a j ∈ C k n) ∧
201       d = ∑ j ∈ Finset.range L, p k n ^ j * a j := by
202   let b := pairStage_rightOrePBWBasis (B := IteratedPairStage k n)
203   let c := b.repr (presentedToIterated k (n + 1) d)
204   let L := c.support.sup id + 1
205   have hsub : c.support ⊆ Finset.range L := by
206     intro j hj
207     exact Finset.mem_range.mpr (Nat.lt_succ_of_le (Finset.le_sup (f := id) hj))
208   refine ⟨L, fun j => coordinateStageToPresented k n (c j).unop,
209     fun j => (C_eq_range k n).symm ▸ ⟨_, rfl⟩, ?_⟩
210   have hd : presentedToIterated k (n + 1) d =
211       ∑ j ∈ Finset.range L, c j • rightPBWMonomial (coordinateDerivation :
212         OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n))) j := by
213     conv_lhs => rw [← b.linearCombination_repr (presentedToIterated k (n + 1) d)]
214     rw [Finsupp.linearCombination_apply, Finsupp.sum_of_support_subset _ hsub _ (by simp)]
215     apply Finset.sum_congr rfl
216     intro j _
217     simp [b, pairStage_rightOrePBWBasis]
218     rfl
219   have hback : d = stageToPresented k n (presentedToIterated k (n + 1) d) :=
220     (DFunLike.congr_fun (iteratedToPresented_comp_presentedToIterated k (n + 1)) d).symm
221   rw [hback, hd, map_sum]
222   apply Finset.sum_congr rfl
223   intro j _
224   exact stageToPresented_rightTerm k n j (c j).unop

226 /-- **Convention 2.2, normal form (uniqueness).** If `∑_{j<L} p^j c_j = 0`
227 with `c_j ∈ C`, then every `c_j = 0`. -/
228 theorem convention_2_2_unique (n L : ℕ) (a : ℕ → PresentedWeyl k (n + 1))
229     (ha : ∀ j, a j ∈ C k n) (h0 : ∑ j ∈ Finset.range L, p k n ^ j * a j = 0) :
230     ∀ j < L, a j = 0 := by
231   have hpre : ∀ j, ∃ z, coordinateStageToPresented k n z = a j := fun j => by
232     have := ha j
233     rw [C_eq_range] at this
234     exact this
235   choose b hb using hpre
236   let D : OreDivisionDerivation (CoordinateStage (B := IteratedPairStage k n)) :=
237     coordinateDerivation
238   have hsum : rightPBWCombination D
239       (∑ j ∈ Finset.range L, Finsupp.single j (MulOpposite.op (b j))) = 0 := by
240     rw [map_sum]
241     simp only [rightPBWCombination_single]
242     have himg : stageToPresented k n
243         (∑ j ∈ Finset.range L, MulOpposite.op (b j) • rightPBWMonomial D j) = 0 := by
244       rw [map_sum, ← h0]
245       apply Finset.sum_congr rfl
246       intro j _
247       rw [stageToPresented_rightTerm, hb]
248     have := congrArg (presentedToIterated k (n + 1)) himg
249     rwa [presentedToIterated_stageToPresented, map_zero] at this
250   have hzero := (rightPBWCombination_injective D) (hsum.trans (map_zero _).symm)
251   intro j hj
252   have hj' := congrArg (fun f : ℕ →₀ (CoordinateStage (B := IteratedPairStage k n))ᵐᵒᵖ => f j)
253     hzero
254   simp only [Finsupp.coe_finsetSum, Finset.sum_apply, Finsupp.single_apply,
255     Finset.sum_ite_eq', Finset.mem_range, ite_eq_left hj, Finsupp.coe_zero, Pi.zero_apply] at hj'
256   rw [← hb, MulOpposite.op_injective (hj'.trans MulOpposite.op_zero.symm), map_zero]
257 
258 end
