12 # Section 7: consequences without a printed proof (tex 1629–1634)
13 
14 The paper lists consequences of Theorem 1.1 that "have no printed proof".
15 Each is stated here in the Mathlib-only form of `Stafford38.PaperClaimDefinitions`
16 and proved directly; those that need Theorem 1.1 use
17 `Stafford38.Paper.theorem_1_1`.
18 
19 * `leftHandedIdentity`: `1 = Rd + SdF`, the transpose (formal adjoint) of
20   Theorem 1.1 applied to the transpose of `d`.
21 * `corollary_6_3_left`: the printed proof of Corollary 6.3 for right modules
22   over `A_n(k)ᵐᵒᵖ`, i.e. left `A_n(k)`-modules, with the left-handed identity
23   in place of `1 = dR + FdS` and the left Ore condition (via the transpose).
24 * `localizedDifferentialIdentity`: Theorem 1.1 for `A_n(k)` transported to
25   `D_k(S⁻¹k[x])` by clearing a left denominator of each operator.
26 * `evolutionaryIdentity`: the Euler-root certificate for
27   `d = p^r - ∑ a_j x^{m_j}`, which does not use Theorem 1.1.
28 -/

42 /-- `A_n(k)` is a left Ore domain: the right Ore condition of `A_n(k)ᵐᵒᵖ`,
43 obtained from the right Ore condition of `A_n(k)` through the transpose. -/
44 theorem weyl_leftOreCondition (k : Type u) [Field k] [CharZero k] (n : ℕ) :
45     AlgebraicAnalysis.OreRightIntersection.RightOreCondition (WeylAlg k n)ᵐᵒᵖ := by
46   intro a b ha hb
47   have hne : ∀ c : (WeylAlg k n)ᵐᵒᵖ, c ≠ 0 → transpose k n c.unop ≠ 0 := by
48     intro c hc h
49     apply hc
50     apply MulOpposite.unop_injective
51     rw [MulOpposite.unop_zero]
52     simpa only [transpose_transpose, transpose_zero] using congrArg (transpose k n) h
53   obtain ⟨x, y, hxy, hx0⟩ :=
54     Section6.weyl_rightOreCondition (k := k) n (hne a ha) (hne b hb)
55   have hxy' := congrArg (transpose k n) hxy
56   simp only [transpose_mul, transpose_transpose] at hxy'
57   refine ⟨MulOpposite.op (transpose k n x), MulOpposite.op (transpose k n y), ?_, ?_⟩
58   · apply MulOpposite.unop_injective
59     simpa only [MulOpposite.unop_mul, MulOpposite.unop_op] using hxy'
60   · intro h
61     apply hx0
62     have h' := congrArg (fun c => transpose k n c.unop) h
63     simpa only [MulOpposite.unop_mul, MulOpposite.unop_op, MulOpposite.unop_zero,
64       transpose_mul, transpose_transpose, transpose_zero] using h'

66 /-- **Corollary 6.3 for left torsion modules.** Every finitely generated torsion
67 left `A_n(k)`-module is cyclic.  The left analogue of the printed proof
68 (`Section6.corollary_6_3_of` for the ring `A_n(k)ᵐᵒᵖ`, whose right modules are
69 the left `A_n(k)`-modules), fed with the left-handed identity `1 = Rd + SdF`
70 (`leftHandedIdentity`) and the left Ore condition; the two generators of the
71 left version of [Sta78, Thm. 3.7] come from the same identity. -/
72 theorem corollary_6_3_left : LeftCorollary_6_3.{u, v} := by
73   intro k _ _ n M _ _ _ hM
74   let ψ : (WeylAlg k n)ᵐᵒᵖᵐᵒᵖ →+* WeylAlg k n := (RingEquiv.opOp (WeylAlg k n)).symm.toRingHom
75   let : Module (WeylAlg k n)ᵐᵒᵖᵐᵒᵖ M := Module.compHom M ψ
76   have hsmul : ∀ (a : WeylAlg k n) (m : M),
77       (MulOpposite.op (MulOpposite.op a) : (WeylAlg k n)ᵐᵒᵖᵐᵒᵖ) • m = a • m := fun _ _ => rfl
78   have hone : ∀ d : (WeylAlg k n)ᵐᵒᵖ, d ≠ 0 →
79       ∃ F R S : (WeylAlg k n)ᵐᵒᵖ, (1 : (WeylAlg k n)ᵐᵒᵖ) = d * R + F * d * S := by
80     intro d hd
81     obtain ⟨R, S, F, h⟩ := leftHandedIdentity k n d.unop
82       (fun h0 => hd (MulOpposite.unop_injective (by rw [h0, MulOpposite.unop_zero])))
83     refine ⟨MulOpposite.op F, MulOpposite.op R, MulOpposite.op S, ?_⟩
84     apply MulOpposite.unop_injective
85     simp only [MulOpposite.unop_one, MulOpposite.unop_add, MulOpposite.unop_mul,
86       MulOpposite.unop_op, h, mul_assoc]
87   have : Module.Finite (WeylAlg k n)ᵐᵒᵖᵐᵒᵖ M := by
88     obtain ⟨s, hs⟩ := (Module.Finite.fg_top : (⊤ : Submodule (WeylAlg k n) M).FG)
89     refine ⟨⟨s, ?_⟩⟩
90     let N : Submodule (WeylAlg k n) M :=
91       { carrier := Submodule.span (WeylAlg k n)ᵐᵒᵖᵐᵒᵖ (s : Set M)
92         add_mem' := fun ha hb => Submodule.add_mem _ ha hb
93         zero_mem' := Submodule.zero_mem _
94         smul_mem' := fun c x hx => by
95           change c • x ∈ Submodule.span (WeylAlg k n)ᵐᵒᵖᵐᵒᵖ (s : Set M)
96           rw [← hsmul]
97           exact Submodule.smul_mem _ _ hx }
98     have hN : (⊤ : Submodule (WeylAlg k n) M) ≤ N := by
99       rw [← hs]
100       exact Submodule.span_le.2 fun x hx => Submodule.subset_span hx
101     exact eq_top_iff.2 fun x _ => hN Submodule.mem_top
102   have hT : Stafford38.Paper.Section2.IsTorsion (WeylAlg k n)ᵐᵒᵖ M := by
103     intro m
104     obtain ⟨d, hd, hdm⟩ := hM m
105     refine ⟨MulOpposite.op d, fun h => hd (MulOpposite.op_injective (by rw [h, MulOpposite.op_zero])),
106       ?_⟩
107     rw [hsmul, hdm]
108   have hOre := weyl_leftOreCondition k n
109   obtain ⟨m, hm⟩ := (Section6.corollary_6_3_of hOre hone hT
110     ⟨Section6.two_generators_of_identity hOre hone hT,
111       fun l hl => Section6.stably_generated_of_identity hOre hone hT l (by omega)⟩).1
112   refine ⟨m, fun x => ?_⟩
113   obtain ⟨a, ha⟩ := hm x
114   exact ⟨a.unop, ha⟩

122 /-- The iterated-commutator order of `Stafford38Challenge` is the recursive
123 order filtration `order N` of `AlgebraicAnalysis`. -/
124 theorem isDifferentialOperatorOfOrder_iff (N : ℕ) (P : Module.End k B) :
125     IsDifferentialOperatorOfOrder k B N P ↔ P ∈ order (k := k) (R := B) N := by
126   induction N generalizing P with
127   | zero =>
128       rw [mem_order_zero_iff]
129       constructor
130       · intro h a
131         exact h [a] rfl
132       · intro h b hb
133         match b, hb with
134         | [a], _ => exact h a
135   | succ N ih =>
136       rw [mem_order_succ_iff]
137       constructor
138       · intro h a
139         rw [← ih]
140         intro b hb
141         exact h (a :: b) (by simp [hb])
142       · intro h b hb
143         match b, hb with
144         | a :: b, hb => exact (ih _).2 (h a) b (by simpa using hb)

180 theorem foldl_conjEnd (e : B' ≃ₐ[k] B) :
181     ∀ (l : List B') (P : Module.End k B'),
182       (l.map e).foldl (fun Q c => Q * LinearMap.mulLeft k c - LinearMap.mulLeft k c * Q)
183           (conjEnd e P) =
184         conjEnd e (l.foldl (fun Q c => Q * LinearMap.mulLeft k c - LinearMap.mulLeft k c * Q) P)
185   | [], _ => rfl
186   | b :: l, P => by
187       rw [List.map_cons, List.foldl_cons, List.foldl_cons, ← foldl_conjEnd e l]
188       congr 1
189       rw [map_sub, map_mul, map_mul, conjEnd_mulLeft]

191 theorem isDifferentialOperator_conjEnd (e : B' ≃ₐ[k] B) {P : Module.End k B'}
192     (h : IsDifferentialOperator k B' P) : IsDifferentialOperator k B (conjEnd e P) := by
193   obtain ⟨N, hN⟩ := h
194   refine ⟨N, fun l hl => ?_⟩
195   have hl' : (l.map e.symm).map e = l := by
196     clear hl
197     induction l with
198     | nil => rfl
199     | cons b l ih => simp only [List.map_cons, AlgEquiv.apply_symm_apply, ih]
200   rw [← hl', foldl_conjEnd, hN _ (by simpa using hl), map_zero]

202 /-- The identity in `D_k(B')` transfers along a `k`-algebra isomorphism `B' ≅ B`. -/
203 theorem hasDifferentialIdentity_of_algEquiv (e : B' ≃ₐ[k] B)
204     (h : HasDifferentialIdentity k B') : HasDifferentialIdentity k B := by
205   intro d hd hd0
206   let P := conjEnd e.symm d
207   have hP : conjEnd e P = d := by
208     ext x
209     simp [P]
210   have hP0 : P ≠ 0 := fun h0 => hd0 (by rw [← hP, h0, map_zero])
211   obtain ⟨F, R, S, hF, hR, hS, hone⟩ := h P (isDifferentialOperator_conjEnd e.symm hd) hP0
212   refine ⟨conjEnd e F, conjEnd e R, conjEnd e S, isDifferentialOperator_conjEnd e hF,
213     isDifferentialOperator_conjEnd e hR, isDifferentialOperator_conjEnd e hS, ?_⟩
214   have := congrArg (conjEnd e) hone
215   simp only [map_one, map_add, map_mul, hP] at this
216   exact this

220 /-- **Localizations.** For every localization `B` of `k[x_1, …, x_n]`, every
221 nonzero finite-order differential operator `d` on `B` has finite-order `F, R, S`
222 with `1 = dR + FdS`.  Theorem 1.1 transfers to `D_k(S⁻¹k[x])` by clearing a left
223 denominator of each operator (`s38_localized_differential`, for `Localization S`)
224 and then along the isomorphism `Localization S ≅ B`. -/
225 theorem localizedDifferentialIdentity : LocalizedDifferentialIdentity.{u, v} := by
226   intro k _ _ n S B _ _ _ _ _
227   have h := hasDifferentialIdentity_of
228     (Stafford38.LocalizedDifferentialCorollaries.s38_localized_differential S (Localization S)
229       fun d hd => Stafford38.Paper.theorem_1_1 k n d hd)
230   exact hasDifferentialIdentity_of_algEquiv
231     ((IsLocalization.algEquiv S (Localization S) B).restrictScalars k) h

233 /-- `potentialSum` over the list of pairs `(a_j, m_j)` is `∑_j a_j x^{m_j}`. -/
234 theorem potentialSum_ofFn {D : Type v} [Ring D] (x : D) :
235     ∀ (J : ℕ) (a : Fin J → D) (m : Fin J → ℕ),
236       Stafford38.Evolution.potentialSum x (List.ofFn fun j => (a j, m j)) =
237         ∑ j, a j * x ^ m j
238   | 0, _, _ => by simp [Stafford38.Evolution.potentialSum]
239   | J + 1, a, m => by
240       rw [List.ofFn_succ, Fin.sum_univ_succ, Stafford38.Evolution.potentialSum,
241         potentialSum_ofFn x J (fun j => a j.succ) (fun j => m j.succ)]
242 
243 /-- **Evolutionary identity.** For `r ≥ 1` and `px = xp + 1` in a `k`-algebra,
244 `d = p^r - ∑_j a_j x^{m_j}` with each `a_j` commuting with `x` and `p` has
245 `1 = dR + x^r d S`. -/
246 theorem evolutionaryIdentity : EvolutionaryIdentity.{u, v} := by
247   intro k _ _ D _ _ x p hw r hr J a m hx hp
248   have hmem : ∀ t ∈ List.ofFn (fun j => (a j, m j)), ∃ j, t = (a j, m j) := by
249     intro t ht
250     obtain ⟨j, rfl⟩ := List.mem_ofFn.1 ht
251     exact ⟨j, rfl⟩
252   obtain ⟨R, S, h⟩ := Stafford38.Evolution.evolutionaryCorollary k hw hr
253     (List.ofFn fun j => (a j, m j))
254     (fun t ht => by obtain ⟨j, rfl⟩ := hmem t ht; exact hx j)
255     (fun t ht => by obtain ⟨j, rfl⟩ := hmem t ht; exact hp j) rfl
256   exact ⟨R, S, by rw [← potentialSum_ofFn x J a m]; exact h⟩
