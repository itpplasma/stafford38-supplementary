8 /-!
9 # The associated graded `k[x, ξ]`-module of a filtered right `A_n(K)`-module
10 
11 For a right `A_n(K)`-module `M` with a compatible order filtration `M_•`
12 (Definition 2.4, `IsCompatibleFiltration`), `gr M = ⊕_i M_i/M_{i-1}` (`gr`)
13 is a module over `gr A_n = K[x, ξ]`: the variable `x_j` acts by right
14 multiplication with the generator `x_j ∈ F_0` and `ξ_j` by right
15 multiplication with `∂_j ∈ F_1`, each raising the degree by its order weight.
16 The two actions commute on `gr M` because the commutators of the generators
17 are scalars, of lower order.
18 -/

44 theorem gen_mem (v : PhaseVar n) : gen K n v ∈ orderFiltration K n (orderWeight v) := by
45   change Stafford.freeWeylGenerator (Matrix.J (Fin n) K) v ∈ orderPiece K n (orderWeight v)
46   rw [orderPiece, mem_presentedWeightPiece, presentedNormalFormLinearEquiv_generator]
47   intro m hm
48   simp only [MvPolynomial.coeff_X] at hm
49   split at hm
50   · subst m
51     simp [monomialWeight]
52   · exact absurd rfl hm

54 theorem gen_commutator (v w : PhaseVar n) :
55     gen K n v * gen K n w - gen K n w * gen K n v =
56       algebraMap K (PresentedWeyl K n) (Matrix.J (Fin n) K v w) := by
57   have h := RingQuot.mkAlgHom_rel K
58     (s := Stafford.freeWeylRelation (Matrix.J (Fin n) K)) ⟨v, w, rfl, rfl⟩
59   rw [map_sub, map_mul, map_mul, AlgHom.commutes] at h
60   exact h

62 theorem algebraMap_mem_order (c : K) :
63     algebraMap K (PresentedWeyl K n) c ∈ orderFiltration K n 0 := by
64   rw [Algebra.algebraMap_eq_smul_one]
65   exact Submodule.smul_mem _ c (orderPieceOne (n := n) K).property

67 /-- The commutator of two generators lies in `F_{w(v)+w(w)-1}`; for two
68 coordinates it is zero. -/
69 theorem gen_commutator_mem_lower (v w : PhaseVar n) :
70     gen K n v * gen K n w - gen K n w * gen K n v ∈
71       lowerPiece (orderFiltration K n) (orderWeight v + orderWeight w) := by
72   rw [gen_commutator]
73   have hmono := (isFiltration_order K n).mono
74   rcases v with a | a <;> rcases w with b | b
75   · simp [Matrix.J, orderWeight, fibreWeight, lowerPiece]
76   · exact algebraMap_mem_order _
77   · exact algebraMap_mem_order _
78   · exact hmono (Nat.zero_le 1) (algebraMap_mem_order _)

98 /-- The class of `x` in `M_i/M_{i-1}` (zero when `x ∉ M_i`). -/
99 def cls (Γ : ℕ → Submodule K M) (i : ℕ) (x : M) : GrModule Γ :=
100   if h : x ∈ Γ i then DirectSum.of (grPiece Γ) i (Submodule.Quotient.mk ⟨x, h⟩) else 0

104 theorem cls_of_mem {i : ℕ} {x : M} (h : x ∈ Γ i) :
105     cls Γ i x = DirectSum.of (grPiece Γ) i (Submodule.Quotient.mk ⟨x, h⟩) := by
106   unfold cls
107   exact dite_eq_left h
108 
109 theorem cls_add {i : ℕ} {x y : M} (hx : x ∈ Γ i) (hy : y ∈ Γ i) :
110     cls Γ i (x + y) = cls Γ i x + cls Γ i y := by
111   rw [cls_of_mem hx, cls_of_mem hy, cls_of_mem (add_mem hx hy)]
112   change _ = DirectSum.of (grPiece Γ) i _ + DirectSum.of (grPiece Γ) i _
113   rw [← map_add]
114   rfl
115 
116 theorem cls_smul {i : ℕ} {x : M} (hx : x ∈ Γ i) (c : K) :
117     cls Γ i (c • x) = c • cls Γ i x := by
118   rw [cls_of_mem hx, cls_of_mem (Submodule.smul_mem _ c hx)]
119   change DirectSum.lof K ℕ (grPiece Γ) i _ = c • DirectSum.lof K ℕ (grPiece Γ) i _
120   rw [← map_smul]
121   rfl
122 
123 theorem cls_zero (i : ℕ) : cls Γ i (0 : M) = 0 := by
124   rw [cls_of_mem (zero_mem _)]
125   exact (DirectSum.of (grPiece Γ) i).map_zero
126 
127 theorem cls_neg {i : ℕ} {x : M} (hx : x ∈ Γ i) : cls Γ i (-x) = -cls Γ i x := by
128   have h := cls_add hx (neg_mem hx)
129   rw [add_neg_cancel, cls_zero] at h
130   exact (neg_eq_of_add_eq_zero_right h.symm).symm
131 
132 theorem cls_sub {i : ℕ} {x y : M} (hx : x ∈ Γ i) (hy : y ∈ Γ i) :
133     cls Γ i (x - y) = cls Γ i x - cls Γ i y := by
134   rw [sub_eq_add_neg, cls_add hx (neg_mem hy), cls_neg hy, ← sub_eq_add_neg]
135 
136 theorem cls_eq_zero_of_mem_lower {i : ℕ} {x : M} (h : x ∈ lowerPiece Γ i) :
137     cls Γ i x = 0 := by
138   by_cases hx : x ∈ Γ i
139   · have h0 : (Submodule.Quotient.mk ⟨x, hx⟩ : grPiece Γ i) = 0 :=
140       (Submodule.Quotient.mk_eq_zero _).mpr h
141     rw [cls_of_mem hx, h0, map_zero]
142     rfl
143   · unfold cls
144     exact dite_eq_right hx

146 /-- `gr M` is spanned by the classes `cls i x`, `x ∈ M_i`. -/
147 theorem gr_induction {P : GrModule Γ → Prop} (z : GrModule Γ) (h0 : P 0)
148     (hcls : ∀ i x, x ∈ Γ i → P (cls Γ i x)) (hadd : ∀ a b, P a → P b → P (a + b)) : P z := by
149   change gr Γ at z
150   induction z using DirectSum.induction_on with
151   | zero => exact h0
152   | of i y =>
153       induction y using Submodule.Quotient.induction_on with
154       | H y =>
155           obtain ⟨x, hx⟩ := y
156           have := hcls i x hx
157           rwa [cls_of_mem hx] at this
158   | add a b ha hb => exact hadd a b ha hb
159 
160 theorem gr_ext {X : Type*} [AddCommGroup X] [Module K X] {f g : GrModule Γ →ₗ[K] X}
161     (h : ∀ i x, x ∈ Γ i → f (cls Γ i x) = g (cls Γ i x)) : f = g := by
162   ext z
163   induction z using gr_induction with
164   | h0 => simp
165   | hcls i x hx => exact h i x hx
166   | hadd a b ha hb => rw [map_add, map_add, ha, hb]

169 def ev (Γ : ℕ → Submodule K M) (j : ℕ) : GrModule Γ →ₗ[K] M ⧸ lowerPiece Γ j :=
170   (Submodule.mapQ _ _ (Γ j).subtype le_rfl).comp (DirectSum.component K ℕ (grPiece Γ) j)

172 theorem ev_cls (j : ℕ) {i : ℕ} {x : M} (hx : x ∈ Γ i) :
173     ev Γ j (cls Γ i x) = if i = j then Submodule.Quotient.mk x else 0 := by
174   rw [cls_of_mem hx]
175   by_cases hij : i = j
176   · subst hij
177     rw [ite_eq_left rfl]
178     change Submodule.mapQ _ _ (Γ i).subtype le_rfl
179       (DirectSum.component K ℕ (grPiece Γ) i (DirectSum.lof K ℕ (grPiece Γ) i _)) = _

183     change Submodule.mapQ _ _ (Γ j).subtype le_rfl
184       (DirectSum.component K ℕ (grPiece Γ) j (DirectSum.lof K ℕ (grPiece Γ) i _)) = _

234 /-- Right multiplication by `a` as a `K`-linear map of `M`. -/
235 def actL (a : PresentedWeyl K n) : M →ₗ[K] M where
236   toFun x := MulOpposite.op a • x
237   map_add' x y := smul_add _ x y
238   map_smul' c x := (smul_comm c (MulOpposite.op a) x).symm

240 /-- Right multiplication by `a ∈ F_j`, from `M_i/M_{i-1}` to `M_{i+j}/M_{i+j-1}`. -/
241 def pieceShift {a : PresentedWeyl K n} {j : ℕ} (ha : a ∈ orderFiltration K n j) (i : ℕ) :
242     grPiece Γ i →ₗ[K] grPiece Γ (i + j) :=
243   Submodule.mapQ _ _ ((actL a).restrict (p := Γ i) (q := Γ (i + j))
244     fun _ hx => hΓ.smul_mem hx ha) (by
245       rintro ⟨x, hx⟩ hlow

248 /-- Right multiplication by `a ∈ F_j` on `gr M`, raising degrees by `j`. -/
249 def shiftOp {a : PresentedWeyl K n} {j : ℕ} (ha : a ∈ orderFiltration K n j) :
250     GrModule Γ →ₗ[K] GrModule Γ :=
251   DirectSum.toModule K ℕ (gr Γ) fun i =>
252     (DirectSum.lof K ℕ (grPiece Γ) (i + j)).comp (pieceShift hΓ ha i)

254 theorem shiftOp_cls {a : PresentedWeyl K n} {j : ℕ} (ha : a ∈ orderFiltration K n j)
255     {i : ℕ} {x : M} (hx : x ∈ Γ i) :
256     shiftOp hΓ ha (cls Γ i x) = cls Γ (i + j) (MulOpposite.op a • x) := by
257   rw [cls_of_mem hx, cls_of_mem (hΓ.smul_mem hx ha)]
258   change DirectSum.toModule K ℕ (gr Γ) _ (DirectSum.lof K ℕ (grPiece Γ) i _) = _
259   rw [DirectSum.toModule_lof]
260   rfl

262 /-- `T_v`: right multiplication by the generator `v` on `gr M`. -/
263 def genOp (v : PhaseVar n) : GrModule Γ →ₗ[K] GrModule Γ :=
264   shiftOp hΓ (gen_mem v)

266 theorem genOp_cls (v : PhaseVar n) {i : ℕ} {x : M} (hx : x ∈ Γ i) :
267     genOp hΓ v (cls Γ i x) =
268       cls Γ (i + orderWeight v) (MulOpposite.op (gen K n v) • x) :=
269   shiftOp_cls hΓ _ hx

271 /-- On `gr M` the generators commute: their commutators are scalars, which
272 lower the degree. -/
273 theorem genOp_commute (v w : PhaseVar n) : Commute (genOp hΓ v) (genOp hΓ w) := by
274   change genOp hΓ v * genOp hΓ w = genOp hΓ w * genOp hΓ v
275   apply gr_ext
276   intro i x hx
277   have hw := hΓ.smul_mem hx (gen_mem w)
278   have hv := hΓ.smul_mem hx (gen_mem v)
279   have hvw := hΓ.smul_mem hw (gen_mem v)
280   have hwv := hΓ.smul_mem hv (gen_mem w)
281   rw [Module.End.mul_apply, Module.End.mul_apply, genOp_cls hΓ w hx, genOp_cls hΓ v hw,
282     genOp_cls hΓ v hx, genOp_cls hΓ w hv,
283     cls_congr (show i + orderWeight v + orderWeight w = i + orderWeight w + orderWeight v by
284       omega), ← sub_eq_zero, ← cls_sub hvw (by rwa [show i + orderWeight w + orderWeight v =
285         i + orderWeight v + orderWeight w by omega])]
286   apply cls_eq_zero_of_mem_lower
287   have hcomm := smul_mem_lowerPiece hΓ hx (gen_commutator_mem_lower w v)
288   rw [← add_assoc] at hcomm
289   convert hcomm using 1
290   rw [MulOpposite.op_sub, sub_smul, MulOpposite.op_mul, MulOpposite.op_mul, mul_smul, mul_smul]
291 
292 end Construction

398       rw [genOp_cls hΓ v hx, grMap_cls φ hφ (hΓ.smul_mem hx (gen_mem v)), grMap_cls φ hφ hx,

421 /-- The filtration `ι⁻¹(M_i)` induced on a submodule `ι : M₁ ↪ M`. -/
422 def subFilt (i : ℕ) : Submodule K M₁ :=
423   (Γ i).comap (ι.restrictScalars K)
424 
425 /-- The image filtration `(M_i + N)/N` on the quotient `M/N`, `N = ι(M₁)`. -/
426 def quotFilt (i : ℕ) : Submodule K (M ⧸ LinearMap.range ι) :=
427   (Γ i).map ((LinearMap.range ι).mkQ.restrictScalars K)

431 theorem subFilt_isCompatible (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ) :
432     IsCompatibleFiltration (orderFiltration K n) (subFilt Γ ι) where
433   mono _ _ h _ hx := hΓ.mono h hx
434   smul_mem := by
435     intro i j x a hx ha
436     change ι (MulOpposite.op a • x) ∈ Γ (i + j)
437     rw [map_smul]
438     exact hΓ.smul_mem hx ha
439   exhaustive x := hΓ.exhaustive (ι x)

443     IsCompatibleFiltration (orderFiltration K n) (quotFilt Γ ι) where
444   mono _ _ h := Submodule.map_mono (hΓ.mono h)

446     rintro i j _ a ⟨x, hx, rfl⟩ ha

493   change DFinsupp.mapRange (fun i => pieceMap φ hφ i) (fun i => (pieceMap φ hφ i).map_zero)

505         at hy
506       rw [Submodule.mem_bot, Submodule.Quotient.mk_eq_zero]

518 theorem grQuot_surjective :
519     Function.Surjective (grMap (LinearMap.range ι).mkQ (quotFilt_map (Γ := Γ))) := by
520   intro z
521   induction z using gr_induction with
522   | h0 => exact ⟨0, map_zero _⟩
523   | hcls i y hy =>
524       obtain ⟨x, hx, rfl⟩ := hy
525       exact ⟨cls Γ i x, grMap_cls _ _ hx⟩
526   | hadd a b ha hb =>
527       obtain ⟨a', rfl⟩ := ha
528       obtain ⟨b', rfl⟩ := hb
529       exact ⟨a' + b', map_add _ _ _⟩

531 theorem pieceMap_quot_exact (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ) (i : ℕ)
532     (q : grPiece Γ i) (hq : pieceMap (LinearMap.range ι).mkQ (quotFilt_map (Γ := Γ)) i q = 0) :
533     ∃ y, pieceMap ι (subFilt_map (Γ := Γ)) i y = q := by
534   induction q using Submodule.Quotient.induction_on with
535   | H q =>
536       obtain ⟨x, hx⟩ := q
537       rw [pieceMap, Submodule.mapQ_apply, Submodule.Quotient.mk_eq_zero] at hq
538       obtain ⟨x', hx', hxx'⟩ := mem_lowerPiece_quotFilt.mp hq
539       have hN : x - x' ∈ LinearMap.range ι := by
540         rw [← Submodule.Quotient.eq]
541         exact hxx'.symm
542       obtain ⟨x₁, hx₁⟩ := hN
543       have hΓx : ι x₁ ∈ Γ i := hx₁ ▸ sub_mem hx (lowerPiece_le hΓ i hx')
544       refine ⟨Submodule.Quotient.mk ⟨x₁, hΓx⟩, ?_⟩
545       rw [pieceMap, Submodule.mapQ_apply, Submodule.Quotient.eq]
546       change ι x₁ - x ∈ lowerPiece Γ i
547       rw [hx₁, sub_sub_cancel_left]
548       exact neg_mem hx'

550 theorem grQuot_exact (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ) (z : GrModule Γ)
551     (hz : grMap (LinearMap.range ι).mkQ (quotFilt_map (Γ := Γ)) z = 0) :
552     ∃ y, grMap ι (subFilt_map (Γ := Γ)) y = z := by
553   classical
554   have hpiece : ∀ i, ∃ y, pieceMap ι (subFilt_map (Γ := Γ)) i y = comp Γ i z := by
555     intro i
556     refine pieceMap_quot_exact hΓ i _ ?_
557     rw [← grMap_apply, hz, map_zero]
558   choose y hy using hpiece
559   obtain ⟨s, hs⟩ := sum_ofG_comp z
560   refine ⟨∑ i ∈ s, ofG (subFilt Γ ι) i (y i), ?_⟩
561   rw [map_sum, hs]
562   refine Finset.sum_congr rfl fun i _ => ?_
563   rw [grMap_of, hy]

582 theorem annihilator_mul_le :
583     Module.annihilator (SymbolRing K n) (GrModule (subFilt Γ ι)) *
584         Module.annihilator (SymbolRing K n) (GrModule (quotFilt Γ ι)) ≤

592   obtain ⟨y, hy⟩ := grQuot_exact hΓ _ hker
593   rw [mul_smul, ← hy, ← grMap_smul _ _ (subFilt_isCompatible hΓ) hΓ, hr, map_zero]

596 /-- `√Ann gr M = √Ann gr N ∩ √Ann gr(M/N)` for the induced filtrations. -/
