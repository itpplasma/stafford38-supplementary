9 # Theorem 2.11 (Gabber) for finitely generated modules with a good filtration
10 
11 **Theorem 2.11.** Let `K` be a field of characteristic zero and `M` a finitely
12 generated right `A_n(K)`-module with a good order filtration `M_•`.  Then
13 `J = √(Ann_{K[x,ξ]} gr M)` is involutive, `{J, J} ⊆ J`.
14 
15 The theorem is cited from the literature [Gab81]; the proof here reduces it to
16 the cyclic case `M = A_n(K)/I` with the filtration `qF_m`, which the repository
17 proves (`GabberGlobalAssembly.associatedGraded_radical_isInvolutive`):
18 
19 1. `√Ann gr M` does not depend on the good filtration (`GabberGood`), so for a
20    cyclic module any good filtration may be replaced by `qF_m`, whose `gr` is
21    the repository's `grCyclic` as a `K[x, ξ]`-module (`grCyclicEquivS`).
22 2. For `0 → N → M → M/N → 0` with induced filtrations, `gr N → gr M → gr(M/N)`
23    is exact, so `√Ann gr M = √Ann gr N ∩ √Ann gr(M/N)`; both induced filtrations
24    are good because `K[x, ξ]` is Noetherian.
25 3. Induction on the number of generators, splitting off a cyclic submodule;
26    an intersection of involutive ideals is involutive.
27 
28 [Gab81] O. Gabber, *The integrability of the characteristic variety*,
29 Amer. J. Math. 103 (1981), 445–468.
30 -/

86 theorem grCyclic_induction {P : grCyclic K I → Prop} (y : grCyclic K I) (h0 : P 0)
87     (hof : ∀ N (z : orderPiece K n N),
88       P (orderAssociatedGradedOf K I N (orderPieceToQuotientGraded K I N z)))
89     (hadd : ∀ a b, P a → P b → P (a + b)) : P y :=
90   DirectSum.induction_on (motive := fun y => P y) y h0
91     (fun N q => by
92       obtain ⟨z, rfl⟩ := orderPieceToQuotientGraded_surjective K I N q
93       exact hof N z)
94     (fun a b ha hb => hadd a b ha hb)

96 theorem grCyclicEquiv_X_smul (v : PhaseVar n) (y : grCyclic K I) :
97     grCyclicEquivG K I ((MvPolynomial.X v : SymbolRing K n) • y) =
98       (MvPolynomial.X v : SymbolRing K n) • grCyclicEquivG K I y := by
99   induction y using grCyclic_induction with
100   | h0 => rw [smul_zero, map_zero, smul_zero]
101   | hof N z =>
102       have h := smul_orderAssociatedGradedOf_mk_eq_of_mul K I z ⟨gen K n v, gen_mem v⟩
103       rw [principal_gen] at h
104       rw [h, grCyclicEquiv_of, grCyclicEquiv_of]
105       have hz : (Submodule.Quotient.mk (z : PresentedWeyl K n) : RightQuotient I) ∈
106           qFilt K I N := ⟨z, z.property, orbitMap_qClass K I z⟩
107       rw [X_smul_cls (cyclicFiltration_order_isCompatible K I) v hz]
108       rfl
109   | hadd a b ha hb => rw [smul_add, map_add, ha, hb, map_add, smul_add]

111 /-- `gr Q` for `Q_m = qF_m`, with the `K[x, ξ]`-structure of `GrModule`, is the
112 repository's `grCyclic` (`OrderAssociatedGradedModule`). -/
113 def grCyclicEquivS : grCyclic K I ≃ₗ[SymbolRing K n] GrModule (qFilt K I) :=
114   { grCyclicEquivG K I with
115     map_smul' := by
116       intro f y
117       change grCyclicEquivG K I (f • y) = f • grCyclicEquivG K I y
118       induction f using MvPolynomial.induction_on generalizing y with
119       | C c =>
120           rw [C_smul, ← MvPolynomial.algebraMap_eq, algebraMap_smul, map_smul]
121       | add p q hp hq => rw [add_smul, map_add, hp, hq, add_smul]
122       | mul_X p v hp => rw [mul_smul, hp, grCyclicEquiv_X_smul, mul_smul] }

169 theorem isInvolutive_of_subsingleton [Subsingleton M] (Γ : ℕ → Submodule K M) :
170     IsInvolutive (Module.annihilator (SymbolRing K n) (GrModule Γ)).radical := by
171   have hsub : Subsingleton (GrModule Γ) := by
172     refine ⟨fun a b => ?_⟩
173     have h0 : ∀ z : GrModule Γ, z = 0 := fun z => by
174       induction z using gr_induction with
175       | h0 => rfl
176       | hcls i x _ => rw [Subsingleton.elim x 0, cls_zero]
177       | hadd a b ha hb => rw [ha, hb, add_zero]
178     rw [h0 a, h0 b]
179   rw [Module.annihilator_eq_top_iff.mpr hsub, Ideal.radical_top]
180   exact fun _ _ _ _ => Submodule.mem_top

182 /-- `a ↦ m a`, the orbit map of `m` as a map of right modules. -/
183 def orbitHom (m : M) : PresentedWeyl K n →ₗ[(PresentedWeyl K n)ᵐᵒᵖ] M where
184   toFun a := MulOpposite.op a • m
185   map_add' a b := by rw [MulOpposite.op_add, add_smul]
186   map_smul' c a := by
187     change MulOpposite.op (a * c.unop) • m = c • MulOpposite.op a • m
188     rw [MulOpposite.op_mul, MulOpposite.op_unop, mul_smul]

210 theorem isInvolutive_of_card (r : ℕ) :
211     ∀ {M : Type w} [AddCommGroup M] [Module (PresentedWeyl K n)ᵐᵒᵖ M] [Module K M]
212       [IsScalarTower K (PresentedWeyl K n)ᵐᵒᵖ M] (s : Finset M), s.card ≤ r →
213       Submodule.span (PresentedWeyl K n)ᵐᵒᵖ (s : Set M) = ⊤ →
214       ∀ Γ : ℕ → Submodule K M, IsCompatibleFiltration (orderFiltration K n) Γ →
215         Module.Finite (SymbolRing K n) (GrModule Γ) →
216         IsInvolutive (Module.annihilator (SymbolRing K n) (GrModule Γ)).radical := by
217   classical
218   induction r with
219   | zero =>
220       intro M _ _ _ _ s hs hspan Γ _ _
221       have : s = ∅ := Finset.card_eq_zero.mp (Nat.le_zero.mp hs)
222       subst this
223       rw [Finset.coe_empty, Submodule.span_empty] at hspan
224       have : Subsingleton M := subsingleton_of_span_eq_top_empty hspan
225       exact isInvolutive_of_subsingleton Γ
226   | succ r ih =>
227       intro M _ _ _ _ s hs hspan Γ hΓ hgood
228       by_cases hne : s = ∅
229       · subst hne
230         rw [Finset.coe_empty, Submodule.span_empty] at hspan
231         have : Subsingleton M := subsingleton_of_span_eq_top_empty hspan
232         exact isInvolutive_of_subsingleton Γ
233       obtain ⟨m, hm⟩ := Finset.nonempty_iff_ne_empty.mpr hne
234       obtain ⟨ι, hι, hmι⟩ : ∃ ι : RightQuotient (LinearMap.ker (orbitHom (K := K) (n := n) m))
235           →ₗ[(PresentedWeyl K n)ᵐᵒᵖ] M, Function.Injective ι ∧ m ∈ LinearMap.range ι :=
236         ⟨_, cyclicInclusion_injective m, mem_range_cyclicInclusion m⟩
237       rw [radical_annihilator_eq hΓ hι]
238       apply isInvolutive_inf
239       · exact cyclic_isInvolutive K _ (subFilt_isCompatible hΓ) (subFilt_good hι hΓ)
240       · have hgoodq : Module.Finite (SymbolRing K n) (GrModule (quotFilt Γ ι)) :=
241           quotFilt_good hΓ
242         refine ih ((s.erase m).image (LinearMap.range ι).mkQ) ?_ ?_ _
243           (quotFilt_isCompatible hΓ) hgoodq
244         · have := Finset.card_erase_of_mem hm
245           have := Finset.card_image_le (s := s.erase m) (f := (LinearMap.range ι).mkQ)
246           omega
247         · rw [Finset.coe_image, Submodule.span_image, eq_top_iff]
248           rintro y -
249           obtain ⟨x, rfl⟩ := (LinearMap.range ι).mkQ_surjective y
250           have hx : x ∈ Submodule.span (PresentedWeyl K n)ᵐᵒᵖ
251               (insert m ((s.erase m : Finset M) : Set M)) := by
252             rw [← Finset.coe_insert, Finset.insert_erase hm, hspan]
253             exact Submodule.mem_top
254           obtain ⟨a, z, hz, rfl⟩ := Submodule.mem_span_insert.mp hx
255           refine ⟨z, hz, ?_⟩
256           have hm0 : (LinearMap.range ι).mkQ m = 0 :=
257             (Submodule.Quotient.mk_eq_zero _).mpr hmι
258           rw [map_add, map_smul, hm0, smul_zero, zero_add]

264 /-- **Theorem 2.11 (Gabber).** Let `K` be a field of characteristic zero and `M`
265 a finitely generated right `A_n(K)`-module with a good order filtration `M_•`
266 (Definition 2.4: compatible, exhaustive, `gr M` finitely generated over
267 `K[x, ξ]`).  Then `J = √(Ann_{K[x,ξ]} gr M)` is involutive, `{J, J} ⊆ J`.
268 
269 The proof reduces to the cyclic case proved in the repository
270 (`GabberGlobalAssembly.associatedGraded_radical_isInvolutive`); see the module
271 docstring. -/
272 theorem theorem_2_11_general (K : Type u) [Field K] [CharZero K] (n : ℕ)
273     {M : Type w} [AddCommGroup M] [Module (Stafford38Challenge.WeylAlg K n)ᵐᵒᵖ M] [Module K M]
274     [IsScalarTower K (Stafford38Challenge.WeylAlg K n)ᵐᵒᵖ M]
275     [Module.Finite (Stafford38Challenge.WeylAlg K n)ᵐᵒᵖ M]
276     (Γ : ℕ → Submodule K M) (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ)
277     (hgood : Module.Finite (SymbolRing K n) (GrModule Γ)) :
278     IsInvolutive (Module.annihilator (SymbolRing K n) (GrModule Γ)).radical := by
279   obtain ⟨s, hs⟩ := Module.Finite.fg_top (R := (PresentedWeyl K n)ᵐᵒᵖ) (M := M)
280   exact isInvolutive_of_card s.card s le_rfl hs Γ hΓ hgood
