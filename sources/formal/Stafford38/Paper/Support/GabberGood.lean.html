7 /-!
8 # Good filtrations: shift comparison and independence of `√Ann gr M`
9 
10 Two facts about compatible order filtrations of a right `A_n(K)`-module `M`:
11 
12 * a good filtration `M_•` is dominated by every compatible exhaustive filtration
13   `M'_•` up to a shift, `M_i ⊆ M'_{i+c}`;
14 * if two compatible filtrations dominate each other up to shifts, the radicals
15   of the annihilators of their associated graded modules agree.
16 
17 Hence `√(Ann_{K[x,ξ]} gr M)` does not depend on the good filtration
18 [HTT, Lemma D.3.1].  Both rest on writing the action of `f ∈ K[x, ξ]` on
19 `gr M` through finitely many `b_e ∈ F_e` (`exists_lift`).
20 -/

56 /-- The action of `f ∈ K[x, ξ]` on `gr M` is right multiplication by finitely
57 many `b_e ∈ F_e`, the same for every compatible filtration:
58 `f · [x]_d = Σ_{e<D} [x b_e]_{d+e}`. -/
59 theorem exists_lift (f : SymbolRing K n) :
60     ∃ (D : ℕ) (b : ℕ → PresentedWeyl K n), (∀ e, b e ∈ orderFiltration K n e) ∧
61       ∀ (Γ : ℕ → Submodule K M), IsCompatibleFiltration (orderFiltration K n) Γ →
62         ∀ d x, x ∈ Γ d →
63           f • cls Γ d x = ∑ e ∈ Finset.range D, cls Γ (d + e) (MulOpposite.op (b e) • x) := by
64   classical
65   induction f using MvPolynomial.induction_on with
66   | C c =>
67       refine ⟨1, fun e => if e = 0 then algebraMap K _ c else 0, ?_, ?_⟩
68       · intro e
69         dsimp only
70         split_ifs with he
71         · subst he; exact algebraMap_mem_order c
72         · exact zero_mem _
73       · intro Γ hΓ d x hx
74         dsimp only
75         rw [Finset.sum_range_one, ite_eq_left rfl, add_zero, C_smul, ← cls_smul hx]
76         congr 1
77         rw [← MulOpposite.algebraMap_apply, algebraMap_smul]
78   | add p q hp hq =>
79       obtain ⟨D₁, b₁, hb₁, h₁⟩ := hp
80       obtain ⟨D₂, b₂, hb₂, h₂⟩ := hq
81       refine ⟨max D₁ D₂, fun e => (if e < D₁ then b₁ e else 0) + (if e < D₂ then b₂ e else 0),
82         fun e => add_mem (by split_ifs <;> simp [hb₁ e, zero_mem])
83           (by split_ifs <;> simp [hb₂ e, zero_mem]), ?_⟩
84       intro Γ hΓ d x hx
85       have hmem : ∀ (p : Prop) [Decidable p] (b : ℕ → PresentedWeyl K n),
86           (∀ e, b e ∈ orderFiltration K n e) → ∀ e,
87           MulOpposite.op (if p then b e else 0) • x ∈ Γ (d + e) := by
88         intro p _ b hb e
89         split_ifs
90         · exact hΓ.smul_mem hx (hb e)
91         · simp [zero_mem]
92       rw [add_smul, h₁ Γ hΓ d x hx, h₂ Γ hΓ d x hx]
93       simp_rw [MulOpposite.op_add, add_smul]
94       rw [Finset.sum_congr rfl fun e _ => cls_add (hmem (e < D₁) b₁ hb₁ e)
95         (hmem (e < D₂) b₂ hb₂ e), Finset.sum_add_distrib]
96       simp_rw [cls_ite_smul]
97       rw [sum_range_ite (le_max_left _ _), sum_range_ite (le_max_right _ _)]
98   | mul_X p v hp =>
99       obtain ⟨D, b, hb, h⟩ := hp
100       let w := orderWeight v
101       refine ⟨w + D, fun e => if w ≤ e then gen K n v * b (e - w) else 0, ?_, ?_⟩
102       · intro e
103         dsimp only
104         split_ifs with he
105         · have := mul_mem_orderPiece K (gen_mem v) (hb (e - w))
106           rwa [show orderWeight v + (e - w) = e by omega] at this
107         · exact zero_mem _
108       · intro Γ hΓ d x hx
109         dsimp only
110         rw [mul_smul, X_smul_cls hΓ v hx,
111           h Γ hΓ _ _ (hΓ.smul_mem hx (gen_mem v)), Finset.sum_range_add]
112         rw [Finset.sum_eq_zero (s := Finset.range w) (fun e he => by
113           rw [ite_eq_right (by simp at he; omega), MulOpposite.op_zero, zero_smul, cls_zero]),
114           zero_add]
115         refine Finset.sum_congr rfl fun e _ => ?_
116         rw [ite_eq_left (Nat.le_add_right _ _), show w + e - w = e by omega,
117           cls_congr (show d + (w + e) = d + orderWeight v + e by omega),
118           MulOpposite.op_mul, mul_smul]

128 /-- Classes `[x]_i` with `x ∈ M_i ∩ M'_{i+c}`. -/
129 def boundedSpan (Γ Γ' : ℕ → Submodule K M) (c : ℕ) : Submodule K (GrModule Γ) :=
130   Submodule.span K {z | ∃ i x, x ∈ Γ i ∧ x ∈ Γ' (i + c) ∧ z = cls Γ i x}

132 theorem symbol_smul_mem_boundedSpan (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ)
133     (hΓ' : IsCompatibleFiltration (orderFiltration K n) Γ') (c : ℕ) (f : SymbolRing K n)
134     {z : GrModule Γ} (hz : z ∈ boundedSpan Γ Γ' c) : f • z ∈ boundedSpan Γ Γ' c := by
135   obtain ⟨D, b, hb, hlift⟩ := exists_lift (M := M) f
136   induction hz using Submodule.span_induction with
137   | mem z hz =>
138       obtain ⟨i, x, hx, hx', rfl⟩ := hz
139       rw [hlift Γ hΓ i x hx]
140       refine Submodule.sum_mem _ fun e _ => Submodule.subset_span ⟨i + e, _,
141         hΓ.smul_mem hx (hb e), ?_, rfl⟩
142       have := hΓ'.smul_mem hx' (hb e)
143       rwa [show i + c + e = i + e + c by omega] at this
144   | zero => rw [smul_zero]; exact zero_mem _
145   | add a b' _ _ ha hb' => rw [smul_add]; exact add_mem ha hb'
146   | smul a z _ hz =>
147       rw [smul_def, map_smul, ← smul_def]
148       exact Submodule.smul_mem _ a hz

158 theorem exists_mem_boundedSpan (hΓ' : IsCompatibleFiltration (orderFiltration K n) Γ')

171 theorem ev_of_mem_boundedSpan {c : ℕ} {z : GrModule Γ} (hz : z ∈ boundedSpan Γ Γ' c)
172     (j : ℕ) : ∃ y ∈ Γ' (j + c), ev Γ j z = Submodule.Quotient.mk y := by
173   induction hz using Submodule.span_induction with
174   | mem z hz =>
175       obtain ⟨i, x, hx, hx', rfl⟩ := hz
176       rw [ev_cls j hx]
177       split_ifs with hij
178       · subst hij; exact ⟨x, hx', rfl⟩
179       · exact ⟨0, zero_mem _, (Submodule.Quotient.mk_zero _).symm⟩
180   | zero => exact ⟨0, zero_mem _, by rw [map_zero, Submodule.Quotient.mk_zero]⟩
181   | add a b _ _ ha hb =>
182       obtain ⟨y₁, hy₁, e₁⟩ := ha
183       obtain ⟨y₂, hy₂, e₂⟩ := hb
184       exact ⟨y₁ + y₂, add_mem hy₁ hy₂, by rw [map_add, e₁, e₂, Submodule.Quotient.mk_add]⟩
185   | smul a z _ hz =>
186       obtain ⟨y, hy, e⟩ := hz
187       exact ⟨a • y, Submodule.smul_mem _ a hy, by rw [map_smul, e, Submodule.Quotient.mk_smul]⟩

189 /-- A good filtration is dominated, up to a shift, by every compatible
190 exhaustive filtration: `M_i ⊆ M'_{i+c}`. -/
191 theorem exists_shift_le (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ)
192     (hgood : Module.Finite (SymbolRing K n) (GrModule Γ))
193     (hΓ' : IsCompatibleFiltration (orderFiltration K n) Γ') :
194     ∃ c, ∀ i, Γ i ≤ Γ' (i + c) := by
195   classical
196   obtain ⟨s, hs⟩ := hgood.fg_top
197   choose cz hcz using exists_mem_boundedSpan (Γ := Γ) hΓ'
198   let C := s.sup cz
199   let W : Submodule (SymbolRing K n) (GrModule Γ) :=
200     { boundedSpan Γ Γ' C with
201       smul_mem' := fun f _ hz => symbol_smul_mem_boundedSpan hΓ hΓ' C f hz }
202   have htop : W = ⊤ := by
203     rw [eq_top_iff, ← hs, Submodule.span_le]
204     intro z hz
205     exact boundedSpan_mono hΓ' (Finset.le_sup (f := cz) hz) (hcz z)
206   refine ⟨C, fun i => ?_⟩
207   induction i with
208   | zero =>
209       intro x hx
210       have hmem : cls Γ 0 x ∈ W := htop ▸ Submodule.mem_top
211       obtain ⟨y, hy, hev⟩ := ev_of_mem_boundedSpan hmem 0
212       rw [ev_cls 0 hx, ite_eq_left rfl, Submodule.Quotient.eq] at hev
213       have : x = y := sub_eq_zero.mp ((Submodule.mem_bot K).mp hev)
214       rwa [this]
215   | succ i ih =>
216       intro x hx
217       have hmem : cls Γ (i + 1) x ∈ W := htop ▸ Submodule.mem_top
218       obtain ⟨y, hy, hev⟩ := ev_of_mem_boundedSpan hmem (i + 1)
219       rw [ev_cls _ hx, ite_eq_left rfl, Submodule.Quotient.eq] at hev
220       have hlow : x - y ∈ Γ' (i + C) := ih hev
221       have : x = (x - y) + y := by abel
222       rw [this]
223       exact add_mem (hΓ'.mono (by omega) hlow) hy

233 /-- `M_z` for an integer index `z`, zero for `z < 0`. -/
234 def zPiece (Γ : ℕ → Submodule K M) (z : ℤ) : Submodule K M :=
235   if 0 ≤ z then Γ z.toNat else ⊥
236 
237 theorem zPiece_nat (i : ℕ) : zPiece Γ i = Γ i := by
238   simp [zPiece]
239 
240 theorem lowerPiece_eq_zPiece (i : ℕ) : lowerPiece Γ i = zPiece Γ ((i : ℤ) - 1) := by
241   cases i with
242   | zero => simp [lowerPiece, zPiece]
243   | succ i =>
244       simp only [lowerPiece, zPiece]
245       rw [ite_eq_left (by omega)]
246       congr 1
247       omega

249 theorem zPiece_le_shift {c : ℕ} (h : ∀ i, Γ' i ≤ Γ (i + c)) (z : ℤ) :
250     zPiece Γ' z ≤ zPiece Γ (z + c) := by
251   unfold zPiece
252   split_ifs with h₁ h₂
253   · have := h z.toNat
254     rwa [show z.toNat + c = (z + c).toNat by omega] at this
255   · omega
256   · exact bot_le
257   · exact bot_le

259 /-- The classes `[x]_j`, `x ∈ M'_j ∩ M_{j+c-N}`. -/
260 def loweredSpan (Γ Γ' : ℕ → Submodule K M) (c : ℕ) (N : ℕ) : Submodule K (GrModule Γ') :=
261   Submodule.span K {z | ∃ j x, x ∈ Γ' j ∧ x ∈ zPiece Γ ((j : ℤ) + c - N) ∧ z = cls Γ' j x}

263 /-- If `M'_i ⊆ M_{i+c}` and `M_i ⊆ M'_{i+c'}`, then
264 `Ann gr_M ⊆ √(Ann gr_{M'})`. -/
265 theorem annihilator_le_radical (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ)
266     (hΓ' : IsCompatibleFiltration (orderFiltration K n) Γ') {c c' : ℕ}
267     (h₁ : ∀ i, Γ' i ≤ Γ (i + c)) (h₂ : ∀ i, Γ i ≤ Γ' (i + c')) :
268     Module.annihilator (SymbolRing K n) (GrModule Γ) ≤
269       (Module.annihilator (SymbolRing K n) (GrModule Γ')).radical := by
270   intro f hf
271   obtain ⟨D, b, hb, hlift⟩ := exists_lift (M := M) f
272   -- `f` annihilates `gr_M`, so every `b_e` lowers the `M`-degree by one.
273   have hlower : ∀ e, e < D → ∀ (z : ℤ) x, x ∈ zPiece Γ z →
274       MulOpposite.op (b e) • x ∈ zPiece Γ (z + e - 1) := by
275     intro e he z x hx
276     by_cases hz : 0 ≤ z
277     · have hx' : x ∈ Γ z.toNat := by simpa [zPiece, hz] using hx
278       have hzero := (Module.mem_annihilator.mp hf) (cls Γ z.toNat x)
279       rw [hlift Γ hΓ _ x hx'] at hzero
280       have hev := congrArg (ev Γ (z.toNat + e)) hzero
281       rw [map_sum, map_zero, Finset.sum_eq_single e] at hev
282       · rw [ev_cls _ (hΓ.smul_mem hx' (hb e)), ite_eq_left rfl,
283           Submodule.Quotient.mk_eq_zero, lowerPiece_eq_zPiece] at hev
284         convert hev using 2
285         omega
286       · intro e' _ he'
287         rw [ev_cls _ (hΓ.smul_mem hx' (hb e')), ite_eq_right (by omega)]
288       · intro h; exact absurd (Finset.mem_range.mpr he) h
289     · have : x = 0 := by simpa [zPiece, hz] using hx
290       subst this
291       rw [smul_zero]
292       exact zero_mem _
293   have hstep : ∀ N, ∀ z ∈ loweredSpan Γ Γ' c N, f • z ∈ loweredSpan Γ Γ' c (N + 1) := by
294     intro N z hz
295     induction hz using Submodule.span_induction with
296     | mem z hz =>
297         obtain ⟨j, x, hx', hx, rfl⟩ := hz
298         rw [hlift Γ' hΓ' j x hx']
299         refine Submodule.sum_mem _ fun e he => Submodule.subset_span
300           ⟨j + e, _, hΓ'.smul_mem hx' (hb e), ?_, rfl⟩
301         have := hlower e (Finset.mem_range.mp he) _ x hx
302         convert this using 2
303         push_cast
304         ring
305     | zero => rw [smul_zero]; exact zero_mem _
306     | add a b' _ _ ha hb' => rw [smul_add]; exact add_mem ha hb'
307     | smul a z _ hz =>
308         rw [smul_def, map_smul, ← smul_def]
309         exact Submodule.smul_mem _ a hz
310   have hpow : ∀ N d x, x ∈ Γ' d → f ^ N • cls Γ' d x ∈ loweredSpan Γ Γ' c N := by
311     intro N
312     induction N with
313     | zero =>
314         intro d x hx
315         rw [pow_zero, one_smul]
316         refine Submodule.subset_span ⟨d, x, hx, ?_, rfl⟩
317         rw [show (d : ℤ) + c - ((0 : ℕ) : ℤ) = ((d + c : ℕ) : ℤ) by push_cast; ring, zPiece_nat]
318         exact h₁ d hx
319     | succ N ih =>
320         intro d x hx
321         rw [pow_succ', mul_smul]
322         exact hstep N _ (ih d x hx)
323   have hvanish : ∀ z ∈ loweredSpan Γ Γ' c (c + c' + 1), z = 0 := by
324     intro z hz
325     induction hz using Submodule.span_induction with
326     | mem z hz =>
327         obtain ⟨j, x, hx', hx, rfl⟩ := hz
328         apply cls_eq_zero_of_mem_lower
329         rw [lowerPiece_eq_zPiece]
330         have := zPiece_le_shift h₂ _ hx
331         convert this using 2
332         push_cast
333         ring
334     | zero => rfl
335     | add a b _ _ ha hb => rw [ha, hb, add_zero]
336     | smul a z _ hz => rw [hz, smul_zero]
337   refine ⟨c + c' + 1, ?_⟩
338   rw [Module.mem_annihilator]
339   intro z
340   induction z using gr_induction with
341   | h0 => exact smul_zero _
342   | hcls d x hx => exact hvanish _ (hpow _ d x hx)
343   | hadd a b ha hb => rw [smul_add, ha, hb, add_zero]

345 /-- Shift-comparable compatible filtrations have the same `√(Ann gr M)`. -/
346 theorem radical_annihilator_eq_of_shift (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ)
347     (hΓ' : IsCompatibleFiltration (orderFiltration K n) Γ') {c c' : ℕ}
348     (h₁ : ∀ i, Γ' i ≤ Γ (i + c)) (h₂ : ∀ i, Γ i ≤ Γ' (i + c')) :
349     (Module.annihilator (SymbolRing K n) (GrModule Γ)).radical =
350       (Module.annihilator (SymbolRing K n) (GrModule Γ')).radical :=
351   le_antisymm (Ideal.radical_le_radical_iff.mpr (annihilator_le_radical hΓ hΓ' h₁ h₂))
352     (Ideal.radical_le_radical_iff.mpr (annihilator_le_radical hΓ' hΓ h₂ h₁))
353 
354 /-- **`√(Ann gr M)` is independent of the good filtration** [HTT, Lemma D.3.1]. -/
355 theorem radical_annihilator_eq_of_good (hΓ : IsCompatibleFiltration (orderFiltration K n) Γ)
356     (hΓ' : IsCompatibleFiltration (orderFiltration K n) Γ')
357     (hgood : Module.Finite (SymbolRing K n) (GrModule Γ))
358     (hgood' : Module.Finite (SymbolRing K n) (GrModule Γ')) :
359     (Module.annihilator (SymbolRing K n) (GrModule Γ)).radical =
360       (Module.annihilator (SymbolRing K n) (GrModule Γ')).radical := by
361   obtain ⟨c, hc⟩ := exists_shift_le hΓ' hgood' hΓ
362   obtain ⟨c', hc'⟩ := exists_shift_le hΓ hgood hΓ'
363   exact radical_annihilator_eq_of_shift hΓ hΓ' hc hc'
