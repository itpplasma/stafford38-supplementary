11 # Section 4, first part: the Euler grading, Lemmas 4.1, 4.2 and Proposition 4.3
12 
13 `E = x₁ p` with `p = ∂₁`.  The paragraph before Lemma 4.1 introduces the
14 Euler grading, the nonnegative part `A_{≥0}` and the normality of `x₁` in it.
15 Lemma 4.1 computes `x₁^m p^m` and `p^m x₁^m` as products of Euler factors,
16 Lemma 4.2 produces `1 + u₀ x₁ ∈ 𝓘`, and Proposition 4.3 proves `Q = Q x₁`.
17 
18 The ring-theoretic paragraphs are stated for an arbitrary ring `A` with
19 `p x₁ = x₁ p + 1`; the paper's statements are their specializations to the
20 presented Weyl algebra `A_n = PresentedWeyl k (n+1)` (Lean indexes the
21 coordinates from `0`, so the paper's `x₁` is `presentedCoordinate k n` and the
22 transverse algebra `A_{n-1}` is the image of `previousWeylEmbedding k n`), the
23 normalized element `d̃` (`IsPBWMonicAt … N d`) and the right ideal
24 `𝓘 = d̃ A_n + x₁^N d̃ A_n` (`presentedCanonicalRightIdeal n N d`).
25 -/

46 variable {A : Type*} [Ring A]
47 
48 /-- The Euler element `E = x₁ p`. -/
49 abbrev eulerElement (x p : A) : A := x * p
50 
51 /-- `A_{≥0}`: the subring generated by the transverse algebra `C`, by `E` and
52 by `x₁`.  This is the repository's canonical owner `eulerSubring`; the PBW
53 description by monomials of nonnegative Euler weight is
54 `PaperEulerGrading.pairEulerSubring_eq_closure_nonnegativeEulerPBWMonomials`. -/
55 abbrev nonnegativePart (C : Subring A) (x p : A) : Subring A :=
56   eulerSubring C x p
57 
58 /-- **Equation (nonnegative-pbw).**  `E x₁ = x₁(E+1)`, `x₁E = (E−1)x₁`,
59 `Ep = p(E−1)`, `pE = (E+1)p`, from `p x₁ = x₁ p + 1`. -/
60 theorem eq_nonnegative_pbw (x p : A) (hweyl : p * x = x * p + 1) :
61     eulerElement x p * x = x * (eulerElement x p + 1) ∧
62       x * eulerElement x p = (eulerElement x p - 1) * x ∧
63       eulerElement x p * p = p * (eulerElement x p - 1) ∧
64       p * eulerElement x p = (eulerElement x p + 1) * p := by
65   simp only [eulerElement]
66   refine ⟨?_, ?_, ?_, ?_⟩
67   · rw [mul_assoc, hweyl]
68   · rw [sub_mul, one_mul, mul_assoc x p x, hweyl]
69     noncomm_ring
70   · rw [mul_sub, mul_one, ← mul_assoc p x p, hweyl]
71     noncomm_ring
72   · rw [← mul_assoc, hweyl]
73 
74 /-- **Normality of `x₁` in `A_{≥0}`.**  `x₁` commutes with the transverse
75 algebra, so for every `r ∈ A_{≥0}` there is `r' ∈ A_{≥0}` with `x₁ r = r' x₁`.
76 Only this one-sided normality is used later. -/
77 theorem step_coordinate_normal (C : Subring A) (x p : A)
78     (hweyl : p * x = x * p + 1)
79     (hcomm : ∀ c : C, x * (c : A) = (c : A) * x) :
80     ∀ r ∈ nonnegativePart C x p, ∃ r' ∈ nonnegativePart C x p, x * r = r' * x := by
81   intro r hr
82   obtain ⟨s, hs⟩ := coordinate_normal_eulerSubring C x p hweyl hcomm ⟨r, hr⟩
83   exact ⟨s, s.property, hs⟩
84 
85 end Grading

93 /-- The factors `E − i` and `E + i` are polynomials in the single element `E`,
94 hence commute with one another. -/
95 theorem step_4_1_factors_commute (x p : A) :
96     (∀ i j : ℕ, Commute (eulerElement x p - i) (eulerElement x p - j)) ∧
97       ∀ i j : ℕ, Commute (eulerElement x p + i) (eulerElement x p + j) := by
98   refine ⟨fun i j => ?_, fun i j => ?_⟩
99   · exact ((Commute.refl _).sub_right (Nat.cast_commute j _).symm).sub_left
100       ((Nat.cast_commute i _).sub_right (Nat.cast_commute i (j : A)))
101   · exact ((Commute.refl _).add_right (Nat.cast_commute j _).symm).add_left
102       ((Nat.cast_commute i _).add_right (Nat.cast_commute i (j : A)))
103 
104 /-- `x₁ ∏(E − c) = ∏(E − 1 − c) x₁` for any list of shifts, from
105 `x₁E = (E−1)x₁`. -/
106 theorem step_4_1_coordinate_shift (x p : A) (hweyl : p * x = x * p + 1)
107     (l : List ℕ) :
108     x * (l.map fun i : ℕ => eulerElement x p - i).prod =
109       (l.map fun i : ℕ => eulerElement x p - 1 - i).prod * x := by
110   have hxE := (eq_nonnegative_pbw x p hweyl).2.1
111   induction l with
112   | nil => simp
113   | cons a l ih =>
114       simp only [List.map_cons, List.prod_cons]
115       rw [← mul_assoc, mul_sub, hxE, ← (Nat.cast_commute a x).eq, ← sub_mul,
116         mul_assoc, ih, mul_assoc]
117 
118 /-- `p ∏(E + c) = ∏(E + 1 + c) p` for any list of shifts, from `pE = (E+1)p`. -/
119 theorem step_4_1_momentum_shift (x p : A) (hweyl : p * x = x * p + 1)
120     (l : List ℕ) :
121     p * (l.map fun i : ℕ => eulerElement x p + i).prod =
122       (l.map fun i : ℕ => eulerElement x p + 1 + i).prod * p := by
123   have hpE := (eq_nonnegative_pbw x p hweyl).2.2.2
124   induction l with
125   | nil => simp
126   | cons a l ih =>
127       simp only [List.map_cons, List.prod_cons]
128       rw [← mul_assoc, mul_add, hpE, ← (Nat.cast_commute a p).eq, ← add_mul,
129         mul_assoc, ih, mul_assoc]
130 
131 /-- Induction step of the first identity:
132 `x₁^{m+1}p^{m+1} = x₁(∏_{i<m}(E−i))p = (∏_{i<m}(E−1−i))x₁p = ∏_{i≤m}(E−i)`. -/
133 theorem step_4_1_falling_succ (x p : A) (hweyl : p * x = x * p + 1) (m : ℕ)
134     (ih : x ^ m * p ^ m = ((List.range m).map fun i : ℕ => eulerElement x p - i).prod) :
135     x ^ (m + 1) * p ^ (m + 1) =
136       ((List.range (m + 1)).map fun i : ℕ => eulerElement x p - i).prod := by
137   calc x ^ (m + 1) * p ^ (m + 1) = x * (x ^ m * p ^ m) * p := by
138         rw [pow_succ', pow_succ]; simp only [mul_assoc]
139     _ = ((List.range m).map fun i : ℕ => eulerElement x p - 1 - i).prod * eulerElement x p := by
140         rw [ih, step_4_1_coordinate_shift x p hweyl, mul_assoc]
141     _ = eulerElement x p * ((List.range m).map fun i : ℕ => eulerElement x p - 1 - i).prod := by
142         refine (Commute.list_prod_left _ _ fun y hy => ?_).eq
143         obtain ⟨i, -, rfl⟩ := List.mem_map.mp hy
144         exact ((Commute.refl _).sub_left (Commute.one_left _)).sub_left
145           (Nat.cast_commute i _)
146     _ = ((List.range (m + 1)).map fun i : ℕ => eulerElement x p - i).prod := by
147         rw [List.range_succ_eq_map, List.map_cons, List.prod_cons, List.map_map]
148         congr 1
149         · simp
150         · congr 1
151           refine List.map_congr_left fun i _ => ?_
152           simp only [Function.comp_apply, Nat.succ_eq_add_one]
153           push_cast
154           abel
155 
156 /-- Induction step of the second identity, from `Ep = p(E−1)`, i.e.
157 `pE = (E+1)p`:
158 `p^{m+1}x₁^{m+1} = p(∏_{i=1}^m(E+i))x₁ = (∏_{i=1}^m(E+1+i))px₁ = ∏_{i=1}^{m+1}(E+i)`. -/
159 theorem step_4_1_rising_succ (x p : A) (hweyl : p * x = x * p + 1) (m : ℕ)
160     (ih : p ^ m * x ^ m = ((List.range' 1 m).map fun i : ℕ => eulerElement x p + i).prod) :
161     p ^ (m + 1) * x ^ (m + 1) =
162       ((List.range' 1 (m + 1)).map fun i : ℕ => eulerElement x p + i).prod := by
163   have hpx : p * x = eulerElement x p + 1 := hweyl
164   calc p ^ (m + 1) * x ^ (m + 1) = p * (p ^ m * x ^ m) * x := by
165         rw [pow_succ', pow_succ]; simp only [mul_assoc]
166     _ = ((List.range' 1 m).map fun i : ℕ => eulerElement x p + 1 + i).prod *
167           (eulerElement x p + 1) := by
168         rw [ih, step_4_1_momentum_shift x p hweyl, mul_assoc, hpx]
169     _ = (eulerElement x p + 1) *
170           ((List.range' 1 m).map fun i : ℕ => eulerElement x p + 1 + i).prod := by
171         refine (Commute.list_prod_left _ _ fun y hy => ?_).eq
172         obtain ⟨i, -, rfl⟩ := List.mem_map.mp hy
173         exact (Commute.refl (eulerElement x p + 1)).add_left (Nat.cast_commute i _)
174     _ = ((List.range' 1 (m + 1)).map fun i : ℕ => eulerElement x p + i).prod := by
175         rw [List.range'_succ, List.map_cons, List.prod_cons]
176         congr 1
177         · simp
178         · rw [List.range'_eq_map_range, List.range'_eq_map_range, List.map_map,
179             List.map_map]
180           congr 1
181           refine List.map_congr_left fun i _ => ?_
182           simp only [Function.comp_apply]
183           push_cast
184           rw [add_assoc _ 1, ← add_assoc 1 1, one_add_one_eq_two]
185 
186 /-- Lemma 4.1 for any ring with elements `x, p` satisfying `p x = x p + 1`:
187 for every `m ≥ 1`, `x^m p^m = ∏_{i=0}^{m-1}(E−i)` and `p^m x^m = ∏_{i=1}^{m}(E+i)`
188 with `E = x p`, the factors in each product commuting with one another.  The
189 products are taken in the listed order of `i`; by the last two clauses the order
190 does not matter. -/
191 theorem lemma_4_1_of_weylPair (x p : A) (hweyl : p * x = x * p + 1) (m : ℕ) (hm : 1 ≤ m) :
192     x ^ m * p ^ m = ((List.range m).map fun i : ℕ => eulerElement x p - i).prod ∧
193       p ^ m * x ^ m = ((List.range' 1 m).map fun i : ℕ => eulerElement x p + i).prod ∧
194       (∀ i j : ℕ, Commute (eulerElement x p - i) (eulerElement x p - j)) ∧
195       ∀ i j : ℕ, Commute (eulerElement x p + i) (eulerElement x p + j) := by
196   refine ⟨?_, ?_, step_4_1_factors_commute x p⟩
197   · -- induction from the trivial case `m = 1`
198     induction m, hm using Nat.le_induction with
199     | base => simp [eulerElement]
200     | succ m _ ih => exact step_4_1_falling_succ x p hweyl m ih
201   · induction m, hm using Nat.le_induction with
202     | base => simp [eulerElement, hweyl]
203     | succ m _ ih => exact step_4_1_rising_succ x p hweyl m ih
204 
205 end Lemma41
