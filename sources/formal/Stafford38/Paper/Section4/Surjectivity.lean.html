8 /-!
9 # Proposition 4.3: right multiplication by `x₁` on `Q = A_n/𝓘` is surjective
10 
11 The printed proof works with `q A_{≥0}`, the classes `q r` of the elements
12 `r ∈ A_{≥0}`, where `q` is the class of `1`.  Its paragraphs are: `q ∈ (qA_{≥0})x₁`;
13 `qA_{≥0} = (qA_{≥0})x₁` by normality; `qA_{≥0}` is stable under `p` and under
14 `C = A_{n-1}[x₁]`; hence `qA_{≥0} = Q`.  They are stated below for a ring with a
15 right ideal, then applied to the presented Weyl algebra.
16 -/

36 /-- `q A_{≥0}`: the classes `q r = [r]` with `r ∈ R`. -/
37 def classSet : Set (RightQuotient I) := {z | ∃ r ∈ R, z = qmk I r}

41 /-- **`q ∈ (qA_{≥0})x₁`.**  From `1 + u₀x₁ ∈ 𝓘`, `q = −q u₀ x₁`. -/
42 theorem step_4_3_q_mem (x u₀ : A) (hres : 1 + u₀ * x ∈ I) :
43     qmk I 1 = rightMul I x (qmk I (-u₀)) := by
44   rw [← qmk_right_mul]
45   refine ((Submodule.Quotient.eq I).2 ?_).symm
46   convert I.neg_mem hres using 1
47   noncomm_ring
48 
49 /-- **Equation (normal-surj): `qA_{≥0} = (qA_{≥0})x₁`.**  `⊇` because `x₁ ∈ A_{≥0}`;
50 `⊆` because `qr = −(qu₀)x₁r = −(qu₀r')x₁` for `x₁r = r'x₁`. -/
51 theorem step_4_3_normal_surj (x u₀ : A) (hx : x ∈ R) (hu₀ : u₀ ∈ R)
52     (hres : 1 + u₀ * x ∈ I)
53     (hnormal : ∀ r ∈ R, ∃ r' ∈ R, x * r = r' * x) :
54     classSet I R = rightMul I x '' classSet I R := by
55   ext z
56   constructor
57   · rintro ⟨r, hr, rfl⟩
58     obtain ⟨r', hr', hxr⟩ := hnormal r hr
59     refine ⟨qmk I (-(u₀ * r')), ⟨_, R.neg_mem (R.mul_mem hu₀ hr'), rfl⟩, ?_⟩
60     calc rightMul I x (qmk I (-(u₀ * r'))) = qmk I (-u₀ * (r' * x)) := by
61           rw [← qmk_right_mul]; congr 1; noncomm_ring
62       _ = qmk I (-u₀ * x * r) := by rw [← hxr, mul_assoc]
63       _ = rightMul I r (rightMul I x (qmk I (-u₀))) := by
64           rw [← qmk_right_mul, ← qmk_right_mul]
65       _ = qmk I r := by rw [← step_4_3_q_mem x u₀ hres, ← qmk_right_mul, one_mul]
66   · rintro ⟨w, ⟨r, hr, rfl⟩, rfl⟩
67     exact ⟨r * x, R.mul_mem hr hx, (qmk_right_mul I r x).symm⟩
68 
69 /-- **Stability under `p`.**  If `m = m'x₁` with `m' ∈ qA_{≥0}`, then
70 `mp = m'(x₁p) = m'E ∈ qA_{≥0}`. -/
71 theorem step_4_3_stable_momentum (x p : A) (hE : x * p ∈ R)
72     (hsurj : classSet I R = rightMul I x '' classSet I R) :
73     ∀ z ∈ classSet I R, rightMul I p z ∈ classSet I R := by
74   intro z hz
75   rw [hsurj] at hz
76   obtain ⟨_, ⟨r, hr, rfl⟩, rfl⟩ := hz
77   refine ⟨r * (x * p), R.mul_mem hr hE, ?_⟩
78   rw [← qmk_right_mul, ← qmk_right_mul, mul_assoc]
79 
80 /-- **Stability under `C`.**  `C = A_{n-1}[x₁] ⊆ A_{≥0}`. -/
81 theorem step_4_3_stable_coefficients (C : Subring A) (hC : C ≤ R) :
82     ∀ c ∈ C, ∀ z ∈ classSet I R, rightMul I c z ∈ classSet I R := by
83   rintro c hc _ ⟨r, hr, rfl⟩
84   exact ⟨r * c, R.mul_mem hr (hC hc), (qmk_right_mul I r c).symm⟩

87 /-- The right multipliers preserving `qA_{≥0}` form a subring. -/
88 def classSetStabilizer : Subring A where
89   carrier := {a | ∀ z ∈ classSet I R, rightMul I a z ∈ classSet I R}
90   zero_mem' := by
91     intro z _
92     refine ⟨0, R.zero_mem, ?_⟩
93     change MulOpposite.op (0 : A) • z = _
94     rw [MulOpposite.op_zero, zero_smul]
95     rfl
96   one_mem' := by
97     intro z hz
98     change MulOpposite.op (1 : A) • z ∈ _
99     rwa [MulOpposite.op_one, one_smul]
100   add_mem' := by
101     intro a b ha hb z hz
102     obtain ⟨r, hr, hra⟩ := ha z hz
103     obtain ⟨s, hs, hsb⟩ := hb z hz
104     refine ⟨r + s, R.add_mem hr hs, ?_⟩
105     change MulOpposite.op (a + b) • z = _
106     rw [MulOpposite.op_add, add_smul]
107     change rightMul I a z + rightMul I b z = _
108     rw [hra, hsb]
109     rfl
110   neg_mem' := by
111     intro a ha z hz
112     obtain ⟨r, hr, hra⟩ := ha z hz
113     refine ⟨-r, R.neg_mem hr, ?_⟩
114     change MulOpposite.op (-a) • z = _
115     rw [MulOpposite.op_neg, neg_smul]
116     change -rightMul I a z = _
117     rw [hra]
118     rfl
119   mul_mem' := by
120     intro a b ha hb z hz
121     have h := hb _ (ha z hz)
122     change MulOpposite.op (a * b) • z ∈ _
123     rw [MulOpposite.op_mul, mul_smul]
124     exact h

126 /-- **`qA_{≥0} = Q`.**  `qA_{≥0}` is stable under `C` and `p`, which generate `A`. -/
127 theorem step_4_3_classSet_eq_univ (C : Subring A) (p : A)
128     (hC : ∀ c ∈ C, ∀ z ∈ classSet I R, rightMul I c z ∈ classSet I R)
129     (hp : ∀ z ∈ classSet I R, rightMul I p z ∈ classSet I R)
130     (hgenerate : Subring.closure ((C : Set A) ∪ {p}) = ⊤) :
131     classSet I R = Set.univ := by
132   have htop : classSetStabilizer I R = ⊤ := by
133     apply top_unique
134     rw [← hgenerate]
135     apply Subring.closure_le.2
136     rintro a (ha | ha)
137     · exact hC a ha
138     · rw [Set.mem_singleton_iff.mp ha]
139       exact hp
140   apply Set.eq_univ_of_forall
141   intro z
142   obtain ⟨a, rfl⟩ := Submodule.Quotient.mk_surjective I z
143   have ha : a ∈ classSetStabilizer I R := htop ▸ Subring.mem_top a
144   have h := ha (qmk I 1) ⟨1, R.one_mem, rfl⟩
145   rwa [← qmk_right_mul, one_mul] at h

147 /-- Proposition 4.3 for a ring with a right ideal, assembled from the
148 paragraphs: equation (normal-surj) with `qA_{≥0} = Q` reads `Q = Qx₁`. -/
149 theorem proposition_4_3_of (C : Subring A) (x p u₀ : A)
150     (hx : x ∈ R) (hE : x * p ∈ R) (hC : C ≤ R)
151     (hnormal : ∀ r ∈ R, ∃ r' ∈ R, x * r = r' * x)
152     (hu₀ : u₀ ∈ R) (hres : 1 + u₀ * x ∈ I)
153     (hgenerate : Subring.closure ((C : Set A) ∪ {p}) = ⊤) :
154     Function.Surjective (rightMul I x) := by
155   have hsurj := step_4_3_normal_surj x u₀ hx hu₀ hres hnormal
156   have huniv := step_4_3_classSet_eq_univ C p
157     (step_4_3_stable_coefficients C hC) (step_4_3_stable_momentum x p hE hsurj) hgenerate
158   intro z
159   have hz : z ∈ rightMul I x '' classSet I R := by
160     rw [← hsurj, huniv]
161     trivial
162   obtain ⟨w, -, hw⟩ := hz
163   exact ⟨w, hw⟩
164 
165 end Generic

177 /-- `C ⊆ A_{≥0}`: no element of `C` contains `p`. -/
178 theorem transverseCoordinateAlgebra_le :
179     transverseCoordinateAlgebra k n ≤ presentedNonnegativePart k n := by

186   · rw [Set.mem_singleton_iff.mp ha]

189 /-- `C` and `p` together generate `A_n`. -/
190 theorem closure_transverseCoordinateAlgebra_momentum [Algebra ℚ k] :
191     Subring.closure ((transverseCoordinateAlgebra k n : Set (PresentedWeyl k (n + 1))) ∪
192       {presentedMomentum k n}) = ⊤ := by
193   apply top_unique
194   intro z _
195   have hz : presentedToIterated k (n + 1) z ∈
196       (⊤ : Subring (PairStage (B := IteratedPairStage k n))) := trivial
197   let := Stafford38.WeylEulerRemainder.iteratedRatAlgebra k n
198   rw [← closure_pairOldSubring_union_coordinate_momentum_eq_top] at hz
199   have hle : Subring.closure (((pairOldSubring (IteratedPairStage k n) :
200         Subring (PairStage (B := IteratedPairStage k n))) : Set _) ∪
201         {pairCoordinate, pairMomentum}) ≤
202       (Subring.closure ((transverseCoordinateAlgebra k n : Set (PresentedWeyl k (n + 1))) ∪
203         {presentedMomentum k n})).comap (fromIterated k n) := by
204     apply Subring.closure_le.2
205     rintro a (⟨b, rfl⟩ | ha)
206     · exact Subring.subset_closure (Or.inl (Subring.subset_closure
207         (Or.inl (fromIterated_coefficient_mem_transverse k n b))))
208     · simp only [Set.mem_insert_iff, Set.mem_singleton_iff] at ha
209       rcases ha with rfl | rfl
210       · rw [SetLike.mem_coe, Subring.mem_comap, fromIterated_coordinate]
211         exact Subring.subset_closure (Or.inl (Subring.subset_closure (Or.inr rfl)))
212       · rw [SetLike.mem_coe, Subring.mem_comap, fromIterated_momentum]
213         exact Subring.subset_closure (Or.inr rfl)
214   have h : fromIterated k n (presentedToIterated k (n + 1) z) ∈
215       Subring.closure ((transverseCoordinateAlgebra k n : Set (PresentedWeyl k (n + 1))) ∪
216         {presentedMomentum k n}) := Subring.mem_comap.mp (hle hz)
217   rwa [fromIterated_toIterated] at h

221 /-- **Proposition 4.3.** For `d̃` in the form (3.1), right multiplication by `x₁`
222 on `Q = A_n/𝓘` (Equation (3.2)) is surjective, i.e. `Q = Qx₁`. -/
223 theorem proposition_4_3 [Algebra ℚ k] {N : ℕ} {d : PresentedWeyl k (n + 1)}
224     (hform : Stafford38.Paper.Section3.IsMonicNormalForm k N d) :
225     Function.Surjective
226       (rightMul (Stafford38.Paper.Section3.idealI k N d) (presentedCoordinate k n)) := by
227   obtain ⟨u₀, hu₀, hres⟩ := lemma_4_2 hform
228   exact proposition_4_3_of (transverseCoordinateAlgebra k n)
229     (presentedCoordinate k n) (presentedMomentum k n) u₀
230     (coordinate_mem_nonnegativePart (Stafford38.Paper.Section2.convention_2_2_relations k n).1)
231     (euler_mem_nonnegativePart _ _)
232     (transverseCoordinateAlgebra_le k n)
233     (presented_coordinate_normal k n)
234     hu₀ hres (closure_transverseCoordinateAlgebra_momentum k n)
