11 /-!
12 # Lemma 4.10: the strict inequality `ℓ(U_𝔮) > ℓ(V_𝔮)`
13 
14 Paragraphs of the printed proof: Gabber's theorem (Theorem 2.11) makes
15 `J = √Ann M` involutive, hence its minimal primes [HTT D.3.4], and the bracket
16 argument keeps `x₁` out of them; this persists after localizing at `S = T∖𝔮`; the stable
17 power kernel `K_∞`; its two elementary facts; `ℓ(K_∞/x₁K_∞) = ℓ(V_𝔮)`; the split
18 `ℓ(U_𝔮) = ℓ(V_𝔮) + ℓ(Ẽ/x₁Ẽ)` with `Ẽ = M_𝔮/K_∞`; `Ẽ/x₁Ẽ ≠ 0` by Nakayama.
19 
20 The middle paragraphs are stated for a base ring `R` (where lengths are
21 measured, `R = T_𝔮`), a Noetherian coefficient algebra `C` (`C = k[x,ξ]_𝔮`), a
22 finite `C`-module `E` (`E = M_𝔮`) and `x ∈ C`, with `f = x·` viewed `R`-linearly.
23 In Lean `Ẽ/x₁Ẽ` is `E ⧸ (K_∞ ⊔ x₁E)`, which is the same module by the third
24 isomorphism theorem (`Submodule.quotientQuotientEquivQuotientSup`).
25 -/

48 /-- **`K_∞`.**  The chain `ker f ⊆ ker f² ⊆ ⋯` stabilizes (`E` is Noetherian), at
49 `K_∞ = ker f^m = ker f^{m+1}`; the layers `ker f^{i+1}/ker f^i` inject into `V`
50 via `v ↦ x₁^i v`, so `K_∞` has finite length with `V`. -/
51 theorem step_4_10_stable_kernel [IsNoetherianRing C] [Module.Finite C E] (x : C)
52     (hV : IsFiniteLength R (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R))) :
53     ∃ m : ℕ,
54       LinearMap.ker (((LinearMap.lsmul C E x).restrictScalars R) ^ m) =
55           LinearMap.ker (((LinearMap.lsmul C E x).restrictScalars R) ^ (m + 1)) ∧
56         IsFiniteLength R (LinearMap.ker (((LinearMap.lsmul C E x).restrictScalars R) ^ m)) := by
57   let fC : Module.End C E := LinearMap.lsmul C E x
58   let fR : Module.End R E := fC.restrictScalars R
59   obtain ⟨m, hm⟩ := PrincipalKoszulPositivity.exists_stable_kernel_power_smul (E := E) x
60   have hpow : ∀ j, fR ^ j = (fC ^ j).restrictScalars R := by
61     intro j
62     induction j with
63     | zero => ext z; simp
64     | succ j ih =>
65         rw [pow_succ, pow_succ, ih]
66         ext z
67         simp [fR, LinearMap.restrictScalars_apply, Module.End.mul_apply]
68   refine ⟨m, ?_, PrincipalKoszulFiniteTorsion.isFiniteLength_kernel_power fR hV m⟩
69   change LinearMap.ker (fR ^ m) = LinearMap.ker (fR ^ (m + 1))
70   rw [hpow, hpow, LinearMap.ker_restrictScalars, LinearMap.ker_restrictScalars]
71   exact congrArg (Submodule.restrictScalars R) hm

73 /-- **The two elementary facts.**  For `K_∞ = ker f^m = ker f^{m+1}`:
74 `f⁻¹K_∞ = K_∞` and `K_∞ ∩ fE = f(K_∞)`. -/
75 theorem step_4_10_two_facts (f : Module.End R E) (m : ℕ)
76     (hstable : LinearMap.ker (f ^ m) = LinearMap.ker (f ^ (m + 1))) :
77     (LinearMap.ker (f ^ m)).comap f = LinearMap.ker (f ^ m) ∧
78       LinearMap.ker (f ^ m) ⊓ LinearMap.range f = (LinearMap.ker (f ^ m)).map f := by
79   have hcomap : (LinearMap.ker (f ^ m)).comap f = LinearMap.ker (f ^ m) := by
80     ext z
81     rw [Submodule.mem_comap]
82     change (f ^ m) (f z) = 0 ↔ (f ^ m) z = 0
83     rw [← Module.End.mul_apply, (Commute.self_pow f m).eq.symm, Module.End.mul_apply]
84     simpa [Module.End.pow_apply, Function.iterate_succ_apply'] using
85       congrArg (fun K : Submodule R E ↦ z ∈ K) hstable.symm
86   refine ⟨hcomap, le_antisymm ?_ ?_⟩
87   · rintro w ⟨hwK, v, rfl⟩
88     have hv : v ∈ (LinearMap.ker (f ^ m)).comap f := hwK
89     rw [hcomap] at hv
90     exact ⟨v, hv, rfl⟩
91   · rintro w ⟨v, hv, rfl⟩
92     refine ⟨?_, v, rfl⟩
93     have : v ∈ (LinearMap.ker (f ^ m)).comap f := by rw [hcomap]; exact hv
94     exact this

96 /-- **`ℓ(K_∞/x₁K_∞) = ℓ(V)`.**  For an endomorphism `φ` of a finite-length module,
97 additivity along `0 → ker φ → L → φL → 0` and `0 → φL → L → L/φL → 0` gives
98 `ℓ(ker φ) = ℓ(L/φL)`; for `φ = f|_{K_∞}`, `ker φ = ker f` as `ker f ⊆ K_∞`. -/
99 theorem step_4_10_kernel_length (f : Module.End R E) (K : Submodule R E)
100     (hK : IsFiniteLength R K) (hcomap : K.comap f = K) :
101     Module.length R (K ⧸ LinearMap.range (f.restrict (p := K) (q := K)
102         (fun z hz => show z ∈ K.comap f from hcomap.symm ▸ hz))) =
103       Module.length R (LinearMap.ker f) := by
104   let fK : Module.End R K := f.restrict (fun z hz => show z ∈ K.comap f from hcomap.symm ▸ hz)
105   have hparts := isFiniteLength_iff_isNoetherian_isArtinian.mp hK
106   let : IsNoetherian R K := hparts.1
107   let : IsArtinian R K := hparts.2
108   have hkerle : LinearMap.ker f ≤ K := by
109     intro z hz
110     rw [← hcomap]
111     change f z ∈ K
112     rw [LinearMap.mem_ker.mp hz]
113     exact K.zero_mem
114   have hker : Module.length R (LinearMap.ker fK) = Module.length R (LinearMap.ker f) := by
115     have hk : LinearMap.ker fK = (LinearMap.ker f).comap K.subtype := by
116       simp [fK, LinearMap.ker_restrict]
117     rw [hk]
118     exact (Submodule.comapSubtypeEquivOfLe hkerle).length_eq
119   have hsource := (definition_4_6 (T := R) (M := K)).2 (LinearMap.ker fK).subtype
120     fK.rangeRestrict (Submodule.subtype_injective _)
121     (LinearMap.range_eq_top.mp (LinearMap.range_rangeRestrict fK))
122     (by rw [LinearMap.exact_iff, Submodule.range_subtype, LinearMap.ker_rangeRestrict])
123   have htarget := (definition_4_6 (T := R) (M := K)).2 (LinearMap.range fK).subtype
124     (LinearMap.range fK).mkQ (Submodule.subtype_injective _) (Submodule.mkQ_surjective _)
125     (LinearMap.exact_subtype_mkQ _)
126   rw [← hker]
127   apply ENat.add_left_injective_of_ne_top (Module.length_ne_top (R := R) (M := LinearMap.range fK))
128   change Module.length R (K ⧸ LinearMap.range fK) + Module.length R (LinearMap.range fK) =
129     Module.length R (LinearMap.ker fK) + Module.length R (LinearMap.range fK)
130   rw [add_comm (Module.length R (K ⧸ LinearMap.range fK)), ← htarget]
131   exact hsource

133 /-- **The split `ℓ(U) = ℓ(V) + ℓ(Ẽ/x₁Ẽ)`.**  `E/fE → Ẽ/fẼ` is onto with kernel
134 `(K_∞ + fE)/fE ≅ K_∞/(K_∞ ∩ fE) = K_∞/fK_∞`; additivity and
135 `ℓ(K_∞/fK_∞) = ℓ(V)` give the split. -/
136 theorem step_4_10_length_split (f : Module.End R E) (K : Submodule R E)
137     (hK : IsFiniteLength R K) (hcomap : K.comap f = K)
138     (hinter : K ⊓ LinearMap.range f = K.map f) :
139     Module.length R (E ⧸ LinearMap.range f) =
140       Module.length R (LinearMap.ker f) + Module.length R (E ⧸ (K ⊔ LinearMap.range f)) := by
141   have hstable : ∀ z ∈ K, f z ∈ K := fun z hz =>
142     show z ∈ K.comap f from hcomap.symm ▸ hz
143   let fK : Module.End R K := f.restrict hstable
144   have hpre : (LinearMap.range f).comap K.subtype = LinearMap.range fK := by
145     ext z
146     constructor
147     · intro hz
148       have hz' : (z : E) ∈ K.map f := hinter ▸ ⟨z.2, hz⟩
149       obtain ⟨w, hw, hwz⟩ := hz'
150       exact ⟨⟨w, hw⟩, Subtype.ext hwz⟩
151     · rintro ⟨w, rfl⟩
152       exact ⟨w.val, rfl⟩
153   let i : (K ⧸ LinearMap.range fK) →ₗ[R] (E ⧸ LinearMap.range f) :=
154     (LinearMap.range fK).mapQ (LinearMap.range f) K.subtype hpre.ge
155   let j : (E ⧸ LinearMap.range f) →ₗ[R] (E ⧸ (K ⊔ LinearMap.range f)) :=
156     (LinearMap.range f).mapQ (K ⊔ LinearMap.range f) LinearMap.id le_sup_right
157   have hi : Function.Injective i := by
158     rw [← LinearMap.ker_eq_bot]
159     dsimp [i]
160     rw [Submodule.ker_mapQ, hpre, Submodule.mkQ_map_self]
161   have hj : Function.Surjective j := by
162     rw [← LinearMap.range_eq_top]
163     simp [j, Submodule.range_mapQ]
164   have hexact : Function.Exact i j := by
165     rw [LinearMap.exact_iff]
166     simp [i, j, Submodule.range_mapQ, Submodule.ker_mapQ, Submodule.map_sup,
167       Submodule.mkQ_map_self]
168   have hsum := (definition_4_6 (T := R) (M := E ⧸ LinearMap.range f)).2 i j hi hj hexact
169   rw [hsum, step_4_10_kernel_length f K hK hcomap]

171 /-- **Nakayama paragraph: `Ẽ/x₁Ẽ ≠ 0`.**  Pick `𝔓` in the support of `E/x₁E ≠ 0`
172 and `𝔭' ⊆ 𝔓` minimal in `Supp E`; `x₁ ∉ 𝔭'`, so `K_∞` vanishes at `𝔭'` and
173 `Ẽ_{𝔭'} = E_{𝔭'} ≠ 0`; hence `Ẽ_𝔓 ≠ 0`, and as `x₁ ∈ 𝔓`, Nakayama gives
174 `(Ẽ/x₁Ẽ)_𝔓 ≠ 0`. -/
175 theorem step_4_10_residual_nonzero [IsNoetherianRing C] [Module.Finite C E] (x : C)
176     (havoid : ∀ p ∈ (Module.annihilator C E).minimalPrimes, x ∉ p)
177     (hU : Nontrivial (QuotSMulTop x E)) (m : ℕ) :
178     Nontrivial (E ⧸ (LinearMap.ker (((LinearMap.lsmul C E x).restrictScalars R) ^ m) ⊔
179       LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R))) := by
180   let fC : Module.End C E := LinearMap.lsmul C E x
181   let fR : Module.End R E := fC.restrictScalars R
182   -- `𝔓 ∈ Supp(E/x₁E)` and a minimal `𝔭' ⊆ 𝔓` of `Supp E`
183   obtain ⟨P, hPquot⟩ := Module.nonempty_support_of_nontrivial (R := C) (M := QuotSMulTop x E)
184   have hPE : P ∈ Module.support C E := by
185     have h := hPquot
186     rw [Module.support_quotSMulTop] at h
187     exact h.1
188   obtain ⟨p, hpmin, hpP⟩ := Ideal.exists_minimalPrimes_le (I := Module.annihilator C E)
189     (J := P.asIdeal) (Module.mem_support_iff_of_finite.mp hPE)
190   let p' : PrimeSpectrum C := ⟨p, hpmin.1.1⟩
191   have hp' : p' ∈ Module.support C E := Module.mem_support_iff_of_finite.mpr hpmin.1.2
192   have hnonzeroC : Nontrivial (E ⧸ (LinearMap.ker (fC ^ m) ⊔ LinearMap.range fC)) :=
193     StableTorsionResidualSupport.residual_nontrivial_of_support x m p' P hp' hpP
194       (havoid p hpmin) hPquot
195   have hpow : ∀ j, fR ^ j = (fC ^ j).restrictScalars R := by
196     intro j
197     induction j with
198     | zero => ext z; simp
199     | succ j ih =>
200         rw [pow_succ, pow_succ, ih]
201         ext z
202         simp [fR, LinearMap.restrictScalars_apply, Module.End.mul_apply]
203   have hPR : LinearMap.ker (fR ^ m) ⊔ LinearMap.range fR =
204       (LinearMap.ker (fC ^ m) ⊔ LinearMap.range fC).restrictScalars R := by
205     rw [hpow m]
206     simp only [LinearMap.ker_restrictScalars, Submodule.restrictScalars_sup]
207     rfl
208   apply Submodule.Quotient.nontrivial_iff.mpr
209   intro htop
210   apply (Submodule.Quotient.nontrivial_iff.mp hnonzeroC)
211   apply (Submodule.restrictScalars_eq_top_iff R C E).mp
212   change LinearMap.ker (fR ^ m) ⊔ LinearMap.range fR = ⊤ at htop
213   rw [← hPR, htop]

215 /-- The length comparison over a base ring, assembled from the paragraphs above:
216 `ℓ(U) = ℓ(V) + ℓ(Ẽ/x₁Ẽ) > ℓ(V)`. -/
217 theorem step_4_10_core [IsNoetherianRing C] [Module.Finite C E] (x : C)
218     (havoid : ∀ p ∈ (Module.annihilator C E).minimalPrimes, x ∉ p)
219     (hU : Nontrivial (QuotSMulTop x E))
220     (hV : IsFiniteLength R (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R))) :
221     Module.length R (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R)) >
222       Module.length R (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R)) := by
223   let f : Module.End R E := (LinearMap.lsmul C E x).restrictScalars R
224   obtain ⟨m, hstable, hK⟩ := step_4_10_stable_kernel x hV
225   obtain ⟨hcomap, hinter⟩ := step_4_10_two_facts f m hstable
226   have hsplit := step_4_10_length_split f _ hK hcomap hinter
227   have hres := step_4_10_residual_nonzero (R := R) x havoid hU m
228   have hpos : 0 < Module.length R (E ⧸ (LinearMap.ker (f ^ m) ⊔ LinearMap.range f)) :=
229     Module.length_pos
230   change Module.length R (E ⧸ LinearMap.range f) > Module.length R (LinearMap.ker f)
231   rw [hsplit]
232   have hVtop : Module.length R (LinearMap.ker f) ≠ ⊤ := Module.length_ne_top_iff.mpr hV
233   simpa [add_comm] using ENat.lt_add_left hVtop hpos

235 /-- **Localizing at `S = T∖𝔮`.**  `M_𝔮 = S⁻¹M` over `T_𝔮` is the coefficient
236 localization `k[x,ξ]_𝔮 = S⁻¹k[x,ξ]` acting on `S⁻¹M`; there `U_𝔮 = M_𝔮/x₁M_𝔮 ≠ 0`
237 (`𝔮 ∈ Supp_T U`) and `V_𝔮 = ker(x₁ on M_𝔮)`, so `step_4_10_core` applies. -/
238 theorem step_4_10_localized [IsNoetherianRing C] [Module.Finite C E] (x : C)
239     (q : PrimeSpectrum R)
240     (hqmem : q ∈ Module.support R
241       (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R)))
242     (hkerfinite : IsFiniteLength (Localization q.asIdeal.primeCompl)
243       (LocalizedModule q.asIdeal.primeCompl
244         (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R))))
245     (havoid : ∀ Q ∈ (Module.annihilator
246         (Localization (Algebra.algebraMapSubmonoid C q.asIdeal.primeCompl))
247         (LocalizedModule (Algebra.algebraMapSubmonoid C q.asIdeal.primeCompl) E)).minimalPrimes,
248       algebraMap C (Localization (Algebra.algebraMapSubmonoid C q.asIdeal.primeCompl)) x ∉ Q) :
249     Module.length (Localization q.asIdeal.primeCompl)
250         (LocalizedModule q.asIdeal.primeCompl
251           (E ⧸ LinearMap.range ((LinearMap.lsmul C E x).restrictScalars R))) >
252       Module.length (Localization q.asIdeal.primeCompl)
253         (LocalizedModule q.asIdeal.primeCompl
254           (LinearMap.ker ((LinearMap.lsmul C E x).restrictScalars R))) := by
255   let S : Submonoid R := q.asIdeal.primeCompl
256   let SC : Submonoid C := Algebra.algebraMapSubmonoid C S
257   let fC : Module.End C E := LinearMap.lsmul C E x
258   let fR : Module.End R E := fC.restrictScalars R
259   -- compare the base localization `M_𝔮 = S⁻¹M` with the coefficient localization
260   let : Module (Localization S) (LocalizedModule SC E) :=
261     Module.compHom _ (algebraMap (Localization S) (Localization SC))
262   let : IsScalarTower (Localization S) (Localization SC) (LocalizedModule SC E) :=
263     IsScalarTower.of_compHom (Localization S) (Localization SC) (LocalizedModule SC E)
264   let e : LocalizedModule S E ≃ₗ[Localization S] LocalizedModule SC E :=
265     localizedModuleComparison S
266   let gR : Module.End (Localization S) (LocalizedModule S E) := localizedMap S fR
267   let y : Localization SC := algebraMap C (Localization SC) x
268   let gC : Module.End (Localization SC) (LocalizedModule SC E) :=
269     LinearMap.lsmul (Localization SC) (LocalizedModule SC E) y
270   have heq (z : LocalizedModule S E) : e (gR z) = gC (e z) := by
271     rw [show e (gR z) = LocalizedModule.map SC fC (e z) by
272       exact localizedModuleComparison_natural S fC z]
273     induction e z using LocalizedModule.induction_on with
274     | _ m s => simp [gC, y, fC, LocalizedModule.smul'_mk]
275   let ek : LinearMap.ker gR ≃ₗ[Localization S]
276       LinearMap.ker (gC.restrictScalars (Localization S)) :=
277     LinearEquiv.ofBijective
278       (e.toLinearMap.domRestrict (LinearMap.ker gR) |>.codRestrict
279         (LinearMap.ker (gC.restrictScalars (Localization S))) (fun z => by
280           exact LinearMap.mem_ker.mpr (by simpa using (heq z).symm)))
281       (by
282         constructor
283         · intro a b hab
284           exact Subtype.ext (e.injective (congrArg Subtype.val hab))
285         · intro z
286           refine ⟨⟨e.symm z, ?_⟩, Subtype.ext (e.apply_symm_apply z)⟩
287           exact LinearMap.mem_ker.mpr (e.injective (by
288             rw [heq]
289             simpa using z.property)))
290   have herange : Submodule.map e.toLinearMap (LinearMap.range gR) =
291       LinearMap.range (gC.restrictScalars (Localization S)) := by
292     ext z
293     constructor
294     · rintro ⟨_, ⟨a, rfl⟩, rfl⟩
295       exact ⟨e a, by simpa using (heq a).symm⟩
296     · rintro ⟨a, rfl⟩
297       refine ⟨gR (e.symm a), ⟨e.symm a, rfl⟩, ?_⟩
298       simpa using heq (e.symm a)
299   let ec : (LocalizedModule S E ⧸ LinearMap.range gR) ≃ₗ[Localization S]
300       (LocalizedModule SC E ⧸ LinearMap.range (gC.restrictScalars (Localization S))) :=
301     Submodule.Quotient.equiv _ _ e herange
302   -- `𝔮 ∈ Supp_T U`, so `U_𝔮 = M_𝔮/x₁M_𝔮 ≠ 0`
303   have hnontrivBase : Nontrivial (LocalizedModule S (E ⧸ LinearMap.range fR)) := by
304     exact Module.mem_support_iff.mp hqmem
305   have hnontrivCokerR : Nontrivial (LocalizedModule S E ⧸ LinearMap.range gR) :=
306     not_subsingleton_iff_nontrivial.mp (by
307       intro hs
308       have : Subsingleton (LocalizedModule S (E ⧸ LinearMap.range fR)) :=
309         (localizedCokernelEquiv S fR).toEquiv.subsingleton_congr.mpr hs
310       exact not_subsingleton_iff_nontrivial.mpr hnontrivBase this)
311   have hnontrivCokerC : Nontrivial
312       (LocalizedModule SC E ⧸ LinearMap.range (gC.restrictScalars (Localization S))) :=
313     not_subsingleton_iff_nontrivial.mp (by
314       intro hs
315       exact not_subsingleton_iff_nontrivial.mpr hnontrivCokerR
316         (ec.toEquiv.subsingleton_congr.mpr hs))
317   have hrange : LinearMap.range gC =
318       y • (⊤ : Submodule (Localization SC) (LocalizedModule SC E)) := by
319     ext z
320     rw [LinearMap.mem_range, Submodule.mem_smul_pointwise_iff_exists]
321     simp only [Submodule.mem_top, true_and]
322     rfl
323   have hnontrivQuot : Nontrivial (QuotSMulTop y (LocalizedModule SC E)) := by
324     change Nontrivial (LocalizedModule SC E ⧸
325       y • (⊤ : Submodule (Localization SC) (LocalizedModule SC E)))
326     rw [← hrange]
327     exact hnontrivCokerC
328   have hkbase : IsFiniteLength (Localization S) (LinearMap.ker gR) :=
329     (localizedKernelEquiv S fR).isFiniteLength hkerfinite
330   have hkfinite : IsFiniteLength (Localization S)
331       (LinearMap.ker (gC.restrictScalars (Localization S))) :=
332     ek.isFiniteLength hkbase
333   -- the strict inequality over `T_𝔮`, from the paragraphs above
334   have hpos := step_4_10_core (R := Localization S) (C := Localization SC)
335     (E := LocalizedModule SC E) y havoid hnontrivQuot hkfinite
336   have h1 : Module.length (Localization S) (LocalizedModule S (E ⧸ LinearMap.range fR)) =
337       Module.length (Localization S)
338         (LocalizedModule SC E ⧸ LinearMap.range (gC.restrictScalars (Localization S))) :=
339     (localizedCokernelEquiv S fR).length_eq.trans ec.length_eq
340   have hek : Module.length (Localization S) (LinearMap.ker gR) =
341       Module.length (Localization S) (LinearMap.ker (gC.restrictScalars (Localization S))) := by
342     exact ek.length_eq
343   have h2 : Module.length (Localization S) (LocalizedModule S (LinearMap.ker fR)) =
344       Module.length (Localization S) (LinearMap.ker (gC.restrictScalars (Localization S))) :=
345     (localizedKernelEquiv S fR).length_eq.trans hek
346   change Module.length (Localization S) (LocalizedModule S (LinearMap.ker fR)) <
347     Module.length (Localization S) (LocalizedModule S (E ⧸ LinearMap.range fR))
348   rw [h1, h2]
349   exact hpos
