10 # Lemma 4.9: the page inequality `ℓ(U_𝔮) ≤ ℓ(V_𝔮)`
11 
12 The filtered complex `Q --f--> Q`, `G^p = F_{-p}Q`, is
13 `CanonicalFilteredTwoTerm.canonicalFilteredTwoTerm` (`complex k n N d hd`); its
14 source and target pages `A_r`, `B_r` are `SourceTotal r`, `TargetTotal r`
15 (direct sums over `p` of the printed quotients), `d_r` is `tangentialDrop r`.
16 The steps below are the paragraphs of the printed proof: the pages and their
17 `T`-structure; localization and finite length; exhaustion `(B_R)_𝔮 = 0` from
18 `Q = Qx₁` (Proposition 4.3); equation (additivity); the telescoping sum.
19 -/

48 /-- **Later pages have finite length.**  `A_{r+1} ⊆ A_r` and `B_{r+1}` is a
49 quotient of `B_r`, so finite length of `A_1, B_1` passes to all `r ≥ 1`. -/
50 theorem step_4_9_pages_finite_length
51     (d : ∀ r, A r →ₗ[R] B r)
52     (sourceSucc : ∀ r, A (r + 1) ≃ₗ[R] LinearMap.ker (d r))
53     (targetSucc : ∀ r, B (r + 1) ≃ₗ[R] B r ⧸ LinearMap.range (d r))
54     (hA1 : IsFiniteLength R (A 1)) (hB1 : IsFiniteLength R (B 1)) :
55     ∀ r, 1 ≤ r → IsFiniteLength R (A r) ∧ IsFiniteLength R (B r) := by
56   intro r hr
57   induction r, hr using Nat.le_induction with
58   | base => exact ⟨hA1, hB1⟩
59   | succ r _ ih =>
60       exact ⟨(sourceSucc r).symm.isFiniteLength
61           (IsFiniteLength.of_injective ih.1 (Submodule.subtype_injective _)),
62         (targetSucc r).symm.isFiniteLength
63           (IsFiniteLength.of_surjective ih.2 (Submodule.mkQ_surjective _))⟩
64 
65 /-- **Equation (additivity).**  `0 → A_{r+1} → A_r → B_r → B_{r+1} → 0` splits at
66 `im d_r` into two short exact sequences; additivity of length (Definition 4.6)
67 gives `ℓ(B_r) + ℓ(A_{r+1}) = ℓ(A_r) + ℓ(B_{r+1})`. -/
68 theorem step_4_9_additivity {R : Type*} [Ring R] {A₀ A₁ B₀ B₁ : Type v}
69     [AddCommGroup A₀] [Module R A₀] [AddCommGroup A₁] [Module R A₁]
70     [AddCommGroup B₀] [Module R B₀] [AddCommGroup B₁] [Module R B₁]
71     (d : A₀ →ₗ[R] B₀) (es : A₁ ≃ₗ[R] LinearMap.ker d)
72     (et : B₁ ≃ₗ[R] B₀ ⧸ LinearMap.range d) :
73     Module.length R B₀ + Module.length R A₁ = Module.length R A₀ + Module.length R B₁ := by
74   have hsource : Module.length R A₀ =
75       Module.length R (LinearMap.ker d) + Module.length R (LinearMap.range d) :=
76     (definition_4_6 (T := R) (M := A₀)).2 (LinearMap.ker d).subtype d.rangeRestrict
77       (Submodule.subtype_injective _)
78       (LinearMap.range_eq_top.mp (LinearMap.range_rangeRestrict d))
79       (by rw [LinearMap.exact_iff, Submodule.range_subtype, LinearMap.ker_rangeRestrict])
80   have htarget : Module.length R B₀ =
81       Module.length R (LinearMap.range d) + Module.length R (B₀ ⧸ LinearMap.range d) :=
82     (definition_4_6 (T := R) (M := B₀)).2 (LinearMap.range d).subtype (LinearMap.range d).mkQ
83       (Submodule.subtype_injective _) (Submodule.mkQ_surjective _)
84       (LinearMap.exact_subtype_mkQ _)
85   rw [es.length_eq, et.length_eq, hsource, htarget]
86   ac_rfl
87 
88 /-- **Summing (additivity) over `r = 1,…,R−1`.**  The intermediate terms cancel:
89 `ℓ(B_1) + ℓ(A_R) = ℓ(A_1) + ℓ(B_R)`. -/
90 theorem step_4_9_telescope (a b : ℕ → ℕ∞) (ha : ∀ r, a r ≠ ⊤) (hb : ∀ r, b r ≠ ⊤)
91     (hadd : ∀ r, 1 ≤ r → b r + a (r + 1) = a r + b (r + 1)) :
92     ∀ R, 1 ≤ R → b 1 + a R = a 1 + b R := by
93   have ha' : ∀ r, a r = ((a r).toNat : ℕ∞) := fun r => (ENat.natCast_toNat (ha r)).symm
94   have hb' : ∀ r, b r = ((b r).toNat : ℕ∞) := fun r => (ENat.natCast_toNat (hb r)).symm
95   have hadd' : ∀ r, 1 ≤ r → (b r).toNat + (a (r + 1)).toNat = (a r).toNat + (b (r + 1)).toNat :=
96     fun r hr => by
97       have h := hadd r hr
98       rw [ha' r, ha' (r + 1), hb' r, hb' (r + 1)] at h
99       exact_mod_cast h
100   intro R hR
101   rw [ha' 1, ha' R, hb' 1, hb' R]
102   norm_cast
103   induction R, hR using Nat.le_induction with
104   | base => exact add_comm _ _
105   | succ R hR ih =>
106       have h := hadd' R hR
107       omega
108 
109 end Generic

120 /-- **The pages and their `T`-structure.**  `A_{r+1} = ker d_r` and
121 `B_{r+1} = coker d_r`, `T`-linearly, and `A_1 ≅ V`, `B_1 ≅ U`.  The `T`-action on
122 the pages comes from the symbols of the transverse generators, whose commutators
123 lower the filtration (`generator_commutator_lowers`). -/
124 theorem step_4_9_pages (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
125     (∀ r, Nonempty ((complex k n N d hd).SourceTotal (r + 1) ≃ₗ[Tangential k n]
126         LinearMap.ker (tangentialDrop k n N d hd r))) ∧
127       (∀ r, Nonempty ((complex k n N d hd).TargetTotal (r + 1) ≃ₗ[Tangential k n]
128         ((complex k n N d hd).TargetTotal r ⧸ LinearMap.range (tangentialDrop k n N d hd r)))) ∧
129       Nonempty ((complex k n N d hd).SourceTotal 1 ≃ₗ[Tangential k n] kernelV k n N d) ∧
130       Nonempty ((complex k n N d hd).TargetTotal 1 ≃ₗ[Tangential k n] cokernelU k n N d) :=
131   ⟨fun r => ⟨tangentialSourceSuccEquiv k n N d hd r⟩,
132     fun r => ⟨tangentialTargetSuccEquiv k n N d hd r⟩,
133     ⟨firstSourceGradedKernelEquiv k n N d hd⟩, ⟨firstTargetGradedCokernelEquiv k n N d hd⟩⟩
134 
135 /-- `f : Q → Q` of the filtered complex is onto: this is `Q = Qx₁`
136 (Proposition 4.3). -/
137 theorem step_4_9_f_surjective (hform : IsMonicNormalForm k N d) :
138     Function.Surjective (complex k n N d (isPBWMonicAt_of_isMonicNormalForm k hform)).f := by
139   intro q
140   obtain ⟨z, rfl⟩ := Submodule.Quotient.mk_surjective
141     (rightIdealKSubmodule k (presentedCanonicalRightIdeal (k := k) n N d)) q
142   obtain ⟨y, hy⟩ := proposition_4_3 hform (qmk (presentedCanonicalRightIdeal (k := k) n N d) z)
143   obtain ⟨y', hy'⟩ := Submodule.Quotient.mk_surjective
144     (rightIdealKSubmodule k (presentedCanonicalRightIdeal (k := k) n N d)) y
145   rw [← hy'] at hy
146   refine ⟨Submodule.Quotient.mk y', ?_⟩
147   change rightMulLinearMap k (presentedCanonicalRightIdeal (k := k) n N d)
148       (presentedCoordinate k n) (Submodule.Quotient.mk y') = Submodule.Quotient.mk z
149   rw [rightMulLinearMap_mk]
150   exact hy

152 /-- **Exhaustion: `(B_R)_𝔮 = 0` for some `R`.**  The maps `B_1 → B_r` are onto with
153 increasing kernels `K_r`.  A homogeneous class `z ∈ G^p` of `B_1` has a preimage
154 `z = wx₁` (Proposition 4.3) with `w ∈ G^s`, so `z` vanishes in `B_r` once
155 `p − r + 1 ≤ s`; hence `⋃ K_r = (B_1)_𝔮`.  As `(B_1)_𝔮` is Noetherian the chain
156 stabilizes at some `K_R = (B_1)_𝔮`, and `(B_R)_𝔮 = 0`. -/
157 theorem step_4_9_exhaustion (hform : IsMonicNormalForm k N d)
158     (S : Submonoid (Tangential k n))
159     [IsNoetherian (Localization S) (LocalizedModule S
160       ((complex k n N d (isPBWMonicAt_of_isMonicNormalForm k hform)).TargetTotal 1))] :
161     ∃ R, Subsingleton (LocalizedModule S
162       ((complex k n N d (isPBWMonicAt_of_isMonicNormalForm k hform)).TargetTotal (R + 1))) :=
163   let hd := isPBWMonicAt_of_isMonicNormalForm k hform
164   AlgebraicAnalysis.exists_uniform_subsingleton_localized S
165     (fun r => tangentialBoundaryMap k n N d hd r)
166     (tangentialBoundaryMap_ker_mono k n N d hd)
167     (fun z => (complex k n N d hd).totalBoundaryMap_eventually_zero
168       (canonicalOrderFiltration_exhaustive k _) (step_4_9_f_surjective hform) z)
169     (fun r => tangentialBoundaryMap_surjective k n N d hd r)
170 
171 /-- **Lemma 4.9.** For every minimal prime `𝔮` of `Supp_T U`,
172 `ℓ_{T_𝔮}(U_𝔮) ≤ ℓ_{T_𝔮}(V_𝔮)`, for `d̃` in the form (3.1). -/
173 theorem lemma_4_9 (hform : IsMonicNormalForm k N d)
174     (q : PrimeSpectrum (Tangential k n))
175     (hqmem : q ∈ Module.support (Tangential k n) (cokernelU k n N d))
176     (hqmin : ∀ p ∈ Module.support (Tangential k n) (cokernelU k n N d),
177       p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal) :
178     Module.length (Localization q.asIdeal.primeCompl)
179         (LocalizedModule q.asIdeal.primeCompl (cokernelU k n N d)) ≤
180       Module.length (Localization q.asIdeal.primeCompl)
181         (LocalizedModule q.asIdeal.primeCompl (kernelV k n N d)) := by
182   have hd := isPBWMonicAt_of_isMonicNormalForm k hform
183   let S := q.asIdeal.primeCompl
184   let A : ℕ → Type _ := fun r => LocalizedModule S ((complex k n N d hd).SourceTotal r)
185   let B : ℕ → Type _ := fun r => LocalizedModule S ((complex k n N d hd).TargetTotal r)
186   let : ∀ r, AddCommGroup (A r) := fun r => by dsimp only [A]; infer_instance
187   let : ∀ r, Module (Localization S) (A r) := fun r => by dsimp only [A]; infer_instance
188   let : ∀ r, AddCommGroup (B r) := fun r => by dsimp only [B]; infer_instance
189   let : ∀ r, Module (Localization S) (B r) := fun r => by dsimp only [B]; infer_instance
190   -- the pages and their localization; exactness keeps the successor identities
191   obtain ⟨hsucc, htsucc, ⟨eV⟩, ⟨eU⟩⟩ := step_4_9_pages hd
192   let dd : ∀ r, A r →ₗ[Localization S] B r := fun r =>
193     localizedMap S (tangentialDrop k n N d hd r)
194   let es : ∀ r, A (r + 1) ≃ₗ[Localization S] LinearMap.ker (dd r) := fun r =>
195     (localizedEquiv S (hsucc r).some).trans (localizedKernelEquiv S _)
196   let et : ∀ r, B (r + 1) ≃ₗ[Localization S] B r ⧸ LinearMap.range (dd r) := fun r =>
197     (localizedEquiv S (htsucc r).some).trans (localizedCokernelEquiv S _)
198   let eV' : A 1 ≃ₗ[Localization S] LocalizedModule S (kernelV k n N d) := localizedEquiv S eV
199   let eU' : B 1 ≃ₗ[Localization S] LocalizedModule S (cokernelU k n N d) :=
200     localizedEquiv S eU
201   -- Lemma 4.8: `(A_1)_𝔮 = V_𝔮` and `(B_1)_𝔮 = U_𝔮` have finite length
202   obtain ⟨-, -, -, hfin⟩ := lemma_4_8 hform
203   obtain ⟨hU, hV⟩ := hfin q hqmem hqmin
204   have hfl := step_4_9_pages_finite_length dd es et (eV'.symm.isFiniteLength hV)
205     (eU'.symm.isFiniteLength hU)
206   -- exhaustion: `(B_R)_𝔮 = 0`
207   let : IsNoetherian (Localization S) (B 1) :=
208     (isFiniteLength_iff_isNoetherian_isArtinian.mp (hfl 1 le_rfl).2).1
209   obtain ⟨R', hR'⟩ := step_4_9_exhaustion hform S
210   -- equation (additivity) and the telescoping sum
211   have hlen := step_4_9_telescope
212     (fun r => if 1 ≤ r then Module.length (Localization S) (A r) else 0)
213     (fun r => if 1 ≤ r then Module.length (Localization S) (B r) else 0)
214     (fun r => by
215       split_ifs with h
216       · exact Module.length_ne_top_iff.mpr (hfl r h).1
217       · exact ENat.zero_ne_top)
218     (fun r => by
219       split_ifs with h
220       · exact Module.length_ne_top_iff.mpr (hfl r h).2
221       · exact ENat.zero_ne_top)
222     (fun r hr => by
223       simp only [hr, ite_true, show 1 ≤ r + 1 by omega]
224       exact step_4_9_additivity (dd r) (es r) (et r))
225     (R' + 1) (by omega)
226   simp only [le_refl, ite_true, show 1 ≤ R' + 1 by omega] at hlen
227   have hBR : Module.length (Localization S) (B (R' + 1)) = 0 :=
228     Module.length_eq_zero_iff.mpr hR'
229   -- `ℓ(U_𝔮) + ℓ((A_R)_𝔮) = ℓ(V_𝔮)`, and `ℓ((A_R)_𝔮) ≥ 0`
230   calc Module.length (Localization S) (LocalizedModule S (cokernelU k n N d))
231         = Module.length (Localization S) (B 1) := eU'.length_eq.symm
232     _ ≤ Module.length (Localization S) (B 1) + Module.length (Localization S) (A (R' + 1)) :=
233         le_self_add
234     _ = Module.length (Localization S) (A 1) := by rw [hlen, hBR, add_zero]
235     _ = Module.length (Localization S) (LocalizedModule S (kernelV k n N d)) := eV'.length_eq
236
