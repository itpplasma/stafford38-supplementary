29 /-- **Proof of Corollary 4.11.**  If `U ≠ 0` it has a minimal support prime `𝔮`;
30 Lemma 4.8 makes the lengths finite, and Lemmas 4.9 and 4.10 contradict one
31 another.  Hence `U = M/x₁M = 0`. -/
32 theorem step_4_11_cokernel_zero (hN : 0 < N)
33     (hform : IsMonicNormalForm k N d) :
34     Subsingleton (cokernelU k n N d) := by
35   obtain ⟨hU, -, -, -⟩ := lemma_4_8 hform
36   by_contra h
37   have : Nontrivial (cokernelU k n N d) := not_subsingleton_iff_nontrivial.mp h
38   obtain ⟨q, hqmem, hqmin⟩ :=
39     AlgebraicAnalysis.MinimalSupportExistence.exists_minimal_support_prime
40       (R := Tangential k n) (U := cokernelU k n N d)
41   exact not_lt_of_ge (lemma_4_9 hform q hqmem hqmin) (lemma_4_10 hN hform q hqmem hqmin)

43 /-- **Corollary 4.11.** Multiplication by `x₁` on `gr Q` is surjective, i.e.
44 `gr Q = (gr Q) x₁`, for `d̃` in the form (3.1) with `N ≥ 1`. -/
45 theorem corollary_4_11 (hN : 0 < N) (hform : IsMonicNormalForm k N d) :
46     Function.Surjective (fun z : GrQ k n N d => symbolCoordinate k n • z) := by
47   have hU := step_4_11_cokernel_zero hN hform
48   have htop : LinearMap.range (coordinateMap k n N d) = ⊤ :=
49     Submodule.Quotient.subsingleton_iff.mp hU
50   intro z
51   obtain ⟨w, hw⟩ := LinearMap.range_eq_top.mp htop z
52   exact ⟨w, (setup_coordinateMap_apply k n N d w).symm.trans hw⟩

54 /-- Corollary 4.11 in the form `M/x₁M = 0` over `k[x,ξ]` (`QuotSMulTop`), as
55 consumed by the support-avoidance argument of Section 6. -/
56 theorem corollary_4_11_quotSMulTop (hN : 0 < N)
57     (hform : IsMonicNormalForm k N d) :
58     Subsingleton (QuotSMulTop (symbolCoordinate k n) (GrQ k n N d)) := by
59   have hsurj := corollary_4_11 hN hform
60   change Subsingleton (GrQ k n N d ⧸ symbolCoordinate k n • (⊤ : Submodule (Symbol k n) _))
61   rw [Submodule.Quotient.subsingleton_iff, eq_top_iff]
62   intro z _
63   obtain ⟨w, rfl⟩ := hsurj z
64   exact Submodule.smul_mem_pointwise_smul w _ ⊤ trivial
