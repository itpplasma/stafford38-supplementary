10 # Lemma 4.2: the Euler residue `1 + u₀ x₁ ∈ 𝓘`
11 
12 The proof of Lemma 4.2 has three paragraphs: the Bezout identity for the
13 Euler products `R_±` of Lemma 4.1, the expansion `d̃ = p^N + Σ` giving
14 `1 + w ∈ 𝓘`, and the positivity `w ∈ A_{≥0} x₁`.  Each is a declaration below,
15 first for an arbitrary ring with a Weyl pair, then for the presented Weyl
16 algebra.
17 -/

36 /-- Products over lists of pairwise coprime factors are coprime. -/
37 theorem isCoprime_list_prod {R : Type*} [CommRing R] (l₁ l₂ : List R)
38     (h : ∀ a ∈ l₁, ∀ b ∈ l₂, IsCoprime a b) : IsCoprime l₁.prod l₂.prod := by
39   induction l₁ with
40   | nil => exact isCoprime_one_left
41   | cons a l₁ ih =>
42       rw [List.prod_cons]
43       refine IsCoprime.mul_left ?_ (ih fun a' ha' => h a' (List.mem_cons_of_mem _ ha'))
44       have ha := h a List.mem_cons_self
45       clear ih h
46       induction l₂ with
47       | nil => exact isCoprime_one_right
48       | cons b l₂ ih₂ =>
49           rw [List.prod_cons]
50           exact (ha b List.mem_cons_self).mul_right
51             (ih₂ fun b' hb' => ha b' (List.mem_cons_of_mem _ hb'))

53 /-- `R_+(E) = ∏_{r=1}^N (E+r)` and `R_-(E) = ∏_{r=0}^{N-1}(E−r)` have the disjoint
54 root sets `{−1,…,−N}` and `{0,…,N−1}`, so `gcd(R_+, R_-) = 1` in `ℚ[E]`. -/
55 theorem step_4_2_coprime (N : ℕ) :
56     IsCoprime
57       ((List.range' 1 N).map fun r : ℕ => (Polynomial.X + Polynomial.C (r : ℚ))).prod
58       ((List.range N).map fun r : ℕ => (Polynomial.X - Polynomial.C (r : ℚ))).prod := by
59   apply isCoprime_list_prod
60   intro a ha b hb
61   obtain ⟨r, hr, rfl⟩ := List.mem_map.mp ha
62   obtain ⟨s, -, rfl⟩ := List.mem_map.mp hb
63   have hr1 : 1 ≤ r := (List.mem_range'_1.mp hr).1
64   have hunit : IsUnit (-(r : ℚ) - s) := by
65     apply isUnit_iff_ne_zero.mpr
66     have : (1 : ℚ) ≤ r := by exact_mod_cast hr1
67     have : (0 : ℚ) ≤ s := by positivity
68     linarith
69   have := Polynomial.isCoprime_X_sub_C_of_isUnit_sub hunit
70   simpa [sub_eq_add_neg] using this

74 /-- Evaluating `R_±` at `E` gives the products of Lemma 4.1. -/
75 theorem step_4_2_eval_products (x p : A) (N : ℕ) :
76     eulerPolynomialEval x p
77         ((List.range' 1 N).map fun r : ℕ => (Polynomial.X + Polynomial.C (r : ℚ))).prod =
78       ((List.range' 1 N).map fun i : ℕ => eulerElement x p + i).prod ∧
79     eulerPolynomialEval x p
80         ((List.range N).map fun r : ℕ => (Polynomial.X - Polynomial.C (r : ℚ))).prod =
81       ((List.range N).map fun i : ℕ => eulerElement x p - i).prod := by
82   constructor <;>
83   · rw [map_list_prod, List.map_map]
84     congr 1
85     refine List.map_congr_left fun r _ => ?_
86     simp [eulerElement]

88 /-- **First paragraph of the proof of Lemma 4.2.**  By Lemma 4.1 (the
89 hypothesis `h41`, the product identities for `m = N`) and the extended Euclidean
90 algorithm in `ℚ[E]` there are `u, v ∈ ℚ[E]` with
91 `p^N x₁^N u(E) + x₁^N p^N v(E) = 1`. -/
92 theorem step_4_2_bezout (x p : A) (N : ℕ)
93     (h41 : 1 ≤ N →
94       x ^ N * p ^ N = ((List.range N).map fun i : ℕ => eulerElement x p - i).prod ∧
95         p ^ N * x ^ N = ((List.range' 1 N).map fun i : ℕ => eulerElement x p + i).prod) :
96     ∃ u v : Polynomial ℚ,
97       (p ^ N * x ^ N) * eulerPolynomialEval x p u +
98         (x ^ N * p ^ N) * eulerPolynomialEval x p v = 1 := by
99   obtain ⟨u, v, huv⟩ := step_4_2_coprime N
100   refine ⟨u, v, ?_⟩
101   rw [mul_comm u, mul_comm v] at huv
102   have h := congrArg (eulerPolynomialEval x p) huv
103   rw [map_add, map_mul, map_mul, map_one] at h
104   rcases Nat.eq_zero_or_pos N with rfl | hN
105   · simpa [add_comm] using h
106   obtain ⟨hminus, hplus⟩ := h41 hN
107   rw [(step_4_2_eval_products x p N).1, (step_4_2_eval_products x p N).2,
108     ← hplus, ← hminus] at h
109   exact h

111 /-- **Second paragraph.**  With `d̃ = p^N + Σ`,
112 `(d̃ x₁^N) u(E) + (x₁^N d̃) v(E) = 1 + w ∈ 𝓘`, where
113 `w = Σ x₁^N u(E) + x₁^N Σ v(E)` and `𝓘 = d̃A + x₁^N d̃A`. -/
114 theorem step_4_2_residue (x p d sigma : A) (N : ℕ) (u v : Polynomial ℚ)
115     (hd : d = p ^ N + sigma)
116     (hbezout : (p ^ N * x ^ N) * eulerPolynomialEval x p u +
117         (x ^ N * p ^ N) * eulerPolynomialEval x p v = 1) :
118     (d * x ^ N) * eulerPolynomialEval x p u + (x ^ N * d) * eulerPolynomialEval x p v =
119         1 + ((sigma * x ^ N) * eulerPolynomialEval x p u +
120           (x ^ N * sigma) * eulerPolynomialEval x p v) ∧
121       1 + ((sigma * x ^ N) * eulerPolynomialEval x p u +
122           (x ^ N * sigma) * eulerPolynomialEval x p v) ∈ canonicalRightIdeal x d N := by
123   have hid := positiveEulerResidue_identity x p d sigma
124     (eulerPolynomialEval x p u) (eulerPolynomialEval x p v) N hd hbezout
125   rw [← mul_assoc] at hid
126   refine ⟨hid, ?_⟩
127   rw [← hid]
128   refine (canonicalRightIdeal x d N).add_mem ?_ ?_
129   · rw [mul_assoc]
130     change MulOpposite.op (x ^ N * eulerPolynomialEval x p u) • d ∈
131       canonicalRightIdeal x d N
132     exact (canonicalRightIdeal x d N).smul_mem _ (firstGenerator_mem x d N)
133   · change MulOpposite.op (eulerPolynomialEval x p v) • (x ^ N * d) ∈
134       canonicalRightIdeal x d N
135     exact (canonicalRightIdeal x d N).smul_mem _ (secondGenerator_mem x d N)
136 
137 /-- **Third paragraph.**  Every term of `w` has positive Euler degree, hence
138 `w = u₀ x₁` with `u₀ ∈ A_{≥0}`.  The input is that `Σ x₁^N` and `x₁^N Σ` lie in
139 `A_{≥0} x₁` (positive Euler degree of `Σ x₁^N` and `x₁^N Σ`, supplied for the
140 presented `d̃` by `step_4_2_presented_remainder`); `u(E), v(E)` have degree zero
141 and pass to the left of `x₁` by normality. -/
142 theorem step_4_2_positive (C : Subring A) (x p sigma : A) (N : ℕ)
143     (hweyl : p * x = x * p + 1)
144     (hcomm : ∀ c : C, x * (c : A) = (c : A) * x)
145     (hscalar : ∀ q : ℚ, algebraMap ℚ A q ∈ C)
146     (U V : A) (hU : U ∈ nonnegativePart C x p) (hV : V ∈ nonnegativePart C x p)
147     (hright : sigma * x ^ N = U * x) (hleft : x ^ N * sigma = V * x)
148     (u v : Polynomial ℚ) :
149     ∃ u₀ ∈ nonnegativePart C x p,
150       (sigma * x ^ N) * eulerPolynomialEval x p u +
151         (x ^ N * sigma) * eulerPolynomialEval x p v = u₀ * x := by
152   obtain ⟨su, hsu⟩ := coordinate_mul_eulerPolynomial C x p hweyl hcomm hscalar u
153   obtain ⟨sv, hsv⟩ := coordinate_mul_eulerPolynomial C x p hweyl hcomm hscalar v
154   refine ⟨U * su + V * sv,
155     Subring.add_mem _ (Subring.mul_mem _ hU su.2) (Subring.mul_mem _ hV sv.2), ?_⟩
156   rw [hright, hleft, mul_assoc, mul_assoc, hsu, hsv]
157   noncomm_ring

159 /-- Lemma 4.2 for a Weyl pair, assembled from the three paragraphs, given the
160 identities of Lemma 4.1 for `m = N`. -/
161 theorem lemma_4_2_of_positive (C : Subring A) (x p d sigma : A) (N : ℕ)
162     (hweyl : p * x = x * p + 1)
163     (h41 : 1 ≤ N →
164       x ^ N * p ^ N = ((List.range N).map fun i : ℕ => eulerElement x p - i).prod ∧
165         p ^ N * x ^ N = ((List.range' 1 N).map fun i : ℕ => eulerElement x p + i).prod)
166     (hcomm : ∀ c : C, x * (c : A) = (c : A) * x)
167     (hscalar : ∀ q : ℚ, algebraMap ℚ A q ∈ C)
168     (hd : d = p ^ N + sigma)
169     (U V : A) (hU : U ∈ nonnegativePart C x p) (hV : V ∈ nonnegativePart C x p)
170     (hright : sigma * x ^ N = U * x) (hleft : x ^ N * sigma = V * x) :
171     ∃ u₀ ∈ nonnegativePart C x p, 1 + u₀ * x ∈ canonicalRightIdeal x d N := by
172   obtain ⟨u, v, hbez⟩ := step_4_2_bezout x p N h41
173   obtain ⟨-, hmem⟩ := step_4_2_residue x p d sigma N u v hd hbez
174   obtain ⟨u₀, hu₀, hw⟩ :=
175     step_4_2_positive C x p sigma N hweyl hcomm hscalar U V hU hV hright hleft u v
176   exact ⟨u₀, hu₀, hw ▸ hmem⟩
177 
178 end Generic

278 /-- **Positivity of the remainder for `d̃`.**  For the normalized `d̃` with
279 `d̃ = p^N + Σ`, both `Σ x₁^N` and `x₁^N Σ` lie in `A_{≥0} x₁`.  The paper argues by
280 Euler degree of the terms `p^j x₁^l b` (`j < N`); Lean writes `Σ` in the PBW
281 basis of the Ore tower and factors each monomial `b x₁^a p^j` with `j < N`
282 (`presentedRemainder_hasPositiveCoordinateFactor`), then transports back. -/
283 theorem step_4_2_presented_remainder {N : ℕ} {d : PresentedWeyl k (n + 1)}
284     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
285     ∃ U ∈ presentedNonnegativePart k n, ∃ V ∈ presentedNonnegativePart k n,
286       (d - presentedMomentum k n ^ N) * presentedCoordinate k n ^ N =
287           U * presentedCoordinate k n ∧
288         presentedCoordinate k n ^ N * (d - presentedMomentum k n ^ N) =
289           V * presentedCoordinate k n := by
290   obtain ⟨U, V, hU, hV⟩ := presentedRemainder_hasPositiveCoordinateFactor k n N hd
291   obtain ⟨σ, hσdef⟩ : ∃ σ : PairStage (B := IteratedPairStage k n),
292       σ = presentedToIterated k (n + 1) d - stageMomentum k n ^ N := ⟨_, rfl⟩
293   have hU' : σ * pairCoordinate ^ N = (U : PairStage (B := IteratedPairStage k n)) *
294       pairCoordinate := by rw [hσdef]; exact hU
295   have hV' : pairCoordinate ^ N * σ = (V : PairStage (B := IteratedPairStage k n)) *
296       pairCoordinate := by rw [hσdef]; exact hV
297   have hσ : fromIterated k n σ = d - presentedMomentum k n ^ N := by
298     rw [hσdef]
299     change (presentedIteratedEquiv k (n + 1)).symm
300       (presentedIteratedEquiv k (n + 1) d - stageMomentum k n ^ N) = _
301     rw [map_sub, AlgEquiv.symm_apply_apply, map_pow]
302     congr 2
303     exact fromIterated_momentum k n
304   refine ⟨fromIterated k n U, iterated_eulerSubring_le k n U U.2,
305     fromIterated k n V, iterated_eulerSubring_le k n V V.2, ?_, ?_⟩
306   · have h := congrArg (fromIterated k n) hU'
307     rw [RingHom.map_mul (fromIterated k n) σ] at h
308     rw [RingHom.map_mul (fromIterated k n) (U : PairStage (B := IteratedPairStage k n))] at h
309     rw [RingHom.map_pow (fromIterated k n) pairCoordinate N] at h
310     rw [hσ, fromIterated_coordinate] at h
311     exact h
312   · have h := congrArg (fromIterated k n) hV'
313     rw [RingHom.map_mul (fromIterated k n) _ σ] at h
314     rw [RingHom.map_mul (fromIterated k n) (V : PairStage (B := IteratedPairStage k n))] at h
315     rw [RingHom.map_pow (fromIterated k n) pairCoordinate N] at h
316     rw [hσ, fromIterated_coordinate] at h
317     exact h

319 /-- **Lemma 4.2.** For `d̃` in the form (3.1) and `𝓘 = d̃A_n + x₁^N d̃A_n`
320 (Equation (3.2)), there is `u₀ ∈ A_{≥0}` with `1 + u₀ x₁ ∈ 𝓘`.  The proof uses
321 (3.1) only through its consequence `IsPBWMonicAt`
322 (`Section3.isPBWMonicAt_of_isMonicNormalForm`), and cites Lemma 4.1 for `m = N`. -/
323 theorem lemma_4_2 {N : ℕ} {d : PresentedWeyl k (n + 1)}
324     (hform : Stafford38.Paper.Section3.IsMonicNormalForm k N d) :
325     ∃ u₀ ∈ presentedNonnegativePart k n,
326       1 + u₀ * presentedCoordinate k n ∈ Stafford38.Paper.Section3.idealI k N d := by
327   let := presentedRatAlgebra k n
328   have hd := Stafford38.Paper.Section3.isPBWMonicAt_of_isMonicNormalForm k hform
329   obtain ⟨U, hU, V, hV, hright, hleft⟩ := step_4_2_presented_remainder hd
330   exact lemma_4_2_of_positive (transverse k n) (presentedCoordinate k n)
331     (presentedMomentum k n) d (d - presentedMomentum k n ^ N) N
332     (Stafford38.Paper.Section2.convention_2_2_relations k n).1
333     (fun hN => ⟨(lemma_4_1 k n N hN).1, (lemma_4_1 k n N hN).2.1⟩)
334     (transverse_commutes_coordinate k n) (rat_mem_transverse k n) (by abel) U V hU hV hright hleft
