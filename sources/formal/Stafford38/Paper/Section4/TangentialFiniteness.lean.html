10 # Lemma 4.8: tangential finiteness
11 
12 The three paragraphs of the printed proof: `U, V` are finite over `T`;
13 `Supp_T V ⊆ Supp_T U` (Lemma 4.7 and Vasconcelos' theorem); at a minimal prime
14 `𝔮` of `Supp_T U`, `U_𝔮` and `V_𝔮` have finite length.  The last two are proved
15 for any commutative Noetherian algebra `T → C` and finite `C`-module.
16 -/

28 /-- **Vasconcelos paragraph.**  For a finite module `M` over a commutative
29 Noetherian ring `C` and `f ∈ End_C M`, `Supp_C(ker f) ⊆ Supp_C(coker f)`: if
30 `(coker f)_𝔔 = coker(f_𝔔) = 0`, then `f_𝔔` is a surjective endomorphism of a finite
31 module, hence injective, and `(ker f)_𝔔 = ker(f_𝔔) = 0`. -/
32 theorem step_4_8_vasconcelos {C E : Type*} [CommRing C] [IsNoetherianRing C]
33     [AddCommGroup E] [Module C E] [Module.Finite C E] (f : Module.End C E) :
34     Module.support C (LinearMap.ker f) ⊆ Module.support C (E ⧸ LinearMap.range f) := by
35   intro Q hQ
36   by_contra hc
37   let S := Q.asIdeal.primeCompl
38   have hzero : Subsingleton (LocalizedModule S (E ⧸ LinearMap.range f)) :=
39     Module.notMem_support_iff.mp hc
40   -- exactness of localization: `(coker f)_𝔔 = coker(f_𝔔)`
41   have hexc := (definition_4_4 S).2.2 f (LinearMap.range f).mkQ
42     (LinearMap.exact_map_mkQ_range f)
43   have hsurj : Function.Surjective (LocalizedModule.map S f) := by
44     intro z
45     exact (hexc z).mp (Subsingleton.elim _ _)
46   -- Vasconcelos: a surjective endomorphism of a finite module is injective
47   have hinj : Function.Injective (LocalizedModule.map S f) :=
48     IsNoetherian.injective_of_surjective_endomorphism _ hsurj
49   -- exactness again: `(ker f)_𝔔 = ker(f_𝔔) = 0`
50   have hexk := (definition_4_4 S).2.2 (LinearMap.ker f).subtype f
51     (LinearMap.exact_subtype_ker_map f)
52   have hsub : Subsingleton (LocalizedModule S (LinearMap.ker f)) := by
53     refine ⟨fun a b => LocalizedModule.map_injective S _ (Submodule.subtype_injective _) ?_⟩
54     have ha : LocalizedModule.map S f (LocalizedModule.map S (LinearMap.ker f).subtype a) =
55         LocalizedModule.map S f (LocalizedModule.map S (LinearMap.ker f).subtype b) := by
56       rw [hexk.apply_apply_eq_zero, hexk.apply_apply_eq_zero]
57     exact hinj ha
58   exact Module.notMem_support_iff.mpr hsub hQ

60 /-- **Second paragraph: `Supp_T V ⊆ Supp_T U`.**  For `𝔭 ∈ Supp_T V` pick `v` with
61 `Ann_T(v) ⊆ 𝔭` (Lemma 4.7); `Ann_{k[x,ξ]}(v)` misses `S = T∖𝔭`, so lies in a prime
62 `𝔔` with `𝔔 ∩ S = ∅`; `𝔔 ∈ Supp(ker f)` (Lemma 4.7), hence `𝔔 ∈ Supp(coker f)`
63 (Vasconcelos), so some `u ∈ U` has `Ann(u) ⊆ 𝔔` (Lemma 4.7), whence
64 `Ann_T(u) ⊆ 𝔭` and `𝔭 ∈ Supp_T U` (Lemma 4.7). -/
65 theorem step_4_8_support_inclusion {R C E : Type*} [CommRing R] [CommRing C] [Algebra R C]
66     [AddCommGroup E] [Module C E] [Module R E] [IsScalarTower R C E]
67     [IsNoetherianRing C] [Module.Finite C E] (f : Module.End C E) :
68     Module.support R (LinearMap.ker (f.restrictScalars R)) ⊆
69       Module.support R (E ⧸ LinearMap.range (f.restrictScalars R)) := by
70   intro p hp
71   obtain ⟨v, hv⟩ := (lemma_4_7 p).mp hp
72   let vC : LinearMap.ker f := ⟨v.1, v.2⟩
73   let I : Ideal C := (C ∙ vC).annihilator
74   have hdisj : Disjoint (I : Set C) (p.asIdeal.primeCompl.map (algebraMap R C)) := by
75     rw [Set.disjoint_left]
76     rintro c hc ⟨r, hr, rfl⟩
77     have hrv : r • v = 0 := by
78       apply Subtype.ext
79       change r • v.1 = 0
80       have hcv : algebraMap R C r • vC = 0 :=
81         (Submodule.mem_annihilator_span_singleton _ _).mp hc
82       simpa [vC, IsScalarTower.algebraMap_smul C r v.1] using congrArg Subtype.val hcv
83     exact hr (hv ((Submodule.mem_annihilator_span_singleton _ _).mpr hrv))
84   obtain ⟨Q, hQprime, hIQ, hQdisj⟩ := I.exists_le_prime_disjoint _ hdisj
85   let Q' : PrimeSpectrum C := ⟨Q, hQprime⟩
86   have hQker : Q' ∈ Module.support C (LinearMap.ker f) := (lemma_4_7 Q').mpr ⟨vC, hIQ⟩
87   have hQcoker := step_4_8_vasconcelos f hQker
88   obtain ⟨u, hu⟩ := (lemma_4_7 Q').mp hQcoker
89   refine (lemma_4_7 p).mpr ⟨u, fun r hr => ?_⟩
90   have hmap : algebraMap R C r ∈ (C ∙ u).annihilator := by
91     rw [Submodule.mem_annihilator_span_singleton]
92     have hru : r • u = 0 := (Submodule.mem_annihilator_span_singleton _ _).mp hr
93     simpa [IsScalarTower.algebraMap_smul C r u] using hru
94   by_contra hrp
95   exact Set.disjoint_left.mp hQdisj (hu hmap) ⟨r, hrp, rfl⟩

97 /-- A minimal prime of `Supp_T L` (`L` finite over the Noetherian `T`) is minimal
98 over `Ann_T L`, since `Supp L = V(Ann L)` [Stacks 00L2], so `L_𝔮` has finite
99 length [Atiyah–Macdonald 8.5, Stacks 00KH]. -/
100 theorem step_4_8_localized_isFiniteLength {R L : Type*} [CommRing R] [IsNoetherianRing R]
101     [AddCommGroup L] [Module R L] [Module.Finite R L] (q : PrimeSpectrum R)
102     (hqmem : q ∈ Module.support R L)
103     (hq : ∀ p ∈ Module.support R L, p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal) :
104     IsFiniteLength (Localization q.asIdeal.primeCompl)
105       (LocalizedModule q.asIdeal.primeCompl L) := by
106   have hmin : q.asIdeal ∈ (Module.annihilator R L).minimalPrimes := by
107     obtain ⟨p, hp, hpq⟩ := Ideal.exists_minimalPrimes_le
108       (Module.mem_support_iff_of_finite.mp hqmem)
109     have hp' : (⟨p, hp.1.1⟩ : PrimeSpectrum R) ∈ Module.support R L :=
110       Module.mem_support_iff_of_finite.mpr hp.1.2
111     have heq : p = q.asIdeal := le_antisymm hpq (hq _ hp' hpq)
112     simpa [heq] using hp
113   exact AlgebraicAnalysis.MinimalPrimeFiniteLengthLocalization.localizedModule_isFiniteLength
114     q.asIdeal hmin

116 /-- **Third paragraph.**  At a minimal prime `𝔮` of `Supp_T U`, the supports of
117 `U_𝔮` and `V_𝔮` are contained in `{𝔮T_𝔮}` (for `V` by `Supp V ⊆ Supp U`), so
118 both have finite length over `T_𝔮` (`V_𝔮` possibly zero). -/
119 theorem step_4_8_finite_length {R U V : Type*} [CommRing R] [IsNoetherianRing R]
120     [AddCommGroup U] [Module R U] [Module.Finite R U]
121     [AddCommGroup V] [Module R V] [Module.Finite R V]
122     (hsupp : Module.support R V ⊆ Module.support R U) (q : PrimeSpectrum R)
123     (hqmem : q ∈ Module.support R U)
124     (hq : ∀ p ∈ Module.support R U, p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal) :
125     IsFiniteLength (Localization q.asIdeal.primeCompl)
126         (LocalizedModule q.asIdeal.primeCompl U) ∧
127       IsFiniteLength (Localization q.asIdeal.primeCompl)
128         (LocalizedModule q.asIdeal.primeCompl V) := by
129   refine ⟨step_4_8_localized_isFiniteLength q hqmem hq, ?_⟩
130   by_cases hk : q ∈ Module.support R V
131   · exact step_4_8_localized_isFiniteLength q hk fun p hp hpq => hq p (hsupp hp) hpq
132   · have : Subsingleton (LocalizedModule q.asIdeal.primeCompl V) :=
133       Module.notMem_support_iff.mp hk
134     exact IsFiniteLength.of_subsingleton

145 /-- **First paragraph: `U` and `V` are finite over `T`.**  `M = gr Q` is finite
146 over `T[x₁]` because `P` annihilates `M` and is monic of degree `N` in `ξ₁`

156 /-- **Lemma 4.8.** `U` and `V` are finite over `T`, `Supp_T V ⊆ Supp_T U`, and at a
157 minimal prime `𝔮` of `Supp_T U`, `U_𝔮` and `V_𝔮` have finite length over `T_𝔮`.
158 Here `d̃` is in the form (3.1), used through `IsPBWMonicAt`. -/
159 theorem lemma_4_8 (hform : IsMonicNormalForm k N d) :
160     Module.Finite (Tangential k n) (cokernelU k n N d) ∧
161       Module.Finite (Tangential k n) (kernelV k n N d) ∧
162       Module.support (Tangential k n) (kernelV k n N d) ⊆
163         Module.support (Tangential k n) (cokernelU k n N d) ∧
164       ∀ q ∈ Module.support (Tangential k n) (cokernelU k n N d),
165         (∀ p ∈ Module.support (Tangential k n) (cokernelU k n N d),
166           p.asIdeal ≤ q.asIdeal → q.asIdeal ≤ p.asIdeal) →
167         IsFiniteLength (Localization q.asIdeal.primeCompl)
168             (LocalizedModule q.asIdeal.primeCompl (cokernelU k n N d)) ∧
169           IsFiniteLength (Localization q.asIdeal.primeCompl)
170             (LocalizedModule q.asIdeal.primeCompl (kernelV k n N d)) := by
171   have hd := isPBWMonicAt_of_isMonicNormalForm k hform
172   obtain ⟨hU, hV⟩ := step_4_8_finite hd
173   have hsupp : Module.support (Tangential k n) (kernelV k n N d) ⊆
174       Module.support (Tangential k n) (cokernelU k n N d) := by
175     have h := step_4_8_support_inclusion (R := Tangential k n)
176       (LinearMap.lsmul (Symbol k n) (GrQ k n N d) (symbolCoordinate k n))
177     rwa [setup_coordinateMap_eq] at h
178   exact ⟨hU, hV, hsupp, fun q hqmem hq => step_4_8_finite_length hsupp q hqmem hq⟩
