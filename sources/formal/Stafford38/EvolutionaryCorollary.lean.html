22 variable {D : Type*} [Ring D] {x p : D}
23 
24 /-- For a Weyl pair and a finite polynomial potential whose coefficients
25 commute separately with both generators, the two displayed right multiples
26 give a Stafford certificate with fixed left source `x ^ r`. -/
27 theorem evolution_stafford (k : Type*) [Field k] [CharZero k] [Algebra k D]
28     (hw : p * x = x * p + 1) {r : ℕ} (hr : 0 < r) (terms : List (D × ℕ))
29     (hx : ∀ t ∈ terms, t.1 * x = x * t.1)
30     (hp : ∀ t ∈ terms, t.1 * p = p * t.1) :
31     ∃ R S : D, (p ^ r - potentialSum x terms) * R
32         + x ^ r * (p ^ r - potentialSum x terms) * S = 1 := by
33   obtain ⟨α, β, hαβ⟩ := exists_bezout (k := k) (D := D) hr terms
34   exact evolution_stafford_certificate hw r terms hx hp α β hαβ
35 
36 /-- Exact paper-facing form. No commutativity among the coefficients is
37 assumed: each displayed coefficient need only commute with `x` and `p`. -/
38 theorem evolutionaryCorollary (k : Type*) [Field k] [CharZero k] [Algebra k D]
39     (hw : p * x = x * p + 1) {r : ℕ} (hr : 0 < r) (terms : List (D × ℕ))
40     (hx : ∀ t ∈ terms, t.1 * x = x * t.1)
41     (hp : ∀ t ∈ terms, t.1 * p = p * t.1)
42     {d : D} (hd : d = p ^ r - potentialSum x terms) :
43     ∃ R S : D, 1 = d * R + x ^ r * d * S := by
44   obtain ⟨R, S, h⟩ := evolution_stafford k hw hr terms hx hp
45   exact ⟨R, S, by simpa [hd] using h.symm⟩

53 local notation "ιB" => (Algebra.TensorProduct.includeLeft : B →ₐ[k] Dᵣ)
54 local notation "ιW" => (Algebra.TensorProduct.includeRight : W₁ →ₐ[k] Dᵣ)

56 /-- The evolutionary certificate in the literal tensor algebra
57 `B ⊗[k] A₁(k)`. This asserts no identification with a geometric ring of
58 differential operators. -/
59 theorem tensorEvolutionaryCorollary {r : ℕ} (hr : 0 < r)
60     (terms : List (B × ℕ)) {d : Dᵣ}
61     (hd : d = ιW (presentedMomentum k 0) ^ r -
62       potentialSum (ιW (presentedCoordinate k 0))
63         (terms.map fun t => (ιB t.1, t.2))) :
64     ∃ R S : Dᵣ, 1 = d * R + ιW (presentedCoordinate k 0) ^ r * d * S := by
65   let lifted : List (Dᵣ × ℕ) := terms.map fun t => (ιB t.1, t.2)
66   have hcomm : ∀ (b : B) (w : W₁), ιB b * ιW w = ιW w * ιB b := by
67     intro b w
68     simp [Algebra.TensorProduct.includeLeft_apply,
69       Algebra.TensorProduct.includeRight_apply,
70       Algebra.TensorProduct.tmul_mul_tmul]
71   have hx : ∀ t ∈ lifted,
72       t.1 * ιW (presentedCoordinate k 0) =
73         ιW (presentedCoordinate k 0) * t.1 := by
74     rintro t ht
75     rcases List.mem_map.mp ht with ⟨⟨b, n⟩, _, rfl⟩
76     exact hcomm b (presentedCoordinate k 0)
77   have hp : ∀ t ∈ lifted,
78       t.1 * ιW (presentedMomentum k 0) =
79         ιW (presentedMomentum k 0) * t.1 := by
80     rintro t ht
81     rcases List.mem_map.mp ht with ⟨⟨b, n⟩, _, rfl⟩
82     exact hcomm b (presentedMomentum k 0)
83   have hw : ιW (presentedMomentum k 0) * ιW (presentedCoordinate k 0) =
84       ιW (presentedCoordinate k 0) * ιW (presentedMomentum k 0) + 1 := by
85     calc
86       _ = ιW (presentedMomentum k 0 * presentedCoordinate k 0) :=
87         (map_mul ιW _ _).symm
88       _ = ιW (presentedCoordinate k 0 * presentedMomentum k 0 + 1) :=
89         congrArg ιW (presentedMomentum_mul_coordinate k 0)
90       _ = _ := by rw [map_add, map_mul, map_one]
91   exact evolutionaryCorollary
92     (D := Dᵣ) (x := ιW (presentedCoordinate k 0))
93     (p := ιW (presentedMomentum k 0)) k hw hr lifted hx hp hd
94
