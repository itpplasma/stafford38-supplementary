20 /-- Stafford's *stably cyclic* [Sta78, p. 435], as quoted in the proof of
21 Corollary 6.3 (tex line 1619): "given `l ≥ 1` and `a_1, …, a_{l+1} ∈ T` that
22 generate `T`, there exist `f_i ∈ A_n` such that `a_i + a_{l+1} f_i`,
23 `i = 1, …, l`, also generate `T`". -/
24 def IsStablyCyclic (A : Type u) (T : Type v) [Ring A] [AddCommGroup T] [Module Aᵐᵒᵖ T] :
25     Prop :=
26   ∀ l ≥ 1, ∀ a : Fin (l + 1) → T, Submodule.span Aᵐᵒᵖ (Set.range a) = ⊤ →
27     ∃ f : Fin l → A, Submodule.span Aᵐᵒᵖ
28       (Set.range fun i : Fin l => a i.castSucc + (MulOpposite.op (f i) : Aᵐᵒᵖ) • a (Fin.last l)) =
29         ⊤
30 
31 /-- **Corollary 6.3, stable cyclicity** (tex lines 75, 108, 1619).  Every
32 finitely generated torsion right `A_n(k)`-module is stably cyclic. -/
33 def PaperCorollary_6_3_stablyCyclic : Prop :=
34   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (M : Type v) [AddCommGroup M]
35     [Module (WeylAlg k n)ᵐᵒᵖ M] [Module.Finite (WeylAlg k n)ᵐᵒᵖ M],
36     Stafford38.TorsionCyclicity.IsRightTorsion (A := WeylAlg k n) (M := M) →
37       IsStablyCyclic (WeylAlg k n) M

39 /-- **Section 7, left-handed identity** (tex lines 1629–1630): every nonzero
40 `d ∈ A_n(k)` has `R, S, F` with `1 = Rd + SdF`. -/
41 def LeftHandedIdentity : Prop :=
42   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (d : WeylAlg k n), d ≠ 0 →
43     ∃ R S F : WeylAlg k n, (1 : WeylAlg k n) = R * d + S * d * F
44 
45 /-- Every element of a left module is killed by some nonzero scalar. -/
46 def IsLeftTorsion (A : Type u) (M : Type v) [Ring A] [AddCommGroup M] [Module A M] : Prop :=
47   ∀ m : M, ∃ d : A, d ≠ 0 ∧ d • m = 0
48 
49 /-- A left module `M` is cyclic: `M = Am` for some `m ∈ M`. -/
50 def IsLeftCyclicModule (A : Type u) (M : Type v) [Ring A] [AddCommGroup M] [Module A M] :
51     Prop :=
52   ∃ m : M, ∀ x : M, ∃ a : A, x = a • m
53 
54 /-- **Section 7, Corollary 6.3 for left torsion modules** (tex line 1630):
55 every finitely generated torsion left `A_n(k)`-module is cyclic. -/
56 def LeftCorollary_6_3 : Prop :=
57   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (M : Type v) [AddCommGroup M]
58     [Module (WeylAlg k n) M] [Module.Finite (WeylAlg k n) M],
59     IsLeftTorsion (WeylAlg k n) M → IsLeftCyclicModule (WeylAlg k n) M

61 /-- `P` is a `k`-linear differential operator of order at most `N` on the
62 commutative `k`-algebra `B` (Grothendieck): every iterated commutator
63 `[…[[P, b₀], b₁], …, b_N]` with multiplication operators vanishes, where
64 `[Q, b] = Q ∘ b − b ∘ Q`. -/
65 def IsDifferentialOperatorOfOrder (k : Type u) (B : Type v) [CommRing k] [CommRing B]
66     [Algebra k B] (N : ℕ) (P : Module.End k B) : Prop :=
67   ∀ b : List B, b.length = N + 1 →
68     b.foldl (fun Q c => Q * LinearMap.mulLeft k c - LinearMap.mulLeft k c * Q) P = 0
69 
70 /-- `P` is a `k`-linear differential operator of finite order on `B`. -/
71 def IsDifferentialOperator (k : Type u) (B : Type v) [CommRing k] [CommRing B]
72     [Algebra k B] (P : Module.End k B) : Prop :=
73   ∃ N, IsDifferentialOperatorOfOrder k B N P
74 
75 /-- The identity `1 = dR + FdS` in the ring `D_k(B)` of finite-order
76 `k`-linear differential operators on `B`: for every nonzero `d ∈ D_k(B)` there
77 are `F, R, S ∈ D_k(B)` with `1 = dR + FdS`. -/
78 def HasDifferentialIdentity (k : Type u) (B : Type v) [CommRing k] [CommRing B]
79     [Algebra k B] : Prop :=
80   ∀ d : Module.End k B, IsDifferentialOperator k B d → d ≠ 0 →
81     ∃ F R S : Module.End k B, IsDifferentialOperator k B F ∧ IsDifferentialOperator k B R ∧
82       IsDifferentialOperator k B S ∧ (1 : Module.End k B) = d * R + F * d * S
83 
84 /-- **Section 7, localizations** (tex lines 1630–1631): the identity for
85 finite-order differential operators on every localization `B = S⁻¹k[x_1, …, x_n]`. -/
86 def LocalizedDifferentialIdentity : Prop :=
87   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (S : Submonoid (MvPolynomial (Fin n) k))
88     (B : Type v) [CommRing B] [Algebra (MvPolynomial (Fin n) k) B] [Algebra k B]
89     [IsScalarTower k (MvPolynomial (Fin n) k) B] [IsLocalization S B],
90     HasDifferentialIdentity k B

92 /-- **Section 7, evolutionary identity** (tex lines 1631–1634): for `r ≥ 1`, in
93 every `k`-algebra `D` with `px = xp + 1`, the element
94 `d = p^r − ∑_j a_j x^{m_j}`, each `a_j` commuting with `x` and `p`, has
95 `R, S` with `1 = dR + x^r d S`. -/
96 def EvolutionaryIdentity : Prop :=
97   ∀ (k : Type u) [Field k] [CharZero k] (D : Type v) [Ring D] [Algebra k D] (x p : D),
98     p * x = x * p + 1 → ∀ r : ℕ, 1 ≤ r → ∀ (J : ℕ) (a : Fin J → D) (m : Fin J → ℕ),
99       (∀ j, a j * x = x * a j) → (∀ j, a j * p = p * a j) →
100         ∃ R S : D, (1 : D) = (p ^ r - ∑ j, a j * x ^ m j) * R +
101           x ^ r * (p ^ r - ∑ j, a j * x ^ m j) * S
