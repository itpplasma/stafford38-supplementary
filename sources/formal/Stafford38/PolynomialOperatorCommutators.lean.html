16 abbrev PolynomialRing := MvPolynomial (Fin n) k
17 abbrev Operator := Module.End k (PolynomialRing k n)
18 abbrev OperatorEnd := Module.End k (Operator k n)

20 def coordinateCommutator (i : Fin n) : OperatorEnd k n where
21   toFun P := commutator P (MvPolynomial.X i)
22   map_add' P Q := by
23     apply LinearMap.ext
24     intro f
25     simp [commutator_apply, sub_eq_add_neg, add_mul]
26     noncomm_ring
27   map_smul' c P := by
28     apply LinearMap.ext
29     intro f
30     simp [commutator_apply, smul_sub]
31 
32 @[simp] theorem coordinateCommutator_apply (i : Fin n) (P : Operator k n) :
33     coordinateCommutator k n i P = commutator P (MvPolynomial.X i) := rfl
34 
35 theorem coordinateCommutator_comm (i j : Fin n) :
36     coordinateCommutator k n i * coordinateCommutator k n j =
37       coordinateCommutator k n j * coordinateCommutator k n i := by
38   apply LinearMap.ext
39   intro P
40   apply LinearMap.ext
41   intro f
42   simp [coordinateCommutator, commutator_apply, Module.End.mul_apply,
43     mul_add, add_mul, sub_eq_add_neg, mul_assoc, mul_comm, mul_left_comm]
44   abel

46 def iteratedCoordinateCommutator (l : List (Fin n)) : OperatorEnd k n :=
47   l.foldr (fun i T => coordinateCommutator k n i * T) 1
48 
49 @[simp] theorem iteratedCoordinateCommutator_nil :
50     iteratedCoordinateCommutator k n [] = 1 := rfl

52 theorem iteratedCoordinateCommutator_cons (i : Fin n) (l : List (Fin n)) :
53     iteratedCoordinateCommutator k n (i :: l) =
54       coordinateCommutator k n i * iteratedCoordinateCommutator k n l := rfl
55 
56 theorem iteratedCoordinateCommutator_append (l₁ l₂ : List (Fin n)) :
57     iteratedCoordinateCommutator k n (l₁ ++ l₂) =
58       iteratedCoordinateCommutator k n l₁ * iteratedCoordinateCommutator k n l₂ := by
59   induction l₁ with
60   | nil => simp
61   | cons i l ih =>
62       change coordinateCommutator k n i * iteratedCoordinateCommutator k n (l ++ l₂) = _
63       rw [ih]
64       rfl

66 theorem iteratedCoordinateCommutator_swap (i j : Fin n) (l : List (Fin n)) :
67     iteratedCoordinateCommutator k n (i :: j :: l) =
68       iteratedCoordinateCommutator k n (j :: i :: l) := by
69   rw [iteratedCoordinateCommutator_cons, iteratedCoordinateCommutator_cons,
70     iteratedCoordinateCommutator_cons, iteratedCoordinateCommutator_cons]
71   rw [← mul_assoc, ← mul_assoc, coordinateCommutator_comm k n i j, mul_assoc]

73 theorem coordinateCommutator_order_lower {m : ℕ} (i : Fin n)
74     (P : Operator k n) (hP : P ∈ order (k := k) (R := PolynomialRing k n) (m + 1)) :
75     coordinateCommutator k n i P ∈ order (k := k) (R := PolynomialRing k n) m :=
76   hP (MvPolynomial.X i)

78 theorem coordinateCommutator_order_zero (i : Fin n) (P : Operator k n)
79     (hP : P ∈ order (k := k) (R := PolynomialRing k n) 0) :
80     coordinateCommutator k n i P = 0 :=
81   hP (MvPolynomial.X i)

83 /-- An iterated coordinate commutator lowers intrinsic order by its exact
84 list length.  The statement avoids truncated subtraction: the input order is
85 written as the desired residual order plus the number of commutators. -/
86 theorem iteratedCoordinateCommutator_order_lower (l : List (Fin n)) {m : ℕ}
87     (P : Operator k n)
88     (hP : P ∈ order (k := k) (R := PolynomialRing k n) (m + l.length)) :
89     iteratedCoordinateCommutator k n l P ∈
90       order (k := k) (R := PolynomialRing k n) m := by
91   induction l generalizing m with
92   | nil => simpa using hP
93   | cons i l ih =>
94       rw [iteratedCoordinateCommutator_cons, Module.End.mul_apply]
95       apply coordinateCommutator_order_lower k n i
96       apply ih (m := m + 1)
97       simpa [Nat.add_assoc, Nat.add_comm, Nat.add_left_comm] using hP
98 
99 /-- More commutators than the intrinsic order annihilate an operator. -/
100 theorem iteratedCoordinateCommutator_eq_zero_of_length_gt {m : ℕ}
101     (l : List (Fin n)) (P : Operator k n)
102     (hP : P ∈ order (k := k) (R := PolynomialRing k n) m)
103     (hl : m < l.length) :
104     iteratedCoordinateCommutator k n l P = 0 := by
105   induction l generalizing m P with
106   | nil => simp at hl
107   | cons i l ih =>
108       simp only [List.length_cons] at hl
109       rw [iteratedCoordinateCommutator_cons, Module.End.mul_apply]
110       by_cases htail : m < l.length
111       · rw [ih P hP htail]
112         exact map_zero _
113       · have hlen : l.length = m := by omega
114         apply coordinateCommutator_order_zero k n i
115         apply iteratedCoordinateCommutator_order_lower (k := k) (n := n) l
116         simpa [hlen] using hP

118 /-- Coordinate commutators depend only on the multiset of coordinates in the
119 iteration list.  This is the order-independent interface needed to encode a
120 multiindex by any list containing the prescribed multiplicities. -/
121 theorem iteratedCoordinateCommutator_eq_of_perm {l₁ l₂ : List (Fin n)}
122     (h : l₁.Perm l₂) :
123     iteratedCoordinateCommutator k n l₁ =
124       iteratedCoordinateCommutator k n l₂ := by
125   induction h with
126   | nil => rfl
127   | cons i h ih =>
128       rw [iteratedCoordinateCommutator_cons, iteratedCoordinateCommutator_cons, ih]
129   | swap i j l => exact (iteratedCoordinateCommutator_swap k n i j l).symm
130   | trans _ _ ih₁ ih₂ => exact ih₁.trans ih₂
131 
132 end
