13 /-- Finite generation is unchanged when the coefficient algebra is transported
14 along an isomorphism of coefficient rings. -/
15 theorem finiteType_of_equiv_base
16     {k E₀ E A : Type*} [CommSemiring k] [CommSemiring E₀] [CommSemiring E]
17     [CommSemiring A] [Algebra k E₀] [Algebra k E]
18     [Algebra E₀ A] [Algebra E A]
19     (e : E₀ ≃ₐ[k] E)
20     (hcompat : ∀ x : E₀,
21       algebraMap E₀ A x = algebraMap E A (e x))
22     (hfinite : Algebra.FiniteType E₀ A) : Algebra.FiniteType E A := by
23   rw [Algebra.FiniteType.iff_quotient_mvPolynomial''] at hfinite ⊢
24   obtain ⟨n, f, hf⟩ := hfinite
25   let g : MvPolynomial (Fin n) E →ₐ[E] A := {
26     toRingHom := f.toRingHom.comp (MvPolynomial.map e.symm.toRingHom)
27     commutes' := by
28       intro x
29       change f (MvPolynomial.map e.symm.toRingHom
30         (algebraMap E (MvPolynomial (Fin n) E) x)) = algebraMap E A x
31       rw [MvPolynomial.algebraMap_eq, MvPolynomial.map_C]
32       change f (algebraMap E₀ (MvPolynomial (Fin n) E₀) (e.symm x)) =
33         algebraMap E A x
34       rw [f.commutes]
35       simpa using hcompat (e.symm x)
36   }
37   refine ⟨n, g, ?_⟩
38   intro a
39   obtain ⟨p, hp⟩ := hf a
40   obtain ⟨q, hq⟩ := MvPolynomial.map_surjective e.symm.toRingHom e.symm.surjective p
41   refine ⟨q, ?_⟩
42   change f (MvPolynomial.map e.symm.toRingHom q) = a
43   rw [hq, hp]
44 
45 end Stafford38.Geometry.FieldEquivFiniteType

49 /-- An algebra isomorphic to a domain has the same fraction field after
50 transporting the canonical map into the fraction field. -/
51 theorem fractionField_of_equiv_domain
52     {k P R : Type*} [CommSemiring k] [CommRing P] [IsDomain P]
53     [CommRing R] [IsDomain R] [Algebra k P] [Algebra k R]
54     (e : P ≃ₐ[k] R) :
55     let E := FractionRing P
56     let f : R →+* E := (algebraMap P E).comp e.symm.toRingHom
57     letI : Algebra R E := f.toAlgebra
58     IsFractionRing R E := by
59   let E := FractionRing P
60   letI : Field E := FractionRing.field P
61   let f : R →+* E := (algebraMap P E).comp e.symm.toRingHom
62   letI : Algebra R E := f.toAlgebra
63   have hinj : Function.Injective f := by
64     intro a b hab
65     apply e.symm.injective
66     exact IsFractionRing.injective P E (by simpa [f] using hab)
67   letI : FaithfulSMul R E := (faithfulSMul_iff_algebraMap_injective R E).mpr hinj
68   apply IsFractionRing.of_field R E
69   intro z
70   obtain ⟨p, q, hq, hz⟩ := IsFractionRing.div_surjective P z
71   refine ⟨e p, e q, ?_⟩
72   change z = algebraMap P E (e.symm (e p)) /
73     algebraMap P E (e.symm (e q))
74   rw [e.symm_apply_apply, e.symm_apply_apply]
75   exact hz.symm
76 
77 end Stafford38.Geometry.FieldEquivFiniteType

110 /-- A further localization of a domain with the same fraction field does not
111 change that fraction field. The returned map is the canonical extension of
112 `B → F`; every inverted coefficient is nonzero because `R → B` is injective. -/
113 theorem exists_localized_fractionField_map
114     {R B F : Type*} [CommRing R] [IsDomain R] [CommRing B] [IsDomain B]
115     [Field F] [Algebra R B] [Algebra B F] [Algebra R F] [IsScalarTower R B F]
116     [IsFractionRing B F]
117     (hinj : Function.Injective (algebraMap R B)) :
118     ∃ f : Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R)) →+* F,
119       letI : Algebra (Localization
120         (Algebra.algebraMapSubmonoid B (nonZeroDivisors R))) F := f.toAlgebra
121       (∀ b : B, f (algebraMap B
122         (Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R))) b) =
123         algebraMap B F b) ∧
124       IsFractionRing (Localization
125         (Algebra.algebraMapSubmonoid B (nonZeroDivisors R))) F := by
126   let S := Algebra.algebraMapSubmonoid B (nonZeroDivisors R)
127   let L := Localization S
128   have hS : S ≤ nonZeroDivisors B := by
129     change (nonZeroDivisors R).map (algebraMap R B) ≤ nonZeroDivisors B
130     rw [Submonoid.map_le_iff_le_comap]
131     intro r hr
132     apply mem_nonZeroDivisors_of_ne_zero
133     intro hzero
134     exact (mem_nonZeroDivisors_iff_ne_zero.mp hr) (hinj (by simpa using hzero))
135   have hunit : ∀ s : S, IsUnit (algebraMap B F s.1) := by
136     intro s
137     have hsB : s.1 ∈ nonZeroDivisors B := hS s.property
138     have hsne : s.1 ≠ 0 := (mem_nonZeroDivisors_iff_ne_zero.mp hsB)
139     apply isUnit_iff_ne_zero.mpr
140     intro hzero
141     exact hsne (IsFractionRing.injective B F (by simpa using hzero))
142   let f : L →+* F := IsLocalization.lift hunit
143   refine ⟨f, ?_⟩
144   letI : Algebra L F := f.toAlgebra
145   letI : IsScalarTower B L F := IsScalarTower.of_algebraMap_eq fun b => by
146     change algebraMap B F b = f (algebraMap B L b)
147     change algebraMap B F b = IsLocalization.lift hunit (algebraMap B L b)
148     exact (IsLocalization.lift_eq hunit b).symm
149   constructor
150   · intro b
151     change IsLocalization.lift hunit (algebraMap B L b) = algebraMap B F b
152     exact IsLocalization.lift_eq hunit b
153   · exact IsFractionRing.isFractionRing_of_isDomain_of_isLocalization S L F

155 end Stafford38.Geometry.FieldEquivFiniteType

159 /-- The selected coefficient polynomial fraction field maps canonically to
160 its normalization localization. The map extends the actual inclusion of each
161 selected polynomial generator, transported through the chosen algebra
162 isomorphism `eA`. -/
163 theorem exists_fractionField_map_to_localization
164     {k P R B : Type*} [Field k] [CommRing P] [IsDomain P]
165     [CommRing R] [IsDomain R] [CommRing B] [IsDomain B]
166     [Algebra k P] [Algebra k R] [Algebra k B]
167     [Algebra R B] [IsScalarTower k R B]
168     (eA : P ≃ₐ[k] R) :
169     ∃ fR : FractionRing R →ₐ[R]
170       Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R)),
171     ∃ fE : FractionRing P →ₐ[k]
172       Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R)),
173       (∀ r : R, fR (algebraMap R (FractionRing R) r) =
174         algebraMap B (Localization (Algebra.algebraMapSubmonoid B
175           (nonZeroDivisors R))) (algebraMap R B r)) ∧
176       (∀ p : P, fE (algebraMap P (FractionRing P) p) =
177         algebraMap B (Localization (Algebra.algebraMapSubmonoid B
178           (nonZeroDivisors R))) (algebraMap R B (eA p))) ∧
179       fE = (AlgHom.restrictScalars k fR).comp
180         (IsFractionRing.algEquivOfAlgEquiv eA).toAlgHom := by
181   let L := Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R))
182   let S := Algebra.algebraMapSubmonoid B (nonZeroDivisors R)
183   let idR : R →ₐ[R] L := IsScalarTower.toAlgHom R R L
184   have hunitR : ∀ s : nonZeroDivisors R, IsUnit (idR s.1) := by
185     intro s
186     let sB : S := ⟨algebraMap R B s.1,
187       Submonoid.mem_map.mpr ⟨s.1, s.property, rfl⟩⟩
188     have hu : IsUnit (algebraMap B L sB.1) := IsLocalization.map_units L sB
189     change IsUnit (algebraMap R L s.1)
190     rw [IsScalarTower.algebraMap_apply R B L]
191     exact hu
192   let fR : FractionRing R →ₐ[R] L :=
193     IsLocalization.liftAlgHom (A := R) (M := nonZeroDivisors R)
194       (f := idR) hunitR
195   let eFrac : FractionRing P ≃ₐ[k] FractionRing R :=
196     IsFractionRing.algEquivOfAlgEquiv eA
197   let fE : FractionRing P →ₐ[k] L :=
198     (AlgHom.restrictScalars k fR).comp eFrac.toAlgHom
199   refine ⟨fR, fE, ?_, ?_, rfl⟩
200   · intro r
201     rw [IsLocalization.liftAlgHom_apply (f := idR) hunitR]
202     rw [IsLocalization.lift_eq (g := idR.toRingHom) hunitR r]
203     change algebraMap R L r = _
204     rw [IsScalarTower.algebraMap_apply R B L]
205   · intro p
206     change fR (eFrac (algebraMap P (FractionRing P) p)) = _
207     rw [IsFractionRing.algEquivOfAlgEquiv_algebraMap]
208     rw [IsLocalization.liftAlgHom_apply (f := idR) hunitR]
209     rw [IsLocalization.lift_eq (g := idR.toRingHom) hunitR (eA p)]
210     change algebraMap R L (eA p) = _
211     rw [IsScalarTower.algebraMap_apply R B L]
212 
213 end Stafford38.Geometry.FieldEquivFiniteType

217 /-- Finite generation of the normalized chart localization over the selected
218 residue-coordinate fraction field. The returned algebra map is the canonical
219 extension from the selected polynomial coordinates. -/
220 theorem exists_finiteType_selected_localization
221     {k P R B : Type*} [Field k] [CommRing P] [IsDomain P]
222     [CommRing R] [IsDomain R] [CommRing B] [IsDomain B]
223     [Algebra k P] [Algebra k R] [Algebra k B]
224     [Algebra R B] [IsScalarTower k R B]
225     [Algebra.FiniteType k B]
226     (eA : P ≃ₐ[k] R) :
227     ∃ fE : FractionRing P →ₐ[k]
228       Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R)),
229       letI : Algebra (FractionRing P)
230         (Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R))) :=
231           fE.toRingHom.toAlgebra
232       (∀ p : P, fE (algebraMap P (FractionRing P) p) =
233         algebraMap B (Localization (Algebra.algebraMapSubmonoid B
234           (nonZeroDivisors R))) (algebraMap R B (eA p))) ∧
235       Algebra.FiniteType (FractionRing P)
236         (Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R))):= by
237   let L := Localization (Algebra.algebraMapSubmonoid B (nonZeroDivisors R))
238   let eFrac : FractionRing P ≃ₐ[k] FractionRing R :=
239     IsFractionRing.algEquivOfAlgEquiv eA
240   obtain ⟨fR, fE, hRvar, hvar, hfcomp⟩ :=
241     exists_fractionField_map_to_localization (k := k) (P := P) (R := R) (B := B) eA
242   letI : Algebra (FractionRing P) L := fE.toRingHom.toAlgebra
243   have hRbase (r : R) :
244       algebraMap R L r = algebraMap B L (algebraMap R B r) := by
245     rw [← fR.commutes]
246     exact hRvar r
247   have hfRcanon : fR.toRingHom = algebraMap (FractionRing R) L := by
248     apply IsLocalization.ringHom_ext (M := nonZeroDivisors R)
249     apply RingHom.ext
250     intro r
251     change fR (algebraMap R (FractionRing R) r) =
252       algebraMap (FractionRing R) L (algebraMap R (FractionRing R) r)
253     calc
254       fR (algebraMap R (FractionRing R) r) =
255           algebraMap B L (algebraMap R B r) := hRvar r
256       _ = algebraMap R L r := (hRbase r).symm
257       _ = algebraMap (FractionRing R) L (algebraMap R (FractionRing R) r) :=
258         IsScalarTower.algebraMap_apply R (FractionRing R) L r
259   have hcompat : ∀ x : FractionRing R,
260       algebraMap (FractionRing R) L x =
261         algebraMap (FractionRing P) L (eFrac.symm x) := by
262     intro x
263     change algebraMap (FractionRing R) L x = fE (eFrac.symm x)
264     calc
265       algebraMap (FractionRing R) L x = fR x := by
266         have h := congrArg (fun g : FractionRing R →+* L => g x) hfRcanon
267         exact h.symm
268       _ = fE (eFrac.symm x) := by
269         rw [hfcomp]
270         change fR x = fR (eFrac (eFrac.symm x))
271         rw [eFrac.apply_symm_apply]
272   have hfinite : Algebra.FiniteType (FractionRing P) L :=
273     finiteType_localization_transport (k := k) (R := R) (B := B)
274       (E := FractionRing P) eFrac.symm hcompat
275   exact ⟨fE, hvar, hfinite⟩

318 /-- If a path from the selected fraction field through a coefficient
319 localization agrees with another field map on the original coefficient
320 ring, localization uniqueness identifies the field maps and supplies the
321 corresponding scalar tower. -/
322 theorem fractionField_map_coherence
323     {k P L F : Type*} [Field k] [CommRing P] [IsDomain P]
324     [CommRing L] [CommRing F] [Algebra k P]
325     [Algebra k (FractionRing P)] [Algebra k L] [Algebra k F]
326     (fEA : FractionRing P →ₐ[k] L)
327     (fAF : L →+* F)
328     (fEF : FractionRing P →ₐ[k] F)
329     (hgen : ∀ p : P,
330       fAF (fEA (algebraMap P (FractionRing P) p)) =
331         fEF (algebraMap P (FractionRing P) p)) :
332     letI : Algebra (FractionRing P) L := fEA.toRingHom.toAlgebra
333     letI : Algebra L F := fAF.toAlgebra
334     letI : Algebra (FractionRing P) F := fEF.toRingHom.toAlgebra
335     (fAF.comp fEA.toRingHom = fEF.toRingHom) ∧
336       IsScalarTower (FractionRing P) L F := by
337   letI : Algebra (FractionRing P) L := fEA.toRingHom.toAlgebra
338   letI : Algebra L F := fAF.toAlgebra
339   letI : Algebra (FractionRing P) F := fEF.toRingHom.toAlgebra
340   have hcomp : fAF.comp fEA.toRingHom = fEF.toRingHom := by
341     apply IsLocalization.ringHom_ext (M := nonZeroDivisors P)
342     apply RingHom.ext
343     intro p
344     exact hgen p
345   refine ⟨hcomp, ?_⟩
346   exact IsScalarTower.of_algebraMap_eq fun x => by
347     change fEF.toRingHom x = fAF (fEA.toRingHom x)
348     have hx := congrArg (fun g : FractionRing P →+* F => g x) hcomp
349     simpa only [RingHom.comp_apply] using hx.symm
