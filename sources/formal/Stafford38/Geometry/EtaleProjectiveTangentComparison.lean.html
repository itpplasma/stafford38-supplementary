46 /-- Actual local projective functions and the parameter derivations of one
47 common étale chart generate exactly the projective tangent cone of the
48 original affine presentation. The cone equality is proved, not supplied. -/
49 theorem projectiveTangentCone_eq_span_chartDerivation_columns
50     (q : Fin (n + 1) → C) (h0 : algebraMap C L (q 0) ≠ 0)
51     (hfactor : ∀ i, q i.succ = q 0 *
52       algebraMap (MvPolynomial (Fin n) k ⧸ I) C
53         (Ideal.Quotient.mk I (MvPolynomial.X i))) :
54     projectiveTangentCone (fun i => algebraMap C L (q i))
55       (zariskiTangentSpace (dehomogenizedPoint (fun i => algebraMap C L (q i)))
56         (I.map (MvPolynomial.map (algebraMap k L)))) =
57       Submodule.span L (Set.range fun j : Option (σ) => fun i =>
58         match j with
59         | none => algebraMap C L (q i)
60         | some j => coordinateDerivation
61             (k := k) (σ := σ) (B := C) (L := L) j (q i)) := by
62   classical
63   let y : Fin n → C := fun i =>
64     algebraMap (MvPolynomial (Fin n) k ⧸ I) C
65       (Ideal.Quotient.mk I (MvPolynomial.X i))
66   let B : Matrix (Fin (n + 1)) (Option (σ)) L := fun i j =>
67     match j with
68     | none => algebraMap C L (q i)
69     | some j => coordinateDerivation
70         (k := k) (σ := σ) (B := C) (L := L) j (q i)
71   have hpoint : ∀ i, dehomogenizedPoint (fun i => algebraMap C L (q i)) i =
72       algebraMap C L (y i) := by
73     intro i
74     change algebraMap C L (q i.succ) / algebraMap C L (q 0) = _
75     rw [hfactor i, map_mul]
76     field_simp [h0]
77     rfl
78   have hcols : ∀ j, dehomogenizedTangentColumn
79       (fun i => algebraMap C L (q i)) (fun i => B i j) =
80       match j with
81       | none => 0
82       | some j => fun i => coordinateDerivation
83           (k := k) (σ := σ) (B := C) (L := L) j (y i) := by
84     intro j
85     cases j with
86     | none =>
87         ext i
88         simp [B]
89     | some j =>
90         exact dehomogenizedTangentColumn_derivation
91           (coordinateDerivation (k := k) (σ := σ) (B := C) (L := L) j)
92           q y h0 hfactor
93   have hspan : zariskiTangentSpace
94         (dehomogenizedPoint (fun i => algebraMap C L (q i)))
95         (I.map (MvPolynomial.map (algebraMap k L))) =
96       dehomogenizedTangentSpan (fun i => algebraMap C L (q i)) B := by
97     rw [Stafford38.Geometry.EtaleTangentChartSpan.FiniteParameters.zariskiTangentSpace_eq_span_parameterDerivations
98       (σ := σ) _ hpoint]
99     apply le_antisymm
100     · apply Submodule.span_le.mpr
101       rintro _ ⟨j, rfl⟩
102       change (fun i => coordinateDerivation j (y i)) ∈ _
103       exact Submodule.subset_span ⟨some j, hcols (some j)⟩
104     · apply Submodule.span_le.mpr
105       rintro _ ⟨j, rfl⟩
106       dsimp only
107       rw [hcols j]
108       cases j with
109       | none => exact Submodule.zero_mem _
110       | some j => exact Submodule.subset_span ⟨j, rfl⟩
111   apply projectiveTangentCone_eq_span_of_position_and_tangent
112     (fun i => algebraMap C L (q i)) h0 _ B _ hspan
113   exact Submodule.subset_span ⟨none, rfl⟩
114 
115 end FiniteParameters
