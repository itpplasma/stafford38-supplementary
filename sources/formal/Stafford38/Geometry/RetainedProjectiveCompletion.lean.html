8 /-!
9 # Transport from a retained valuation ring to its power-series completion
10 
11 The completed-DVR equivalence turns every valuation-ring coordinate into a
12 power series over the actual residue field.  Krull intersection makes this
13 coordinate map injective, so projective nonvanishing is preserved.
14 -/

31 /-- Map retained valuation-ring coordinates into the corresponding
32 power-series ring via the actual adic completion. -/
33 def retainedToCompletedPowerSeries
34     {k K : Type u} [Field k] [CharZero k]
35     [Field K] [Algebra k K] {x : K} (W : Data k K x) :
36     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
37       W.ambientAlgebra
38     let V := W.place.valuation.toSubring
39     letI : IsDiscreteValuationRing V := W.place.isDiscrete
40     letI : Algebra W.coefficientField V :=
41       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
42     V →+* PowerSeries (ResidueField V) := by
43   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
44     W.ambientAlgebra
45   let V := W.place.valuation.toSubring
46   letI : IsDiscreteValuationRing V := W.place.isDiscrete
47   letI : Algebra W.coefficientField V :=
48     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
49   exact W.completedPowerSeriesEquiv.symm.toRingHom.comp
50     (algebraMap V (AdicCompletion (maximalIdeal V) V))

64     W.ambientAlgebra
65   let V := W.place.valuation.toSubring
66   letI : IsDiscreteValuationRing V := W.place.isDiscrete

73 /-- A retained nonunit has zero constant coefficient in the completed
74 power-series chart. -/
75 theorem retainedToCompletedPowerSeries_constantCoeff_eq_zero_of_nonunit
76     {k K : Type u} [Field k] [CharZero k]
77     [Field K] [Algebra k K] {x : K} (W : Data k K x) :
78     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
79       W.ambientAlgebra
80     let V := W.place.valuation.toSubring
81     letI : IsDiscreteValuationRing V := W.place.isDiscrete
82     letI : Algebra W.coefficientField V :=
83       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
84     ∀ v : ↥V, ¬IsUnit v →
85       PowerSeries.constantCoeff (R := ResidueField V)
86         (retainedToCompletedPowerSeries W v) = 0 := by
87   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
88     W.ambientAlgebra
89   let V := W.place.valuation.toSubring
90   letI : IsDiscreteValuationRing V := W.place.isDiscrete
91   letI : Algebra W.coefficientField V :=
92     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
93   dsimp only
94   intro v hv
95   by_contra hconstant
96   have hseries : IsUnit (retainedToCompletedPowerSeries W v) :=
97     PowerSeries.isUnit_iff_constantCoeff.mpr
98       (isUnit_iff_ne_zero.mpr hconstant)
99   have hcompletion : IsUnit
100       (algebraMap V (AdicCompletion (maximalIdeal V) V) v) := by
101     have hmap := hseries.map W.completedPowerSeriesEquiv.toMonoidHom
102     simpa [retainedToCompletedPowerSeries] using hmap
103   have hlevel := hcompletion.map
104     (AdicCompletion.evalₐ (maximalIdeal V) 1).toRingHom.toMonoidHom
105   have heval : AdicCompletion.evalₐ (maximalIdeal V) 1
106       (algebraMap V (AdicCompletion (maximalIdeal V) V) v) =
107       Ideal.Quotient.mk ((maximalIdeal V) ^ 1) v := by
108     exact (AdicCompletion.evalₐ (maximalIdeal V) 1).commutes v
109   change IsUnit (AdicCompletion.evalₐ (maximalIdeal V) 1
110     (algebraMap V (AdicCompletion (maximalIdeal V) V) v)) at hlevel
111   rw [heval] at hlevel
112   have hvmax : v ∈ maximalIdeal V := by
113     rw [mem_maximalIdeal, mem_nonunits_iff]
114     exact hv
115   have hvpow : v ∈ (maximalIdeal V) ^ 1 := by simpa only [pow_one] using hvmax
116   have hzero : Ideal.Quotient.mk ((maximalIdeal V) ^ 1) v = 0 :=
117     Ideal.Quotient.eq_zero_iff_mem.mpr hvpow
118   rw [hzero] at hlevel
119   letI : Nontrivial (V ⧸ (maximalIdeal V) ^ 1) :=
120     Ideal.Quotient.nontrivial_iff.mpr (by
121       simpa only [pow_one] using (maximalIdeal.isMaximal V).ne_top)
122   exact not_isUnit_zero hlevel

124 /-- A normalized valuation-ring projective family remains normalized and
125 nonzero after transport to the power-series completion. -/
126 theorem retainedCompleted_family_properties
127     {k K : Type u} [Field k] [CharZero k]
128     [Field K] [Algebra k K] {x : K} (W : Data k K x) :
129     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
130       W.ambientAlgebra
131     let V := W.place.valuation.toSubring
132     letI : IsDiscreteValuationRing V := W.place.isDiscrete
133     letI : Algebra W.coefficientField V :=
134       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
135     ∀ {ι : Type v} [Fintype ι]
136       (q : ι → V) (chart zero axis : ι),
137       q chart = 1 → q zero ≠ 0 →
138       q axis = q zero * W.place.parameter →
139       let qhat := fun i ↦ retainedToCompletedPowerSeries W (q i)
140       qhat chart = 1 ∧ qhat zero ≠ 0 ∧
141         qhat axis = qhat zero *
142           retainedToCompletedPowerSeries W W.place.parameter := by
143   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
144     W.ambientAlgebra
145   let V := W.place.valuation.toSubring
146   letI : IsDiscreteValuationRing V := W.place.isDiscrete
147   letI : Algebra W.coefficientField V :=
148     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
149   dsimp only
150   intro ι _ q chart zero axis hchart hzero haxis
151   dsimp
152   refine ⟨by rw [hchart, map_one], ?_, ?_⟩
153   · simpa using (retainedToCompletedPowerSeries_injective W).ne hzero
154   · rw [haxis, map_mul]

156 /-- The retained parameter and a nonunit projective denominator supply all
157 nonvanishing and residue-vanishing fields needed by the projective order-gap
158 consumer after completion. -/
159 theorem retainedCompleted_projective_order_properties
160     {k K : Type u} [Field k] [CharZero k]
161     [Field K] [Algebra k K] {x : K} (W : Data k K x) :
162     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
163       W.ambientAlgebra
164     let V := W.place.valuation.toSubring
165     letI : IsDiscreteValuationRing V := W.place.isDiscrete
166     letI : Algebra W.coefficientField V :=
167       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
168     ∀ {ι : Type v} [Fintype ι]
169       (q : ι → V) (chart zero axis : ι),
170       q chart = 1 → q zero ≠ 0 → ¬IsUnit (q zero) →
171       q axis = q zero * W.place.parameter →
172       let qhat := fun i ↦ retainedToCompletedPowerSeries W (q i)
173       let ratio := retainedToCompletedPowerSeries W W.place.parameter
174       qhat chart = 1 ∧ qhat zero ≠ 0 ∧ ratio ≠ 0 ∧
175         PowerSeries.constantCoeff (R := ResidueField V) (qhat zero) = 0 ∧
176         PowerSeries.constantCoeff (R := ResidueField V) ratio = 0 ∧
177         qhat axis = qhat zero * ratio := by
178   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K :=
179     W.ambientAlgebra
180   let V := W.place.valuation.toSubring
181   letI : IsDiscreteValuationRing V := W.place.isDiscrete
182   letI : Algebra W.coefficientField V :=
183     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
184   dsimp only
185   intro ι _ q chart zero axis hchart hzero hzero_nonunit haxis
186   obtain ⟨hqchart, hqzero, hqaxis⟩ :=
187     retainedCompleted_family_properties W q chart zero axis
188       hchart hzero haxis
189   refine ⟨hqchart, hqzero,
190     (by simpa using
191       (retainedToCompletedPowerSeries_injective W).ne W.place.parameter_ne),
192     retainedToCompletedPowerSeries_constantCoeff_eq_zero_of_nonunit W
193       (q zero) hzero_nonunit,
194     retainedToCompletedPowerSeries_constantCoeff_eq_zero_of_nonunit W
195       W.place.parameter W.place.parameter_nonunit,
196     hqaxis⟩
