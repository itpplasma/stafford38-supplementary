29 /-- A finite residue basis has distinct representatives among the chart coordinates. -/
30 theorem exists_selected_residue_basis_indices
31     {k V : Type u} [Field k] [CommRing V] [IsLocalRing V]
32     [Algebra k V] [Algebra k (ResidueField V)]
33     [IsScalarTower k V (ResidueField V)]
34     {m : ℕ} (q : Fin (m + 1) → V) (chart : Fin (m + 1))
35     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
36     (hqchart : q chart = 1) (hq0res : residue V (q 0) = 0)
37     (halg : Algebra.IsAlgebraic
38       (IntermediateField.adjoin k
39         (Set.range fun j : Fin m => residue V (q (Fin.succ j))))
40       (ResidueField V)) :
41     ∃ t : Set (ResidueField V), t.Finite ∧
42       IsTranscendenceBasis k ((↑) : t → ResidueField V) ∧
43       ∃ index : t → Fin m, Function.Injective index ∧
44         ∀ z : t, residue V (q (e (index z)).1) = (z : ResidueField V) := by
45   classical
46   obtain ⟨t, htFinite, htSubset, htBasis⟩ :=
47     exists_finite_chartCoordinate_residue_transcendenceBasis
48       (k := k) (m := m) (V := V) q chart e hqchart hq0res halg
49   have hchoose : ∀ z : t, ∃ i : Fin m,
50       residue V (q (e i).1) = (z : ResidueField V) := fun z => htSubset z.property
51   choose index hindex using hchoose
52   have hinj : Function.Injective index := by
53     intro z₁ z₂ h
54     apply Subtype.ext
55     calc
56       (z₁ : ResidueField V) = residue V (q (e (index z₁)).1) := (hindex z₁).symm
57       _ = residue V (q (e (index z₂)).1) := by rw [h]
58       _ = (z₂ : ResidueField V) := hindex z₂
59   exact ⟨t, htFinite, htBasis, index, hinj, hindex⟩

61 /-- Select a finite residue transcendence basis from the actual retained

63 theorem exists_actual_selected_residue_basis
64     {k : Type u} [Field k] [CharZero k]
65     {m : ℕ} (hm : 0 < m)
66     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
67     (w : GeneralDivisorialVisibleFrameWitness hm P) :
68     let C := w.column
69     let W := C.W
70     let F := ComponentFractionField P
71     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
72     let V := W.place.valuation.toSubring
73     letI : IsDiscreteValuationRing V := W.place.isDiscrete
74     letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
75     letI : Algebra W.coefficientField V :=
76       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
77     let κ := ResidueField V
78     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
79     ∃ t : Set κ, t.Finite ∧
80       IsTranscendenceBasis k ((↑) : t → κ) ∧
81       ∃ index : t → Fin m, Function.Injective index ∧
82         ∀ z : t, residue V
83           (C.q ((chartAffineCoordinateEquiv C.chart) (index z)).1) = (z : κ) := by
84   classical
85   dsimp only
86   let C := w.column
87   let W := C.W
88   let F := ComponentFractionField P
89   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
90   let V := W.place.valuation.toSubring
91   letI : IsDiscreteValuationRing V := W.place.isDiscrete
92   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
93   letI : Algebra W.coefficientField V :=
94     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
95   letI : Algebra k V := C.groundCoeff.toAlgebra
96   let κ := ResidueField V
97   let retained : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
98   letI : Algebra k κ := retained
99   have hretainedMap : @algebraMap k κ _ _ retained =
100       (residue V).comp C.groundCoeff := by
101     simpa [C, W, V, κ, C.groundCoeff_eq_retained] using w.retainedGroundMap
102   letI : IsScalarTower k V κ := IsScalarTower.of_algebraMap_eq fun c => by
103     have h := RingHom.congr_fun hretainedMap c
104     change algebraMap k κ c = residue V (C.groundCoeff c)
105     exact h
106   let e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) C.chart :=
107     chartAffineCoordinateEquiv C.chart
108   exact exists_selected_residue_basis_indices (k := k) (V := V) (m := m)
109     C.q C.chart e C.chart_one (witness_q0_residue_eq_zero hm P w) w.halg
