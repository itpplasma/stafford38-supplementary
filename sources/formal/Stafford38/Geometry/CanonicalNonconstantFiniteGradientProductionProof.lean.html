38 /-- The reusable one-row package for a normalized retained-place chart.
39 Its `conormal` field is the explicit affine statement at the completed
40 Laurent point. The projective and residue fields are retained as certificate
41 data; the extraction theorem uses `conormal` to find finite gradient
42 coefficients. `PaperActualWitnessConormalData` constructs it from the same
43 retained witness. -/
44 structure RegularizedOneRowConormalData
45     {m : ℕ} (hm : 0 < m)
46     (I : Ideal (MvPolynomial (Fin m) k))
47     (q : Fin (m + 1) → PowerSeries K) where
48   ell : Fin (m + 1) → PowerSeries K
49   q_origin_ne : q 0 ≠ 0
50   projective_annihilation :
51     ∑ i, laurentColumn ell i * laurentColumn q i = 0
52   base_vanish :
53     ∀ f ∈ I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m)),
54       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0
55   conormal :
56     coordinateCovector (fun i ↦ laurentColumn ell i.succ) ∈
57       affineConormalSpace (dehomogenizedPoint (laurentColumn q))
58         (I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m)))
59   residue_axis :
60     residueColumn (fun i : Fin m ↦ ell i.succ) =
61       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0)
62 
63 /-- A regularized one-row conormal datum yields the finite gradient certificate
64 over the same residue-field extension.  The only mathematical input is the
65 explicit `RegularizedOneRowConormalData`; all finite equations and Laurent
66 coefficients are extracted by the checked affine conormal-span theorem. -/
67 theorem finiteGradientBoundaryCertificateOver_of_regularizedOneRowConormalData
68     {m : ℕ} (hm : 0 < m)
69     (I : Ideal (MvPolynomial (Fin m) k))
70     (q : Fin (m + 1) → PowerSeries K)
71     (W : RegularizedOneRowConormalData (k := k) (K := K) hm I q) :
72     Nonempty (FiniteGradientBoundaryCertificateOver
73       (k := k) (K := K) m hm I) := by
74   let Iext := I.map
75     (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m))
76   let y := dehomogenizedPoint (laurentColumn q)
77   let xi : Fin m → LaurentSeries K :=
78     fun i ↦ laurentColumn W.ell i.succ
79   obtain ⟨r, equations, coefficients, hgradient⟩ :=
80     exists_fin_gradient_identity_of_mem_affineConormalSpace
81       Iext y xi W.conormal
82   exact ⟨{
83     equationCount := r
84     q := q
85     ell := W.ell
86     q_origin_ne := W.q_origin_ne
87     projective_annihilation := W.projective_annihilation
88     base_vanish := W.base_vanish
89     equations := equations
90     coefficients := coefficients
91     gradient_identity := by
92       intro i
93       exact hgradient i
94     residue_axis := W.residue_axis
95   }⟩
96 
97 /-- The finite-gradient certificate immediately supplies the terminal
98 conormal-axis witness.  This is a named adapter so a future retained-place
99 producer can target the smaller row datum directly. -/
100 theorem exists_groundConormalAxis_of_regularizedOneRowConormalData
101     {m : ℕ} (hm : 0 < m)
102     (I : Ideal (MvPolynomial (Fin m) k))
103     (q : Fin (m + 1) → PowerSeries K)
104     (W : RegularizedOneRowConormalData (k := k) (K := K) hm I q) :
105     ∃ (y : Fin m → LaurentSeries K)
106       (xi : Fin m → PowerSeries K),
107       Sum.elim y
108           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
109         groundEquationConormalLocus (k := k) (K := K) I ∧
110       residueColumn xi =
111         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
112   obtain ⟨certificate⟩ :=
113     finiteGradientBoundaryCertificateOver_of_regularizedOneRowConormalData
114       (k := k) (K := K) hm I q W
115   exact exists_groundConormalAxis_of_finiteGradientBoundaryCertificateOver
116     (k := k) (K := K) hm I certificate
