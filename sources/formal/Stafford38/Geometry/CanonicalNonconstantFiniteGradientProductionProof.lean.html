36 /-- The reusable one-row package for a normalized retained-place chart.
37 Its `conormal` field is the explicit affine statement at the completed
38 Laurent point. The projective and residue fields are retained as certificate
39 data; the extraction theorem uses `conormal` to find finite gradient
40 coefficients. `PaperActualWitnessConormalData` constructs it from the same
41 retained witness. -/
42 structure RegularizedOneRowConormalData
43     {m : ℕ} (hm : 0 < m)
44     (I : Ideal (MvPolynomial (Fin m) k))
45     (q : Fin (m + 1) → PowerSeries K) where
46   ell : Fin (m + 1) → PowerSeries K
47   q_origin_ne : q 0 ≠ 0
48   projective_annihilation :
49     ∑ i, laurentColumn ell i * laurentColumn q i = 0
50   base_vanish :
51     ∀ f ∈ I.map (groundPolynomialMap (k := k) (K := K) (Fin m)),
52       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0
53   conormal :
54     coordinateCovector (fun i ↦ laurentColumn ell i.succ) ∈
55       affineConormalSpace (dehomogenizedPoint (laurentColumn q))
56         (I.map (groundPolynomialMap (k := k) (K := K) (Fin m)))
57   residue_axis :
58     residueColumn (fun i : Fin m ↦ ell i.succ) =
59       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0)

61 /-- A regularized one-row conormal datum yields the finite gradient certificate
62 over the same residue-field extension.  The only mathematical input is the
63 explicit `RegularizedOneRowConormalData`; all finite equations and Laurent
64 coefficients are extracted by the checked affine conormal-span theorem. -/
65 theorem finiteGradientBoundaryCertificateOver_of_regularizedOneRowConormalData
66     {m : ℕ} (hm : 0 < m)
67     (I : Ideal (MvPolynomial (Fin m) k))
68     (q : Fin (m + 1) → PowerSeries K)
69     (W : RegularizedOneRowConormalData (k := k) (K := K) hm I q) :
70     Nonempty (FiniteGradientBoundaryCertificateOver
71       (k := k) (K := K) m hm I) := by
72   let Iext := I.map
73     (groundPolynomialMap (k := k) (K := K) (Fin m))
74   let y := dehomogenizedPoint (laurentColumn q)
75   let xi : Fin m → LaurentSeries K :=
76     fun i ↦ laurentColumn W.ell i.succ
77   obtain ⟨r, equations, coefficients, hgradient⟩ :=
78     exists_fin_gradient_identity_of_mem_affineConormalSpace
79       Iext y xi W.conormal
80   exact ⟨{
81     equationCount := r
82     q := q
83     ell := W.ell
84     q_origin_ne := W.q_origin_ne
85     projective_annihilation := W.projective_annihilation
86     base_vanish := W.base_vanish
87     equations := equations
88     coefficients := coefficients
89     gradient_identity := by
90       intro i
91       exact hgradient i
92     residue_axis := W.residue_axis
93   }⟩

95 /-- The finite-gradient certificate immediately supplies the terminal
96 conormal-axis witness.  This is a named adapter so a future retained-place
97 producer can target the smaller row datum directly. -/
98 theorem exists_groundConormalAxis_of_regularizedOneRowConormalData
99     {m : ℕ} (hm : 0 < m)
100     (I : Ideal (MvPolynomial (Fin m) k))
101     (q : Fin (m + 1) → PowerSeries K)
102     (W : RegularizedOneRowConormalData (k := k) (K := K) hm I q) :
103     ∃ (y : Fin m → LaurentSeries K)
104       (xi : Fin m → PowerSeries K),
105       Sum.elim y
106           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
107         groundEquationConormalLocus (k := k) (K := K) I ∧
108       residueColumn xi =
109         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
110   obtain ⟨certificate⟩ :=
111     finiteGradientBoundaryCertificateOver_of_regularizedOneRowConormalData
112       (k := k) (K := K) hm I q W
113   exact exists_groundConormalAxis_of_finiteGradientBoundaryCertificateOver
114     (k := k) (K := K) hm I certificate
