18 /-- If a map from a domain has injective image in the residue field, it extends
19 from the domain's fraction ring to the localization at that residue kernel. -/
20 noncomputable def fractionRingToAtPrime
21     {A : Type u} [CommRing A] [IsDomain A]
22     {B : Type v} [CommRing B] {κ : Type w} [Field κ]
23     (f : A →+* B) (rho : B →+* κ)
24     (P : Ideal B) [P.IsPrime] (hP : P = RingHom.ker rho)
25     (hinj : Function.Injective (rho.comp f)) :
26     FractionRing A →+* Localization.AtPrime P := by
27   let g : A →+* Localization.AtPrime P :=
28     (algebraMap B (Localization.AtPrime P)).comp f
29   have hg : ∀ y : nonZeroDivisors A, IsUnit (g y) := by
30     intro y
31     have hnot : f y ∉ P := by
32       intro hmem
33       have hz : rho (f y) = 0 := by
34         rw [hP, RingHom.mem_ker] at hmem
35         exact hmem
36       have hz' : (rho.comp f) y = 0 := by
37         simpa only [RingHom.comp_apply] using hz
38       apply (nonZeroDivisors.ne_zero y.2)
39       exact hinj (by simpa using hz')
40     exact IsLocalization.map_units (Localization.AtPrime P)
41       (⟨f y, hnot⟩ : P.primeCompl)
42   exact IsLocalization.lift hg

44 @[simp]
45 theorem fractionRingToAtPrime_algebraMap
46     {A : Type u} [CommRing A] [IsDomain A]
47     {B : Type v} [CommRing B] {κ : Type w} [Field κ]
48     (f : A →+* B) (rho : B →+* κ)
49     (P : Ideal B) [P.IsPrime] (hP : P = RingHom.ker rho)
50     (hinj : Function.Injective (rho.comp f)) (a : A) :
51     fractionRingToAtPrime f rho P hP hinj
52       (algebraMap A (FractionRing A) a) =
53       algebraMap B (Localization.AtPrime P) (f a) := by
54   simp [fractionRingToAtPrime]

56 /-- The coefficient field maps into the two local rings are compatible with
57 the canonical localization-in-stages equivalence. The only selected-data
58 condition is injectivity of the chosen domain in the residue field; the right
59 coefficient map is constructed by the localization universal property. -/
60 theorem stageFractionCoefficientMap_commutes
61     {A : Type u} [CommRing A] [IsDomain A]
62     {B : Type v} [CommRing B] {κ : Type w} [Field κ]
63     (M : Submonoid B) (p : Ideal (Localization M)) [p.IsPrime]
64     (rhoL : Localization M →+* κ) (rhoB : B →+* κ)
65     (hp : p = RingHom.ker rhoL)
66     (hLoc : rhoL.comp (algebraMap B (Localization M)) = rhoB)
67     (f : A →+* B)
68     (hinj : Function.Injective (rhoB.comp f))
69     (coeffL : FractionRing A →+* Localization M)
70     (hcoeffL : coeffL.comp (algebraMap A (FractionRing A)) =
71       (algebraMap B (Localization M)).comp f) :
72     ∃ e : Localization.AtPrime p ≃ₐ[B]
73         Localization.AtPrime (p.comap (algebraMap B (Localization M))),
74       ∃ hcenter : p.comap (algebraMap B (Localization M)) = RingHom.ker rhoB,
75       e.toRingEquiv.toRingHom.comp
76           ((algebraMap (Localization M) (Localization.AtPrime p)).comp coeffL) =
77         fractionRingToAtPrime f rhoB
78           (p.comap (algebraMap B (Localization M)))
79           hcenter hinj := by
80   obtain ⟨e, hcenter, _hmap⟩ :=
81     kernel_center_localization_equiv M p rhoL rhoB hp hLoc
82   refine ⟨e, hcenter, ?_⟩
83   apply IsLocalization.ringHom_ext (M := nonZeroDivisors A)
84   apply RingHom.ext
85   intro a
86   change e (algebraMap (Localization M) (Localization.AtPrime p)
87       (coeffL (algebraMap A (FractionRing A) a))) =
88     fractionRingToAtPrime f rhoB
89       (p.comap (algebraMap B (Localization M))) hcenter hinj
90       (algebraMap A (FractionRing A) a)
91   have hcoeffA : coeffL (algebraMap A (FractionRing A) a) =
92       algebraMap B (Localization M) (f a) := by
93     exact RingHom.congr_fun hcoeffL a
94   rw [hcoeffA]
95   calc
96     e (algebraMap (Localization M) (Localization.AtPrime p)
97         (algebraMap B (Localization M) (f a))) =
98       algebraMap B (Localization.AtPrime (p.comap (algebraMap B (Localization M))))
99         (f a) := _hmap (f a)
100     _ = fractionRingToAtPrime f rhoB
101           (p.comap (algebraMap B (Localization M))) hcenter hinj
102           (algebraMap A (FractionRing A) a) := by
103             symm
104             exact fractionRingToAtPrime_algebraMap f rhoB
105               (p.comap (algebraMap B (Localization M))) hcenter hinj a

107 /-- If a ring equivalence preserves the coefficient map and the distinguished
108 parameter, it is automatically an equivalence over the induced polynomial
109 parameter algebra. The two `hEval` premises record that the chosen polynomial
110 algebra structures are exactly evaluation at those data. -/
111 noncomputable def polynomialAlgEquivOfCompatibleEvaluation
112     {E : Type u} [CommSemiring E]
113     {T₁ : Type v} [CommSemiring T₁] {T₂ : Type w} [CommSemiring T₂]
114     [Algebra E T₁] [Algebra E T₂]
115     [Algebra (Polynomial E) T₁] [Algebra (Polynomial E) T₂]
116     (f : T₁ ≃+* T₂)
117     (hcoeff : ∀ c : E, f (algebraMap E T₁ c) = algebraMap E T₂ c)
118     (a₁ : T₁) (a₂ : T₂) (ha : f a₁ = a₂)
119     (hEval₁ : ∀ p : Polynomial E,
120       algebraMap (Polynomial E) T₁ p = Polynomial.aeval a₁ p)
121     (hEval₂ : ∀ p : Polynomial E,
122       algebraMap (Polynomial E) T₂ p = Polynomial.aeval a₂ p) :
123     T₁ ≃ₐ[Polynomial E] T₂ := by
124   let fE : T₁ ≃ₐ[E] T₂ := AlgEquiv.ofRingEquiv (f := f) hcoeff
125   have hpoly : fE.toAlgHom.comp (Polynomial.aeval a₁) = Polynomial.aeval a₂ := by
126     apply Polynomial.algHom_ext
127     simpa [fE] using ha
128   apply AlgEquiv.ofRingEquiv (f := f)
129   intro p
130   rw [hEval₁ p, hEval₂ p]
131   have hp := congrArg (fun g : Polynomial E →ₐ[E] T₂ => g p) hpoly
132   simpa [fE] using hp
