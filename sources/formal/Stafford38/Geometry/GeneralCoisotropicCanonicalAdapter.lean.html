16 /-!
17 # Adapter from canonical order support to the arbitrary coisotropic theorem
18 
19 The canonical order support is presented on the prime spectrum, while the
20 general exclusion theorem is stated for field-valued phase points.  This file
21 performs that change of presentation at the reduced order-support ideal.  The
22 canonical support-avoidance input supplies the coordinate hypothesis, and the
23 Gabber involutivity result supplies the self-involutivity of the same radical
24 ideal.
25 
26 This adapter is deliberately separate from the canonical support assembly, so
27 the general theorem can be integrated there without an import cycle.
28 -/

82 private theorem zeroLocus_isFibreConical_of_isHomogeneous
83     {n : ℕ} (J : Ideal (SymbolRing k n))
84     (hhom : J.IsHomogeneous
85       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n))) :
86     IsFibreConical (MvPolynomial.zeroLocus k J) := by
87   intro q hq a ha
88   let y : Fin n → k := fun i => q (.inl i)
89   let ξ : Fin n → k := fun i => q (.inr i)
90   have hqsplit : q = Sum.elim y ξ := by
91     funext i
92     rcases i with i | i <;> rfl
93   have hline (f : SymbolRing k n) (hf : f ∈ J) :
94       fibreLinePolynomial y ξ f = 0 := by
95     apply Polynomial.ext
96     intro d
97     rw [coeff_fibreLinePolynomial_eq_eval_weightedHomogeneousComponent]
98     rw [← hqsplit]
99     exact (MvPolynomial.mem_zeroLocus_iff.mp hq) _ (hhom d hf)
100   intro f hf
101   have heval := eval_fibreLinePolynomial y ξ f a
102   rw [hline f hf, Polynomial.eval_zero] at heval
103   simpa [y, ξ] using heval.symm

105 /-! The canonical reduced support ideal is an admissible input to the
106 general coisotropic exclusion theorem.  The only geometric fact used here is
107 Gabber's involutivity of the radical associated graded annihilator; its
108 base-relative fragment is obtained by restricting the first bracket entry to
109 the base polynomial subring. -/
110 
111 theorem algebraicallyClosedCanonicalSupportVanishing_using
112     (axisClosure : ∀ {k : Type u} [Field k] [IsAlgClosed k] [CharZero k],
113       CoordinateAxisClosureInput (k := k))
114     (hunit : CanonicalStrictUnitCoordinatePreimage.{u}) :
115     AlgebraicallyClosedCanonicalSupportVanishing.{u} := by
116   intro k _ _ _ n N d hN hd
117   let I := canonicalRightIdeal (presentedCoordinate k n) d N
118   let J : Ideal (SymbolRing k (n + 1)) := reducedOrderSupportIdeal k I
119   have hdisjoint : Disjoint
120       (orderCharacteristicSupport k I)
121       (PrimeSpectrum.zeroLocus
122         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
123           Set (SymbolRing k (n + 1)))) := by
124     exact canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
125       k n N (hunit k n N d hN hd)
126   by_contra hsupp
127   have hnonempty : (orderCharacteristicSupport k I).Nonempty :=
128     Set.nonempty_iff_ne_empty.mpr hsupp
129   have hproper : J ≠ ⊤ := by
130     intro htop
131     obtain ⟨p, hp⟩ := hnonempty
132     have hpzero : p ∈ PrimeSpectrum.zeroLocus J := by
133       rw [← orderCharacteristicSupport_eq_zeroLocus_reduced k I]
134       exact hp
135     rw [htop] at hpzero
136     simpa using hpzero
137   have hrad : J.IsRadical := by
138     simpa only [J] using reducedOrderSupportIdeal_isRadical k I
139   have hhom : J.IsHomogeneous
140       (MvPolynomial.weightedHomogeneousSubmodule k
141         (@orderWeight (n + 1))) := by
142     simpa only [J, reducedOrderSupportIdeal] using
143       (orderInitialIdeal_isHomogeneous (k := k) I).radical
144   have hinv : IsInvolutive J := by
145     change IsInvolutive (orderInitialIdeal k I).radical
146     rw [← annihilator_orderAssociatedGradedModule]
147     exact associatedGraded_radical_isInvolutive (k := k) I
148   have hW : IsFibreConical (MvPolynomial.zeroLocus k J) :=
149     zeroLocus_isFibreConical_of_isHomogeneous J hhom
150   have hnonemptyW : (MvPolynomial.zeroLocus k J).Nonempty := by
151     obtain ⟨M, hM, hJM⟩ := Ideal.exists_le_maximal J hproper
152     obtain ⟨q, hqM⟩ :=
153       MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hM
154     refine ⟨q, ?_⟩
155     rw [MvPolynomial.mem_zeroLocus_iff]
156     intro f hf
157     have hfM : f ∈ M := hJM hf
158     rw [hqM, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hfM
159     exact hfM
160   have hpoissonW : ∀ f ∈ MvPolynomial.vanishingIdeal k
161       (MvPolynomial.zeroLocus k J),
162       ∀ g ∈ MvPolynomial.vanishingIdeal k (MvPolynomial.zeroLocus k J),
163         poissonBracket f g ∈ MvPolynomial.vanishingIdeal k
164           (MvPolynomial.zeroLocus k J) := by
165     intro f hf g hg
166     rw [MvPolynomial.vanishingIdeal_zeroLocus_eq_radical] at hf hg ⊢
167     exact Ideal.le_radical (hinv f (hrad hf) g (hrad hg))
168   obtain ⟨P, hP⟩ := exists_canonical_fibrePolynomial n N hd
169   have hPinitial : fibreLift P ∈ J := by
170     apply orderInitialIdeal_le_reducedOrderSupportIdeal k I
171     rw [hP]
172     exact canonical_orderPrincipalComponent_mem_initialIdeal k n N hd
173   have hPaxis : MvPolynomial.eval
174       (fun i : Fin (n + 1) =>
175         if i = ⟨0, Nat.zero_lt_succ n⟩ then (1 : k) else 0) P = 1 := by
176     have hcanonical := canonical_orderPrincipalComponent_eval_pureMomentumAxis
177       n N hd
178     rw [← hP] at hcanonical
179     have heval := eval₂_fibreLift (K := k) P
180       (fun _ : Fin (n + 1) => (0 : k))
181       (fun i : Fin (n + 1) =>
182         if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)
183     have hsplit : Sum.elim (fun _ : Fin (n + 1) => (0 : k))
184         (fun i : Fin (n + 1) =>
185           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) =
186         axisPoint k (.inr (0 : Fin (n + 1))) := by
187       funext i
188       rcases i with i | i
189       · simp [axisPoint]
190       · simp [axisPoint, Sum.inr.injEq] <;> rfl
191     rw [hsplit] at heval
192     simpa only [← MvPolynomial.aeval_def, MvPolynomial.aeval_eq_eval] using
193       heval.symm.trans hcanonical
194   obtain ⟨q, hq, hqcoord⟩ :=
195     exists_zero_base_coordinate_of_isFibreConical_using axisClosure
196       (k := k) (m := n + 1) (Nat.zero_lt_succ n)
197       (MvPolynomial.zeroLocus k J) hnonemptyW ⟨J, rfl⟩ hW hpoissonW P
198       (by
199         rw [MvPolynomial.vanishingIdeal_zeroLocus_eq_radical]
200         exact Ideal.le_radical hPinitial)
201       (by rw [hPaxis]; exact one_ne_zero)
202   have hqSupport : MvPolynomial.pointToPoint q ∈
203       orderCharacteristicSupport k I := by
204     rw [orderCharacteristicSupport_eq_zeroLocus_reduced k I,
205       PrimeSpectrum.mem_zeroLocus]
206     intro f hf
207     change f ∈ MvPolynomial.vanishingIdeal k {q}
208     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff]
209     exact (MvPolynomial.mem_zeroLocus_iff.mp hq) f hf
210   have hqCoordinate : MvPolynomial.pointToPoint q ∈
211       PrimeSpectrum.zeroLocus
212         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
213           Set (SymbolRing k (n + 1))) := by
214     rw [PrimeSpectrum.mem_zeroLocus]
215     intro f hf
216     change f ∈ MvPolynomial.vanishingIdeal k {q}
217     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff,
218       Set.mem_singleton_iff.mp hf]
219     simpa using hqcoord
220   exact Set.disjoint_left.mp hdisjoint hqSupport hqCoordinate
