18 /-!
19 # Adapter from canonical order support to the arbitrary coisotropic theorem
20 
21 The canonical order support is presented on the prime spectrum, while the
22 general exclusion theorem is stated for field-valued phase points.  This file
23 performs that change of presentation at the reduced order-support ideal.  The
24 canonical support-avoidance input supplies the coordinate hypothesis, and the
25 Gabber involutivity result supplies the self-involutivity of the same radical
26 ideal.
27 
28 This adapter is deliberately separate from the canonical support assembly, so
29 the general theorem can be integrated there without an import cycle.
30 -/

84 private theorem zeroLocus_isFibreConical_of_isHomogeneous
85     {n : ℕ} (J : Ideal (SymbolRing k n))
86     (hhom : J.IsHomogeneous
87       (MvPolynomial.weightedHomogeneousSubmodule k (@orderWeight n))) :
88     IsFibreConical (MvPolynomial.zeroLocus k J) := by
89   intro q hq a ha
90   let y : Fin n → k := fun i => q (.inl i)
91   let ξ : Fin n → k := fun i => q (.inr i)
92   have hqsplit : q = Sum.elim y ξ := by
93     funext i
94     rcases i with i | i <;> rfl
95   have hline (f : SymbolRing k n) (hf : f ∈ J) :
96       fibreLinePolynomial y ξ f = 0 := by
97     apply Polynomial.ext
98     intro d
99     rw [coeff_fibreLinePolynomial_eq_eval_weightedHomogeneousComponent]
100     rw [← hqsplit]
101     exact (MvPolynomial.mem_zeroLocus_iff.mp hq) _ (hhom d hf)
102   intro f hf
103   have heval := eval_fibreLinePolynomial y ξ f a
104   rw [hline f hf, Polynomial.eval_zero] at heval
105   simpa [y, ξ] using heval.symm

107 /-! The canonical reduced support ideal is an admissible input to the
108 general coisotropic exclusion theorem.  The only geometric fact used here is
109 Gabber's involutivity of the radical associated graded annihilator; its
110 base-relative fragment is obtained by restricting the first bracket entry to
111 the base polynomial subring. -/
112 
113 theorem algebraicallyClosedCanonicalSupportVanishing_using
114     (axisClosure : ∀ {k : Type u} [Field k] [IsAlgClosed k] [CharZero k],
115       CoordinateAxisClosureInput (k := k))
116     (hunit : CanonicalStrictUnitCoordinatePreimage.{u}) :
117     AlgebraicallyClosedCanonicalSupportVanishing.{u} := by
118   intro k _ _ _ n N d hN hd
119   let I := canonicalRightIdeal (presentedCoordinate k n) d N
120   let J : Ideal (SymbolRing k (n + 1)) := reducedOrderSupportIdeal k I
121   have hdisjoint : Disjoint
122       (orderCharacteristicSupport k I)
123       (PrimeSpectrum.zeroLocus
124         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
125           Set (SymbolRing k (n + 1)))) := by
126     exact canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
127       k n N (hunit k n N d hN hd)
128   by_contra hsupp
129   have hnonempty : (orderCharacteristicSupport k I).Nonempty :=
130     Set.nonempty_iff_ne_empty.mpr hsupp
131   have hproper : J ≠ ⊤ := by
132     intro htop
133     obtain ⟨p, hp⟩ := hnonempty
134     have hpzero : p ∈ PrimeSpectrum.zeroLocus J := by
135       rw [← orderCharacteristicSupport_eq_zeroLocus_reduced k I]
136       exact hp
137     rw [htop] at hpzero
138     simpa using hpzero
139   have hrad : J.IsRadical := by
140     simpa only [J] using reducedOrderSupportIdeal_isRadical k I
141   have hhom : J.IsHomogeneous
142       (MvPolynomial.weightedHomogeneousSubmodule k
143         (@orderWeight (n + 1))) := by
144     simpa only [J, reducedOrderSupportIdeal] using
145       (orderInitialIdeal_isHomogeneous (k := k) I).radical
146   have hinv : IsInvolutive J := by
147     change IsInvolutive (orderInitialIdeal k I).radical
148     rw [← annihilator_orderAssociatedGradedModule]
149     exact associatedGraded_radical_isInvolutive (k := k) I
150   have hW : IsFibreConical (MvPolynomial.zeroLocus k J) :=
151     zeroLocus_isFibreConical_of_isHomogeneous J hhom
152   have hnonemptyW : (MvPolynomial.zeroLocus k J).Nonempty := by
153     obtain ⟨M, hM, hJM⟩ := Ideal.exists_le_maximal J hproper
154     obtain ⟨q, hqM⟩ :=
155       MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hM
156     refine ⟨q, ?_⟩
157     rw [MvPolynomial.mem_zeroLocus_iff]
158     intro f hf
159     have hfM : f ∈ M := hJM hf
160     rw [hqM, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hfM
161     exact hfM
162   have hpoissonW : ∀ f ∈ MvPolynomial.vanishingIdeal k
163       (MvPolynomial.zeroLocus k J),
164       ∀ g ∈ MvPolynomial.vanishingIdeal k (MvPolynomial.zeroLocus k J),
165         poissonBracket f g ∈ MvPolynomial.vanishingIdeal k
166           (MvPolynomial.zeroLocus k J) := by
167     intro f hf g hg
168     rw [MvPolynomial.vanishingIdeal_zeroLocus_eq_radical] at hf hg ⊢
169     exact Ideal.le_radical (hinv f (hrad hf) g (hrad hg))
170   obtain ⟨P, hP⟩ := exists_canonical_fibrePolynomial n N hd
171   have hPinitial : fibreLift P ∈ J := by
172     apply orderInitialIdeal_le_reducedOrderSupportIdeal k I
173     rw [hP]
174     exact canonical_orderPrincipalComponent_mem_initialIdeal k n N hd
175   have hPaxis : MvPolynomial.eval
176       (fun i : Fin (n + 1) =>
177         if i = ⟨0, Nat.zero_lt_succ n⟩ then (1 : k) else 0) P = 1 := by
178     have hcanonical := canonical_orderPrincipalComponent_eval_pureMomentumAxis
179       n N hd
180     rw [← hP] at hcanonical
181     have heval := eval₂_fibreLift (K := k) P
182       (fun _ : Fin (n + 1) => (0 : k))
183       (fun i : Fin (n + 1) =>
184         if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)
185     have hsplit : Sum.elim (fun _ : Fin (n + 1) => (0 : k))
186         (fun i : Fin (n + 1) =>
187           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) =
188         axisPoint k (.inr (0 : Fin (n + 1))) := by
189       funext i
190       rcases i with i | i
191       · simp [axisPoint]
192       · simp [axisPoint, Sum.inr.injEq] <;> rfl
193     rw [hsplit] at heval
194     simpa only [← MvPolynomial.aeval_def, MvPolynomial.aeval_eq_eval] using
195       heval.symm.trans hcanonical
196   obtain ⟨q, hq, hqcoord⟩ :=
197     exists_zero_base_coordinate_of_isFibreConical_using axisClosure
198       (k := k) (m := n + 1) (Nat.zero_lt_succ n)
199       (MvPolynomial.zeroLocus k J) hnonemptyW ⟨J, rfl⟩ hW hpoissonW P
200       (by
201         rw [MvPolynomial.vanishingIdeal_zeroLocus_eq_radical]
202         exact Ideal.le_radical hPinitial)
203       (by rw [hPaxis]; exact one_ne_zero)
204   have hqSupport : MvPolynomial.pointToPoint q ∈
205       orderCharacteristicSupport k I := by
206     rw [orderCharacteristicSupport_eq_zeroLocus_reduced k I,
207       PrimeSpectrum.mem_zeroLocus]
208     intro f hf
209     change f ∈ MvPolynomial.vanishingIdeal k {q}
210     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff]
211     exact (MvPolynomial.mem_zeroLocus_iff.mp hq) f hf
212   have hqCoordinate : MvPolynomial.pointToPoint q ∈
213       PrimeSpectrum.zeroLocus
214         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
215           Set (SymbolRing k (n + 1))) := by
216     rw [PrimeSpectrum.mem_zeroLocus]
217     intro f hf
218     change f ∈ MvPolynomial.vanishingIdeal k {q}
219     rw [MvPolynomial.mem_vanishingIdeal_singleton_iff,
220       Set.mem_singleton_iff.mp hf]
221     simpa using hqcoord
222   exact Set.disjoint_left.mp hdisjoint hqSupport hqCoordinate
