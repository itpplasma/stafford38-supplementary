34 abbrev retainedGroundMap
35     {k : Type u} [Field k] [CharZero k]
36     {m : ℕ} (hm : 0 < m)
37     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
38     (column : GeneralDivisorialVisibleFrameColumn hm P) :=
39   letI : Algebra (CoordinateZeroLocalRing column.W.coefficientField)
40       (ComponentFractionField P) := column.W.ambientAlgebra
41   let V := column.W.place.valuation.toSubring
42   letI : IsDiscreteValuationRing V := column.W.place.isDiscrete
43   letI : Algebra column.W.coefficientField V :=
44       (relativeCoefficientMap column.W.coefficientField column.W.place).toAlgebra
45   letI : Algebra k (ResidueField V) :=
46       retainedResidueGroundAlgebra P ⟨0, hm⟩ column.W
47   algebraMap k (ResidueField V) =
48     (residue V).comp
49       (retainedComponentCoefficientMap P ⟨0, hm⟩ column.W)

53     {m : ℕ} (hm : 0 < m)
54     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
55     (column : GeneralDivisorialVisibleFrameColumn hm P) :=
