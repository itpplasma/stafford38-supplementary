27 /-- A residue transcendence basis cannot contain zero: injective multivariate
28 evaluation sends its variable to the selected element. -/
29 theorem transcendenceBasis_value_ne_zero
30     {k κ : Type u} [Field k] [Field κ] [Algebra k κ]
31     {t : Set κ}
32     (ht : IsTranscendenceBasis k ((↑) : t → κ)) (z : t) :
33     (z : κ) ≠ 0 := by
34   intro hz
35   have hinj : Function.Injective
36       (MvPolynomial.aeval (fun i : t => (i : κ)) : MvPolynomial t k →ₐ[k] κ) :=
37     (algebraicIndependent_iff_injective_aeval).mp ht.1
38   have hX : (MvPolynomial.X z : MvPolynomial t k) ≠ 0 :=
39     MvPolynomial.X_ne_zero z
40   apply hX
41   apply hinj
42   simp [hz]

44 /-- If the row-zero and row-one coordinates both have zero residue, rows
45 representing a residue transcendence basis avoid both. The chart row is
46 excluded by the complement subtype itself. -/
47 theorem selected_basis_rows_avoid_zero_one_chart
48     {k κ : Type u} [Field k] [Field κ] [Algebra k κ]
49     {m : ℕ} (hm : 0 < m) (chart : Fin (m + 1))
50     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
51     (value : Fin (m + 1) → κ) (t : Set κ)
52     (ht : IsTranscendenceBasis k ((↑) : t → κ))
53     (index : t → Fin m)
54     (hrep : ∀ z : t, value ((e (index z)).1) = (z : κ))
55     (hzero : value 0 = 0)
56     (hone : value (Fin.succ ⟨0, hm⟩) = 0) :
57     ∀ z : t,
58       (e (index z)).1 ≠ 0 ∧
59       (e (index z)).1 ≠ Fin.succ ⟨0, hm⟩ ∧
60       (e (index z)).1 ≠ chart := by
61   intro z
62   have hz0 : (z : κ) ≠ 0 := transcendenceBasis_value_ne_zero ht z
63   refine ⟨?_, ?_, (e (index z)).2⟩
64   · intro hrow
65     apply hz0
66     calc
67       (z : κ) = value ((e (index z)).1) := (hrep z).symm
68       _ = value 0 := congrArg value hrow
69       _ = 0 := hzero
70   · intro hrow
71     apply hz0
72     calc
73       (z : κ) = value ((e (index z)).1) := (hrep z).symm
74       _ = value (Fin.succ ⟨0, hm⟩) := congrArg value hrow
75       _ = 0 := hone

77 /-- The actual retained witness also makes the first two homogeneous
78 coordinates vanish in the same residue field. The second vanishing follows
79 from the stored relation `q₁ = q₀ * parameter`; it does not identify `q₀` with
80 the parameter. -/
81 theorem witness_q1_residue_eq_zero
82     {k : Type u} [Field k] [CharZero k]
83     {m : ℕ} (hm : 0 < m)
84     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
85     (w : GeneralDivisorialVisibleFrameWitness hm P) :
86     letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
87         (ComponentFractionField P) := w.column.W.ambientAlgebra
88     let V := w.column.W.place.valuation.toSubring
89     letI : IsLocalRing V := w.column.W.place.isDiscrete.toIsLocalRing
90     residue V (w.column.q (Fin.succ ⟨0, hm⟩)) = 0 := by
91   let C := w.column
92   let W := C.W
93   let F := ComponentFractionField P
94   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
95   let V := W.place.valuation.toSubring
96   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
97   change residue V (C.q (Fin.succ ⟨0, hm⟩)) = 0
98   apply (IsLocalRing.residue_eq_zero_iff _).2
99   have hq0 : C.q 0 ∈ maximalIdeal V :=
100     GeneralDivisorialVisibleFrame.witness_q0_mem_maximal hm P w
101   rw [C.q_parameter]
102   exact Ideal.mul_mem_right _ _ hq0
