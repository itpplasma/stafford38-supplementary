21 /-- Formal smoothness lifts a residue map from an algebra to the completion of its base. -/
22 theorem exists_formalSmooth_lift_to_baseCompletion [Algebra.FormallySmooth A B]
23     (r : B →ₐ[A] A ⧸ I) :
24     ∃ ψ : B →ₐ[A] AdicCompletion I A,
25       ((AdicCompletion.evalOneₐ I).restrictScalars A).comp ψ = r := by
26   exact Algebra.FormallySmooth.exists_adicCompletionEvalOneₐ_comp_eq r

28 /-- The ideal extension identity forced by an `A`-algebra map into `Â`. -/
29 theorem extendedIdeal_eq_map_of_lift (hJ : J = I.map (algebraMap A B))
30     (ψ : B →ₐ[A] AdicCompletion I A) :
31     I.map (algebraMap A (AdicCompletion I A)) = J.map (ψ : B →+* AdicCompletion I A) := by
32   have hcomp : (ψ : B →+* AdicCompletion I A).comp (algebraMap A B) =
33       algebraMap A (AdicCompletion I A) := by
34     exact RingHom.ext fun a => ψ.commutes a
35   calc
36     I.map (algebraMap A (AdicCompletion I A)) =
37         I.map ((ψ : B →+* AdicCompletion I A).comp (algebraMap A B)) := by
38       rw [hcomp]
39     _ = (I.map (algebraMap A B)).map (ψ : B →+* AdicCompletion I A) := by
40       rw [Ideal.map_map]

43 /-- A formal-smooth lift to `Â` extends continuously to the completion at an ideal
44 which is the extension of the base ideal. -/
45 def extendFormalSmoothLift (hJ : J = I.map (algebraMap A B))
46     (hfg : I.FG) (ψ : B →ₐ[A] AdicCompletion I A) :
47     AdicCompletion J B →+* AdicCompletion I A := by
48   let IC : Ideal (AdicCompletion I A) := I.map (algebraMap A (AdicCompletion I A))
49   have hIC : IC = J.map (ψ : B →+* AdicCompletion I A) := by
50     simpa [IC] using extendedIdeal_eq_map_of_lift I J hJ ψ
51   letI : IsAdicComplete IC (AdicCompletion I A) := by
52     rw [IsAdicComplete.map_algebraMap_iff]
53     exact AdicCompletion.isAdicComplete hfg
54   let e : AdicCompletion I A ≃ₐ[AdicCompletion I A]
55       AdicCompletion IC (AdicCompletion I A) := AdicCompletion.ofAlgEquiv IC

63     extendFormalSmoothLift I J hJ hfg ψ (AdicCompletion.of J B b) = ψ b := by
64   let IC : Ideal (AdicCompletion I A) := I.map (algebraMap A (AdicCompletion I A))
