19 /-- The induced equivalence of prime localizations for a ring equivalence and
20 the corresponding comapped prime. -/
21 noncomputable def atPrimeEquiv
22     {B C : Type u} [CommRing B] [CommRing C]
23     (e : B ≃+* C) (P : Ideal B) (Q : Ideal C) [P.IsPrime] [Q.IsPrime]
24     (hP : P = Q.comap e) : Localization.AtPrime P ≃+* Localization.AtPrime Q := by
25   have hcomplement : P.primeCompl.map e = Q.primeCompl := by
26     simpa only [hP] using e.map_primeCompl_comap_eq Q
27   exact IsLocalization.ringEquivOfRingEquiv (Localization.AtPrime P)
28     (Localization.AtPrime Q) e hcomplement

30 @[simp] theorem atPrimeEquiv_algebraMap
31     {B C : Type u} [CommRing B] [CommRing C]
32     (e : B ≃+* C) (P : Ideal B) (Q : Ideal C) [P.IsPrime] [Q.IsPrime]
33     (hP : P = Q.comap e) (b : B) :
34     atPrimeEquiv e P Q hP (algebraMap B (Localization.AtPrime P) b) =
35       algebraMap C (Localization.AtPrime Q) (e b) := by
36   have hloc : P.primeCompl ≤ Submonoid.comap (e : B →+* C) Q.primeCompl := by
37     intro x hx
38     change x ∉ P at hx
39     change e x ∉ Q
40     intro hxQ
41     apply hx
42     rw [hP]
43     exact hxQ
44   change (IsLocalization.map (Localization.AtPrime Q) (e : B →+* C) hloc)
45     (algebraMap B (Localization.AtPrime P) b) = _
46   exact IsLocalization.map_eq _ _

48 /-- A principal maximal-ideal generator transports across the induced
49 localization equivalence. -/
50 theorem span_transport_across_ring_equiv
51     {B C : Type u} [CommRing B] [CommRing C]
52     (e : B ≃+* C) (P : Ideal B) (Q : Ideal C) [P.IsPrime] [Q.IsPrime]
53     (hP : P = Q.comap e)
54     (s : C)
55     (hs : Ideal.span {algebraMap C (Localization.AtPrime Q) s} =
56       maximalIdeal (Localization.AtPrime Q)) :
57     Ideal.span {algebraMap B (Localization.AtPrime P) (e.symm s)} =
58       maximalIdeal (Localization.AtPrime P) := by
59   let eLoc := atPrimeEquiv e P Q hP
60   have heval' : eLoc.toRingHom (algebraMap B (Localization.AtPrime P) (e.symm s)) =
61       algebraMap C (Localization.AtPrime Q) s := by
62     simpa [eLoc] using atPrimeEquiv_algebraMap e P Q hP (e.symm s)
63   have hspanMap : (Ideal.span {algebraMap B (Localization.AtPrime P) (e.symm s)}).map
64       eLoc.toRingHom = (maximalIdeal (Localization.AtPrime P)).map eLoc.toRingHom := by
65     rw [Ideal.map_span]
66     rw [IsLocalRing.map_maximalIdeal_of_surjective eLoc.toRingHom eLoc.surjective]
67     simpa only [Set.image_singleton, heval'] using hs
68   have hmapInjective : Function.Injective (Ideal.map eLoc.toRingHom) := by
69     intro I J h
70     have h' := congrArg (Ideal.map eLoc.symm.toRingHom) h
71     simpa using h'
72   exact hmapInjective hspanMap

74 /-- Transport both order factorizations and the parameter selected on the
75 normalization subring to the integral-closure algebra carrier. The image
76 units are carried by the same induced local equivalence. -/
77 theorem transport_parameter_and_two_orders
78     {B C : Type u} [CommRing B] [CommRing C]
79     (e : B ≃+* C) (P : Ideal B) (Q : Ideal C) [P.IsPrime] [Q.IsPrime]
80     (hP : P = Q.comap e)
81     (s : C) (hs0 : s ≠ 0) (hsP : s ∈ Q)
82     {q₀ q₁ : B} {n₀ n₁ : ℕ} {u₀ u₁ : Localization.AtPrime Q}
83     (hspan : Ideal.span {algebraMap C (Localization.AtPrime Q) s} =
84       maximalIdeal (Localization.AtPrime Q))
85     (h₀ : algebraMap C (Localization.AtPrime Q) (e q₀) =
86       (algebraMap C (Localization.AtPrime Q) s) ^ n₀ * u₀)
87     (h₁ : algebraMap C (Localization.AtPrime Q) (e q₁) =
88       (algebraMap C (Localization.AtPrime Q) s) ^ n₁ * u₁)
89     (hu₀ : IsUnit u₀) (hu₁ : IsUnit u₁) :
90     let sB := e.symm s
91     ∃ u₀B u₁B : Localization.AtPrime P,
92       sB ≠ 0 ∧ sB ∈ P ∧
93       Ideal.span {algebraMap B (Localization.AtPrime P) sB} =
94         maximalIdeal (Localization.AtPrime P) ∧
95       IsUnit u₀B ∧ IsUnit u₁B ∧
96       algebraMap B (Localization.AtPrime P) q₀ =
97         (algebraMap B (Localization.AtPrime P) sB) ^ n₀ * u₀B ∧
98       algebraMap B (Localization.AtPrime P) q₁ =
99         (algebraMap B (Localization.AtPrime P) sB) ^ n₁ * u₁B := by
100   classical
101   dsimp only
102   let eLoc := atPrimeEquiv e P Q hP
103   let sB := e.symm s
104   let u₀B := eLoc.symm u₀
105   let u₁B := eLoc.symm u₁
106   have hsB0 : sB ≠ 0 := by
107     intro h
108     apply hs0
109     exact e.symm.injective (by simpa [sB] using h)
110   have hsBP : sB ∈ P := by
111     rw [hP]
112     change e sB ∈ Q
113     simpa [sB] using hsP
114   have hspanB := span_transport_across_ring_equiv e P Q hP s hspan
115   have hu₀B : IsUnit u₀B := IsUnit.map eLoc.symm.toMonoidHom hu₀
116   have hu₁B : IsUnit u₁B := IsUnit.map eLoc.symm.toMonoidHom hu₁
117   have h₀B : algebraMap B (Localization.AtPrime P) q₀ =
118       (algebraMap B (Localization.AtPrime P) sB) ^ n₀ * u₀B := by
119     apply eLoc.injective
120     rw [map_mul, map_pow]
121     simp [eLoc, u₀B, sB, atPrimeEquiv_algebraMap, h₀]
122   have h₁B : algebraMap B (Localization.AtPrime P) q₁ =
123       (algebraMap B (Localization.AtPrime P) sB) ^ n₁ * u₁B := by
124     apply eLoc.injective
125     rw [map_mul, map_pow]
126     simp [eLoc, u₁B, sB, atPrimeEquiv_algebraMap, h₁]
127   exact ⟨u₀B, u₁B, hsB0, hsBP, hspanB, hu₀B, hu₁B, h₀B, h₁B⟩
