8 /-!
9 # The exact completed-chart to arc-frame adapter boundary
10 
11 The completed projective certificate canonically supplies an affine
12 power-series arc after deleting its normalized chart coordinate.  Its
13 dehomogenized first jet is exactly the uniformizer member of
14 `residueFrameVector`.
15 
16 The remaining projective residue columns cannot be interpreted as
17 coefficient-field derivations over the field stored in the certificate: every
18 `k`-derivation of `k` is zero, whereas those columns are independent.  Thus a
19 smaller base field and geometric residue-field derivations are genuinely new
20 data, not consequences of `CompletedProjectiveBoundaryChart`.
21 
22 For the separate closed-tangent obligation, independent Jacobian rows from
23 the actual boundary ideal give the sharp annihilator-dimension formula and
24 therefore the finrank bound consumed by `ArcFrameConormal`.  This regularity
25 criterion concerns the closed point directly; no generic-rank specialization
26 is used.
27 -/

45 /-! ## The canonical affine arc and its first jet -/
46 
47 /-- The complement of one coordinate in `Fin (m+1)` has `m` elements. -/
48 def chartAffineCoordinateEquiv {m : ℕ} (chart : Fin (m + 1)) :
49     Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart := by
50   classical
51   apply (Fintype.equivFinOfCardEq (α := ChartAffineIndex (Fin (m + 1)) chart) ?_).symm
52   rw [Fintype.card_subtype_compl]
53   simp
54 
55 /-- Affine power-series coordinates obtained by deleting the normalized
56 projective chart coordinate.  Since that coordinate is exactly one, no
57 power-series division is needed. -/
58 def completedChartAffineArc [CharZero k]
59     {m : ℕ} (hm : 0 < m) (I : Ideal (MvPolynomial (Fin m) k))
60     (W : CompletedProjectiveBoundaryChart k m hm I) :
61     Fin m → PowerSeries k :=
62   fun i ↦ W.q ((chartAffineCoordinateEquiv W.chart i).1)
63 
64 /-- The actual uniformizer residue vector of the affine chart arc is exactly
65 the dehomogenized projective first jet. -/
66 theorem completedChart_uniformizerResidueFrame_eq_dehomogenizedFirstJet
67     [CharZero k] {m r : ℕ} (hm : 0 < m)
68     (I : Ideal (MvPolynomial (Fin m) k))
69     (W : CompletedProjectiveBoundaryChart k m hm I)
70     (D : Fin r → Derivation k k k) (i : Fin m) :
71     residueFrameVector (completedChartAffineArc hm I W) D none i =
72       chartDehomogenizedTangentColumn W.chart
73         (residueColumn W.q) (powerSeriesFirstCoefficient W.q)
74         (chartAffineCoordinateEquiv W.chart i) := by
75   simp [completedChartAffineArc, chartDehomogenizedTangentColumn,
76     powerSeriesFirstCoefficient, residueColumn, W.q_chart]
77   simpa [powerSeriesFirstCoefficient] using
78     congrFun (powerSeriesFirstCoefficient_eq_constantCoeff_derivative W.q)
79       ((chartAffineCoordinateEquiv W.chart i).1)

81 /-! ## Why the remaining columns are not a self-relative arc frame -/
82 
83 /-- Every derivation of a field relative to itself is zero. -/
84 theorem selfDerivation_apply_eq_zero (D : Derivation k k k) (a : k) :
85     D a = 0 := by
86   simpa using D.map_algebraMap a
87 
88 /-- Consequently every coefficient member of a `k/k` residue frame is the
89 zero tangent vector. -/
90 theorem residueFrameVector_coefficient_self_eq_zero
91     {m : ℕ} (q : Fin m → PowerSeries k)
92     {κ : Type v} (D : κ → Derivation k k k) (j : κ) :
93     residueFrameVector q D (some j) = 0 := by
94   funext i
95   rw [residueFrameVector_coefficient]
96   exact selfDerivation_apply_eq_zero (D j) _
97 
98 /-- If the completed chart has at least one divisor-tangent column, its
99 independent projective residue frame cannot equal a coefficient-derivation
100 frame over `k/k`, even after the canonical affine-coordinate reindexing. -/
101 theorem completedChart_no_selfRelative_fullResidueFrameIdentification
102     [CharZero k] {m : ℕ} (hm : 0 < m)
103     (I : Ideal (MvPolynomial (Fin m) k))
104     (W : CompletedProjectiveBoundaryChart k m hm I)
105     (hpositive : 0 < W.tangentCount) :
106     ¬ ∃ (q : Fin m → PowerSeries k)
107         (D : Fin W.tangentCount → Derivation k k k),
108       ∀ (j : Fin W.tangentCount) (i : Fin m),
109         residueFrameVector q D (some j) i =
110           chartDehomogenizedTangentColumn W.chart
111             (residueColumn W.q)
112             (fun a ↦ PowerSeries.constantCoeff
113               (nonpositionPowerSeriesMatrix W.Z W.tau a (Sum.inl j)))
114             (chartAffineCoordinateEquiv W.chart i) := by
115   rintro ⟨q, D, hidentify⟩
116   let j : Fin W.tangentCount := ⟨0, hpositive⟩
117   have hprojective :=
118     (completedBoundaryChart_residueFrame_linearIndependent hm I W).ne_zero
119       (Sum.inl j)
120   apply hprojective
121   funext a
122   let i : Fin m := (chartAffineCoordinateEquiv W.chart).symm a
123   have hz := congrFun (residueFrameVector_coefficient_self_eq_zero q D j) i
124   have hid := hidentify j i
125   have hres : residueColumn W.q = fun i ↦ PowerSeries.constantCoeff (W.q i) := rfl
126   simpa [i, hres, Equiv.apply_symm_apply] using hid.symm.trans hz

130 /-- Independent Jacobian rows have the expected common-kernel dimension. -/
131 theorem finrank_jacobianTangentSpace_eq_sub_of_linearIndependent
132     {n : ℕ} {ι : Type v} [Fintype ι]
133     (y : Fin n → k) (equations : ι → MvPolynomial (Fin n) k)
134     (hindependent :
135       LinearIndependent k (fun i ↦ differentialCovector y (equations i))) :
136     Module.finrank k (jacobianTangentSpace y equations) =
137       n - Fintype.card ι := by
138   have hspan : Module.finrank k (jacobianCovectorSpan y equations) =
139       Fintype.card ι := by
140     exact finrank_span_eq_card hindependent
141   have hdimension :=
142     Subspace.finrank_add_finrank_dualCoannihilator_eq
143       (jacobianCovectorSpan y equations)
144   rw [hspan, Module.finrank_pi, Fintype.card_fin] at hdimension
145   rw [jacobianTangentSpace]
146   omega

148 /-- Independent differentials of equations belonging to `I` give a sharp
149 upper bound on the equation-defined closed tangent space. -/
150 theorem zariskiTangent_finrank_le_sub_of_independentEquations
151     {n : ℕ} {ι : Type v} [Fintype ι]
152     (y : Fin n → k) (I : Ideal (MvPolynomial (Fin n) k))
153     (equations : ι → MvPolynomial (Fin n) k)
154     (hequations : ∀ i, equations i ∈ I)
155     (hindependent :
156       LinearIndependent k (fun i ↦ differentialCovector y (equations i))) :
157     Module.finrank k (zariskiTangentSpace y I) ≤
158       n - Fintype.card ι := by
159   have hcovectors :
160       jacobianCovectorSpan y equations ≤ equationCovectorSpan y I :=
161     jacobianCovectorSpan_le_equationCovectorSpan y I equations hequations
162   have htangent : zariskiTangentSpace y I ≤
163       jacobianTangentSpace y equations := by
164     exact Submodule.dualCoannihilator_anti hcovectors
165   calc
166     Module.finrank k (zariskiTangentSpace y I) ≤
167         Module.finrank k (jacobianTangentSpace y equations) :=
168       Submodule.finrank_mono htangent
169     _ = n - Fintype.card ι :=
170       finrank_jacobianTangentSpace_eq_sub_of_linearIndependent
171         y equations hindependent

173 /-- Cardinal form matching the `ArcFrameConormal` consumer: enough
174 independent closed-point Jacobian equations discharge its finrank input. -/
175 theorem zariskiTangent_finrank_le_frameCard_of_independentEquations
176     {n frameCount : ℕ} {ι : Type v} [Fintype ι]
177     (y : Fin n → k) (I : Ideal (MvPolynomial (Fin n) k))
178     (equations : ι → MvPolynomial (Fin n) k)
179     (hequations : ∀ i, equations i ∈ I)
180     (hindependent :
181       LinearIndependent k (fun i ↦ differentialCovector y (equations i)))
182     (hcodimension : n ≤ Fintype.card ι + frameCount) :
183     Module.finrank k (zariskiTangentSpace y I) ≤ frameCount := by
184   have hbound := zariskiTangent_finrank_le_sub_of_independentEquations
185     y I equations hequations hindependent
186   omega

188 /-- An annihilating arc with an independent residue frame spans the closed
189 tangent space once enough independent boundary Jacobian equations are
190 exhibited.  The finrank premise of `ArcFrameConormal` is now a conclusion. -/
191 theorem tangent_le_span_residueFrame_of_independentBoundaryEquations
192     {K : Type u} [Field K] [Algebra k K]
193     {m : ℕ} {κ : Type v} [Fintype κ]
194     {ι : Type v} [Fintype ι]
195     (I : Ideal (MvPolynomial (Fin m) k))
196     (q : Fin m → PowerSeries K)
197     (D : κ → Derivation k K K)
198     (equations : ι → MvPolynomial (Fin m) K)
199     (hequations : ∀ i,
200       equations i ∈ I.map (MvPolynomial.map (algebraMap k K)))
201     (hjacobian : LinearIndependent K (fun i ↦
202       differentialCovector
203         (fun a ↦ PowerSeries.constantCoeff (q a)) (equations i)))
204     (hcodimension :
205       m ≤ Fintype.card ι + Fintype.card (FrameIndex κ))
206     (hq : ∀ f ∈ I,
207       MvPolynomial.eval₂ (algebraMap k (PowerSeries K)) q f = 0)
208     (hframe : LinearIndependent K (residueFrameVector q D)) :
209     zariskiTangentSpace
210         (fun i ↦ PowerSeries.constantCoeff (q i))
211         (I.map (MvPolynomial.map (algebraMap k K))) ≤
212       Submodule.span K (Set.range (residueFrameVector q D)) := by
213   apply tangent_le_span_residueFrame_of_annihilatingArc I q D hq hframe
214   exact zariskiTangent_finrank_le_frameCard_of_independentEquations
215     _ _ equations hequations hjacobian hcodimension
