12 # Affine conormals are finite spans of equation differentials
13 
14 For an ideal of equations in affine space and a rational point, define the
15 embedded Zariski tangent space as the common kernel of the evaluated
16 differentials.  Its annihilator is exactly the span of those differentials.
17 Because the ambient affine space is finite-dimensional, every conormal
18 covector is therefore an explicit finite linear combination of equation
19 differentials.
20 
21 The final theorem feeds that finite representation into the pointwise
22 Hamiltonian-translation theorem.  No smoothness assumption is needed for this
23 linear-algebraic bridge.  What is not proved here is the scheme-geometric
24 identification of this equation-defined tangent space with the tangent fibre
25 of a smooth locus, or the passage from pointwise fibres to the closure of the
26 smooth conormal bundle.
27 -/

45 abbrev AffineTangentVector (k : Type*) (n : ℕ) := Fin n → k

47 /-- The differential of `f` at `y`, viewed as a linear functional on the
48 ambient affine tangent space. -/
49 def differentialCovector (y : Fin n → k) (f : MvPolynomial (Fin n) k) :
50     Module.Dual k (AffineTangentVector k n) where
51   toFun v := ∑ i, differentialAt y f i * v i
52   map_add' u v := by simp [mul_add, Finset.sum_add_distrib]
53   map_smul' a v := by simp [mul_assoc, mul_left_comm, mul_comm, Finset.mul_sum]
54 
55 @[simp]
56 theorem differentialCovector_apply (y : Fin n → k)
57     (f : MvPolynomial (Fin n) k) (v : AffineTangentVector k n) :
58     differentialCovector y f v = ∑ i, differentialAt y f i * v i :=
59   rfl

61 /-- Product rule for the evaluated differential covector.  All consumers,
62 including scalar extensions, use this low-level identity rather than carrying
63 private copies of the polynomial product-rule calculation. -/
64 theorem differentialCovector_mul (y : Fin n → k)
65     (p q : MvPolynomial (Fin n) k) :
66     differentialCovector y (p * q) =
67       MvPolynomial.eval y q • differentialCovector y p +
68         MvPolynomial.eval y p • differentialCovector y q := by
69   apply LinearMap.ext
70   intro v
71   change (∑ i, differentialAt y (p * q) i * v i) =
72     MvPolynomial.eval y q * (∑ i, differentialAt y p i * v i) +
73       MvPolynomial.eval y p * (∑ i, differentialAt y q i * v i)
74   simp only [differentialAt, MvPolynomial.pderiv_mul, map_add, map_mul,
75     Finset.mul_sum, ← Finset.sum_add_distrib]
76   refine Finset.sum_congr rfl fun i _ ↦ ?_
77   ring

79 /-- Covectors generated by the differentials of all equations in `I`. -/
80 def equationCovectorSpan (y : Fin n → k)
81     (I : Ideal (MvPolynomial (Fin n) k)) :
82     Submodule k (Module.Dual k (AffineTangentVector k n)) :=
83   Submodule.span k (Set.range fun f : I ↦ differentialCovector y f.1)
84 
85 /-- The embedded Zariski tangent space cut out by the first-order parts of all
86 equations in `I`. -/
87 def zariskiTangentSpace (y : Fin n → k)
88     (I : Ideal (MvPolynomial (Fin n) k)) :
89     Submodule k (AffineTangentVector k n) :=
90   (equationCovectorSpan y I).dualCoannihilator
91 
92 /-- The embedded conormal space, defined as the annihilator of the embedded
93 Zariski tangent space. -/
94 def affineConormalSpace (y : Fin n → k)
95     (I : Ideal (MvPolynomial (Fin n) k)) :
96     Submodule k (Module.Dual k (AffineTangentVector k n)) :=
97   (zariskiTangentSpace y I).dualAnnihilator
98 
99 /-- The embedded conormal is exactly the span of evaluated differentials of
100 the equation ideal.  This is finite-dimensional double-annihilator duality. -/
101 theorem affineConormalSpace_eq_equationCovectorSpan
102     (y : Fin n → k) (I : Ideal (MvPolynomial (Fin n) k)) :
103     affineConormalSpace y I = equationCovectorSpan y I := by
104   classical
105   exact Subspace.dualCoannihilator_dualAnnihilator_eq

107 /-- Every embedded conormal covector has an explicit finite-support expansion
108 in differentials of equations. -/
109 theorem mem_affineConormalSpace_iff_exists_finsupp
110     (y : Fin n → k) (I : Ideal (MvPolynomial (Fin n) k))
111     (phi : Module.Dual k (AffineTangentVector k n)) :
112     phi ∈ affineConormalSpace y I ↔
113       ∃ c : I →₀ k,
114         Finsupp.linearCombination k (fun f : I ↦ differentialCovector y f.1) c = phi := by
115   classical
116   rw [affineConormalSpace_eq_equationCovectorSpan]
117   exact Finsupp.mem_span_range_iff_exists_finsupp

119 /-- The coordinate covector associated with a fibre coordinate `xi`. -/
120 def coordinateCovector (xi : Fin n → k) :
121     Module.Dual k (AffineTangentVector k n) where
122   toFun v := ∑ i, xi i * v i
123   map_add' u v := by simp [mul_add, Finset.sum_add_distrib]
124   map_smul' a v := by simp [mul_assoc, mul_left_comm, mul_comm, Finset.mul_sum]
125 
126 @[simp]
127 theorem coordinateCovector_apply (xi v : Fin n → k) :
128     coordinateCovector xi v = ∑ i, xi i * v i :=
129   rfl

131 /-- Equality of coordinate covectors recovers equality of their coordinates. -/
132 theorem coordinateCovector_injective :
133     Function.Injective (coordinateCovector (k := k) (n := n)) := by
134   intro xi eta h
135   funext i
136   have hi := LinearMap.congr_fun h (Pi.single i 1)
137   simpa [coordinateCovector, Pi.single_apply] using hi

139 /-- A conormal coordinate covector admits a finite-support expansion in
140 gradients of equations. -/
141 theorem coordinate_mem_affineConormalSpace_exists_finsupp
142     (y xi : Fin n → k) (I : Ideal (MvPolynomial (Fin n) k)) :
143     coordinateCovector xi ∈ affineConormalSpace y I →
144       ∃ c : I →₀ k, ∀ i,
145         xi i = ∑ f ∈ c.support, c f * differentialAt y f.1 i := by
146   classical
147   intro hxi
148   obtain ⟨c, hc⟩ :=
149     (mem_affineConormalSpace_iff_exists_finsupp y I (coordinateCovector xi)).mp hxi
150   refine ⟨c, fun i ↦ ?_⟩
151   have hi := LinearMap.congr_fun hc (Pi.single i 1)
152   simpa [Finsupp.linearCombination_apply, differentialCovector, coordinateCovector,
153     Pi.single_apply, Finsupp.sum, mul_comm, mul_left_comm, mul_assoc] using hi.symm

155 /-- The genuine conormal consumer: under the existing base-relative Poisson
156 hypothesis, every coordinate covector in the equation-defined affine
157 conormal space gives a common zero in the corresponding support fibre. -/
158 theorem affineConormal_coordinatePoint_isCommonZero
159     [CharZero k]
160     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
161     (y : Fin n → k)
162     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
163     (I : Ideal (MvPolynomial (Fin n) k))
164     (hlift : ∀ f : I, baseLift f.1 ∈ J)
165     (xi : Fin n → k)
166     (hxi : coordinateCovector xi ∈ affineConormalSpace y I) :
167     ∀ g ∈ J, MvPolynomial.eval (Sum.elim y xi) g = 0 := by
168   classical
169   obtain ⟨c, hc⟩ :=
170     coordinate_mem_affineConormalSpace_exists_finsupp y xi I hxi
171   let ι := {f : I // f ∈ c.support}
172   let equations : ι → MvPolynomial (Fin n) k := fun f ↦ f.1.1
173   let coefficients : ι → k := fun f ↦ c f.1
174   have hpoint : differentialCombinationPoint y coefficients equations = Sum.elim y xi := by
175     funext q
176     rcases q with i | i
177     · rfl
178     · rw [differentialCombinationPoint_inr]
179       simp only [coefficients, equations, Sum.elim_inr]
180       rw [hc i]
181       exact (Finset.sum_subtype c.support (by simp)
182         (fun f ↦ c f * differentialAt y f.1 i)).symm
183   rw [← hpoint]
184   exact differentialCombinationPoint_isCommonZero J hJ y hzero equations
185     (fun f ↦ hlift f.1) coefficients

187 /-- Concrete characteristic-support form.  If `(y, eta)` lies in the reduced
188 order support and the required base-relative Gabber fragment holds, then the
189 entire equation-defined conormal space of the contracted base ideal over `y`
190 lies in the same reduced support fibre. -/
191 theorem reducedOrderSupport_affineConormal_coordinatePoint_isCommonZero
192     [CharZero k]
193     (W : RightIdeal (WeylAlg k n))
194     (hJ : IsBaseRelativePoisson (reducedOrderSupportIdeal k W))
195     (y eta : Fin n → k)
196     (hpoint : ∀ g ∈ reducedOrderSupportIdeal k W,
197       MvPolynomial.eval (Sum.elim y eta) g = 0)
198     (xi : Fin n → k)
199     (hxi : coordinateCovector xi ∈
200       affineConormalSpace y (reducedOrderBaseIdeal k W)) :
201     ∀ g ∈ reducedOrderSupportIdeal k W,
202       MvPolynomial.eval (Sum.elim y xi) g = 0 := by
203   apply affineConormal_coordinatePoint_isCommonZero
204     (reducedOrderSupportIdeal k W) hJ y
205     (zeroSection_mem_of_mem_reducedOrderSupport_zeroSet k W y eta hpoint)
206     (reducedOrderBaseIdeal k W)
207   · intro f
208     exact (mem_reducedOrderBaseIdeal_iff k W f.1).mp f.2
209   · exact hxi
