86 /-- The factors of a tilted point-local arc remain nonzero after passage to
87 its common open (paper proof, common-open step). -/
88 def CommonOpenArcFactors
89     {k A Q : Type u} [Field k] [CommRing A] [CommRing Q]
90     [Algebra k A] [Algebra Q A] {d : ℕ}
91     [Algebra (MvPolynomial (Option (Fin d)) k) A]
92     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) A]
93     [Algebra.EssFiniteType (MvPolynomial (Option (Fin d)) k) A]
94     (M : Ideal A) [M.IsMaximal]
95     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) (Localization.AtPrime M)]
96     (f : Q) (e : Localization.Away f ≃ₐ[Q] Localization.Away (algebraMap Q A f))
97     (g : Q) (eM : (A ⧸ M) ≃ₐ[k] k) (alpha : Fin d → k)
98     (bad r q0 q1 : A) : Prop :=
99   let ρA := originalToPointLocalFinSuccArcLaurentSeries
100     (k := k) (d := d) (A := A) M eM alpha
101   ∃ (hf : IsUnit (ρA (algebraMap Q A f)))
102     (hg : IsUnit (ρA (algebraMap Q A g))),
103     let hunitM : ∀ b, b ∉ M → IsUnit (ρA b) := fun b hb => by
104       exact IsUnit.map
105         (pointLocalFinSuccArcLaurentSeries (k := k) (d := d) (A := A) M eM alpha).toRingHom
106         (IsLocalization.map_units (Localization.AtPrime M) (M := M.primeCompl) ⟨b, hb⟩)
107     let ρU := genericArcToGenericOpenExtraAwayB M f e ρA hf hunitM g hg
108     (ρU.comp (genericOpenExtraAwayBMap M f e g) = ρA) ∧
109     ((ρU.comp (genericOpenExtraAwayBMap M f e g)).comp
110       (algebraMap k A) = algebraMap k (LaurentSeries k)) ∧
111     (ρU (genericOpenExtraAwayBMap M f e g
112       bad) ≠ 0) ∧
113     (ρU (genericOpenExtraAwayBMap M f e g r) ≠ 0) ∧
114     (ρU (genericOpenExtraAwayBMap M f e g
115       q0) ≠ 0) ∧
116     (ρU (genericOpenExtraAwayBMap M f e g
117       q1) ≠ 0)

119 /-- The tilted-product avoidance certificate yields the nonvanishing
120 common-open factors (paper proof, common-open step). -/
121 theorem commonOpenArcFactors_of_tiltedProduct
122     {k A Q : Type u} [Field k] [CommRing A] [CommRing Q]
123     [Algebra k A] [Algebra Q A] {d : ℕ}
124     [Algebra (MvPolynomial (Option (Fin d)) k) A]
125     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) A]
126     [Algebra.EssFiniteType (MvPolynomial (Option (Fin d)) k) A]
127     (M : Ideal A) [M.IsMaximal]
128     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) (Localization.AtPrime M)]
129     (f : Q) (e : Localization.Away f ≃ₐ[Q] Localization.Away (algebraMap Q A f))
130     (g : Q) (eM : (A ⧸ M) ≃ₐ[k] k) (alpha : Fin d → k)
131     (bad r q0 q1 : A)
132     (hbad : bad = algebraMap Q A f * r)
133     (hq0 : q0 = algebraMap Q A g)
134     (havoid : tiltedArc (k := k) alpha
135       (localToFinSuccPowerSeries (k := k) (B := A) (d := d) M eM
136           (algebraMap A (Localization.AtPrime M) bad) *
137        localToFinSuccPowerSeries (k := k) (B := A) (d := d) M eM
138           (algebraMap A (Localization.AtPrime M) q0) *
139        localToFinSuccPowerSeries (k := k) (B := A) (d := d) M eM
140           (algebraMap A (Localization.AtPrime M) q1)) ≠ 0) :
141     CommonOpenArcFactors M f e g eM alpha bad r q0 q1 := by
142   exact commonOpen_factors_ne_zero_of_tiltedProduct
143     M f e g eM alpha r bad q0 q1 hbad hq0 havoid

520 theorem exists_common_open_map_data
521     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]

525     (setup : ChartSetup hm P w)
526     (M : Ideal (actualSelectedNormalization P w)) [M.IsMaximal]
527     (hsel : SelectedAffineChartQuotient (k := k) P setup.j ≃ₐ[k]
528       actualSelectedChartAlgebra P w)
529     (qQ : Fin (m + 1) → actualSelectedChartAlgebra P w)
530     (hqchartQ : qQ (Fin.succ setup.j) = 1)
531     (hqQF : ∀ a, (qQ a : ComponentFractionField P) =
532       retainedColumnValue P w a)
533     (hq0 : qQ 0 = hsel (selectedAffineChartDenominator P setup.j))
534     (hqvars : ∀ i : Fin m, ∀ hij : i ≠ setup.j,
535       qQ (Fin.succ i) = hsel (selectedAffineChartVariableClass P setup.j i hij)) :
536     let Q := actualSelectedChartAlgebra P w
537     let B := actualSelectedNormalization P w
538     let g := hsel (selectedAffineChartDenominator P setup.j)
539     let Cq := genericOpenRing M setup.f setup.e
540     let U := genericOpenExtraAwayB M setup.f setup.e g
541     let qU : Fin (m + 1) → U := fun a =>
542       algebraMap Cq U (algebraMap Q Cq (qQ a))
543     let qT : Fin (m + 1) → Localization.AtPrime M := fun a =>
544       algebraMap B (Localization.AtPrime M)
545         ((retainedColumns hm P w) a)
546     ∃ hxj : componentCoordinate P setup.j ≠ 0,
547       (∀ a, (retainedColumns hm P w) a = algebraMap Q B (qQ a)) ∧
548       (∀ a, qU a = pointLocalToCommonOpen (A := B) M setup.f setup.e g (qT a)) ∧
549       (∀ i : Fin m, qU i.succ = qU 0 *
550         originalAffineChartToCommonOpen P setup.j hxj hsel M setup.f setup.e
551           (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i))) ∧
552       (retainedColumns hm P w) 0 =
553         algebraMap Q B g ∧
554       (genericOpenExtraAwayBMap M setup.f setup.e g).comp
555           (algebraMap Q B) = (algebraMap Cq U).comp (algebraMap Q Cq) := by
556   dsimp only
557   let Q := actualSelectedChartAlgebra P w
558   let B := actualSelectedNormalization P w
559   let g := hsel (selectedAffineChartDenominator P setup.j)
560   let Cq := genericOpenRing M setup.f setup.e
561   let U := genericOpenExtraAwayB M setup.f setup.e g
562   let qU : Fin (m + 1) → U := fun a =>
563     algebraMap Cq U (algebraMap Q Cq (qQ a))
564   let qB : Fin (m + 1) → B := (retainedColumns hm P w)
565   let qT : Fin (m + 1) → Localization.AtPrime M := fun a =>
566     algebraMap B (Localization.AtPrime M) (qB a)
567   have hqBspec :=
568     letI : Algebra
569         (Stafford38.Geometry.AsymptoticDivisorExistence.CoordinateZeroLocalRing
570           w.column.W.coefficientField)
571         (ComponentFractionField P) := w.column.W.ambientAlgebra
572     letI : IsLocalRing w.column.W.place.valuation.toSubring :=
573       w.column.W.place.isDiscrete.toIsLocalRing
574     actualNormalizedProjectiveColumnInIntegralClosure_spec hm P w
575   have hqBf : ∀ a, ((qB a : B) : ComponentFractionField P) =
576       retainedColumnValue P w a := by
577     simpa [qB] using hqBspec.2.1
578   have hqA : ∀ a, qB a = algebraMap Q B (qQ a) := by
579     intro a
580     exact actualColumn_eq_chartImage Q qQ qB
581       (fun a => retainedColumnValue P w a)
582       hqQF hqBf a
583   have hqUPoint : ∀ a, qU a =
584       pointLocalToCommonOpen (A := B) M setup.f setup.e g (qT a) := by
585     have hroute := pointLocal_commonOpen_column (A := B) M setup.f setup.e g
586       qQ qB hqA qT (fun a => rfl)
587     intro a
588     exact (hroute a).symm
589   have hxj : componentCoordinate P setup.j ≠ 0 :=
590     actualWitness_selected_coordinate_ne_zero hm P w setup.j setup.hchart
591   let φ := originalAffineChartToCommonOpen P setup.j hxj hsel M setup.f setup.e
592   have hchartU := q_column_commonOpen_coordinates P setup.j hxj hsel M setup.f
593     setup.e qQ hqchartQ hq0 hqvars
594   have hq0B : qB 0 = algebraMap Q B g := by
595     exact (hqA 0).trans (congrArg (algebraMap Q B) hq0)
596   have hbaseMap : (genericOpenExtraAwayBMap M setup.f setup.e g).comp
597       (algebraMap Q B) =
598       (algebraMap Cq U).comp (algebraMap Q Cq) := by
599     apply RingHom.ext
600     intro x
601     change algebraMap Cq U
602         (genericOpenBMap M setup.f setup.e (algebraMap Q B x)) =
603       algebraMap Cq U (algebraMap Q Cq x)
604     exact congrArg (algebraMap Cq U) (genericOpenBMap_base_eq M setup.f setup.e x)
