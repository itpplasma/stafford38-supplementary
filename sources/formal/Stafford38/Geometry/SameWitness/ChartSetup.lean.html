28 /-- The chart, smooth affine open, and homogenized numerator data retained
29 from one same-witness construction. The smooth open is the original affine
30 component open used in the paper's conormal closure argument. -/
31 structure ChartSetup
32     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
33     {m : ℕ} (hm : 0 < m)
34     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
35     (w : GeneralDivisorialVisibleFrameWitness hm P) where
36   fbar : MvPolynomial (Fin m) k ⧸ P.asIdeal
37   hfbar : fbar ≠ 0
38   p : MvPolynomial (Fin m) k
39   hrep : Ideal.Quotient.mk P.asIdeal p = fbar
40   hp : Ideal.Quotient.mk P.asIdeal p ≠ 0
41   hsmooth : Algebra.Smooth k (Localization.Away fbar)
42   j : Fin m
43   hchart : w.column.chart = Fin.succ j
44   f : actualSelectedChartAlgebra P w
45   hf : f ≠ 0
46   e : Localization.Away f ≃ₐ[actualSelectedChartAlgebra P w]
47     Localization.Away
48       (algebraMap (actualSelectedChartAlgebra P w)
49         (actualSelectedNormalization P w) f)
50   r : actualSelectedChartAlgebra P w
51   hr : r ≠ 0
52   hrF : (r : ComponentFractionField P) = MvPolynomial.eval
53     (fun a : Fin (m + 1) => (w.column.q a : ComponentFractionField P))
54     (MvPolynomial.map (algebraMap k (ComponentFractionField P))
55       (homogenizeAtZero p))
56   hrformula : (r : ComponentFractionField P) =
57     (w.column.q 0 : ComponentFractionField P) ^
58       (MvPolynomial.map (algebraMap k (ComponentFractionField P)) p).totalDegree *
59         componentAffineGenericPointMap P p
60   qToB_injective : Function.Injective
61     (chartSubalgebraToIntegralClosure (actualSelectedChartAlgebra P w))
62   fB_mul_rB_ne_zero :
63     chartSubalgebraToIntegralClosure (actualSelectedChartAlgebra P w) f *
64       chartSubalgebraToIntegralClosure (actualSelectedChartAlgebra P w) r ≠ 0

66 /-- The same-witness chart setup exists from a smooth principal open of the
67 original affine component. -/
68 theorem nonempty_chartSetup
69     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
70     {m : ℕ} (hm : 0 < m)
71     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
72     (w : GeneralDivisorialVisibleFrameWitness hm P)
73     (hsmoothOpen : ∃ fbar : MvPolynomial (Fin m) k ⧸ P.asIdeal,
74       fbar ≠ 0 ∧ Algebra.Smooth k (Localization.Away fbar)) :
75     Nonempty (ChartSetup hm P w) := by
76   classical
77   obtain ⟨fbar, hfbar, hsmooth⟩ := hsmoothOpen
78   obtain ⟨p, hrep⟩ := Ideal.Quotient.mk_surjective (I := P.asIdeal) fbar
79   have hp : Ideal.Quotient.mk P.asIdeal p ≠ 0 := by
80     rw [hrep]
81     exact hfbar
82   obtain ⟨j, hchart⟩ := exists_succ_chart_index hm P w
83   obtain ⟨f, hf, he⟩ := actual_selected_normalization_is_away_equiv P w
84   let e := Classical.choice he
85   obtain ⟨r, hrF, hrformula, hr⟩ :=
86     exists_nonzero_homogenized_numerator hm P w p hp
87   let Q := actualSelectedChartAlgebra P w
88   let qToB : Q →+* actualSelectedNormalization P w :=
89     chartSubalgebraToIntegralClosure Q
90   have hqToB_injective : Function.Injective qToB := by
91     intro x y hxy
92     apply Subtype.ext
93     calc
94       (x : ComponentFractionField P) = ((qToB x : actualSelectedNormalization P w) :
95           ComponentFractionField P) :=
96         (coe_chartSubalgebraToIntegralClosure Q x).symm
97       _ = ((qToB y : actualSelectedNormalization P w) :
98           ComponentFractionField P) := congrArg
99             (fun z : actualSelectedNormalization P w => (z : ComponentFractionField P)) hxy
100       _ = (y : ComponentFractionField P) := coe_chartSubalgebraToIntegralClosure Q y
101   have hfB : qToB f ≠ 0 := by
102     intro hz
103     apply hf
104     apply hqToB_injective
105     simpa using hz
106   have hrB : qToB r ≠ 0 := by
107     intro hz
108     apply hr
109     apply hqToB_injective
110     simpa using hz
111   have hbad : qToB f * qToB r ≠ 0 := mul_ne_zero hfB hrB
112   exact ⟨{
113     fbar := fbar
114     hfbar := hfbar
115     p := p
116     hrep := hrep
117     hp := hp
118     hsmooth := hsmooth
119     j := j
120     hchart := hchart
121     f := f
122     hf := hf
123     e := e
124     r := r
125     hr := hr
126     hrF := hrF
127     hrformula := hrformula
128     qToB_injective := hqToB_injective
129     fB_mul_rB_ne_zero := hbad
130   }⟩
131 
132 end Stafford38.Geometry.SameWitness
