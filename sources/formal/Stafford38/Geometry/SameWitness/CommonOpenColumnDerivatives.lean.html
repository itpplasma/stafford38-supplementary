34 def commonOpenDerivativeIdentities
35     {k U : Type u} [Field k] [CommRing U] {d n : ℕ}

41     (rho : U →+* LaurentSeries k)
42     (hground : rho.comp b.algebraMap = algebraMap k (LaurentSeries k))
43     (alpha : Fin d → k) (qPre : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)
44     (qU : Fin (n + 1) → U) : Prop :=
45     let R := MvPolynomial (Option (Fin d)) k
46     letI : Algebra R U := a
47     letI : SMul R U := a.toSMul
48     letI : Algebra k U := b
49     letI : SMul k U := b.toSMul
50     letI : IsScalarTower k R U := tower
51     letI : Algebra.FormallyEtale R U := etale
52     letI : Algebra U (LaurentSeries k) :=
53       RingHom.toAlgebra' rho (by intro x y; exact mul_comm _ _)
54     letI : Algebra k (LaurentSeries k) :=
55       RingHom.toAlgebra' (algebraMap k (LaurentSeries k))
56         (by intro x y; exact mul_comm _ _)
57     letI : Module k (LaurentSeries k) := Algebra.toModule
58     letI : SMul k (LaurentSeries k) :=
59       (Algebra.toModule : Module k (LaurentSeries k)).toSMul
60     letI : IsScalarTower k U (LaurentSeries k) :=
61       IsScalarTower.of_algebraMap_eq' hground.symm

63       ⟨fun x y z => by
64         change (algebraMap U (LaurentSeries k) x * y) * z = _
65         exact mul_assoc _ _ _⟩
66     (∀ j i, coordinateDerivation (k := k)
67     (σ := Option (Fin d))
68     (B := U) (L := LaurentSeries k) (some j)
69     (qU i) = algebraMap (PowerSeries k) (LaurentSeries k)
70       (tiltedTransverseDerivativeMatrix alpha qPre i j)) ∧
