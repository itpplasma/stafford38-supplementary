25 /-- The original-affine chart map into the common open is the identity on the
26 ground field: it commutes with `k` mapped through the original affine chart
27 algebra and `k` mapped through the ground ring `Q`. -/
28 theorem originalAffineChartToCommonOpen_groundMap
29     {k : Type u} [Field k] {m : ℕ}
30     {Q : Type v} [CommRing Q] [Algebra k Q]
31     {B : Type w} [CommRing B] [Algebra Q B]
32     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (j : Fin m)
33     (hxj : componentCoordinate P j ≠ 0)
34     (hsel : SelectedAffineChartQuotient (k := k) P j ≃ₐ[k] Q)
35     (M : Ideal B) [M.IsPrime] (f : Q)
36     (e : Localization.Away f ≃ₐ[Q]
37       Localization.Away (algebraMap Q B f)) :
38     let g := hsel (selectedAffineChartDenominator P j)
39     let Cq := genericOpenRing M f e
40     let U := genericOpenExtraAwayB M f e g
41     letI : Semiring U := (inferInstance : CommSemiring U).toSemiring
42     letI : Algebra Q Cq := inferInstance
43     letI : Algebra Q U := Algebra.compHom U (algebraMap Q Cq)
44     (originalAffineChartToCommonOpen P j hxj hsel M f e).comp
45         (algebraMap k (OriginalAffineChartQuotient (k := k) P)) =
46       (algebraMap Q U).comp (algebraMap k Q) := by
47   dsimp only
48   let g := hsel (selectedAffineChartDenominator P j)
49   let Cq := genericOpenRing M f e
50   let U := genericOpenExtraAwayB M f e g
51   letI : Semiring U := (inferInstance : CommSemiring U).toSemiring
52   letI : Algebra Q Cq := inferInstance
53   letI : Algebra Q U := Algebra.compHom U (algebraMap Q Cq)
54   let A₀ := OriginalAffineChartQuotient (k := k) P
55   let S₀ := OriginalAffineChartLocalization (k := k) P j
56   let T := Localization.Away g
57   let overlap : S₀ ≃ₐ[k] T :=
58     (originalAffineChartOverlapEquiv P j hxj).trans
59       (selectedChartAwayEquivOfQuotientEquiv P j hsel)
60   have hz : genericOpenBMap M f e (algebraMap Q B g) = algebraMap Q Cq g :=
61     genericOpenBMap_base_eq M f e g
62   letI : IsLocalization.Away (algebraMap Q Cq g) U := by
63     rw [← hz]
64     infer_instance
65   let ψ : T →+* U := IsLocalization.Away.map
66     (S := T) (Q := U) (algebraMap Q Cq) g
67   apply RingHom.ext
68   intro c
69   change ψ (overlap (algebraMap A₀ S₀ (algebraMap k A₀ c))) =
70     algebraMap Q U (algebraMap k Q c)
71   rw [← IsScalarTower.algebraMap_apply k A₀ S₀ c, overlap.commutes]
72   rw [IsScalarTower.algebraMap_apply k Q T c]
73   simp [ψ, IsLocalization.Away.map]
74   rfl
75 
76 end Stafford38.Geometry.SameWitness
