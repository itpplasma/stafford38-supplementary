69 /-- Homogenized evaluation of the retained projective column is the selected
70 chart numerator, which stays nonzero on the common-open Laurent arc. -/
71 theorem commonOpen_selectedNumerator_ne_zero
72     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
73     {m : ℕ} (hm : 0 < m)
74     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
75     (w : GeneralDivisorialVisibleFrameWitness hm P)
76     (setup : ChartSetup hm P w)
77     (coords : CoordinatePresentation hm P w setup)
78     (arc : CommonOpenArcData hm P w setup coords)
79     (etale : CommonOpenEtaleData hm P w setup coords arc)
80     (positions : CommonOpenPositionData hm P w setup coords arc) :
81     MvPolynomial.eval positions.qL
82       (MvPolynomial.map (algebraMap k (LaurentSeries k))
83         (homogenizeAtZero setup.p)) ≠ 0 := by
84   classical
85   letI : arc.common.M.IsMaximal := arc.common.hM
86   let B := actualSelectedNormalization P w
87   let F := ComponentFractionField P
88   let qB : Fin (m + 1) → B := actualNormalizedProjectiveColumnInIntegralClosure hm P w
89   have hqBspec := actualNormalizedProjectiveColumnInIntegralClosure_spec hm P w
90   have hqBf : ∀ a, (qB a : F) = (w.column.q a : F) := hqBspec.2.1
91   have hgroundCoeff : (algebraMap B F).comp coords.coeff = algebraMap k F := by
92     ext c
93     change ((coords.coeff c : B) : F) = algebraMap k F c
94     rw [coords.hcoeff]
95     rfl
96   have hEvalB : MvPolynomial.eval₂ coords.coeff qB
97       (homogenizeAtZero setup.p) = algebraMap
98         (actualSelectedChartAlgebra P w) B setup.r := by
99     apply Subtype.ext
100     change algebraMap B F (MvPolynomial.eval₂ coords.coeff qB
101       (homogenizeAtZero setup.p)) = (setup.r : F)
102     calc
103       _ = MvPolynomial.eval₂ ((algebraMap B F).comp coords.coeff)
104           (fun a => algebraMap B F (qB a)) (homogenizeAtZero setup.p) := by
105             exact (MvPolynomial.eval₂_comp_left (algebraMap B F)
106               coords.coeff qB (homogenizeAtZero setup.p))
107       _ = MvPolynomial.eval₂ (algebraMap k F)
108           (fun a => (w.column.q a : F)) (homogenizeAtZero setup.p) := by
109             rw [hgroundCoeff]
110             congr 1
111             funext a
112             exact hqBf a
113       _ = MvPolynomial.eval (fun a => (w.column.q a : F))
114           (MvPolynomial.map (algebraMap k F) (homogenizeAtZero setup.p)) := by
115             exact (MvPolynomial.eval_map (algebraMap k F)
116               (fun a => (w.column.q a : F)) (homogenizeAtZero setup.p)).symm
117       _ = (setup.r : F) := setup.hrF.symm
118   have hEval := @eval_map_eq_arc_eval₂ k B arc.common.U (LaurentSeries k)
119     (Fin (m + 1)) _ _ _ _ coords.coeff.toAlgebra inferInstance
120     (genericOpenExtraAwayBMap arc.common.M setup.f setup.e arc.common.g) arc.rhoU
121     arc.hgroundB' qB arc.common.qU positions.qL
122     positions.hcoords positions.hpositionL (homogenizeAtZero setup.p)
123   have hEval' : MvPolynomial.eval positions.qL
124       (MvPolynomial.map (algebraMap k (LaurentSeries k))
125         (homogenizeAtZero setup.p)) =
126       arc.rhoU (genericOpenExtraAwayBMap arc.common.M setup.f setup.e
127         arc.common.g (MvPolynomial.eval₂ coords.coeff qB (homogenizeAtZero setup.p))) := hEval
128   rw [hEval', hEvalB]
129   exact arc.hrU

131 /-- Derive the endpoint derivative and numerator facts from the retained
132 point-local columns without installing scalar actions on selected rings. -/
133 theorem commonOpenColumnsData_of_arc
134     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
135     {m : ℕ} (hm : 0 < m)
136     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
137     (w : GeneralDivisorialVisibleFrameWitness hm P)
138     (setup : ChartSetup hm P w)
139     (coords : CoordinatePresentation hm P w setup)
140     (arc : CommonOpenArcData hm P w setup coords)
141     (etale : CommonOpenEtaleData hm P w setup coords arc)
142     (positions : CommonOpenPositionData hm P w setup coords arc) :
143     CommonOpenColumnsData hm P w setup coords arc etale positions := by
144   exact ⟨commonOpen_arcDerivativeData hm P w setup coords arc etale positions,
