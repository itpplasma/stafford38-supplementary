25     {k A Q : Type u} [Field k] [CommRing A] [CommRing Q]

27     (M : Ideal A) [M.IsMaximal] (f : Q)
28     (e : Localization.Away f ≃ₐ[Q] Localization.Away (algebraMap Q A f))
29     (g : Q) (coeff : k →+* A) (canonical : A →+* LaurentSeries k)
30     (bad rB q0 q1 : A) where
31   rhoA : A →+* LaurentSeries k
32   hcanonicalRhoA : rhoA = canonical
33   hf : IsUnit (rhoA (algebraMap (Q)
34     (A) f))
35   hg : IsUnit (rhoA (algebraMap (Q)
36     (A) g))
37   hunitM : ∀ b, b ∉ M → IsUnit (rhoA b)
38   rhoU : genericOpenExtraAwayB M f e g →+* LaurentSeries k
39   hcanonicalRhoU :
40     rhoU = genericArcToGenericOpenExtraAwayB
41     M f e rhoA hf hunitM g hg
42   hcomp' :
43     rhoU.comp
44       (genericOpenExtraAwayBMap M f e g) = rhoA
45   hgroundB' :
46     (rhoU.comp
47       (genericOpenExtraAwayBMap M f e g)).comp
48       coeff =
49     algebraMap k (LaurentSeries k)
50   hbaseMap :
51     (genericOpenExtraAwayBMap M f e g).comp
52       coeff =
53     algebraMap k (genericOpenExtraAwayB M f e g)
54   hgroundU : rhoU.comp (algebraMap k (genericOpenExtraAwayB M f e g)) = algebraMap k (LaurentSeries k)
55   hbadU :

59     rhoU (genericOpenExtraAwayBMap M f e g

62     rhoU (genericOpenExtraAwayBMap M f e g
63     q0) ≠ 0

71 private theorem pointLocalArc_unit_of_not_mem
72     {k A : Type u} [Field k] [CommRing A] [Algebra k A] {d : ℕ}
73     [Algebra (MvPolynomial (Option (Fin d)) k) A]
74     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) A]
75     [Algebra.EssFiniteType (MvPolynomial (Option (Fin d)) k) A]
76     (M : Ideal A) [M.IsMaximal]
77     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) (Localization.AtPrime M)]
78     (eM : (A ⧸ M) ≃ₐ[k] k) (alpha : Fin d → k)
79     (b : A) (hb : b ∉ M) :
80     IsUnit (originalToPointLocalFinSuccArcLaurentSeries
81       (k := k) (d := d) (A := A) M eM alpha b) := by
82   exact IsUnit.map
83     (pointLocalFinSuccArcLaurentSeries (k := k) (d := d) (A := A) M eM alpha).toRingHom
84     (IsLocalization.map_units (Localization.AtPrime M) (M := M.primeCompl) ⟨b, hb⟩)

86 /-- The tilted-product factors determine the Laurent arc transport and its
87 restriction to the ground field (paper proof, common-open arc step). -/
88 private theorem exists_commonOpenArcTransport
89     {k A Q : Type u} [Field k] [CommRing A] [CommRing Q]
90     [Algebra k A] [Algebra k Q] [Algebra Q A] {d : ℕ}
91     [Algebra (MvPolynomial (Option (Fin d)) k) A]
92     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) A]
93     [Algebra.EssFiniteType (MvPolynomial (Option (Fin d)) k) A]
94     (M : Ideal A) [M.IsMaximal]
95     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) (Localization.AtPrime M)]
96     (f : Q) (e : Localization.Away f ≃ₐ[Q] Localization.Away (algebraMap Q A f))
97     (g : Q) (eM : (A ⧸ M) ≃ₐ[k] k) (alpha : Fin d → k)
98     (bad rB q0 q1 : A)
99     (hfactor : CommonOpenArcFactors M f e g eM alpha bad rB q0 q1)
100     (hbaseMap : (genericOpenExtraAwayBMap M f e g).comp (algebraMap k A) =
101       algebraMap k (genericOpenExtraAwayB M f e g)) :
102     Nonempty (CommonOpenArcTransport M f e g (algebraMap k A)
103       (originalToPointLocalFinSuccArcLaurentSeries (k := k) (d := d) (A := A) M eM alpha)
104       bad rB q0 q1) := by
105   classical
106   let rhoA := originalToPointLocalFinSuccArcLaurentSeries
107     (k := k) (d := d) (A := A) M eM alpha
108   unfold CommonOpenArcFactors at hfactor
109   rcases hfactor with ⟨hf, hg, hcomp, hgroundB, hbadU, hrU, hq0U, hq1U⟩
110   let hunitM : ∀ b, b ∉ M → IsUnit (rhoA b) := fun b hb =>
111     pointLocalArc_unit_of_not_mem M eM alpha b hb
112   let rhoU := genericArcToGenericOpenExtraAwayB M f e rhoA hf hunitM g hg
113   have hgroundU : rhoU.comp (algebraMap k (genericOpenExtraAwayB M f e g)) =
114       algebraMap k (LaurentSeries k) := by
115     exact (congrArg (fun h => rhoU.comp h) hbaseMap).symm.trans hgroundB
116   exact ⟨{
117     rhoA := rhoA, hcanonicalRhoA := rfl
118     hf := hf, hg := hg, hunitM := hunitM
119     rhoU := rhoU, hcanonicalRhoU := rfl
120     hcomp' := hcomp, hgroundB' := hgroundB, hbaseMap := hbaseMap
121     hgroundU := hgroundU, hbadU := hbadU, hrU := hrU
122     hq0U := hq0U, hq1U := hq1U
123   }⟩

127 structure CommonOpenArcData
128     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
129     {m : ℕ} (hm : 0 < m)
130     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
131     (w : GeneralDivisorialVisibleFrameWitness hm P)
132     (setup : ChartSetup hm P w)
133     (coords : CoordinatePresentation hm P w setup) where
134   common : CommonOpenData hm P w setup coords
135   transport :
136     letI : Algebra
137         (Stafford38.Geometry.AsymptoticDivisorExistence.CoordinateZeroLocalRing
138           w.column.W.coefficientField)
139         (Stafford38.Geometry.ComponentProjectiveClosure.ComponentFractionField P) :=
140       w.column.W.ambientAlgebra
141     letI : IsLocalRing w.column.W.place.valuation.toSubring :=
142       w.column.W.place.isDiscrete.toIsLocalRing
143     letI : common.M.IsMaximal := common.hM
144     let B := actualSelectedNormalization P w
145     let R := MvPolynomial (Option (Fin (@Fintype.card coords.t coords.htFinite))) k
146     letI : Algebra k B := coords.coeff.toAlgebra
147     letI : Algebra R B := coords.fFin.toRingHom.toAlgebra
148     letI : IsScalarTower k R B := IsScalarTower.of_algHom coords.fFin
149     letI : Algebra.FiniteType k B := coords.hBfinite
150     letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
151     letI : Algebra R (Localization.AtPrime common.M) := inferInstance
152     letI : Algebra.FormallyEtale R (Localization.AtPrime common.M) := common.hEtM
153     CommonOpenArcTransport common.M setup.f setup.e common.g
154       (@algebraMap k B _ _ coords.coeff.toAlgebra)
155       (originalToPointLocalFinSuccArcLaurentSeries
156         (k := k) (d := @Fintype.card coords.t coords.htFinite)
157         (A := B) common.M common.eM common.alpha)
158       (algebraMap (actualSelectedChartAlgebra P w) B setup.f *
159         algebraMap (actualSelectedChartAlgebra P w) B setup.r)
160       (algebraMap (actualSelectedChartAlgebra P w) B setup.r)
161       (Stafford38.Geometry.ActualOptionColumnBinding.actualNormalizedProjectiveColumnInIntegralClosure hm P w 0)
162       (Stafford38.Geometry.ActualOptionColumnBinding.actualNormalizedProjectiveColumnInIntegralClosure hm P w
163         (Fin.succ (⟨0, hm⟩ : Fin m)))

217 /-- The selected common-open map carries the canonical coefficient map to
218 the existing ground map on its localization. -/
219 private theorem commonOpenMap_comp_groundMap
220     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
221     {m : ℕ} (hm : 0 < m)
222     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
223     (w : GeneralDivisorialVisibleFrameWitness hm P)
224     (setup : ChartSetup hm P w)
225     (coords : CoordinatePresentation hm P w setup)
226     (common : CommonOpenData hm P w setup coords) :
227     letI : common.M.IsMaximal := common.hM
228     (genericOpenExtraAwayBMap common.M setup.f setup.e common.g).comp
229       (@algebraMap k (actualSelectedNormalization P w) _ _ coords.coeff.toAlgebra) =
230     algebraMap k common.U := by
231   letI : common.M.IsMaximal := common.hM
232   let B := actualSelectedNormalization P w
233   let Q := actualSelectedChartAlgebra P w
234   have hcoeff : @algebraMap k B _ _ coords.coeff.toAlgebra =
235       (algebraMap Q B).comp (algebraMap k Q) := by
236     rw [coords.hcoeff]
237     apply RingHom.ext
238     intro c
239     rfl
240   apply RingHom.ext
241   intro c
242   have h := congrArg
243     (fun f : Q →+* common.U => f (algebraMap k Q c)) common.hbaseMap
244   calc
245     genericOpenExtraAwayBMap common.M setup.f setup.e common.g
246         (coords.coeff c) =
247       genericOpenExtraAwayBMap common.M setup.f setup.e common.g
248         (algebraMap Q B (algebraMap k Q c)) :=
249           congrArg (genericOpenExtraAwayBMap common.M setup.f setup.e common.g)
250             (RingHom.congr_fun hcoeff c)
251     _ = (algebraMap common.Cq common.U)
252         (algebraMap Q common.Cq (algebraMap k Q c)) := h
253     _ = algebraMap k common.U c := by
254       exact (congrArg (algebraMap common.Cq common.U)
255         (IsScalarTower.algebraMap_apply k Q common.Cq c).symm).trans
256           (IsScalarTower.algebraMap_apply k common.Cq common.U c).symm

258 /-- Construct the Laurent-series arc on the common open from the retained
259 ground-point tilt and its factor-avoidance certificate. -/
260 theorem exists_commonOpenArcData
261     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
262     {m : ℕ} (hm : 0 < m)
263     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
264     (w : GeneralDivisorialVisibleFrameWitness hm P)
265     (setup : ChartSetup hm P w)
266     (coords : CoordinatePresentation hm P w setup)
267     (common : CommonOpenData hm P w setup coords) :
268     Nonempty (CommonOpenArcData hm P w setup coords) := by
269   classical
270   letI : common.M.IsMaximal := common.hM
271   let B := actualSelectedNormalization P w
272   let Q := actualSelectedChartAlgebra P w
273   refine ⟨{ common := common, transport := ?_ }⟩
274   apply Classical.choice
275   apply @exists_commonOpenArcTransport k B Q
276     inferInstance inferInstance inferInstance coords.coeff.toAlgebra
277     (inferInstance : Algebra k Q) (inferInstance : Algebra Q B)
278     (@Fintype.card coords.t coords.htFinite)
279     (@RingHom.toAlgebra
280       (MvPolynomial (Option (Fin (@Fintype.card coords.t coords.htFinite))) k) B
281       inferInstance inferInstance
282       (@AlgHom.toRingHom k
283         (MvPolynomial (Option (Fin (@Fintype.card coords.t coords.htFinite))) k) B
284         inferInstance inferInstance inferInstance inferInstance
285         coords.coeff.toAlgebra coords.fFin))
286     (@IsScalarTower.of_algHom k
287       (MvPolynomial (Option (Fin (@Fintype.card coords.t coords.htFinite))) k) B
288       inferInstance inferInstance inferInstance inferInstance
289       coords.coeff.toAlgebra coords.fFin)
290     (@essFiniteType_of_coordinateMap k B inferInstance inferInstance
291       coords.coeff.toAlgebra coords.hBfinite
292       (@Fintype.card coords.t coords.htFinite) coords.fFin)
293     common.M common.hM common.hEtM
294   · exact common.hfactor
295   · exact commonOpenMap_comp_groundMap hm P w setup coords common
