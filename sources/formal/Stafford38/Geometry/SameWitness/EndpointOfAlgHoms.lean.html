28     {k E : Type u} [Field k] [IsAlgClosed k] [CharZero k]
29     [CommRing E] [Algebra k E] {n d : ℕ}
30     {I : Ideal (MvPolynomial (Fin n) k)} [I.IsPrime]
31     (φ : (MvPolynomial (Fin n) k ⧸ I) →ₐ[k] E)
32     (ψ : MvPolynomial (Option (Fin d)) k →ₐ[k] E)
33     (hφEtale : letI := φ.toRingHom.toAlgebra
34       Algebra.FormallyEtale (MvPolynomial (Fin n) k ⧸ I) E)
35     (hψEtale : letI := ψ.toRingHom.toAlgebra
36       Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) E)
37     (rho : E →+* LaurentSeries k)
38     (hground : rho.comp (algebraMap k E) = algebraMap k (LaurentSeries k)) :
39     letI := ψ.toRingHom.toAlgebra
40     letI : IsScalarTower k (MvPolynomial (Option (Fin d)) k) E :=
41       IsScalarTower.of_algebraMap_eq' ψ.comp_algebraMap.symm
42     letI : Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) E := hψEtale
43     letI : Algebra E (LaurentSeries k) :=
44       RingHom.toAlgebra' rho (by intro x y; exact mul_comm _ _)
45     letI : Algebra k (LaurentSeries k) :=
46       RingHom.toAlgebra' (algebraMap k (LaurentSeries k))
47         (by intro x y; exact mul_comm _ _)
48     letI : Module k (LaurentSeries k) := Algebra.toModule
49     letI : SMul k (LaurentSeries k) :=
50       (Algebra.toModule : Module k (LaurentSeries k)).toSMul

52       IsScalarTower.of_algebraMap_eq' hground.symm

54     (qPre : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)

56     (a b : ℕ) (beta alpha : Fin d → k)

63         (tiltedArc alpha (qPre i)))
64     (htransverse : ∀ j i,

72       coordinateDerivation (k := k) (σ := Option (Fin d)) (B := E)
73         (L := LaurentSeries k) none (qC i) +
74         ∑ j, algebraMap (PowerSeries k) (LaurentSeries k)
75           (PowerSeries.C (alpha j)) *
76             coordinateDerivation (k := k) (σ := Option (Fin d)) (B := E)
77               (L := LaurentSeries k) (some j) (qC i))

87         (homogenizeAtZero p)) ≠ 0),
