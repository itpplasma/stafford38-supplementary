15 /-- Transport a unit-power factorization from an away localization to the
16 localization at a prime avoiding its denominator. -/
17 theorem factorization_to_atPrime
18     {B : Type u} [CommRing B] (M : Ideal B) [M.IsPrime]
19     (f : B) (hf : f ∉ M) (q s : B) (n : ℕ)
20     (v : Localization.Away f) (hv : IsUnit v)
21     (h : algebraMap B (Localization.Away f) q =
22       (algebraMap B (Localization.Away f) s) ^ n * v) :
23     ∃ u : Localization.AtPrime M, IsUnit u ∧
24       algebraMap B (Localization.AtPrime M) q =
25         (algebraMap B (Localization.AtPrime M) s) ^ n * u := by
26   let ψ : Localization.Away f →+* Localization.AtPrime M :=
27     IsLocalization.Away.lift f
28       (IsLocalization.map_units (Localization.AtPrime M)
29         (M := M.primeCompl) ⟨f, hf⟩)
30   have hψ : ψ.comp (algebraMap B (Localization.Away f)) =
31       algebraMap B (Localization.AtPrime M) :=
32     IsLocalization.Away.lift_comp (R := B) (S := Localization.Away f)
33       (P := Localization.AtPrime M) (x := f) _
34   have hψb (b : B) :
35       ψ (algebraMap B (Localization.Away f) b) =
36         algebraMap B (Localization.AtPrime M) b := by
37     exact DFunLike.congr_fun hψ b
38   refine ⟨ψ v, hv.map ψ, ?_⟩
39   have hh := congrArg ψ h
40   calc
41     algebraMap B (Localization.AtPrime M) q =
42         ψ (algebraMap B (Localization.Away f) q) := (hψb q).symm
43     _ = ψ ((algebraMap B (Localization.Away f) s) ^ n * v) := hh
44     _ = algebraMap B (Localization.AtPrime M) s ^ n * ψ v := by
45       rw [map_mul, map_pow, hψb]

47 /-- Transport both divisor-order factorizations from the one ground-point
48 away open to its local ring. -/
49 theorem pair_factorizations_to_atPrime
50     {B : Type u} [CommRing B] (M : Ideal B) [M.IsPrime]
51     (f : B) (hf : f ∉ M) (s q₀ q₁ : B) (e₀ e₁ : ℕ)
52     (v₀ v₁ : Localization.Away f)
53     (hv₀ : IsUnit v₀) (hv₁ : IsUnit v₁)
54     (h₀ : algebraMap B (Localization.Away f) q₀ =
55       (algebraMap B (Localization.Away f) s) ^ e₀ * v₀)
56     (h₁ : algebraMap B (Localization.Away f) q₁ =
57       (algebraMap B (Localization.Away f) s) ^ e₁ * v₁) :
58     ∃ u₀ u₁ : Localization.AtPrime M,
59       IsUnit u₀ ∧ IsUnit u₁ ∧
60       algebraMap B (Localization.AtPrime M) q₀ =
61         (algebraMap B (Localization.AtPrime M) s) ^ e₀ * u₀ ∧
62       algebraMap B (Localization.AtPrime M) q₁ =
63         (algebraMap B (Localization.AtPrime M) s) ^ e₁ * u₁ := by
64   obtain ⟨u₀, hu₀, hq₀⟩ := factorization_to_atPrime M f hf q₀ s e₀ v₀ hv₀ h₀
65   obtain ⟨u₁, hu₁, hq₁⟩ := factorization_to_atPrime M f hf q₁ s e₁ v₁ hv₁ h₁
66   exact ⟨u₀, u₁, hu₀, hu₁, hq₀, hq₁⟩
