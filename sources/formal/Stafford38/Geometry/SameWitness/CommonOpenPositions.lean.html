24 /-- The prescribed completion chart as a ring homomorphism, with all
25 coordinate scalar structures installed only over the abstract ring B. -/
26 noncomputable def pointLocalPowerSeriesChart
27     {k B : Type u} [Field k] [CommRing B] {d : ℕ}
28     (coeff : k →+* B)
29     (fFin : @AlgHom k (MvPolynomial (Option (Fin d)) k) B _ _ _ _ coeff.toAlgebra)
30     (hfinite : @Algebra.FiniteType k B _ _ coeff.toAlgebra)
31     (M : Ideal B) [M.IsMaximal] :
32     letI : Algebra k B := coeff.toAlgebra
33     let R := MvPolynomial (Option (Fin d)) k
34     letI : Algebra R B := fFin.toRingHom.toAlgebra
35     letI : IsScalarTower k R B := IsScalarTower.of_algebraMap_eq' (by
36       ext c; exact (fFin.commutes c).symm)
37     letI : Algebra.FiniteType k B := hfinite
38     letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
39     ∀ (eM : (B ⧸ M) ≃ₐ[k] k)
40     (hEtM : Algebra.FormallyEtale R (Localization.AtPrime M)),
41     Localization.AtPrime M →+* MvPowerSeries (Fin (d + 1)) k := by
42   letI : Algebra k B := coeff.toAlgebra
43   dsimp only
44   intro eM hEtM
45   let R := MvPolynomial (Option (Fin d)) k
46   letI : Algebra R B := fFin.toRingHom.toAlgebra
47   letI : IsScalarTower k R B := IsScalarTower.of_algebraMap_eq' (by
48     ext c; exact (fFin.commutes c).symm)
49   letI : Algebra.FiniteType k B := hfinite
50   letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
51   letI : Algebra.FormallyEtale R (Localization.AtPrime M) := hEtM
52   exact (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM).toRingHom

54 /-- The existing point-local column theorem yields the common-open position.
55 The ring variables stay abstract while its required instances are installed. -/
56 theorem commonOpen_position_of_pointColumns
57     {k B Q : Type u} [Field k] [CharZero k] [IsAlgClosed k]
58     [CommRing B] [CommRing Q]
59     {d : ℕ} (coeff : k →+* B) (qToB : Q →+* B)
60     (fFin : @AlgHom k (MvPolynomial (Option (Fin d)) k) B _ _ _ _ coeff.toAlgebra)
61     (hfinite : @Algebra.FiniteType k B _ _ coeff.toAlgebra)
62     (M : Ideal B) [M.IsPrime] [M.IsMaximal]
63     : letI : Algebra k B := coeff.toAlgebra
64     letI : Algebra Q B := qToB.toAlgebra
65     let R := MvPolynomial (Option (Fin d)) k
66     letI : Algebra R B := fFin.toRingHom.toAlgebra
67     letI : IsScalarTower k R B := IsScalarTower.of_algebraMap_eq' (by
68       ext c; exact (fFin.commutes c).symm)
69     letI : Algebra.FiniteType k B := hfinite
70     letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
71       ∀ (f : Q) (e : Localization.Away f ≃ₐ[Q]
72         Localization.Away (algebraMap Q B f)) (g : Q)
73       (eM : (B ⧸ M) ≃ₐ[k] k)
74       (hEtM : Algebra.FormallyEtale R (Localization.AtPrime M)),
75       letI : Algebra.FormallyEtale R (Localization.AtPrime M) := hEtM
76       ∀ (alpha : Fin d → k)
77       (hf : IsUnit ((originalToPointLocalFinSuccArcLaurentSeries
78       (k := k) (d := d) (A := B) M eM alpha) (algebraMap Q B f)))
79       (hunitM : ∀ b, b ∉ M → IsUnit ((originalToPointLocalFinSuccArcLaurentSeries
80         (k := k) (d := d) (A := B) M eM alpha) b))
81       (hg : IsUnit ((originalToPointLocalFinSuccArcLaurentSeries
82       (k := k) (d := d) (A := B) M eM alpha) (algebraMap Q B g)))
83       (n : ℕ) (qT : Fin (n + 1) → Localization.AtPrime M)
84       (qU : Fin (n + 1) → genericOpenExtraAwayB M f e g)
85       (hq : ∀ i, qU i = pointLocalToCommonOpen (A := B) M f e g (qT i))
86       (rho : genericOpenExtraAwayB M f e g →+* LaurentSeries k)
87       (hcanonical : rho = genericArcToGenericOpenExtraAwayB M f e
88         (originalToPointLocalFinSuccArcLaurentSeries
89           (k := k) (d := d) (A := B) M eM alpha) hf hunitM g hg)
90       (qPre : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)
91       (hqPre : qPre = fun i =>
92         pointLocalPowerSeriesChart coeff fFin hfinite M eM hEtM (qT i))
93       (qL : Fin (n + 1) → LaurentSeries k)
94       (hqL : qL = fun i => algebraMap (PowerSeries k) (LaurentSeries k)
95         (tiltedArc alpha (qPre i))),
96       ∀ i, rho (qU i) = qL i := by
97   letI : Algebra k B := coeff.toAlgebra
98   letI : Algebra Q B := qToB.toAlgebra
99   dsimp only
100   intro f e g eM hEtM alpha hf hunitM hg n qT qU hq
101     rho hcanonical qPre hqPre qL hqL
102   let R := MvPolynomial (Option (Fin d)) k
103   letI : Algebra R B := fFin.toRingHom.toAlgebra
104   letI : IsScalarTower k R B := IsScalarTower.of_algebraMap_eq' (by
105     ext c; exact (fFin.commutes c).symm)
106   letI : Algebra.FiniteType k B := hfinite
107   letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
108   letI : Algebra R (Localization.AtPrime M) := inferInstance
109   letI : Algebra.FormallyEtale R (Localization.AtPrime M) := hEtM
110   have hchart (x : Localization.AtPrime M) :
111       pointLocalPowerSeriesChart coeff fFin hfinite M eM hEtM x =
112         localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM x := by
113     unfold pointLocalPowerSeriesChart
114     dsimp only [id_eq]
115     exact congrFun (AlgHom.coe_toRingHom
116       (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM)) x
117   intro i
118   rw [hcanonical, hqL, hqPre]
119   exact ((actual_option_columns (k := k) (d := d) (A := B) M f e g eM alpha
120     hf hunitM hg n qT qU hq).1 i).trans
121       (congrArg (fun z => algebraMap (PowerSeries k) (LaurentSeries k)
122         (tiltedArc alpha z)) (hchart (qT i)).symm)

125 /-- Equality of source arcs transports the localization lift together with
126 its dependent unit certificates, without installing coefficient actions. -/
127 theorem genericOpenArc_eq_of_source_eq
128     {Q B L : Type u} [CommRing Q] [CommRing B] [Algebra Q B] [Field L]
129     (M : Ideal B) [M.IsPrime] (f : Q)
130     (e : Localization.Away f ≃ₐ[Q] Localization.Away (algebraMap Q B f))
131     (g : Q) (rho canonical : B →+* L)
132     (rhoU : genericOpenExtraAwayB M f e g →+* L)
133     (hf : IsUnit (rho (algebraMap Q B f)))
134     (hunitM : ∀ b, b ∉ M → IsUnit (rho b))
135     (hg : IsUnit (rho (algebraMap Q B g)))
136     (hsource : rho = canonical)
137     (hU : rhoU = genericArcToGenericOpenExtraAwayB M f e rho hf hunitM g hg) :
138     rhoU = genericArcToGenericOpenExtraAwayB M f e canonical
139       (hsource ▸ hf) (hsource ▸ hunitM) g (hsource ▸ hg) := by
140   cases hsource
141   exact hU

169 /-- Build the positions from the retained point-local columns and their
170 common-open images, preserving the same witness throughout. -/
171 noncomputable def commonOpenPositionData_of_arc
172     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
173     {m : ℕ} (hm : 0 < m)
174     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
175     (w : GeneralDivisorialVisibleFrameWitness hm P)
176     (setup : ChartSetup hm P w)
177     (coords : CoordinatePresentation hm P w setup)
178     (arc : CommonOpenArcData hm P w setup coords) :
179     CommonOpenPositionData hm P w setup coords arc := by
180   classical
181   letI : arc.common.M.IsMaximal := arc.common.hM
182   let d := @Fintype.card coords.t coords.htFinite
183   let B := actualSelectedNormalization P w
184   let Q := actualSelectedChartAlgebra P w
185   let qB : Fin (m + 1) → B :=
186     actualNormalizedProjectiveColumnInIntegralClosure hm P w
187   let qPre : Fin (m + 1) → MvPowerSeries (Fin (d + 1)) k := fun i =>
188     pointLocalPowerSeriesChart coords.coeff coords.fFin coords.hBfinite
189       arc.common.M arc.common.eM arc.common.hEtM (arc.common.qT i)
190   have hchartPre : qPre (Fin.succ setup.j) = 1 := by
191     have hqT : arc.common.qT (Fin.succ setup.j) = 1 := by
192       change algebraMap B (Localization.AtPrime arc.common.M)
193         (actualNormalizedProjectiveColumnInIntegralClosure hm P w
194           (Fin.succ setup.j)) = 1
195       rw [← setup.hchart, coords.hqchartB]
196       exact map_one _
197     change (pointLocalPowerSeriesChart coords.coeff coords.fFin coords.hBfinite
198       arc.common.M arc.common.eM arc.common.hEtM)
199       (arc.common.qT (Fin.succ setup.j)) = 1
200     rw [hqT, map_one]
201   have hf := arc.hf
202   rw [arc.hcanonicalRhoA] at hf
203   have hunitM := arc.hunitM
204   rw [arc.hcanonicalRhoA] at hunitM
205   have hg := arc.hg
206   rw [arc.hcanonicalRhoA] at hg
207   have hcanonicalRhoU := genericOpenArc_eq_of_source_eq
208     arc.common.M setup.f setup.e arc.common.g arc.rhoA _ arc.rhoU
209     arc.hf arc.hunitM arc.hg arc.hcanonicalRhoA arc.hcanonicalRhoU
210   let qL : Fin (m + 1) → LaurentSeries k :=
211     Stafford38.Geometry.FormalDivisorLaurentConormal.laurentColumn
212       (fun i => tiltedArc arc.common.alpha (qPre i))
213   have hcols := commonOpen_position_of_pointColumns
214     (k := k) (B := B) (Q := Q) (d := d)
215     (coeff := coords.coeff)
216     (qToB := chartSubalgebraToIntegralClosure Q)
217     (fFin := coords.fFin) coords.hBfinite arc.common.M setup.f setup.e arc.common.g arc.common.eM arc.common.hEtM arc.common.alpha
218     hf hunitM hg (n := m) arc.common.qT arc.common.qU arc.common.hqUPoint
219     arc.rhoU hcanonicalRhoU qPre rfl qL rfl
220   have hcoords : ∀ i, arc.common.qU i = genericOpenExtraAwayBMap
221       arc.common.M setup.f setup.e arc.common.g (qB i) := by
222     intro i
223     calc
224       arc.common.qU i = pointLocalToCommonOpen (A := B) arc.common.M setup.f setup.e
225           arc.common.g (arc.common.qT i) := arc.common.hqUPoint i
226       _ = pointLocalToCommonOpen (A := B) arc.common.M setup.f setup.e
227           arc.common.g (algebraMap B (Localization.AtPrime arc.common.M) (qB i)) := rfl
228       _ = genericOpenExtraAwayBMap arc.common.M setup.f setup.e arc.common.g (qB i) := by
229         have h := congrArg (fun φ : B →+* arc.common.U => φ (qB i))
230           (pointLocalToCommonOpen_comp_algebraMap (A := B)
231             arc.common.M setup.f setup.e arc.common.g)
232         simpa only [RingHom.comp_apply] using h
233   exact ⟨qPre, qL, rfl, hchartPre, hcoords, hcols⟩
