30 /-- The finite coordinate presentation and the ground-point chart output
31 retained from one same-witness chart setup. -/
32 structure CoordinatePresentation
33     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
34     {m : ℕ} (hm : 0 < m)
35     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
36     (w : GeneralDivisorialVisibleFrameWitness hm P)
37     (setup : ChartSetup hm P w) where
38   t : Type u
39   htFinite : Fintype t
40   coeff : k →+* actualSelectedNormalization P w
41   hcoeff : coeff = actualSelectedNormalizationCoefficients P w
42   hBfinite : @Algebra.FiniteType k (actualSelectedNormalization P w)
43     _ _ coeff.toAlgebra
44   hPB : (actualSelectedNormalizationCenterPrime P w
45     (chartGenericPointSubalgebra_le_valuationSubring hm P w)).IsPrime
46   τ : Fin (@Fintype.card t htFinite) ≃ t
47   rows : Fin (@Fintype.card t htFinite) ↪ Fin (m + 1)
48   qRow : t → actualSelectedNormalization P w
49   fOption : @AlgHom k (MvPolynomial (Option t) k)
50     (actualSelectedNormalization P w) _ _ _ _ coeff.toAlgebra
51   s : actualSelectedNormalization P w
52   hsPB : s ∈ actualSelectedNormalizationCenterPrime P w
53     (chartGenericPointSubalgebra_le_valuationSubring hm P w)
54   hnone : fOption (MvPolynomial.X (R := k) none) = s
55   hsome : ∀ z, fOption (MvPolynomial.X (R := k) (some z)) = qRow z
56   hrows : ∀ i, qRow (τ i) =
57     actualNormalizedProjectiveColumnInIntegralClosure hm P w (rows i)
58   hqchartB : actualNormalizedProjectiveColumnInIntegralClosure hm P w
59     w.column.chart = 1
60   hzero : actualNormalizedProjectiveColumnInIntegralClosure hm P w 0 =
61     actualNormalizedProjectiveColumnInIntegralClosure hm P w 0
62   haxis : actualNormalizedProjectiveColumnInIntegralClosure hm P w
63       (Fin.succ (⟨0, hm⟩ : Fin m)) =
64     actualNormalizedProjectiveColumnInIntegralClosure hm P w
65       (Fin.succ (⟨0, hm⟩ : Fin m))
66   hOutput :
67     letI : Algebra (Stafford38.Geometry.AsymptoticDivisorExistence.CoordinateZeroLocalRing
68       w.column.W.coefficientField)
69       (Stafford38.Geometry.ComponentProjectiveClosure.ComponentFractionField P) :=
70         w.column.W.ambientAlgebra
71     letI : IsLocalRing w.column.W.place.valuation.toSubring :=
72       w.column.W.place.isDiscrete.toIsLocalRing
73     @GroundPointChartOutput k (actualSelectedNormalization P w) t
74       inferInstance inferInstance inferInstance coeff.toAlgebra inferInstance
75       hBfinite htFinite
76       (actualSelectedNormalizationCenterPrime P w
77         (chartGenericPointSubalgebra_le_valuationSubring hm P w)) hPB
78       fOption s
79       (actualNormalizedProjectiveColumnInIntegralClosure hm P w 0)
80       (actualNormalizedProjectiveColumnInIntegralClosure hm P w
81         (Fin.succ (⟨0, hm⟩ : Fin m)))
82       qRow τ w.differential.core.D.a
83       (w.differential.core.D.a + w.differential.core.D.e)

85 /-- Reindex the retained coordinate presentation along its chosen finite
86 index equivalence (paper proof, étale-coordinate step). -/
87 noncomputable abbrev CoordinatePresentation.fFin
88     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
89     {m : ℕ} {hm : 0 < m}
90     {P : PrimeSpectrum (MvPolynomial (Fin m) k)}
91     {w : GeneralDivisorialVisibleFrameWitness hm P}
92     {setup : ChartSetup hm P w}
93     (coords : CoordinatePresentation hm P w setup) :
94     @AlgHom k (MvPolynomial (Option (Fin (@Fintype.card coords.t coords.htFinite))) k)
95       (actualSelectedNormalization P w) _ _ _ _ coords.coeff.toAlgebra :=
96   @AlgHom.comp k
97     (MvPolynomial (Option (Fin (@Fintype.card coords.t coords.htFinite))) k)
98     (MvPolynomial (Option coords.t) k) (actualSelectedNormalization P w)
99     _ _ _ _ _ _ coords.coeff.toAlgebra coords.fOption
100     (MvPolynomial.renameEquiv k coords.τ.optionCongr).toAlgHom

102 /-- Distinct retained rows remain distinct in the selected affine chart. -/
103 theorem chart_rows_injective
104     {m d : ℕ} (j : Fin (m + 1)) {t : Type u}
105     (index : t → Fin m) (hindex : Function.Injective index)
106     (τ : Fin d ≃ t) :
107     Function.Injective (fun i => (chartAffineCoordinateEquiv j (index (τ i))).1) := by
108   intro i i' hii
109   have hcoord : chartAffineCoordinateEquiv j (index (τ i)) =
110       chartAffineCoordinateEquiv j (index (τ i')) := Subtype.ext hii
111   exact τ.injective (hindex ((chartAffineCoordinateEquiv j).injective hcoord))

113 /-- Retained normalization rows agree with the selected chart's coordinates. -/
114 theorem selected_rows_chart_eq
115     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
116     {m : ℕ} (hm : 0 < m)
117     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
118     (w : GeneralDivisorialVisibleFrameWitness hm P)
119     (setup : ChartSetup hm P w) {t : Type u} (index : t → Fin m) (z : t) :
120     actualSelectedNormalizationRows P w index z =
121       actualNormalizedProjectiveColumnInIntegralClosure hm P w
122         (chartAffineCoordinateEquiv (Fin.succ setup.j) (index z)).1 := by
123   change actualNormalizedProjectiveColumnInIntegralClosure hm P w
124       (chartAffineCoordinateEquiv w.column.chart (index z)).1 = _
125   congr 1
126   exact congrArg (fun c => (chartAffineCoordinateEquiv c (index z)).1) setup.hchart

128 /-- A same-witness chart setup and its retained ground-point output determine
129 the corresponding finite coordinate presentation. -/
130 theorem nonempty_coordinatePresentation
131     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
132     {m : ℕ} (hm : 0 < m)
133     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
134     (w : GeneralDivisorialVisibleFrameWitness hm P)
135     (setup : ChartSetup hm P w)
136     (hpoint : actualSameWitnessGroundPointOutput hm P w) :
137     Nonempty (CoordinatePresentation hm P w setup) := by
138   classical
139   let B := actualSelectedNormalization P w
140   letI : IsLocalRing (
141     letI : Algebra
142       (Stafford38.Geometry.AsymptoticDivisorExistence.CoordinateZeroLocalRing
143         w.column.W.coefficientField)
144       (Stafford38.Geometry.ComponentProjectiveClosure.ComponentFractionField P) :=
145         w.column.W.ambientAlgebra;
146     w.column.W.place.valuation.toSubring) :=
147     letI : Algebra
148       (Stafford38.Geometry.AsymptoticDivisorExistence.CoordinateZeroLocalRing
149         w.column.W.coefficientField)
150       (Stafford38.Geometry.ComponentProjectiveClosure.ComponentFractionField P) :=
151         w.column.W.ambientAlgebra
152     w.column.W.place.isDiscrete.toIsLocalRing
153   unfold actualSameWitnessGroundPointOutput at hpoint
154   dsimp only at hpoint
155   rcases hpoint with ⟨hBfinite, t, htFinite, htBasis, index, hindex,
156     hresidue, hPB', s, hs0, hsPB, u₀, u₁, hu₀, hu₁, hspan,
157     hfactor₀, hfactor₁, hGroundPoint⟩
158   letI : Fintype t := htFinite.fintype
159   let τ : Fin (Fintype.card t) ≃ t := (Fintype.equivFin t).symm
160   let rows : Fin (Fintype.card t) ↪ Fin (m + 1) :=
161     ⟨fun i => (chartAffineCoordinateEquiv (Fin.succ setup.j)
162       (index (τ i))).1, chart_rows_injective (Fin.succ setup.j) index hindex τ⟩
163   let qRow : t → B := actualSelectedNormalizationRows P w index
164   let coeff := actualSelectedNormalizationCoefficients P w
165   let fOption : @AlgHom k (MvPolynomial (Option t) k) B _ _ _ _ coeff.toAlgebra :=
166     @actualOptionMap k _ t B _ coeff.toAlgebra s qRow
167   have hrows : ∀ i : Fin (Fintype.card t), qRow (τ i) =
168       actualNormalizedProjectiveColumnInIntegralClosure hm P w (rows i) :=
169     fun i => selected_rows_chart_eq hm P w setup index (τ i)
170   rcases hGroundPoint with ⟨cert, hGroundPoint⟩
171   have hOutput : @GroundPointChartOutput k B t inferInstance inferInstance
172       inferInstance coeff.toAlgebra inferInstance hBfinite htFinite.fintype
173       (actualSelectedNormalizationCenterPrime P w
174         (chartGenericPointSubalgebra_le_valuationSubring hm P w)) hPB'
175       fOption s
176       (actualNormalizedProjectiveColumnInIntegralClosure hm P w 0)
177       (actualNormalizedProjectiveColumnInIntegralClosure hm P w
178         (Fin.succ (⟨0, hm⟩ : Fin m)))
179       qRow τ w.differential.core.D.a
180       (w.differential.core.D.a + w.differential.core.D.e) := hGroundPoint
181   exact ⟨{
182     t := t, htFinite := htFinite.fintype
183     coeff := coeff, hcoeff := rfl, hBfinite := hBfinite
184     hPB := hPB', τ := τ, rows := rows, qRow := qRow
185     fOption := fOption, s := s, hsPB := hsPB
186     hnone := @actualOptionMap_none k _ t B _ coeff.toAlgebra s qRow
187     hsome := @actualOptionMap_some k _ t B _ coeff.toAlgebra s qRow
188     hrows := hrows
189     hqchartB := (actualNormalizedProjectiveColumnInIntegralClosure_spec hm P w).1
190     hzero := rfl, haxis := rfl, hOutput := hOutput }⟩
191 
192 end Stafford38.Geometry.SameWitness
