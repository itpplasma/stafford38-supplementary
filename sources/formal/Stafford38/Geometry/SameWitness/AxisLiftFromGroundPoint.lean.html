23 /-- Extract the chosen maximal point and the same tilted arc from the
24 closed-ground-point output. The two divisor factorizations are transported
25 from its single away open to the local ring at that same point. -/
26 theorem exists_axis_lift_of_groundPointChartOutput
27     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
28     {B : Type u} [CommRing B] [Algebra k B] [IsDomain B]
29     [Algebra.FiniteType k B] {σ : Type v} [Fintype σ]
30     (P : Ideal B) [P.IsPrime]
31     (fOption : MvPolynomial (Option σ) k →ₐ[k] B)
32     (s q₀ q₁ : B) (qRow : σ → B)
33     (hnone : fOption (MvPolynomial.X none) = s)
34     (hsP : s ∈ P)
35     (hsome : ∀ z, fOption (MvPolynomial.X (some z)) = qRow z)
36     (τ : Fin (Fintype.card σ) ≃ σ) (e₀ e₁ : ℕ)
37     (hOutput : GroundPointChartOutput P fOption s q₀ q₁ qRow τ e₀ e₁)
38     {n : ℕ} (qB : Fin (n + 1) → B) (chart : Fin (n + 1)) (axis : Fin n)
39     (hqchart : qB chart = 1)
40     (hzero : qB 0 = q₀) (haxis : qB axis.succ = q₁)
41     (rows : Fin (Fintype.card σ) ↪ Fin (n + 1))
42     (hrows : ∀ i, qRow (τ i) = qB (rows i))
43     (bad : B) (hbad : bad ≠ 0) :
44     ∃ (M : Ideal B) (hM : M.IsMaximal) (eM : (B ⧸ M) ≃ₐ[k] k)
45       (fUnit : B) (hfUnitM : fUnit ∉ M),
46       let R := MvPolynomial (Option (Fin (Fintype.card σ))) k
47       let fFin : R →ₐ[k] B := fOption.comp (MvPolynomial.renameEquiv k τ.optionCongr).toAlgHom
48       letI : Algebra R B := fFin.toRingHom.toAlgebra
49       letI : IsScalarTower k R B := IsScalarTower.of_algebraMap_eq' (by
50         ext c
51         exact (fFin.commutes c).symm)
52       letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
53       letI : Algebra R (Localization.AtPrime M) := inferInstance
54       ∃ (u₀ u₁ : Localization.AtPrime M) (hu₀ : IsUnit u₀) (hu₁ : IsUnit u₁)
55         (hq₀ : algebraMap B (Localization.AtPrime M) (qB 0) =
56           (algebraMap B (Localization.AtPrime M) s) ^ e₀ * u₀)
57         (hq₁ : algebraMap B (Localization.AtPrime M) (qB axis.succ) =
58           (algebraMap B (Localization.AtPrime M) s) ^ e₁ * u₁)
59         (hEtM : Algebra.FormallyEtale R (Localization.AtPrime M))
60         (α : Fin (Fintype.card σ) → k),
61         tiltedArc (k := k) α
62           ((localToFinSuccPowerSeries (k := k) (B := B)
63               (d := Fintype.card σ) M eM (algebraMap B (Localization.AtPrime M) bad)) *
64             localToFinSuccPowerSeries (k := k) (B := B)
65               (d := Fintype.card σ) M eM
66               (algebraMap B (Localization.AtPrime M) (qB 0)) *
67             localToFinSuccPowerSeries (k := k) (B := B)
68               (d := Fintype.card σ) M eM
69               (algebraMap B (Localization.AtPrime M) (qB axis.succ))) ≠ 0 ∧
70         SelectedCoordinateAxisLiftData
71           (fun i => localToFinSuccPowerSeries (k := k) (B := B)
72             (d := Fintype.card σ) M eM (algebraMap B (Localization.AtPrime M) (qB i)))
73           rows chart 0 axis.succ e₀ e₁
74           (fun j => Stafford38.Geometry.PrescribedAffineResidueCompletion.residueCoordinates
75             M eM (some j)) α
76           (localToFinSuccPowerSeries (k := k) (B := B)
77             (d := Fintype.card σ) M eM
78             u₀)
79           (localToFinSuccPowerSeries (k := k) (B := B)
80             (d := Fintype.card σ) M eM
81             u₁) := by
82   classical
83   unfold GroundPointChartOutput at hOutput
84   rcases hOutput with ⟨fUnit, fEtale, M, hM, eM, v₀, v₁, chartMap, U₀, U₁,
85     hfUnitP, hfUnitM, hPM, hfEtaleP, hfEtaleM, hStandard,
86     hv₀, hv₁, hAway₀, hAway₁, hOrders, hU₀, hU₁, hcc₀, hcc₁,
87     hsChart, hrowChart, hq₀Chart, hq₁Chart, hEtM, hchartEq⟩
88   let R := MvPolynomial (Option (Fin (Fintype.card σ))) k
89   let fFin : R →ₐ[k] B := fOption.comp (MvPolynomial.renameEquiv k τ.optionCongr).toAlgHom
90   letI : Algebra R B := fFin.toRingHom.toAlgebra
91   letI : IsScalarTower k R B := IsScalarTower.of_algebraMap_eq' (by
92     ext c
93     exact (fFin.commutes c).symm)
94   letI : M.IsMaximal := hM
95   letI : Algebra R (Localization.AtPrime M) := inferInstance
96   have hEt' : Algebra.FormallyEtale R (Localization.AtPrime M) := hEtM
97   letI : Algebra.FormallyEtale R (Localization.AtPrime M) := hEt'
98   letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
99   have hmap : algebraMap R B = fFin.toRingHom :=
100     RingHom.algebraMap_toAlgebra fFin.toRingHom
101   have hnoneFin : fFin (MvPolynomial.X (R := k) none) = s := by
102     simp [fFin, MvPolynomial.renameEquiv, MvPolynomial.rename_X, hnone]
103   have hrowsFin (i : Fin (Fintype.card σ)) :
104       fFin (MvPolynomial.X (R := k) (some i)) = qB (rows i) := by
105     calc
106       fFin (MvPolynomial.X (R := k) (some i)) = qRow (τ i) := by
107         simp [fFin, MvPolynomial.renameEquiv, MvPolynomial.rename_X, hsome]
108       _ = qB (rows i) := hrows i
109   have hq₀ : algebraMap B (Localization.Away fUnit) (qB 0) =
110       (algebraMap B (Localization.Away fUnit) s) ^ e₀ * v₀ := by
111     simpa [hzero] using hAway₀
112   have hq₁ : algebraMap B (Localization.Away fUnit) (qB axis.succ) =
113       (algebraMap B (Localization.Away fUnit) s) ^ e₁ * v₁ := by
114     simpa [haxis] using hAway₁
115   obtain ⟨u₀, u₁, hu₀, hu₁, hq₀M, hq₁M⟩ :=
116     SameWitness.pair_factorizations_to_atPrime M fUnit hfUnitM s (qB 0)
117       (qB axis.succ) e₀ e₁ v₀ v₁ hv₀ hv₁ hq₀ hq₁
118   have hsM : s ∈ M := hPM hsP
119   obtain ⟨α, havoid, hdata⟩ :=
120     exists_actual_point_axis_lift M eM fFin hmap s bad hnoneFin hsM hbad qB chart axis
121       hqchart e₀ e₁ hOrders.1 hOrders.2 u₀ u₁ hu₀ hu₁ hq₀M hq₁M rows hrowsFin
122   exact ⟨M, hM, eM, fUnit, hfUnitM, u₀, u₁, hu₀, hu₁, hq₀M, hq₁M,
123     hEt', α, havoid, hdata⟩
