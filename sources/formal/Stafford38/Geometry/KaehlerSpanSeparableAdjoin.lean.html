7 /-!
8 # Kähler span after a separable residue-field extension
9 
10 Let `k → E → K` be a tower of fields.  If a family `q : ι → E` generates
11 `E` as an intermediate field over `k`, then its universal differentials span
12 `Ω[E⁄k]`.  If `K/E` is formally étale, the images of the same differentials
13 span `Ω[K⁄k]`; separability supplies the formally-étale instance for fields.
14 
15 For a finite generating family, this also supplies the finite-dimensional
16 instance needed by `KaehlerVisibleDerivationFrame` and hence a dual visible
17 derivation frame over `K`.
18 
19 This file does not construct a projective boundary divisor, prove that the
20 residue coordinates at a chosen divisor generate an intermediate field, or
21 produce any completed boundary chart.  Those are separate geometric inputs.
22 -/

36 /-- A family that generates a field over the ground field has universal
37 differentials spanning the full Kähler module. -/
38 theorem kaehler_span_of_intermediateField_adjoin_eq_top
39     {ι : Type v} (q : ι → E)
40     (hgen : IntermediateField.adjoin k (Set.range q) = ⊤) :
41     Submodule.span E
42       (Set.range fun i ↦ KaehlerDifferential.D k E (q i)) = ⊤ := by
43   let W : Submodule E (Ω[E⁄k]) :=
44     Submodule.span E (Set.range fun i ↦ KaehlerDifferential.D k E (q i))
45   have hD (x : E) : KaehlerDifferential.D k E x ∈ W := by
46     have hx : x ∈ IntermediateField.adjoin k (Set.range q) := by
47       rw [hgen]
48       trivial
49     apply IntermediateField.adjoin_induction k
50         (p := fun y _ ↦ KaehlerDifferential.D k E y ∈ W)
51     · intro y hy
52       change y ∈ Set.range q at hy
53       obtain ⟨i, rfl⟩ := hy
54       exact Submodule.subset_span ⟨i, rfl⟩
55     · intro a
56       simp
57     · intro a b _ _ ha hb
58       simpa only [map_add] using W.add_mem ha hb
59     · intro a _ ha
60       rw [Derivation.leibniz_inv]
61       exact W.smul_mem _ ha
62     · intro a b _ _ ha hb
63       rw [Derivation.leibniz]
64       exact W.add_mem (W.smul_mem _ hb) (W.smul_mem _ ha)
65     · exact hx
66   apply top_unique
67   rw [← KaehlerDifferential.span_range_derivation, Submodule.span_le]
68   rintro _ ⟨x, rfl⟩
69   exact hD x

71 /-- Formally-étale base change transports a field-generating differential
72 family to a spanning family over the extension field. -/
73 theorem kaehler_span_of_formallyEtale_adjoin_eq_top
74     [Algebra.FormallyEtale E K]
75     {ι : Type v} (q : ι → E)
76     (hgen : IntermediateField.adjoin k (Set.range q) = ⊤) :
77     Submodule.span K
78       (Set.range fun i ↦
79         KaehlerDifferential.D k K (algebraMap E K (q i))) = ⊤ := by
80   let W : Submodule K (Ω[K⁄k]) :=
81     Submodule.span K (Set.range fun i ↦
82       KaehlerDifferential.D k K (algebraMap E K (q i)))
83   have hE : Submodule.span E
84       (Set.range fun i ↦ KaehlerDifferential.D k E (q i)) = ⊤ :=
85     kaehler_span_of_intermediateField_adjoin_eq_top q hgen
86   have hmap (w : Ω[E⁄k]) :
87       KaehlerDifferential.map k k E K w ∈ W := by
88     have hw : w ∈ Submodule.span E
89         (Set.range fun i ↦ KaehlerDifferential.D k E (q i)) := by
90       rw [hE]
91       trivial
92     induction hw using Submodule.span_induction with
93     | mem x hx =>
94         obtain ⟨i, rfl⟩ := hx
95         rw [KaehlerDifferential.map_D]
96         exact Submodule.subset_span ⟨i, rfl⟩
97     | zero => exact W.zero_mem
98     | add x y hx hy hx' hy' =>
99         simpa only [map_add] using W.add_mem hx' hy'
100     | smul a x hx hx' =>
101         simpa only [LinearMap.map_smul_of_tower,
102           IsScalarTower.algebraMap_smul] using
103           W.smul_mem (algebraMap E K a) hx'
104   let e := KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale k E K
105   have he (y : K ⊗[E] Ω[E⁄k]) : e y ∈ W := by
106     induction y using TensorProduct.induction_on with
107     | zero =>
108         rw [show e (0 : K ⊗[E] Ω[E⁄k]) = 0 from e.map_zero]
109         exact W.zero_mem
110     | add x y hx hy =>
111         simpa only [map_add] using W.add_mem hx hy
112     | tmul a w =>
113         rw [KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale_apply,
114           KaehlerDifferential.mapBaseChange_tmul]
115         exact W.smul_mem _ (hmap w)
116   apply top_unique
117   intro z _
118   obtain ⟨y, rfl⟩ := e.surjective z
119   exact he y

121 /-- The formally-étale span theorem specialized to a separable extension of
122 fields.  No finite-degree hypothesis is needed. -/
123 theorem kaehler_span_of_separable_adjoin_eq_top
124     [Algebra.IsSeparable E K]
125     {ι : Type v} (q : ι → E)
126     (hgen : IntermediateField.adjoin k (Set.range q) = ⊤) :
127     Submodule.span K
128       (Set.range fun i ↦
129         KaehlerDifferential.D k K (algebraMap E K (q i))) = ⊤ := by
130   letI : Algebra.FormallyEtale E K :=
131     Algebra.FormallyEtale.of_isSeparable E K
132   exact kaehler_span_of_formallyEtale_adjoin_eq_top q hgen

134 /-- A finite field-generating family gives a visible dual derivation frame
135 after separable extension. -/
136 theorem exists_visible_derivation_frame_of_finite_separable_adjoin
137     [Algebra.IsSeparable E K]
138     {ι : Type v} [Fintype ι] (q : ι → E)
139     (hgen : IntermediateField.adjoin k (Set.range q) = ⊤) :
140     ∃ (rows : Fin (Module.finrank K (Ω[K⁄k])) ↪ ι)
141       (D : Fin (Module.finrank K (Ω[K⁄k])) → Derivation k K K),
142       ∀ i j, D j (algebraMap E K (q (rows i))) =
143         if i = j then 1 else 0 := by
144   have hspan := kaehler_span_of_separable_adjoin_eq_top
145     (k := k) (E := E) (K := K) q hgen
146   letI : FiniteDimensional K (Ω[K⁄k]) := by
147     letI : FiniteDimensional K
148         (Submodule.span K (Set.range fun i ↦
149           KaehlerDifferential.D k K (algebraMap E K (q i)))) :=
150       FiniteDimensional.span_of_finite K (Set.finite_range _)
151     letI : FiniteDimensional K (⊤ : Submodule K (Ω[K⁄k])) :=
152       (LinearEquiv.ofEq _ _ hspan).finiteDimensional
153     exact Submodule.topEquiv.finiteDimensional
154   exact Stafford38.Geometry.KaehlerVisibleDerivationFrame.exists_visible_derivation_frame_of_kaehler_span
155     (fun i ↦ algebraMap E K (q i)) hspan
