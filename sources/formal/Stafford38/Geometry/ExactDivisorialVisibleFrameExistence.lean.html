45 /-- The affine coordinates generate the function field of a prime component.
46 This is the exact generation hypothesis consumed by the lane-C stage
47 assembly. -/
48 theorem componentCoordinate_adjoin_eq_top
49     {k : Type u} [Field k] {m : ℕ}
50     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
51     IntermediateField.adjoin k (Set.range (componentCoordinate P)) = ⊤ := by
52   let R := MvPolynomial (Fin m) k ⧸ P.asIdeal
53   let K := FractionRing R
54   let L : IntermediateField k K :=
55     IntermediateField.adjoin k (Set.range (componentCoordinate P))
56   have hpoly (p : MvPolynomial (Fin m) k) :
57       algebraMap R K (Ideal.Quotient.mk P.asIdeal p) ∈ L := by
58     induction p using MvPolynomial.induction_on with
59     | C c =>
60         rw [← MvPolynomial.algebraMap_eq,
61           Ideal.Quotient.mk_algebraMap,
62           ← IsScalarTower.algebraMap_apply k R K]
63         exact L.algebraMap_mem c
64     | add p q hp hq =>
65         rw [map_add, map_add]
66         exact L.add_mem hp hq
67     | mul_X p i hp =>
68         rw [map_mul, map_mul]
69         apply L.mul_mem hp
70         exact IntermediateField.subset_adjoin k _ ⟨i, rfl⟩
71   have hquot (z : R) : algebraMap R K z ∈ L := by
72     refine Quotient.inductionOn z ?_
73     intro p
74     exact hpoly p
75   apply top_unique
76   intro z _
77   obtain ⟨a, b, hb, hab⟩ := IsFractionRing.div_surjective R z
78   rw [← hab]
79   exact L.div_mem (hquot a) (hquot b)

97 /-- The coordinate-zero local ring maps to a valuation ring that contains
98 the coefficient field and the selected element in its maximal ideal. -/
99 def coordinateZeroLocalFactor
100     {k K : Type u} [Field k] [Field K] [Algebra k K]
101     (E : IntermediateField k K)
102     (V : ValuationSubring K) [IsLocalRing V.toSubring]
103     (hEV : ∀ z : E, (z : K) ∈ V.toSubring)
104     (x : K) (hxV : x ∈ V.toSubring)
105     (hxm : (⟨x, hxV⟩ : V.toSubring) ∈ maximalIdeal V.toSubring) :
106     CoordinateZeroLocalRing E →+* V.toSubring := by
107   let c : E →+* V.toSubring :=
108     Stafford38.Geometry.LaneC.coeffHom E V hEV
109   let xV : V.toSubring := ⟨x, hxV⟩
110   let g : Polynomial E →+* V.toSubring := Polynomial.eval₂RingHom c xV
111   exact IsLocalization.lift (S := CoordinateZeroLocalRing E)
112     (M := (coordinateZeroPrime E).primeCompl) (g := g) fun p ↦ by
113       have hp0 : (p : Polynomial E).coeff 0 ≠ 0 := by
114         intro hp0
115         apply p.property
116         change (p : Polynomial E) ∈ Ideal.span {Polynomial.X}
117         rw [Ideal.mem_span_singleton, Polynomial.X_dvd_iff]
118         exact hp0
119       have hcunit : IsUnit (c ((p : Polynomial E).coeff 0)) :=
120         (isUnit_iff_ne_zero.mpr hp0).map c
121       apply (residue_ne_zero_iff_isUnit (g (p : Polynomial E))).mp
122       have hresx : residue V.toSubring xV = 0 :=
123         (residue_eq_zero_iff xV).mpr hxm
124       have hresconst : residue V.toSubring
125           (c ((p : Polynomial E).coeff 0)) ≠ 0 :=
126         (residue_ne_zero_iff_isUnit
127           (c ((p : Polynomial E).coeff 0))).mpr hcunit
128       have hdecomp := Polynomial.X_mul_divX_add (p : Polynomial E)
129       have heval : residue V.toSubring (g (p : Polynomial E)) =
130           residue V.toSubring (c ((p : Polynomial E).coeff 0)) := by
131         calc
132           residue V.toSubring (g (p : Polynomial E)) =
133               residue V.toSubring
134                 (g (Polynomial.X * Polynomial.divX (p : Polynomial E) +
135                   Polynomial.C ((p : Polynomial E).coeff 0))) :=
136             congrArg (fun z ↦ residue V.toSubring (g z)) hdecomp.symm
137           _ = residue V.toSubring (c ((p : Polynomial E).coeff 0)) := by
138             rw [map_add, map_mul]
139             change residue V.toSubring (Polynomial.eval₂ c xV Polynomial.X) *
140                 residue V.toSubring
141                   (Polynomial.eval₂ c xV (Polynomial.divX (p : Polynomial E))) +
142                 residue V.toSubring
143                   (Polynomial.eval₂ c xV
144                     (Polynomial.C ((p : Polynomial E).coeff 0))) = _
145             rw [Polynomial.eval₂_X, Polynomial.eval₂_C, hresx,
146               zero_mul, zero_add]
147       rw [heval]
148       exact hresconst

163 theorem coordinateZeroLocalFactor_isLocal
164     {k K : Type u} [Field k] [Field K] [Algebra k K]
165     (E : IntermediateField k K)
166     (V : ValuationSubring K) [IsLocalRing V.toSubring]
167     (hEV : ∀ z : E, (z : K) ∈ V.toSubring)
168     (x : K) (hxV : x ∈ V.toSubring)
169     (hxm : (⟨x, hxV⟩ : V.toSubring) ∈ maximalIdeal V.toSubring) :
170     IsLocalHom (coordinateZeroLocalFactor E V hEV x hxV hxm) := by
171   let f := coordinateZeroLocalFactor E V hEV x hxV hxm
172   apply (IsLocalRing.local_hom_TFAE f).out 3 1 |>.mp
173   intro z hz
174   rw [coordinateZeroLocal_maximalIdeal_eq_span, Ideal.map_span, Set.image_singleton,

178     maximalIdeal V.toSubring
179   rw [coordinateZeroLocalFactor_map_X E]
180   exact Ideal.mul_mem_right _ _ hxm

237 /-- Package the particular DVR produced by the stage assembly as retained
238 boundary data, without replacing its parameter by a uniformizer. -/
239 def retainedDataOfValuation
240     {k K : Type u} [Field k] [Field K] [Algebra k K]
241     (E : IntermediateField k K) (V : ValuationSubring K)
242     (hEV : ∀ z : E, (z : K) ∈ V.toSubring)
243     (hVdvr : IsDiscreteValuationRing V.toSubring)
244     (x : K) (hxV : x ∈ V.toSubring)
245     (htrans : Transcendental E x)
246     (hxm : (⟨x, hxV⟩ : V.toSubring) ∈ maximalIdeal V.toSubring)
247     (hfinite :
248       letI : IsLocalRing V.toSubring := hVdvr.toIsLocalRing
249       letI : Algebra E V.toSubring :=
250         (Stafford38.Geometry.LaneC.coeffHom E V hEV).toAlgebra
251       Module.Finite E (ResidueField V.toSubring)) : Data k K x := by
252   letI : IsLocalRing V.toSubring := hVdvr.toIsLocalRing
253   let xV : V.toSubring := ⟨x, hxV⟩
254   let f : CoordinateZeroLocalRing E →+* V.toSubring :=
255     coordinateZeroLocalFactor E V hEV x hxV hxm
256   let hf : IsLocalHom f :=
257     coordinateZeroLocalFactor_isLocal E V hEV x hxV hxm
258   let ambient : Algebra (CoordinateZeroLocalRing E) K :=
259     (V.toSubring.subtype.comp f).toAlgebra
260   letI : Algebra (CoordinateZeroLocalRing E) K := ambient
261   have tower : IsScalarTower E (CoordinateZeroLocalRing E) K := by
262     apply IsScalarTower.of_algebraMap_eq
263     intro z
264     change (z : K) =
265       ((f (algebraMap E (CoordinateZeroLocalRing E) z) : V.toSubring) : K)
266     rw [IsScalarTower.algebraMap_apply E (Polynomial E)
267       (CoordinateZeroLocalRing E) z]
268     dsimp only [f]
269     rw [coordinateZeroLocalFactor, IsLocalization.lift_eq]
270     simp [Stafford38.Geometry.LaneC.coeffHom]
271   have coordinate_eq :
272       algebraMap (CoordinateZeroLocalRing E) K
273           (algebraMap (Polynomial E) (CoordinateZeroLocalRing E) Polynomial.X) = x := by
274     change ((f
275       (algebraMap (Polynomial E) (CoordinateZeroLocalRing E) Polynomial.X) :
276         V.toSubring) : K) = x
277     rw [coordinateZeroLocalFactor_map_X E]
278   have xV_ne : xV ≠ 0 := by
279     intro hx0
280     apply htrans
281     have hx0K : x = 0 := congrArg Subtype.val hx0
282     exact hx0K ▸ isAlgebraic_zero
283   have xV_nonunit : ¬ IsUnit xV := by
284     rw [← mem_nonunits_iff, ← IsLocalRing.mem_maximalIdeal]
285     exact hxm
286   have hresfinite : ResidueExtensionFinite f hf :=
287     coordinateZeroLocalFactor_residue_finite E V hEV x hxV hxm hfinite
288   let place : RetainedDVRPlace (CoordinateZeroLocalRing E) (L := K)
289       (algebraMap (Polynomial E) (CoordinateZeroLocalRing E) Polynomial.X) :=
290     { valuation := V
291       isDiscrete := hVdvr
292       parameter := xV
293       parameter_eq := coordinate_eq.symm
294       parameter_ne := xV_ne
295       parameter_nonunit := xV_nonunit
296       factor := f
297       factor_commutes := rfl
298       factor_isLocal := hf
299       residue_finite := hresfinite }
300   exact
301     { coefficientField := E
302       coordinate_transcendental := htrans
303       ambientAlgebra := ambient
304       coefficientTower := tower
305       coordinate_eq := coordinate_eq
306       place := place }
