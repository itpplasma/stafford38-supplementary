35 private def basePart (g : SymbolRing k n) : MvPolynomial (Fin n) k :=
36   MvPolynomial.eval₂Hom (MvPolynomial.C : k →+* MvPolynomial (Fin n) k)
37     (Sum.elim MvPolynomial.X (fun _ => 0)) g

39 private theorem baseLift_basePart_eq_zeroComponent (g : SymbolRing k n) :
40     baseLift (basePart g) =
41       (DirectSum.decompose (orderDecomposition (k := k) (n := n)) g 0 :
42         SymbolRing k n) := by
43   induction g using MvPolynomial.induction_on with
44   | C a =>
45       have hC : MvPolynomial.C a ∈ orderDecomposition (k := k) (n := n) 0 :=
46         MvPolynomial.isWeightedHomogeneous_C (@orderWeight n) a
47       let C0 : orderDecomposition (k := k) (n := n) 0 := ⟨MvPolynomial.C a, hC⟩
48       have hd := congrArg (fun z => z 0)
49         (DirectSum.decompose_coe (orderDecomposition (k := k) (n := n)) C0)
50       have hd' : (DirectSum.decompose (orderDecomposition (k := k) (n := n))
51           (MvPolynomial.C a) 0 : SymbolRing k n) = MvPolynomial.C a := by
52         simpa [C0] using congrArg Subtype.val hd
53       rw [hd']
54       simp [basePart]
55   | add P Q hP hQ =>
56       rw [DirectSum.decompose_add]
57       change baseLift (basePart (P + Q)) =
58         (DirectSum.decompose (orderDecomposition (k := k) (n := n)) P 0 :
59           SymbolRing k n) +
60         (DirectSum.decompose (orderDecomposition (k := k) (n := n)) Q 0 :
61           SymbolRing k n)
62       rw [← hP, ← hQ]
63       simp [basePart]
64   | mul_X P i hP =>
65       rcases i with i | i
66       · have hX : MvPolynomial.X (Sum.inl i : PhaseVar n) ∈
67             orderDecomposition (k := k) (n := n) 0 :=
68           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) _
69         rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
70           (orderDecomposition (k := k) (n := n)) hX (Nat.zero_le 0)]
71         rw [← hP]
72         simp [basePart, baseLift]
73       · have hX : MvPolynomial.X (Sum.inr i : PhaseVar n) ∈
74             orderDecomposition (k := k) (n := n) 1 :=
75           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) _
76         rw [DirectSum.coe_decompose_mul_of_right_mem_of_not_le
77           (orderDecomposition (k := k) (n := n)) hX (by omega)]
78         simp [basePart]

80 private theorem eval_basePart (y : Fin n → k) (g : SymbolRing k n) :
81     MvPolynomial.eval y (basePart g) =
82       MvPolynomial.eval (zeroSectionPoint y) g := by
83   induction g using MvPolynomial.induction_on with
84   | C a => simp [basePart]
85   | add P Q hP hQ =>
86       simp only [basePart] at hP hQ
87       simp only [basePart, map_add]
88       rw [hP, hQ]
89   | mul_X P i hP =>
90       rcases i with i | i
91       · simp only [basePart, map_mul, MvPolynomial.eval₂Hom_X', Sum.elim_inl,
92           MvPolynomial.eval_X]
93         simp only [basePart] at hP
94         rw [hP]
95         simp [zeroSectionPoint]
96       · simp [basePart, zeroSectionPoint]

98 theorem zeroSection_commonZero_of_isHomogeneous
99     (J : Ideal (SymbolRing k n))
100     (hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := n)))
101     (y : Fin n → k)
102     (hy : ∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0) :
103     ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0 := by
104   intro g hg
105   let f := basePart g
106   have hf : baseLift f ∈ J := by
107     rw [baseLift_basePart_eq_zeroComponent]
108     exact (hhom.mem_iff.mp hg 0)
109   have hzero := hy f hf
110   rw [eval_basePart] at hzero
111   exact hzero

113 /-- The equation-defined conormal over `V(J.comap baseLift)` is contained in
114 `V(J)`, provided the zero section over every base zero is a common zero of
115 `J`.  This is the exact generic bridge; no characteristic-support predicate
116 or radicality assumption is used here. -/
117 theorem equationConormalLocus_subset_zeroLocus
118     [CharZero k]
119     (J : Ideal (SymbolRing k n))
120     (hJ : IsBaseRelativePoisson J)
121     (hzero : ∀ y : Fin n → k,
122       (∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0) →
123         ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0) :
124     equationConormalLocus (J.comap baseLift) ⊆
125       MvPolynomial.zeroLocus k J := by
126   intro q hq
127   let y : Fin n → k := fun i => q (Sum.inl i)
128   let ξ : Fin n → k := fun i => q (Sum.inr i)
129   have hy : ∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0 := by
130     intro f hf
131     exact hq.1 f hf
132   have hs := Stafford38.Geometry.PaperHamiltonianFlow.affineConormal_coordinatePoint_isCommonZero J hJ y
133     (hzero y hy) (J.comap baseLift)
134     (fun f => f.2) ξ hq.2
135   have hsplit : Sum.elim y ξ = q := by
136     funext i
137     rcases i with i | i <;> rfl
138   rw [← hsplit]
139   exact hs

141 theorem equationConormalClosure_subset_zeroLocus
142     [CharZero k]
143     (J : Ideal (SymbolRing k n))
144     (hJ : IsBaseRelativePoisson J)
145     (hzero : ∀ y : Fin n → k,
146       (∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0) →
147         ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0) :
148     equationConormalClosure (J.comap baseLift) ⊆
149       MvPolynomial.zeroLocus k J := by
150   exact zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
151     (equationConormalLocus (J.comap baseLift)) J
152     (equationConormalLocus_subset_zeroLocus J hJ hzero)

154 /-- Fibre-homogeneity supplies the zero-section input required by the generic
155 conormal argument. -/
156 theorem equationConormalClosure_subset_zeroLocus_of_isHomogeneous
157     [CharZero k]
158     (J : Ideal (SymbolRing k n))
159     (hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := n)))
160     (hJ : IsBaseRelativePoisson J) :
161     equationConormalClosure (J.comap baseLift) ⊆
162       MvPolynomial.zeroLocus k J := by
163   apply equationConormalClosure_subset_zeroLocus J hJ
164   intro y hy
165   exact zeroSection_commonZero_of_isHomogeneous J hhom y hy
