37 private def basePart (g : SymbolRing k n) : MvPolynomial (Fin n) k :=
38   MvPolynomial.eval₂Hom (MvPolynomial.C : k →+* MvPolynomial (Fin n) k)
39     (Sum.elim MvPolynomial.X (fun _ => 0)) g

41 private theorem baseLift_basePart_eq_zeroComponent (g : SymbolRing k n) :
42     baseLift (basePart g) =
43       (DirectSum.decompose (orderDecomposition (k := k) (n := n)) g 0 :
44         SymbolRing k n) := by
45   induction g using MvPolynomial.induction_on with
46   | C a =>
47       have hC : MvPolynomial.C a ∈ orderDecomposition (k := k) (n := n) 0 :=
48         MvPolynomial.isWeightedHomogeneous_C (@orderWeight n) a
49       let C0 : orderDecomposition (k := k) (n := n) 0 := ⟨MvPolynomial.C a, hC⟩
50       have hd := congrArg (fun z => z 0)
51         (DirectSum.decompose_coe (orderDecomposition (k := k) (n := n)) C0)
52       have hd' : (DirectSum.decompose (orderDecomposition (k := k) (n := n))
53           (MvPolynomial.C a) 0 : SymbolRing k n) = MvPolynomial.C a := by
54         simpa [C0] using congrArg Subtype.val hd
55       rw [hd']
56       simp [basePart]
57   | add P Q hP hQ =>
58       rw [DirectSum.decompose_add]
59       change baseLift (basePart (P + Q)) =
60         (DirectSum.decompose (orderDecomposition (k := k) (n := n)) P 0 :
61           SymbolRing k n) +
62         (DirectSum.decompose (orderDecomposition (k := k) (n := n)) Q 0 :
63           SymbolRing k n)
64       rw [← hP, ← hQ]
65       simp [basePart]
66   | mul_X P i hP =>
67       rcases i with i | i
68       · have hX : MvPolynomial.X (Sum.inl i : PhaseVar n) ∈
69             orderDecomposition (k := k) (n := n) 0 :=
70           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) _
71         rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
72           (orderDecomposition (k := k) (n := n)) hX (Nat.zero_le 0)]
73         rw [← hP]
74         simp [basePart, baseLift]
75       · have hX : MvPolynomial.X (Sum.inr i : PhaseVar n) ∈
76             orderDecomposition (k := k) (n := n) 1 :=
77           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) _
78         rw [DirectSum.coe_decompose_mul_of_right_mem_of_not_le
79           (orderDecomposition (k := k) (n := n)) hX (by omega)]
80         simp [basePart]

82 private theorem eval_basePart (y : Fin n → k) (g : SymbolRing k n) :
83     MvPolynomial.eval y (basePart g) =
84       MvPolynomial.eval (zeroSectionPoint y) g := by
85   induction g using MvPolynomial.induction_on with
86   | C a => simp [basePart]
87   | add P Q hP hQ =>
88       simp only [basePart] at hP hQ
89       simp only [basePart, map_add]
90       rw [hP, hQ]
91   | mul_X P i hP =>
92       rcases i with i | i
93       · simp only [basePart, map_mul, MvPolynomial.eval₂Hom_X', Sum.elim_inl,
94           MvPolynomial.eval_X]
95         simp only [basePart] at hP
96         rw [hP]
97         simp [zeroSectionPoint]
98       · simp [basePart, zeroSectionPoint]

100 theorem zeroSection_commonZero_of_isHomogeneous
101     (J : Ideal (SymbolRing k n))
102     (hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := n)))
103     (y : Fin n → k)
104     (hy : ∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0) :
105     ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0 := by
106   intro g hg
107   let f := basePart g
108   have hf : baseLift f ∈ J := by
109     rw [baseLift_basePart_eq_zeroComponent]
110     exact (hhom.mem_iff.mp hg 0)
111   have hzero := hy f hf
112   rw [eval_basePart] at hzero
113   exact hzero

115 /-- The equation-defined conormal over `V(J.comap baseLift)` is contained in
116 `V(J)`, provided the zero section over every base zero is a common zero of
117 `J`.  This is the exact generic bridge; no characteristic-support predicate
118 or radicality assumption is used here. -/
119 theorem equationConormalLocus_subset_zeroLocus
120     [CharZero k]
121     (J : Ideal (SymbolRing k n))
122     (hJ : IsBaseRelativePoisson J)
123     (hzero : ∀ y : Fin n → k,
124       (∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0) →
125         ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0) :
126     equationConormalLocus (J.comap baseLift) ⊆
127       MvPolynomial.zeroLocus k J := by
128   intro q hq
129   let y : Fin n → k := fun i => q (Sum.inl i)
130   let ξ : Fin n → k := fun i => q (Sum.inr i)
131   have hy : ∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0 := by
132     intro f hf
133     exact hq.1 f hf
134   have hs := Stafford38.Geometry.PaperHamiltonianFlow.affineConormal_coordinatePoint_isCommonZero J hJ y
135     (hzero y hy) (J.comap baseLift)
136     (fun f => f.2) ξ hq.2
137   have hsplit : Sum.elim y ξ = q := by
138     funext i
139     rcases i with i | i <;> rfl
140   rw [← hsplit]
141   exact hs

143 theorem equationConormalClosure_subset_zeroLocus
144     [CharZero k]
145     (J : Ideal (SymbolRing k n))
146     (hJ : IsBaseRelativePoisson J)
147     (hzero : ∀ y : Fin n → k,
148       (∀ f ∈ J.comap baseLift, MvPolynomial.eval y f = 0) →
149         ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0) :
150     equationConormalClosure (J.comap baseLift) ⊆
151       MvPolynomial.zeroLocus k J := by
152   exact zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
153     (equationConormalLocus (J.comap baseLift)) J
154     (equationConormalLocus_subset_zeroLocus J hJ hzero)

156 /-- Fibre-homogeneity supplies the zero-section input required by the generic
157 conormal argument. -/
158 theorem equationConormalClosure_subset_zeroLocus_of_isHomogeneous
159     [CharZero k]
160     (J : Ideal (SymbolRing k n))
161     (hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := n)))
162     (hJ : IsBaseRelativePoisson J) :
163     equationConormalClosure (J.comap baseLift) ⊆
164       MvPolynomial.zeroLocus k J := by
165   apply equationConormalClosure_subset_zeroLocus J hJ
166   intro y hy
167   exact zeroSection_commonZero_of_isHomogeneous J hhom y hy
