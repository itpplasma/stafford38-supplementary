36 private theorem exists_separator
37     (I P : Ideal (MvPolynomial (Fin n) k))
38     (hI : I.IsRadical) (hP : P ∈ I.minimalPrimes) :
39     ∃ f ∉ P, ∀ h ∈ P, f * h ∈ I := by
40   classical
41   have hfin := Ideal.finite_minimalPrimes_of_isNoetherianRing
42     (MvPolynomial (Fin n) k) I
43   set T := hfin.toFinset.erase P with hT
44   have hchoice : ∀ Q ∈ T, ∃ s ∈ Q, s ∉ P := by
45     intro Q hQ
46     rw [hT, Finset.mem_erase, hfin.mem_toFinset] at hQ
47     by_contra hcon
48     push_neg at hcon
49     exact hQ.1 (le_antisymm hcon (hP.2 hQ.2.1 hcon))
50   choose s hsQ hsP using hchoice
51   let f := ∏ Q ∈ T.attach, s Q.1 Q.2
52   have hfP : f ∉ P := by
53     intro hf
54     obtain ⟨Q, -, hQ⟩ := (hP.1.1.prod_mem_iff).1 hf
55     exact hsP Q.1 Q.2 hQ
56   refine ⟨f, hfP, ?_⟩
57   intro h hp
58   rw [← hI.radical, ← Ideal.sInf_minimalPrimes, Ideal.mem_sInf]
59   intro Q hQ
60   by_cases hQP : Q = P
61   · subst hQP
62     exact Ideal.mul_mem_left _ _ hp
63   · have hQT : Q ∈ T := by
64       rw [hT, Finset.mem_erase, hfin.mem_toFinset]
65       exact ⟨hQP, hQ⟩
66     refine Ideal.mul_mem_right _ _ ?_
67     exact Ideal.mem_of_dvd _
68       (Finset.dvd_prod_of_mem (fun Q : T ↦ s Q.1 Q.2)
69         (Finset.mem_attach _ ⟨Q, hQT⟩))
70       (hsQ Q hQT)

72 theorem equationConormalClosure_minimalPrime_subset_zeroLocus
73     (J : Ideal (SymbolRing k n))
74     (hJrad : J.IsRadical)
75     (hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := n)))
76     (hJpoisson : IsBaseRelativePoisson J)
77     (P : Ideal (MvPolynomial (Fin n) k))
78     (hP : P ∈ (J.comap baseLift).minimalPrimes) :
79     equationConormalClosure P ⊆ MvPolynomial.zeroLocus k J := by
80   let I : Ideal (MvPolynomial (Fin n) k) :=
81     J.comap (baseLift (k := k) (n := n)).toRingHom
82   have hIrad : I.IsRadical := by
83     exact Ideal.IsRadical.comap
84       (f := (baseLift (k := k) (n := n)).toRingHom) hJrad
85   change P ∈ I.minimalPrimes at hP
86   obtain ⟨f, hfP, hfp⟩ := exists_separator I P hIrad hP
87   let S : Set (Fin n → k) := {y | y ∈ MvPolynomial.zeroLocus k P ∧
88     MvPolynomial.eval y f ≠ 0}
89   have hS : S ⊆ MvPolynomial.zeroLocus k P := by
90     intro y hy
91     exact hy.1
92   have hopen : ∀ y : Fin n → k, y ∈ MvPolynomial.zeroLocus k P →
93       MvPolynomial.eval y f ≠ 0 → y ∈ S := by
94     intro y hy hfy
95     exact ⟨hy, hfy⟩
96   have hrestricted :
97       ConormalPrincipalOpenDensity.restrictedEquationConormalLocus P S ⊆
98         MvPolynomial.zeroLocus k J := by
99     intro q hq
100     let y : Fin n → k := fun i ↦ q (Sum.inl i)
101     let ξ : Fin n → k := fun i ↦ q (Sum.inr i)
102     have hyP : y ∈ MvPolynomial.zeroLocus k P := hq.2.1
103     have hfy : MvPolynomial.eval y f ≠ 0 := hq.2.2
104     have hyI : ∀ g ∈ I, MvPolynomial.eval y g = 0 := by
105       intro g hg
106       exact hq.1.1 g (hP.1.2 hg)
107     have hinc : affineConormalSpace y P ≤ affineConormalSpace y I := by
108       rw [affineConormalSpace_eq_equationCovectorSpan y P,
109         affineConormalSpace_eq_equationCovectorSpan y I]
110       apply Submodule.span_le.2
111       intro z hz
112       rcases hz with ⟨h', rfl⟩
113       rcases h' with ⟨h, hh⟩
114       obtain hfh := hfp h hh
115       have hgen : differentialCovector y (f * h) ∈
116           equationCovectorSpan y I := by
117         apply Submodule.subset_span
118         exact ⟨⟨f * h, hfh⟩, rfl⟩
119       have hevalh : MvPolynomial.eval y h = 0 := hq.1.1 h hh
120       rw [differentialCovector_mul, hevalh, zero_smul, zero_add] at hgen
121       have hscaled := (equationCovectorSpan y I).smul_mem
122         (MvPolynomial.eval y f)⁻¹ hgen
123       rw [smul_smul, inv_mul_cancel₀ hfy, one_smul] at hscaled
124       exact hscaled
125     have hxiI : coordinateCovector ξ ∈ affineConormalSpace y I :=
126       hinc hq.1.2
127     have hqI : q ∈ equationConormalLocus I := ⟨hyI, hxiI⟩
128     exact (GeneralConormalContainment.equationConormalLocus_subset_zeroLocus
129         J hJpoisson (fun z hz ↦
130         zeroSection_commonZero_of_isHomogeneous J hhom z hz)) hqI
131   have hclosure := equationConormalClosure_restricted_eq P hP.1.1 f hfP S hS hopen
132   intro q hq
133   apply (zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
134     (ConormalPrincipalOpenDensity.restrictedEquationConormalLocus P S)
135     J hrestricted)
136   rw [hclosure]
137   exact hq
