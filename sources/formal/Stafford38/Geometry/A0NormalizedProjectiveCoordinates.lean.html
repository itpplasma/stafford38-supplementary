162 /-- Any selected-chart column with the canonical denominator and variable
163 entries has its original-affine projective coordinates on the common open. -/
164 theorem q_column_commonOpen_coordinates
165     {k : Type u} [Field k] {m : ℕ}
166     {Q : Type v} [CommRing Q] [Algebra k Q]
167     {B : Type w} [CommRing B] [Algebra Q B]
168     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (j : Fin m)
169     (hxj : componentCoordinate P j ≠ 0)
170     (hsel : SelectedAffineChartQuotient (k := k) P j ≃ₐ[k] Q)
171     (M : Ideal B) [M.IsPrime] (f : Q)
172     (e : Localization.Away f ≃ₐ[Q]
173       Localization.Away (algebraMap Q B f))
174     (qQ : Fin (m + 1) → Q)
175     (hchart : qQ (Fin.succ j) = 1)
176     (hzero : qQ 0 = hsel (selectedAffineChartDenominator P j))
177     (hrows : ∀ i : Fin m, ∀ hij : i ≠ j,
178       qQ (Fin.succ i) = hsel (selectedAffineChartVariableClass P j i hij)) :
179     let g := hsel (selectedAffineChartDenominator P j)
180     let Cq := genericOpenRing M f e
181     let C := genericOpenExtraAwayB M f e g
182     letI : Semiring C := (inferInstance : CommSemiring C).toSemiring
183     letI : Algebra Q Cq := inferInstance
184     letI : Algebra Cq C := inferInstance
185     letI : Algebra Q C := Algebra.compHom C (algebraMap Q Cq)
186     let qU : Fin (m + 1) → C := fun i =>
187       algebraMap Cq C (algebraMap Q Cq (qQ i))
188     ∀ i : Fin m,
189       qU (Fin.succ i) = qU 0 *
190         originalAffineChartToCommonOpen P j hxj hsel M f e
191           (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i)) := by
192   classical
193   dsimp
194   let g := hsel (selectedAffineChartDenominator P j)
195   let Cq := genericOpenRing M f e
196   let C := genericOpenExtraAwayB M f e g
197   letI : Semiring C := (inferInstance : CommSemiring C).toSemiring
198   letI : Algebra Q Cq := inferInstance
199   letI : Algebra Cq C := inferInstance
200   letI : Algebra Q C := Algebra.compHom C (algebraMap Q Cq)
201   intro i
202   by_cases hij : i = j
203   · subst i
204     simpa [hzero, hchart] using
205       (selectedGenerator_commonOpen_product P j hxj hsel M f e).symm
206   · have hother := originalGenerator_commonOpen_identity
207       P j i hij hxj hsel M f e
208     simpa [hzero, hrows i hij] using hother
