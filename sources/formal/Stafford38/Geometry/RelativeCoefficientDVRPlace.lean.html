8 # Relative coefficients in a retained DVR place
9 
10 For the source DVR `E[X]_(X)`, the retained local factor contains the
11 coefficient field `E`.  This file constructs that coefficient map, checks its
12 compatibility with the ambient-field embedding, and proves that the resulting
13 residue extension is finite and separable in characteristic zero.
14 
15 No inverse limit, completion, or power-series chart is constructed here.
16 -/

31 abbrev SourceDVR (E : Type u) [Field E] :=
32   CoordinateZeroLocalRing E
33 
34 /-- Evaluation at zero on the source coordinate DVR. -/
35 def coordinateConstantTerm
36     (E : Type u) [Field E] : SourceDVR E →+* E :=
37   IsLocalization.lift (S := SourceDVR E)
38     (M := (coordinateZeroPrime E).primeCompl)
39     (g := Polynomial.evalRingHom (0 : E)) fun y ↦ by
40       rw [isUnit_iff_ne_zero]
41       intro hy
42       apply y.2
43       have hker :
44           RingHom.ker (Polynomial.evalRingHom (0 : E)) =
45             coordinateZeroPrime E := by
46         rw [Polynomial.ker_evalRingHom, coordinateZeroPrime]
47         have hCzero : Polynomial.C (0 : E) = 0 :=
48           map_zero Polynomial.C
49         rw [hCzero, sub_zero]
50       exact hker.le (RingHom.mem_ker.mpr hy)
51 
52 /-- Evaluation at zero is onto because constant polynomials survive the
53 localization. -/
54 theorem coordinateConstantTerm_surjective
55     (E : Type u) [Field E] :
56     Function.Surjective (coordinateConstantTerm E) := by
57   intro e
58   refine ⟨algebraMap (Polynomial E) (SourceDVR E) (Polynomial.C e), ?_⟩
59   rw [coordinateConstantTerm, IsLocalization.lift_eq]
60   exact Polynomial.eval_C
61 
62 /-- The residue field of `E[X]_(X)` is canonically `E`. -/
63 noncomputable def coordinateResidueEquiv
64     (E : Type u) [Field E] :
65     ResidueField (SourceDVR E) ≃+* E := by
66   let f := coordinateConstantTerm E
67   letI : IsLocalHom f := (coordinateConstantTerm_surjective E).isLocalHom
68   let g : ResidueField (SourceDVR E) →+* E := ResidueField.lift f
69   apply RingEquiv.ofBijective g
70   refine ⟨g.injective, ?_⟩
71   intro e
72   obtain ⟨a, rfl⟩ := coordinateConstantTerm_surjective E e
73   exact ⟨residue (SourceDVR E) a, ResidueField.lift_residue_apply f a⟩
74 
75 @[simp]
76 theorem coordinateResidueEquiv_algebraMap
77     (E : Type u) [Field E] (e : E) :
78     coordinateResidueEquiv E
79         (algebraMap E (ResidueField (SourceDVR E)) e) = e := by
80   letI : IsLocalHom (coordinateConstantTerm E) :=
81     (coordinateConstantTerm_surjective E).isLocalHom
82   change ResidueField.lift (coordinateConstantTerm E)
83       (residue (SourceDVR E)
84         (algebraMap E (SourceDVR E) e)) = e
85   rw [ResidueField.lift_residue_apply]
86   change coordinateConstantTerm E
87       (algebraMap (Polynomial E) (SourceDVR E) (Polynomial.C e)) = e
88   rw [coordinateConstantTerm, IsLocalization.lift_eq]
89   exact Polynomial.eval_C

91 /-- The retained local factor restricted to constant coefficients. -/
92 def relativeCoefficientMap
93     (E : Type u) [Field E]
94     {L : Type v} [Field L] [Algebra (SourceDVR E) L]
95     {a : SourceDVR E}
96     (D : RetainedDVRPlace (SourceDVR E) (L := L) a) :
97     E →+* D.valuation.toSubring :=
98   D.factor.comp (algebraMap E (SourceDVR E))
99 
100 /-- The coefficient map retained in the valuation subring is the original
101 ambient embedding of `E`. -/
102 theorem relativeCoefficientMap_commutes
103     (E : Type u) [Field E]
104     {L : Type v} [Field L] [Algebra E L]
105     [Algebra (SourceDVR E) L] [IsScalarTower E (SourceDVR E) L]
106     {a : SourceDVR E}
107     (D : RetainedDVRPlace (SourceDVR E) (L := L) a) :
108     D.valuation.toSubring.subtype.comp (relativeCoefficientMap E D) =
109       algebraMap E L := by
110   ext e
111   change (D.factor (algebraMap E (SourceDVR E) e) : L) =
112     algebraMap E L e
113   calc
114     (D.factor (algebraMap E (SourceDVR E) e) : L) =
115         algebraMap (SourceDVR E) L
116           (algebraMap E (SourceDVR E) e) :=
117       DFunLike.congr_fun D.factor_commutes
118         (algebraMap E (SourceDVR E) e)
119     _ = algebraMap E L e :=
120       (IsScalarTower.algebraMap_apply E (SourceDVR E) L e).symm

122 /-- The finite residue extension retained from the source DVR remains finite
123 over its coefficient field. -/
124 theorem relativeResidue_finite
125     (E : Type u) [Field E]
126     {L : Type v} [Field L] [Algebra (SourceDVR E) L]
127     {a : SourceDVR E}
128     (D : RetainedDVRPlace (SourceDVR E) (L := L) a) :
129     letI : IsDiscreteValuationRing D.valuation.toSubring := D.isDiscrete
130     letI : Algebra E D.valuation.toSubring :=
131       (relativeCoefficientMap E D).toAlgebra
132     Module.Finite E (ResidueField D.valuation.toSubring) := by
133   let A := SourceDVR E
134   let V := D.valuation.toSubring
135   letI : IsDiscreteValuationRing V := D.isDiscrete
136   letI : Algebra A V := D.factor.toAlgebra
137   letI : IsLocalHom (algebraMap A V) := D.factor_isLocal
138   letI : Algebra E V := (relativeCoefficientMap E D).toAlgebra
139   letI : IsScalarTower E A V :=
140     IsScalarTower.of_algebraMap_eq fun _ ↦ rfl
141   have hsourceSurjective :
142       Function.Surjective (algebraMap E (ResidueField A)) := by
143     intro z
144     refine ⟨coordinateResidueEquiv E z, ?_⟩
145     apply (coordinateResidueEquiv E).injective
146     exact coordinateResidueEquiv_algebraMap E
147       (coordinateResidueEquiv E z)
148   have hsourceFinite : Module.Finite E (ResidueField A) :=
149     Module.Finite.of_surjective
150       (Algebra.linearMap E (ResidueField A)) hsourceSurjective
151   letI : Module.Finite E (ResidueField A) := hsourceFinite
152   have htargetFinite :
153       Module.Finite (ResidueField A) (ResidueField V) := by
154     exact D.residue_finite
155   letI : Module.Finite (ResidueField A) (ResidueField V) := htargetFinite
156   letI : IsScalarTower E (ResidueField A) (ResidueField V) :=
157     IsScalarTower.of_algebraMap_eq fun _ ↦ rfl
158   exact Module.Finite.trans (ResidueField A) (ResidueField V)

160 /-- In characteristic zero, the finite residue extension induced by the
161 retained coefficient map is separable. -/
162 theorem relativeResidue_isSeparable
163     (E : Type u) [Field E] [CharZero E]
164     {L : Type v} [Field L] [Algebra (SourceDVR E) L]
165     {a : SourceDVR E}
166     (D : RetainedDVRPlace (SourceDVR E) (L := L) a) :
167     letI : IsDiscreteValuationRing D.valuation.toSubring := D.isDiscrete
168     letI : Algebra E D.valuation.toSubring :=
169       (relativeCoefficientMap E D).toAlgebra
170     Algebra.IsSeparable E (ResidueField D.valuation.toSubring) := by
171   letI : IsDiscreteValuationRing D.valuation.toSubring := D.isDiscrete
172   letI : Algebra E D.valuation.toSubring :=
173     (relativeCoefficientMap E D).toAlgebra
174   letI : Module.Finite E (ResidueField D.valuation.toSubring) :=
175     relativeResidue_finite E D
176   letI : Algebra.IsIntegral E (ResidueField D.valuation.toSubring) :=
177     ⟨fun y ↦ (IsAlgebraic.of_finite E y).isIntegral⟩
178   exact Algebra.IsSeparable.of_integral E
179     (ResidueField D.valuation.toSubring)
