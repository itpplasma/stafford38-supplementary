20 /-- Translation sending the polynomial variable `X i` to `X i - p i`. -/
21 def translate (p : σ → k) : MvPolynomial σ k ≃ₐ[k] MvPolynomial σ k := by
22   let f : MvPolynomial σ k →ₐ[k] MvPolynomial σ k :=
23     MvPolynomial.aeval (fun i => X i - C (p i))
24   let g : MvPolynomial σ k →ₐ[k] MvPolynomial σ k :=
25     MvPolynomial.aeval (fun i => X i + C (p i))
26   apply AlgEquiv.ofAlgHom f g
27   · apply MvPolynomial.algHom_ext
28     intro i
29     simp [f, g]
30   · apply MvPolynomial.algHom_ext
31     intro i
32     simp [f, g]

34 /-- The kernel of evaluation at the origin is the ideal generated by the variables. -/
35 theorem idealOfVars_eq_ker_eval_zero :
36     MvPolynomial.idealOfVars σ k =
37       RingHom.ker (MvPolynomial.aeval (fun _ : σ => (0 : k))) := by
38   ext p
39   change p ∈ Ideal.span (Set.range (X : σ → MvPolynomial σ k)) ↔ _
40   rw [← Set.image_univ]
41   rw [MvPolynomial.mem_ideal_span_X_image]
42   simp only [Set.mem_univ, true_and, RingHom.mem_ker]
43   have heval : MvPolynomial.aeval (fun _ : σ => (0 : k)) p =
44       MvPolynomial.constantCoeff p := by
45     simp
46   rw [heval]
47   constructor
48   · intro h
49     by_contra hconst
50     have hzero : (0 : σ →₀ ℕ) ∈ p.support := MvPolynomial.mem_support_iff.mpr hconst
51     obtain ⟨i, hi⟩ := h 0 hzero
52     exact hi (by simp)
53   · intro h
54     have hconst : MvPolynomial.coeff 0 p = 0 := by
55       simpa [MvPolynomial.constantCoeff_eq] using h
56     intro m hm
57     have hm0 : m ≠ 0 := by
58       intro hz
59       subst hz
60       exact (MvPolynomial.mem_support_iff.mp hm) hconst
61     have hcoords : ∃ i, m i ≠ 0 := by
62       by_contra hnone
63       apply hm0
64       apply Finsupp.ext
65       intro i
66       by_contra hne
67       exact hnone ⟨i, hne⟩
68     exact hcoords

70 /-- The maximal evaluation ideal of the affine point `p`. -/
71 def pointIdeal (p : σ → k) : Ideal (MvPolynomial σ k) :=
72   RingHom.ker (MvPolynomial.aeval p)
73 
74 /-- Translation carries the origin ideal onto the evaluation ideal at `p`. -/
75 theorem pointIdeal_eq_map_translate (p : σ → k) :
76     pointIdeal p = (MvPolynomial.idealOfVars σ k).map
77       (translate p).toRingEquiv.toRingHom := by
78   let e := (translate p).toRingEquiv.toRingHom
79   have heval : (MvPolynomial.aeval p).comp (translate p).toAlgHom =
80       MvPolynomial.aeval (fun _ : σ => (0 : k)) := by
81     apply MvPolynomial.algHom_ext
82     intro i
83     simp [translate]
84   have hcomap : Ideal.comap e (pointIdeal p) = MvPolynomial.idealOfVars σ k := by
85     ext f
86     change ((MvPolynomial.aeval p).comp (translate p).toAlgHom) f = 0 ↔
87       f ∈ MvPolynomial.idealOfVars σ k
88     rw [heval]
89     rw [idealOfVars_eq_ker_eval_zero]
90     rfl
91   have hsurj : Function.Surjective e := (translate p).toRingEquiv.surjective
92   have hmap := Ideal.map_comap_of_surjective e hsurj (pointIdeal p)
93   rw [hcomap] at hmap
94   exact hmap.symm

96 /-- The completed affine polynomial ring at a rational point is the multivariate
97 power-series ring, as a ring. This is the affine-chart model; it does not assert
98 that an arbitrary smooth local ring has this completion. -/
99 def powerSeriesCompletionAtPoint (p : σ → k) :
100     MvPowerSeries σ k ≃+* AdicCompletion (pointIdeal p) (MvPolynomial σ k) := by
101   let I := MvPolynomial.idealOfVars σ k
102   let J := pointIdeal p
103   have hJ : J = I.map (translate p).toRingEquiv.toRingHom := by
104     simpa [I, J] using pointIdeal_eq_map_translate p
105   exact (MvPowerSeries.toAdicCompletionAlgEquiv σ k).toRingEquiv.trans
106     (AdicCompletionRingEquiv.ofRingEquiv (translate p).toRingEquiv I J hJ)
107 
108 end
