11 # Relative transcendence tower containing a selected coordinate
12 
13 This file extends a prescribed transcendental coordinate to a transcendence
14 basis, removes it to obtain a relative coefficient field, and proves
15 finiteness and separability over the full basis field. The concrete fraction
16 field comparison for `E[X]_(X)` and retained DVR place are assembled in
17 `RelativeRetainedBoundaryPlace.exists_data_of_fg_charZero`.

28 /-- A transcendental element belongs to some transcendence basis.  Unlike the
29 empty-seed library theorem, this version retains the prescribed element. -/
30 theorem exists_transcendenceBasis_containing
31     (k : Type u) [Field k]
32     {K : Type v} [Field K] [Algebra k K]
33     (x : K) (hx : Transcendental k x) :
34     ∃ s : Set K, x ∈ s ∧
35       IsTranscendenceBasis k ((↑) : s → K) := by
36   have hsingleton :
37       AlgebraicIndependent k ((↑) : ({x} : Set K) → K) := by
38     rw [algebraicIndependent_singleton_iff ⟨x, Set.mem_singleton x⟩]
39     exact hx
40   obtain ⟨s, hxs, hsmax⟩ :=
41     exists_maximal_algebraicIndependent ({x} : Set K) Set.univ
42       (Set.subset_univ _) hsingleton
43   refine ⟨s, hxs (Set.mem_singleton x), hsmax.1.1, ?_⟩
44   intro t ht hst
45   have hrs : Set.range ((↑) : s → K) = s := Subtype.range_coe
46   rw [hrs] at hst ⊢
47   apply hsmax.eq_of_le ⟨ht, Set.subset_univ _⟩
48   exact hst

50 /-- The relative coefficient field obtained by deleting `x` from a basis. -/
51 abbrev coefficientField
52     (k : Type u) [Field k]
53     {K : Type v} [Field K] [Algebra k K]
54     (s : Set K) (x : K) : IntermediateField k K :=
55   IntermediateField.adjoin k (s \ {x})

57 /-- The field generated by a set is algebraic over the ordinary algebra
58 generated by that set.  Concretely, field-adjoin induction adds only inverses
59 to the algebraic closure operations. -/
60 theorem intermediateAdjoin_isAlgebraic_over_algebraAdjoin
61     (k : Type u) [Field k]
62     {K : Type v} [Field K] [Algebra k K]
63     (t : Set K) :
64     let R := Algebra.adjoin k t
65     let E := IntermediateField.adjoin k t
66     letI : Algebra R E :=
67       (Subalgebra.inclusion
68         (IntermediateField.algebra_adjoin_le_adjoin k t)).toAlgebra
69     Algebra.IsAlgebraic R E := by
70   dsimp only
71   let R := Algebra.adjoin k t
72   let E := IntermediateField.adjoin k t
73   let hRE : R ≤ E.toSubalgebra :=
74     IntermediateField.algebra_adjoin_le_adjoin k t
75   letI : Algebra R E := (Subalgebra.inclusion hRE).toAlgebra
76   refine ⟨fun y ↦ ?_⟩
77   apply IntermediateField.adjoin_induction k
78       (p := fun z hz ↦ IsAlgebraic R (⟨z, hz⟩ : E))
79   · intro z hz
80     have hmem : z ∈ E := IntermediateField.subset_adjoin k t hz
81     have heq : (⟨z, hmem⟩ : E) = algebraMap R E
82         (⟨z, Algebra.subset_adjoin hz⟩ : R) := by
83       apply Subtype.ext
84       rfl
85     rw [heq]
86     exact isAlgebraic_algebraMap _
87   · intro a
88     have heq : (⟨algebraMap k K a,
89         (IntermediateField.adjoin k t).algebraMap_mem a⟩ : E) =
90         algebraMap R E (algebraMap k R a) := by
91       apply Subtype.ext
92       rfl
93     rw [heq]
94     exact isAlgebraic_algebraMap _
95   · intro a b _ _ ha hb
96     exact ha.add hb
97   · intro a _ ha
98     exact ha.inv
99   · intro a b _ _ ha hb
100     exact ha.mul hb

102 /-- Deleting a prescribed element from a transcendence basis leaves a
103 coefficient field over which that element is still transcendental.  This is
104 the load-bearing relative-basis lemma needed to evaluate `E[X]` at `x`. -/
105 theorem transcendental_over_coefficientField
106     (k : Type u) [Field k]
107     {K : Type v} [Field K] [Algebra k K]
108     {s : Set K} (hs : IsTranscendenceBasis k ((↑) : s → K))
109     {x : K} (hx : x ∈ s) :
110     Transcendental (coefficientField k s x) x := by
111   let i : s := ⟨x, hx⟩
112   let t : Set s := {j | (j : K) ≠ x}
113   have hi : i ∉ t := by simp [i, t]
114   have ht : ((fun j : s ↦ (j : K)) '' t) = s \ {x} := by
115     ext y
116     constructor
117     · rintro ⟨j, hj, rfl⟩
118       exact ⟨j.property, by simpa [t] using hj⟩
119     · rintro ⟨hy, hyx⟩
120       exact ⟨⟨y, hy⟩, by simpa [t] using hyx, rfl⟩
121   have htr : Transcendental (Algebra.adjoin k (s \ {x})) x := by
122     have h := hs.1.transcendental_adjoin (s := t) hi
123     rw [ht] at h
124     exact h
125   let R := Algebra.adjoin k (s \ {x})
126   let E := coefficientField k s x
127   let hRE : R ≤ E.toSubalgebra :=
128     IntermediateField.algebra_adjoin_le_adjoin k (s \ {x})
129   letI : Algebra R E := (Subalgebra.inclusion hRE).toAlgebra
130   letI : Algebra R K := R.val.toRingHom.toAlgebra
131   letI : IsScalarTower R E K :=
132     IsScalarTower.of_algebraMap_eq (fun _ ↦ rfl)
133   letI : Algebra.IsAlgebraic R E :=
134     intermediateAdjoin_isAlgebraic_over_algebraAdjoin k (s \ {x})
135   exact htr.extendScalars E

137 /-- Adjoining `x` to the coefficient field obtained from `s \ {x}` recovers
138 the field generated by all of `s`, provided `x ∈ s`.  This is the exact
139 relative decomposition used by the valuation construction. -/
140 theorem restrictScalars_adjoin_coordinate
141     (k : Type u) [Field k]
142     {K : Type v} [Field K] [Algebra k K]
143     (s : Set K) (x : K) (hx : x ∈ s) :
144     (IntermediateField.adjoin (coefficientField k s x) ({x} : Set K)).restrictScalars k =
145       IntermediateField.adjoin k s := by
146   rw [IntermediateField.adjoin_adjoin_left]
147   congr 1
148   rw [Set.union_comm, ← Set.insert_eq, Set.insert_diff_singleton,
149     Set.insert_eq_of_mem hx]

151 /-- A finitely generated field extension is finite over the field generated
152 by any transcendence basis.  Finite generation here is field-theoretic
153 (`IntermediateField.FG`), not finite type as an algebra (which would be
154 impossible for a transcendental field extension). -/
155 theorem finiteDimensional_over_adjoin_transcendenceBasis
156     (k : Type u) [Field k]
157     {K : Type v} [Field K] [Algebra k K]
158     (hfg : (⊤ : IntermediateField k K).FG)
159     {s : Set K} (hs : IsTranscendenceBasis k ((↑) : s → K)) :
160     FiniteDimensional (IntermediateField.adjoin k s) K := by
161   let B : IntermediateField k K := IntermediateField.adjoin k s
162   haveI : Algebra.IsAlgebraic B K := by
163     have halg := hs.isAlgebraic_field
164     have hrange : Set.range ((↑) : s → K) = s := Subtype.range_coe
165     rw [hrange] at halg
166     exact halg
167   obtain ⟨t, ht⟩ := hfg
168   have hBt : IntermediateField.adjoin B (t : Set K) = ⊤ := by
169     rw [eq_top_iff]
170     intro y _
171     have hy : y ∈ IntermediateField.adjoin k (t : Set K) := by
172       rw [ht]
173       trivial
174     apply IntermediateField.adjoin_induction k
175         (p := fun z _ ↦ z ∈ IntermediateField.adjoin B (t : Set K))
176     · intro z hz
177       exact IntermediateField.subset_adjoin B (t : Set K) hz
178     · intro a
179       exact (IntermediateField.adjoin B (t : Set K)).algebraMap_mem
180         ⟨algebraMap k K a, B.algebraMap_mem a⟩
181     · exact fun _ _ _ _ ha hb ↦ add_mem ha hb
182     · exact fun _ _ ha ↦ inv_mem ha
183     · exact fun _ _ _ _ ha hb ↦ mul_mem ha hb
184     · exact hy
185   have hfieldAlg := IntermediateField.adjoin_toSubalgebra_of_isAlgebraic
186     (F := B) (E := K) (S := (t : Set K))
187     (fun z _ ↦ Algebra.IsAlgebraic.isAlgebraic z)
188   rw [hBt] at hfieldAlg
189   letI : Algebra.FiniteType B K := ⟨⟨t, hfieldAlg.symm⟩⟩
190   exact Algebra.IsIntegral.finite

192 /-- The full relative tower produced from a prescribed transcendental
193 coordinate.  It records actual fields `E ⊆ F ⊆ K`, the equality
194 `F|_k = k(s)`, membership of `x`, and finiteness of `K/F`.
195 
196 This is stronger than merely postulating a convenient coefficient field: the
197 fields are constructed from a basis containing the supplied coordinate. -/
198 theorem exists_relative_finite_tower
199     (k : Type u) [Field k]
200     {K : Type v} [Field K] [Algebra k K]
201     (hfg : (⊤ : IntermediateField k K).FG)
202     (x : K) (hx : Transcendental k x) :
203     ∃ (s : Set K) (E : IntermediateField k K)
204         (F : IntermediateField E K),
205       x ∈ s ∧
206       IsTranscendenceBasis k ((↑) : s → K) ∧
207       E = coefficientField k s x ∧
208       F = IntermediateField.adjoin E ({x} : Set K) ∧
209       F.restrictScalars k = IntermediateField.adjoin k s ∧
210       Transcendental E x ∧
211       x ∈ F ∧
212       FiniteDimensional F K := by
213   obtain ⟨s, hxs, hs⟩ := exists_transcendenceBasis_containing k x hx
214   let E : IntermediateField k K := coefficientField k s x
215   let F : IntermediateField E K := IntermediateField.adjoin E ({x} : Set K)
216   have hF : F.restrictScalars k = IntermediateField.adjoin k s := by
217     exact restrictScalars_adjoin_coordinate k s x hxs
218   have hfinB : FiniteDimensional (IntermediateField.adjoin k s) K :=
219     finiteDimensional_over_adjoin_transcendenceBasis k hfg hs
220   letI : FiniteDimensional (IntermediateField.adjoin k s) K := hfinB
221   -- `F` and `k(s)` have the same elements in `K`; use the induced inclusion
222   -- tower to transfer algebraicity and finite type to the relative field.
223   have hB_le_F : IntermediateField.adjoin k s ≤ F.restrictScalars k := hF.ge
224   letI : Algebra (IntermediateField.adjoin k s) F :=
225     (IntermediateField.inclusion hB_le_F).toAlgebra
226   letI : IsScalarTower (IntermediateField.adjoin k s) F K :=
227     IsScalarTower.of_algebraMap_eq (fun _ ↦ rfl)
228   haveI : Algebra.IsAlgebraic (IntermediateField.adjoin k s) K :=
229     Algebra.IsAlgebraic.of_finite (IntermediateField.adjoin k s) K
230   haveI : Algebra.IsAlgebraic F K :=
231     Algebra.IsAlgebraic.extendScalars
232       (IntermediateField.inclusion_injective hB_le_F)
233   letI : Algebra.FiniteType F K :=
234     Algebra.FiniteType.of_restrictScalars_finiteType
235       (IntermediateField.adjoin k s) F K
236   letI : Algebra.IsIntegral F K :=
237     ⟨fun y ↦ (Algebra.IsAlgebraic.isAlgebraic y).isIntegral⟩
238   have hfinF : FiniteDimensional F K := Algebra.IsIntegral.finite
239   have hxE : Transcendental E x :=
240     transcendental_over_coefficientField k hs hxs
241   refine ⟨s, E, F, hxs, hs, rfl, rfl, hF, hxE, ?_, hfinF⟩
242   exact IntermediateField.subset_adjoin E ({x} : Set K) (Set.mem_singleton x)

244 /-- In characteristic zero, the finite relative extension just constructed is
245 automatically separable.  This closes the separability hypothesis required by
246 `DivisorialBoundaryExtension`. -/
247 theorem exists_relative_finite_separable_tower_of_charZero
248     (k : Type u) [Field k] [CharZero k]
249     {K : Type v} [Field K] [Algebra k K]
250     (hfg : (⊤ : IntermediateField k K).FG)
251     (x : K) (hx : Transcendental k x) :
252     ∃ (s : Set K) (E : IntermediateField k K)
253         (F : IntermediateField E K),
254       x ∈ s ∧
255       IsTranscendenceBasis k ((↑) : s → K) ∧
256       E = coefficientField k s x ∧
257       F = IntermediateField.adjoin E ({x} : Set K) ∧
258       F.restrictScalars k = IntermediateField.adjoin k s ∧
259       Transcendental E x ∧
260       x ∈ F ∧
261       FiniteDimensional F K ∧
262       Algebra.IsSeparable F K := by
263   obtain ⟨s, E, F, hxs, hs, hE, hFdef, hFrestrict, hxE, hxF, hfin⟩ :=
264     exists_relative_finite_tower k hfg x hx
265   letI : FiniteDimensional F K := hfin
266   letI : CharZero F :=
267     charZero_of_injective_algebraMap (algebraMap k F).injective
268   letI : Algebra.IsIntegral F K :=
269     ⟨fun y ↦ (IsAlgebraic.of_finite F y).isIntegral⟩
270   let hsep : Algebra.IsSeparable F K :=
271     Algebra.IsSeparable.of_integral F K
272   exact ⟨s, E, F, hxs, hs, hE, hFdef, hFrestrict, hxE, hxF, hfin, hsep⟩
