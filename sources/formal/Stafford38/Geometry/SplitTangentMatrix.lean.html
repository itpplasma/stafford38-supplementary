8 /-!
9 # Splitting a rectangular power-series matrix from a selected minor
10 
11 Let `B` be a matrix whose columns are indexed by `κ`.  An embedding
12 `rows : κ ↪ ι` selects a square minor of `B`.  If the determinant of that
13 minor is a unit, its nonsingular inverse, followed by the row-selector matrix,
14 is an explicit left inverse of `B`.
15 
16 For matrices over `k[[X]]`, it is enough to assume that the determinant of the
17 selected minor has nonzero constant coefficient.  This file does not assert
18 that a suitable minor exists for a geometric tangent family.
19 -/
20

25 universe u v

27 variable {R : Type u} [CommRing R]
28 
29 /-- The square submatrix obtained by retaining the explicitly selected rows. -/
30 def selectedMinor {ι κ : Type*} (B : Matrix ι κ R) (rows : κ ↪ ι) : Matrix κ κ R :=
31   fun a b => B (rows a) b
32 
33 /-- The matrix which restricts a column indexed by `ι` to the selected rows. -/
34 def rowSelector {ι κ : Type*} [DecidableEq ι] (rows : κ ↪ ι) : Matrix κ ι R :=
35   fun a i => if rows a = i then 1 else 0
36 
37 theorem rowSelector_mul
38     {ι κ : Type*} [Fintype ι] [DecidableEq ι]
39     (B : Matrix ι κ R) (rows : κ ↪ ι) :
40     rowSelector (R := R) rows * B = selectedMinor B rows := by
41   ext a b
42   simp [Matrix.mul_apply, rowSelector, selectedMinor]
43 
44 /--
45 The constructive left inverse supplied by an invertible selected minor.
46 
47 The nonsingular inverse is the adjugate divided by the determinant in
48 Mathlib's matrix API; no inverse of the rectangular matrix is used.
49 -/
50 def selectedMinorLeftInverse
51     {ι κ : Type*} [Fintype ι] [Fintype κ]
52     [DecidableEq ι] [DecidableEq κ]
53     (B : Matrix ι κ R) (rows : κ ↪ ι) : Matrix κ ι R :=
54   (selectedMinor B rows)⁻¹ * rowSelector (R := R) rows
55 
56 /-- A unit determinant of the selected square minor makes the construction a
57 left inverse of the original rectangular matrix. -/
58 theorem selectedMinorLeftInverse_mul
59     {ι κ : Type*} [Fintype ι] [Fintype κ]
60     [DecidableEq ι] [DecidableEq κ]
61     (B : Matrix ι κ R) (rows : κ ↪ ι)
62     (hdet : IsUnit (selectedMinor B rows).det) :
63     selectedMinorLeftInverse B rows * B = 1 := by
64   rw [selectedMinorLeftInverse, Matrix.mul_assoc, rowSelector_mul]
65   exact Matrix.nonsing_inv_mul _ hdet
66 
67 /-- The selected-minor construction, packaged as an explicit split matrix. -/
68 theorem exists_leftInverse_of_selectedMinor_det_isUnit
69     {ι κ : Type*} [Fintype ι] [Fintype κ]
70     [DecidableEq ι] [DecidableEq κ]
71     (B : Matrix ι κ R) (rows : κ ↪ ι)
72     (hdet : IsUnit (selectedMinor B rows).det) :
73     ∃ C : Matrix κ ι R, C * B = 1 :=
74   ⟨selectedMinorLeftInverse B rows,
75     selectedMinorLeftInverse_mul B rows hdet⟩

77 /-! ## Power-series criterion -/
78 
79 variable {k : Type v} [Field k]
80 
81 /-- Over `k[[X]]`, a nonzero constant coefficient makes the selected
82 determinant a unit and hence gives the same explicit left inverse. -/
83 theorem powerSeries_selectedMinorLeftInverse_mul
84     {ι κ : Type*} [Fintype ι] [Fintype κ]
85     [DecidableEq ι] [DecidableEq κ]
86     (B : Matrix ι κ (PowerSeries k)) (rows : κ ↪ ι)
87     (hconst : PowerSeries.constantCoeff (selectedMinor B rows).det ≠ 0) :
88     selectedMinorLeftInverse B rows * B = 1 := by
89   apply selectedMinorLeftInverse_mul
90   rw [PowerSeries.isUnit_iff_constantCoeff]
91   exact isUnit_iff_ne_zero.mpr hconst
92 
93 /-- Existence form used by the tangent-limit annihilator construction. -/
94 theorem powerSeries_exists_leftInverse_of_selectedMinor_constantCoeff_ne_zero
95     {ι κ : Type*} [Fintype ι] [Fintype κ]
96     [DecidableEq ι] [DecidableEq κ]
97     (B : Matrix ι κ (PowerSeries k)) (rows : κ ↪ ι)
98     (hconst : PowerSeries.constantCoeff (selectedMinor B rows).det ≠ 0) :
99     ∃ C : Matrix κ ι (PowerSeries k), C * B = 1 :=
100   ⟨selectedMinorLeftInverse B rows,
101     powerSeries_selectedMinorLeftInverse_mul B rows hconst⟩
