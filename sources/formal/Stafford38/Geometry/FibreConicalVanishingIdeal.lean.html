23 /-- Stability under every nonzero scalar dilation in each cotangent fibre. -/
24 def IsFibreConical (W : Set (PhaseVar n → k)) : Prop :=
25   ∀ q ∈ W, ∀ a : k, a ≠ 0 →
26     Sum.elim (fun i => q (.inl i)) (fun i => a * q (.inr i)) ∈ W
27 
28 /-- The coefficient of fibre degree `d` on a fibre line is evaluation of the
29 `d`th order-weight homogeneous component. -/
30 theorem coeff_fibreLinePolynomial_eq_eval_weightedHomogeneousComponent
31     (y ξ : Fin n → k) (g : SymbolRing k n) (d : ℕ) :
32     (fibreLinePolynomial y ξ g).coeff d =
33       MvPolynomial.eval (Sum.elim y ξ)
34         (DirectSum.decompose (orderDecomposition (k := k) (n := n)) g d :
35           SymbolRing k n) := by
36   induction g using MvPolynomial.induction_on generalizing d with
37   | C c =>
38       have hC : MvPolynomial.C c ∈ orderDecomposition (k := k) (n := n) 0 :=
39         MvPolynomial.isWeightedHomogeneous_C (@orderWeight n) c
40       cases d with
41       | zero =>
42           rw [DirectSum.decompose_of_mem_same
43             (orderDecomposition (k := k) (n := n)) hC]
44           simp
45       | succ d =>
46           rw [DirectSum.decompose_of_mem_ne
47             (orderDecomposition (k := k) (n := n)) hC (Nat.succ_ne_zero d).symm]
48           simp
49   | add P Q hP hQ =>
50       rw [fibreLinePolynomial_add, Polynomial.coeff_add,
51         DirectSum.decompose_add]
52       change _ = MvPolynomial.eval (Sum.elim y ξ)
53         ((DirectSum.decompose (orderDecomposition (k := k) (n := n)) P d :
54           SymbolRing k n) +
55          (DirectSum.decompose (orderDecomposition (k := k) (n := n)) Q d :
56           SymbolRing k n))
57       rw [MvPolynomial.eval_add, hP, hQ]
58   | mul_X P i hP =>
59       rcases i with i | i
60       · have hX : MvPolynomial.X (Sum.inl i : PhaseVar n) ∈
61             orderDecomposition (k := k) (n := n) 0 :=
62           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) _
63         rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
64           (orderDecomposition (k := k) (n := n)) hX (Nat.zero_le d)]
65         simp [fibreLinePolynomial_mul, hP]
66       · have hX : MvPolynomial.X (Sum.inr i : PhaseVar n) ∈
67             orderDecomposition (k := k) (n := n) 1 :=
68           MvPolynomial.isWeightedHomogeneous_X k (@orderWeight n) _
69         cases d with
70         | zero =>
71             rw [DirectSum.coe_decompose_mul_of_right_mem_of_not_le
72               (orderDecomposition (k := k) (n := n)) hX (by omega)]
73             simp [fibreLinePolynomial_mul]
74         | succ d =>
75             rw [DirectSum.coe_decompose_mul_of_right_mem_of_le
76               (orderDecomposition (k := k) (n := n)) hX (by omega)]
77             rw [fibreLinePolynomial_mul, fibreLinePolynomial_X_fibre, ← mul_assoc,
78               Polynomial.coeff_mul_X, Polynomial.coeff_mul_C]
79             simp [hP]

81 /-- Fibre-conicality makes the vanishing ideal homogeneous for fibre degree. -/
82 theorem vanishingIdeal_isHomogeneous_of_isFibreConical
83     (W : Set (PhaseVar n → k)) (hW : IsFibreConical W) :
84     (MvPolynomial.vanishingIdeal k W).IsHomogeneous
85       (orderDecomposition (k := k) (n := n)) := by
86   intro d g hg
87   rw [MvPolynomial.mem_vanishingIdeal_iff] at hg ⊢
88   intro q hq
89   let y : Fin n → k := fun i => q (.inl i)
90   let ξ : Fin n → k := fun i => q (.inr i)
91   have hline : fibreLinePolynomial y ξ g = 0 := by
92     apply Polynomial.eq_zero_of_infinite_isRoot
93     refine Set.Infinite.mono ?_ (Set.infinite_univ.sdiff (Set.finite_singleton 0))
94     intro a ha
95     simp only [Set.mem_diff, Set.mem_univ, Set.mem_singleton_iff, true_and] at ha
96     change Polynomial.eval a (fibreLinePolynomial y ξ g) = 0
97     rw [eval_fibreLinePolynomial]
98     exact hg _ (hW q hq a ha)
99   have hqeq : q = Sum.elim y ξ := by
100     funext i
101     cases i <;> rfl
102   rw [hqeq]
103   change MvPolynomial.eval (Sum.elim y ξ)
104       (DirectSum.decompose (orderDecomposition (k := k) (n := n)) g d :
105         SymbolRing k n) = 0
106   rw [← coeff_fibreLinePolynomial_eq_eval_weightedHomogeneousComponent y ξ g d,
107     hline, Polynomial.coeff_zero]
108 
109 end
110 end Stafford38.Geometry.FibreConicalVanishingIdeal
