9 /-!
10 # Vanishing of the normalized projective denominator
11 
12 Axis avoidance makes the distinguished affine coordinate a unit on every
13 component of the reduced base variety.  At the retained boundary place that
14 coordinate is a nonunit.  Consequently, a common projective normalization of
15 all affine coordinates cannot have unit homogeneous denominator.
16 -/

43 /-- Support avoidance forces the contracted reduced base ideal and the
44 distinguished coordinate to generate the unit ideal. -/
45 theorem reducedOrderBaseIdeal_sup_coordinate_eq_top_of_support_disjoint
46     {k : Type u} [Field k] [IsAlgClosed k] {n : ℕ}
47     (I : RightIdeal (PresentedWeyl k n)) (i : Fin n)
48     (hdisjoint : Disjoint
49       (orderCharacteristicSupport k I)
50       (PrimeSpectrum.zeroLocus
51         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n)))) :
52     reducedOrderBaseIdeal k I ⊔
53         Ideal.span ({MvPolynomial.X i} : Set (MvPolynomial (Fin n) k)) = ⊤ := by
54   classical
55   let B := reducedOrderBaseIdeal k I
56   let J := B ⊔ Ideal.span
57     ({MvPolynomial.X i} : Set (MvPolynomial (Fin n) k))
58   by_contra hJ
59   obtain ⟨M, hMmax, hJM⟩ := Ideal.exists_le_maximal J hJ
60   obtain ⟨y, hMy⟩ :=
61     MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hMmax
62   have hyB : ∀ f ∈ B, MvPolynomial.eval y f = 0 := by
63     intro f hf
64     have hfM : f ∈ M := hJM (Ideal.mem_sup_left hf)
65     rw [hMy, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hfM
66     exact hfM
67   have hyi : y i = 0 := by
68     have hXM : MvPolynomial.X i ∈ M := by
69       apply hJM
70       exact Ideal.mem_sup_right (Ideal.mem_span_singleton_self _)
71     rw [hMy, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hXM
72     simpa using hXM
73   exact (coordinate_ne_zero_of_baseZero_of_support_disjoint
74     I i hdisjoint y hyB) hyi

76 /-- On every component above the contracted base ideal, axis avoidance gives
77 an explicit polynomial inverse for the distinguished coordinate. -/
78 theorem exists_componentCoordinate_polynomial_inverse
79     {k : Type u} [Field k] [IsAlgClosed k] {n : ℕ}
80     (I : RightIdeal (PresentedWeyl k n)) (i : Fin n)
81     (hdisjoint : Disjoint
82       (orderCharacteristicSupport k I)
83       (PrimeSpectrum.zeroLocus
84         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
85     (P : PrimeSpectrum (MvPolynomial (Fin n) k))
86     (hBP : reducedOrderBaseIdeal k I ≤ P.asIdeal) :
87     ∃ g : MvPolynomial (Fin n) k,
88       MvPolynomial.X i * g - 1 ∈ P.asIdeal := by
89   classical
90   let B := reducedOrderBaseIdeal k I
91   have htop := reducedOrderBaseIdeal_sup_coordinate_eq_top_of_support_disjoint
92     I i hdisjoint
93   have hone : (1 : MvPolynomial (Fin n) k) ∈
94       B ⊔ Ideal.span ({MvPolynomial.X i} : Set (MvPolynomial (Fin n) k)) := by
95     rw [htop]
96     exact Submodule.mem_top
97   obtain ⟨b, hb, z, hz, hbz⟩ := Submodule.mem_sup.mp hone
98   obtain ⟨g, hg⟩ := Ideal.mem_span_singleton.mp hz
99   refine ⟨g, ?_⟩
100   have hbP : b ∈ P.asIdeal := hBP hb
101   rw [← Ideal.Quotient.eq_zero_iff_mem]
102   rw [map_sub, map_mul, map_one]
103   have hsum := congrArg (Ideal.Quotient.mk P.asIdeal) hbz
104   rw [map_add, map_one, Ideal.Quotient.eq_zero_iff_mem.mpr hbP] at hsum
105   have hzq : Ideal.Quotient.mk P.asIdeal z =
106       Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i * g) := by
107     rw [hg]
108   rw [← map_mul, ← hzq]
109   apply sub_eq_zero.mpr
110   simpa using hsum

112 /-- If a nonunit parameter has a polynomial inverse in affine coordinates,
113 then the common denominator of an integral projective normalization of those
114 coordinates is a nonunit. -/
115 theorem normalized_denominator_nonunit_of_polynomial_inverse
116     {k : Type u} [Field k] {K : Type v} [Field K] [Algebra k K]
117     (V : ValuationSubring K) {n : ℕ}
118     (coeff : k →+* V)
119     (hcoeff : V.toSubring.subtype.comp coeff = algebraMap k K)
120     (x : Fin n → K) (qzero : V) (q : Fin n → V) (scale : K)
121     (hqzero : (qzero : K) = scale)
122     (hq : ∀ j, (q j : K) = scale * x j)
123     (i : Fin n) (parameter : V)
124     (hparameter : (parameter : K) = x i)
125     (hparameter_nonunit : ¬IsUnit parameter)
126     (g : MvPolynomial (Fin n) k)
127     (hinverse : x i * MvPolynomial.eval₂ (algebraMap k K) x g = 1) :
128     ¬IsUnit qzero := by
129   intro hqzero_unit
130   obtain ⟨u, hu⟩ := hqzero_unit
131   let xV : Fin n → V := fun j ↦ q j * (↑(u⁻¹) : V)
132   have hxV : ∀ j, (xV j : K) = x j := by
133     intro j
134     have huK : ((u : V) : K) = scale :=
135       (congrArg (fun z : V ↦ (z : K)) hu).trans hqzero
136     have huinvK : ((u : V) : K) * (((↑(u⁻¹) : V) : K)) = 1 := by
137       have huinvV : (u : V) * (↑(u⁻¹) : V) = 1 :=
138         (Units.mul_inv_eq_one).2 rfl
139       change ((((u : V) * (↑(u⁻¹) : V) : V) : K)) = 1
140       rw [huinvV]
141       exact map_one V.toSubring.subtype
142     change ((q j : V) : K) * (((↑(u⁻¹) : V) : K)) = x j
143     calc
144       ((q j : V) : K) * (((↑(u⁻¹) : V) : K)) =
145           (scale * x j) * (((↑(u⁻¹) : V) : K)) := by rw [hq j]
146       _ = x j * (((u : V) : K) * (((↑(u⁻¹) : V) : K))) := by
147         rw [huK]
148         ring
149       _ = x j := by rw [huinvK, mul_one]
150   let gV : V := MvPolynomial.eval₂Hom coeff xV g
151   have hgV : (gV : K) = MvPolynomial.eval₂ (algebraMap k K) x g := by
152     calc
153       (gV : K) = MvPolynomial.eval₂Hom
154           (V.toSubring.subtype.comp coeff) (fun j ↦ (xV j : K)) g :=
155         MvPolynomial.map_eval₂Hom coeff xV V.toSubring.subtype g
156       _ = MvPolynomial.eval₂Hom (algebraMap k K) x g := by
157         apply MvPolynomial.eval₂Hom_congr hcoeff
158         · funext j
159           exact hxV j
160         · rfl
161       _ = MvPolynomial.eval₂ (algebraMap k K) x g := rfl
162   apply hparameter_nonunit
163   rw [isUnit_iff_exists_inv]
164   refine ⟨gV, ?_⟩
165   apply Subtype.ext
166   change (parameter : K) * (gV : K) = (1 : V)
167   simpa [hparameter, hgV] using hinverse

169 /-- The denominator in the retained normalization of a component point is a
170 nonunit whenever the ambient support avoids the selected coordinate axis. -/
171 theorem normalizedComponentProjectivePoint_zero_nonunit
172     {k : Type u} [Field k] [IsAlgClosed k] {n : ℕ}
173     (I : RightIdeal (PresentedWeyl k n)) (i : Fin n)
174     (hdisjoint : Disjoint
175       (orderCharacteristicSupport k I)
176       (PrimeSpectrum.zeroLocus
177         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
178     (P : PrimeSpectrum (MvPolynomial (Fin n) k))
179     (hBP : reducedOrderBaseIdeal k I ≤ P.asIdeal)
180     (W : Data k (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal))
181       (componentCoordinate P i)) :
182     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
183         (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)) :=
184       W.ambientAlgebra
185     ∀ (q : Fin (n + 1) → W.place.valuation.toSubring)
186       (scale : FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)),
187       (∀ a, (q a : FractionRing
188           (MvPolynomial (Fin n) k ⧸ P.asIdeal)) =
189         scale * componentProjectivePoint P a) →
190       ¬IsUnit (q 0) := by
191   let F := FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)
192   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F :=
193     W.ambientAlgebra
194   intro q scale hq
195   letI : IsScalarTower W.coefficientField
196       (CoordinateZeroLocalRing W.coefficientField) F := W.coefficientTower
197   let V := W.place.valuation.toSubring
198   obtain ⟨g, hg⟩ := exists_componentCoordinate_polynomial_inverse
199     I i hdisjoint P hBP
200   let phi : MvPolynomial (Fin n) k →+* F :=
201     (algebraMap (MvPolynomial (Fin n) k ⧸ P.asIdeal) F).comp
202       (Ideal.Quotient.mk P.asIdeal)
203   have hphiC : phi.comp MvPolynomial.C = algebraMap k F := by
204     ext c
205     exact IsScalarTower.algebraMap_apply k
206       (MvPolynomial (Fin n) k ⧸ P.asIdeal) F c
207   have hphiX : ∀ j, phi (MvPolynomial.X j) = componentCoordinate P j := by
208     intro j
209     rfl
210   have hpoly : phi g = MvPolynomial.eval₂ (algebraMap k F)
211       (fun j ↦ componentCoordinate P j) g := by
212     rw [MvPolynomial.map_mvPolynomial_eq_eval₂ phi g]
213     change MvPolynomial.eval₂Hom (phi.comp MvPolynomial.C)
214         (fun j ↦ phi (MvPolynomial.X j)) g =
215       MvPolynomial.eval₂Hom (algebraMap k F)
216         (fun j ↦ componentCoordinate P j) g
217     apply MvPolynomial.eval₂Hom_congr hphiC
218     · funext j
219       exact hphiX j
220     · rfl
221   have hinverse : componentCoordinate P i *
222       MvPolynomial.eval₂ (algebraMap k F)
223         (fun j ↦ componentCoordinate P j) g = 1 := by
224     have hzero : phi (MvPolynomial.X i * g - 1) = 0 := by
225       have hmk : Ideal.Quotient.mk P.asIdeal
226           (MvPolynomial.X i * g - 1) = 0 :=
227         Ideal.Quotient.eq_zero_iff_mem.mpr hg
228       simpa [phi] using (congrArg
229         (algebraMap (MvPolynomial (Fin n) k ⧸ P.asIdeal) F) hmk)
230     rw [map_sub, map_mul, map_one, sub_eq_zero, hphiX, hpoly] at hzero
231     exact hzero
232   let coeff : k →+* V :=
233     (relativeCoefficientMap W.coefficientField W.place).comp
234       (algebraMap k W.coefficientField)
235   have hcoeff : W.place.valuation.toSubring.subtype.comp coeff =
236       algebraMap k F := by
237     ext c
238     change ((relativeCoefficientMap W.coefficientField W.place
239       (algebraMap k W.coefficientField c) : V) : F) = algebraMap k F c
240     calc
241       ((relativeCoefficientMap W.coefficientField W.place
242           (algebraMap k W.coefficientField c) : V) : F) =
243           algebraMap W.coefficientField F
244             (algebraMap k W.coefficientField c) :=
245         DFunLike.congr_fun
246           (relativeCoefficientMap_commutes W.coefficientField W.place)
247           (algebraMap k W.coefficientField c)
248       _ = algebraMap k F c :=
249         IsScalarTower.algebraMap_apply k W.coefficientField F c
250   apply normalized_denominator_nonunit_of_polynomial_inverse
251     (V := W.place.valuation) (coeff := coeff) hcoeff
252     (x := fun j ↦ componentCoordinate P j)
253     (qzero := q 0) (q := fun j ↦ q (Fin.succ j)) (scale := scale)
254     (i := i) (parameter := W.place.parameter) (g := g)
255   · simpa [componentProjectivePoint] using hq 0
256   · intro j
257     simpa [componentProjectivePoint] using hq (Fin.succ j)
258   · exact W.parameter_eq_coordinate
259   · exact W.place.parameter_nonunit
260   · exact hinverse

262 /-- The normalized homogeneous zeroth coordinate therefore vanishes in the
263 completed residue chart. -/
264 theorem normalizedComponentProjectivePoint_zero_vanish
265     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k] {n : ℕ}
266     (I : RightIdeal (PresentedWeyl k n)) (i : Fin n)
267     (hdisjoint : Disjoint
268       (orderCharacteristicSupport k I)
269       (PrimeSpectrum.zeroLocus
270         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
271     (P : PrimeSpectrum (MvPolynomial (Fin n) k))
272     (hBP : reducedOrderBaseIdeal k I ≤ P.asIdeal)
273     (W : Data k (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal))
274       (componentCoordinate P i)) :
275     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
276         (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)) :=
277       W.ambientAlgebra
278     let V := W.place.valuation.toSubring
279     letI : IsDiscreteValuationRing V := W.place.isDiscrete
280     letI : Algebra W.coefficientField V :=
281       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
282     ∀ (q : Fin (n + 1) → W.place.valuation.toSubring)
283       (scale : FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)),
284       (∀ a, (q a : FractionRing
285           (MvPolynomial (Fin n) k ⧸ P.asIdeal)) =
286         scale * componentProjectivePoint P a) →
287       PowerSeries.constantCoeff
288         (retainedToCompletedPowerSeries W (q 0)) = 0 := by
289   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
290       (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)) :=
291     W.ambientAlgebra
292   let V := W.place.valuation.toSubring
293   letI : IsDiscreteValuationRing V := W.place.isDiscrete
294   letI : Algebra W.coefficientField V :=
295     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
296   dsimp only
297   intro q scale hq
298   have hnonunit : ¬IsUnit (q 0) :=
299     normalizedComponentProjectivePoint_zero_nonunit
300       I i hdisjoint P hBP W q scale hq
301   exact retainedToCompletedPowerSeries_constantCoeff_eq_zero_of_nonunit W
302     (q 0) hnonunit
