11 /-!
12 # Vanishing of the normalized projective denominator
13 
14 Axis avoidance makes the distinguished affine coordinate a unit on every
15 component of the reduced base variety.  At the retained boundary place that
16 coordinate is a nonunit.  Consequently, a common projective normalization of
17 all affine coordinates cannot have unit homogeneous denominator.
18 -/

45 /-- Support avoidance forces the contracted reduced base ideal and the
46 distinguished coordinate to generate the unit ideal. -/
47 theorem reducedOrderBaseIdeal_sup_coordinate_eq_top_of_support_disjoint
48     {k : Type u} [Field k] [IsAlgClosed k] {n : ℕ}
49     (I : RightIdeal (WeylAlg k n)) (i : Fin n)
50     (hdisjoint : Disjoint
51       (orderCharacteristicSupport k I)
52       (PrimeSpectrum.zeroLocus
53         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n)))) :
54     reducedOrderBaseIdeal k I ⊔
55         Ideal.span ({MvPolynomial.X i} : Set (MvPolynomial (Fin n) k)) = ⊤ := by
56   classical
57   let B := reducedOrderBaseIdeal k I
58   let J := B ⊔ Ideal.span
59     ({MvPolynomial.X i} : Set (MvPolynomial (Fin n) k))
60   by_contra hJ
61   obtain ⟨M, hMmax, hJM⟩ := Ideal.exists_le_maximal J hJ
62   obtain ⟨y, hMy⟩ :=
63     MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hMmax
64   have hyB : ∀ f ∈ B, MvPolynomial.eval y f = 0 := by
65     intro f hf
66     have hfM : f ∈ M := hJM (Ideal.mem_sup_left hf)
67     rw [hMy, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hfM
68     exact hfM
69   have hyi : y i = 0 := by
70     have hXM : MvPolynomial.X i ∈ M := by
71       apply hJM
72       exact Ideal.mem_sup_right (Ideal.mem_span_singleton_self _)
73     rw [hMy, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hXM
74     simpa using hXM
75   exact (coordinate_ne_zero_of_baseZero_of_support_disjoint
76     I i hdisjoint y hyB) hyi

78 /-- On every component above the contracted base ideal, axis avoidance gives
79 an explicit polynomial inverse for the distinguished coordinate. -/
80 theorem exists_componentCoordinate_polynomial_inverse
81     {k : Type u} [Field k] [IsAlgClosed k] {n : ℕ}
82     (I : RightIdeal (WeylAlg k n)) (i : Fin n)
83     (hdisjoint : Disjoint
84       (orderCharacteristicSupport k I)
85       (PrimeSpectrum.zeroLocus
86         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
87     (P : PrimeSpectrum (MvPolynomial (Fin n) k))
88     (hBP : reducedOrderBaseIdeal k I ≤ P.asIdeal) :
89     ∃ g : MvPolynomial (Fin n) k,
90       MvPolynomial.X i * g - 1 ∈ P.asIdeal := by
91   classical
92   let B := reducedOrderBaseIdeal k I
93   have htop := reducedOrderBaseIdeal_sup_coordinate_eq_top_of_support_disjoint
94     I i hdisjoint
95   have hone : (1 : MvPolynomial (Fin n) k) ∈
96       B ⊔ Ideal.span ({MvPolynomial.X i} : Set (MvPolynomial (Fin n) k)) := by
97     rw [htop]
98     exact Submodule.mem_top
99   obtain ⟨b, hb, z, hz, hbz⟩ := Submodule.mem_sup.mp hone
100   obtain ⟨g, hg⟩ := Ideal.mem_span_singleton.mp hz
101   refine ⟨g, ?_⟩
102   have hbP : b ∈ P.asIdeal := hBP hb
103   rw [← Ideal.Quotient.eq_zero_iff_mem]
104   rw [map_sub, map_mul, map_one]
105   have hsum := congrArg (Ideal.Quotient.mk P.asIdeal) hbz
106   rw [map_add, map_one, Ideal.Quotient.eq_zero_iff_mem.mpr hbP] at hsum
107   have hzq : Ideal.Quotient.mk P.asIdeal z =
108       Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i * g) := by
109     rw [hg]
110   rw [← map_mul, ← hzq]
111   apply sub_eq_zero.mpr
112   simpa using hsum

114 /-- If a nonunit parameter has a polynomial inverse in affine coordinates,
115 then the common denominator of an integral projective normalization of those
116 coordinates is a nonunit. -/
117 theorem normalized_denominator_nonunit_of_polynomial_inverse
118     {k : Type u} [Field k] {K : Type v} [Field K] [Algebra k K]
119     (V : ValuationSubring K) {n : ℕ}
120     (coeff : k →+* V)
121     (hcoeff : V.toSubring.subtype.comp coeff = algebraMap k K)
122     (x : Fin n → K) (qzero : V) (q : Fin n → V) (scale : K)
123     (hqzero : (qzero : K) = scale)
124     (hq : ∀ j, (q j : K) = scale * x j)
125     (i : Fin n) (parameter : V)
126     (hparameter : (parameter : K) = x i)
127     (hparameter_nonunit : ¬IsUnit parameter)
128     (g : MvPolynomial (Fin n) k)
129     (hinverse : x i * MvPolynomial.eval₂ (algebraMap k K) x g = 1) :
130     ¬IsUnit qzero := by
131   intro hqzero_unit
132   obtain ⟨u, hu⟩ := hqzero_unit
133   let xV : Fin n → V := fun j ↦ q j * (↑(u⁻¹) : V)
134   have hxV : ∀ j, (xV j : K) = x j := by
135     intro j
136     have huK : ((u : V) : K) = scale :=
137       (congrArg (fun z : V ↦ (z : K)) hu).trans hqzero
138     have huinvK : ((u : V) : K) * (((↑(u⁻¹) : V) : K)) = 1 := by
139       have huinvV : (u : V) * (↑(u⁻¹) : V) = 1 :=
140         (Units.mul_inv_eq_one).2 rfl
141       change ((((u : V) * (↑(u⁻¹) : V) : V) : K)) = 1
142       rw [huinvV]
143       exact map_one V.toSubring.subtype
144     change ((q j : V) : K) * (((↑(u⁻¹) : V) : K)) = x j
145     calc
146       ((q j : V) : K) * (((↑(u⁻¹) : V) : K)) =
147           (scale * x j) * (((↑(u⁻¹) : V) : K)) := by rw [hq j]
148       _ = x j * (((u : V) : K) * (((↑(u⁻¹) : V) : K))) := by
149         rw [huK]
150         ring
151       _ = x j := by rw [huinvK, mul_one]
152   let gV : V := MvPolynomial.eval₂Hom coeff xV g
153   have hgV : (gV : K) = MvPolynomial.eval₂ (algebraMap k K) x g := by
154     calc
155       (gV : K) = MvPolynomial.eval₂Hom
156           (V.toSubring.subtype.comp coeff) (fun j ↦ (xV j : K)) g :=
157         MvPolynomial.map_eval₂Hom coeff xV V.toSubring.subtype g
158       _ = MvPolynomial.eval₂Hom (algebraMap k K) x g := by
159         apply MvPolynomial.eval₂Hom_congr hcoeff
160         · funext j
161           exact hxV j
162         · rfl
163       _ = MvPolynomial.eval₂ (algebraMap k K) x g := rfl
164   apply hparameter_nonunit
165   rw [isUnit_iff_exists_inv]
166   refine ⟨gV, ?_⟩
167   apply Subtype.ext
168   change (parameter : K) * (gV : K) = (1 : V)
169   simpa [hparameter, hgV] using hinverse

171 /-- The denominator in the retained normalization of a component point is a
172 nonunit whenever the ambient support avoids the selected coordinate axis. -/
173 theorem normalizedComponentProjectivePoint_zero_nonunit
174     {k : Type u} [Field k] [IsAlgClosed k] {n : ℕ}
175     (I : RightIdeal (WeylAlg k n)) (i : Fin n)
176     (hdisjoint : Disjoint
177       (orderCharacteristicSupport k I)
178       (PrimeSpectrum.zeroLocus
179         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
180     (P : PrimeSpectrum (MvPolynomial (Fin n) k))
181     (hBP : reducedOrderBaseIdeal k I ≤ P.asIdeal)
182     (W : Data k (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal))
183       (componentCoordinate P i)) :
184     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
185         (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)) :=
186       W.ambientAlgebra
187     ∀ (q : Fin (n + 1) → W.place.valuation.toSubring)
188       (scale : FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)),
189       (∀ a, (q a : FractionRing
190           (MvPolynomial (Fin n) k ⧸ P.asIdeal)) =
191         scale * componentProjectivePoint P a) →
192       ¬IsUnit (q 0) := by
193   let F := FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)
194   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F :=
195     W.ambientAlgebra
196   intro q scale hq
197   letI : IsScalarTower W.coefficientField
198       (CoordinateZeroLocalRing W.coefficientField) F := W.coefficientTower
199   let V := W.place.valuation.toSubring
200   obtain ⟨g, hg⟩ := exists_componentCoordinate_polynomial_inverse
201     I i hdisjoint P hBP
202   let phi : MvPolynomial (Fin n) k →+* F :=
203     (algebraMap (MvPolynomial (Fin n) k ⧸ P.asIdeal) F).comp
204       (Ideal.Quotient.mk P.asIdeal)
205   have hphiC : phi.comp MvPolynomial.C = algebraMap k F := by
206     ext c
207     exact IsScalarTower.algebraMap_apply k
208       (MvPolynomial (Fin n) k ⧸ P.asIdeal) F c
209   have hphiX : ∀ j, phi (MvPolynomial.X j) = componentCoordinate P j := by
210     intro j
211     rfl
212   have hpoly : phi g = MvPolynomial.eval₂ (algebraMap k F)
213       (fun j ↦ componentCoordinate P j) g := by
214     rw [MvPolynomial.map_mvPolynomial_eq_eval₂ phi g]
215     change MvPolynomial.eval₂Hom (phi.comp MvPolynomial.C)
216         (fun j ↦ phi (MvPolynomial.X j)) g =
217       MvPolynomial.eval₂Hom (algebraMap k F)
218         (fun j ↦ componentCoordinate P j) g
219     apply MvPolynomial.eval₂Hom_congr hphiC
220     · funext j
221       exact hphiX j
222     · rfl
223   have hinverse : componentCoordinate P i *
224       MvPolynomial.eval₂ (algebraMap k F)
225         (fun j ↦ componentCoordinate P j) g = 1 := by
226     have hzero : phi (MvPolynomial.X i * g - 1) = 0 := by
227       have hmk : Ideal.Quotient.mk P.asIdeal
228           (MvPolynomial.X i * g - 1) = 0 :=
229         Ideal.Quotient.eq_zero_iff_mem.mpr hg
230       simpa [phi] using (congrArg
231         (algebraMap (MvPolynomial (Fin n) k ⧸ P.asIdeal) F) hmk)
232     rw [map_sub, map_mul, map_one, sub_eq_zero, hphiX, hpoly] at hzero
233     exact hzero
234   let coeff : k →+* V :=
235     (relativeCoefficientMap W.coefficientField W.place).comp
236       (algebraMap k W.coefficientField)
237   have hcoeff : W.place.valuation.toSubring.subtype.comp coeff =
238       algebraMap k F := by
239     ext c
240     change ((relativeCoefficientMap W.coefficientField W.place
241       (algebraMap k W.coefficientField c) : V) : F) = algebraMap k F c
242     calc
243       ((relativeCoefficientMap W.coefficientField W.place
244           (algebraMap k W.coefficientField c) : V) : F) =
245           algebraMap W.coefficientField F
246             (algebraMap k W.coefficientField c) :=
247         DFunLike.congr_fun
248           (relativeCoefficientMap_commutes W.coefficientField W.place)
249           (algebraMap k W.coefficientField c)
250       _ = algebraMap k F c :=
251         IsScalarTower.algebraMap_apply k W.coefficientField F c
252   apply normalized_denominator_nonunit_of_polynomial_inverse
253     (V := W.place.valuation) (coeff := coeff) hcoeff
254     (x := fun j ↦ componentCoordinate P j)
255     (qzero := q 0) (q := fun j ↦ q (Fin.succ j)) (scale := scale)
256     (i := i) (parameter := W.place.parameter) (g := g)
257   · simpa [componentProjectivePoint] using hq 0
258   · intro j
259     simpa [componentProjectivePoint] using hq (Fin.succ j)
260   · exact W.parameter_eq_coordinate
261   · exact W.place.parameter_nonunit
262   · exact hinverse

264 /-- The normalized homogeneous zeroth coordinate therefore vanishes in the
265 completed residue chart. -/
266 theorem normalizedComponentProjectivePoint_zero_vanish
267     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k] {n : ℕ}
268     (I : RightIdeal (WeylAlg k n)) (i : Fin n)
269     (hdisjoint : Disjoint
270       (orderCharacteristicSupport k I)
271       (PrimeSpectrum.zeroLocus
272         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
273     (P : PrimeSpectrum (MvPolynomial (Fin n) k))
274     (hBP : reducedOrderBaseIdeal k I ≤ P.asIdeal)
275     (W : Data k (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal))
276       (componentCoordinate P i)) :
277     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
278         (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)) :=
279       W.ambientAlgebra
280     let V := W.place.valuation.toSubring
281     letI : IsDiscreteValuationRing V := W.place.isDiscrete
282     letI : Algebra W.coefficientField V :=
283       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
284     ∀ (q : Fin (n + 1) → W.place.valuation.toSubring)
285       (scale : FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)),
286       (∀ a, (q a : FractionRing
287           (MvPolynomial (Fin n) k ⧸ P.asIdeal)) =
288         scale * componentProjectivePoint P a) →
289       PowerSeries.constantCoeff
290         (retainedToCompletedPowerSeries W (q 0)) = 0 := by
291   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
292       (FractionRing (MvPolynomial (Fin n) k ⧸ P.asIdeal)) :=
293     W.ambientAlgebra
294   let V := W.place.valuation.toSubring
295   letI : IsDiscreteValuationRing V := W.place.isDiscrete
296   letI : Algebra W.coefficientField V :=
297     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
298   dsimp only
299   intro q scale hq
300   have hnonunit : ¬IsUnit (q 0) :=
301     normalizedComponentProjectivePoint_zero_nonunit
302       I i hdisjoint P hBP W q scale hq
303   exact retainedToCompletedPowerSeries_constantCoeff_eq_zero_of_nonunit W
304     (q 0) hnonunit
