64 theorem differentialCovector_mul (y : Fin m → F) (p q : MvPolynomial (Fin m) F) :
65     differentialCovector y (p * q) =
66       MvPolynomial.eval y q • differentialCovector y p +
67         MvPolynomial.eval y p • differentialCovector y q :=

70 theorem differentialCovector_add (y : Fin m → F) (p q : MvPolynomial (Fin m) F) :
71     differentialCovector y (p + q) = differentialCovector y p + differentialCovector y q := by
72   apply LinearMap.ext
73   intro v
74   change (∑ i, differentialAt y (p + q) i * v i) =
75     (∑ i, differentialAt y p i * v i) + ∑ i, differentialAt y q i * v i
76   simp only [differentialAt, map_add, add_mul, Finset.sum_add_distrib]

78 theorem differentialCovector_zero (y : Fin m → F) :
79     differentialCovector y (0 : MvPolynomial (Fin m) F) = 0 := by
80   apply LinearMap.ext
81   intro v
82   change (∑ i, differentialAt y 0 i * v i) = 0
83   simp [differentialAt]

85 /-- For a radical ideal `I` and a minimal prime `P` of `I`, every element of
86 `P` becomes an element of `I` after multiplication by an element outside `P`.
87 The multiplier is a product of elements of the other minimal primes. -/
88 theorem exists_mul_mem_of_mem_minimalPrimes_of_isRadical
89     {R : Type*} [CommRing R] [IsNoetherianRing R]
90     {I P : Ideal R} (hI : I.IsRadical) (hP : P ∈ I.minimalPrimes)
91     {f : R} (hf : f ∈ P) : ∃ s ∉ P, s * f ∈ I := by
92   classical
93   have hfin := Ideal.finite_minimalPrimes_of_isNoetherianRing R I
94   set T := hfin.toFinset.erase P with hT
95   have hchoice : ∀ Q ∈ T, ∃ s ∈ Q, s ∉ P := by
96     intro Q hQ
97     rw [hT, Finset.mem_erase, hfin.mem_toFinset] at hQ
98     by_contra hcon
99     push_neg at hcon
100     exact hQ.1 (le_antisymm hcon (hP.2 hQ.2.1 hcon))
101   choose s hsQ hsP using hchoice
102   refine ⟨∏ Q ∈ T.attach, s Q.1 Q.2, ?_, ?_⟩
103   · intro h
104     obtain ⟨Q, -, hQ⟩ := (hP.1.1.prod_mem_iff).1 h
105     exact hsP Q.1 Q.2 hQ
106   · rw [← hI.radical, ← Ideal.sInf_minimalPrimes, Ideal.mem_sInf]
107     intro Q hQ
108     by_cases hQP : Q = P
109     · subst hQP
110       exact Ideal.mul_mem_left _ _ hf
111     · have hQT : Q ∈ T := by
112         rw [hT, Finset.mem_erase, hfin.mem_toFinset]
113         exact ⟨hQP, hQ⟩
114       refine Ideal.mul_mem_right _ _ ?_
115       exact Ideal.mem_of_dvd _
116         (Finset.dvd_prod_of_mem (fun Q : T ↦ s Q.1 Q.2) (Finset.mem_attach _ ⟨Q, hQT⟩))
117         (hsQ Q hQT)

119 /-- At a point whose vanishing ideal is exactly the minimal prime `P` of `I`,
120 the equation-defined conormal space of `P` is contained in that of `I`. -/
121 theorem affineConormalSpace_map_minimalPrime_le [Algebra k F]
122     (I P : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical) (hP : P ∈ I.minimalPrimes)
123     (y : Fin m → F)
124     (hy : ∀ f, MvPolynomial.eval y (MvPolynomial.map (algebraMap k F) f) = 0 ↔ f ∈ P) :
125     affineConormalSpace y (P.map (MvPolynomial.map (algebraMap k F))) ≤
126       affineConormalSpace y (I.map (MvPolynomial.map (algebraMap k F))) := by
127   classical
128   rw [affineConormalSpace_eq_equationCovectorSpan, affineConormalSpace_eq_equationCovectorSpan]
129   set S := equationCovectorSpan y (I.map (MvPolynomial.map (algebraMap k F)))
130   refine Submodule.span_le.2 ?_
131   rintro _ ⟨⟨f', hf'⟩, rfl⟩
132   show differentialCovector y f' ∈ S
133   have key : MvPolynomial.eval y f' = 0 ∧ differentialCovector y f' ∈ S := by
134     refine Submodule.span_induction (p := fun x _ ↦
135       MvPolynomial.eval y x = 0 ∧ differentialCovector y x ∈ S) ?_ ?_ ?_ ?_ hf'
136     · rintro _ ⟨f, hf, rfl⟩
137       refine ⟨(hy f).2 hf, ?_⟩
138       obtain ⟨g, hgP, hfg⟩ := exists_mul_mem_of_mem_minimalPrimes_of_isRadical hI hP hf
139       have hg : MvPolynomial.eval y (MvPolynomial.map (algebraMap k F) g) ≠ 0 :=
140         fun h ↦ hgP ((hy g).1 h)
141       have hprod : differentialCovector y
142           (MvPolynomial.map (algebraMap k F) (g * f)) ∈ S :=
143         Submodule.subset_span ⟨⟨_, Ideal.mem_map_of_mem _ hfg⟩, rfl⟩
144       rw [map_mul, differentialCovector_mul, (hy f).2 hf, zero_smul, zero_add] at hprod
145       have := S.smul_mem (MvPolynomial.eval y (MvPolynomial.map (algebraMap k F) g))⁻¹ hprod
146       rwa [smul_smul, inv_mul_cancel₀ hg, one_smul] at this
147     · exact ⟨by simp, by rw [differentialCovector_zero]; exact S.zero_mem⟩
148     · rintro x y' _ _ ⟨hx0, hx⟩ ⟨hy0, hy'⟩
149       exact ⟨by rw [map_add, hx0, hy0, add_zero],
150         by rw [differentialCovector_add]; exact S.add_mem hx hy'⟩
151     · rintro p x _ ⟨hx0, hx⟩
152       refine ⟨by rw [smul_eq_mul, map_mul, hx0, mul_zero], ?_⟩
153       rw [smul_eq_mul, differentialCovector_mul, hx0, zero_smul, zero_add]
154       exact S.smul_mem _ hx
155   exact key.2
156

159 /-- A retained boundary place on a component together with a normalized
160 projective column and a visible divisor frame attached to it. -/
161 def HasVisibleDivisorFrame
162     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (hm : 0 < m)
163     [CharZero k] : Prop :=
164   ∃ W : Data k (ComponentFractionField P) (componentCoordinate P ⟨0, hm⟩),
165     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
166         (ComponentFractionField P) := W.ambientAlgebra
167     let V := W.place.valuation.toSubring
168     letI : IsDiscreteValuationRing V := W.place.isDiscrete
169     ∃ (q : Fin (m + 1) → V) (scale : ComponentFractionField P),
170       q 0 ≠ 0 ∧
171       (∀ a, (q a : ComponentFractionField P) =
172         scale * componentProjectivePoint P a) ∧
173       q (Fin.succ ⟨0, hm⟩) = q 0 * W.place.parameter ∧
174       ∃ D : VisibleDivisorFrame (V := V)
175         (KaehlerDifferential.D k (ComponentFractionField P)) (Fin m),
176         D.Q₀ = q 0 ∧ D.Q₁ = q (Fin.succ ⟨0, hm⟩) ∧ ∀ j, D.Q j = q (Fin.succ j)

178 /-- The generic-point bridge instantiated on a component: its statement is
179 the `hbridge` hypothesis of the retained-place transport. -/
180 theorem component_kaehler_bridge (P : PrimeSpectrum (MvPolynomial (Fin m) k))
181     (xi : Fin m → ComponentFractionField P)
182     (h : ∑ j, xi j • KaehlerDifferential.D k (ComponentFractionField P)
183       (componentCoordinate P j) = 0) :
184     Stafford38.Geometry.AffineConormalSpan.coordinateCovector xi ∈
185       Stafford38.Geometry.AffineConormalSpan.affineConormalSpace (componentCoordinate P)
186         (P.asIdeal.map (MvPolynomial.map
187           (algebraMap k (ComponentFractionField P)))) := by
188   haveI : P.asIdeal.IsPrime := P.isPrime
189   exact coordinateCovector_mem_affineConormalSpace_of_kaehler_sum_eq_zero
190     (I := P.asIdeal) (F := ComponentFractionField P) xi h

192 /-- A visible divisor frame on a component yields a finite-gradient boundary
193 certificate over the residue field of its place, for the radical ideal of
194 which the component is a minimal prime. -/
195 theorem exists_finiteGradientBoundaryCertificateOver_of_hasVisibleDivisorFrame
196     [CharZero k]
197     (I : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
198     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (hP : P.asIdeal ∈ I.minimalPrimes)
199     (hm : 0 < m) (h : HasVisibleDivisorFrame P hm) :
200     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K),
201       Nonempty (FiniteGradientBoundaryCertificateOver (k := k) (K := K) m hm I) := by
202   classical
203   obtain ⟨W, q, scale, hq0, hq, hratio, D, hQ₀, hQ₁, hQ⟩ := h
204   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
205       (ComponentFractionField P) := W.ambientAlgebra
206   let V := W.place.valuation.toSubring
207   letI : IsDiscreteValuationRing V := W.place.isDiscrete
208   letI : Algebra W.coefficientField V :=
209     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
210   letI : Algebra k (ResidueField V) := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
211   -- The generic point kills exactly the minimal prime.
212   have hy : ∀ f, MvPolynomial.eval (componentCoordinate P)
213       (MvPolynomial.map (algebraMap k (ComponentFractionField P)) f) = 0 ↔ f ∈ P.asIdeal := by
214     intro f
215     rw [MvPolynomial.eval_map, ← componentAffineGenericPointMap_eq_eval₂,
216       ← RingHom.mem_ker, componentAffineGenericPointMap_ker]
217   have hbridge : ∀ xi : Fin m → ComponentFractionField P,
218       ∑ j, xi j • KaehlerDifferential.D k (ComponentFractionField P)
219         (componentCoordinate P j) = 0 →
220       Stafford38.Geometry.AffineConormalSpan.coordinateCovector xi ∈
221         Stafford38.Geometry.AffineConormalSpan.affineConormalSpace (componentCoordinate P)
222           (I.map (MvPolynomial.map (algebraMap k (ComponentFractionField P)))) :=
223     fun xi hxi ↦ affineConormalSpace_map_minimalPrime_le I P.asIdeal hI hP _ hy
224       (component_kaehler_bridge P xi hxi)
225   have hq0F : (q 0 : ComponentFractionField P) ≠ 0 := fun h ↦ hq0 (Subtype.ext h)
226   have hIy : ∀ g ∈ I,
227       MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
228         (componentCoordinate P) g = 0 := by
229     intro g hg
230     rw [← componentAffineGenericPointMap_eq_eval₂]
231     exact componentAffineGenericPointMap_eq_zero_of_mem P (hP.1.2 hg)
232   obtain ⟨R⟩ := regularizedOneRowConormalData_of_transport hm I (componentCoordinate P)
233     hIy Subtype.val_injective (retainedToCompletedPowerSeries W)
234     (retainedToCompletedPowerSeries_injective W)
235     (retainedToCompletedPowerSeries_constantCoeff_eq_zero_of_nonunit W)
236     (retainedLaurentLift P ⟨0, hm⟩ W)
237     (retainedLaurentLift_algebraMap P ⟨0, hm⟩ W)
238     (retainedLaurentLift_comp_algebraMap P ⟨0, hm⟩ W)
239     q hq0 (componentCoordinate_eq_div P (fun a ↦ (q a : ComponentFractionField P))
240       scale hq hq0F)
241     W.place.parameter hratio D hQ₀ hQ₁ hQ hbridge
242   exact ⟨ResidueField V, inferInstance, inferInstance,
243     finiteGradientBoundaryCertificateOver_of_regularizedOneRowConormalData hm I _ R⟩

245 /-- The reduced lane C input: every relevant transcendental component carries
246 a retained place with a visible divisor frame. -/
247 def HigherDimensionalCanonicalVisibleDivisorFrameProduction : Prop :=
248   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
249     (n N : ℕ) (d : WeylAlg k (n + 1)),
250     0 < n →
251     Disjoint
252       (orderCharacteristicSupport k
253         (canonicalRightIdeal (presentedCoordinate k n) d N))
254       (PrimeSpectrum.zeroLocus
255         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
256           Set (SymbolRing k (n + 1)))) →
257     ∀ P : PrimeSpectrum (MvPolynomial (Fin (n + 1)) k),
258       P.asIdeal ∈
259         (reducedOrderBaseIdeal k
260           (canonicalRightIdeal (presentedCoordinate k n) d N)).minimalPrimes →
261       Transcendental k (componentCoordinate P ⟨0, Nat.zero_lt_succ n⟩) →
262       HasVisibleDivisorFrame P (Nat.zero_lt_succ n)
263 
264 /-- The visible-frame producer discharges the finite-gradient interface. -/
265 theorem higherDimensionalCanonicalResidueExtensionNonconstantFiniteGradientProduction_of_visibleDivisorFrame
266     (h : HigherDimensionalCanonicalVisibleDivisorFrameProduction.{u}) :
267     HigherDimensionalCanonicalResidueExtensionNonconstantFiniteGradientProduction.{u} := by
268   intro k _ _ _ n N d hn hdisjoint P hP htrans
269   exact exists_finiteGradientBoundaryCertificateOver_of_hasVisibleDivisorFrame _
270     (reducedOrderBaseIdeal_isRadical k (canonicalRightIdeal (presentedCoordinate k n) d N))
271     P hP (Nat.zero_lt_succ n) (h k n N d hn hdisjoint P hP htrans)
