10 /-- Polynomial evaluation follows the same common-open map and the same arc.
11 This transports the nonvanishing numerator used to select the smooth locus. -/
12 theorem eval_map_eq_arc_eval₂
13     {k Q E L σ : Type*} [CommSemiring k] [CommSemiring Q]
14     [CommSemiring E] [CommSemiring L] [Algebra k Q] [Algebra k L]
15     (g : Q →+* E) (rho : E →+* L)
16     (hground : (rho.comp g).comp (algebraMap k Q) = algebraMap k L)
17     (qQ : σ → Q) (qC : σ → E) (qL : σ → L)
18     (hcoords : ∀ i, qC i = g (qQ i))
19     (hposition : ∀ i, rho (qC i) = qL i)
20     (p : MvPolynomial σ k) :
21     MvPolynomial.eval qL (MvPolynomial.map (algebraMap k L) p) =
22       rho (g (MvPolynomial.eval₂ (algebraMap k Q) qQ p)) := by
23   rw [MvPolynomial.eval_map]
24   have hcolumns : (rho.comp g) ∘ qQ = qL := by
25     funext i
26     change rho (g (qQ i)) = qL i
27     rw [← hcoords i]
28     exact hposition i
29   have h := MvPolynomial.eval₂_comp_left (rho.comp g) (algebraMap k Q) qQ p
30   rw [hground, hcolumns] at h
31   exact h.symm
32 
33 end Stafford38.Geometry.CommonOpenEvaluation
