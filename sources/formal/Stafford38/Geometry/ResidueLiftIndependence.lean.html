11 /-- Algebraic independence survives lifting elements from a residue field into
12 an injectively embedded local domain. A polynomial relation among the lifts
13 would reduce to the same relation among their residues. -/
14 theorem algebraicIndependent_of_residue
15     {k V K ι : Type*} [Field k] [CommRing V] [IsLocalRing V]
16     [Field K] [Algebra k V] [Algebra k K] [Algebra V K]
17     [IsScalarTower k V K] [Algebra k (IsLocalRing.ResidueField V)]
18     [IsScalarTower k V (IsLocalRing.ResidueField V)]
19     (hinj : Function.Injective (algebraMap V K))
20     (x : ι → V)
21     (hx : AlgebraicIndependent k (fun i ↦ IsLocalRing.residue V (x i))) :
22     AlgebraicIndependent k (fun i ↦ algebraMap V K (x i)) := by
23   rw [algebraicIndependent_iff]
24   intro p hp
25   have hcompK :
26       (algebraMap V K) (MvPolynomial.aeval x p) =
27         MvPolynomial.aeval (fun i ↦ algebraMap V K (x i)) p := by
28     have hcoeff : (algebraMap V K).comp (algebraMap k V) = algebraMap k K := by
29       ext c
30       exact (IsScalarTower.algebraMap_apply k V K c).symm
31     rw [MvPolynomial.map_aeval, MvPolynomial.aeval_eq_eval₂Hom, hcoeff]
32   have hzeroV : MvPolynomial.aeval x p = 0 := by
33     apply hinj
34     rw [map_zero, hcompK]
35     exact hp
36   have hcompκ : IsLocalRing.residue V (MvPolynomial.aeval x p) =
37       MvPolynomial.aeval (fun i ↦ IsLocalRing.residue V (x i)) p := by
38     have hcoeff : (IsLocalRing.residue V).comp (algebraMap k V) =
39         algebraMap k (IsLocalRing.ResidueField V) := by
40       ext c
41       exact (IsScalarTower.algebraMap_apply k V (IsLocalRing.ResidueField V) c).symm
42     rw [MvPolynomial.map_aeval, MvPolynomial.aeval_eq_eval₂Hom, hcoeff]
43   have hzeroκ : MvPolynomial.aeval (fun i ↦ IsLocalRing.residue V (x i)) p = 0 := by
44     rw [← hcompκ, hzeroV, map_zero]
45   exact hx.eq_zero_of_aeval_eq_zero p hzeroκ

48 end Stafford38.Geometry.ResidueLiftIndependence
