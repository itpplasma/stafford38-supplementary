61 /-! ## Local consumer theorem -/
62 
63 /--
64 The completed-DVR/projective certificate supplies the Laurent conormal axis.
65 
66 The proof has four explicit interfaces: the projective order-gap theorem
67 constructs the formal annihilating row; homogeneous equation vanishing gives
68 the base-equation condition after dehomogenization; tangent inclusion feeds
69 the weaker Laurent conormal bridge; and the axis equation identifies the
70 regular fibre residue with the pure first momentum direction.
71 -/
72 theorem exists_conormalAxis_of_completedProjectiveBoundaryChart
73     {k : Type u} [Field k] [CharZero k]
74     {m : ℕ} (hm : 0 < m)
75     (I : Ideal (MvPolynomial (Fin m) k))
76     (W : CompletedProjectiveBoundaryChart k m hm I) :
77     ∃ (y : Fin m → LaurentSeries k)
78       (xi : Fin m → PowerSeries k),
79       Sum.elim y
80           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈
81         equationConormalLocus
82           (I.map (scalarPolynomialMap
83             (k := k) (K := LaurentSeries k) (Fin m))) ∧
84       residueColumn xi =
85         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
86   simpa [Stafford38.Geometry.LaurentConormalResidueExtension.groundEquationConormalLocus,
87     Stafford38.Geometry.ScalarExtensionPoints.scalarPolynomialMap,
88     Stafford38.Geometry.ScalarExtensionPoints.scalarPolynomialMap] using
89     Stafford38.Geometry.LaurentConormalResidueExtension.exists_conormalAxis_of_completedProjectiveBoundaryChartOver
90       (k := k) (K := k) hm I W

92 /-! ## Global chart-production interface -/
93 
94 /-- Global production of a completed chart for the canonical reduced base
95 ideal. The chart stores the geometric construction and tangent inclusion;
96 it has no support-vanishing conclusion as a field. -/
97 def CanonicalBoundaryChartProduction : Prop :=
98   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
99     (n N : ℕ)
100     (d : WeylAlg k (n + 1)),
101     0 < N →
102     IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
103     Disjoint
104       (orderCharacteristicSupport k
105         (canonicalRightIdeal (presentedCoordinate k n) d N))
106       (PrimeSpectrum.zeroLocus
107         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
108           Set (SymbolRing k (n + 1)))) →
109     (orderCharacteristicSupport k
110       (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty →
111     Nonempty (CompletedProjectiveBoundaryChart k (n + 1) (Nat.zero_lt_succ n)
112       (reducedOrderBaseIdeal k
113         (canonicalRightIdeal (presentedCoordinate k n) d N)))
114 
115 /--
116 The exact adapter from global boundary-chart production to the canonical
117 Laurent producer.  All support hypotheses are used only to request a chart;
118 the chart-to-conormal proof itself has no support-vanishing premise.
119 -/
120 theorem canonicalAsymptoticLaurentProducer_of_boundaryChartProduction
121     (hproduction : CanonicalBoundaryChartProduction.{u}) :
122     CanonicalAsymptoticLaurentProducer.{u} := by
123   intro k _ _ _ n N d hN hd hdisjoint hnonempty
124   obtain ⟨W⟩ :=
125     hproduction k n N d hN hd hdisjoint hnonempty
126   obtain ⟨y, xi, hmem, haxis⟩ :=
127     exists_conormalAxis_of_completedProjectiveBoundaryChart
128       (k := k) (m := n + 1) (Nat.zero_lt_succ n)
129       (reducedOrderBaseIdeal k
130         (canonicalRightIdeal (presentedCoordinate k n) d N)) W
131   exact ⟨y, xi, hmem, haxis⟩

136 end
