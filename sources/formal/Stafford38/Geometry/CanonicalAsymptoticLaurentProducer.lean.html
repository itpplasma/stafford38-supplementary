59 /-! ## Local consumer theorem -/
60 
61 /--
62 The completed-DVR/projective certificate supplies the Laurent conormal axis.
63 
64 The proof has four explicit interfaces: the projective order-gap theorem
65 constructs the formal annihilating row; homogeneous equation vanishing gives
66 the base-equation condition after dehomogenization; tangent inclusion feeds
67 the weaker Laurent conormal bridge; and the axis equation identifies the
68 regular fibre residue with the pure first momentum direction.
69 -/
70 theorem exists_conormalAxis_of_completedProjectiveBoundaryChart
71     {k : Type u} [Field k] [CharZero k]
72     {m : ℕ} (hm : 0 < m)
73     (I : Ideal (MvPolynomial (Fin m) k))
74     (W : CompletedProjectiveBoundaryChart k m hm I) :
75     ∃ (y : Fin m → LaurentSeries k)
76       (xi : Fin m → PowerSeries k),
77       Sum.elim y
78           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈
79         equationConormalLocus
80           (I.map (scalarPolynomialMap
81             (k := k) (K := LaurentSeries k) (Fin m))) ∧
82       residueColumn xi =
83         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
84   simpa [Stafford38.Geometry.LaurentConormalResidueExtension.groundEquationConormalLocus,
85     Stafford38.Geometry.LaurentConormalResidueExtension.groundPolynomialMap,
86     Stafford38.Geometry.ScalarExtensionPoints.scalarPolynomialMap] using
87     Stafford38.Geometry.LaurentConormalResidueExtension.exists_conormalAxis_of_completedProjectiveBoundaryChartOver
88       (k := k) (K := k) hm I W

90 /-! ## Global chart-production interface -/
91 
92 /-- Global production of a completed chart for the canonical reduced base
93 ideal. The chart stores the geometric construction and tangent inclusion;
94 it has no support-vanishing conclusion as a field. -/
95 def CanonicalBoundaryChartProduction : Prop :=
96   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
97     (n N : ℕ)
98     (d : PresentedWeyl k (n + 1)),
99     0 < N →
100     IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
101     Disjoint
102       (orderCharacteristicSupport k
103         (canonicalRightIdeal (presentedCoordinate k n) d N))
104       (PrimeSpectrum.zeroLocus
105         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
106           Set (SymbolRing k (n + 1)))) →
107     (orderCharacteristicSupport k
108       (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty →
109     Nonempty (CompletedProjectiveBoundaryChart k (n + 1) (Nat.zero_lt_succ n)
110       (reducedOrderBaseIdeal k
111         (canonicalRightIdeal (presentedCoordinate k n) d N)))
112 
113 /--
114 The exact adapter from global boundary-chart production to the canonical
115 Laurent producer.  All support hypotheses are used only to request a chart;
116 the chart-to-conormal proof itself has no support-vanishing premise.
117 -/
118 theorem canonicalAsymptoticLaurentProducer_of_boundaryChartProduction
119     (hproduction : CanonicalBoundaryChartProduction.{u}) :
120     CanonicalAsymptoticLaurentProducer.{u} := by
121   intro k _ _ _ n N d hN hd hdisjoint hnonempty
122   obtain ⟨W⟩ :=
123     hproduction k n N d hN hd hdisjoint hnonempty
124   obtain ⟨y, xi, hmem, haxis⟩ :=
125     exists_conormalAxis_of_completedProjectiveBoundaryChart
126       (k := k) (m := n + 1) (Nat.zero_lt_succ n)
127       (reducedOrderBaseIdeal k
128         (canonicalRightIdeal (presentedCoordinate k n) d N)) W
129   exact ⟨y, xi, hmem, haxis⟩

134 end
