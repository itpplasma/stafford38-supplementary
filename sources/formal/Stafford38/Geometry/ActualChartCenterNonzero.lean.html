31 /-- The zeroth projective coordinate is nonzero and belongs to the center in
32 the integral closure of the chart coordinate algebra. -/
33 theorem actual_chart_normalization_center_ne_bot
34     {k : Type u} [Field k] [CharZero k]
35     {m : ℕ} (hm : 0 < m)
36     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
37     (w : GeneralDivisorialVisibleFrameWitness hm P) :
38     letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
39       (ComponentFractionField P) := w.column.W.ambientAlgebra
40     let U := w.column.W.place.valuation
41     let V : Subring (ComponentFractionField P) := w.column.W.place.valuation.toSubring
42     let Q : Subalgebra k (ComponentFractionField P) :=
43       Algebra.adjoin k (Set.range (chartGenericPoint P w.column.chart
44         (chartAffineCoordinateEquiv w.column.chart)))
45     let hQV : Q.toSubring ≤ V := by
46       intro z hz
47       exact chartGenericPointSubalgebra_le_valuationSubring hm P w z hz
48     ProjectiveChartNormalizationCenter.integralClosureCenter Q.toSubring U hQV ≠ ⊥ := by
49   classical
50   let C := w.column
51   let W := C.W
52   let F := ComponentFractionField P
53   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
54   let U := W.place.valuation
55   let V : Subring F := W.place.valuation.toSubring
56   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
57   letI : Algebra W.coefficientField V :=
58     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
59   letI : Algebra k V := C.groundCoeff.toAlgebra
60   letI : Algebra V F := V.subtype.toAlgebra
61   letI : IsScalarTower k V F := C.groundTower
62   let e := chartAffineCoordinateEquiv C.chart
63   let Q : Subalgebra k F := Algebra.adjoin k (Set.range (chartGenericPoint P C.chart e))
64   have hQV : Q.toSubring ≤ V := by
65     intro z hz
66     exact chartGenericPointSubalgebra_le_valuationSubring hm P w z hz
67   have hq0max : C.q 0 ∈ maximalIdeal V :=
68     witness_q0_mem_maximal hm P w
69   have hchart : 0 ≠ C.chart := by
70     intro hc
71     have hq1 : C.q 0 = 1 := by simpa [hc] using C.chart_one
72     have hnot : (1 : V) ∉ maximalIdeal V :=
73       (IsLocalRing.notMem_maximalIdeal).mpr isUnit_one
74     exact hnot (hq1 ▸ hq0max)
75   let i0 : Fin m := e.symm ⟨0, hchart⟩
76   have hi0 : (e i0).1 = 0 := by
77     have heq := Equiv.apply_symm_apply e ⟨0, hchart⟩
78     exact congrArg Subtype.val heq
79   have hnorm := chartGenericPoint_eq_normalized_lift
80     (P := P) (V := V) C.chart e C.q C.scale C.chart_one C.q_commonScale
81   have hrmem : chartGenericPoint P C.chart e i0 ∈ Q :=
82     Algebra.subset_adjoin (R := k) (A := F) (Set.mem_range_self i0)
83   let r : Q.toSubring := ⟨chartGenericPoint P C.chart e i0, hrmem⟩
84   have hmap : Subring.inclusion hQV r = C.q 0 := by
85     apply Subtype.ext
86     change chartGenericPoint P C.chart e i0 = (C.q 0 : V)
87     calc
88       chartGenericPoint P C.chart e i0 =
89           algebraMap V F (C.q (e i0).1) := congrFun hnorm i0
90       _ = algebraMap V F (C.q 0) := by rw [hi0]
91       _ = (C.q 0 : F) := rfl
92   have hr0 : (r : F) ≠ 0 := by
93     intro hz
94     have hz' : Subring.inclusion hQV r = 0 := Subtype.ext hz
95     rw [hmap] at hz'
96     exact C.q0_ne hz'
97   have hrmax : Subring.inclusion hQV r ∈ maximalIdeal U.toLocalSubring.toSubring := by
98     rw [hmap]
99     exact hq0max
100   exact ProjectiveChartNormalizationCenter.integralClosureCenter_ne_bot_of_nonzero_maximal
101     Q.toSubring U hQV r hr0 hrmax
