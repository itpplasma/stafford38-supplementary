8 /-!
9 # Finite-separable coefficient sections through nilpotent thickenings
10 
11 Let `K/E` be a separable field extension.  Formal etaleness constructs a
12 unique `E`-algebra section of every surjective `E`-algebra map to `K` whose
13 kernel is nilpotent.  These sections are automatically compatible with maps
14 between such thickenings.
15 
16 Applied to the quotients of an equicharacteristic DVR by positive powers of
17 its maximal ideal, this supplies the finite-level coefficient fields needed
18 for a completed power-series chart.  The passage from this compatible family
19 to the inverse limit is outside this module's finite-level scope; the
20 completion comparison is supplied by the downstream completed-DVR chart.
21 -/

33 /-- The canonical coefficient-field lift through a nilpotent thickening.
34 
35 The construction uses only the formal smoothness of the separable field
36 extension `K/E`; in particular it does not assume a coefficient field in `R`.
37 -/
38 def finiteSeparableSection
39     (hsep : Algebra.IsSeparable E K)
40     (residue : R →ₐ[E] K)
41     (hsurj : Function.Surjective residue)
42     (hnil : IsNilpotent (RingHom.ker residue.toRingHom)) :
43     K →ₐ[E] R := by
44   letI : Algebra.IsSeparable E K := hsep
45   letI : Algebra.FormallyEtale E K :=
46     Algebra.FormallyEtale.of_isSeparable E K
47   exact Algebra.FormallySmooth.liftOfSurjective
48     (AlgHom.id E K) residue hsurj hnil
49 
50 /-- The canonical lift is a section of the residue homomorphism. -/
51 @[simp]
52 theorem residue_comp_finiteSeparableSection
53     (hsep : Algebra.IsSeparable E K)
54     (residue : R →ₐ[E] K)
55     (hsurj : Function.Surjective residue)
56     (hnil : IsNilpotent (RingHom.ker residue.toRingHom)) :
57     residue.comp (finiteSeparableSection E K R hsep residue hsurj hnil) =
58       AlgHom.id E K := by
59   letI : Algebra.IsSeparable E K := hsep
60   letI : Algebra.FormallyEtale E K :=
61     Algebra.FormallyEtale.of_isSeparable E K
62   exact Algebra.FormallySmooth.comp_liftOfSurjective
63     (AlgHom.id E K) residue hsurj hnil
64 
65 /-- Every coefficient-field lift through the same nilpotent thickening is the
66 canonical one.  This is the formal-unramified half of separability. -/
67 theorem finiteSeparableSection_unique
68     (hsep : Algebra.IsSeparable E K)
69     (residue : R →ₐ[E] K)
70     (hsurj : Function.Surjective residue)
71     (hnil : IsNilpotent (RingHom.ker residue.toRingHom))
72     (lift : K →ₐ[E] R)
73     (hlift : residue.comp lift = AlgHom.id E K) :
74     lift = finiteSeparableSection E K R hsep residue hsurj hnil := by
75   letI : Algebra.IsSeparable E K := hsep
76   letI : Algebra.FormallyEtale E K :=
77     Algebra.FormallyEtale.of_isSeparable E K
78   apply Algebra.FormallyUnramified.lift_unique' residue hnil
79   rw [hlift, residue_comp_finiteSeparableSection]
80 
81 /-- A separable residue field has a unique coefficient-field section through
82 every surjective nilpotent `E`-algebra thickening. -/
83 theorem existsUnique_finiteSeparableSection
84     (hsep : Algebra.IsSeparable E K)
85     (residue : R →ₐ[E] K)
86     (hsurj : Function.Surjective residue)
87     (hnil : IsNilpotent (RingHom.ker residue.toRingHom)) :
88     ∃! lift : K →ₐ[E] R,
89       residue.comp lift = AlgHom.id E K := by
90   refine ⟨finiteSeparableSection E K R hsep residue hsurj hnil,
91     residue_comp_finiteSeparableSection E K R hsep residue hsurj hnil, ?_⟩
92   intro lift hlift
93   exact finiteSeparableSection_unique E K R hsep residue hsurj hnil
94     lift hlift

100 /-- The finite-level coefficient sections are compatible with every
101 `E`-algebra transition map commuting with residue specialization.  Thus no
102 extra choices have to be synchronized along an Artinian quotient tower. -/
103 theorem finiteSeparableSection_naturality
104     (hsep : Algebra.IsSeparable E K)
105     (residueR : R →ₐ[E] K)
106     (hsurjR : Function.Surjective residueR)
107     (hnilR : IsNilpotent (RingHom.ker residueR.toRingHom))
108     (residueS : S →ₐ[E] K)
109     (hsurjS : Function.Surjective residueS)
110     (hnilS : IsNilpotent (RingHom.ker residueS.toRingHom))
111     (transition : S →ₐ[E] R)
112     (htransition : residueR.comp transition = residueS) :
113     transition.comp
114         (finiteSeparableSection E K S hsep residueS hsurjS hnilS) =
115       finiteSeparableSection E K R hsep residueR hsurjR hnilR := by
116   apply finiteSeparableSection_unique E K R hsep residueR hsurjR hnilR
117   rw [← AlgHom.comp_assoc, htransition,
118     residue_comp_finiteSeparableSection]
119 
120 end Compatibility
