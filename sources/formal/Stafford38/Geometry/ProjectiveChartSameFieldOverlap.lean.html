236 theorem selectedLocalizationToField_injective
237     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (j : Fin m)
238     (hxj : componentCoordinate P j ≠ 0) :
239     Function.Injective (selectedAffineChartToFunctionField P j hxj) := by
240   letI : (SelectedAffineChartIdeal (k := k) P j).IsPrime := selectedIdeal_isPrime P j hxj
241   letI : IsDomain (SelectedAffineChartQuotient (k := k) P j) :=
242     Ideal.Quotient.isDomain (SelectedAffineChartIdeal (k := k) P j)
243   have hg : Function.Injective (selectedQuotientToField P j hxj) := by
244     intro x y hxy
245     obtain ⟨x, rfl⟩ := Ideal.Quotient.mk_surjective x
246     obtain ⟨y, rfl⟩ := Ideal.Quotient.mk_surjective y
247     have hmem : x - y ∈ SelectedAffineChartIdeal (k := k) P j := by

249       rw [componentChartEquationIdeal_eq_genericEvalKer P (Fin.succ j)

251       apply RingHom.mem_ker.mpr
252       change (MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
253         (selectedPoint P j)) (x - y) = 0
254       have hxy' : MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
255           (selectedPoint P j) x =
256         MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
257           (selectedPoint P j) y := by

260         (selectedPoint P j)) x =
261         (MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))

263       rw [map_sub, hxy', sub_self]

265   exact away_lift_injective (r := selectedAffineChartDenominator P j) (by
