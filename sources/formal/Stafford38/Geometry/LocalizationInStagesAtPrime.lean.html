14 /-- If `L` is a localization of `B`, the local rings over compatible residue
15 kernels are canonically equivalent. The center equality is derived from the
16 commuting residue maps; no height or DVR hypothesis is used. -/
17 theorem kernel_center_localization_equiv
18     {B : Type u} [CommRing B] {κ : Type v} [Field κ]
19     (M : Submonoid B)
20     (p : Ideal (Localization M)) [p.IsPrime]
21     (rhoL : Localization M →+* κ) (rhoB : B →+* κ)
22     (hp : p = RingHom.ker rhoL)
23     (hLoc : rhoL.comp (algebraMap B (Localization M)) = rhoB) :
24     ∃ e : Localization.AtPrime p ≃ₐ[B]
25         Localization.AtPrime (p.comap (algebraMap B (Localization M))),
26       p.comap (algebraMap B (Localization M)) = RingHom.ker rhoB ∧
27       ∀ b : B,
28         e (algebraMap (Localization M) (Localization.AtPrime p)
29           (algebraMap B (Localization M) b)) =
30           algebraMap B
31             (Localization.AtPrime (p.comap (algebraMap B (Localization M)))) b := by
32   have hcenter : p.comap (algebraMap B (Localization M)) = RingHom.ker rhoB := by
33     ext b
34     rw [Ideal.mem_comap, hp, RingHom.mem_ker]
35     change rhoL (algebraMap B (Localization M) b) = 0 ↔ rhoB b = 0
36     have heq : rhoL (algebraMap B (Localization M) b) = rhoB b := by
37       simpa only [RingHom.comp_apply] using
38         congrArg (fun f : B →+* κ => f b) hLoc
39     rw [heq]
40   let e0 := IsLocalization.localizationLocalizationAtPrimeIsoLocalization M p
41   have e : Localization.AtPrime p ≃ₐ[B]
42       Localization.AtPrime (p.comap (algebraMap B (Localization M))) := e0.symm
43   refine ⟨e, hcenter, ?_⟩
44   intro b
45   have htower :
46       algebraMap (Localization M) (Localization.AtPrime p)
47         (algebraMap B (Localization M) b) =
48       algebraMap B (Localization.AtPrime p) b :=
49     IsScalarTower.algebraMap_apply B (Localization M) (Localization.AtPrime p) b
50   rw [htower]
51   exact e.commutes b
52 
53 end Stafford38.Geometry.LocalizationInStagesAtPrime
