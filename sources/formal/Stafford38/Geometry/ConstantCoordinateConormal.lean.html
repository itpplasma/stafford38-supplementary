8 /-!
9 # Constant-coordinate conormal directions
10 
11 The boundary-divisor argument is needed only when the distinguished affine
12 coordinate is nonconstant on the relevant base component.  If
13 `X_i - c` already belongs to the reduced base ideal, its differential is the
14 pure `i`-th coordinate covector at every base point.  The desired conormal
15 axis is then present at a constant point, with no projective boundary or
16 completion.
17 -/

33 /-- The differential of `X_i - c` is the pure `i`-th coordinate covector. -/
34 theorem differentialCovector_coordinate_sub_constant
35     (y : Fin m → k) (i : Fin m) (c : k) :
36     differentialCovector y (MvPolynomial.X i - MvPolynomial.C c) =
37       coordinateCovector (fun j ↦ if j = i then 1 else 0) := by
38   ext v
39   simp [differentialCovector, coordinateCovector, differentialAt,
40     Pi.single_apply]

42 /-- A constant-coordinate equation places the corresponding pure axis in the
43 equation-defined affine conormal space. -/
44 theorem pureCoordinate_mem_affineConormalSpace_of_sub_constant_mem
45     (I : Ideal (MvPolynomial (Fin m) k))
46     (y : Fin m → k) (i : Fin m) (c : k)
47     (hcoordinate : MvPolynomial.X i - MvPolynomial.C c ∈ I) :
48     coordinateCovector (fun j ↦ if j = i then 1 else 0) ∈
49       affineConormalSpace y I := by
50   rw [affineConormalSpace_eq_equationCovectorSpan]
51   rw [← differentialCovector_coordinate_sub_constant y i c]
52   apply Submodule.subset_span
53   exact ⟨⟨MvPolynomial.X i - MvPolynomial.C c, hcoordinate⟩, rfl⟩

55 /-- The constant-coordinate branch produces an actual equation-conormal
56 point over every common zero of the base ideal. -/
57 theorem constantCoordinate_phasePoint_mem_equationConormalLocus
58     (I : Ideal (MvPolynomial (Fin m) k))
59     (y : Fin m → k)
60     (hy : ∀ f ∈ I, MvPolynomial.eval y f = 0)
61     (i : Fin m) (c : k)
62     (hcoordinate : MvPolynomial.X i - MvPolynomial.C c ∈ I) :
63     Sum.elim y (fun j ↦ if j = i then 1 else 0) ∈
64       equationConormalLocus I := by
65   refine ⟨?_, ?_⟩
66   · intro f hf
67     exact hy f hf
68   · exact pureCoordinate_mem_affineConormalSpace_of_sub_constant_mem
69       I y i c hcoordinate

71 /-- The same constant-coordinate branch gives the exact Laurent-series point
72 used by the terminal asymptotic consumer.  All coordinates are constant
73 series; no boundary valuation is involved. -/
74 theorem exists_laurentConormalAxis_of_coordinate_sub_constant_mem
75     (I : Ideal (MvPolynomial (Fin m) k))
76     (y : Fin m → k)
77     (hy : ∀ f ∈ I, MvPolynomial.eval y f = 0)
78     (i : Fin m) (c : k)
79     (hcoordinate : MvPolynomial.X i - MvPolynomial.C c ∈ I) :
80     ∃ (yL : Fin m → LaurentSeries k)
81       (xi : Fin m → PowerSeries k),
82       Sum.elim yL
83           (fun j ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi j)) ∈
84         equationConormalLocus
85           (I.map (scalarPolynomialMap
86             (k := k) (K := LaurentSeries k) (Fin m))) ∧
87       residueColumn xi = (fun j ↦ if j = i then 1 else 0) := by
88   let yL : Fin m → LaurentSeries k := fun j ↦ algebraMap k (LaurentSeries k) (y j)
89   let xi : Fin m → PowerSeries k := fun j ↦
90     PowerSeries.C (if j = i then 1 else 0)
91   let IL := I.map
92     (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m))
93   have hyL : ∀ f ∈ IL, MvPolynomial.eval yL f = 0 := by
94     rw [show (∀ f ∈ IL, MvPolynomial.eval yL f = 0) ↔
95         yL ∈ MvPolynomial.zeroLocus (LaurentSeries k) IL by rfl]
96     rw [mem_zeroLocus_map_iff]
97     intro f hf
98     change MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
99       ((algebraMap k (LaurentSeries k)) ∘ y) f = 0
100     rw [← MvPolynomial.eval₂_comp, hy f hf, map_zero]
101   have hcoordinateL :
102       MvPolynomial.X i -
103           MvPolynomial.C (algebraMap k (LaurentSeries k) c) ∈ IL := by
104     have hmapped := Ideal.mem_map_of_mem
105       (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m))
106       hcoordinate
107     change (scalarPolynomialMap
108       (k := k) (K := LaurentSeries k) (Fin m))
109         (MvPolynomial.X i - MvPolynomial.C c) ∈ IL at hmapped
110     rw [(scalarPolynomialMap
111       (k := k) (K := LaurentSeries k) (Fin m)).map_sub] at hmapped
112     simpa [scalarPolynomialMap] using hmapped
113   have hphase := constantCoordinate_phasePoint_mem_equationConormalLocus
114     IL yL hyL i (algebraMap k (LaurentSeries k) c) hcoordinateL
115   refine ⟨yL, xi, ?_, ?_⟩
116   · convert hphase using 1
117     funext q
118     rcases q with j | j
119     · rfl
120     · simp [xi]
121   · funext j
122     simp [xi, residueColumn]
