16 /-- A residue map from a localization identifies its contraction with a given
17 normalization center. If the residue extension is algebraic and the ambient
18 function field has relative transcendence degree one, the center has height
19 one. All maps and center identities are explicit inputs, so this lemma does
20 not choose a different valuation or normalization model. -/
21 theorem height_one_of_localized_algebraic_residue
22     {E : Type u} {B : Type v} {A : Type w} {F : Type x}
23     {κ : Type y} {C : Type z}
24     [Field E] [CommRing B] [IsDomain B] {S : Submonoid B}
25     [CommRing A] [IsDomain A] [Algebra B A] [IsLocalization S A]
26     [Algebra E A] [Algebra.FiniteType E A]
27     [Field F] [Algebra A F] [IsFractionRing A F]
28     [Algebra E F] [IsScalarTower E A F]
29     [Field κ] [Algebra E κ] [Algebra.IsAlgebraic E κ]
30     [CommRing C]
31     (rhoA : A →+* κ) (rhoE : A →ₐ[E] κ)
32     (hρ : (rhoE : A →+* κ) = rhoA)
33     (rhoB : B →+* κ)
34     (hrestrict : rhoA.comp (algebraMap B A) = rhoB)
35     (e : B ≃+* C) (J : Ideal C)
36     (hcenter : RingHom.ker rhoB = J.comap e.toRingHom)
37     (hJ0 : J ≠ ⊥)
38     (htrdeg : Algebra.trdeg E F = 1) : J.height = 1 := by
39   have hmax' : (RingHom.ker (rhoE : A →+* κ)).IsMaximal :=
40     Stafford38.Geometry.ResidueBasisLocalization.ker_isMaximal_of_algebraic_residue
41       (E := E) (A := A) (K := κ) rhoE
42   have hmax : (RingHom.ker rhoA).IsMaximal := by
43     rw [← hρ]
44     exact hmax'
45   let p : PrimeSpectrum A := ⟨RingHom.ker rhoA, hmax.isPrime⟩
46   letI : p.asIdeal.IsMaximal := hmax
47   have hcontract :=
48     Stafford38.Geometry.ActualChartCenterContraction.under_ker_eq_ker_of_restriction
49       (B := B) (A := A) (K := κ) (ρA := rhoA) (ρB := rhoB) hrestrict
50   have hcontractCenter : Ideal.under B p.asIdeal = J.comap e.toRingHom := by
51     change Ideal.under B (RingHom.ker rhoA) = J.comap e.toRingHom
52     exact hcontract.trans hcenter
53   have hcontract0 : Ideal.under B p.asIdeal ≠ ⊥ := by
54     rw [hcontractCenter]
55     intro hbot
56     apply hJ0
57     apply le_antisymm
58     · intro c hc
59       obtain ⟨b, rfl⟩ := e.surjective c
60       have hb : b ∈ J.comap e.toRingHom := hc
61       rw [hbot] at hb
62       simpa using hb
63     · exact bot_le
64   have hcomap0 : J.comap e.toRingHom ≠ ⊥ := by
65     rw [← hcontractCenter]
66     exact hcontract0
67   exact Stafford38.Geometry.ActualChartCenterHeight.normalized_center_height_one_of_localized_curve
68     (E := E) (R := B) (A := A) (F := F) (C := C) (S := S)
69     e p J hcontractCenter.symm hcomap0 htrdeg
70 
71 end Stafford38.Geometry.ActualResidueCenterHeight
