35 /-! ## Finite reindexing of an equation-differential expansion -/
36 
37 /-- Membership of a coordinate covector in the equation conormal space gives
38 an honest `Fin r`-indexed gradient identity.  This is the finite-support
39 content of `Submodule.span`; it does not require the whole ideal to be
40 finitely generated. -/
41 theorem exists_fin_gradient_identity_of_mem_affineConormalSpace
42     {m : ℕ}
43     (I : Ideal (MvPolynomial (Fin m) k))
44     (y xi : Fin m → k)
45     (hxi : coordinateCovector xi ∈ affineConormalSpace y I) :
46     ∃ (r : ℕ) (equations : Fin r → I) (coefficients : Fin r → k),
47       ∀ i, xi i = ∑ j, coefficients j * differentialAt y (equations j).1 i := by
48   classical
49   obtain ⟨c, hc⟩ :=
50     coordinate_mem_affineConormalSpace_exists_finsupp y xi I hxi
51   let ι := {f : I // f ∈ c.support}
52   let e : ι ≃ Fin (Fintype.card ι) := Fintype.equivFin ι
53   let equations : Fin (Fintype.card ι) → I := fun j ↦ (e.symm j).1
54   let coefficients : Fin (Fintype.card ι) → k := fun j ↦ c (e.symm j).1
55   refine ⟨Fintype.card ι, equations, coefficients, fun i ↦ ?_⟩
56   rw [hc i]
57   calc
58     ∑ f ∈ c.support, c f * differentialAt y f.1 i =
59         ∑ f : ι, c f.1 * differentialAt y f.1.1 i := by
60           exact Finset.sum_subtype c.support (by simp)
61             (fun f ↦ c f * differentialAt y f.1 i)
62     _ = ∑ j : Fin (Fintype.card ι),
63         coefficients j * differentialAt y (equations j).1 i := by
64           exact Fintype.sum_equiv e _ _ (fun f ↦ by
65             simp [coefficients, equations])

67 /-! ## Completed formal chart to finite-gradient certificate -/
68 
69 /-- A formal projective row plus the standard tangent-inclusion condition
70 constructs the complete finite-gradient boundary certificate.
71 
72 This theorem identifies the exact higher-dimensional local bridge left to
73 the normalization argument: prove the Zariski tangent space of the
74 scalar-extended ambient ideal is contained in the dehomogenized span of the
75 formal divisor and normalized transverse columns.  Once that inclusion is
76 available, Mathlib's finite-dimensional annihilator and finite-support span
77 machinery supplies the finite equations and coefficients automatically. -/
78 theorem exists_finiteGradientBoundaryCertificate_of_zariski_le_formalSpan
79     {m : ℕ} (hm : 0 < m)
80     (I : Ideal (MvPolynomial (Fin m) k))
81     {κ : Type*} [Fintype κ]
82     (q ell : Fin (m + 1) → PowerSeries k)
83     (Z : Matrix (Fin (m + 1)) κ (PowerSeries k))
84     (tau : Fin (m + 1) → PowerSeries k)
85     (hq0 : q 0 ≠ 0)
86     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
87     (hbase : ∀ f ∈
88         I.map (scalarPolynomialMap
89           (k := k) (K := LaurentSeries k) (Fin m)),
90       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0)
91     (htangent :
92       zariskiTangentSpace (dehomogenizedPoint (laurentColumn q))
93           (I.map (scalarPolynomialMap
94             (k := k) (K := LaurentSeries k) (Fin m))) ≤
95         dehomogenizedTangentSpan (laurentColumn q)
96           (laurentNonpositionTangentMatrix Z tau))
97     (hresidue :
98       residueColumn (fun i : Fin m ↦ ell i.succ) =
99         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0)) :
100     Nonempty (FiniteGradientBoundaryCertificate k m hm I) := by
101   let Iext := I.map
102     (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m))
103   let y := dehomogenizedPoint (laurentColumn q)
104   let xi : Fin m → LaurentSeries k := fun i ↦ laurentColumn ell i.succ
105   have hconormal : coordinateCovector xi ∈ affineConormalSpace y Iext := by
106     exact coordinateCovector_mem_affineConormalSpace_of_zariski_le_span
107       Iext (laurentColumn q) (laurentColumn ell)
108         (laurentNonpositionTangentMatrix Z tau)
109         (laurentColumn_ne_zero_of_ne_zero q hq0)
110         (laurentColumn_dot_eq_zero_of_formalTangent_rowMul q ell Z tau hrow)
111         (laurentNonposition_rowMul_eq_zero_of_formalTangent_rowMul
112           q ell Z tau hrow)
113         htangent
114   obtain ⟨r, equations, coefficients, hgradient⟩ :=
115     exists_fin_gradient_identity_of_mem_affineConormalSpace
116       Iext y xi hconormal
117   exact ⟨{
118     equationCount := r
119     q := q
120     ell := ell
121     q_origin_ne := hq0
122     projective_annihilation :=
123       laurentColumn_dot_eq_zero_of_formalTangent_rowMul q ell Z tau hrow
124     base_vanish := hbase
125     equations := equations
126     coefficients := coefficients
127     gradient_identity := hgradient
128     residue_axis := hresidue
129   }⟩

134 end
