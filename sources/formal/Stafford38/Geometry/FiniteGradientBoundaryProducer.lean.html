9 /-!
10 # A finite-gradient boundary producer
11 
12 The completed-boundary consumer does not need a comparison of the entire
13 Zariski tangent space with a chosen matrix of tangent columns.  It needs only
14 one conormal covector.  This file replaces that global comparison by an
15 explicit finite expression of the required covector as a linear combination
16 of differentials of equations in the scalar-extended affine ideal.
17 
18 This is the algebraic certificate naturally supplied at a smooth generic
19 point by a Jacobian/conormal calculation.  It is strictly local to the one
20 annihilating row and is independently checkable coordinate by coordinate.
21 No normalization, projective closure, boundary divisor, or existence of such
22 a certificate is asserted here.
23 -/

54 /-- A coordinate covector is conormal when its coordinates are an explicit
55 finite linear combination of gradients of equations in the ideal.  This
56 avoids any equality or inclusion assertion about the whole tangent space. -/
57 theorem coordinateCovector_mem_affineConormalSpace_of_finite_gradient
58     {m r : ℕ}
59     (I : Ideal (MvPolynomial (Fin m) K))
60     (y xi : Fin m → K)
61     (equations : Fin r → I)
62     (coefficients : Fin r → K)
63     (hgradient : ∀ i,
64       xi i = ∑ j, coefficients j * differentialAt y (equations j).1 i) :
65     coordinateCovector xi ∈ affineConormalSpace y I := by
66   classical
67   rw [affineConormalSpace_eq_equationCovectorSpan]
68   let g := baseLinearCombination coefficients (fun j ↦ (equations j).1)
69   have hg : g ∈ I := by
70     dsimp only [g, baseLinearCombination]
71     apply Ideal.sum_mem
72     intro j hj
73     exact I.mul_mem_left (MvPolynomial.C (coefficients j)) (equations j).2
74   let gI : I := ⟨g, hg⟩
75   have hxi : xi = fun i ↦ differentialAt y g i := by
76     funext i
77     rw [hgradient, differentialAt_baseLinearCombination]
78   have heq : coordinateCovector xi = differentialCovector y g := by
79     apply LinearMap.ext
80     intro v
81     simp only [coordinateCovector_apply, differentialCovector_apply, hxi]
82   rw [heq]
83   apply Submodule.subset_span
84   refine ⟨gI, ?_⟩
85   rfl

163 /-- The corrected finite-gradient certificate produces exactly the Laurent
164 conormal witness required by the canonical asymptotic step. -/
165 theorem exists_conormalAxis_of_finiteGradientBoundaryCertificate
166     {k : Type u} [Field k]
167     {m : ℕ} (hm : 0 < m)
168     (I : Ideal (MvPolynomial (Fin m) k))
169     (W : FiniteGradientBoundaryCertificate k m hm I) :
170     ∃ (y : Fin m → LaurentSeries k)
171       (xi : Fin m → PowerSeries k),
172       Sum.elim y
173           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈
174         equationConormalLocus
175           (I.map (scalarPolynomialMap
176             (k := k) (K := LaurentSeries k) (Fin m))) ∧
177       residueColumn xi =
178         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
179   let Iext := I.map
180     (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m))
181   let y := dehomogenizedPoint (laurentColumn W.q)
182   let xi : Fin m → PowerSeries k := fun i ↦ W.ell i.succ
183   have hconormal :
184       coordinateCovector (fun i ↦ laurentColumn W.ell i.succ) ∈
185         affineConormalSpace y Iext := by
186     exact coordinateCovector_mem_affineConormalSpace_of_finite_gradient
187       Iext y (fun i ↦ laurentColumn W.ell i.succ)
188         W.equations W.coefficients W.gradient_identity
189   refine ⟨y, xi, ?_, W.residue_axis⟩
190   refine ⟨?_, ?_⟩
191   · simpa [Iext, y] using W.base_vanish
192   · simpa [xi, laurentColumn, y, Iext] using hconormal

194 /-! ## Pointwise handoff to the canonical producer -/
195 
196 /-- For fixed canonical data, a finite-gradient boundary certificate has
197 exactly the existential conclusion demanded by
198 `CanonicalAsymptoticLaurentProducer`.  Thus the remaining nonconstant
199 geometric producer may target this certificate instead of the stronger
200 all-tangent-space interface. -/
201 theorem canonicalAsymptoticLaurentWitness_of_finiteGradientBoundaryCertificate
202     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k]
203     (n N : ℕ) (d : WeylAlg k (n + 1))
204     (W : FiniteGradientBoundaryCertificate k (n + 1)
205       (Nat.zero_lt_succ n)
206       (reducedOrderBaseIdeal k
207         (canonicalRightIdeal (presentedCoordinate k n) d N))) :
208     ∃ (y : Fin (n + 1) → LaurentSeries k)
209       (xi : Fin (n + 1) → PowerSeries k),
210       Sum.elim y
211           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈
212         equationConormalLocus
213           ((reducedOrderBaseIdeal k
214             (canonicalRightIdeal (presentedCoordinate k n) d N)).map
215               (scalarPolynomialMap
216                 (k := k) (K := LaurentSeries k) (Fin (n + 1)))) ∧
217       residueColumn xi = pureMomentumFibreAxis k n := by
218   obtain ⟨y, xi, hmem, haxis⟩ :=
219     exists_conormalAxis_of_finiteGradientBoundaryCertificate
220       (k := k) (m := n + 1) (Nat.zero_lt_succ n)
221       (reducedOrderBaseIdeal k
222         (canonicalRightIdeal (presentedCoordinate k n) d N)) W
223   exact ⟨y, xi, hmem, haxis⟩
