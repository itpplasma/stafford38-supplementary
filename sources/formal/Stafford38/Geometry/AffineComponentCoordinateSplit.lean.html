11 # Constant or transcendental coordinates on an affine component
12 
13 For a prime component over an algebraically closed field, a coordinate is
14 either constant on the component or transcendental in its function field.
15 Together with existence of a minimal prime over every proper affine ideal,
16 this gives the component split used before the divisorial-boundary
17 construction.
18 -/

30 /-- The image of an affine coordinate in the fraction field of a prime
31 component. -/
32 def componentCoordinate
33     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m) :
34     FractionRing
35       (MvPolynomial (Fin m) k ⧸ P.asIdeal) :=
36   algebraMap
37       (MvPolynomial (Fin m) k ⧸ P.asIdeal)
38       (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal))
39     (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i))

41 /-- On a prime affine component over an algebraically closed field, a
42 coordinate is either represented by a ground-field constant or is
43 transcendental in the component function field. -/
44 theorem coordinate_constant_or_transcendental
45     [IsAlgClosed k]
46     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m) :
47     (∃ c : k, MvPolynomial.X i - MvPolynomial.C c ∈ P.asIdeal) ∨
48       Transcendental k (componentCoordinate P i) := by
49   classical
50   by_cases htr : Transcendental k (componentCoordinate P i)
51   · exact Or.inr htr
52   · left
53     have halg : IsAlgebraic k (componentCoordinate P i) := by
54       simpa only [Transcendental, not_not] using htr
55     have hint : IsIntegral k (componentCoordinate P i) :=
56       isAlgebraic_iff_isIntegral.mp halg
57     let p := minpoly k (componentCoordinate P i)
58     have hpmonic : p.leadingCoeff = 1 := minpoly.monic hint
59     have hpdegree : p.degree = 1 :=
60       IsAlgClosed.degree_eq_one_of_irreducible k (minpoly.irreducible hint)
61     have hroot : aeval (componentCoordinate P i) p = 0 :=
62       minpoly.aeval k (componentCoordinate P i)
63     rw [eq_X_add_C_of_degree_eq_one hpdegree, hpmonic, Polynomial.C_1,
64       one_mul, aeval_add, aeval_X, aeval_C, add_eq_zero_iff_eq_neg] at hroot
65     let c : k := -p.coeff 0
66     refine ⟨c, ?_⟩
67     have hfield :
68         algebraMap k
69             (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)) c =
70           componentCoordinate P i := by
71       simpa only [c, map_neg] using hroot.symm
72     have hquotient :
73         algebraMap k (MvPolynomial (Fin m) k ⧸ P.asIdeal) c =
74           Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i) := by
75       apply IsFractionRing.injective
76         (MvPolynomial (Fin m) k ⧸ P.asIdeal)
77         (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal))
78       change algebraMap k
79           (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)) c =
80         componentCoordinate P i
81       exact hfield
82     rw [← Ideal.Quotient.eq_zero_iff_mem]
83     rw [map_sub, ← MvPolynomial.algebraMap_eq,
84       Ideal.Quotient.mk_algebraMap]
85     exact sub_eq_zero.mpr hquotient.symm

87 /-- Every proper affine ideal has a minimal-prime component on which the
88 distinguished coordinate is either constant or transcendental in the
89 component function field.  Radicality is not needed for this component
90 selection theorem. -/
91 theorem exists_minimalPrime_coordinate_constant_or_transcendental
92     [IsAlgClosed k]
93     (I : Ideal (MvPolynomial (Fin m) k)) (hproper : I ≠ ⊤) (i : Fin m) :
94     ∃ P : PrimeSpectrum (MvPolynomial (Fin m) k),
95       P.asIdeal ∈ I.minimalPrimes ∧
96       I ≤ P.asIdeal ∧
97       ((∃ c : k, MvPolynomial.X i - MvPolynomial.C c ∈ P.asIdeal) ∨
98         Transcendental k (componentCoordinate P i)) := by
99   classical
100   let P₀ : I.minimalPrimes := Classical.choice (I.nonempty_minimalPrimes hproper)
101   let P : PrimeSpectrum (MvPolynomial (Fin m) k) :=
102     ⟨P₀.1, P₀.2.isPrime⟩
103   exact ⟨P, P₀.2, P₀.2.le, coordinate_constant_or_transcendental P i⟩
