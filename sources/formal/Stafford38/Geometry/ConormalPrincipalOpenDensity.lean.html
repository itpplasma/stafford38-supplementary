10 # Density of the equation conormal over a principal open
11 
12 The first lemma is the algebraic Nullstellensatz/primality core used by the
13 principal-open density argument.  The conormal-specific graph construction is
14 kept as a separate theorem below, so its hypotheses and its use of the

29 theorem mem_of_vanishes_on_principal_open
30     (I : Ideal (MvPolynomial (Fin n) k)) (hI : I.IsPrime)
31     {f p : MvPolynomial (Fin n) k} (hf : f ∉ I)
32     (hp : ∀ y : Fin n → k, y ∈ MvPolynomial.zeroLocus k I →
33       MvPolynomial.eval y f ≠ 0 → MvPolynomial.eval y p = 0) :
34     p ∈ I := by
35   have hfp : f * p ∈ MvPolynomial.vanishingIdeal k
36       (MvPolynomial.zeroLocus k I) := by
37     rw [MvPolynomial.mem_vanishingIdeal_iff]
38     intro y hy
39     by_cases hfy : MvPolynomial.eval y f = 0
40     · simp [hfy]
41     · rw [map_mul]
42       change MvPolynomial.eval y f * MvPolynomial.eval y p = 0
43       rw [hp y hy hfy]
44       simp
45   letI : I.IsPrime := hI
46   have hfpI : f * p ∈ I := by
47     rw [MvPolynomial.IsPrime.vanishingIdeal_zeroLocus I] at hfp
48     exact hfp
49   rcases hI.mem_or_mem hfpI with hfi | hpi
50   · exact False.elim (hf hfi)
51   · exact hpi

53 def restrictedEquationConormalLocus
54     (I : Ideal (MvPolynomial (Fin n) k)) (S : Set (Fin n → k)) :
55     Set (PhaseVar n → k) :=
56   {q | q ∈ equationConormalLocus I ∧
57     (fun i => q (Sum.inl i)) ∈ S}

59 theorem restricted_subset_equationConormalLocus
60     (I : Ideal (MvPolynomial (Fin n) k)) (S : Set (Fin n → k)) :
61     restrictedEquationConormalLocus I S ⊆ equationConormalLocus I := by
62   intro q hq
63   exact hq.1

65 /-- A covector in the equation conormal is the gradient of one equation of
66 the ideal; finite linear combinations can be absorbed into that equation. -/
67 theorem exists_equation_with_differentialAt_eq
68     (I : Ideal (MvPolynomial (Fin n) k)) (y ξ : Fin n → k)
69     (hξ : coordinateCovector ξ ∈ affineConormalSpace y I) :
70     ∃ h ∈ I, ∀ i, differentialAt y h i = ξ i := by
71   classical
72   obtain ⟨c, hc⟩ :=
73     coordinate_mem_affineConormalSpace_exists_finsupp y ξ I hξ
74   let h : MvPolynomial (Fin n) k :=
75     ∑ p ∈ c.support, MvPolynomial.C (c p) * p.1
76   refine ⟨h, ?_, fun i => ?_⟩
77   · apply Ideal.sum_mem
78     intro p hp
79     exact I.mul_mem_left _ p.2
80   · rw [hc i]
81     simp only [h, differentialAt, map_sum, MvPolynomial.pderiv_mul,
82       MvPolynomial.pderiv_C, zero_mul, zero_add, MvPolynomial.eval_sum,
83       MvPolynomial.eval_mul, MvPolynomial.eval_C]
84 
85 def gradientGraphSubstitution (h : MvPolynomial (Fin n) k) :
86     SymbolRing k n →+* MvPolynomial (Fin n) k :=
87   MvPolynomial.eval₂Hom MvPolynomial.C
88     (Sum.elim MvPolynomial.X (fun i => MvPolynomial.pderiv i h))
89 
90 theorem eval_gradientGraphSubstitution
91     (h : MvPolynomial (Fin n) k) (P : SymbolRing k n) (y : Fin n → k) :
92     MvPolynomial.eval y (gradientGraphSubstitution h P) =
93       MvPolynomial.eval (Sum.elim y (differentialAt y h)) P := by
94   induction P using MvPolynomial.induction_on with
95   | C a => simp [gradientGraphSubstitution]
96   | add P Q hP hQ =>
97       simp only [gradientGraphSubstitution] at hP hQ
98       simp only [gradientGraphSubstitution, map_add, MvPolynomial.eval_add]
99       rw [hP, hQ]
100   | mul_X P i hP =>
101       simp only [gradientGraphSubstitution] at hP
102       rcases i with i | i
103       · simp only [gradientGraphSubstitution, map_mul, MvPolynomial.eval₂Hom_X',
104           Sum.elim_inl, MvPolynomial.eval_mul, MvPolynomial.eval_X]
105         rw [hP]
106       · simp only [gradientGraphSubstitution, map_mul, MvPolynomial.eval₂Hom_X',
107           Sum.elim_inr, MvPolynomial.eval_mul]
108         rw [hP]
109         simp [differentialAt]

111 /-- Restricting the equation conormal to any subset containing the principal
112 open part of the prime base variety does not change its algebraic closure. -/
113 theorem equationConormalClosure_restricted_eq
114     (I : Ideal (MvPolynomial (Fin n) k)) (hI : I.IsPrime)
115     (f : MvPolynomial (Fin n) k) (hf : f ∉ I)
116     (S : Set (Fin n → k))
117     (hS : S ⊆ MvPolynomial.zeroLocus k I)
118     (hopen : ∀ y : Fin n → k, y ∈ MvPolynomial.zeroLocus k I →
119       MvPolynomial.eval y f ≠ 0 → y ∈ S) :
120     MvPolynomial.zeroLocus k
121         (MvPolynomial.vanishingIdeal k (restrictedEquationConormalLocus I S)) =
122       equationConormalClosure I := by
123   apply Set.Subset.antisymm
124   · exact zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
125       (restrictedEquationConormalLocus I S)
126       (MvPolynomial.vanishingIdeal k (equationConormalLocus I))
127       (fun q hq P hP => hP q hq.1)
128   · apply zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
129     intro q qlocus P hP
130     let y : Fin n → k := fun i => q (.inl i)
131     let ξ : Fin n → k := fun i => q (.inr i)
132     obtain ⟨h, hhI, hgrad⟩ :=
133       exists_equation_with_differentialAt_eq I y ξ qlocus.2
134     have hrho : gradientGraphSubstitution h P ∈ I := by
135       apply mem_of_vanishes_on_principal_open I hI hf
136       intro z hzI hzf
137       have hzS : z ∈ S := hopen z hzI hzf
138       have hgraph : Sum.elim z (differentialAt z h) ∈
139           restrictedEquationConormalLocus I S := by
140         refine ⟨⟨hzI, ?_⟩, hzS⟩
141         rw [affineConormalSpace_eq_equationCovectorSpan]
142         apply Submodule.subset_span
143         exact ⟨⟨h, hhI⟩, rfl⟩
144       rw [eval_gradientGraphSubstitution]
145       exact hP _ hgraph
146     have hyrho := qlocus.1 _ hrho
147     rw [eval_gradientGraphSubstitution] at hyrho
148     change MvPolynomial.eval (Sum.elim y (differentialAt y h)) P = 0 at hyrho
149     have hgradfun : differentialAt y h = ξ := funext hgrad
150     have hsplit : Sum.elim y ξ = q := by
151       funext i
152       rcases i with i | i <;> rfl
153     rw [hgradfun, hsplit] at hyrho
154     exact hyrho
