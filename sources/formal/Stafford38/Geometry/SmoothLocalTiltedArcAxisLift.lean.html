10 /-!
11 # Conditional tilted-arc adapter for a smooth formal local chart
12 
13 This file starts *after* a chosen smooth local model has been identified with
14 `k[[t,z₁,…,z_d]]`.  It does not construct that identification for an arbitrary
15 smooth local ring.  Given the projective chart coordinates, their formal
16 transverse derivatives, the strict t-order factorizations, and the closed-point
17 transversality minor in this model, it tilts the t-axis away from one nonzero
18 bad-locus series.  The substitution preserves constants and unit minors, so
19 `FormalDivisorAxisLift` supplies the printed normalized tangent columns and
20 axis conormal.  The local-model and transversality data remain explicit inputs.
21 -/

40 /-- The coefficient vector `(1, α₁, …, α_d)` with a constant codomain. -/
41 def tiltedArcCoefficients {d : ℕ} (α : Fin d → k) : Fin (d + 1) → k :=
42   @Fin.cons d (fun _ => k) (1 : k) α

44 @[simp] theorem tiltedArcCoefficients_zero {d : ℕ} (α : Fin d → k) :
45     tiltedArcCoefficients (k := k) α 0 = 1 := by
46   simp [tiltedArcCoefficients]

49     tiltedArcCoefficients (k := k) α j.succ = α j := by

52 /-- The formal curve `t ↦ (t, α₁t, …, α_dt)` in the `PUnit` one-variable
53 presentation used by the existing bad-locus avoidance theorem. -/
54 def tiltedArcPointPUnit {d : ℕ} (α : Fin d → k) :
55     Fin (d + 1) → MvPowerSeries PUnit.{1} k :=
56   fun i => MvPowerSeries.C ((Fin.cons 1 α : Fin (d + 1) → k) i) *
57     MvPowerSeries.X (PUnit.unit : PUnit.{1})
58 
59 theorem tiltedArcHasSubstPUnit {d : ℕ} (α : Fin d → k) :
60     MvPowerSeries.HasSubst (tiltedArcPointPUnit (k := k) α) := by
61   apply MvPowerSeries.hasSubst_of_constantCoeff_zero
62   intro i
63   change MvPowerSeries.constantCoeff
64     (MvPowerSeries.C (tiltedArcCoefficients (k := k) α i) *
65       MvPowerSeries.X (PUnit.unit : PUnit.{1})) = 0
66   simp

68 /-- Restriction to the same tilted arc in Mathlib's `PUnit` target. -/
69 noncomputable def tiltedArcMapPUnit {d : ℕ} (α : Fin d → k) :
70     MvPowerSeries (Fin (d + 1)) k →+* MvPowerSeries PUnit.{1} k :=
71   (MvPowerSeries.substAlgHom (tiltedArcHasSubstPUnit (k := k) α)).toRingHom

73 /-- The canonical singleton-index identification from the avoidance target to
74 `PowerSeries`; this makes nonvanishing refer to exactly the same formal arc as
75 all tangent columns below. -/
76 noncomputable def punitToPowerSeries {k : Type u} [Field k] :
77     MvPowerSeries PUnit.{1} k ≃ₐ[k] PowerSeries k :=
78   MvPowerSeries.renameEquiv k (Equiv.equivPUnit Unit).symm

80 /-- The power-series arc map, obtained by transporting the existing avoidance

83     MvPowerSeries (Fin (d + 1)) k →+* PowerSeries k :=
84   (punitToPowerSeries (k := k)).toRingEquiv.toRingHom.comp

101 @[simp] theorem tiltedArc_X_t {d : ℕ} (α : Fin d → k) :
102     tiltedArc (k := k) α (MvPowerSeries.X 0) = PowerSeries.X := by
103   change MvPowerSeries.rename (Equiv.equivPUnit Unit).symm
104       ((MvPowerSeries.substAlgHom (tiltedArcHasSubstPUnit (k := k) α))
105         (MvPowerSeries.X 0)) = _
106   rw [MvPowerSeries.substAlgHom_X]
107   change MvPowerSeries.rename (Equiv.equivPUnit Unit).symm
108     (MvPowerSeries.C (1 : k) * MvPowerSeries.X (PUnit.unit : PUnit.{1})) = _
109   rw [map_mul, MvPowerSeries.rename_C, MvPowerSeries.rename_X]
110   simp [PowerSeries.X_apply]

112 private theorem tiltedArc_t_power_mul {d : ℕ} (α : Fin d → k)
113     (r : ℕ) (f : MvPowerSeries (Fin (d + 1)) k) :
114     tiltedArc (k := k) α ((MvPowerSeries.X 0) ^ r * f) =
115       (PowerSeries.X : PowerSeries k) ^ r * tiltedArc (k := k) α f := by
116   change (tiltedArcMap (k := k) α) ((MvPowerSeries.X 0) ^ r * f) = _
117   rw [(tiltedArcMap (k := k) α).map_mul, (tiltedArcMap (k := k) α).map_pow]
118   have hx : tiltedArcMap (k := k) α (MvPowerSeries.X 0) = PowerSeries.X := by
119     simpa [tiltedArc] using tiltedArc_X_t (k := k) α

123 private theorem constantCoeff_tiltedArcPUnit {d : ℕ} (α : Fin d → k)
124     (f : MvPowerSeries (Fin (d + 1)) k) :
125     MvPowerSeries.constantCoeff (tiltedArcMapPUnit (k := k) α f) =
126       MvPowerSeries.constantCoeff f := by
127   change MvPowerSeries.constantCoeff
128       ((MvPowerSeries.substAlgHom (tiltedArcHasSubstPUnit (k := k) α)) f) = _
129   rw [MvPowerSeries.substAlgHom_apply]
130   have hsub := tiltedArcHasSubstPUnit (k := k) α
131   have hzero : ∀ i, MvPowerSeries.constantCoeff
132       (tiltedArcPointPUnit (k := k) α i) = 0 := by
133     intro i
134     change MvPowerSeries.constantCoeff
135       (MvPowerSeries.C (tiltedArcCoefficients (k := k) α i) *
136         MvPowerSeries.X (PUnit.unit : PUnit.{1})) = 0
137     simp
138   have hsplit : f = MvPowerSeries.C (MvPowerSeries.constantCoeff f) +
139       (f - MvPowerSeries.C (MvPowerSeries.constantCoeff f)) := by ring
140   have htail : MvPowerSeries.constantCoeff
141       (f - MvPowerSeries.C (MvPowerSeries.constantCoeff f)) = 0 := by simp
142   rw [hsplit, MvPowerSeries.subst_add hsub]
143   rw [MvPowerSeries.subst_C, map_add,
144     MvPowerSeries.constantCoeff_subst_eq_zero hsub hzero htail]
145   simp

148     (f : MvPowerSeries (Fin (d + 1)) k) :
149     PowerSeries.constantCoeff (tiltedArc (k := k) α f) =

153       (tiltedArcMapPUnit (k := k) α f)) = _
154   rw [MvPowerSeries.constantCoeff_rename]

157 private theorem tiltedArc_ne_zero_of_avoidance {d : ℕ}
158     (α : Fin d → k) (f : MvPowerSeries (Fin (d + 1)) k)
159     (havoid : MvPowerSeries.subst (tiltedArcPointPUnit (k := k) α) f ≠ 0) :
160     tiltedArc (k := k) α f ≠ 0 := by
161   have hPUnit : tiltedArcMapPUnit (k := k) α f ≠ 0 := by
162     simpa [tiltedArcMapPUnit, MvPowerSeries.substAlgHom_apply] using havoid
163   intro hzero
164   apply hPUnit
165   have hzero' : (punitToPowerSeries (k := k))
166       (tiltedArcMapPUnit (k := k) α f) = 0 := by
167     change (punitToPowerSeries (k := k))
168       (tiltedArcMapPUnit (k := k) α f) = 0 at hzero
169     exact hzero
170   have hkernel : (tiltedArcMapPUnit (k := k) α f) = 0 :=
171     (punitToPowerSeries (k := k)).injective (by simpa using hzero')
172   exact hkernel

187 private theorem pderiv_t_factor {d : ℕ} (j : Fin d) (r : ℕ)
188     (u : MvPowerSeries (Fin (d + 1)) k) :
189     MvPowerSeries.pderiv (Fin.succ j)
190         ((MvPowerSeries.X 0 : MvPowerSeries (Fin (d + 1)) k) ^ r * u) =
191       (MvPowerSeries.X 0) ^ r * MvPowerSeries.pderiv (Fin.succ j) u := by
192   have hdt : MvPowerSeries.pderiv (Fin.succ j)
193       (MvPowerSeries.X 0 : MvPowerSeries (Fin (d + 1)) k) = 0 := by
194     exact MvPowerSeries.pderiv_X_of_ne (Fin.succ_ne_zero j).symm
195   have hdtr : MvPowerSeries.pderiv (Fin.succ j)
196       ((MvPowerSeries.X 0 : MvPowerSeries (Fin (d + 1)) k) ^ r) = 0 := by
197     rw [MvPowerSeries.pderiv_pow]
198     simp [hdt]
199   rw [(MvPowerSeries.pderiv (Fin.succ j)).leibniz]
200   simpa [hdtr, smul_eq_mul]

369 /-- A conditional closed-point tilted-arc producer.  The premises are exactly
370 local power-series data: the chart map `q`, its genuine formal transverse
371 partial derivatives, strict divisor-axis factors, and a full-rank transverse
372 minor at the closed point.  The selected arc avoids the supplied nonzero
373 bad-locus series; substitution preserves the unit constants and minor.  The
374 canonical `FormalDivisorAxisLift` then constructs the correction, normalized
375 tangent column, split matrix, and axis conormal. -/
376 theorem exists_tilted_local_axis_lift
377     [Infinite k] [CharZero k]
378     {d : ℕ} {ι : Type v} [Fintype ι] [DecidableEq ι]
379     (q : ι → MvPowerSeries (Fin (d + 1)) k)
380     (bad : MvPowerSeries (Fin (d + 1)) k) (hbad : bad ≠ 0)
381     (rows : Fin d ↪ ι) (chart zero axis : ι) (a b : ℕ)
382     (u₀ u₁ : MvPowerSeries (Fin (d + 1)) k)
383     (hqchart : q chart = 1)
384     (ha : 0 < a) (hab : a < b)
385     (hqzero : q zero = (MvPowerSeries.X 0) ^ a * u₀)
386     (hu₀ : MvPowerSeries.constantCoeff u₀ ≠ 0)
387     (hqaxis : q axis = (MvPowerSeries.X 0) ^ b * u₁)
388     (hu₁ : MvPowerSeries.constantCoeff u₁ ≠ 0)
389     (hminor : MvPowerSeries.constantCoeff
390       (selectedMinor (localTransverseDerivativeMatrix (k := k) q) rows).det ≠ 0) :
391     ∃ α : Fin d → k,
392       tiltedArc (k := k) α bad ≠ 0 ∧
393       (tiltedArc (k := k) α (q zero) =
394         (PowerSeries.X : PowerSeries k) ^ a * tiltedArc (k := k) α u₀) ∧
395       PowerSeries.constantCoeff (tiltedArc (k := k) α u₀) ≠ 0 ∧
396       (tiltedArc (k := k) α (q axis) =

468 /-- In the selected-coordinate chart, the tilted selected coordinate is
469 `β_j + α_j t`; its transverse derivative row is the identity and its raw
470 `t`-derivative is `α_j`. -/
471 theorem selected_coordinate_rows_on_tilted_arc
472     {d : ℕ} {ι : Type*} (q : ι → MvPowerSeries (Fin (d + 1)) k)
473     (rows : Fin d ↪ ι) (β α : Fin d → k)
474     (hrows : ∀ j, q (rows j) =
475       MvPowerSeries.C (β j) + MvPowerSeries.X (Fin.succ j)) :
476     (∀ j, tiltedArc (k := k) α (q (rows j)) =
477       PowerSeries.C (β j) + PowerSeries.C (α j) * PowerSeries.X) ∧
478     (∀ j l, tiltedTransverseDerivativeMatrix (k := k) α q (rows j) l =
479       if j = l then 1 else 0) ∧
480     (∀ j, PowerSeries.derivative (R := k)
481       (tiltedArc (k := k) α (q (rows j))) = PowerSeries.C (α j)) := by
482   have hC (b : k) : (tiltedArcMap (k := k) α) (MvPowerSeries.C b) =
483       PowerSeries.C b := tiltedArc_C (k := k) α b
484   have hX (j : Fin d) : (tiltedArcMap (k := k) α)
485       (MvPowerSeries.X (Fin.succ j)) = PowerSeries.C (α j) * PowerSeries.X :=
486     tiltedArc_X_succ (k := k) α j
487   have hrowArc (j : Fin d) : tiltedArc (k := k) α (q (rows j)) =
488       PowerSeries.C (β j) + PowerSeries.C (α j) * PowerSeries.X := by
489     have h := congrArg (tiltedArcMap (k := k) α) (hrows j)
490     change (tiltedArcMap (k := k) α) (q (rows j)) =
491       (tiltedArcMap (k := k) α) (MvPowerSeries.C (β j) +
492         MvPowerSeries.X (Fin.succ j)) at h
493     rw [map_add, hC, hX] at h
494     exact h
495   have hderiv (j l : Fin d) :
496       MvPowerSeries.pderiv (Fin.succ l) (q (rows j)) =
497         if j = l then (1 : MvPowerSeries (Fin (d + 1)) k) else 0 := by
498     rw [hrows j]
499     by_cases hjl : j = l
500     · subst l
501       simp
502     · have hne : Fin.succ j ≠ Fin.succ l := by
503         intro h
504         exact hjl ((Fin.succ_injective d) h)
505       simp [hjl, MvPowerSeries.pderiv_X_of_ne, hne]
506   constructor
507   · exact hrowArc
508   constructor
509   · intro j l
510     change tiltedArc (k := k) α
511       (MvPowerSeries.pderiv (Fin.succ l) (q (rows j))) = _
512     rw [hderiv j l]
513     by_cases hjl : j = l <;> simp [tiltedArc, hjl]
514   · intro j
515     rw [hrowArc j]
516     rw [map_add, (PowerSeries.derivative (R := k)).leibniz]
517     simp
