49 /-- The compatibility predicate used by the normalized-column adapter. It
50 requires a visible differential frame on the same retained column, including
51 the explicit modulo-`D.t` inclusion. The complete producer is
52 `ExactDivisorialVisibleFrameExistence`. -/
53 def HasCompatibleVisibleFrame
54     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (hm : 0 < m)
55     [CharZero k]
56     (W : Data k (ComponentFractionField P) (componentCoordinate P ⟨0, hm⟩)) :
57   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
58       (ComponentFractionField P) := W.ambientAlgebra
59   (Fin (m + 1) → W.place.valuation.toSubring) → Prop := by
60     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
61         (ComponentFractionField P) := W.ambientAlgebra
62     let V := W.place.valuation.toSubring
63     letI : IsDiscreteValuationRing V := W.place.isDiscrete
64     exact fun q ↦
65       ∃ D : VisibleDivisorFrame (V := V)
66           (KaehlerDifferential.D k (ComponentFractionField P)) (Fin m),
67         D.Q₀ = q 0 ∧
68           D.Q₁ = q (Fin.succ ⟨0, hm⟩) ∧
69           ∀ j, D.Q j = q (Fin.succ j)

71 /-- Once the compatible frame is supplied, the retained data, common-scale
72 normalization, and exact parameter identity are exactly the fields required by

74 theorem hasVisibleDivisorFrame_of_compatible_normalized_column
75     [CharZero k]
76     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (hm : 0 < m)
77     (W : Data k (ComponentFractionField P) (componentCoordinate P ⟨0, hm⟩)) :
78     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
79         (ComponentFractionField P) := W.ambientAlgebra
80     let V := W.place.valuation.toSubring
81     letI : IsDiscreteValuationRing V := W.place.isDiscrete
82     ∀ (q : Fin (m + 1) → V) (scale : ComponentFractionField P),
83       q 0 ≠ 0 →
84       (∀ a, (q a : ComponentFractionField P) =
85         scale * componentProjectivePoint P a) →
86       q (Fin.succ ⟨0, hm⟩) = q 0 * W.place.parameter →
87       HasCompatibleVisibleFrame P hm W q →
88       HasVisibleDivisorFrame P hm := by
89   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
90       (ComponentFractionField P) := W.ambientAlgebra
91   let V := W.place.valuation.toSubring
92   letI : IsDiscreteValuationRing V := W.place.isDiscrete
93   dsimp only
94   intro q scale hq0 hq hratio hframe
95   obtain ⟨D, hQ₀, hQ₁, hQ⟩ := hframe
96   exact ⟨W, q, scale, hq0, hq, hratio, D, hQ₀, hQ₁, hQ⟩

98 /-- A trust-zero prefix exposing normalized retained-place data and the
99 compatibility premise consumed by the adapter. The complete higher-dimensional
100 producer is `ExactDivisorialVisibleFrameExistence`. -/
101 theorem exists_normalized_column_with_exact_frame_obligation
102     [CharZero k]
103     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (hm : 0 < m)
104     (hi : Transcendental k (componentCoordinate P ⟨0, hm⟩)) :
105     ∃ W : Data k (ComponentFractionField P) (componentCoordinate P ⟨0, hm⟩),
106       letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
107           (ComponentFractionField P) := W.ambientAlgebra
108       let V := W.place.valuation.toSubring
109       letI : IsDiscreteValuationRing V := W.place.isDiscrete
110       ∃ (chart : Fin (m + 1)) (q : Fin (m + 1) → V)
111           (scale : ComponentFractionField P),
112         scale ≠ 0 ∧ q chart = 1 ∧ q 0 ≠ 0 ∧
113         (∀ a, (q a : ComponentFractionField P) =
114           scale * componentProjectivePoint P a) ∧
115         q (Fin.succ ⟨0, hm⟩) = q 0 * W.place.parameter ∧
116         (HasCompatibleVisibleFrame P hm W q → HasVisibleDivisorFrame P hm) := by
117   obtain ⟨W, chart, q, scale, hscale, hchart, hq0, hq, hratio⟩ :=
118     exists_normalizedProjectivePoint_relativeRetainedBoundaryPlace P ⟨0, hm⟩ hi
119   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
120       (ComponentFractionField P) := W.ambientAlgebra
121   let V := W.place.valuation.toSubring
122   letI : IsDiscreteValuationRing V := W.place.isDiscrete
123   refine ⟨W, chart, q, scale, hscale, hchart, hq0, hq, hratio, ?_⟩
124   exact hasVisibleDivisorFrame_of_compatible_normalized_column P hm W q scale
125     hq0 hq hratio

168 /-- A proposition packaging retained `Data`, a normalized projective column,

171 the canonical application. -/
172 def HasNormalizedCompatibleVisibleFrame
173     [CharZero k]
174     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (hm : 0 < m) : Prop :=
175   ∃ W : Data k (ComponentFractionField P) (componentCoordinate P ⟨0, hm⟩),
176     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
177         (ComponentFractionField P) := W.ambientAlgebra
178     let V := W.place.valuation.toSubring
179     letI : IsDiscreteValuationRing V := W.place.isDiscrete
180     ∃ (chart : Fin (m + 1)) (q : Fin (m + 1) → V)
181         (scale : ComponentFractionField P),
182       scale ≠ 0 ∧ q chart = 1 ∧ q 0 ≠ 0 ∧
183       (∀ a, (q a : ComponentFractionField P) =
184         scale * componentProjectivePoint P a) ∧
185       q (Fin.succ ⟨0, hm⟩) = q 0 * W.place.parameter ∧

198 /-- Repaired residual with exactly the ambient hypotheses of the actual lane C
199 consumer.  The target-context hypotheses are retained because, in particular,
200 axis avoidance is needed to force the homogeneous denominator to vanish; the
201 component-only strengthening is false in general. -/
202 def ExactDivisorialVisibleFrameExistence : Prop :=
203   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
204     (n N : ℕ) (d : WeylAlg k (n + 1)),
205     0 < n →
206     Disjoint
207       (orderCharacteristicSupport k
208         (canonicalRightIdeal (presentedCoordinate k n) d N))
209       (PrimeSpectrum.zeroLocus
210         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
211           Set (SymbolRing k (n + 1)))) →
212     ∀ P : PrimeSpectrum (MvPolynomial (Fin (n + 1)) k),
213       P.asIdeal ∈
214         (reducedOrderBaseIdeal k
215           (canonicalRightIdeal (presentedCoordinate k n) d N)).minimalPrimes →
216       Transcendental k
217         (componentCoordinate P ⟨0, Nat.zero_lt_succ n⟩) →
218       HasNormalizedCompatibleVisibleFrame P (Nat.zero_lt_succ n)
219 
220 /-- The repaired residual has a direct, trust-zero adapter to the actual
221 higher-dimensional producer. -/
222 theorem higherDimensionalCanonicalVisibleDivisorFrameProduction_of_exactResidual
223     (h : ExactDivisorialVisibleFrameExistence.{u}) :
224     HigherDimensionalCanonicalVisibleDivisorFrameProduction.{u} := by
225   intro k _ _ _ n N d hn hdisjoint P hP htrans
226   exact hasVisibleDivisorFrame_of_normalizedCompatibleVisibleFrame P
227     (Nat.zero_lt_succ n) (h k n N d hn hdisjoint P hP htrans)
