12 # Laurent conormal axes for arbitrary affine components
13 
14 The prime-component visible frame passes through the generic finite-gradient
15 adapter to a Laurent equation-conormal point with the prescribed residue.
16 This is an algebraic witness; the comparison with smooth projective conormal
17 directions is a separate statement.

53 theorem exists_groundConormalAxis_of_minimalPrime_unit_transcendental
54     {k : Type u} [Field k] [CharZero k]
55     {m : ℕ} (hm : 0 < m)
56     (I : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
57     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
58     (hP : P.asIdeal ∈ I.minimalPrimes)
59     (hunit : ∃ g : MvPolynomial (Fin m) k,
60       MvPolynomial.X ⟨0, hm⟩ * g - 1 ∈ P.asIdeal)
61     (htrans : Transcendental k (componentCoordinate P ⟨0, hm⟩)) :
62     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
63       (y : Fin m → LaurentSeries K) (xi : Fin m → PowerSeries K),
64       Sum.elim y
65           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
66         groundEquationConormalLocus (k := k) (K := K) I ∧
67       residueColumn xi =
68         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
69   classical
70   obtain ⟨w⟩ :=
71     generalDivisorialVisibleFrameWithResidueAlgebraicity hm P hunit htrans
72   let C := w.column
73   let W := C.W
74   let F := ComponentFractionField P
75   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
76   let V := W.place.valuation.toSubring
77   letI : IsDiscreteValuationRing V := W.place.isDiscrete
78   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
79   letI : Algebra W.coefficientField V :=
80     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
81   let κ := ResidueField V
82   letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
83   letI : CharZero κ :=
84     charZero_of_injective_algebraMap (FaithfulSMul.algebraMap_injective k κ)
85   let q : Fin (m + 1) → PowerSeries κ :=
86     fun a ↦ retainedToCompletedPowerSeries W (C.q a)
87   obtain ⟨data⟩ :=
88     exists_regularizedOneRowConormalData_of_actual_witness
89       hm I hI P hP w hunit
90   obtain ⟨certificate⟩ :=
91     finiteGradientBoundaryCertificateOver_of_regularizedOneRowConormalData
92       (k := k) (K := κ) hm I q data
93   obtain ⟨y, xi, hmem, hres⟩ :=
94     exists_groundConormalAxis_of_finiteGradientBoundaryCertificateOver
95       (k := k) (K := κ) hm I certificate
96   exact ⟨κ, inferInstance, inferInstance, y, xi, hmem, hres⟩
