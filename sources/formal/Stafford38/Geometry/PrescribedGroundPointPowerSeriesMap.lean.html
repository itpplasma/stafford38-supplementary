30 /-- Ground scalars pass through the prescribed completion as the same constants
31 in the multivariate power-series chart. -/
32 theorem prescribedCompletion_constant
33     [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
34     [Algebra.FormallyEtale R (Localization.AtPrime M)] (c : k) :
35     powerSeriesCompletionAtGroundPoint (σ := σ) M eM (MvPowerSeries.C c) =
36       AdicCompletion.of J T (algebraMap k T c) := by
37   let x := residueCoordinates (σ := σ) M eM
38   let q := Stafford38.Geometry.AffinePointCompletion.pointIdeal x
39   letI : q.IsMaximal := residuePoint_isMaximal (σ := σ) M eM
40   letI : q.IsPrime := inferInstance
41   let ρ := residueEvaluation (σ := σ) M eM
42   letI : Algebra R k := ρ.toRingHom.toAlgebra
43   letI : IsScalarTower R k k := IsScalarTower.of_algebraMap_eq (fun _ => by simp)
44   let eQ := residuePointQuotientEquiv (σ := σ) M eM
45   letI : IsScalarTower R k (B ⧸ M) := IsScalarTower.of_algebraMap_eq (fun a => by
46     apply eM.injective
47     rw [← Ideal.Quotient.mk_algebraMap R M a, eM.commutes]
48     change eM (Ideal.Quotient.mk M (algebraMap R B a)) = ρ a
49     rfl)
50   letI : IsScalarTower R k (R ⧸ q) := IsScalarTower.of_algebraMap_eq (fun a => by
51     apply eQ.injective
52     calc
53       eQ (algebraMap R (R ⧸ q) a) = ρ a := by
54         change eQ (Ideal.Quotient.mk q a) = ρ a
55         simpa [ρ] using residuePointQuotientEquiv_apply (σ := σ) M eM a
56       _ = eQ (algebraMap k (R ⧸ q) (ρ a)) := by
57         rw [eQ.commutes]
58         rfl)
59   have hq : Ideal.under R M = q := by
60     simpa [x, q] using residuePoint_under_eq (σ := σ) M eM
61   let e := residueEquivOverCoordinates (σ := σ) M eM
62   letI : IsScalarTower k R T := inferInstance
63   have hscalar : algebraMap R T (MvPolynomial.C c) = algebraMap k T c := by
64     change algebraMap R T ((algebraMap k R) c) = algebraMap k T c
65     exact (IsScalarTower.algebraMap_apply k R T c).symm
66   have hAffine :
67       Stafford38.Geometry.AffinePointCompletion.powerSeriesCompletionAtPoint x
68           (MvPowerSeries.C c) = AdicCompletion.of q R (MvPolynomial.C c) := by
69     simpa [Stafford38.Geometry.AffinePointCompletion.powerSeriesCompletionAtPoint] using
70       Stafford38.Geometry.SmoothLocalParameterCompletion.affinePointCompletion_constant_explicit
71         x c
72   change (Stafford38.Geometry.EtalePointCompletion.completionEquivOfEtaleResidue
73     q M hq e) (Stafford38.Geometry.AffinePointCompletion.powerSeriesCompletionAtPoint x
74       (MvPowerSeries.C c)) = _
75   rw [hAffine]
76   rw [← hscalar]
77   exact Stafford38.Geometry.PrescribedAffineResidueCompletion.etaleCompletion_apply_of
78     q M hq e (MvPolynomial.C c)

80 /-- The local ring at the prescribed point maps canonically to its power-series
81 completion chart. The map is the inverse completed-chart equivalence after the
82 canonical dense map into the local completion. -/
83 def localToPowerSeries
84     [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
85     [Algebra.FormallyEtale R (Localization.AtPrime M)] :
86     T →ₐ[k] MvPowerSeries σ k := by
87   let E := powerSeriesCompletionAtGroundPoint (σ := σ) M eM
88   let f : T →+* MvPowerSeries σ k :=
89     (E.symm : _ ≃+* _).toRingHom.comp
90       (algebraMap T (AdicCompletion J T))
91   refine { toRingHom := f, commutes' := ?_ }
92   intro c
93   apply E.injective
94   calc
95     E (f (algebraMap k T c)) = AdicCompletion.of J T (algebraMap k T c) := by
96       simp [f, AdicCompletion.algebraMap_apply]
97     _ = E (algebraMap k (MvPowerSeries σ k) c) := by
98       rw [show algebraMap k (MvPowerSeries σ k) c = MvPowerSeries.C c by rfl]
99       exact (prescribedCompletion_constant (σ := σ) M eM c).symm

125 /-- The source coordinate is sent to its residue constant plus the centered formal
126 variable. This is the coefficient-preserving form of the prescribed-chart map. -/
127 theorem originalAlgebraToPowerSeries_coordinate
128     [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
129     [Algebra.FormallyEtale R (Localization.AtPrime M)] (i : σ) :
130     originalAlgebraToPowerSeries (σ := σ) M eM
131         (algebraMap R B (MvPolynomial.X i)) =
132       MvPowerSeries.C (residueCoordinates (σ := σ) M eM i) + MvPowerSeries.X i := by
133   let x := residueCoordinates (σ := σ) M eM
134   let E := powerSeriesCompletionAtGroundPoint (σ := σ) M eM
135   letI : IsScalarTower k R T := inferInstance
136   letI : IsScalarTower R B T := inferInstance
137   have hscalar : algebraMap R T (MvPolynomial.C (x i)) = algebraMap k T (x i) := by
138     change algebraMap R T ((algebraMap k R) (x i)) = algebraMap k T (x i)
139     exact (IsScalarTower.algebraMap_apply k R T (x i)).symm
140   have hcoordinates :
141       algebraMap B T (algebraMap R B (MvPolynomial.X i)) =
142         algebraMap k T (x i) +
143           algebraMap B T (algebraMap R B
144             (MvPolynomial.X i - MvPolynomial.C (x i))) := by
145     have hpoly : algebraMap R T (MvPolynomial.X i) =
146         algebraMap R T (MvPolynomial.C (x i)) +
147           algebraMap R T (MvPolynomial.X i - MvPolynomial.C (x i)) := by
148       rw [← map_add]
149       congr 1
150       ring
151     calc
152       algebraMap B T (algebraMap R B (MvPolynomial.X i)) =
153           algebraMap R T (MvPolynomial.X i) := by
154         change ((algebraMap B T).comp (algebraMap R B)) _ = _
155         rw [← IsScalarTower.algebraMap_eq R B T]
156       _ = algebraMap k T (x i) + algebraMap R T
157             (MvPolynomial.X i - MvPolynomial.C (x i)) := by
158         rw [hpoly, hscalar]
159       _ = algebraMap k T (x i) +
160           algebraMap B T (algebraMap R B
161             (MvPolynomial.X i - MvPolynomial.C (x i))) := by
162         exact congrArg (fun y => algebraMap k T (x i) + y)
163           (IsScalarTower.algebraMap_apply R B T _)
164   have hcenter : E (MvPowerSeries.X i) =
165       AdicCompletion.of J T (algebraMap R T
166         (MvPolynomial.X i - MvPolynomial.C (x i))) := by
167     simpa [E, x] using
168       Stafford38.Geometry.PrescribedAffineResidueCompletionConsumer.prescribedCompletion_centeredGenerator
169         (σ := σ) M eM i
170   have hcenter' : E (MvPowerSeries.X i) =
171       AdicCompletion.of J T (algebraMap B T (algebraMap R B
172         (MvPolynomial.X i - MvPolynomial.C (x i)))) := by
173     calc
174       E (MvPowerSeries.X i) =
175           AdicCompletion.of J T (algebraMap R T
176             (MvPolynomial.X i - MvPolynomial.C (x i))) := hcenter
177       _ = AdicCompletion.of J T (algebraMap B T (algebraMap R B
178             (MvPolynomial.X i - MvPolynomial.C (x i)))) := by
179         exact congrArg (AdicCompletion.of J T)
180           (IsScalarTower.algebraMap_apply R B T _)
181   apply E.injective
182   calc
183     E (originalAlgebraToPowerSeries (σ := σ) M eM
184         (algebraMap R B (MvPolynomial.X i))) =
185         AdicCompletion.of J T (algebraMap B T (algebraMap R B (MvPolynomial.X i))) := by
186       rw [originalAlgebraToPowerSeries_apply, localToPowerSeries_apply]
187       change E (E.symm (AdicCompletion.of J T
188         (algebraMap B T (algebraMap R B (MvPolynomial.X i))))) = _
189       exact E.apply_symm_apply _
190     _ = AdicCompletion.of J T
191           (algebraMap k T (x i) + algebraMap B T (algebraMap R B
192             (MvPolynomial.X i - MvPolynomial.C (x i)))) := by
193       exact congrArg (AdicCompletion.of J T) hcoordinates
194     _ = AdicCompletion.of J T (algebraMap k T (x i)) +
195         AdicCompletion.of J T (algebraMap B T (algebraMap R B
196           (MvPolynomial.X i - MvPolynomial.C (x i)))) := by
197       simp
198     _ = E (MvPowerSeries.C (x i) + MvPowerSeries.X i) := by
199       rw [map_add, prescribedCompletion_constant (σ := σ) M eM (x i), hcenter']

366 /-- The genuine local completion chart loses no local-ring elements: its
367 completion factor is injective by the Noetherian local Krull intersection
368 property, and the completed coordinate chart is an equivalence. -/
369 theorem localToPowerSeries_injective
370     (M : Ideal B) [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
371     [Algebra.FormallyEtale (MvPolynomial σ k) (Localization.AtPrime M)] :
372     Function.Injective (localToPowerSeries (σ := σ) M eM) := by
373   intro a b hab
374   apply PrescribedCompletionNonzero.localizationCompletion_of_injective
375     (k := k) (σ := σ) (B := B) M
376   have h := congrArg (PrescribedAffineResidueCompletion.powerSeriesCompletionAtGroundPoint
377     (σ := σ) M eM) hab
378   simpa only [localToPowerSeries_apply, RingEquiv.apply_symm_apply,
379     AdicCompletion.algebraMap_apply, Algebra.algebraMap_self_apply] using h
