24 /-- A centered coordinate vanishes at the chosen residue point but gives the
25 corresponding nonzero formal variable in the completed local chart. This is a
26 concrete check that bad-locus nonvanishing in the completion does not require
27 the closed point to avoid that locus. -/
28 theorem centeredCoordinate_vanishes_but_series_is_nonzero
29     (M : Ideal B) [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
30     [Algebra.EssFiniteType R B]
31     [Algebra.FormallyEtale R (Localization.AtPrime M)]
32     [IsDomain B]
33     (hinjR : Function.Injective (algebraMap R B))
34     (i : σ) :
35     residueEvaluation (σ := σ) M eM
36         (MvPolynomial.X i -
37           MvPolynomial.C (residueCoordinates (σ := σ) M eM i)) = 0 ∧
38     (powerSeriesCompletionAtGroundPoint (σ := σ) M eM).symm
39       (AdicCompletion.of
40         (IsLocalRing.maximalIdeal (Localization.AtPrime M))
41         (Localization.AtPrime M)
42         (algebraMap R (Localization.AtPrime M)
43           (MvPolynomial.X i -
44             MvPolynomial.C (residueCoordinates (σ := σ) M eM i)))) =
45       MvPowerSeries.X i ∧
46     (powerSeriesCompletionAtGroundPoint (σ := σ) M eM).symm
47       (AdicCompletion.of
48         (IsLocalRing.maximalIdeal (Localization.AtPrime M))
49         (Localization.AtPrime M)
50         (algebraMap R (Localization.AtPrime M)
51           (MvPolynomial.X i -
52             MvPolynomial.C (residueCoordinates (σ := σ) M eM i)))) ≠ 0 := by
53   let f : R := MvPolynomial.X i -
54     MvPolynomial.C (residueCoordinates (σ := σ) M eM i)
55   have hpoint : residueEvaluation (σ := σ) M eM f = 0 := by
56     rw [residueEvaluation_eq_aeval]
57     simp [f]
58   have hf : f ≠ 0 := by
59     intro h
60     have hi : Finsupp.single i 1 ≠ (0 : σ →₀ ℕ) := by
61       intro hsingle
62       have hcoord := congrArg (fun d : σ →₀ ℕ => d i) hsingle
63       simp at hcoord
64     have hc := congrArg (MvPolynomial.coeff (Finsupp.single i 1)) h
65     dsimp [f] at hc
66     simp only [MvPolynomial.coeff_sub, MvPolynomial.coeff_X_same,
67       MvPolynomial.coeff_C_of_ne_zero hi] at hc
68     exact one_ne_zero (sub_eq_zero.mp hc)
69   have hB : algebraMap R B f ≠ 0 := by
70     intro h
71     apply hf
72     exact hinjR (by simpa using h)
73   have hmap :
74       algebraMap R (Localization.AtPrime M) f =
75         algebraMap B (Localization.AtPrime M) (algebraMap R B f) :=
76     IsScalarTower.algebraMap_apply R B (Localization.AtPrime M) f
77   let E := powerSeriesCompletionAtGroundPoint (σ := σ) M eM
78   have hcoordinate : E (MvPowerSeries.X i) =
79       AdicCompletion.of
80         (IsLocalRing.maximalIdeal (Localization.AtPrime M))
81         (Localization.AtPrime M) (algebraMap R (Localization.AtPrime M) f) := by
82     simpa [E, f] using prescribedCompletion_centeredGenerator (σ := σ) M eM i
83   have hseries : E.symm
84       (AdicCompletion.of
85         (IsLocalRing.maximalIdeal (Localization.AtPrime M))
86         (Localization.AtPrime M) (algebraMap R (Localization.AtPrime M) f)) =
87       MvPowerSeries.X i := by
88     apply E.injective
89     rw [RingEquiv.apply_symm_apply]
90     exact hcoordinate.symm
91   refine ⟨?_, ?_, ?_⟩
92   · simpa [f] using hpoint
93   · simpa [f, E] using hseries
94   · have hnonzero :=
95       prescribedSeries_image_ne_zero (k := k) (σ := σ) (B := B) M eM
96         (algebraMap R B f) hB
97     rw [← hmap] at hnonzero
98     exact hnonzero
