6 /-!
7 # Annihilation by the completed chart arc
8 
9 `CompletedProjectiveBoundaryChart` stores affine-ideal containment after
10 dehomogenizing in coordinate zero, while its normalized power-series chart
11 coordinate is the unrelated index `chart`.  This file makes the necessary
12 coordinate repair explicit.
13 
14 The minimal strengthened certificate below retains the existing completed
15 chart and asks for the same kind of ideal containment after dehomogenizing in
16 its actual normalized chart, with precisely the coordinate ordering used by
17 `completedChartAffineArc`.  Projective equation vanishing then proves arc
18 annihilation; annihilation itself is not a field of the certificate.
19 -/

36 /-- Insert affine coordinates into an arbitrary projective chart, using the
37 same complement ordering as `completedChartAffineArc`. -/
38 abbrev projectiveChartPoint {m : ℕ} (chart : Fin (m + 1))
39     (y : Fin m → K) : Fin (m + 1) → K :=
40   insertProjectiveChart chart (chartAffineCoordinateEquiv chart) y
41 
42 /-- Substitute the chart coordinate by one and the remaining projective
43 coordinates by the correspondingly ordered affine variables. -/
44 abbrev projectiveDehomogenizeAt {m : ℕ} (chart : Fin (m + 1)) :
45     MvPolynomial (Fin (m + 1)) K →ₐ[K] MvPolynomial (Fin m) K :=
46   dehomogenizeProjectiveChart chart (chartAffineCoordinateEquiv chart)
47 
48 /-- Evaluation after arbitrary-chart dehomogenization is evaluation at the
49 inserted projective chart point. -/
50 theorem eval_projectiveDehomogenizeAt {m : ℕ} (chart : Fin (m + 1))
51     (p : MvPolynomial (Fin (m + 1)) K) (y : Fin m → K) :
52     MvPolynomial.eval y (projectiveDehomogenizeAt chart p) =
53       MvPolynomial.eval (projectiveChartPoint chart y) p := by
54   exact eval_dehomogenizeProjectiveChart chart (chartAffineCoordinateEquiv chart) p y
55 
56 /-- The affine ideal generated by dehomogenizing projective equations in the
57 specified chart. -/
58 abbrev chartDehomogenizedEquationIdeal {m : ℕ} {ι : Type*}
59     (chart : Fin (m + 1))
60     (equations : ι → MvPolynomial (Fin (m + 1)) K) :
61     Ideal (MvPolynomial (Fin m) K) :=
62   Stafford38.Geometry.ProjectiveChartCoordinates.dehomogenizedEquationIdeal
63     chart (chartAffineCoordinateEquiv chart) equations
64 
65 /-- Every polynomial in the chart-dehomogenized equation ideal vanishes at
66 an affine chart point whose inserted projective point kills the supplied
67 projective equations. -/
68 theorem eval_eq_zero_of_mem_chartDehomogenizedEquationIdeal
69     {m : ℕ} {ι : Type*} (chart : Fin (m + 1))
70     (equations : ι → MvPolynomial (Fin (m + 1)) K)
71     (y : Fin m → K)
72     (hvanish : ∀ i,
73       MvPolynomial.eval (projectiveChartPoint chart y) (equations i) = 0)
74     (f : MvPolynomial (Fin m) K)
75     (hf : f ∈ chartDehomogenizedEquationIdeal chart equations) :
76     MvPolynomial.eval y f = 0 := by
77   exact Stafford38.Geometry.ProjectiveChartCoordinates.eval_eq_zero_of_mem_dehomogenizedEquationIdeal
78     chart (chartAffineCoordinateEquiv chart) equations y hvanish f hf

80 /-! ## Minimal completed-chart repair -/
81 
82 /-- A completed projective chart together with affine-ideal containment in
83 the chart which is actually normalized to one.
84 
85 This adds only a chart-relative algebraic containment, parallel to the
86 coordinate-zero containment already stored in `CompletedProjectiveBoundaryChart`.
87 It does not assume that the resulting arc annihilates the ideal. -/
88 structure ChartArcCompatibleBoundaryChart
89     (k : Type u) [Field k] [CharZero k]
90     (m : ℕ) (hm : 0 < m)
91     (I : Ideal (MvPolynomial (Fin m) k)) where
92   toCompleted : CompletedProjectiveBoundaryChart k m hm I
93   chart_ideal_containment :
94     I.map (scalarPolynomialMap
95       (k := k) (K := LaurentSeries k) (Fin m)) ≤
96       chartDehomogenizedEquationIdeal toCompleted.chart
97         toCompleted.equations
98 
99 /-- Inserting the completed affine chart arc recovers the original
100 projective Laurent column exactly. -/
101 theorem projectiveChartPoint_completedChartAffineArc
102     {k : Type u} [Field k] [CharZero k]
103     {m : ℕ} (hm : 0 < m)
104     (I : Ideal (MvPolynomial (Fin m) k))
105     (W : CompletedProjectiveBoundaryChart k m hm I) :
106     projectiveChartPoint W.chart
107         (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k)
108           (completedChartAffineArc hm I W i)) =
109       laurentColumn W.q := by
110   funext a
111   by_cases h : a = W.chart
112   · subst a
113     simp [projectiveChartPoint, insertProjectiveChart, laurentColumn, W.q_chart]
114   · let i : Fin m :=
115       (chartAffineCoordinateEquiv W.chart).symm ⟨a, h⟩
116     have hi : chartAffineCoordinateEquiv W.chart i = ⟨a, h⟩ :=
117       (chartAffineCoordinateEquiv W.chart).apply_symm_apply ⟨a, h⟩
118     simp [projectiveChartPoint, insertProjectiveChart, h, completedChartAffineArc, laurentColumn,
119       i, hi]

121 /-- The minimally repaired completed chart proves, rather than stores, that
122 its actual power-series affine chart arc annihilates the boundary ideal. -/
123 theorem completedChartAffineArc_annihilates
124     {k : Type u} [Field k] [CharZero k]
125     {m : ℕ} (hm : 0 < m)
126     (I : Ideal (MvPolynomial (Fin m) k))
127     (W : ChartArcCompatibleBoundaryChart k m hm I) :
128     ∀ f ∈ I,
129       MvPolynomial.eval₂ (algebraMap k (PowerSeries k))
130         (completedChartAffineArc hm I W.toCompleted) f = 0 := by
131   intro f hf
132   let qLaurent : Fin m → LaurentSeries k := fun i ↦
133     algebraMap (PowerSeries k) (LaurentSeries k)
134       (completedChartAffineArc hm I W.toCompleted i)
135   have hprojective : ∀ j,
136       MvPolynomial.eval
137         (projectiveChartPoint W.toCompleted.chart qLaurent)
138         (W.toCompleted.equations j) = 0 := by
139     intro j
140     rw [projectiveChartPoint_completedChartAffineArc]
141     exact W.toCompleted.equations_vanish j
142   have hlaurent :
143       MvPolynomial.eval₂ (algebraMap k (LaurentSeries k)) qLaurent f = 0 := by
144     have hmem := W.chart_ideal_containment
145       (Ideal.mem_map_of_mem
146         (scalarPolynomialMap
147           (k := k) (K := LaurentSeries k) (Fin m)) hf)
148     have heval := eval_eq_zero_of_mem_chartDehomogenizedEquationIdeal
149       W.toCompleted.chart W.toCompleted.equations qLaurent hprojective
150       (scalarPolynomialMap
151         (k := k) (K := LaurentSeries k) (Fin m) f) hmem
152     simpa [eval_scalarPolynomialMap] using heval
153   apply HahnSeries.ofPowerSeries_injective (Γ := ℤ)
154   have hmap := MvPolynomial.eval₂_comp_left
155     (algebraMap (PowerSeries k) (LaurentSeries k))
156     (algebraMap k (PowerSeries k))
157     (completedChartAffineArc hm I W.toCompleted) f
158   have hcoeff :
159       (algebraMap (PowerSeries k) (LaurentSeries k)).comp
160           (algebraMap k (PowerSeries k)) =
161         algebraMap k (LaurentSeries k) := by
162     ext a
163     simp [PowerSeries.algebraMap_eq, PowerSeries.coe_C,
164       LaurentSeries.algebraMap_apply]
165   rw [hcoeff] at hmap
166   have hpoint :
167       (algebraMap (PowerSeries k) (LaurentSeries k)) ∘
168           completedChartAffineArc hm I W.toCompleted = qLaurent := by
169     rfl
170   rw [hpoint, hlaurent] at hmap
171   simpa using hmap
