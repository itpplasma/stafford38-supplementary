34   let M := Submonoid.powers (selectedAffineChartDenominator P j)
35   let N := Submonoid.powers (hsel (selectedAffineChartDenominator P j))

48 private theorem selectedChartAwayEquiv_invSelf

51     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (j : Fin m)
52     (hsel : SelectedAffineChartQuotient (k := k) P j ≃ₐ[k] Q) :
53     selectedChartAwayEquivOfQuotientEquiv P j hsel
54       (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) =
55       IsLocalization.Away.invSelf (hsel (selectedAffineChartDenominator P j)) := by

57   let S := SelectedAffineChartLocalization (k := k) P j
58   let T := Localization.Away (hsel (selectedAffineChartDenominator P j))
59   let M := Submonoid.powers (selectedAffineChartDenominator P j)
60   let N := Submonoid.powers (hsel (selectedAffineChartDenominator P j))

62   letI : IsLocalization.Away
63       (hsel.toRingEquiv.toRingHom (selectedAffineChartDenominator P j)) T :=
64     Localization.isLocalization
65   have hmap : Submonoid.map hsel M = N := by
66     simp [M, N, Submonoid.map_powers]
67   let E := selectedChartAwayEquivOfQuotientEquiv P j hsel
68   have hbase := selectedChartAwayEquiv_base P j hsel
69     (selectedAffineChartDenominator P j)
70   have hprod :
71       algebraMap Q T (hsel (selectedAffineChartDenominator P j)) *
72           E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) = 1 := by
73     calc
74       _ = E (algebraMap R S (selectedAffineChartDenominator P j)) *
75           E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) := by
76             rw [hbase]
77       _ = E (algebraMap R S (selectedAffineChartDenominator P j) *
78           IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) := by

82   calc
83     E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) =
84         1 * E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) := by simp
85     _ = (algebraMap Q T (hsel (selectedAffineChartDenominator P j)) *
86         IsLocalization.Away.invSelf (hsel (selectedAffineChartDenominator P j))) *
87         E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) := by

90         (algebraMap Q T (hsel (selectedAffineChartDenominator P j)) *
91           E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j))) := by ac_rfl
92     _ = IsLocalization.Away.invSelf (hsel (selectedAffineChartDenominator P j)) * 1 := by

96 /-- The original affine variable in the common open is the selected chart

98 is the multiplicative identity before any map to the valuation ring or field. -/
99 theorem originalGenerator_commonOpen_identity
100     {k : Type u} [Field k] {m : ℕ}
101     {Q : Type v} [CommRing Q] [Algebra k Q]
102     {B : Type w} [CommRing B] [Algebra Q B]
103     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (j i : Fin m)
104     (hij : i ≠ j) (hxj : componentCoordinate P j ≠ 0)
105     (hsel : SelectedAffineChartQuotient (k := k) P j ≃ₐ[k] Q)
106     (M : Ideal B) [M.IsPrime] (f : Q)
107     (e : Localization.Away f ≃ₐ[Q]
108       Localization.Away (algebraMap Q B f)) :
109     let g := hsel (selectedAffineChartDenominator P j)
110     let Cq := genericOpenRing M f e
111     let C := genericOpenExtraAwayB M f e g
112     letI : Semiring C := (inferInstance : CommSemiring C).toSemiring
113     letI : Algebra Q Cq := inferInstance
114     letI : Algebra Cq C := inferInstance
115     letI : Algebra Q C := Algebra.compHom C (algebraMap Q Cq)
116     algebraMap Cq C
117         (algebraMap Q Cq
118           (hsel (selectedAffineChartVariableClass P j i hij))) =
119       algebraMap Cq C (algebraMap Q Cq g) *
120         originalAffineChartToCommonOpen P j hxj hsel M f e
121           (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i)) := by
122   dsimp
123   let g := hsel (selectedAffineChartDenominator P j)
124   let Cq := genericOpenRing M f e
125   let C := genericOpenExtraAwayB M f e g
126   let A₀ := OriginalAffineChartQuotient (k := k) P
127   let S := OriginalAffineChartLocalization (k := k) P j
128   let T := Localization.Away g
129   letI : Semiring C := (inferInstance : CommSemiring C).toSemiring
130   letI : Algebra Q Cq := inferInstance
131   letI : Algebra Cq C := inferInstance
132   letI : Algebra Q C := Algebra.compHom C (algebraMap Q Cq)
133   have hz : genericOpenBMap M f e (algebraMap Q B g) = algebraMap Q Cq g :=
134     genericOpenBMap_base_eq M f e g
135   letI : IsLocalization.Away (algebraMap Q Cq g) C := by
136     rw [← hz]
137     infer_instance
138   let h : S ≃ₐ[k] T :=
139     (originalAffineChartOverlapEquiv P j hxj).trans
140       (selectedChartAwayEquivOfQuotientEquiv P j hsel)
141   let ψ : T →+* C := IsLocalization.Away.map
142     (S := T) (Q := C) (algebraMap Q Cq) g
143   have hXi :
144       originalAffineChartToCommonOpen P j hxj hsel M f e
145         (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i)) =
146       ψ (algebraMap Q T
147         (hsel (selectedAffineChartVariableClass P j i hij)) *
148         IsLocalization.Away.invSelf g) := by
149     change ψ (h (algebraMap A₀ S
150       (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i)))) = _
151     have hov : h (algebraMap A₀ S
152         (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i))) =
153       algebraMap Q T
154         (hsel (selectedAffineChartVariableClass P j i hij)) *
155           IsLocalization.Away.invSelf g := by
156       change selectedChartAwayEquivOfQuotientEquiv P j hsel
157         ((originalAffineChartOverlapEquiv P j hxj)
158           (algebraMap A₀ S
159             (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X i)))) = _
160       rw [originalAffineChartOverlapEquiv_Xi P j i hij hxj]
161       rw [map_mul, selectedChartAwayEquiv_base, selectedChartAwayEquiv_invSelf]
162     rw [hov]
163   have hψv :
164       ψ (algebraMap Q T

166       algebraMap Cq C (algebraMap Q Cq
167         (hsel (selectedAffineChartVariableClass P j i hij))) := by
168     simp [ψ, IsLocalization.Away.map, IsLocalization.map_mk']

170     simp [ψ, IsLocalization.Away.map, IsLocalization.map_mk']

179       _ = algebraMap Q T (hsel (selectedAffineChartVariableClass P j i hij)) := by

188           (hsel (selectedAffineChartVariableClass P j i hij)) *
189           IsLocalization.Away.invSelf g) := by
190       rw [← map_mul]

193         originalAffineChartToCommonOpen P j hxj hsel M f e

199 theorem selectedGenerator_commonOpen_product
200     {k : Type u} [Field k] {m : ℕ}
201     {Q : Type v} [CommRing Q] [Algebra k Q]
202     {B : Type w} [CommRing B] [Algebra Q B]
203     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (j : Fin m)
204     (hxj : componentCoordinate P j ≠ 0)
205     (hsel : SelectedAffineChartQuotient (k := k) P j ≃ₐ[k] Q)
206     (M : Ideal B) [M.IsPrime] (f : Q)
207     (e : Localization.Away f ≃ₐ[Q]
208       Localization.Away (algebraMap Q B f)) :
209     let g := hsel (selectedAffineChartDenominator P j)
210     let Cq := genericOpenRing M f e
211     let C := genericOpenExtraAwayB M f e g
212     letI : Semiring C := (inferInstance : CommSemiring C).toSemiring
213     letI : Algebra Q Cq := inferInstance
214     letI : Algebra Cq C := inferInstance
215     letI : Algebra Q C := Algebra.compHom C (algebraMap Q Cq)
216     algebraMap Cq C (algebraMap Q Cq g) *
217       originalAffineChartToCommonOpen P j hxj hsel M f e
218         (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X j)) = 1 := by
219   dsimp
220   let g := hsel (selectedAffineChartDenominator P j)
221   let Cq := genericOpenRing M f e
222   let C := genericOpenExtraAwayB M f e g
223   let A₀ := OriginalAffineChartQuotient (k := k) P
224   let S₀ := OriginalAffineChartLocalization (k := k) P j
225   let T := Localization.Away g
226   let E := selectedChartAwayEquivOfQuotientEquiv P j hsel
227   letI : Semiring C := (inferInstance : CommSemiring C).toSemiring
228   letI : Algebra Q Cq := inferInstance
229   letI : Algebra Cq C := inferInstance
230   letI : Algebra Q C := Algebra.compHom C (algebraMap Q Cq)
231   have hz : genericOpenBMap M f e (algebraMap Q B g) = algebraMap Q Cq g :=
232     genericOpenBMap_base_eq M f e g
233   letI : IsLocalization.Away (algebraMap Q Cq g) C := by
234     rw [← hz]
235     infer_instance
236   let ψ : T →+* C := IsLocalization.Away.map
237     (S := T) (Q := C) (algebraMap Q Cq) g
238   have hprod : algebraMap Q T g *
239       E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) = 1 := by
240     change algebraMap Q T g *
241       selectedChartAwayEquivOfQuotientEquiv P j hsel
242         (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j)) = 1
243     rw [selectedChartAwayEquiv_invSelf P j hsel]
244     rw [IsLocalization.Away.mul_invSelf]
245   have hψg : ψ (algebraMap Q T g) = algebraMap Cq C (algebraMap Q Cq g) := by
246     simp [ψ, IsLocalization.Away.map]

248       (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X j)) =
249       ψ (E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j))) := by
250     change ψ (E ((originalAffineChartOverlapEquiv P j hxj)
251       (algebraMap A₀ S₀
252         (Ideal.Quotient.mk P.asIdeal (MvPolynomial.X j))) )) = _
253     rw [originalAffineChartOverlapEquiv_Xj]
254   calc
255     _ = ψ (algebraMap Q T g) *
256         ψ (E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j))) := by
257       rw [hψg, hXj]
258     _ = ψ (algebraMap Q T g *
259         E (IsLocalization.Away.invSelf (selectedAffineChartDenominator P j))) := by
260       rw [map_mul]
261     _ = 1 := by rw [hprod]; simp
