10 # Evaluation after homogenization in the zeroth projective chart
11 
12 These wrappers expose the exact denominator-clearing identity for an arbitrary
13 affine polynomial and its behavior under the canonical power-series-to-Laurent
14 embedding. The homogeneous scaling calculation is reused from
15 `ProjectiveEquationFormalChart`.

31 /-- Homogenization clears the chart denominator by exactly the total-degree
32 power. -/
33 theorem eval_homogenizeAtZero_eq_pow_mul_dehomogenizedPoint
34     (p : MvPolynomial (Fin n) K) (q : Fin (n + 1) → K)
35     (hq₀ : q 0 ≠ 0) :
36     MvPolynomial.eval q (homogenizeAtZero p) =
37       q 0 ^ p.totalDegree * MvPolynomial.eval (dehomogenizedPoint q) p := by
38   rw [eval_eq_pow_mul_eval_normalizedProjectivePoint
39     (homogenizeAtZero p) p.totalDegree
40     (homogenizeAtZero_isHomogeneous p) q (q 0) hq₀]
41   rw [normalizedProjectivePoint_eq_chartPoint q hq₀,
42     ← eval_projectiveDehomogenize (homogenizeAtZero p)
43       (dehomogenizedPoint q),
44     projectiveDehomogenize_homogenizeAtZero]
45 
46 /-- For a nonzero zeroth coordinate, homogenizing preserves nonvanishing of
47 the original affine evaluation. -/
48 theorem eval_homogenizeAtZero_ne_zero_iff
49     (p : MvPolynomial (Fin n) K) (q : Fin (n + 1) → K)
50     (hq₀ : q 0 ≠ 0) :
51     MvPolynomial.eval q (homogenizeAtZero p) ≠ 0 ↔
52       MvPolynomial.eval (dehomogenizedPoint q) p ≠ 0 := by
53   rw [eval_homogenizeAtZero_eq_pow_mul_dehomogenizedPoint p q hq₀]
54   constructor
55   · intro h
56     exact (mul_ne_zero_iff.mp h).2
57   · intro h
58     exact mul_ne_zero (pow_ne_zero _ hq₀) h

60 /-- Coefficient maps commute with taking a homogeneous component. -/
61 private theorem map_homogeneousComponent
62     {R S : Type*} [CommSemiring R] [CommSemiring S]
63     (f : R →+* S) (d : ℕ) (p : MvPolynomial (Fin n) R) :
64     MvPolynomial.map f (MvPolynomial.homogeneousComponent d p) =
65       MvPolynomial.homogeneousComponent d (MvPolynomial.map f p) := by
66   classical
67   ext e
68   by_cases h : e.degree = d
69   · simp [MvPolynomial.coeff_map, MvPolynomial.coeff_homogeneousComponent, h]
70   · simp [MvPolynomial.coeff_map, MvPolynomial.coeff_homogeneousComponent, h]
71 
72 private theorem totalDegree_map_eq_of_injective
73     {R S : Type*} [CommSemiring R] [CommSemiring S]
74     (f : R →+* S) (hf : Function.Injective f)
75     (p : MvPolynomial (Fin n) R) :
76     (MvPolynomial.map f p).totalDegree = p.totalDegree := by
77   classical
78   simp [MvPolynomial.totalDegree, MvPolynomial.support_map_of_injective p hf]

80 /-- Coefficient maps between fields commute with zeroth-chart homogenization. -/
81 theorem map_homogenizeAtZero
82     {R S : Type*} [Field R] [Field S]
83     (f : R →+* S) (p : MvPolynomial (Fin n) R) :
84     MvPolynomial.map f (homogenizeAtZero p) =
85       homogenizeAtZero (MvPolynomial.map f p) := by
86   have hf : Function.Injective f := RingHom.injective f
87   simp only [homogenizeAtZero, map_sum, map_mul, map_pow,
88     MvPolynomial.map_X, MvPolynomial.map_rename, map_homogeneousComponent]
89   rw [totalDegree_map_eq_of_injective f hf p]

93 /-- The denominator-clearing identity after embedding power-series coordinates
94 and polynomial coefficients into Laurent series. -/
95 theorem laurent_eval_map_homogenizeAtZero_eq_pow_mul_dehomogenizedPoint
96     (p : MvPolynomial (Fin n) κ)
97     (q : Fin (n + 1) → PowerSeries κ) (hq₀ : q 0 ≠ 0) :
98     MvPolynomial.eval (laurentColumn q)
99         (MvPolynomial.map (algebraMap κ (LaurentSeries κ))
100           (homogenizeAtZero p)) =
101       (laurentColumn q 0) ^ p.totalDegree *
102     MvPolynomial.eval (dehomogenizedPoint (laurentColumn q))
103           (MvPolynomial.map (algebraMap κ (LaurentSeries κ)) p) := by
104   calc
105     MvPolynomial.eval (laurentColumn q)
106         (MvPolynomial.map (algebraMap κ (LaurentSeries κ))
107           (homogenizeAtZero p)) =
108         MvPolynomial.eval (laurentColumn q)
109           (homogenizeAtZero
110             (MvPolynomial.map (algebraMap κ (LaurentSeries κ)) p)) := by
111           rw [map_homogenizeAtZero]
112     _ = (laurentColumn q 0) ^
113           (MvPolynomial.map (algebraMap κ (LaurentSeries κ)) p).totalDegree *
114         MvPolynomial.eval (dehomogenizedPoint (laurentColumn q))
115           (MvPolynomial.map (algebraMap κ (LaurentSeries κ)) p) :=
116       eval_homogenizeAtZero_eq_pow_mul_dehomogenizedPoint
117         (MvPolynomial.map (algebraMap κ (LaurentSeries κ)) p)
118         (laurentColumn q)
119         (laurentColumn_ne_zero_of_ne_zero q hq₀)
120     _ = (laurentColumn q 0) ^ p.totalDegree *
121         MvPolynomial.eval (dehomogenizedPoint (laurentColumn q))
122           (MvPolynomial.map (algebraMap κ (LaurentSeries κ)) p) := by
123       rw [totalDegree_map_eq_of_injective
124         (algebraMap κ (LaurentSeries κ))
125         (algebraMap κ (LaurentSeries κ)).injective p]

127 /-- The Laurent-series evaluation of a homogenized numerator is nonzero
128 exactly when the original affine evaluation at the dehomogenized Laurent point
129 is nonzero. -/
130 theorem laurent_eval_map_homogenizeAtZero_ne_zero_iff
131     (p : MvPolynomial (Fin n) κ)
132     (q : Fin (n + 1) → PowerSeries κ) (hq₀ : q 0 ≠ 0) :
133     MvPolynomial.eval (laurentColumn q)
134         (MvPolynomial.map (algebraMap κ (LaurentSeries κ))
135           (homogenizeAtZero p)) ≠ 0 ↔
136       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q))
137           (MvPolynomial.map (algebraMap κ (LaurentSeries κ)) p) ≠ 0 := by
138   rw [laurent_eval_map_homogenizeAtZero_eq_pow_mul_dehomogenizedPoint
139     p q hq₀]
140   constructor
141   · intro h
142     exact (mul_ne_zero_iff.mp h).2
143   · intro h
144     exact mul_ne_zero
145       (pow_ne_zero _ (laurentColumn_ne_zero_of_ne_zero q hq₀)) h
146 
147 end
