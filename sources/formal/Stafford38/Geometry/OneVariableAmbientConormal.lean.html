11 /-!
12 # Ambient rank-one conormal production
13 
14 In one base variable, passing from a radical ambient ideal to one of its
15 prime components loses the direction needed for conormal containment: if
16 `I ≤ P`, then the equation-conormal space of `I` is contained in that of `P`,
17 not conversely.  This file avoids that invalid transfer.
18 
19 Instead, the radical ambient ideal is principal after identifying the
20 one-variable multivariate polynomial ring with `k[X]`.  Its nonzero generator
21 is squarefree.  At every zero, separability therefore makes the derivative
22 nonzero, so the ambient equation itself spans the unique cotangent direction.
23 This produces the Laurent conormal axis for the ambient ideal directly.
24 -/

45 private abbrev finOnePolynomialEquiv (k : Type u) [Field k] :
46     MvPolynomial (Fin 1) k ≃ₐ[k] Polynomial k :=
47   MvPolynomial.uniqueAlgEquiv k (Fin 1)
48 
49 private theorem finOnePolynomialEquiv_eval
50     {k : Type u} [Field k] (f : MvPolynomial (Fin 1) k) (y : Fin 1 → k) :
51     Polynomial.eval (y 0) (finOnePolynomialEquiv k f) =
52       MvPolynomial.eval y f := by
53   change (finOnePolynomialEquiv k f).eval₂ (RingHom.id k) (y 0) =
54     f.eval₂ (RingHom.id k) y
55   exact MvPolynomial.eval₂_uniqueAlgEquiv
56 
57 private theorem finOnePolynomialEquiv_pderiv
58     {k : Type u} [Field k] (f : MvPolynomial (Fin 1) k) :
59     finOnePolynomialEquiv k (MvPolynomial.pderiv (0 : Fin 1) f) =
60       (finOnePolynomialEquiv k f).derivative := by
61   induction' f using MvPolynomial.induction_on with a p q hp hq p i hp
62   · simp [finOnePolynomialEquiv]
63   · simp only [map_add]
64     rw [hp, hq]
65   · fin_cases i
66     simp only [MvPolynomial.pderiv_mul, MvPolynomial.pderiv_X,
67       Pi.single_apply, map_add, map_mul, Polynomial.derivative_mul]
68     rw [hp]
69     simp [finOnePolynomialEquiv]

71 /-- A radical ambient ideal in one variable which avoids the origin has, at
72 each supplied zero, an ambient equation with nonzero differential.  The
73 equation is the pullback of a principal generator after identifying the ring
74 with `k[X]`. -/
75 theorem exists_ambientEquation_differential_ne_zero
76     {k : Type u} [Field k] [IsAlgClosed k]
77     (I : Ideal (MvPolynomial (Fin 1) k))
78     (hIrad : I.IsRadical)
79     (y : Fin 1 → k)
80     (hy : ∀ f ∈ I, MvPolynomial.eval y f = 0)
81     (havoid : ∀ z : Fin 1 → k,
82       (∀ f ∈ I, MvPolynomial.eval z f = 0) → z 0 ≠ 0) :
83     ∃ f ∈ I, differentialAt y f 0 ≠ 0 := by
84   let e := finOnePolynomialEquiv k
85   let Q : Ideal (Polynomial k) := I.map e
86   have hIne : I ≠ ⊥ := by
87     intro hI
88     have hzero : ∀ f ∈ I,
89         MvPolynomial.eval (fun _ : Fin 1 ↦ (0 : k)) f = 0 := by
90       intro f hf
91       have hf0 : f = 0 := by
92         rw [hI] at hf
93         simpa using hf
94       simp [hf0]
95     exact (havoid (fun _ ↦ 0) hzero) rfl
96   have hQrad : Q.IsRadical := by
97     have hQeq : Q = I.comap e.symm := by
98       ext q
99       constructor
100       · intro hq
101         obtain ⟨f, hf, hef⟩ :=
102           (Ideal.mem_map_iff_of_surjective e e.surjective).mp hq
103         change e.symm q ∈ I
104         simpa [← hef] using hf
105       · intro hq
106         apply (Ideal.mem_map_iff_of_surjective e e.surjective).mpr
107         exact ⟨e.symm q, hq, by simp⟩
108     rw [hQeq]
109     exact hIrad.comap e.symm
110   have hQne : Q ≠ ⊥ := by
111     intro hQ
112     apply hIne
113     exact (Ideal.map_eq_bot_iff_of_injective e.injective).mp hQ
114   let g : Polynomial k := Submodule.IsPrincipal.generator Q
115   have hgQ : g ∈ Q := Submodule.IsPrincipal.generator_mem Q
116   have hg0 : g ≠ 0 := by
117     intro hg
118     apply hQne
119     rw [← Ideal.span_singleton_generator Q]
120     have hgenerator : Submodule.IsPrincipal.generator Q = 0 := by
121       simpa [g] using hg
122     rw [hgenerator, Ideal.span_singleton_eq_bot]
123   have hgrad : IsRadical g := by
124     rw [isRadical_iff_span_singleton]
125     have hspan : Ideal.span ({g} : Set (Polynomial k)) = Q := by
126       simp [g]
127     rw [hspan]
128     exact hQrad
129   have hgsq : Squarefree g :=
130     (isRadical_iff_squarefree_of_ne_zero hg0).mp hgrad
131   have hgsep : g.Separable :=
132     PerfectField.separable_iff_squarefree.mpr hgsq
133   let f : MvPolynomial (Fin 1) k := e.symm g
134   have hef : e f = g := e.apply_symm_apply g
135   have hfI : f ∈ I := by
136     have : e.symm g ∈ I := by
137       obtain ⟨f', hf', hef'⟩ :=
138         (Ideal.mem_map_iff_of_surjective e e.surjective).mp hgQ
139       simpa [← hef'] using hf'
140     exact this
141   have hgeval : Polynomial.eval (y 0) g = 0 := by
142     rw [← hef]
143     exact (finOnePolynomialEquiv_eval f y).trans (hy f hfI)
144   have hgderiv : Polynomial.eval (y 0) g.derivative ≠ 0 :=
145     hgsep.eval₂_derivative_ne_zero (RingHom.id k) (by simpa using hgeval)
146   refine ⟨f, hfI, ?_⟩
147   change MvPolynomial.eval y (MvPolynomial.pderiv 0 f) ≠ 0
148   rw [← finOnePolynomialEquiv_eval (MvPolynomial.pderiv 0 f) y,
149     finOnePolynomialEquiv_pderiv]
150   rw [hef]
151   exact hgderiv

153 /-- In rank one, one ambient equation with nonzero differential spans the
154 entire cotangent line. -/
155 theorem pureAxis_mem_affineConormalSpace_of_differential_ne_zero
156     {k : Type u} [Field k]
157     (I : Ideal (MvPolynomial (Fin 1) k))
158     (y : Fin 1 → k) (f : MvPolynomial (Fin 1) k)
159     (hf : f ∈ I) (hdf : differentialAt y f 0 ≠ 0) :
160     coordinateCovector (fun _ : Fin 1 ↦ 1) ∈ affineConormalSpace y I := by
161   rw [affineConormalSpace_eq_equationCovectorSpan]
162   have hgenerator : differentialCovector y f ∈ equationCovectorSpan y I := by
163     apply Submodule.subset_span
164     exact ⟨⟨f, hf⟩, rfl⟩
165   have hscaled := (equationCovectorSpan y I).smul_mem (differentialAt y f 0)⁻¹ hgenerator
166   convert hscaled using 1
167   apply LinearMap.ext
168   intro v
169   simp [differentialCovector, coordinateCovector, hdf]

171 /-- A radical ambient rank-one ideal with a zero and with origin avoidance
172 produces the exact Laurent conormal axis required by the terminal projected
173 consumer.  No prime-component conormal is transferred to the ambient ideal. -/
174 theorem exists_laurentConormalAxis_of_radical_ambient_axis_avoidance
175     {k : Type u} [Field k] [IsAlgClosed k]
176     (I : Ideal (MvPolynomial (Fin 1) k))
177     (hIrad : I.IsRadical)
178     (hexists : ∃ y : Fin 1 → k,
179       ∀ f ∈ I, MvPolynomial.eval y f = 0)
180     (havoid : ∀ y : Fin 1 → k,
181       (∀ f ∈ I, MvPolynomial.eval y f = 0) → y 0 ≠ 0) :
182     ∃ (yL : Fin 1 → LaurentSeries k)
183       (xi : Fin 1 → PowerSeries k),
184       Sum.elim yL
185           (fun j ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi j)) ∈
186         equationConormalLocus
187           (I.map (scalarPolynomialMap
188             (k := k) (K := LaurentSeries k) (Fin 1))) ∧
189       residueColumn xi = (fun _ : Fin 1 ↦ 1) := by
190   obtain ⟨y, hy⟩ := hexists
191   obtain ⟨f, hf, hdf⟩ :=
192     exists_ambientEquation_differential_ne_zero I hIrad y hy havoid
193   let yL : Fin 1 → LaurentSeries k :=
194     fun j ↦ algebraMap k (LaurentSeries k) (y j)
195   let xi : Fin 1 → PowerSeries k := fun _ ↦ PowerSeries.C 1
196   let IL := I.map
197     (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin 1))
198   have hyL : ∀ g ∈ IL, MvPolynomial.eval yL g = 0 := by
199     rw [show (∀ g ∈ IL, MvPolynomial.eval yL g = 0) ↔
200         yL ∈ MvPolynomial.zeroLocus (LaurentSeries k) IL by rfl]
201     rw [mem_zeroLocus_map_iff]
202     intro g hg
203     change MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
204       ((algebraMap k (LaurentSeries k)) ∘ y) g = 0
205     rw [← MvPolynomial.eval₂_comp, hy g hg, map_zero]
206   have hfIL : scalarPolynomialMap
207       (k := k) (K := LaurentSeries k) (Fin 1) f ∈ IL :=
208     Ideal.mem_map_of_mem _ hf
209   have hdfL : differentialAt yL
210       (scalarPolynomialMap
211         (k := k) (K := LaurentSeries k) (Fin 1) f) 0 ≠ 0 := by
212     rw [differentialAt_scalarPolynomialMap]
213     change MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
214       ((algebraMap k (LaurentSeries k)) ∘ y)
215         (MvPolynomial.pderiv 0 f) ≠ 0
216     rw [← MvPolynomial.eval₂_comp]
217     simpa [differentialAt] using
218       (algebraMap k (LaurentSeries k)).injective.ne hdf
219   have haxisL : coordinateCovector (fun _ : Fin 1 ↦ (1 : LaurentSeries k)) ∈
220       affineConormalSpace yL IL :=
221     pureAxis_mem_affineConormalSpace_of_differential_ne_zero IL yL _ hfIL hdfL
222   refine ⟨yL, xi, ?_, ?_⟩
223   · refine ⟨hyL, ?_⟩
224     simpa [xi] using haxisL
225   · funext j
226     simp [xi, residueColumn]

228 /-- Canonical rank-one integration.  Phase-space axis avoidance and nonempty
229 support provide exactly the ambient base hypotheses; radicality of the
230 canonical contracted base ideal then supplies the Laurent conormal axis. -/
231 theorem exists_canonical_rankOne_laurentConormalAxis
232     {k : Type u} [Field k] [IsAlgClosed k]
233     (I : RightIdeal (WeylAlg k 1))
234     (hdisjoint : Disjoint
235       (orderCharacteristicSupport k I)
236       (PrimeSpectrum.zeroLocus
237         ({MvPolynomial.X (.inl (0 : Fin 1))} :
238           Set (SymbolRing k 1))))
239     (hsupp : (orderCharacteristicSupport k I).Nonempty) :
240     ∃ (yL : Fin 1 → LaurentSeries k)
241       (xi : Fin 1 → PowerSeries k),
242       Sum.elim yL
243           (fun j ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi j)) ∈
244         equationConormalLocus
245           ((reducedOrderBaseIdeal k I).map
246             (scalarPolynomialMap
247               (k := k) (K := LaurentSeries k) (Fin 1))) ∧
248       residueColumn xi = (fun _ : Fin 1 ↦ 1) := by
249   obtain ⟨hexists, havoid⟩ :=
250     exists_baseZero_and_all_coordinate_ne_zero I (0 : Fin 1) hdisjoint hsupp
251   exact exists_laurentConormalAxis_of_radical_ambient_axis_avoidance
252     (reducedOrderBaseIdeal k I)
253     (reducedOrderBaseIdeal_isRadical k I) hexists havoid
