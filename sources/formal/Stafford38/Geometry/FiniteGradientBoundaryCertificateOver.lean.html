11 # Shared finite-gradient boundary certificate
12 
13 This is the single record owner for finite-gradient boundary data over a
14 ground field `k` and a boundary residue field `K`.  The base-field producer
15 uses its checked `K = k` specialization; residue-extension consumers retain
16 the full coefficient-extension scope.
17 -/

35 /-- A completed projective arc and one finite gradient identity over the
36 natural boundary residue field `K`, for an affine ideal defined over `k`. -/
37 structure FiniteGradientBoundaryCertificateOver
38     (m : ℕ) (hm : 0 < m)
39     (I : Ideal (MvPolynomial (Fin m) k)) where
40   equationCount : ℕ
41   q : Fin (m + 1) → PowerSeries K
42   ell : Fin (m + 1) → PowerSeries K
43   q_origin_ne : q 0 ≠ 0
44   projective_annihilation :
45     ∑ i, laurentColumn ell i * laurentColumn q i = 0
46   base_vanish :
47     ∀ f ∈ I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m)),
48       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0
49   equations : Fin equationCount →
50     I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m))
51   coefficients : Fin equationCount → LaurentSeries K
52   gradient_identity : ∀ i : Fin m,
53     laurentColumn ell i.succ =
54       ∑ j, coefficients j *
55         differentialAt (dehomogenizedPoint (laurentColumn q))
56           (equations j).1 i
57   residue_axis :
58     residueColumn (fun i : Fin m ↦ ell i.succ) =
59       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0)
60 
61 end

63 end Stafford38.Geometry.FiniteGradientResidueExtension
