17 /-- An isomorphism of rings carrying one adic ideal to another induces an
18 isomorphism of the corresponding adic completions. -/
19 def mapOfRingEquiv (e : A ≃+* B) (I : Ideal A) (J : Ideal B)
20     (hJ : J = I.map (e : A →+* B)) :
21     AdicCompletion I A →+* AdicCompletion J B :=
22   Stafford38.Geometry.AdicCompletionMap.mapOfRingHom (e : A →+* B) I J hJ
23 
24 /-- Evaluation at a finite quotient commutes with the canonical completion map. -/
25 theorem eval_mapOfRingEquiv (e : A ≃+* B) (I : Ideal A) (J : Ideal B)
26     (hJ : J = I.map (e : A →+* B)) (x : AdicCompletion I A) (n : ℕ) :
27     AdicCompletion.evalₐ J n (mapOfRingEquiv e I J hJ x) =
28       Ideal.quotientEquiv (I ^ n) (J ^ n) e (by rw [hJ, Ideal.map_pow])
29         (AdicCompletion.evalₐ I n x) := by
30   rw [mapOfRingEquiv, Stafford38.Geometry.AdicCompletionMap.eval_mapOfRingHom]
31   rfl
32 
33 /-- The canonical map sends the dense polynomial copy through the given ring
34 equivalence. -/
35 @[simp]
36 theorem mapOfRingEquiv_apply_of (e : A ≃+* B) (I : Ideal A) (J : Ideal B)
37     (hJ : J = I.map (e : A →+* B)) (a : A) :
38     mapOfRingEquiv e I J hJ (AdicCompletion.of I A a) =
39       AdicCompletion.of J B (e a) := by
40   exact Stafford38.Geometry.AdicCompletionMap.mapOfRingHom_apply_of
41     (e : A →+* B) I J hJ a

43 /-- A ring equivalence carrying the defining ideals induces an equivalence of
44 adic completions. -/
45 def ofRingEquiv (e : A ≃+* B) (I : Ideal A) (J : Ideal B)
46     (hJ : J = I.map (e : A →+* B)) :
47     AdicCompletion I A ≃+* AdicCompletion J B := by
48   classical
49   let q (n : ℕ) : A ⧸ I ^ n ≃+* B ⧸ J ^ n :=
50     Ideal.quotientEquiv (I ^ n) (J ^ n) e (by rw [hJ, Ideal.map_pow])
51   let forward : AdicCompletion I A →+* AdicCompletion J B :=
52     mapOfRingEquiv e I J hJ
53   have hforward (x : AdicCompletion I A) (n : ℕ) :
54       AdicCompletion.evalₐ J n (forward x) = q n (AdicCompletion.evalₐ I n x) := by
55     simpa [forward, q] using eval_mapOfRingEquiv e I J hJ x n
56   have hJinv : I = J.map (e.symm : B →+* A) := by
57     rw [hJ, Ideal.map_map]; simp
58   let backward : AdicCompletion J B →+* AdicCompletion I A :=
59     mapOfRingEquiv e.symm J I hJinv
60   have hbackward (x : AdicCompletion J B) (n : ℕ) :
61       AdicCompletion.evalₐ I n (backward x) = (q n).symm (AdicCompletion.evalₐ J n x) := by
62     simpa [backward, q] using eval_mapOfRingEquiv e.symm J I hJinv x n
63   refine
64     { toFun := forward
65       invFun := backward
66       left_inv := ?_
67       right_inv := ?_
68       map_mul' := forward.map_mul
69       map_add' := forward.map_add }
70   · intro x
71     apply AdicCompletion.ext_evalₐ
72     intro n
73     rw [hbackward, hforward]
74     exact (q n).symm_apply_apply _
75   · intro x
76     apply AdicCompletion.ext_evalₐ
77     intro n
78     rw [hforward, hbackward]
79     exact (q n).apply_symm_apply _
80 
81 /-- Evaluation of the completion equivalence is the finite-quotient map induced by
82 the original ring equivalence. -/
83 theorem eval_ofRingEquiv (e : A ≃+* B) (I : Ideal A) (J : Ideal B)
84     (hJ : J = I.map (e : A →+* B)) (x : AdicCompletion I A) (n : ℕ) :
85     AdicCompletion.evalₐ J n (ofRingEquiv e I J hJ x) =
86       Ideal.quotientEquiv (I ^ n) (J ^ n) e (by rw [hJ, Ideal.map_pow])
87         (AdicCompletion.evalₐ I n x) := by
88   have hfun : (fun y => ofRingEquiv e I J hJ y) = mapOfRingEquiv e I J hJ := by
89     funext y
90     rfl
91   rw [congrFun hfun x]
92   exact eval_mapOfRingEquiv e I J hJ x n
93 
94 /-- The completion equivalence sends the dense polynomial copy through the given
95 ring equivalence. -/
96 @[simp]
97 theorem ofRingEquiv_apply_of (e : A ≃+* B) (I : Ideal A) (J : Ideal B)
98     (hJ : J = I.map (e : A →+* B)) (a : A) :
99     ofRingEquiv e I J hJ (AdicCompletion.of I A a) =
100       AdicCompletion.of J B (e a) := by
101   apply AdicCompletion.ext_evalₐ
102   intro n
103   rw [eval_ofRingEquiv, AdicCompletion.evalₐ_of]
104   simp
105 
106 end
