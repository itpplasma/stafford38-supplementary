26 /-- If a height-one center on an integral closure has already been established,
27 the canonical localization at that center is the original valuation subring.
28 The final field identity records that this is the localization's canonical map,
29 not a field automorphism. -/
30 theorem exists_localization_equiv_of_height_one
31     (R : Subring K) (V : ValuationSubring K) (hRV : R ≤ V.toSubring)
32     [hNoeth : IsNoetherianRing (integralClosure R K).toSubring]
33     [hIntegrallyClosed : IsIntegrallyClosed (integralClosure R K).toSubring]
34     [hFractionRing : IsFractionRing (integralClosure R K).toSubring K]
35     (hp : (Stafford38.Geometry.ProjectiveChartNormalizationCenter.integralClosureCenter
36       R V hRV).IsPrime)
37     (hh : (Stafford38.Geometry.ProjectiveChartNormalizationCenter.integralClosureCenter
38       R V hRV).height = 1) :
39     let B : Subring K := (integralClosure R K).toSubring
40     let p : Ideal B := Stafford38.Geometry.ProjectiveChartNormalizationCenter.integralClosureCenter
41       R V hRV
42     ∃ e : Localization.AtPrime p ≃+* V.toSubring,
43       ∀ b : B,
44         ((e (algebraMap B (Localization.AtPrime p) b) : V.toSubring) : K) = (b : K) := by
45   dsimp only
46   let B : Subring K := (integralClosure R K).toSubring
47   let p : Ideal B := Stafford38.Geometry.ProjectiveChartNormalizationCenter.integralClosureCenter
48     R V hRV
49   letI : p.IsPrime := hp
50   have hlocal : (LocalSubring.ofPrime B p).toSubring = V.toSubring :=
51     @Stafford38.Geometry.IntegralClosureCenterDVR.localNormalizationCenter_eq_valuationSubring
52       K inferInstance R V hRV hNoeth hIntegrallyClosed hFractionRing hp hh
53   let e0 : Localization.AtPrime p ≃ₐ[B] (LocalSubring.ofPrime B p).toSubring :=
54     LocalSubring.ofPrimeEquiv B p
55   let e : Localization.AtPrime p ≃+* V.toSubring :=
56     e0.toRingEquiv.trans (subringEquivOfEq hlocal)
57   refine ⟨e, ?_⟩
58   intro b
59   change ((subringEquivOfEq hlocal
60       (e0 (algebraMap B (Localization.AtPrime p) b)) : V.toSubring) : K) = (b : K)
61   rw [subringEquivOfEq_coe]
62   change ((e0 (algebraMap B (Localization.AtPrime p) b) : (LocalSubring.ofPrime B p).toSubring) : K) =
63     (b : K)
64   rw [e0.commutes]
65   rfl
