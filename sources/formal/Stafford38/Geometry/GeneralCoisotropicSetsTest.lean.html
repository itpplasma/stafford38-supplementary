14 input.  Their interface is paper-facing: closedness and nonemptiness are
15 hypotheses on `W`, while fibre homogeneity is supplied by `IsFibreConical`

102 /-- The manuscript-facing statement over `ℂ`, with ordinary fibre scaling and
103 pointwise fibre-polynomial vanishing as its inputs.  The exported theorem is
104 therefore checked through the definitions used by the paper rather than by
105 repeating its interface propositions as hypotheses. -/
106 theorem exact_complex_manuscript_coisotropic_consumer
107     {n : ℕ} (hn : 0 < n)
108     (W : Set (PhaseVar n → ℂ))
109     (hnonempty : W.Nonempty)
110     (hclosed : ∃ L : Ideal (SymbolRing ℂ n),
111       W = MvPolynomial.zeroLocus ℂ L)
112     (hscale : ∀ q ∈ W, ∀ a : ℂ, a ≠ 0 →
113       Sum.elim (fun i => q (.inl i)) (fun i => a * q (.inr i)) ∈ W)
114     (hbracket : ∀ f ∈ MvPolynomial.vanishingIdeal ℂ W,
115       ∀ g ∈ MvPolynomial.vanishingIdeal ℂ W,
116       poissonBracket f g ∈ MvPolynomial.vanishingIdeal ℂ W)
117     (P : MvPolynomial (Fin n) ℂ)
118     (hPpoint : ∀ q ∈ W,
119       MvPolynomial.eval (fun i => q (.inr i)) P = 0)
120     (haxis : MvPolynomial.eval
121       (fun i : Fin n => if i = ⟨0, hn⟩ then (1 : ℂ) else 0) P ≠ 0)
122     (havoid : ∀ q ∈ W, q (.inl ⟨0, hn⟩) ≠ 0) :
123     False := by
124   have hW : IsFibreConical W := hscale
125   have hP : fibreLift P ∈ MvPolynomial.vanishingIdeal ℂ W := by
126     rw [MvPolynomial.mem_vanishingIdeal_iff]
127     intro q hq
128     have hqP := hPpoint q hq
129     simpa [fibreLift, MvPolynomial.eval_rename, Function.comp_def] using hqP
130   exact false_of_exported_general_coisotropic_set_avoiding_hyperplane
131     hn W hnonempty hclosed hW hbracket P hP haxis havoid

133 /-! ## A negative control for the coisotropic hypothesis
134 
135 The zero section in one-dimensional phase space has ideal `(ξ)`.  It is
136 self-involutive, but it is not stable under bracketing with arbitrary ambient
137 polynomials because `{ξ, x} = -1`. -/
138 
139 def zeroSectionIdealOne (k : Type*) [Field k] : Ideal (SymbolRing k 1) :=
140   Ideal.span {MvPolynomial.X (Sum.inr (0 : Fin 1) : PhaseVar 1)}
141 
142 theorem zeroSectionIdealOne_isInvolutive
143     {k : Type*} [Field k] :
144     ∀ f ∈ zeroSectionIdealOne k, ∀ g ∈ zeroSectionIdealOne k,
145       poissonBracket f g ∈ zeroSectionIdealOne k := by
146   intro f hf g hg
147   change f ∈ Ideal.span
148     {MvPolynomial.X (Sum.inr (0 : Fin 1) : PhaseVar 1)} at hf
149   change g ∈ Ideal.span
150     {MvPolynomial.X (Sum.inr (0 : Fin 1) : PhaseVar 1)} at hg
151   rw [Ideal.mem_span_singleton] at hf hg
152   obtain ⟨a, ha⟩ := hf
153   obtain ⟨b, hb⟩ := hg
154   rw [ha, hb]
155   let I := zeroSectionIdealOne k
156   have hxi : MvPolynomial.X (Sum.inr (0 : Fin 1) : PhaseVar 1) ∈ I :=
157     Ideal.mem_span_singleton_self _
158   rw [PostScalarExtensionPoisson.poissonBracket_mul_left]
159   apply I.add_mem
160   · exact I.mul_mem_right _ hxi
161   · apply I.mul_mem_left
162     rw [PostScalarExtensionPoisson.poissonBracket_mul_right,
163       poissonBracket_self, mul_zero, add_zero]
164     exact I.mul_mem_right _ hxi
165 
166 theorem zeroSectionIdealOne_not_isPoisson
167     {k : Type*} [Field k] : ¬ IsPoisson (zeroSectionIdealOne k) := by
168   intro hpoisson
169   have hxi : MvPolynomial.X (Sum.inr (0 : Fin 1) : PhaseVar 1) ∈
170       zeroSectionIdealOne k :=
171     Ideal.mem_span_singleton_self _
172   have hbad := hpoisson
173     (MvPolynomial.X (Sum.inr (0 : Fin 1) : PhaseVar 1)) hxi
174     (MvPolynomial.X (Sum.inl (0 : Fin 1) : PhaseVar 1))
175   have hnegone : (-1 : SymbolRing k 1) ∈ zeroSectionIdealOne k := by
176     simpa [poissonBracket, Pi.single_apply] using hbad
177   rw [zeroSectionIdealOne, Ideal.mem_span_singleton] at hnegone
178   rcases hnegone with ⟨a, ha⟩
179   have ha0 := congrArg
180     (MvPolynomial.eval (fun _ : PhaseVar 1 => (0 : k))) ha
181   simpa using ha0
