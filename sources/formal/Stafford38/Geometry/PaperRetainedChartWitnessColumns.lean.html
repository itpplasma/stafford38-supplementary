45 /-- The witness's own normalized projective column and residue algebraicity
46 yield the actual completed coefficientwise derivation columns. -/
47 theorem columns_of_visible_witness
48     (hm : 0 < m) (P : PrimeSpectrum (MvPolynomial (Fin m) k))
49     (w : GeneralDivisorialVisibleFrameWitness hm P) :
50     let C := w.column
51     let W := C.W
52     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
53       (ComponentFractionField P) := W.ambientAlgebra
54     let V := W.place.valuation.toSubring
55     letI : IsDiscreteValuationRing V := W.place.isDiscrete
56     letI : Algebra W.coefficientField V :=
57       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
58     let κ := ResidueField V
59     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
60     ∃ (rows : Fin (Module.finrank κ (Ω[κ⁄k])) ↪ Fin (m + 1))
61       (D : Fin (Module.finrank κ (Ω[κ⁄k])) → Derivation k κ κ),
62       (∀ i j, D j (PowerSeries.constantCoeff
63         (retainedToCompletedPowerSeries W (C.q (rows i)))) =
64           if i = j then 1 else 0) ∧
65       PowerSeries.constantCoeff
66         (selectedMinor
67           (coefficientwiseTangentMatrix
68             (fun a ↦ retainedToCompletedPowerSeries W (C.q a)) D) rows).det = 1 := by
69   let C := w.column
70   let W := C.W
71   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
72       (ComponentFractionField P) := W.ambientAlgebra
73   let V := W.place.valuation.toSubring
74   letI : IsDiscreteValuationRing V := W.place.isDiscrete
75   letI : Algebra W.coefficientField V :=
76     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
77   let κ := ResidueField V
78   letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
79   dsimp only
80   exact exists_retained_projective_derivation_columns P ⟨0, hm⟩
81     W C.q w.halg
