14 /-- A uniformizer in a DVR localization of a domain has a numerator in the
15 original prime ideal which remains a generator of the localized maximal ideal.
16 The denominator is retained to identify the numerator with a unit multiple of
17 the chosen uniformizer. -/
18 theorem exists_numerator_of_dvr_localization_of_irreducible
19     {B : Type u} [CommRing B] [IsDomain B]
20     (P : Ideal B) [P.IsPrime]
21     [IsDiscreteValuationRing (Localization.AtPrime P)]
22     (π : Localization.AtPrime P) (hπ : Irreducible π) :
23     ∃ a : B, ∃ b : P.primeCompl,
24       a ≠ 0 ∧ a ∈ P ∧
25       IsLocalization.mk' (Localization.AtPrime P) a b = π ∧
26       Ideal.span {algebraMap B (Localization.AtPrime P) a} =
27         maximalIdeal (Localization.AtPrime P) := by
28   let R := Localization.AtPrime P
29   obtain ⟨a, b, hrep⟩ := IsLocalization.exists_mk'_eq P.primeCompl π
30   have hmax : maximalIdeal R = Ideal.span {π} := hπ.maximalIdeal_eq
31   have hπmem : π ∈ maximalIdeal R := by
32     rw [hmax]
33     exact Submodule.mem_span_singleton_self π
34   have hbunit : IsUnit (algebraMap B R (b : B)) :=
35     IsLocalization.map_units R b
36   have hmapa : algebraMap B R a = π * algebraMap B R (b : B) := by
37     calc
38       algebraMap B R a = IsLocalization.mk' R a b * algebraMap B R (b : B) :=
39         (IsLocalization.mk'_spec R a b).symm
40       _ = π * algebraMap B R (b : B) := by rw [hrep]
41   have haLoc : algebraMap B R a ∈ maximalIdeal R := by
42     rw [hmapa]
43     exact (maximalIdeal R).mul_mem_right _ hπmem
44   have haP : a ∈ P :=
45     (IsLocalization.AtPrime.to_map_mem_maximal_iff R P a).mp haLoc
46   have ha0 : a ≠ 0 := by
47     intro ha
48     have hzero : π = 0 := by
49       rw [hrep.symm, ha]
50       simp
51     exact hπ.ne_zero hzero
52   have hspan : Ideal.span {algebraMap B R a} = Ideal.span {π} := by
53     rw [hmapa]
54     exact Ideal.span_singleton_mul_right_unit hbunit π
55   exact ⟨a, b, ha0, haP, hrep, by rw [hspan, hmax.symm]⟩

57 /-- A uniformizer in a DVR localization of a domain has a numerator in the
58  original prime ideal which remains a generator of the localized maximal ideal.
59  The denominator is retained to identify the numerator with a unit multiple of
60  the chosen uniformizer. -/
61 theorem exists_numerator_of_dvr_localization
62     {B : Type u} [CommRing B] [IsDomain B]
63     (P : Ideal B) [P.IsPrime]
64     [IsDiscreteValuationRing (Localization.AtPrime P)] :
65     ∃ π : Localization.AtPrime P, Irreducible π ∧
66       ∃ a : B, ∃ b : P.primeCompl,
67         a ≠ 0 ∧ a ∈ P ∧
68         IsLocalization.mk' (Localization.AtPrime P) a b = π ∧
69         Ideal.span {algebraMap B (Localization.AtPrime P) a} =
70           maximalIdeal (Localization.AtPrime P) := by
71   let R := Localization.AtPrime P
72   obtain ⟨π, hπ⟩ := IsDiscreteValuationRing.exists_irreducible R
73   obtain ⟨a, b, ha0, haP, hrep, hspan⟩ :=
74     exists_numerator_of_dvr_localization_of_irreducible P π hπ
75   exact ⟨π, hπ, a, b, ha0, haP, hrep, hspan⟩
