20 /-- The actual affine-chart completion map sends the centered polynomial coordinate to the
21 corresponding formal variable. The public `powerSeriesCompletionAtPoint` is defined by composing
22 this translation-induced completion equivalence with the canonical power-series completion map. -/
23 theorem affinePointCompletion_variable_explicit (p : σ → k) (i : σ) :
24     let I : Ideal (MvPolynomial σ k) := MvPolynomial.idealOfVars σ k
25     let J : Ideal (MvPolynomial σ k) :=
26       Stafford38.Geometry.AffinePointCompletion.pointIdeal p
27     let e := (Stafford38.Geometry.AffinePointCompletion.translate p).toRingEquiv
28     let hJ : J = I.map (e : MvPolynomial σ k →+* MvPolynomial σ k) := by
29       change Stafford38.Geometry.AffinePointCompletion.pointIdeal p =
30         (MvPolynomial.idealOfVars σ k).map
31           ((Stafford38.Geometry.AffinePointCompletion.translate p).toRingEquiv :
32             MvPolynomial σ k →+* MvPolynomial σ k)
33       simpa only [RingEquiv.toRingHom_eq_coe] using
34         Stafford38.Geometry.AffinePointCompletion.pointIdeal_eq_map_translate p
35     Stafford38.Geometry.AdicCompletionRingEquiv.ofRingEquiv e I J hJ
36       (MvPowerSeries.toAdicCompletion σ k
37         (MvPolynomial.toMvPowerSeries (MvPolynomial.X i))) =
38       AdicCompletion.of J (MvPolynomial σ k)
39         (MvPolynomial.X i - MvPolynomial.C (p i)) := by
40   classical
41   dsimp
42   rw [MvPowerSeries.toAdicCompletion_coe]
43   rw [Stafford38.Geometry.AdicCompletionRingEquiv.ofRingEquiv_apply_of]
44   simp [Stafford38.Geometry.AffinePointCompletion.translate]
