12 /-!
13 # Canonical constant-coordinate branch
14 
15 If the distinguished coordinate is constant on a minimal component of a
16 radical affine variety, a minimal-prime multiplier and the Nullstellensatz
17 produce an ambient equation whose differential is the pure coordinate axis.
18 This gives the exact residue-extension endpoint used by the canonical
19 assembly, with residue extension equal to the ground field.
20 -/

46 private theorem exists_zero_eval_ne_of_isRadical_not_mem
47     {k : Type u} [Field k] [IsAlgClosed k] {m : ℕ}
48     (I : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
49     {g : MvPolynomial (Fin m) k} (hg : g ∉ I) :
50     ∃ y : Fin m → k,
51       (∀ f ∈ I, MvPolynomial.eval y f = 0) ∧
52       MvPolynomial.eval y g ≠ 0 := by
53   classical
54   by_contra h
55   have hall : ∀ y : Fin m → k,
56       y ∈ MvPolynomial.zeroLocus k I → MvPolynomial.eval y g = 0 := by
57     intro y hy
58     by_contra hgy
59     exact h ⟨y, hy, hgy⟩
60   have hgvanish : g ∈ MvPolynomial.vanishingIdeal k
61       (MvPolynomial.zeroLocus k I) := hall
62   rw [MvPolynomial.vanishingIdeal_zeroLocus_eq_radical, hI.radical] at hgvanish
63   exact hg hgvanish

65 /-- A coordinate which is constant on one minimal component of a radical
66 affine variety already contributes the pure coordinate covector to the
67 ambient equation-defined conormal at a suitable point.  The multiplier is
68 the standard zero-divisor witness for a minimal prime; radicality and the
69 Nullstellensatz choose a point where that multiplier does not vanish. -/
70 theorem exists_ambientConormalAxis_of_minimalPrime_coordinate_constant
71     {k : Type u} [Field k] [IsAlgClosed k] {m : ℕ}
72     (I P : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
73     (hP : P ∈ I.minimalPrimes) (i : Fin m) (c : k)
74     (hconstant : MvPolynomial.X i - MvPolynomial.C c ∈ P) :
75     ∃ y : Fin m → k,
76       (∀ f ∈ I, MvPolynomial.eval y f = 0) ∧
77       coordinateCovector (fun j ↦ if j = i then 1 else 0) ∈
78         affineConormalSpace y I := by
79   classical
80   let h : MvPolynomial (Fin m) k :=
81     MvPolynomial.X i - MvPolynomial.C c
82   obtain ⟨g, hgI, hhg⟩ :=
83     Ideal.exists_mul_mem_of_mem_minimalPrimes hP hconstant
84   obtain ⟨y, hy, hgy⟩ :=
85     exists_zero_eval_ne_of_isRadical_not_mem I hI hgI
86   have hhzero : MvPolynomial.eval y h = 0 := by
87     have hproduct : MvPolynomial.eval y (h * g) = 0 := hy _ hhg
88     rw [map_mul] at hproduct
89     exact (mul_eq_zero.mp hproduct).resolve_right hgy
90   have hdiff : differentialCovector y (h * g) =
91       MvPolynomial.eval y g •
92         coordinateCovector (fun j ↦ if j = i then 1 else 0) := by
93     apply LinearMap.ext
94     intro v
95     change (∑ j, differentialAt y (h * g) j * v j) =
96       MvPolynomial.eval y g *
97         ∑ j, (if j = i then 1 else 0) * v j
98     rw [Finset.mul_sum]
99     apply Finset.sum_congr rfl
100     intro j hj
101     by_cases hji : j = i
102     · subst j
103       simp [differentialAt, MvPolynomial.pderiv_mul, h, hhzero,
104         mul_assoc, mul_left_comm, mul_comm]
105     · simp [differentialAt, MvPolynomial.pderiv_mul, h, hhzero,
106         hji, Ne.symm hji, mul_assoc, mul_left_comm, mul_comm]
107   have hproductConormal : differentialCovector y (h * g) ∈
108       affineConormalSpace y I := by
109     rw [affineConormalSpace_eq_equationCovectorSpan]
110     apply Submodule.subset_span
111     exact ⟨⟨h * g, hhg⟩, rfl⟩
112   refine ⟨y, hy, ?_⟩
113   have hscaled := (affineConormalSpace y I).smul_mem
114     (MvPolynomial.eval y g)⁻¹ hproductConormal
115   rw [hdiff] at hscaled
116   simpa [hgy] using hscaled

118 /-- The minimal-component constant-coordinate branch reaches the exact
119 residue-extension endpoint used by the canonical asymptotic assembly.  The
120 extension is again `K = k`; the only extra ingredient beyond the ambient
121 branch is the minimal-prime multiplier argument above. -/
122 theorem exists_residueExtensionConormalAxis_of_minimalPrime_coordinate_constant
123     {k : Type u} [Field k] [IsAlgClosed k] {m : ℕ} (hm : 0 < m)
124     (I P : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
125     (hP : P ∈ I.minimalPrimes) (c : k)
126     (hconstant : MvPolynomial.X ⟨0, hm⟩ - MvPolynomial.C c ∈ P) :
127     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
128       (y : Fin m → LaurentSeries K)
129       (xi : Fin m → PowerSeries K),
130       Sum.elim y
131           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
132         groundEquationConormalLocus (k := k) (K := K) I ∧
133       residueColumn xi =
134         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
135   classical
136   let i : Fin m := ⟨0, hm⟩
137   let h : MvPolynomial (Fin m) k :=
138     MvPolynomial.X i - MvPolynomial.C c
139   obtain ⟨g, hgI, hhg⟩ :=
140     Ideal.exists_mul_mem_of_mem_minimalPrimes hP hconstant
141   obtain ⟨y, hy, hgy⟩ :=
142     exists_zero_eval_ne_of_isRadical_not_mem I hI hgI
143   have hhzero : MvPolynomial.eval y h = 0 := by
144     have hproduct : MvPolynomial.eval y (h * g) = 0 := hy _ hhg
145     rw [map_mul] at hproduct
146     exact (mul_eq_zero.mp hproduct).resolve_right hgy
147   let yL : Fin m → LaurentSeries k := fun j ↦
148     algebraMap k (LaurentSeries k) (y j)
149   let xi : Fin m → PowerSeries k := fun j ↦
150     PowerSeries.C (if j = i then 1 else 0)
151   let IL := I.map
152     (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m))
153   let fL := scalarPolynomialMap
154     (k := k) (K := LaurentSeries k) (Fin m) (h * g)
155   have hyL : ∀ f ∈ IL, MvPolynomial.eval yL f = 0 := by
156     rw [show (∀ f ∈ IL, MvPolynomial.eval yL f = 0) ↔
157         yL ∈ MvPolynomial.zeroLocus (LaurentSeries k) IL by rfl]
158     rw [mem_zeroLocus_map_iff]
159     intro f hf
160     change MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
161       ((algebraMap k (LaurentSeries k)) ∘ y) f = 0
162     rw [← MvPolynomial.eval₂_comp, hy f hf, map_zero]
163   have hfL : fL ∈ IL := by
164     exact Ideal.mem_map_of_mem
165       (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m)) hhg
166   have hgyL : algebraMap k (LaurentSeries k) (MvPolynomial.eval y g) ≠ 0 := by
167     simpa only [map_zero] using
168       (algebraMap k (LaurentSeries k)).injective.ne hgy
169   have hdiffAtL (j : Fin m) :
170       differentialAt yL fL j =
171         algebraMap k (LaurentSeries k) (differentialAt y (h * g) j) := by
172     rw [differentialAt_scalarPolynomialMap]
173     change MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
174       ((algebraMap k (LaurentSeries k)) ∘ y)
175         (MvPolynomial.pderiv j (h * g)) = _
176     rw [← MvPolynomial.eval₂_comp]
177     rfl
178   have hdiffL : differentialCovector yL fL =
179       algebraMap k (LaurentSeries k) (MvPolynomial.eval y g) •
180         coordinateCovector
181           (fun j : Fin m ↦ if j = i then 1 else 0) := by
182     apply LinearMap.ext
183     intro v
184     change (∑ j, differentialAt yL fL j * v j) =
185       algebraMap k (LaurentSeries k) (MvPolynomial.eval y g) *
186         ∑ j, (if j = i then 1 else 0) * v j
187     rw [Finset.mul_sum]
188     apply Finset.sum_congr rfl
189     intro j hj
190     rw [hdiffAtL]
191     by_cases hji : j = i
192     · subst j
193       simp [h, differentialAt, MvPolynomial.pderiv_mul, hhzero,
194         mul_assoc, mul_left_comm, mul_comm]
195     · simp [h, differentialAt, MvPolynomial.pderiv_mul, hhzero,
196         hji, Ne.symm hji, mul_assoc, mul_left_comm, mul_comm]
197   have haxisConormal : coordinateCovector
198       (fun j : Fin m ↦ if j = i then 1 else 0) ∈
199         affineConormalSpace yL IL := by
200     have hproductConormal : differentialCovector yL fL ∈
201         affineConormalSpace yL IL := by
202       rw [affineConormalSpace_eq_equationCovectorSpan]
203       apply Submodule.subset_span
204       exact ⟨⟨fL, hfL⟩, rfl⟩
205     have hscaled := (affineConormalSpace yL IL).smul_mem
206       ((algebraMap k (LaurentSeries k) (MvPolynomial.eval y g))⁻¹ :
207         LaurentSeries k) hproductConormal
208     rw [hdiffL] at hscaled
209     simpa only [inv_smul_smul₀ hgyL] using hscaled
210   have hphase : Sum.elim yL
211       (fun j : Fin m ↦ if j = i then 1 else 0) ∈
212         equationConormalLocus IL := ⟨hyL, haxisConormal⟩
213   refine ⟨k, inferInstance, inferInstance, yL, xi, ?_, ?_⟩
214   · change Sum.elim yL
215       (fun j ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi j)) ∈
216         groundEquationConormalLocus (k := k) (K := k) I
217     simpa [xi, i, IL, groundEquationConormalLocus, scalarPolynomialMap,
218       groundLaurentMap, scalarPolynomialMap] using hphase
219   · funext j
220     simp [xi, i, residueColumn]

222 /-- An ambient constant-coordinate equation and one ambient zero produce the
223 exact existential residue-extension conormal axis.  No algebraic closure,
224 boundary valuation, or nonconstant-coordinate argument is needed. -/
225 theorem exists_residueExtensionConormalAxis_of_ambientCoordinate_constant
226     {k : Type u} [Field k] {m : ℕ} (hm : 0 < m)
227     (I : Ideal (MvPolynomial (Fin m) k))
228     (hexists : ∃ y : Fin m → k,
229       ∀ f ∈ I, MvPolynomial.eval y f = 0)
230     (hconstant : ∃ c : k,
231       MvPolynomial.X ⟨0, hm⟩ - MvPolynomial.C c ∈ I) :
232     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
233       (y : Fin m → LaurentSeries K)
234       (xi : Fin m → PowerSeries K),
235       Sum.elim y
236           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
237         groundEquationConormalLocus (k := k) (K := K) I ∧
238       residueColumn xi =
239         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
240   obtain ⟨y, hy⟩ := hexists
241   obtain ⟨c, hc⟩ := hconstant
242   obtain ⟨yL, xi, hphase, hresidue⟩ :=
243     exists_laurentConormalAxis_of_coordinate_sub_constant_mem
244       I y hy ⟨0, hm⟩ c hc
245   refine ⟨k, inferInstance, inferInstance, yL, xi, ?_, hresidue⟩
246   simpa [groundEquationConormalLocus, scalarPolynomialMap, groundLaurentMap,
247     scalarPolynomialMap] using hphase

249 /-- Canonical specialization of the ambient constant-coordinate branch.
250 Nonempty order-characteristic support supplies the required ambient base
251 zero.  The conclusion is exactly the point type consumed by the
252 higher-dimensional residue-extension assembly. -/
253 theorem exists_canonical_residueExtensionConormalAxis_of_baseCoordinate_constant
254     {k : Type u} [Field k] [IsAlgClosed k]
255     (n N : ℕ) (d : WeylAlg k (n + 1))
256     (hsupp :
257       (orderCharacteristicSupport k
258         (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty)
259     (hconstant : ∃ c : k,
260       MvPolynomial.X ⟨0, Nat.zero_lt_succ n⟩ - MvPolynomial.C c ∈
261         reducedOrderBaseIdeal k
262           (canonicalRightIdeal (presentedCoordinate k n) d N)) :
263     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
264       (y : Fin (n + 1) → LaurentSeries K)
265       (xi : Fin (n + 1) → PowerSeries K),
266       Sum.elim y
267           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
268         groundEquationConormalLocus (k := k) (K := K)
269           (reducedOrderBaseIdeal k
270             (canonicalRightIdeal (presentedCoordinate k n) d N)) ∧
271       residueColumn xi =
272         (fun i : Fin (n + 1) ↦
273           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) := by
274   apply exists_residueExtensionConormalAxis_of_ambientCoordinate_constant
275     (Nat.zero_lt_succ n)
276   · exact exists_reducedOrderBaseZero_of_support_nonempty
277       (canonicalRightIdeal (presentedCoordinate k n) d N) hsupp
278   · exact hconstant
