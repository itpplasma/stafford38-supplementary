10 /-!
11 # Canonical constant-coordinate branch
12 
13 If the distinguished coordinate is constant on a minimal component of a
14 radical affine variety, a minimal-prime multiplier and the Nullstellensatz
15 produce an ambient equation whose differential is the pure coordinate axis.
16 This gives the exact residue-extension endpoint used by the canonical
17 assembly, with residue extension equal to the ground field.
18 -/

44 private theorem exists_zero_eval_ne_of_isRadical_not_mem
45     {k : Type u} [Field k] [IsAlgClosed k] {m : ℕ}
46     (I : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
47     {g : MvPolynomial (Fin m) k} (hg : g ∉ I) :
48     ∃ y : Fin m → k,
49       (∀ f ∈ I, MvPolynomial.eval y f = 0) ∧
50       MvPolynomial.eval y g ≠ 0 := by
51   classical
52   by_contra h
53   have hall : ∀ y : Fin m → k,
54       y ∈ MvPolynomial.zeroLocus k I → MvPolynomial.eval y g = 0 := by
55     intro y hy
56     by_contra hgy
57     exact h ⟨y, hy, hgy⟩
58   have hgvanish : g ∈ MvPolynomial.vanishingIdeal k
59       (MvPolynomial.zeroLocus k I) := hall
60   rw [MvPolynomial.vanishingIdeal_zeroLocus_eq_radical, hI.radical] at hgvanish
61   exact hg hgvanish

63 /-- A coordinate which is constant on one minimal component of a radical
64 affine variety already contributes the pure coordinate covector to the
65 ambient equation-defined conormal at a suitable point.  The multiplier is
66 the standard zero-divisor witness for a minimal prime; radicality and the
67 Nullstellensatz choose a point where that multiplier does not vanish. -/
68 theorem exists_ambientConormalAxis_of_minimalPrime_coordinate_constant
69     {k : Type u} [Field k] [IsAlgClosed k] {m : ℕ}
70     (I P : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
71     (hP : P ∈ I.minimalPrimes) (i : Fin m) (c : k)
72     (hconstant : MvPolynomial.X i - MvPolynomial.C c ∈ P) :
73     ∃ y : Fin m → k,
74       (∀ f ∈ I, MvPolynomial.eval y f = 0) ∧
75       coordinateCovector (fun j ↦ if j = i then 1 else 0) ∈
76         affineConormalSpace y I := by
77   classical
78   let h : MvPolynomial (Fin m) k :=
79     MvPolynomial.X i - MvPolynomial.C c
80   obtain ⟨g, hgI, hhg⟩ :=
81     Ideal.exists_mul_mem_of_mem_minimalPrimes hP hconstant
82   obtain ⟨y, hy, hgy⟩ :=
83     exists_zero_eval_ne_of_isRadical_not_mem I hI hgI
84   have hhzero : MvPolynomial.eval y h = 0 := by
85     have hproduct : MvPolynomial.eval y (h * g) = 0 := hy _ hhg
86     rw [map_mul] at hproduct
87     exact (mul_eq_zero.mp hproduct).resolve_right hgy
88   have hdiff : differentialCovector y (h * g) =
89       MvPolynomial.eval y g •
90         coordinateCovector (fun j ↦ if j = i then 1 else 0) := by
91     apply LinearMap.ext
92     intro v
93     change (∑ j, differentialAt y (h * g) j * v j) =
94       MvPolynomial.eval y g *
95         ∑ j, (if j = i then 1 else 0) * v j
96     rw [Finset.mul_sum]
97     apply Finset.sum_congr rfl
98     intro j hj
99     by_cases hji : j = i
100     · subst j
101       simp [differentialAt, MvPolynomial.pderiv_mul, h, hhzero,
102         mul_assoc, mul_left_comm, mul_comm]
103     · simp [differentialAt, MvPolynomial.pderiv_mul, h, hhzero,
104         hji, Ne.symm hji, mul_assoc, mul_left_comm, mul_comm]
105   have hproductConormal : differentialCovector y (h * g) ∈
106       affineConormalSpace y I := by
107     rw [affineConormalSpace_eq_equationCovectorSpan]
108     apply Submodule.subset_span
109     exact ⟨⟨h * g, hhg⟩, rfl⟩
110   refine ⟨y, hy, ?_⟩
111   have hscaled := (affineConormalSpace y I).smul_mem
112     (MvPolynomial.eval y g)⁻¹ hproductConormal
113   rw [hdiff] at hscaled
114   simpa [hgy] using hscaled

116 /-- The minimal-component constant-coordinate branch reaches the exact
117 residue-extension endpoint used by the canonical asymptotic assembly.  The
118 extension is again `K = k`; the only extra ingredient beyond the ambient
119 branch is the minimal-prime multiplier argument above. -/
120 theorem exists_residueExtensionConormalAxis_of_minimalPrime_coordinate_constant
121     {k : Type u} [Field k] [IsAlgClosed k] {m : ℕ} (hm : 0 < m)
122     (I P : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
123     (hP : P ∈ I.minimalPrimes) (c : k)
124     (hconstant : MvPolynomial.X ⟨0, hm⟩ - MvPolynomial.C c ∈ P) :
125     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
126       (y : Fin m → LaurentSeries K)
127       (xi : Fin m → PowerSeries K),
128       Sum.elim y
129           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
130         groundEquationConormalLocus (k := k) (K := K) I ∧
131       residueColumn xi =
132         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
133   classical
134   let i : Fin m := ⟨0, hm⟩
135   let h : MvPolynomial (Fin m) k :=
136     MvPolynomial.X i - MvPolynomial.C c
137   obtain ⟨g, hgI, hhg⟩ :=
138     Ideal.exists_mul_mem_of_mem_minimalPrimes hP hconstant
139   obtain ⟨y, hy, hgy⟩ :=
140     exists_zero_eval_ne_of_isRadical_not_mem I hI hgI
141   have hhzero : MvPolynomial.eval y h = 0 := by
142     have hproduct : MvPolynomial.eval y (h * g) = 0 := hy _ hhg
143     rw [map_mul] at hproduct
144     exact (mul_eq_zero.mp hproduct).resolve_right hgy
145   let yL : Fin m → LaurentSeries k := fun j ↦
146     algebraMap k (LaurentSeries k) (y j)
147   let xi : Fin m → PowerSeries k := fun j ↦
148     PowerSeries.C (if j = i then 1 else 0)
149   let IL := I.map
150     (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m))
151   let fL := scalarPolynomialMap
152     (k := k) (K := LaurentSeries k) (Fin m) (h * g)
153   have hyL : ∀ f ∈ IL, MvPolynomial.eval yL f = 0 := by
154     rw [show (∀ f ∈ IL, MvPolynomial.eval yL f = 0) ↔
155         yL ∈ MvPolynomial.zeroLocus (LaurentSeries k) IL by rfl]
156     rw [mem_zeroLocus_map_iff]
157     intro f hf
158     change MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
159       ((algebraMap k (LaurentSeries k)) ∘ y) f = 0
160     rw [← MvPolynomial.eval₂_comp, hy f hf, map_zero]
161   have hfL : fL ∈ IL := by
162     exact Ideal.mem_map_of_mem
163       (scalarPolynomialMap (k := k) (K := LaurentSeries k) (Fin m)) hhg
164   have hgyL : algebraMap k (LaurentSeries k) (MvPolynomial.eval y g) ≠ 0 := by
165     simpa only [map_zero] using
166       (algebraMap k (LaurentSeries k)).injective.ne hgy
167   have hdiffAtL (j : Fin m) :
168       differentialAt yL fL j =
169         algebraMap k (LaurentSeries k) (differentialAt y (h * g) j) := by
170     rw [differentialAt_scalarPolynomialMap]
171     change MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
172       ((algebraMap k (LaurentSeries k)) ∘ y)
173         (MvPolynomial.pderiv j (h * g)) = _
174     rw [← MvPolynomial.eval₂_comp]
175     rfl
176   have hdiffL : differentialCovector yL fL =
177       algebraMap k (LaurentSeries k) (MvPolynomial.eval y g) •
178         coordinateCovector
179           (fun j : Fin m ↦ if j = i then 1 else 0) := by
180     apply LinearMap.ext
181     intro v
182     change (∑ j, differentialAt yL fL j * v j) =
183       algebraMap k (LaurentSeries k) (MvPolynomial.eval y g) *
184         ∑ j, (if j = i then 1 else 0) * v j
185     rw [Finset.mul_sum]
186     apply Finset.sum_congr rfl
187     intro j hj
188     rw [hdiffAtL]
189     by_cases hji : j = i
190     · subst j
191       simp [h, differentialAt, MvPolynomial.pderiv_mul, hhzero,
192         mul_assoc, mul_left_comm, mul_comm]
193     · simp [h, differentialAt, MvPolynomial.pderiv_mul, hhzero,
194         hji, Ne.symm hji, mul_assoc, mul_left_comm, mul_comm]
195   have haxisConormal : coordinateCovector
196       (fun j : Fin m ↦ if j = i then 1 else 0) ∈
197         affineConormalSpace yL IL := by
198     have hproductConormal : differentialCovector yL fL ∈
199         affineConormalSpace yL IL := by
200       rw [affineConormalSpace_eq_equationCovectorSpan]
201       apply Submodule.subset_span
202       exact ⟨⟨fL, hfL⟩, rfl⟩
203     have hscaled := (affineConormalSpace yL IL).smul_mem
204       ((algebraMap k (LaurentSeries k) (MvPolynomial.eval y g))⁻¹ :
205         LaurentSeries k) hproductConormal
206     rw [hdiffL] at hscaled
207     simpa only [inv_smul_smul₀ hgyL] using hscaled
208   have hphase : Sum.elim yL
209       (fun j : Fin m ↦ if j = i then 1 else 0) ∈
210         equationConormalLocus IL := ⟨hyL, haxisConormal⟩
211   refine ⟨k, inferInstance, inferInstance, yL, xi, ?_, ?_⟩
212   · change Sum.elim yL
213       (fun j ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi j)) ∈
214         groundEquationConormalLocus (k := k) (K := k) I
215     simpa [xi, i, IL, groundEquationConormalLocus, groundPolynomialMap,
216       groundLaurentMap, scalarPolynomialMap] using hphase
217   · funext j
218     simp [xi, i, residueColumn]

220 /-- An ambient constant-coordinate equation and one ambient zero produce the
221 exact existential residue-extension conormal axis.  No algebraic closure,
222 boundary valuation, or nonconstant-coordinate argument is needed. -/
223 theorem exists_residueExtensionConormalAxis_of_ambientCoordinate_constant
224     {k : Type u} [Field k] {m : ℕ} (hm : 0 < m)
225     (I : Ideal (MvPolynomial (Fin m) k))
226     (hexists : ∃ y : Fin m → k,
227       ∀ f ∈ I, MvPolynomial.eval y f = 0)
228     (hconstant : ∃ c : k,
229       MvPolynomial.X ⟨0, hm⟩ - MvPolynomial.C c ∈ I) :
230     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
231       (y : Fin m → LaurentSeries K)
232       (xi : Fin m → PowerSeries K),
233       Sum.elim y
234           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
235         groundEquationConormalLocus (k := k) (K := K) I ∧
236       residueColumn xi =
237         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
238   obtain ⟨y, hy⟩ := hexists
239   obtain ⟨c, hc⟩ := hconstant
240   obtain ⟨yL, xi, hphase, hresidue⟩ :=
241     exists_laurentConormalAxis_of_coordinate_sub_constant_mem
242       I y hy ⟨0, hm⟩ c hc
243   refine ⟨k, inferInstance, inferInstance, yL, xi, ?_, hresidue⟩
244   simpa [groundEquationConormalLocus, groundPolynomialMap, groundLaurentMap,
245     scalarPolynomialMap] using hphase

247 /-- Canonical specialization of the ambient constant-coordinate branch.
248 Nonempty order-characteristic support supplies the required ambient base
249 zero.  The conclusion is exactly the point type consumed by the
250 higher-dimensional residue-extension assembly. -/
251 theorem exists_canonical_residueExtensionConormalAxis_of_baseCoordinate_constant
252     {k : Type u} [Field k] [IsAlgClosed k]
253     (n N : ℕ) (d : PresentedWeyl k (n + 1))
254     (hsupp :
255       (orderCharacteristicSupport k
256         (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty)
257     (hconstant : ∃ c : k,
258       MvPolynomial.X ⟨0, Nat.zero_lt_succ n⟩ - MvPolynomial.C c ∈
259         reducedOrderBaseIdeal k
260           (canonicalRightIdeal (presentedCoordinate k n) d N)) :
261     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
262       (y : Fin (n + 1) → LaurentSeries K)
263       (xi : Fin (n + 1) → PowerSeries K),
264       Sum.elim y
265           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
266         groundEquationConormalLocus (k := k) (K := K)
267           (reducedOrderBaseIdeal k
268             (canonicalRightIdeal (presentedCoordinate k n) d N)) ∧
269       residueColumn xi =
270         (fun i : Fin (n + 1) ↦
271           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) := by
272   apply exists_residueExtensionConormalAxis_of_ambientCoordinate_constant
273     (Nat.zero_lt_succ n)
274   · exact exists_reducedOrderBaseZero_of_support_nonempty
275       (canonicalRightIdeal (presentedCoordinate k n) d N) hsupp
276   · exact hconstant
