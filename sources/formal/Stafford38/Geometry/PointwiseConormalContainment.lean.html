9 /-!
10 # Pointwise conormal containment
11 
12 This file packages the finite-linear-combination consequence of the
13 base-relative Hamiltonian translation theorem.  At a fixed base point, every
14 finite linear combination of differentials of base equations whose lifts lie
15 in the ideal is a common zero of the ideal in the corresponding fibre.
16 
17 The proof forms the same linear combination of the base equations and applies
18 the existing one-direction translation theorem once.  It assumes zero-section
19 vanishing at the chosen point; it does not prove that hypothesis or any
20 homogeneity statement.
21 -/

37 /-- A finite linear combination of base equations. -/
38 def baseLinearCombination {ι : Type*} [Fintype ι]
39     (a : ι → k) (f : ι → MvPolynomial (Fin n) k) :
40     MvPolynomial (Fin n) k :=
41   ∑ j, MvPolynomial.C (a j) * f j
42 
43 /-- The zero-start affine fibre translation by the differential of the
44 corresponding finite linear combination of base equations. -/
45 abbrev differentialCombinationPoint {ι : Type*} [Fintype ι]
46     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k) :
47     PhaseVar n → k
48   := affineFibreTranslatePoint y (fun _ => 0)
49       (differentialAt y (baseLinearCombination a f)) 1
50 
51 /-- Differentiation at a point commutes with the displayed finite linear
52 combination. -/
53 theorem differentialAt_baseLinearCombination {ι : Type*} [Fintype ι]
54     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k)
55     (i : Fin n) :
56     differentialAt y (baseLinearCombination a f) i =
57       ∑ j, a j * differentialAt y (f j) i := by
58   classical
59   simp [baseLinearCombination, differentialAt, MvPolynomial.pderiv_mul]
60 
61 @[simp] theorem differentialCombinationPoint_inl {ι : Type*} [Fintype ι]
62     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k)
63     (i : Fin n) :
64     differentialCombinationPoint y a f (Sum.inl i) = y i := rfl
65 
66 @[simp] theorem differentialCombinationPoint_inr {ι : Type*} [Fintype ι]
67     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k)
68     (i : Fin n) :
69     differentialCombinationPoint y a f (Sum.inr i) =
70       ∑ j, a j * differentialAt y (f j) i := by
71   simp [differentialCombinationPoint, affineFibreTranslatePoint,
72     differentialAt_baseLinearCombination]

74 /-- With scalar parameter one, translation by the differential of the
75 combined equation is exactly the point represented by the combined
76 differentials. -/
77 theorem differentialTranslatePoint_baseLinearCombination_one
78     {ι : Type*} [Fintype ι]
79     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k) :
80     differentialTranslatePoint y (baseLinearCombination a f) 1 =
81       differentialCombinationPoint y a f := by
82   classical
83   funext i
84   rcases i with i | i
85   · rfl
86   · simp [differentialTranslatePoint, affineFibreTranslatePoint,
87       differentialCombinationPoint, differentialAt_baseLinearCombination]

89 /--
90 Pointwise finite-span form of conormal containment.
91 
92 If the zero-section point over `y` is a common zero of `J`, and every displayed
93 base equation lifts into `J`, then the fibre point obtained from any finite
94 linear combination of their differentials is again a common zero of `J`.
95 -/
96 theorem differentialCombinationPoint_isCommonZero
97     [CharZero k] {ι : Type*} [Fintype ι]
98     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
99     (y : Fin n → k)
100     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
101     (f : ι → MvPolynomial (Fin n) k)
102     (hf : ∀ j, baseLift (f j) ∈ J)
103     (a : ι → k) :
104     ∀ g ∈ J, MvPolynomial.eval (differentialCombinationPoint y a f) g = 0 := by
105   classical
106   have hcombined : baseLift (baseLinearCombination a f) ∈ J := by
107     rw [baseLinearCombination, map_sum]
108     apply Ideal.sum_mem
109     intro j hj
110     rw [map_mul]
111     exact J.mul_mem_left _ (hf j)
112   intro g hg
113   rw [← differentialTranslatePoint_baseLinearCombination_one y a f]
114   exact
115     zeroSection_stable_under_differential_translation_of_isBaseRelativePoisson
116       J hJ y hzero (baseLinearCombination a f) hcombined g hg 1
117 
118 /-- Concrete reduced-support form: a point of the actual reduced order
119 support supplies its zero-section point by order homogeneity, after which the
120 base-relative Gabber fragment contains every displayed finite differential
121 span in the same support fibre. -/
122 theorem reducedOrderSupport_differentialCombinationPoint_isCommonZero
123     [CharZero k] {ι : Type*} [Fintype ι]
124     (I : RightIdeal (WeylAlg k n))
125     (hJ : IsBaseRelativePoisson (reducedOrderSupportIdeal k I))
126     (y ξ : Fin n → k)
127     (hpoint : ∀ g ∈ reducedOrderSupportIdeal k I,
128       MvPolynomial.eval (Sum.elim y ξ) g = 0)
129     (f : ι → MvPolynomial (Fin n) k)
130     (hf : ∀ j, baseLift (f j) ∈ reducedOrderSupportIdeal k I)
131     (a : ι → k) :
132     ∀ g ∈ reducedOrderSupportIdeal k I,
133       MvPolynomial.eval (differentialCombinationPoint y a f) g = 0 := by
134   exact differentialCombinationPoint_isCommonZero
135     (reducedOrderSupportIdeal k I) hJ y
136     (zeroSection_mem_of_mem_reducedOrderSupport_zeroSet k I y ξ hpoint)
137     f hf a
