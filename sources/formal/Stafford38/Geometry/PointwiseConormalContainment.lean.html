7 /-!
8 # Pointwise conormal containment
9 
10 This file packages the finite-linear-combination consequence of the
11 base-relative Hamiltonian translation theorem.  At a fixed base point, every
12 finite linear combination of differentials of base equations whose lifts lie
13 in the ideal is a common zero of the ideal in the corresponding fibre.
14 
15 The proof forms the same linear combination of the base equations and applies
16 the existing one-direction translation theorem once.  It assumes zero-section
17 vanishing at the chosen point; it does not prove that hypothesis or any
18 homogeneity statement.
19 -/

35 /-- A finite linear combination of base equations. -/
36 def baseLinearCombination {ι : Type*} [Fintype ι]
37     (a : ι → k) (f : ι → MvPolynomial (Fin n) k) :
38     MvPolynomial (Fin n) k :=
39   ∑ j, MvPolynomial.C (a j) * f j
40 
41 /-- The zero-start affine fibre translation by the differential of the
42 corresponding finite linear combination of base equations. -/
43 abbrev differentialCombinationPoint {ι : Type*} [Fintype ι]
44     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k) :
45     PhaseVar n → k
46   := affineFibreTranslatePoint y (fun _ => 0)
47       (differentialAt y (baseLinearCombination a f)) 1
48 
49 /-- Differentiation at a point commutes with the displayed finite linear
50 combination. -/
51 theorem differentialAt_baseLinearCombination {ι : Type*} [Fintype ι]
52     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k)
53     (i : Fin n) :
54     differentialAt y (baseLinearCombination a f) i =
55       ∑ j, a j * differentialAt y (f j) i := by
56   classical
57   simp [baseLinearCombination, differentialAt, MvPolynomial.pderiv_mul]
58 
59 @[simp] theorem differentialCombinationPoint_inl {ι : Type*} [Fintype ι]
60     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k)
61     (i : Fin n) :
62     differentialCombinationPoint y a f (Sum.inl i) = y i := rfl
63 
64 @[simp] theorem differentialCombinationPoint_inr {ι : Type*} [Fintype ι]
65     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k)
66     (i : Fin n) :
67     differentialCombinationPoint y a f (Sum.inr i) =
68       ∑ j, a j * differentialAt y (f j) i := by
69   simp [differentialCombinationPoint, affineFibreTranslatePoint,
70     differentialAt_baseLinearCombination]

72 /-- With scalar parameter one, translation by the differential of the
73 combined equation is exactly the point represented by the combined
74 differentials. -/
75 theorem differentialTranslatePoint_baseLinearCombination_one
76     {ι : Type*} [Fintype ι]
77     (y : Fin n → k) (a : ι → k) (f : ι → MvPolynomial (Fin n) k) :
78     differentialTranslatePoint y (baseLinearCombination a f) 1 =
79       differentialCombinationPoint y a f := by
80   classical
81   funext i
82   rcases i with i | i
83   · rfl
84   · simp [differentialTranslatePoint, affineFibreTranslatePoint,
85       differentialCombinationPoint, differentialAt_baseLinearCombination]

87 /--
88 Pointwise finite-span form of conormal containment.
89 
90 If the zero-section point over `y` is a common zero of `J`, and every displayed
91 base equation lifts into `J`, then the fibre point obtained from any finite
92 linear combination of their differentials is again a common zero of `J`.
93 -/
94 theorem differentialCombinationPoint_isCommonZero
95     [CharZero k] {ι : Type*} [Fintype ι]
96     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
97     (y : Fin n → k)
98     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
99     (f : ι → MvPolynomial (Fin n) k)
100     (hf : ∀ j, baseLift (f j) ∈ J)
101     (a : ι → k) :
102     ∀ g ∈ J, MvPolynomial.eval (differentialCombinationPoint y a f) g = 0 := by
103   classical
104   have hcombined : baseLift (baseLinearCombination a f) ∈ J := by
105     rw [baseLinearCombination, map_sum]
106     apply Ideal.sum_mem
107     intro j hj
108     rw [map_mul]
109     exact J.mul_mem_left _ (hf j)
110   intro g hg
111   rw [← differentialTranslatePoint_baseLinearCombination_one y a f]
112   exact
113     zeroSection_stable_under_differential_translation_of_isBaseRelativePoisson
114       J hJ y hzero (baseLinearCombination a f) hcombined g hg 1

116 /-- Concrete reduced-support form: a point of the actual reduced order
117 support supplies its zero-section point by order homogeneity, after which the
118 base-relative Gabber fragment contains every displayed finite differential
119 span in the same support fibre. -/
120 theorem reducedOrderSupport_differentialCombinationPoint_isCommonZero
121     [CharZero k] {ι : Type*} [Fintype ι]
122     (I : RightIdeal (PresentedWeyl k n))
123     (hJ : IsBaseRelativePoisson (reducedOrderSupportIdeal k I))
124     (y ξ : Fin n → k)
125     (hpoint : ∀ g ∈ reducedOrderSupportIdeal k I,
126       MvPolynomial.eval (Sum.elim y ξ) g = 0)
127     (f : ι → MvPolynomial (Fin n) k)
128     (hf : ∀ j, baseLift (f j) ∈ reducedOrderSupportIdeal k I)
129     (a : ι → k) :
130     ∀ g ∈ reducedOrderSupportIdeal k I,
131       MvPolynomial.eval (differentialCombinationPoint y a f) g = 0 := by
132   exact differentialCombinationPoint_isCommonZero
133     (reducedOrderSupportIdeal k I) hJ y
134     (zeroSection_mem_of_mem_reducedOrderSupport_zeroSet k I y ξ hpoint)
135     f hf a
