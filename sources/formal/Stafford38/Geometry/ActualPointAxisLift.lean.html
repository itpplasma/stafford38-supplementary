22 /-- From one prescribed formally-etale ground point, its canonical completion,
23 centered selected rows, divisor/axis order factorizations, and nonzero equations
24 for the required generic open, choose one tilt and construct the complete
25 selected-row axis-lift certificate. The avoidance series is the product of
26 the smooth/bad open and both ordered coordinates, so no second tilt is chosen
27 later. -/
28 theorem exists_actual_point_axis_lift
29     {k B : Type u} [Field k] [Infinite k] [CharZero k]
30     [CommRing B] [IsDomain B] [Algebra k B] [Algebra.FiniteType k B]
31     {d n : ℕ} (M : Ideal B) [M.IsMaximal]
32     (eM : (B ⧸ M) ≃ₐ[k] k)
33     (f : MvPolynomial (Option (Fin d)) k →ₐ[k] B)
34     [Algebra (MvPolynomial (Option (Fin d)) k) B]
35     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) B]
36     [Algebra.EssFiniteType (MvPolynomial (Option (Fin d)) k) B]
37     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k)
38       (Localization.AtPrime M)]
39     (hmap : algebraMap (MvPolynomial (Option (Fin d)) k) B = f.toRingHom)
40     (s bad : B) (hs : f (MvPolynomial.X none) = s) (hsM : s ∈ M)
41     (hbad : bad ≠ 0)
42     (qB : Fin (n + 1) → B) (chart : Fin (n + 1)) (axis : Fin n)
43     (hqchart : qB chart = 1)
44     (a b : ℕ) (ha : 0 < a) (hab : a < b)
45     (u₀ u₁ : Localization.AtPrime M)
46     (hu₀ : IsUnit u₀) (hu₁ : IsUnit u₁)
47     (hq₀ : algebraMap B (Localization.AtPrime M) (qB 0) =
48       (algebraMap B (Localization.AtPrime M) s) ^ a * u₀)
49     (hq₁ : algebraMap B (Localization.AtPrime M) (qB axis.succ) =
50       (algebraMap B (Localization.AtPrime M) s) ^ b * u₁)
51     (rows : Fin d ↪ Fin (n + 1))
52     (hrows : ∀ j, f (MvPolynomial.X (some j)) = qB (rows j)) :
53     ∃ α : Fin d → k,
54       tiltedArc (k := k) α
55         ((localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
56           (algebraMap B (Localization.AtPrime M) bad)) *
57          localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
58           (algebraMap B (Localization.AtPrime M) (qB 0)) *
59          localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
60           (algebraMap B (Localization.AtPrime M) (qB axis.succ))) ≠ 0 ∧
61       SelectedCoordinateAxisLiftData (k := k)
62         (fun i => localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
63           (algebraMap B (Localization.AtPrime M) (qB i)))
64         rows chart 0 axis.succ a b
65         (fun j => residueCoordinates M eM (some j)) α
66         (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM u₀)
67         (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM u₁) := by
68   classical
69   let R := MvPolynomial (Option (Fin d)) k
70   let T := Localization.AtPrime M
71   let Φ := localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
72   let q : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k :=
73     fun i => Φ (algebraMap B T (qB i))
74   let U₀ : MvPowerSeries (Fin (d + 1)) k := Φ u₀
75   let U₁ : MvPowerSeries (Fin (d + 1)) k := Φ u₁
76   let badSeries : MvPowerSeries (Fin (d + 1)) k := Φ (algebraMap B T bad)
77   let badProduct : MvPowerSeries (Fin (d + 1)) k :=
78     badSeries * q 0 * q axis.succ
79   have hF (p : R) : f p = algebraMap R B p := by
80     change f.toRingHom p = algebraMap R B p
81     rw [← hmap]
82   have hsMap : f (MvPolynomial.X (R := k) none) =
83       algebraMap R B (MvPolynomial.X none) := hF _
84   have hnoneEq : algebraMap R B (MvPolynomial.X none) = s := by
85     rw [← hsMap]
86     exact hs
87   have hBeta : residueCoordinates (σ := Option (Fin d)) M eM none = 0 := by
88     unfold residueCoordinates
89     rw [Ideal.Quotient.mkₐ_eq_mk]
90     have hmk : Ideal.Quotient.mk M (algebraMap R B (MvPolynomial.X none)) = 0 :=
91       Ideal.Quotient.eq_zero_iff_mem.mpr (hnoneEq ▸ hsM)
92     rw [hmk]
93     simp
94   have hsPhi : Φ (algebraMap B T s) = MvPowerSeries.X (0 : Fin (d + 1)) := by
95     calc
96       Φ (algebraMap B T s) =
97           Φ (algebraMap B T (algebraMap R B (MvPolynomial.X none))) := by
98             rw [hnoneEq]
99       _ = MvPowerSeries.C (residueCoordinates M eM none) +
100           MvPowerSeries.X (0 : Fin (d + 1)) := by
101             exact localToFinSuccPowerSeries_none (k := k) (B := B) (d := d) M eM
102       _ = MvPowerSeries.X (0 : Fin (d + 1)) := by rw [hBeta]; simp
103   have hrow : ∀ j, q (rows j) =
104       MvPowerSeries.C (residueCoordinates M eM (some j)) +
105         MvPowerSeries.X j.succ := by
106     intro j
107     calc
108       q (rows j) = Φ (algebraMap B T (f (MvPolynomial.X (some j)))) := by
109         simp [q, hrows j]
110       _ = Φ (algebraMap B T (algebraMap R B (MvPolynomial.X (some j)))) := by
111         rw [hF]
112       _ = MvPowerSeries.C (residueCoordinates M eM (some j)) +
113           MvPowerSeries.X j.succ := by
114         exact localToFinSuccPowerSeries_some (k := k) (B := B) (d := d)
115           M eM j
116   have hqchart' : q chart = 1 := by
117     simp [q, hqchart]
118   have hq₀' : q 0 = MvPowerSeries.X (0 : Fin (d + 1)) ^ a * U₀ := by
119     calc
120       q 0 = Φ (algebraMap B T (qB 0)) := rfl
121       _ = Φ ((algebraMap B T s) ^ a * u₀) := congrArg Φ hq₀
122       _ = Φ (algebraMap B T s) ^ a * Φ u₀ := by simp
123       _ = MvPowerSeries.X (0 : Fin (d + 1)) ^ a * U₀ := by rw [hsPhi]
124   have hq₁' : q axis.succ =
125       MvPowerSeries.X (0 : Fin (d + 1)) ^ b * U₁ := by
126     calc
127       q axis.succ = Φ (algebraMap B T (qB axis.succ)) := rfl
128       _ = Φ ((algebraMap B T s) ^ b * u₁) := congrArg Φ hq₁
129       _ = Φ (algebraMap B T s) ^ b * Φ u₁ := by simp
130       _ = MvPowerSeries.X (0 : Fin (d + 1)) ^ b * U₁ := by rw [hsPhi]
131   have hU₀ : IsUnit U₀ := IsUnit.map Φ hu₀
132   have hU₁ : IsUnit U₁ := IsUnit.map Φ hu₁
133   have hU₀cc : MvPowerSeries.constantCoeff U₀ ≠ 0 := by
134     have h := MvPowerSeries.isUnit_constantCoeff U₀ hU₀
135     exact isUnit_iff_ne_zero.mp h
136   have hU₁cc : MvPowerSeries.constantCoeff U₁ ≠ 0 := by
137     have h := MvPowerSeries.isUnit_constantCoeff U₁ hU₁
138     exact isUnit_iff_ne_zero.mp h
139   have hbadSne : badSeries ≠ 0 := by
140     exact localToFinSuccPowerSeries_algebraMap_ne_zero
141       (k := k) (B := B) (d := d) M eM bad hbad
142   have hX : MvPowerSeries.X (0 : Fin (d + 1)) ≠
143       (0 : MvPowerSeries (Fin (d + 1)) k) := by
144     intro hx
145     have hc := congrArg
146       (MvPowerSeries.coeff (Finsupp.single (0 : Fin (d + 1)) 1)) hx
147     simpa using hc
148   have hq₀ne : q 0 ≠ 0 := by
149     rw [hq₀']
150     exact mul_ne_zero (pow_ne_zero a hX) hU₀.ne_zero
151   have hq₁ne : q axis.succ ≠ 0 := by
152     rw [hq₁']
153     exact mul_ne_zero (pow_ne_zero b hX) hU₁.ne_zero
154   have hbadProduct : badProduct ≠ 0 := by
155     dsimp [badProduct]
156     exact mul_ne_zero (mul_ne_zero hbadSne hq₀ne) hq₁ne
157   obtain ⟨α, havoid, hdata⟩ :=
158     exists_tilted_local_axis_lift_of_selected_coordinate_rows
159       (k := k) q badProduct hbadProduct rows chart 0 axis.succ a b
160       (fun j => residueCoordinates M eM (some j)) U₀ U₁
161       hqchart' ha hab hq₀' hU₀cc hq₁' hU₁cc hrow
162   refine ⟨α, ?_, hdata⟩
163   simpa [badProduct, badSeries, q, Φ] using havoid
164 
165 end Stafford38.Geometry.ActualPointAxisLift
