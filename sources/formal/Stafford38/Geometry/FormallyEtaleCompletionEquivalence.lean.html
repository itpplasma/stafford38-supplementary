20 private theorem factor_evalₐ_comp (m n : ℕ) (hmn : m ≤ n) (x : AdicCompletion I A) :
21     Ideal.Quotient.factorₐ A (Ideal.pow_le_pow_right hmn) (AdicCompletion.evalₐ I n x) =
22       AdicCompletion.evalₐ I m x := by
23   apply AdicCompletion.induction_on I A x
24   intro c
25   rw [AdicCompletion.evalₐ_mk, AdicCompletion.evalₐ_mk]
26   apply Ideal.Quotient.eq.mpr
27   have hc : c.val m - c.val n ∈ I ^ m := by
28     have hc' := SModEq.sub_mem.mp (c.property hmn)
29     simpa [smul_eq_mul, Ideal.mul_top] using hc'
30   simpa using (I ^ m).neg_mem hc

40     A ⧸ I ^ n →ₐ[A] B ⧸ J ^ n :=

53 /-- A lift to the base completion induces maps in the reverse direction at every
54 finite quotient level. -/
55 def reverseStage (hJ : J = I.map (algebraMap A B))
56     (ψ : B →ₐ[A] AdicCompletion I A) (n : ℕ) : B ⧸ J ^ n →ₐ[A] A ⧸ I ^ n := by
57   let u : B →ₐ[A] A ⧸ I ^ n := (AdicCompletion.evalₐ I n).comp ψ
58   have hkill : ∀ b ∈ J ^ n, u b = 0 := by
59     have hker : (I ^ n).map (algebraMap A B) ≤ RingHom.ker (u : B →+* A ⧸ I ^ n) := by
60       apply (Ideal.map_le_iff_le_comap).2
61       intro a ha
62       change AdicCompletion.evalₐ I n (ψ (algebraMap A B a)) = 0
63       rw [ψ.commutes]
64       simp [Ideal.Quotient.eq_zero_iff_mem, ha]
65     rw [← idealPowMap_eq I J n hJ]
66     exact fun b hb => RingHom.mem_ker.mp (hker hb)
67   exact Ideal.Quotient.liftₐ (R₁ := A) (J ^ n) u hkill

69 private theorem reverseStage_mk (hJ : J = I.map (algebraMap A B))
70     (ψ : B →ₐ[A] AdicCompletion I A) (n : ℕ) (b : B) :
71     reverseStage I J hJ ψ n (Ideal.Quotient.mkₐ A (J ^ n) b) =
72       AdicCompletion.evalₐ I n (ψ b) := by
73   let u : B →ₐ[A] A ⧸ I ^ n := (AdicCompletion.evalₐ I n).comp ψ
74   have hkill : ∀ x ∈ J ^ n, u x = 0 := by
75     unfold u
76     have hker : (I ^ n).map (algebraMap A B) ≤
77         RingHom.ker ((AdicCompletion.evalₐ I n).comp ψ : B →+* A ⧸ I ^ n) := by
78       apply (Ideal.map_le_iff_le_comap).2
79       intro a ha
80       change AdicCompletion.evalₐ I n (ψ (algebraMap A B a)) = 0
81       rw [ψ.commutes]
82       simp [Ideal.Quotient.eq_zero_iff_mem, ha]
83     rw [← idealPowMap_eq I J n hJ]
84     exact fun x hx => RingHom.mem_ker.mp (hker hx)
85   have hcomp := Ideal.Quotient.liftₐ_comp (R₁ := A) (J ^ n) u hkill
86   change ((Ideal.Quotient.liftₐ (R₁ := A) (J ^ n) u hkill).comp
87       (Ideal.Quotient.mkₐ A (J ^ n))) b = _
88   exact DFunLike.congr_fun hcomp b

90 private theorem reverseStage_compatible (hJ : J = I.map (algebraMap A B))
91     (ψ : B →ₐ[A] AdicCompletion I A) {m n : ℕ} (hmn : m ≤ n) :
92     (Ideal.Quotient.factorₐ A (Ideal.pow_le_pow_right hmn)).comp
93         (reverseStage I J hJ ψ n) =
94       (reverseStage I J hJ ψ m).comp
95         (Ideal.Quotient.factorₐ A (Ideal.pow_le_pow_right hmn)) := by
96   apply Ideal.Quotient.algHom_ext A
97   ext b
98   calc
99     Ideal.Quotient.factorₐ A (Ideal.pow_le_pow_right hmn)
100         (reverseStage I J hJ ψ n (Ideal.Quotient.mkₐ A (J ^ n) b)) =
101       Ideal.Quotient.factorₐ A (Ideal.pow_le_pow_right hmn)
102         (AdicCompletion.evalₐ I n (ψ b)) := by
103       rw [reverseStage_mk]
104     _ = AdicCompletion.evalₐ I m (ψ b) := factor_evalₐ_comp I m n hmn (ψ b)
105     _ = reverseStage I J hJ ψ m (Ideal.Quotient.mkₐ A (J ^ m) b) := by
106       exact (reverseStage_mk I J hJ ψ m b).symm

108 theorem reverseCompletionMap_compatible (hJ : J = I.map (algebraMap A B))
109     (ψ : B →ₐ[A] AdicCompletion I A) {m n : ℕ} (hmn : m ≤ n) :
110     (Ideal.Quotient.factorPow I hmn).comp
111         (((reverseStage I J hJ ψ n).toRingHom).comp (AdicCompletion.evalₐ J n).toRingHom) =
112       ((reverseStage I J hJ ψ m).toRingHom).comp (AdicCompletion.evalₐ J m).toRingHom := by
113   ext x
114   change Ideal.Quotient.factorPow I hmn
115       (reverseStage I J hJ ψ n (AdicCompletion.evalₐ J n x)) =
116     reverseStage I J hJ ψ m (AdicCompletion.evalₐ J m x)
117   have hstage := DFunLike.congr_fun
118     (reverseStage_compatible I J hJ ψ hmn) (AdicCompletion.evalₐ J n x)
119   calc
120     Ideal.Quotient.factorPow I hmn
121         (reverseStage I J hJ ψ n (AdicCompletion.evalₐ J n x)) =
122       reverseStage I J hJ ψ m
123         (Ideal.Quotient.factorPow J hmn (AdicCompletion.evalₐ J n x)) := hstage
124     _ = reverseStage I J hJ ψ m (AdicCompletion.evalₐ J m x) := by
125       congr 1
126       exact factor_evalₐ_comp J m n hmn x

128 /-- The reverse map on completions induced by the supplied lift to the base completion. -/

155 private theorem factor_to_first_power_ker_nilpotent (n : ℕ) :
156     IsNilpotent (RingHom.ker
157       (Ideal.Quotient.factor (Ideal.pow_le_pow_right
158         (show 1 ≤ n + 1 by omega)) :
159           B ⧸ J ^ (n + 1) →+* B ⧸ J ^ 1)) := by
160   let hpow : J ^ (n + 1) ≤ J ^ 1 :=
161     Ideal.pow_le_pow_right (show 1 ≤ n + 1 by omega)
162   change IsNilpotent (RingHom.ker (Ideal.Quotient.factor hpow))
163   rw [Ideal.Quotient.factor_ker hpow]
164   refine ⟨n + 1, ?_⟩
165   rw [← Ideal.map_pow]
166   have hp : (J ^ 1) ^ (n + 1) = J ^ (n + 1) := by simp
167   rw [hp]
168   simp

170 private theorem forward_reverse_stage
171     [Algebra.FormallyUnramified A B]
172     (hJ : J = I.map (algebraMap A B))
173     (ψ : B →ₐ[A] AdicCompletion I A)
174     (hres : (forwardStage I J hJ 1).comp (reverseStage I J hJ ψ 1) =
175       AlgHom.id A (B ⧸ J ^ 1)) (n : ℕ) :
176     (forwardStage I J hJ n).comp (reverseStage I J hJ ψ n) =
177       AlgHom.id A (B ⧸ J ^ n) := by
178   cases n with
179   | zero =>
180     apply AlgHom.ext
181     intro b
182     haveI : Subsingleton (B ⧸ J ^ 0) := by simp
183     exact Subsingleton.elim _ _
184   | succ k =>
185     let hn : 1 ≤ k + 1 := by omega
186     let f : B ⧸ J ^ (k + 1) →+* B ⧸ J ^ 1 :=
187       Ideal.Quotient.factor (Ideal.pow_le_pow_right hn)
188     have hnil : IsNilpotent (RingHom.ker f) := by
189       simpa [f, hn] using factor_to_first_power_ker_nilpotent (J := J) k
190     let g₁ : B →ₐ[A] B ⧸ J ^ (k + 1) :=
191       ((forwardStage I J hJ (k + 1)).comp (reverseStage I J hJ ψ (k + 1))).comp
192         (Ideal.Quotient.mkₐ A (J ^ (k + 1)))
193     let g₂ : B →ₐ[A] B ⧸ J ^ (k + 1) := Ideal.Quotient.mkₐ A (J ^ (k + 1))
194     have hred : f.comp (g₁ : B →+* B ⧸ J ^ (k + 1)) =
195         f.comp (g₂ : B →+* B ⧸ J ^ (k + 1)) := by
196       ext b
197       change f (forwardStage I J hJ (k + 1)
198           (reverseStage I J hJ ψ (k + 1) (Ideal.Quotient.mkₐ A (J ^ (k + 1)) b))) =
199         Ideal.Quotient.mk (J ^ 1) b
200       calc
201         f (forwardStage I J hJ (k + 1)
202             (reverseStage I J hJ ψ (k + 1) (Ideal.Quotient.mkₐ A (J ^ (k + 1)) b))) =
203             forwardStage I J hJ 1
204             (Ideal.Quotient.factor (Ideal.pow_le_pow_right hn)
205               (reverseStage I J hJ ψ (k + 1)
206                 (Ideal.Quotient.mkₐ A (J ^ (k + 1)) b))) := by
207             exact DFunLike.congr_fun
208               (forwardStage_compatible I J hJ hn)
209               (reverseStage I J hJ ψ (k + 1) (Ideal.Quotient.mkₐ A (J ^ (k + 1)) b))
210         _ = forwardStage I J hJ 1
211             (reverseStage I J hJ ψ 1 (Ideal.Quotient.mkₐ A (J ^ 1) b)) := by
212             congr 1
213             exact DFunLike.congr_fun
214               (reverseStage_compatible I J hJ ψ hn)
215               (Ideal.Quotient.mkₐ A (J ^ (k + 1)) b)
216         _ = Ideal.Quotient.mk (J ^ 1) b := by
217             have h := DFunLike.congr_fun hres (Ideal.Quotient.mkₐ A (J ^ 1) b)
218             simpa using h
219     have huniq := Algebra.FormallyUnramified.lift_unique_of_ringHom
220       (R := A) (A := B) f hnil g₁ g₂ hred
221     apply Ideal.Quotient.algHom_ext A
222     ext b
223     exact DFunLike.congr_fun huniq b

238 theorem forward_reverse_completion
239     [Algebra.FormallyUnramified A B]
240     (hJ : J = I.map (algebraMap A B))
241     (ψ : B →ₐ[A] AdicCompletion I A)
242     (hres : (forwardStage I J hJ 1).comp (reverseStage I J hJ ψ 1) =
243       AlgHom.id A (B ⧸ J ^ 1))
244     (x : AdicCompletion J B) :
245     forwardCompletionMap I J hJ (reverseCompletionMap I J hJ ψ x) = x := by
246   apply AdicCompletion.ext_evalₐ
247   intro n
248   rw [eval_forwardCompletionMap, eval_reverseCompletionMap]
249   exact DFunLike.congr_fun (forward_reverse_stage I J hJ ψ hres n)

252 theorem reverse_forward_completion
253     (hJ : J = I.map (algebraMap A B))
254     (ψ : B →ₐ[A] AdicCompletion I A)

263 /-- A formally unramified algebra with an adic residue retraction has
264 isomorphic completions. The explicit `ψ` and first-quotient identity are the
265 lift and residue-identification data required by this comparison. -/
266 noncomputable def formalEtaleCompletionEquiv
267     [Algebra.FormallyUnramified A B]
268     (hJ : J = I.map (algebraMap A B))
269     (ψ : B →ₐ[A] AdicCompletion I A)
270     (hres : (forwardStage I J hJ 1).comp (reverseStage I J hJ ψ 1) =
271       AlgHom.id A (B ⧸ J ^ 1)) :
272     AdicCompletion I A ≃+* AdicCompletion J B where
273   toFun := forwardCompletionMap I J hJ
274   invFun := reverseCompletionMap I J hJ ψ
275   left_inv := reverse_forward_completion I J hJ ψ
276   right_inv := forward_reverse_completion I J hJ ψ hres
277   map_mul' := (forwardCompletionMap I J hJ).map_mul
278   map_add' := (forwardCompletionMap I J hJ).map_add
279 
280 end
