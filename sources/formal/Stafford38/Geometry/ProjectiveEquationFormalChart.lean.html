8 /-!
9 # Projective equations on a formal affine chart
10 
11 This file removes the base-equation part of the formal-divisor producer.
12 A homogeneous projective equation which vanishes at a projective point also
13 vanishes after dehomogenizing in any nonzero chart.  The result is extended
14 from a family of homogeneous equations to the affine ideal generated by their
15 dehomogenizations, and then connected directly to the formal Laurent conormal
16 handoff.
17 
18 No projective closure, normalization, divisor, formal parameter, smoothness,
19 or tangent-space equality is constructed here.  Those geometric existence
20 statements remain outside this algebraic chart calculation.
21 -/

38 /-- Normalize every projective coordinate by the chosen chart denominator. -/
39 def normalizedProjectivePoint {m : ℕ} (q : Fin m → K) (q₀ : K) : Fin m → K :=
40   fun i ↦ q i / q₀

42 /-- Substitute `X₀ = 1` and `X_(i+1) = X_i` in a projective polynomial. -/
43 abbrev projectiveDehomogenize :
44     MvPolynomial (Fin (n + 1)) K →ₐ[K] MvPolynomial (Fin n) K :=
45   dehomogenizeProjectiveChart (0 : Fin (n + 1)) (zerothChartEquiv (m := n))

47 /-- Evaluation after projective dehomogenization is evaluation at the
48 projective chart point `(1,y)`. -/
49 theorem eval_projectiveDehomogenize
50     (p : MvPolynomial (Fin (n + 1)) K) (y : Fin n → K) :
51     MvPolynomial.eval y (projectiveDehomogenize p) =
52       MvPolynomial.eval (Fin.cases 1 y) p := by
53   rw [eval_dehomogenizeProjectiveChart, insertProjectiveChart_zero]

55 /-- Scaling all coordinates of a homogeneous polynomial scales its value by
56 the corresponding power.  This explicit finite-product form is the algebraic
57 valuation identity used in the chart argument. -/
58 theorem eval_eq_pow_mul_eval_normalizedProjectivePoint
59     {m : ℕ} (p : MvPolynomial (Fin m) K) (d : ℕ)
60     (hp : p.IsHomogeneous d) (q : Fin m → K) (q₀ : K) (hq₀ : q₀ ≠ 0) :
61     MvPolynomial.eval q p =
62       q₀ ^ d * MvPolynomial.eval (normalizedProjectivePoint q q₀) p := by
63   change p.eval₂ (RingHom.id K) q =
64     q₀ ^ d * p.eval₂ (RingHom.id K) (normalizedProjectivePoint q q₀)
65   rw [MvPolynomial.eval₂_eq', MvPolynomial.eval₂_eq']
66   rw [Finset.mul_sum]
67   apply Finset.sum_congr rfl
68   intro s hs
69   have hdegree : ∑ i, s i = d := by
70     have hweight := hp (MvPolynomial.mem_support_iff.mp hs)
71     rw [Finsupp.weight_apply,
72       Finsupp.sum_fintype s
73         (fun i c ↦ c • (1 : Fin m → ℕ) i) (by simp)] at hweight
74     simpa only [Pi.one_apply, smul_eq_mul, mul_one] using hweight
75   have hcoordinate : ∀ i : Fin m,
76       q₀ * normalizedProjectivePoint q q₀ i = q i := by
77     intro i
78     exact mul_div_cancel₀ (q i) hq₀
79   have hproduct : (∏ i : Fin m, q i ^ s i) =
80       q₀ ^ d * ∏ i : Fin m, normalizedProjectivePoint q q₀ i ^ s i := by
81     calc
82       (∏ i : Fin m, q i ^ s i) =
83           ∏ i : Fin m, (q₀ * normalizedProjectivePoint q q₀ i) ^ s i := by
84             simp_rw [hcoordinate]
85       _ = (∏ i : Fin m, q₀ ^ s i) *
86           ∏ i : Fin m, normalizedProjectivePoint q q₀ i ^ s i := by
87             simp_rw [mul_pow]
88             rw [Finset.prod_mul_distrib]
89       _ = q₀ ^ d *
90           ∏ i : Fin m, normalizedProjectivePoint q q₀ i ^ s i := by
91             rw [Finset.prod_pow_eq_pow_sum, hdegree]
92   rw [hproduct]
93   ring

95 /-- The normalized projective point is exactly `(1, dehomogenizedPoint q)`
96 in the chart with denominator `q 0`. -/
97 theorem normalizedProjectivePoint_eq_chartPoint
98     (q : Fin (n + 1) → K) (hq₀ : q 0 ≠ 0) :
99     normalizedProjectivePoint q (q 0) =
100       Fin.cases 1 (dehomogenizedPoint q) := by
101   funext i
102   refine Fin.cases ?_ (fun j ↦ ?_) i
103   · simp [normalizedProjectivePoint, hq₀]
104   · rfl

106 /-- A homogeneous projective equation vanishing at `q` gives a vanishing
107 affine equation after dehomogenization in the nonzero zeroth chart. -/
108 theorem eval_projectiveDehomogenize_eq_zero_of_homogeneous
109     (p : MvPolynomial (Fin (n + 1)) K) (d : ℕ)
110     (hp : p.IsHomogeneous d) (q : Fin (n + 1) → K)
111     (hq₀ : q 0 ≠ 0) (hvanish : MvPolynomial.eval q p = 0) :
112     MvPolynomial.eval (dehomogenizedPoint q) (projectiveDehomogenize p) = 0 := by
113   have hscale := eval_eq_pow_mul_eval_normalizedProjectivePoint
114     p d hp q (q 0) hq₀
115   rw [hvanish] at hscale
116   have hnormalized :
117       MvPolynomial.eval (normalizedProjectivePoint q (q 0)) p = 0 :=
118     (mul_eq_zero.mp hscale.symm).resolve_left (pow_ne_zero d hq₀)
119   rw [eval_projectiveDehomogenize,
120     ← normalizedProjectivePoint_eq_chartPoint q hq₀]
121   exact hnormalized

123 /-- The affine ideal generated by dehomogenizing a supplied family of
124 projective equations. -/
125 abbrev dehomogenizedEquationIdeal {I : Type*}
126     (equations : I → MvPolynomial (Fin (n + 1)) K) :
127     Ideal (MvPolynomial (Fin n) K) :=
128   Stafford38.Geometry.ProjectiveChartCoordinates.dehomogenizedEquationIdeal
129     0 (zerothChartEquiv (m := n)) equations
130 
131 /-- Vanishing descends not only for each homogeneous generator but for every
132 element of the affine ideal generated by their dehomogenizations. -/
133 theorem eval_eq_zero_of_mem_dehomogenizedEquationIdeal
134     {I : Type*} (equations : I → MvPolynomial (Fin (n + 1)) K)
135     (degree : I → ℕ)
136     (hhomogeneous : ∀ i, (equations i).IsHomogeneous (degree i))
137     (q : Fin (n + 1) → K) (hq₀ : q 0 ≠ 0)
138     (hvanish : ∀ i, MvPolynomial.eval q (equations i) = 0)
139     (f : MvPolynomial (Fin n) K)
140     (hf : f ∈ dehomogenizedEquationIdeal equations) :
141     MvPolynomial.eval (dehomogenizedPoint q) f = 0 := by
142   apply Stafford38.Geometry.ProjectiveChartCoordinates.eval_eq_zero_of_mem_dehomogenizedEquationIdeal
143     0 (zerothChartEquiv (m := n)) equations (dehomogenizedPoint q) ?_ f hf
144   intro i
145   rw [insertProjectiveChart_zero, ← eval_projectiveDehomogenize]
146   exact eval_projectiveDehomogenize_eq_zero_of_homogeneous
147     (equations i) (degree i) (hhomogeneous i) q hq₀ (hvanish i)

149 /-- Direct formal-chart handoff.  For the affine ideal generated by the
150 dehomogenized homogeneous projective equations, projective vanishing supplies
151 the complete base-equation hypothesis required by
152 `FormalDivisorLaurentConormal`.  Only the exact tangent-space identification
153 remains as geometric input. -/
154 theorem laurentPhasePoint_mem_dehomogenizedEquationConormalLocus
155     {k : Type*} [Field k] {I κ : Type*} [Fintype κ]
156     (equations : I → MvPolynomial (Fin (n + 1)) (LaurentSeries k))
157     (degree : I → ℕ)
158     (hhomogeneous : ∀ i, (equations i).IsHomogeneous (degree i))
159     (q ell : Fin (n + 1) → PowerSeries k)
160     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
161     (tau : Fin (n + 1) → PowerSeries k)
162     (hq₀ : q 0 ≠ 0)
163     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
164     (hvanish : ∀ i, MvPolynomial.eval (laurentColumn q) (equations i) = 0)
165     (htangent :
166       dehomogenizedTangentSpan (laurentColumn q)
167           (laurentNonpositionTangentMatrix Z tau) =
168         zariskiTangentSpace (dehomogenizedPoint (laurentColumn q))
169           (dehomogenizedEquationIdeal equations)) :
170     Sum.elim (dehomogenizedPoint (laurentColumn q))
171         (fun i ↦ laurentColumn ell i.succ) ∈
172       equationConormalLocus (dehomogenizedEquationIdeal equations) := by
173   apply laurentPhasePoint_mem_equationConormalLocus
174     (dehomogenizedEquationIdeal equations) q ell Z tau hq₀ hrow
175   · intro f hf
176     exact eval_eq_zero_of_mem_dehomogenizedEquationIdeal
177       equations degree hhomogeneous (laurentColumn q)
178         (laurentColumn_ne_zero_of_ne_zero q hq₀) hvanish f hf
179   · exact htangent
