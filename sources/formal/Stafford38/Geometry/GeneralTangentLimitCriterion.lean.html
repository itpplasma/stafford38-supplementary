16 /-!
17 # The split-lattice tangent-limit calculation
18 
19 This file records the tangent-limit criterion after the affine-chart and
20 tangent-lattice dictionaries have been made explicit.  It uses no
21 normalization theorem.  A
22 power-series matrix is the lattice of projective tangent columns.  A left
23 inverse is the direct-summand certificate.  The axis condition is imposed on
24 the reduced lattice, and the annihilator is constructed by the retraction
25 correction; it is not an input.
26 
27 The Laurent-valued smooth equation-conormal direction hull is transported to
28 the ground-field projective closure by the existing scalar-extension and
29 smooth-fibre vanishing theorems.  `Input` is the internal split-matrix kernel.
30 The paper-facing `DirectSummandInput` instead takes the formal arc and the
31 actual complemented tangent lattice.  The theorem below constructs the
32 matrix presentation and proves the projective-closure and tangent-chart
33 dictionaries before applying that kernel.
34 -/

59 /-- The scalar-extended smooth equation-conormal phase locus used by the
60 Laurent specialization theorem.  The smoothness predicate is the genuine
61 `SmoothAffinePoint` predicate; it is not a dimension label or a placeholder.
62 -/
63 def smoothEquationConormalLocus {n : ℕ}
64     (I : Ideal (MvPolynomial (Fin n) k)) :
65     Set (PhaseVar n → LaurentSeries k) :=
66   {q | q ∈ equationConormalLocus
67       (I.map (scalarPolynomialMap
68         (k := k) (K := LaurentSeries k) (Fin n))) ∧
69     SmoothAffinePoint
70       (I.map (scalarPolynomialMap
71         (k := k) (K := LaurentSeries k) (Fin n)))
72       (fun i ↦ q (Sum.inl i))}

76 /-- Coordinatewise extension of scalars on a finite coordinate module. -/
77 def extendColumn {R K : Type*} [CommSemiring R] [CommSemiring K]
78     [Algebra R K] {ι : Type*} :
79     (ι → R) →ₛₗ[algebraMap R K] (ι → K) where
80   toFun v i := algebraMap R K (v i)
81   map_add' x y := by ext i; simp
82   map_smul' a x := by ext i; simp

84 /-- The ordinary scalar extension of a lattice, presented as the span of its
85 coordinatewise images.  This definition avoids choosing a basis. -/
86 def genericFibre {R K : Type*} [CommSemiring R] [CommSemiring K]
87     [Algebra R K] {ι : Type*} (L : Submodule R (ι → R)) :
88     Submodule K (ι → K) :=
89   Submodule.span K (extendColumn (R := R) (K := K) '' (L : Set (ι → R)))

105 /-- The affine cone over the embedded projective tangent plane in the chart
106 `q 0 ≠ 0`: it is the inverse image of the affine tangent space under the
107 chart derivative.  Its radial kernel contains `q`. -/
108 def projectiveTangentCone {K : Type*} [Field K] {n : ℕ}
109     (q : Fin (n + 1) → K) (T : Submodule K (Fin n → K)) :
110     Submodule K (Fin (n + 1) → K) :=
111   T.comap (dehomogenizedTangentLinearMap q)

113 /-- Equation-level membership in the projective closure in the zeroth chart.
114 The equations are the standard homogenizations of every affine equation, so
115 this is the ordinary homogeneous-equation presentation of the closure. -/
116 def projectiveClosureAtZero {K : Type*} [Field K] {n : ℕ}
117     (I : Ideal (MvPolynomial (Fin n) K))
118     (q : Fin (n + 1) → K) : Prop :=
119   ∀ f ∈ I, MvPolynomial.eval q (homogenizeAtZero f) = 0

121 /-- In the nonzero zeroth chart, the homogeneous-equation presentation of the
122 projective closure is equivalent to membership in the original affine zero
123 locus. -/
124 theorem projectiveClosureAtZero_iff {K : Type*} [Field K] {n : ℕ}
125     (I : Ideal (MvPolynomial (Fin n) K))
126     (q : Fin (n + 1) → K) (hq0 : q 0 ≠ 0) :
127     projectiveClosureAtZero I q ↔
128       dehomogenizedPoint q ∈ MvPolynomial.zeroLocus K I := by
129   constructor
130   · intro h f hf
131     have heval := eval_projectiveDehomogenize_eq_zero_of_homogeneous
132       (homogenizeAtZero f) f.totalDegree
133       (homogenizeAtZero_isHomogeneous f) q hq0 (h f hf)
134     simpa only [projectiveDehomogenize_homogenizeAtZero,
135       MvPolynomial.aeval_eq_eval] using heval
136   · intro h f hf
137     have hnormalized :
138         MvPolynomial.eval (normalizedProjectivePoint q (q 0))
139           (homogenizeAtZero f) = 0 := by
140       rw [normalizedProjectivePoint_eq_chartPoint q hq0,
141         ← eval_projectiveDehomogenize,
142         projectiveDehomogenize_homogenizeAtZero]
143       exact h f hf
144     rw [eval_eq_pow_mul_eval_normalizedProjectivePoint
145       (homogenizeAtZero f) f.totalDegree
146       (homogenizeAtZero_isHomogeneous f) q (q 0) hq0,
147       hnormalized, mul_zero]

149 /-- A formal projective arc represented by power-series coordinates.  The
150 unit-coordinate condition makes the tuple a morphism to projective space;
151 the second field is membership in the homogeneous-equation closure after
152 passing to the generic Laurent point. -/
153 def FormalProjectiveArcInClosure {n : ℕ}
154     (I : Ideal (MvPolynomial (Fin n) k))
155     (q : Fin (n + 1) → PowerSeries k) : Prop :=
156   (∃ i, IsUnit (q i)) ∧
157     projectiveClosureAtZero
158       (I.map (scalarPolynomialMap
159         (k := k) (K := LaurentSeries k) (Fin n)))
160       (laurentColumn q)

180 /-- The columns of a matrix presentation span the generic fibre of the
181 original lattice after scalar extension. -/
182 theorem genericFibre_eq_span_matrixColumns
183     {R K : Type*} [CommRing R] [Field K] [Algebra R K]
184     {ι κ : Type*} [Fintype κ]
185     (L : Submodule R (ι → R)) (B : Matrix ι κ R)
186     (hspan : Submodule.span R (Set.range fun j => fun i => B i j) = L) :
187     genericFibre (K := K) L =
188       Submodule.span K (Set.range fun j => fun i =>
189         algebraMap R K (B i j)) := by
190   let f := extendColumn (R := R) (K := K) (ι := ι)
191   let columns : κ → (ι → R) := fun j i => B i j
192   apply le_antisymm
193   · apply Submodule.span_le.mpr
194     rintro _ ⟨x, hx, rfl⟩
195     have hxspan : x ∈ Submodule.span R (Set.range columns) := by
196       rw [hspan]
197       exact hx
198     obtain ⟨c, hc⟩ :=
199       (Submodule.mem_span_range_iff_exists_fun R).mp hxspan
200     apply (Submodule.mem_span_range_iff_exists_fun K).mpr
201     refine ⟨fun j => algebraMap R K (c j), ?_⟩
202     apply funext
203     intro i
204     simp only [Finset.sum_apply, Pi.smul_apply, smul_eq_mul]
205     have hi : (∑ j, c j * B i j) = x i := by
206       simpa only [columns, Finset.sum_apply, Pi.smul_apply,
207         smul_eq_mul] using congrFun hc i
208     change _ = algebraMap R K (x i)
209     simpa only [map_sum, map_mul] using congrArg (algebraMap R K) hi
210   · apply Submodule.span_le.mpr
211     rintro _ ⟨j, rfl⟩
212     apply Submodule.subset_span
213     refine ⟨columns j, ?_, rfl⟩
214     rw [← hspan]
215     exact Submodule.subset_span ⟨j, rfl⟩

217 /-- The chart derivative is onto whenever the chart denominator is nonzero. -/
218 private theorem dehomogenizedTangentLinearMap_surjective
219     {K : Type*} [Field K] {n : ℕ}
220     (q : Fin (n + 1) → K) (hq0 : q 0 ≠ 0) :
221     Function.Surjective (dehomogenizedTangentLinearMap q) := by
222   intro v
223   refine ⟨Fin.cases 0 (fun i => v i * q 0), ?_⟩
224   ext i
225   change (v i * q 0 * q 0 - q i.succ * 0) / q 0 ^ 2 = v i
226   field_simp
227   ring

229 /-- If the scalar-extended lattice is the projective tangent cone, any matrix
230 whose columns span that lattice contains the projective point and its chart
231 derivatives span exactly the affine tangent space.  This is the dictionary
232 formerly exposed as the separate `hposition` and `htangent` inputs. -/
233 private theorem matrix_tangent_dictionary
234     {K : Type*} [Field K] {n : ℕ} {κ : Type*} [Fintype κ]
235     (q : Fin (n + 1) → K) (hq0 : q 0 ≠ 0)
236     (T : Submodule K (Fin n → K))
237     (B : Matrix (Fin (n + 1)) κ K)
238     (hspan : Submodule.span K
239       (Set.range fun j => fun i => B i j) = projectiveTangentCone q T) :
240     (∃ c : κ → K, ∀ i, q i = ∑ j, B i j * c j) ∧
241       dehomogenizedTangentSpan q B = T := by
242   have hqcone : q ∈ projectiveTangentCone q T := by
243     change dehomogenizedTangentColumn q q ∈ T
244     convert T.zero_mem using 1
245     ext i
246     simp [dehomogenizedTangentColumn]
247   have hqspan : q ∈ Submodule.span K
248       (Set.range fun j => fun i => B i j) := by
249     rw [hspan]
250     exact hqcone
251   obtain ⟨c, hc⟩ :=
252     (Submodule.mem_span_range_iff_exists_fun K).mp hqspan
253   refine ⟨⟨c, fun i => ?_⟩, ?_⟩
254   · have hi := congrFun hc.symm i
255     simpa [Finset.sum_apply, smul_eq_mul, mul_comm] using hi
256   let d := dehomogenizedTangentLinearMap q
257   calc
258     dehomogenizedTangentSpan q B =
259         (Submodule.span K (Set.range fun j => fun i => B i j)).map d := by
260           rw [dehomogenizedTangentSpan, Submodule.map_span]
261           congr 1
262           ext v
263           simp only [Set.mem_image, Set.mem_range]
264           constructor
265           · rintro ⟨j, rfl⟩
266             exact ⟨fun i => B i j, ⟨j, rfl⟩, rfl⟩
267           · rintro ⟨_, ⟨j, rfl⟩, rfl⟩
268             exact ⟨j, rfl⟩
269     _ = (projectiveTangentCone q T).map d := by rw [hspan]
270     _ = T := by
271       exact Submodule.map_comap_eq_of_surjective
272         (dehomogenizedTangentLinearMap_surjective q hq0) T

529 /-- The affine consequence of the paper's tangent-limit argument.  Starting
530 from an actual rank `dimY + 1` direct-summand lattice whose generic fibre is
531 the projective tangent cone, this theorem constructs split matrix coordinates
532 internally and obtains vanishing on the smooth conormal fibre projection.  No
533 annihilator, matrix retraction, position coefficients, or chart tangent
534 dictionary occurs in the input. -/
535 theorem tangent_limit_affine_fibre_closure_of_directSummand
536     {n dimY : ℕ} [IsAlgClosed k]
537     (I : Ideal (MvPolynomial (Fin n) k))
538     (q : Fin (n + 1) → PowerSeries k)
539     (L : Submodule (PowerSeries k)
540       (Fin (n + 1) → PowerSeries k))
541     (D : DirectSummandInput (dimY := dimY) I q L) :
542     (fun i : Fin n => if i = D.axis then (1 : k) else 0) ∈
543       MvPolynomial.zeroLocus k
544         (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I)) := by
545   obtain ⟨P⟩ := exists_splitMatrixPresentation_of_isComplemented
546     L D.isComplemented (dimY + 1) D.rank_eq
547   let B : Matrix (Fin (n + 1)) (Fin (dimY + 1))
548       (PowerSeries k) := P.B
549   let BL : Matrix (Fin (n + 1)) (Fin (dimY + 1))
550       (LaurentSeries k) :=
551     fun i j => algebraMap (PowerSeries k) (LaurentSeries k) (B i j)
552   have hgenericSpan :
553       Submodule.span (LaurentSeries k)
554           (Set.range fun j => fun i => BL i j) =
555         projectiveTangentCone (laurentColumn q)
556           (zariskiTangentSpace
557             (dehomogenizedPoint (laurentColumn q))
558             (I.map (scalarPolynomialMap
559               (k := k) (K := LaurentSeries k) (Fin n)))) := by
560     rw [← D.hgenericTangentCone,
561       genericFibre_eq_span_matrixColumns L B P.columnsSpan]
562   have hq0L : laurentColumn q 0 ≠ 0 :=
563     laurentColumn_ne_zero_of_ne_zero q D.chart_nonzero
564   obtain ⟨hposition, htangent⟩ := matrix_tangent_dictionary
565     (laurentColumn q) hq0L
566     (zariskiTangentSpace
567       (dehomogenizedPoint (laurentColumn q))
568       (I.map (scalarPolynomialMap
569         (k := k) (K := LaurentSeries k) (Fin n))))
570     BL hgenericSpan
571   have hbase : ∀ f ∈ I.map (scalarPolynomialMap
572       (k := k) (K := LaurentSeries k) (Fin n)),
573       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0 := by
574     have hz := (projectiveClosureAtZero_iff
575       (I.map (scalarPolynomialMap
576         (k := k) (K := LaurentSeries k) (Fin n)))
577       (laurentColumn q) hq0L).mp D.arc_mem_closure.2
578     exact hz
579   have haxis : ∀ j,
580       PowerSeries.constantCoeff (B D.axis.succ j) = 0 := by
581     intro j
582     have hmem : (fun i => B i j) ∈ L := by
583       rw [← P.columnsSpan]
584       exact Submodule.subset_span ⟨j, rfl⟩
585     let v : L := ⟨fun i => B i j, hmem⟩
586     simpa [v] using D.residue_le_coordinateHyperplane v
587   let low : Input (dimY := dimY) I q B := {
588     axis := D.axis
589     C := P.C
590     hprime := D.hprime
591     hsplit := P.leftInverse
592     hrank := by simp
593     hq0 := D.chart_nonzero
594     hbase := hbase
595     hsmooth := D.generic_smooth
596     hposition := hposition
597     htangent := htangent
598     haxis := haxis }
599   obtain ⟨ell, hrow, hres, hconormal⟩ :=
600     exists_axis_laurent_smooth_conormal_direction_with_fibre_closure I q B low
601   exact hconormal.2.2.1
