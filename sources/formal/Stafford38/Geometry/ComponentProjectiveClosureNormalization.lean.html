9 /-!
10 # Projective-component equations under retained normalization
11 
12 A homogeneous equation of the generic projective cone continues to vanish
13 after a common rescaling of all projective coordinates into a subring of the
14 component function field.  The proof first checks the equality in the function
15 field and then reflects it through the injective subring map.  Consequently the
16 equality is preserved by every subsequent ring homomorphism, including the
17 retained completion and its Laurent-series embedding.
18 
19 This is only a homogeneous projective statement.  It does not assert that an
20 arbitrary affine chart point satisfies the original affine component ideal.
21 -/

44 /-- A homogeneous projective-cone equation vanishes after any common-scale
45 lift through an injective coefficient ring.  The explicit coefficient
46 compatibility prevents an implicit change of the ground-field embedding. -/
47 theorem eval₂_eq_zero_of_commonScale_componentProjectivePoint
48     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
49     {V : Type u} [CommRing V]
50     (coeff : k →+* V) (ι : V →+* ComponentFractionField P)
51     (hι : Function.Injective ι)
52     (hcoeff : ι.comp coeff = algebraMap k (ComponentFractionField P))
53     {H : MvPolynomial (Fin (m + 1)) k} {d : ℕ}
54     (hhomogeneous : H.IsHomogeneous d)
55     (hH : H ∈ componentProjectiveClosureIdeal P)
56     (q : Fin (m + 1) → V) (scale : ComponentFractionField P)
57     (hq : ∀ a, ι (q a) = scale * componentProjectivePoint P a) :
58     MvPolynomial.eval₂ coeff q H = 0 := by
59   have hgeneric :
60       MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
61         (componentProjectivePoint P) H = 0 := by
62     have hcone : componentProjectiveConeMap P H = 0 := hH
63     have hformula :=
64       componentProjectiveConeMap_eq_eval₂_mul_X_pow_of_isHomogeneous
65         P hhomogeneous
66     rw [hcone] at hformula
67     have hone := congrArg (Polynomial.eval 1) hformula
68     have hpt : componentProjectivePoint P =
69         Fin.cases 1 fun i ↦ componentCoordinate P i := by
70       simp [componentProjectivePoint]
71     simpa [hpt] using hone.symm
72   apply hι
73   rw [map_zero]
74   calc
75     ι (MvPolynomial.eval₂ coeff q H) =
76         MvPolynomial.eval₂ (ι.comp coeff) (fun a ↦ ι (q a)) H :=
77       MvPolynomial.map_eval₂Hom coeff q ι H
78     _ = MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
79         (fun a ↦ componentProjectivePoint P a * scale) H := by
80       apply MvPolynomial.eval₂Hom_congr hcoeff
81       · funext a
82         rw [hq a, mul_comm]
83       · rfl
84     _ = MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
85           (componentProjectivePoint P) H * scale ^ d :=
86       eval₂_mul_common_of_isHomogeneous hhomogeneous
87         (algebraMap k (ComponentFractionField P))
88         (componentProjectivePoint P) scale
89     _ = 0 := by rw [hgeneric, zero_mul]

91 /-- Evaluation is natural under every ring homomorphism out of a common-scale
92 lift: the target coefficient map and every target coordinate are the displayed
93 composites, and the resulting target-ring evaluation is zero. -/
94 theorem map_eval₂_eq_zero_of_commonScale_componentProjectivePoint
95     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
96     {V S : Type u} [CommRing V] [CommRing S]
97     (coeff : k →+* V) (ι : V →+* ComponentFractionField P)
98     (hι : Function.Injective ι)
99     (hcoeff : ι.comp coeff = algebraMap k (ComponentFractionField P))
100     (ψ : V →+* S)
101     {H : MvPolynomial (Fin (m + 1)) k} {d : ℕ}
102     (hhomogeneous : H.IsHomogeneous d)
103     (hH : H ∈ componentProjectiveClosureIdeal P)
104     (q : Fin (m + 1) → V) (scale : ComponentFractionField P)
105     (hq : ∀ a, ι (q a) = scale * componentProjectivePoint P a) :
106     MvPolynomial.eval₂ (ψ.comp coeff) (fun a ↦ ψ (q a)) H = 0 := by
107   calc
108     MvPolynomial.eval₂ (ψ.comp coeff) (fun a ↦ ψ (q a)) H =
109         ψ (MvPolynomial.eval₂ coeff q H) := by
110       simpa only [Function.comp_apply, Function.comp_def] using
111         (MvPolynomial.eval₂_comp_left ψ coeff q H).symm
112     _ = 0 := by
113       rw [eval₂_eq_zero_of_commonScale_componentProjectivePoint P coeff ι hι
114         hcoeff hhomogeneous hH q scale hq, map_zero]

129 /-- Every homogeneous equation of the projective component vanishes on a
130 common-scale lift in the retained valuation subring. -/
131 theorem retained_eval₂_eq_zero_of_commonScale
132     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
133     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
134     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
135         (ComponentFractionField P) := W.ambientAlgebra
136     ∀ {H : MvPolynomial (Fin (m + 1)) k} {d : ℕ},
137       H.IsHomogeneous d → H ∈ componentProjectiveClosureIdeal P →
138       ∀ (q : Fin (m + 1) → W.place.valuation.toSubring)
139         (scale : ComponentFractionField P),
140         (∀ a, (q a : ComponentFractionField P) =
141           scale * componentProjectivePoint P a) →
142         MvPolynomial.eval₂ (retainedComponentCoefficientMap P i W) q H = 0 := by
143   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
144       (ComponentFractionField P) := W.ambientAlgebra
145   letI : IsScalarTower W.coefficientField
146       (CoordinateZeroLocalRing W.coefficientField)
147       (ComponentFractionField P) := W.coefficientTower
148   intro H d hhomogeneous hH q scale hq
149   let V := W.place.valuation.toSubring
150   let coeff : k →+* V := retainedComponentCoefficientMap P i W
151   have hcoeff : W.place.valuation.toSubring.subtype.comp coeff =
152       algebraMap k (ComponentFractionField P) := by
153     ext c
154     change ((relativeCoefficientMap W.coefficientField W.place
155       (algebraMap k W.coefficientField c) : V) : ComponentFractionField P) =
156         algebraMap k (ComponentFractionField P) c
157     calc
158       ((relativeCoefficientMap W.coefficientField W.place
159           (algebraMap k W.coefficientField c) : V) :
160           ComponentFractionField P) =
161           algebraMap W.coefficientField (ComponentFractionField P)
162             (algebraMap k W.coefficientField c) :=
163         DFunLike.congr_fun
164           (relativeCoefficientMap_commutes W.coefficientField W.place)
165           (algebraMap k W.coefficientField c)
166       _ = algebraMap k (ComponentFractionField P) c :=
167         IsScalarTower.algebraMap_apply k W.coefficientField
168           (ComponentFractionField P) c
169   exact eval₂_eq_zero_of_commonScale_componentProjectivePoint
170     P coeff W.place.valuation.toSubring.subtype
171       Subtype.val_injective hcoeff hhomogeneous hH q scale hq

173 /-- The explicit ground-field coefficient map in the completed power-series
174 chart.  No ambient `Algebra k` instance is selected implicitly. -/
175 def retainedCompletedCoefficientMap
176     [CharZero k]
177     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
178     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
179     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
180         (ComponentFractionField P) := W.ambientAlgebra
181     let V := W.place.valuation.toSubring
182     letI : IsDiscreteValuationRing V := W.place.isDiscrete
183     letI : Algebra W.coefficientField V :=
184       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
185     k →+* PowerSeries (ResidueField V) := by
186   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
187       (ComponentFractionField P) := W.ambientAlgebra
188   let V := W.place.valuation.toSubring
189   letI : IsDiscreteValuationRing V := W.place.isDiscrete
190   letI : Algebra W.coefficientField V :=
191     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
192   exact (retainedToCompletedPowerSeries W).comp
193     (retainedComponentCoefficientMap P i W)

196 /-- The same equation vanishes when evaluated at the transported completed
197 coordinates with the explicit transported coefficient map. -/
198 theorem retainedCompleted_eval₂_eq_zero_of_commonScale
199     [CharZero k]
200     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
201     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
202     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
203         (ComponentFractionField P) := W.ambientAlgebra
204     let V := W.place.valuation.toSubring
205     letI : IsDiscreteValuationRing V := W.place.isDiscrete
206     letI : Algebra W.coefficientField V :=
207       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
208     ∀ {H : MvPolynomial (Fin (m + 1)) k} {d : ℕ},
209       H.IsHomogeneous d → H ∈ componentProjectiveClosureIdeal P →
210       ∀ (q : Fin (m + 1) → V) (scale : ComponentFractionField P),
211         (∀ a, (q a : ComponentFractionField P) =
212           scale * componentProjectivePoint P a) →
213         MvPolynomial.eval₂ (retainedCompletedCoefficientMap P i W)
214           (fun a ↦ retainedToCompletedPowerSeries W (q a)) H = 0 := by
215   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
216       (ComponentFractionField P) := W.ambientAlgebra
217   let V := W.place.valuation.toSubring
218   letI : IsDiscreteValuationRing V := W.place.isDiscrete
219   letI : Algebra W.coefficientField V :=
220     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
221   dsimp only
222   intro H d hhomogeneous hH q scale hq
223   change MvPolynomial.eval₂
224       ((retainedToCompletedPowerSeries W).comp
225         (retainedComponentCoefficientMap P i W))
226       (fun a ↦ retainedToCompletedPowerSeries W (q a)) H = 0
227   calc
228     _ = retainedToCompletedPowerSeries W
229         (MvPolynomial.eval₂ (retainedComponentCoefficientMap P i W) q H) := by
230       simpa only [Function.comp_apply, Function.comp_def] using
231         (MvPolynomial.eval₂_comp_left (retainedToCompletedPowerSeries W)
232           (retainedComponentCoefficientMap P i W) q H).symm
233     _ = 0 := by
234       rw [retained_eval₂_eq_zero_of_commonScale P i W
235         hhomogeneous hH q scale hq, map_zero]

260 /-- The completed equation remains zero when evaluated at the corresponding
261 Laurent coordinates. -/
262 theorem retainedLaurent_eval₂_eq_zero_of_commonScale
263     [CharZero k]
264     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
265     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
266     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
267         (ComponentFractionField P) := W.ambientAlgebra
268     let V := W.place.valuation.toSubring
269     letI : IsDiscreteValuationRing V := W.place.isDiscrete
270     letI : Algebra W.coefficientField V :=
271       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
272     ∀ {H : MvPolynomial (Fin (m + 1)) k} {d : ℕ},
273       H.IsHomogeneous d → H ∈ componentProjectiveClosureIdeal P →
274       ∀ (q : Fin (m + 1) → V) (scale : ComponentFractionField P),
275         (∀ a, (q a : ComponentFractionField P) =
276           scale * componentProjectivePoint P a) →
277         MvPolynomial.eval₂ (retainedLaurentCoefficientMap P i W)
278           (fun a ↦ algebraMap (PowerSeries (ResidueField V))
279             (LaurentSeries (ResidueField V))
280               (retainedToCompletedPowerSeries W (q a))) H = 0 := by
281   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
282       (ComponentFractionField P) := W.ambientAlgebra
283   let V := W.place.valuation.toSubring
284   letI : IsDiscreteValuationRing V := W.place.isDiscrete

286     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
287   dsimp only
288   intro H d hhomogeneous hH q scale hq
289   let ψ : PowerSeries (ResidueField V) →+* LaurentSeries (ResidueField V) :=
290     algebraMap (PowerSeries (ResidueField V)) (LaurentSeries (ResidueField V))
291   change MvPolynomial.eval₂
292       (ψ.comp (retainedCompletedCoefficientMap P i W))
293       (fun a ↦ ψ (retainedToCompletedPowerSeries W (q a))) H = 0
294   calc
295     _ = ψ (MvPolynomial.eval₂ (retainedCompletedCoefficientMap P i W)
296         (fun a ↦ retainedToCompletedPowerSeries W (q a)) H) := by
297       simpa only [Function.comp_apply, Function.comp_def] using
298         (MvPolynomial.eval₂_comp_left ψ

303         hhomogeneous hH q scale hq, map_zero]
