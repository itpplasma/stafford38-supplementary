11 /-!
12 # Laurent conormal specialization over a residue-field extension
13 
14 The residue field of a divisorial valuation need not be the original ground
15 field.  This file separates the two coefficient roles which are easy to
16 confuse in that situation:
17 
18 * `k` is the field of coefficients of the original equations and symbol;
19 * `K` is the residue field of the boundary valuation, and the completed
20   coordinates live in `PowerSeries K` and `LaurentSeries K`.
21 
22 Ground coefficients use the canonical map `k -> LaurentSeries K`; the theorem
23 `groundLaurentComposite_eq_algebraMap` proves that it agrees with the tower
24 composite `k -> K -> LaurentSeries K`.  The fibre closure itself is taken over `K`.
25 The resulting contradiction uses only field and algebra-tower structure; in
26 particular it does not require `K` to be algebraically closed and does not
27 choose a map `K -> k`.
28 
29 The final structure and consumer are a residue-extension version of the
30 completed-boundary local interface.  They construct no projective chart,
31 normalization, divisor, or tangent comparison.
32 -/

59 /-! ## The coefficient tower -/
60 
61 /-- The coefficient map used by equations over the ground field after
62 completion at a boundary with residue field `K`. -/
63 abbrev groundLaurentMap : k →+* LaurentSeries K :=
64   algebraMap k (LaurentSeries K)
65 
66 /-- The coefficient tower composite agrees with the canonical structure map.
67 This bridge is needed because the tower instance is not available as a typeclass
68 instance for arbitrary `k → K → LaurentSeries K`. -/
69 theorem groundLaurentComposite_eq_algebraMap :
70     (algebraMap K (LaurentSeries K)).comp (algebraMap k K) =
71       groundLaurentMap (k := k) (K := K) := by
72   refine RingHom.ext fun a => ?_
73   simp only [RingHom.comp_apply, HahnSeries.algebraMap_apply',
74     PowerSeries.algebraMap_apply, Algebra.algebraMap_self_apply]
75 
76 /-- Fibre lift after extending the ground coefficients to `K`. -/
77 def residueFibreLift (P : MvPolynomial (Fin n) k) :
78     MvPolynomial (PhaseVar n) K :=
79   fibreLift (scalarPolynomialMap (k := k) (K := K) (Fin n) P)
80 
81 /-- Extending first to `K` or using the ground coefficient map gives the same
82 Laurent-series evaluation. -/
83 theorem eval_residueFibreLift_eq_ground
84     (P : MvPolynomial (Fin n) k)
85     (q : PhaseVar n → LaurentSeries K) :
86     MvPolynomial.eval₂ (algebraMap K (LaurentSeries K)) q
87         (residueFibreLift (k := k) (K := K) P) =
88       MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
89         (fibreLift P) := by
90   change MvPolynomial.eval₂ (algebraMap K (LaurentSeries K)) q
91       (MvPolynomial.rename Sum.inr
92         (MvPolynomial.map (algebraMap k K) P)) = _
93   rw [← MvPolynomial.map_rename]
94   rw [MvPolynomial.eval₂_map]
95   rw [← groundLaurentComposite_eq_algebraMap (k := k) (K := K)]
96   rfl
97 
98 /-- Evaluation of a ground polynomial agrees with evaluation of its
99 coefficient extension along the residue-field-to-Laurent map. -/
100 theorem eval_residuePolynomialMap_eq_ground
101     (P : MvPolynomial (Fin n) k)
102     (v : Fin n → LaurentSeries K) :
103     MvPolynomial.eval₂ (algebraMap K (LaurentSeries K)) v
104         (scalarPolynomialMap (k := k) (K := K) (Fin n) P) =
105     MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) v P := by
106   change MvPolynomial.eval₂ (algebraMap K (LaurentSeries K)) v
107       (MvPolynomial.map (algebraMap k K) P) = _
108   rw [MvPolynomial.eval₂_map]
109   rw [← groundLaurentComposite_eq_algebraMap (k := k) (K := K)]
110 
111 /-- Fibre-only evaluation ignores the base part of a phase point, also for
112 the explicit ground-to-Laurent coefficient map. -/
113 theorem eval_ground_fibreLift
114     (P : MvPolynomial (Fin n) k)
115     (q : PhaseVar n → LaurentSeries K) :
116     MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
117         (fibreLift P) =
118       MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K))
119         (fun i ↦ q (Sum.inr i)) P := by
120   rw [fibreLift, MvPolynomial.eval₂_rename]
121   rfl

123 /-! ## Fibre closure over the residue field -/
124 
125 /-- The projected closure of Laurent-valued phase points, with regular fibre
126 residue, formed over `K`.  Its base coordinates may have poles. -/
127 abbrev residueExtensionFibreClosure
128     (S : Set (PhaseVar n → LaurentSeries K)) : Set (Fin n → K) :=
129   extensionFibreClosure (k := K) (K := LaurentSeries K) S
130 
131 /-- The existing Laurent-residue theorem, instantiated over the residue field
132 `K`.  No algebraic-closedness hypothesis occurs. -/
133 theorem residue_mem_residueExtensionFibreClosure_of_laurent_generic
134     (S : Set (PhaseVar n → LaurentSeries K))
135     (y : Fin n → LaurentSeries K)
136     (xi : Fin n → PowerSeries K)
137     (hgeneric :
138       Sum.elim y
139           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈ S) :
140     residueColumn xi ∈ residueExtensionFibreClosure (K := K) S := by
141   exact residue_mem_extensionFibreClosure_of_laurent_generic
142     (k := K) S y xi hgeneric

144 /-! ## Ground-field contradiction -/
145 
146 /-- The scalar-extended equation-conormal locus associated with a ground
147 ideal, using the canonical `k -> LaurentSeries K` coefficient map. -/
148 abbrev groundEquationConormalLocus
149     (I : Ideal (MvPolynomial (Fin n) k)) :
150     Set (PhaseVar n → LaurentSeries K) :=
151   equationConormalLocus
152     (I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin n)))
153 
154 /-- A fibre-only ground polynomial which vanishes on the whole completed
155 equation-conormal locus also vanishes after extending its coefficients to
156 `K` and evaluating at every regular fibre residue. -/
157 theorem residueFibreLift_mem_extensionValuedVanishingIdeal_of_ground_vanishing
158     (I : Ideal (MvPolynomial (Fin n) k))
159     (P : MvPolynomial (Fin n) k)
160     (hvanishes :
161       ∀ q ∈ groundEquationConormalLocus (k := k) (K := K) I,
162         MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
163           (fibreLift P) = 0) :
164     scalarPolynomialMap (k := k) (K := K) (Fin n) P ∈
165       extensionValuedVanishingIdeal (k := K) (K := LaurentSeries K)
166         (fibreImage (groundEquationConormalLocus (k := k) (K := K) I)) := by
167   rw [mem_extensionValuedVanishingIdeal_iff]
168   intro q hq
169   rcases hq with ⟨z, hz, rfl⟩
170   rw [eval_residuePolynomialMap_eq_ground]
171   rw [← eval_ground_fibreLift]
172   exact hvanishes z hz
173 
174 /-- Exact contradiction consumer for a residue-field-valued boundary arc.
175 
176 The hypotheses deliberately use the composite coefficient map and an
177 `eval₂ (algebraMap k K)` axis value.  The theorem requires neither
178 `IsAlgClosed K` nor a retraction from `K` to `k`. -/
179 theorem false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
180     (I : Ideal (MvPolynomial (Fin n) k))
181     (P : MvPolynomial (Fin n) k)
182     (axis : Fin n → K)
183     (y : Fin n → LaurentSeries K)
184     (xi : Fin n → PowerSeries K)
185     (hgeneric :
186       Sum.elim y
187           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
188         groundEquationConormalLocus (k := k) (K := K) I)
189     (hresidue : residueColumn xi = axis)
190     (hvanishes :
191       ∀ q ∈ groundEquationConormalLocus (k := k) (K := K) I,
192         MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
193           (fibreLift P) = 0)
194     (haxis : MvPolynomial.eval₂ (algebraMap k K) axis P = 1) : False := by
195   let S : Set (PhaseVar n → LaurentSeries K) :=
196     groundEquationConormalLocus (k := k) (K := K) I
197   have hP :
198       scalarPolynomialMap (k := k) (K := K) (Fin n) P ∈
199         extensionValuedVanishingIdeal (k := K) (K := LaurentSeries K)
200           (fibreImage S) := by
201     exact residueFibreLift_mem_extensionValuedVanishingIdeal_of_ground_vanishing
202       (k := k) (K := K) I P (by
203         simpa [S] using hvanishes)
204   have hclosure :
205       residueColumn xi ∈ residueExtensionFibreClosure (K := K) S :=
206     residue_mem_residueExtensionFibreClosure_of_laurent_generic
207       (K := K) S y xi (by simpa [S] using hgeneric)
208   have hzero :
209       MvPolynomial.eval (residueColumn xi)
210         (scalarPolynomialMap (k := k) (K := K) (Fin n) P) = 0 :=
211     hclosure _ hP
212   have hzero' :
213       MvPolynomial.eval₂ (algebraMap k K) (residueColumn xi) P = 0 := by
214     simpa [scalarPolynomialMap] using hzero
215   rw [hresidue, haxis] at hzero'
216   exact one_ne_zero hzero'

218 /-! ## Completed-boundary adapter over `K` -/
219 
220 /-- A completed projective boundary chart whose residue coefficients are in
221 `K`, while the target affine ideal remains defined over `k`.  Its formal
222 tangent data are one fixed derivative-compatible geometric witness.  The
223 one-sided tangent inclusion uses that same stored transverse column, never an
224 alternative split column. -/
225 structure CompletedProjectiveBoundaryChartOver
226     (m : ℕ) (hm : 0 < m)
227     (I : Ideal (MvPolynomial (Fin m) k)) where
228   equationCount : ℕ
229   tangentCount : ℕ
230   equations : Fin equationCount →
231     MvPolynomial (Fin (m + 1)) (LaurentSeries K)
232   degree : Fin equationCount → ℕ
233   homogeneous : ∀ j, (equations j).IsHomogeneous (degree j)
234   q : Fin (m + 1) → PowerSeries K
235   Z : Matrix (Fin (m + 1)) (Fin tangentCount) (PowerSeries K)
236   rows : Fin tangentCount ↪ Fin (m + 1)
237   chart : Fin (m + 1)
238   zero : Fin (m + 1)
239   axis : Fin (m + 1)
240   ratio : PowerSeries K
241   q_chart : q chart = 1
242   Z_chart : ∀ j, Z chart j = 0
243   q_zero_ne : q zero ≠ 0
244   q_origin_ne : q 0 ≠ 0
245   ratio_ne : ratio ≠ 0
246   q_zero_vanish : PowerSeries.constantCoeff (q zero) = 0
247   ratio_vanish : PowerSeries.constantCoeff ratio = 0
248   q_axis : q axis = q zero * ratio
249   Z_zero_dvd : ∀ j, q zero ∣ Z zero j
250   Z_axis_dvd : ∀ j, q axis ∣ Z axis j
251   selected_minor_nonzero :
252     PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0
253   equations_vanish :
254     ∀ j, MvPolynomial.eval (laurentColumn q) (equations j) = 0
255   ideal_containment :
256     I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m)) ≤
257       dehomogenizedEquationIdeal equations
258   axis_is_first_fibre : axis = Fin.succ ⟨0, hm⟩
259   tau : Fin (m + 1) → PowerSeries K
260   C : Matrix (FormalTangentColumn (Fin tangentCount))
261     (Fin (m + 1)) (PowerSeries K)
262   ell : Fin (m + 1) → PowerSeries K
263   left_inverse : C * formalTangentMatrix q Z tau = 1
264   annihilation : rowMul ell (formalTangentMatrix q Z tau) = 0
265   residue_axis : residueColumn ell = axisRow (k := K) axis
266   tangent_inclusion :
267     zariskiTangentSpace (dehomogenizedPoint (laurentColumn q))
268         (I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m))) ≤
269       dehomogenizedTangentSpan (laurentColumn q)
270         (laurentNonpositionTangentMatrix Z tau)
271 
272 /-- The residue-extension completed chart supplies a Laurent equation-
273 conormal point with pure first fibre residue. -/
274 theorem exists_conormalAxis_of_completedProjectiveBoundaryChartOver
275     [CharZero K]
276     {m : ℕ} (hm : 0 < m)
277     (I : Ideal (MvPolynomial (Fin m) k))
278     (W : CompletedProjectiveBoundaryChartOver (k := k) (K := K) m hm I) :
279     ∃ (y : Fin m → LaurentSeries K)
280       (xi : Fin m → PowerSeries K),
281       Sum.elim y
282           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
283         groundEquationConormalLocus (k := k) (K := K) I ∧
284       residueColumn xi =
285         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
286   classical
287   let Iext := I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m))
288   have hbase : ∀ f ∈ Iext,
289       MvPolynomial.eval (dehomogenizedPoint (laurentColumn W.q)) f = 0 := by
290     intro f hf
291     exact eval_eq_zero_of_mem_dehomogenizedEquationIdeal
292       W.equations W.degree W.homogeneous (laurentColumn W.q)
293       (laurentColumn_ne_zero_of_ne_zero W.q W.q_origin_ne)
294       W.equations_vanish f (W.ideal_containment hf)
295   have htangent :
296       zariskiTangentSpace (dehomogenizedPoint (laurentColumn W.q)) Iext ≤
297         dehomogenizedTangentSpan (laurentColumn W.q)
298           (laurentNonpositionTangentMatrix W.Z W.tau) := by
299     exact W.tangent_inclusion
300   have hphase :=
301     laurentPhasePoint_mem_equationConormalLocus_of_zariski_le_span
302       Iext W.q W.ell W.Z W.tau W.q_origin_ne W.annihilation hbase htangent
303   refine ⟨dehomogenizedPoint (laurentColumn W.q),
304     (fun i : Fin m ↦ W.ell i.succ), ?_, ?_⟩
305   · simpa [Iext, groundEquationConormalLocus, laurentColumn] using hphase
306   · calc
307       residueColumn (fun i : Fin m ↦ W.ell i.succ) =
308           (fun i : Fin m ↦ residueColumn W.ell i.succ) :=
309         residueColumn_tail W.ell
310       _ = (fun i : Fin m ↦ axisRow (k := K) W.axis i.succ) := by
311         rw [W.residue_axis]
312       _ = (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
313         funext i
314         rw [W.axis_is_first_fibre]
315         by_cases hi : i = ⟨0, hm⟩
316         · subst i
317           simp [axisRow]
318         · have hne : i.succ ≠ Fin.succ ⟨0, hm⟩ := by
319             intro h
320             exact hi (Fin.succ_injective m h)
321           change (if i.succ = Fin.succ ⟨0, hm⟩ then 1 else 0) =
322             (if i = ⟨0, hm⟩ then 1 else 0)
323           rw [if_neg hne, if_neg hi]
