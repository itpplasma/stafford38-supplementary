7 /-!
8 # Equality of a centered DVR with a dominating valuation ring
9 
10 This is the final local-algebra step in identifying a divisorial valuation
11 with the local ring at its center. It does not construct a proper model, a
12 center, or prove that the center has codimension one.
13 -/

27 /-- A DVR subring of a field equals any valuation subring of that field which
28 locally dominates it. The domination hypothesis is precisely the algebraic
29 condition obtained after the center is known to be height one. -/
30 theorem dvr_eq_of_dominated_by_valuationSubring
31     [IsDiscreteValuationRing R] [IsFractionRing R K] (V : ValuationSubring K)
32     (hdom : (LocalSubring.mk R : LocalSubring K) ≤ V.toLocalSubring) :
33     R = V.toSubring := by
34   let vR : ValuationSubring K :=
35     ((IsDiscreteValuationRing.maximalIdeal R).valuation K).valuationSubring
36   have hcarrier : vR.toSubring = R := by
37     change ((IsDiscreteValuationRing.maximalIdeal R).valuation K).valuationSubring.toSubring = R
38     rw [← IsDiscreteValuationRing.map_algebraMap_eq_valuationSubring]
39     ext x
40     simp only [Subring.mem_map]
41     constructor
42     · rintro ⟨a, ha, hax⟩
43       change (a : K) = x at hax
44       exact hax ▸ a.property
45     · intro hx
46       refine ⟨⟨x, hx⟩, Set.mem_univ _, ?_⟩
47       rfl
48   have hlocal : vR.toLocalSubring = (LocalSubring.mk R : LocalSubring K) := by
49     apply LocalSubring.toSubring_injective
50     exact hcarrier
51   have hmax : IsMax (LocalSubring.mk R : LocalSubring K) := by
52     rw [← hlocal]
53     exact vR.isMax_toLocalSubring
54   have heq := hmax.eq_of_le hdom
55   exact congrArg LocalSubring.toSubring heq
56 
57 #print axioms dvr_eq_of_dominated_by_valuationSubring
58 
59 end
