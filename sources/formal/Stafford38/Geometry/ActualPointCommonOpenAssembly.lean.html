37 /-- The prescribed local completion arc sends every element outside the
38 chosen maximal ideal to a unit. This is the localization property needed to
39 extend the same arc to the generic common open. -/
40 theorem pointLocalArc_unit_of_not_mem
41     {k B : Type u} [Field k] {d : ℕ}
42     [CommRing B] [Algebra k B]
43     [Algebra (MvPolynomial (Option (Fin d)) k) B]
44     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) B]
45     [Algebra.EssFiniteType (MvPolynomial (Option (Fin d)) k) B]
46     (M : Ideal B) [M.IsMaximal]
47     (eM : (B ⧸ M) ≃ₐ[k] k) (α : Fin d → k)
48     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k)
49       (Localization.AtPrime M)]
50     (b : B) (hb : b ∉ M) :
51     IsUnit (originalToPointLocalFinSuccArcLaurentSeries
52       (k := k) (d := d) (A := B) M eM α b) := by
53   let T := Localization.AtPrime M
54   have hu : IsUnit (algebraMap B T b) :=
55     IsLocalization.map_units T (M := M.primeCompl) ⟨b, hb⟩
56   exact IsUnit.map
57     (pointLocalFinSuccArcLaurentSeries
58       (k := k) (d := d) (A := B) M eM α).toRingHom hu

63 /-- Turn one already-selected prescribed-point tilt and the three genuine
64 common-open column identities into the paper-facing direct-summand input.
65 The chart relation is the checked affine-overlap coordinate identity; closure
66 and smoothness are the two outputs of the same homogenized smooth-open
67 numerator argument. This wrapper does not choose a second tilt or assume a
68 tangent-cone identity. -/
69 theorem exists_directSummandInput_of_selected_axis_lift
70     {k E : Type u} [Field k] [CommRing E] {n d : ℕ}
71     {I : Ideal (MvPolynomial (Fin n) k)}
72     [Algebra k E]
73     [Algebra (MvPolynomial (Fin n) k ⧸ I) E]
74     [Algebra (MvPolynomial (Fin n) k) E]
75     [Algebra (MvPolynomial (Option (Fin d)) k) E]
76     [IsScalarTower k (MvPolynomial (Fin n) k ⧸ I) E]
77     [IsScalarTower k (MvPolynomial (Fin n) k) E]
78     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) E]
79     [IsScalarTower (MvPolynomial (Fin n) k)
80       (MvPolynomial (Fin n) k ⧸ I) E]
81     [Algebra.FormallyEtale (MvPolynomial (Fin n) k ⧸ I) E]
82     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) E]
83     (rho : E →+* LaurentSeries k)
84     (hground : rho.comp (algebraMap k E) = algebraMap k (LaurentSeries k)) :
85     letI : Algebra E (LaurentSeries k) :=
86       RingHom.toAlgebra' rho (by intro x y; exact mul_comm _ _)
87     letI : Algebra k (LaurentSeries k) :=
88       RingHom.toAlgebra' (algebraMap k (LaurentSeries k))
89         (by intro x y; exact mul_comm _ _)
90     letI : Module k (LaurentSeries k) := Algebra.toModule
91     letI : SMul k (LaurentSeries k) :=
92       (Algebra.toModule : Module k (LaurentSeries k)).toSMul
93     letI : IsScalarTower k E (LaurentSeries k) :=
94       IsScalarTower.of_algebraMap_eq' hground.symm
95     ∀ (qC : Fin (n + 1) → E)
96       (qPre : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)
97       (rows : Fin d ↪ Fin (n + 1)) (chart : Fin (n + 1)) (axis : Fin n)
98       (a b : ℕ)
99       (beta alpha : Fin d → k)
100       (u₀ u₁ : MvPowerSeries (Fin (d + 1)) k)
101       (hqchartPre : qPre chart = 1)
102       (hdata : SelectedCoordinateAxisLiftData qPre rows chart 0 axis.succ a b
103         beta alpha u₀ u₁)
104       (hposition : ∀ i, rho (qC i) =
105         algebraMap (PowerSeries k) (LaurentSeries k)
106           (tiltedArc (k := k) alpha (qPre i)))
107       (htransverse : ∀ j i,
108         coordinateDerivation (k := k) (σ := Option (Fin d)) (B := E)
109           (L := LaurentSeries k) (some j) (qC i) =
110           algebraMap (PowerSeries k) (LaurentSeries k)
111             (tiltedTransverseDerivativeMatrix (k := k) alpha qPre i j))
112       (hraw : ∀ i, algebraMap (PowerSeries k) (LaurentSeries k)
113         (PowerSeries.derivative (R := k) (tiltedArc (k := k) alpha (qPre i))) =
114           coordinateDerivation (k := k) (σ := Option (Fin d)) (B := E)
115             (L := LaurentSeries k) none (qC i) +
116           ∑ j : Fin d,
117             algebraMap (PowerSeries k) (LaurentSeries k) (PowerSeries.C (alpha j)) *
118               coordinateDerivation (k := k) (σ := Option (Fin d)) (B := E)
119                 (L := LaurentSeries k) (some j) (qC i))
120       (hchart : ∀ i, qC i.succ = qC 0 *
121         algebraMap (MvPolynomial (Fin n) k ⧸ I) E
122           (Ideal.Quotient.mk I (MvPolynomial.X i)))
123       (hprime : I.IsPrime)
124       (hclosure : dehomogenizedPoint (laurentColumn
125         (fun i => tiltedArc (k := k) alpha (qPre i))) ∈
126           MvPolynomial.zeroLocus (LaurentSeries k)
127             (I.map (scalarPolynomialMap (k := k)
128               (K := LaurentSeries k) (Fin n))))
129       (hsmooth : SmoothAffinePoint
130         (I.map (scalarPolynomialMap (k := k)
131           (K := LaurentSeries k) (Fin n)))
132         (dehomogenizedPoint (laurentColumn
133           (fun i => tiltedArc (k := k) alpha (qPre i))))) ,
134       ∃ (tau : Fin (n + 1) → PowerSeries k)
135         (C : Matrix (FormalTangentColumn (Fin d)) (Fin (n + 1))
136           (PowerSeries k)),
137         C * formalTangentMatrix
138             (fun i => tiltedArc (k := k) alpha (qPre i))
139             (tiltedTransverseDerivativeMatrix (k := k) alpha qPre) tau = 1 ∧
140         (∀ j, PowerSeries.constantCoeff
141           (formalTangentMatrix
142             (fun i => tiltedArc (k := k) alpha (qPre i))
143             (tiltedTransverseDerivativeMatrix (k := k) alpha qPre)
144             tau axis.succ j) = 0) ∧
145         ∃ D : DirectSummandInput (dimY := d + 1) I
146           (fun i => tiltedArc (k := k) alpha (qPre i))
147           (normalizedTangentLattice
148             (fun i => tiltedArc (k := k) alpha (qPre i))
149             (tiltedTransverseDerivativeMatrix (k := k) alpha qPre) tau),
150           D.axis = axis := by
151   letI : Algebra E (LaurentSeries k) :=
152     RingHom.toAlgebra' rho (by intro x y; exact mul_comm _ _)
153   letI : Algebra k (LaurentSeries k) :=
154     RingHom.toAlgebra' (algebraMap k (LaurentSeries k))
155       (by intro x y; exact mul_comm _ _)
156   letI : Module k (LaurentSeries k) := Algebra.toModule
157   letI : SMul k (LaurentSeries k) :=
158     (Algebra.toModule : Module k (LaurentSeries k)).toSMul
159   letI : IsScalarTower k E (LaurentSeries k) :=
160     IsScalarTower.of_algebraMap_eq' hground.symm
161   intro qC qPre rows chart axis a b beta alpha u₀ u₁ hqchartPre hdata
162     hposition htransverse hraw hchart hprime hclosure hsmooth
163   rcases hdata with ⟨hq0factor, hu₀, hqaxis, hu₁, hrowsArc, hminorRows,
164     hdet, c, tau, C, ell, hc, hselected, htauChart, htauRows,
165     hfactor, hprimitive, haxisZero, haxisColumns, hCB, hEll, hEllRes⟩
166   let qArc : Fin (n + 1) → PowerSeries k :=
167     fun i => tiltedArc (k := k) alpha (qPre i)
168   let Z : Matrix (Fin (n + 1)) (Fin d) (PowerSeries k) :=
169     tiltedTransverseDerivativeMatrix (k := k) alpha qPre
170   have hqchart : qArc chart = 1 := by
171     change tiltedArc (k := k) alpha (qPre chart) = 1
172     rw [hqchartPre]
173     exact map_one (tiltedArcMap (k := k) alpha)
174   have hq0nonzero : qArc 0 ≠ 0 := by
175     change tiltedArc (k := k) alpha (qPre 0) ≠ 0
176     rw [hq0factor]
177     have hu : tiltedArc (k := k) alpha u₀ ≠ 0 := by
178       intro hz
179       have hc0 := congrArg (PowerSeries.constantCoeff) hz
180       exact hu₀ (by simpa using hc0)
181     exact mul_ne_zero (pow_ne_zero a (PowerSeries.X_ne_zero)) hu
182   have hq0chart :
183       algebraMap E (LaurentSeries k) (qC 0) ≠ 0 := by
184     have hq0map :
185         algebraMap (PowerSeries k) (LaurentSeries k) (qArc 0) ≠ 0 := by
186       simpa only [map_zero] using
187         (IsFractionRing.injective (PowerSeries k) (LaurentSeries k)).ne
188           hq0nonzero
189     calc
190       algebraMap E (LaurentSeries k) (qC 0) = rho (qC 0) :=
191         congrArg (fun φ : E →+* LaurentSeries k => φ (qC 0))
192           (RingHom.algebraMap_toAlgebra' rho (by intro x y; exact mul_comm _ _))
193       _ = algebraMap (PowerSeries k) (LaurentSeries k) (qArc 0) := by
194         simpa [qArc] using hposition 0
195       _ ≠ 0 := hq0map
196   have hcorrection : ∀ i,
197       PowerSeries.derivative (R := k) (qArc i) - Z.mulVec
198         (fun j => PowerSeries.C (alpha j)) i =
199           (PowerSeries.X : PowerSeries k) ^ c * tau i := by
200     simpa [qArc, Z] using hfactor
201   have haxis : ∀ j,
202       PowerSeries.constantCoeff
203         (formalTangentMatrix qArc Z tau axis.succ j) = 0 := by
204     simpa [qArc, Z] using haxisColumns
205   refine ⟨tau, C, hCB, haxis, ?_⟩
206   exact DirectSummandInputOfActualChart.directSummandInput_of_actual_chart_columns
207     (rho := rho) hground qC qArc Z tau (fun j => PowerSeries.C (alpha j)) c
208     hcorrection hposition htransverse hraw hq0chart hchart chart hqchart hq0nonzero
209     axis C hCB haxis hprime hclosure hsmooth
