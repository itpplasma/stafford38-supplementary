7 /-!
8 # Kähler visibility at a discrete valuation place
9 
10 For a local ring whose maximal ideal is generated by `t`, the kernel of the
11 map from absolute Kähler differentials to residue-field differentials is the
12 sum of the line generated by `d t` and the maximal-ideal multiple of the
13 source module.  If the residue field is algebraic over the field generated by
14 the residues of a coordinate family, those coordinate differentials span the
15 remaining residue cotangent space.  Mapping this statement to the fraction
16 field gives the visibility condition used by `VisibleDivisorFrame`.
17 -/

29 /-- Tensoring a module with `V/I` kills exactly the submodule `I • M`. -/
30 theorem ker_quotient_tensor_mk_one
31     {V M : Type*} [CommRing V]
32     [AddCommGroup M] [Module V M] (I : Ideal V) :
33     LinearMap.ker (TensorProduct.mk V (V ⧸ I) M 1) =
34       I • (⊤ : Submodule V M) := by
35   ext x
36   constructor
37   · intro hx
38     have hzero : TensorProduct.quotTensorEquivQuotSMul M I
39         ((1 : V ⧸ I) ⊗ₜ[V] x) = 0 := by
40       change (TensorProduct.quotTensorEquivQuotSMul M I)
41         ((TensorProduct.mk V (V ⧸ I) M 1) x) = 0
42       rw [LinearMap.mem_ker.mp hx, map_zero]
43     rw [TensorProduct.quotTensorEquivQuotSMul_mk_one_tmul] at hzero
44     exact (Submodule.Quotient.mk_eq_zero _).mp hzero
45   · intro hx
46     apply (TensorProduct.quotTensorEquivQuotSMul M I).injective
47     change (TensorProduct.quotTensorEquivQuotSMul M I) ((1 : V ⧸ I) ⊗ₜ[V] x) =
48       (TensorProduct.quotTensorEquivQuotSMul M I) 0
49     rw [TensorProduct.quotTensorEquivQuotSMul_mk_one_tmul,
50       (Submodule.Quotient.mk_eq_zero _).mpr hx, map_zero]

52 theorem ker_residue_tensor_mk_one
53     {V M : Type*} [CommRing V] [IsLocalRing V]
54     [AddCommGroup M] [Module V M] :
55     LinearMap.ker (TensorProduct.mk V (ResidueField V) M 1) =
56       maximalIdeal V • (⊤ : Submodule V M) := by
57   change LinearMap.ker (TensorProduct.mk V (V ⧸ maximalIdeal V) M 1) = _
58   exact ker_quotient_tensor_mk_one (maximalIdeal V)

60 /-- If the maximal ideal is generated by `t`, reduction of Kähler
61 differentials has kernel `V · d t + 𝔪 · Ω[V/k]`.
62 
63 The summand generated by `d t` is essential: the kernel is generally larger
64 than `𝔪 · Ω[V/k]`. -/
65 theorem ker_kaehler_residue_map_eq
66     {k V : Type*} [CommRing k] [CommRing V] [IsLocalRing V]
67     [Algebra k V] (t : V) (hmax : maximalIdeal V = Ideal.span {t}) :
68     LinearMap.ker
69         (KaehlerDifferential.map k k V (ResidueField V)) =
70       Submodule.span V {KaehlerDifferential.D k V t} ⊔
71         maximalIdeal V • (⊤ : Submodule V (Ω[V⁄k])) := by
72   let f : Ω[V⁄k] →ₗ[V] ResidueField V ⊗[V] Ω[V⁄k] :=
73     TensorProduct.mk V (ResidueField V) Ω[V⁄k] 1
74   let g : ResidueField V ⊗[V] Ω[V⁄k] →ₗ[ResidueField V] Ω[ResidueField V⁄k] :=
75     KaehlerDifferential.mapBaseChange k V (ResidueField V)
76   apply le_antisymm
77   · intro ω hω
78     have hgf : g (f ω) = 0 := by
79       simpa [f, g, KaehlerDifferential.mapBaseChange_tmul] using
80         (LinearMap.mem_ker.mp hω)
81     have hfrange : f ω ∈ LinearMap.range
82         (KaehlerDifferential.kerCotangentToTensor k V (ResidueField V)) := by
83       rw [KaehlerDifferential.range_kerCotangentToTensor k V (ResidueField V)
84         IsLocalRing.residue_surjective]
85       exact hgf
86     obtain ⟨z, hz⟩ := hfrange
87     obtain ⟨x, rfl⟩ := Ideal.toCotangent_surjective _ z
88     have hxmax : (x : V) ∈ maximalIdeal V := by
89       simpa only [IsLocalRing.ker_residue, ResidueField.algebraMap_eq] using x.property
90     rw [hmax, Ideal.mem_span_singleton] at hxmax
91     obtain ⟨a, ha⟩ := hxmax
92     have hfa : f (a • KaehlerDifferential.D k V t) =
93         KaehlerDifferential.kerCotangentToTensor k V (ResidueField V)
94           (Ideal.toCotangent _ x) := by
95       have htresidue : residue V t = 0 :=
96         (IsLocalRing.residue_eq_zero_iff t).2
97           (hmax.symm ▸ Ideal.mem_span_singleton_self t)
98       rw [KaehlerDifferential.kerCotangentToTensor_toCotangent]
99       rw [ha, Derivation.leibniz]
100       simp only [f, map_smul, TensorProduct.tmul_add,
101         TensorProduct.tmul_smul, TensorProduct.smul_tmul']
102       simp [Algebra.smul_def, htresidue]
103       change (algebraMap V (ResidueField V) a) •
104           ((1 : ResidueField V) ⊗ₜ[V] KaehlerDifferential.D k V t) = _
105       rw [ResidueField.algebraMap_eq, TensorProduct.smul_tmul']
106       simp
107     have hdiff : ω - a • KaehlerDifferential.D k V t ∈
108         maximalIdeal V • (⊤ : Submodule V (Ω[V⁄k])) := by
109       rw [← ker_residue_tensor_mk_one]
110       apply LinearMap.mem_ker.mpr
111       rw [map_sub, hfa, ← hz, sub_self]
112     rw [show ω = a • KaehlerDifferential.D k V t +
113         (ω - a • KaehlerDifferential.D k V t) by abel]
114     exact Submodule.add_mem _
115       (Submodule.mem_sup_left (Submodule.smul_mem _ _
116         (Submodule.subset_span (Set.mem_singleton _))))
117       (Submodule.mem_sup_right hdiff)
118   · refine sup_le ?_ ?_
119     · rw [Submodule.span_le]
120       rintro _ rfl
121       apply LinearMap.mem_ker.mpr
122       rw [KaehlerDifferential.map_D,
123         ResidueField.algebraMap_eq,
124         (IsLocalRing.residue_eq_zero_iff t).2 (hmax.symm ▸ Ideal.mem_span_singleton_self t)]
125       simp
126     · intro ω hω
127       apply LinearMap.mem_ker.mpr
128       have hf : f ω = 0 := by
129         rw [← LinearMap.mem_ker, ker_residue_tensor_mk_one]
130         exact hω
131       simpa [f, g, KaehlerDifferential.mapBaseChange_tmul] using
132         congrArg g hf

134 /-- In characteristic zero, coordinate differentials span after an algebraic
135 extension of the field generated by the coordinates. -/
136 theorem kaehler_span_of_isAlgebraic_adjoin
137     {k K : Type u} {ι : Type v} [Field k] [CharZero k] [Field K] [Algebra k K]
138     (q : ι → K)
139     (halg : Algebra.IsAlgebraic
140       (IntermediateField.adjoin k (Set.range q) : IntermediateField k K) K) :
141     Submodule.span K
142       (Set.range fun i ↦ KaehlerDifferential.D k K (q i)) = ⊤ := by
143   let L : IntermediateField k K :=
144     IntermediateField.adjoin k (Set.range q)
145   let qL : ι → L := fun i ↦
146     ⟨q i, IntermediateField.subset_adjoin k (Set.range q) ⟨i, rfl⟩⟩
147   letI : Algebra L K := L.toSubalgebra.toAlgebra
148   letI : IsScalarTower k L K :=
149     IsScalarTower.of_algebraMap_eq (fun _ ↦ rfl)
150   letI : Algebra.IsAlgebraic L K := halg
151   letI : Algebra.IsSeparable L K := inferInstance
152   have hgenL : IntermediateField.adjoin k (Set.range qL) = ⊤ := by
153     apply top_unique
154     rintro ⟨z, hz⟩ -
155     apply IntermediateField.adjoin_induction k
156         (p := fun x hx ↦
157           (⟨x, hx⟩ : L) ∈ IntermediateField.adjoin k (Set.range qL))
158     · intro x hx
159       obtain ⟨i, rfl⟩ := hx
160       exact IntermediateField.subset_adjoin k _ ⟨i, rfl⟩
161     · intro a
162       exact (IntermediateField.adjoin k (Set.range qL)).algebraMap_mem a
163     · intro a b _ _ ha hb
164       exact (IntermediateField.adjoin k (Set.range qL)).add_mem ha hb
165     · intro a _ ha
166       exact (IntermediateField.adjoin k (Set.range qL)).inv_mem ha
167     · intro a b _ _ ha hb
168       exact (IntermediateField.adjoin k (Set.range qL)).mul_mem ha hb
169   have hspan :=
170     Stafford38.Geometry.KaehlerSpanSeparableAdjoin.kaehler_span_of_separable_adjoin_eq_top
171       (k := k) (E := L) (K := K) qL hgenL
172   simpa only [qL, IntermediateField.algebraMap_apply] using hspan

174 /-- The source differential module is generated modulo the maximal ideal by
175 `d t` and the chosen coordinate differentials. -/
176 theorem kaehler_le_coordinate_span_sup_maximalIdeal_smul
177     {k V : Type u} {ι : Type v}
178     [Field k] [CharZero k] [CommRing V] [IsLocalRing V] [Algebra k V]
179     (t : V) (hmax : maximalIdeal V = Ideal.span {t}) (q : ι → V)
180     (halg : Algebra.IsAlgebraic
181       (IntermediateField.adjoin k
182         (Set.range fun i ↦ residue V (q i)) :
183           IntermediateField k (ResidueField V))
184       (ResidueField V)) :
185     (⊤ : Submodule V (Ω[V⁄k])) ≤
186       (Submodule.span V {KaehlerDifferential.D k V t} ⊔
187           Submodule.span V
188             (Set.range fun i ↦ KaehlerDifferential.D k V (q i))) ⊔
189         maximalIdeal V • (⊤ : Submodule V (Ω[V⁄k])) := by
190   let C : Submodule V (Ω[V⁄k]) :=
191     Submodule.span V {KaehlerDifferential.D k V t} ⊔
192       Submodule.span V (Set.range fun i ↦ KaehlerDifferential.D k V (q i))
193   let S : Submodule (ResidueField V) (Ω[ResidueField V⁄k]) :=
194     Submodule.span (ResidueField V)
195       (Set.range fun i ↦ KaehlerDifferential.D k (ResidueField V) (residue V (q i)))
196   have hS : S = ⊤ := kaehler_span_of_isAlgebraic_adjoin
197     (fun i ↦ residue V (q i)) halg
198   intro ω _
199   have himage : KaehlerDifferential.map k k V (ResidueField V) ω ∈ S := by
200     rw [hS]
201     trivial
202   have hlift : ∀ z ∈ S, ∃ η ∈ C,
203       KaehlerDifferential.map k k V (ResidueField V) η = z := by
204     intro z hz
205     induction hz using Submodule.span_induction with
206     | mem z hz =>
207         obtain ⟨i, rfl⟩ := hz
208         refine ⟨KaehlerDifferential.D k V (q i), ?_, ?_⟩
209         · exact Submodule.mem_sup_right (Submodule.subset_span ⟨i, rfl⟩)
210         · rw [KaehlerDifferential.map_D, ResidueField.algebraMap_eq]
211     | zero => exact ⟨0, C.zero_mem, map_zero _⟩
212     | add x y hx hy hx' hy' =>
213         obtain ⟨x', hx'C, hx'map⟩ := hx'
214         obtain ⟨y', hy'C, hy'map⟩ := hy'
215         refine ⟨x' + y', C.add_mem hx'C hy'C, ?_⟩
216         rw [map_add, hx'map, hy'map]
217     | smul a x hx hx' =>
218         obtain ⟨x', hx'C, hx'map⟩ := hx'
219         obtain ⟨a', ha'⟩ := IsLocalRing.residue_surjective a
220         refine ⟨a' • x', C.smul_mem a' hx'C, ?_⟩
221         rw [map_smul, hx'map]
222         change (algebraMap V (ResidueField V) a') • x = a • x
223         rw [ResidueField.algebraMap_eq, ha']
224   obtain ⟨η, hηC, hη⟩ := hlift _ himage
225   have hker : ω - η ∈ LinearMap.ker
226       (KaehlerDifferential.map k k V (ResidueField V)) := by
227     apply LinearMap.mem_ker.mpr
228     rw [map_sub, hη, sub_self]
229   rw [ker_kaehler_residue_map_eq t hmax] at hker
230   obtain ⟨δ, hδdt, μ, hμm, hsum⟩ := Submodule.mem_sup.mp hker
231   rw [show ω = η + (δ + μ) by rw [hsum]; abel]
232   exact Submodule.add_mem _
233     (Submodule.mem_sup_left hηC)
234     (Submodule.add_mem _
235       (Submodule.mem_sup_left
236         (Submodule.mem_sup_left hδdt))
237       (Submodule.mem_sup_right hμm))

239 /-- The image of `Ω[V/k]` in `Ω[F/k]` satisfies the visibility condition
240 for an independent generator `t` of the maximal ideal. -/
241 theorem kaehler_image_visible
242     {k V F : Type u} {ι : Type v}
243     [Fintype ι]
244     [Field k] [CharZero k] [CommRing V] [IsLocalRing V] [Field F]
245     [Algebra k V] [Algebra k F] [Algebra V F] [IsScalarTower k V F]
246     (t : V) (hmax : maximalIdeal V = Ideal.span {t}) (q : ι → V)
247     (halg : Algebra.IsAlgebraic
248       (IntermediateField.adjoin k
249         (Set.range fun i ↦ residue V (q i)) :
250           IntermediateField k (ResidueField V))
251       (ResidueField V)) :
252     LinearMap.range (KaehlerDifferential.map k k V F) ≤
253       Stafford38.Geometry.DivisorTangentLattice.coordinateFrame
254           (KaehlerDifferential.D k F) t q ⊔
255         Ideal.span {t} •
256           LinearMap.range (KaehlerDifferential.map k k V F) := by
257   let φ : Ω[V⁄k] →ₗ[V] Ω[F⁄k] :=
258     KaehlerDifferential.map k k V F
259   let C : Submodule V (Ω[V⁄k]) :=
260     Submodule.span V {KaehlerDifferential.D k V t} ⊔
261       Submodule.span V (Set.range fun i ↦ KaehlerDifferential.D k V (q i))
262   let CF : Submodule V (Ω[F⁄k]) :=
263     Stafford38.Geometry.DivisorTangentLattice.coordinateFrame
264       (KaehlerDifferential.D k F) t q
265   have hsource := kaehler_le_coordinate_span_sup_maximalIdeal_smul
266     t hmax q halg
267   have hcoord : C ≤ Submodule.comap φ CF := by
268     refine sup_le ?_ ?_
269     · rw [Submodule.span_le, Set.singleton_subset_iff]
270       change φ (KaehlerDifferential.D k V t) ∈ CF
271       change KaehlerDifferential.map k k V F
272         (KaehlerDifferential.D k V t) ∈ CF
273       rw [KaehlerDifferential.map_D]
274       exact Submodule.mem_sup_left
275         (Submodule.subset_span (Set.mem_singleton _))
276     · rw [Submodule.span_le]
277       rintro _ ⟨i, rfl⟩
278       change φ (KaehlerDifferential.D k V (q i)) ∈ CF
279       change KaehlerDifferential.map k k V F
280         (KaehlerDifferential.D k V (q i)) ∈ CF
281       rw [KaehlerDifferential.map_D]
282       exact Stafford38.Geometry.DivisorTangentLattice.dQ_mem_coordinateFrame
283         (KaehlerDifferential.D k F) t q i
284   rintro w ⟨ω, rfl⟩
285   have hω := hsource (show ω ∈ (⊤ : Submodule V (Ω[V⁄k])) from trivial)
286   rw [hmax] at hω
287   obtain ⟨η, hηC, μ, hμm, hsum⟩ := Submodule.mem_sup.mp hω
288   have hηCF : φ η ∈ CF := hcoord hηC
289   have hμW : φ μ ∈ Ideal.span {t} • LinearMap.range φ := by
290     have hmapped : φ μ ∈ Submodule.map φ
291         (Ideal.span {t} • (⊤ : Submodule V (Ω[V⁄k]))) :=
292       Submodule.mem_map_of_mem hμm
293     rwa [Submodule.map_smul'', Submodule.map_top] at hmapped
294   rw [← hsum, map_add]
295   exact Submodule.add_mem _
296     (Submodule.mem_sup_left hηCF)
297     (Submodule.mem_sup_right hμW)

299 /-- Package the Kähler image as the lattice of a visible divisor frame once
300 the independent DVR and projective-order data have been supplied. -/
301 theorem exists_visibleDivisorFrame_of_kaehler_image
302     {k V F : Type u} {ι : Type v}
303     [Fintype ι]
304     [Field k] [CharZero k] [CommRing V] [IsLocalRing V] [Field F]
305     [Algebra k V] [Algebra k F] [Algebra V F] [IsScalarTower k V F]
306     [Module.Finite V (Ω[V⁄k])]
307     (Q₀ Q₁ t u w : V) (Q : ι → V) (a e : ℕ) (j₀ : ι)
308     (hmax : maximalIdeal V = Ideal.span {t})
309     (t_ne : t ≠ 0) (u_unit : IsUnit u)
310     (one_le_a : 1 ≤ a) (one_le_e : 1 ≤ e)
311     (Q₀_eq : Q₀ = t ^ a * u) (Q₁_eq : Q₁ = t ^ (a + e) * w)
312     (Q_j₀ : Q j₀ = 1)
313     (halg : Algebra.IsAlgebraic
314       (IntermediateField.adjoin k
315         (Set.range fun i ↦ residue V (Q i)) :
316           IntermediateField k (ResidueField V))
317       (ResidueField V)) :
318     ∃ D : Stafford38.Geometry.DivisorTangentLattice.VisibleDivisorFrame
319         (V := V) (KaehlerDifferential.D k F) ι,
320       D.Q₀ = Q₀ ∧ D.Q₁ = Q₁ ∧ D.t = t ∧ D.u = u ∧ D.w = w ∧
321         D.Q = Q ∧ D.a = a ∧ D.e = e ∧ D.j₀ = j₀ ∧
322         D.W = LinearMap.range (KaehlerDifferential.map k k V F) := by
323   let φ : Ω[V⁄k] →ₗ[V] Ω[F⁄k] :=
324     KaehlerDifferential.map k k V F
325   have W_fg : (LinearMap.range φ).FG := by
326     rw [LinearMap.range_eq_map]
327     exact Module.Finite.fg_top.map φ
328   let D : Stafford38.Geometry.DivisorTangentLattice.VisibleDivisorFrame
329       (V := V) (KaehlerDifferential.D k F) ι :=
330     { Q₀ := Q₀
331       Q₁ := Q₁
332       t := t
333       u := u
334       w := w
335       Q := Q
336       a := a
337       e := e
338       j₀ := j₀
339       W := LinearMap.range φ
340       t_mem := hmax.symm ▸ Ideal.mem_span_singleton_self t
341       t_ne := t_ne
342       u_unit := u_unit
343       a_unit := by
344         have hak : (a : k) ≠ 0 := Nat.cast_ne_zero.mpr
345           (Nat.ne_of_gt (lt_of_lt_of_le Nat.zero_lt_one one_le_a))
346         exact_mod_cast (isUnit_iff_ne_zero.mpr hak).map (algebraMap k V)
347       one_le_a := one_le_a
348       one_le_e := one_le_e
349       Q₀_eq := Q₀_eq
350       Q₁_eq := Q₁_eq
351       Q_j₀ := Q_j₀
352       W_fg := W_fg
353       dt_mem := ⟨KaehlerDifferential.D k V t, by
354         simp only [φ, KaehlerDifferential.map_D]⟩
355       du_mem := ⟨KaehlerDifferential.D k V u, by
356         simp only [φ, KaehlerDifferential.map_D]⟩
357       dw_mem := ⟨KaehlerDifferential.D k V w, by
358         simp only [φ, KaehlerDifferential.map_D]⟩
359       dQ_mem := fun j ↦ ⟨KaehlerDifferential.D k V (Q j), by
360         simp only [φ, KaehlerDifferential.map_D]⟩
361       visible := kaehler_image_visible t hmax Q halg }
362   exact ⟨D, rfl, rfl, rfl, rfl, rfl, rfl, rfl, rfl, rfl, rfl⟩
