18 /-- If an actual coordinate map sends the distinguished variable to the
19 polynomial parameter and each selected coordinate to its coefficient-field
20 image, it is exactly the composition through the coefficient localization.
21 Consequently it is formally etale whenever the second map is. -/
22 theorem formallyEtale_of_optionCoordinate_generator_images
23     {k : Type u} [CommRing k] {σ : Type w} {E : Type v} [CommRing E]
24     {T : Type x} [CommRing T]
25     [Algebra (MvPolynomial σ k) E] [Algebra k E]
26     [IsScalarTower k (MvPolynomial σ k) E]
27     [IsLocalization (nonZeroDivisors (MvPolynomial σ k)) E]
28     [Algebra k T] [Algebra (Polynomial E) T]
29     [IsScalarTower k (Polynomial E) T]
30     (f : MvPolynomial (Option σ) k →ₐ[k] T)
31     (hnone : f (MvPolynomial.X none) =
32       algebraMap (Polynomial E) T (Polynomial.X : Polynomial E))
33     (hsome : ∀ i : σ, f (MvPolynomial.X (some i)) =
34       algebraMap (Polynomial E) T
35         (Polynomial.C (algebraMap (MvPolynomial σ k) E (MvPolynomial.X i))))
36     (hsecond : Algebra.FormallyEtale (Polynomial E) T) :
37     letI : Algebra (MvPolynomial (Option σ) k) T := f.toRingHom.toAlgebra
38     Algebra.FormallyEtale (MvPolynomial (Option σ) k) T := by
39   let optionMap := Stafford38.Geometry.ProjectiveCoefficientLocalization.optionCoordinateMap
40     (k := k) (E := E) (σ := σ)
41   letI : Algebra (MvPolynomial (Option σ) k) (Polynomial E) :=
42     optionMap.toRingHom.toAlgebra
43   letI : Algebra (MvPolynomial (Option σ) k) T := f.toRingHom.toAlgebra
44   have hfactor : f = (IsScalarTower.toAlgHom k (Polynomial E) T).comp optionMap := by
45     apply MvPolynomial.algHom_ext
46     intro i
47     cases i with
48     | none =>
49         simpa [optionMap, ProjectiveCoefficientLocalization.optionCoordinateMap_X_none]
50           using hnone
51     | some j =>
52         simpa [optionMap, ProjectiveCoefficientLocalization.optionCoordinateMap_X_some]
53           using hsome j
54   have hscalarMap :
55       algebraMap (MvPolynomial (Option σ) k) T =
56         (algebraMap (Polynomial E) T).comp
57           (algebraMap (MvPolynomial (Option σ) k) (Polynomial E)) := by
58     apply RingHom.ext
59     intro r
60     change f r = algebraMap (Polynomial E) T (optionMap r)
61     have hr := congrArg (fun g : MvPolynomial (Option σ) k →ₐ[k] T => g r) hfactor
62     simpa [IsScalarTower.toAlgHom_apply] using hr
63   letI : IsScalarTower (MvPolynomial (Option σ) k) (Polynomial E) T :=
64     IsScalarTower.of_algebraMap_eq' hscalarMap
65   have hfirst : Algebra.FormallyEtale (MvPolynomial (Option σ) k) (Polynomial E) := by
66     exact ProjectiveCoefficientLocalization.optionCoordinateMap_formallyEtale
67       (k := k) (E := E) (σ := σ)
68   exact Algebra.FormallyEtale.comp (MvPolynomial (Option σ) k) (Polynomial E) T

70 /-- If the target is finite type over the coefficient ring and the coordinate
71 map extends its coefficient algebra structure, then it is finitely presented
72 over the finite-variable polynomial source. This uses Noetherianity of that
73 source and does not require formal-etaleness. -/
74 theorem finitePresentation_of_finiteType_target
75     {k : Type u} [CommRing k] [IsNoetherianRing k]
76     {σ : Type w} [Fintype σ] {B : Type x} [CommRing B]
77     [Algebra k B] [Algebra (MvPolynomial (Option σ) k) B]
78     [IsScalarTower k (MvPolynomial (Option σ) k) B]
79     (hB : Algebra.FiniteType k B) :
80     Algebra.FinitePresentation (MvPolynomial (Option σ) k) B := by
81   letI : Algebra.FiniteType k B := hB
82   letI : Algebra.FiniteType (MvPolynomial (Option σ) k) B :=
83     Algebra.FiniteType.of_restrictScalars_finiteType k
84       (MvPolynomial (Option σ) k) B
85   exact (Algebra.FinitePresentation.of_finiteType (R := MvPolynomial (Option σ) k)
86     (A := B)).mp inferInstance
