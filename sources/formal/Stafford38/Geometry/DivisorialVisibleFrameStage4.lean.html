18 theorem stage4_residueField_isAlgebraic_of_isIntegral
19     {k K : Type u} [Field k] [Field K] [Algebra k K]
20     (A : Subalgebra k K) (p : Ideal A) [p.IsPrime]
21     (V : ValuationSubring K) [IsLocalRing V.toSubring]
22     [Algebra k V.toSubring] [IsScalarTower k V.toSubring K]
23     (hAV : ∀ a : A, (a : K) ∈ V.toSubring)
24     (hp : ∀ a : A, (⟨a, hAV a⟩ : V.toSubring) ∈ maximalIdeal V.toSubring ↔ a ∈ p)
25     (hsurj : ∀ z : ResidueField V.toSubring, ∃ (a b : A), b ∉ p ∧
26       z = residue V.toSubring ⟨a, hAV a⟩ / residue V.toSubring ⟨b, hAV b⟩)
27     {ι : Type v} (c : ι → K) (hc : ∀ i, c i ∈ A)
28     (hint : ∀ a : A, IsIntegral (Algebra.adjoin k (Set.range c)) (a : K)) :
29     Algebra.IsAlgebraic
30       (IntermediateField.adjoin k (Set.range fun i => residue V.toSubring ⟨c i, hAV ⟨c i, hc i⟩⟩) :
31         IntermediateField k (ResidueField V.toSubring))
32       (ResidueField V.toSubring) := by
33   let C : Subalgebra k K := Algebra.adjoin k (Set.range c)
34   have hCA : C ≤ A := by
35     apply Algebra.adjoin_le
36     rintro z ⟨i, rfl⟩
37     exact hc i
38   let cV : C →+* V.toSubring :=
39     RingHom.codRestrict C.val.toRingHom V.toSubring (fun z => hAV ⟨z, hCA z.property⟩)
40   let ρ : V.toSubring →+* ResidueField V.toSubring := residue V.toSubring
41   let φ : C →+* ResidueField V.toSubring := ρ.comp cV
42   let L : IntermediateField k (ResidueField V.toSubring) :=
43     IntermediateField.adjoin k (Set.range fun i => ρ ⟨c i, hAV ⟨c i, hc i⟩⟩)
44   have hφL : ∀ z : C, φ z ∈ L := by
45     intro z
46     apply Algebra.adjoin_induction
47       (p := fun x hx => φ ⟨x, hx⟩ ∈ L)
48     · intro x hx
49       rcases hx with ⟨i, rfl⟩
50       exact IntermediateField.subset_adjoin k _ ⟨i, rfl⟩
51     · intro a
52       have hcV : cV (algebraMap k C a) = algebraMap k V.toSubring a := by
53         apply Subtype.ext
54         change (algebraMap k K) a =
55           (algebraMap V.toSubring K) ((algebraMap k V.toSubring) a)
56         exact IsScalarTower.algebraMap_apply k V.toSubring K a
57       have hbase : φ (algebraMap k C a) =
58           algebraMap k (ResidueField V.toSubring) a := by
59         calc
60           φ (algebraMap k C a) = ρ (algebraMap k V.toSubring a) := by
61             simp only [φ, RingHom.comp_apply, hcV]
62           _ = algebraMap k (ResidueField V.toSubring) a := by
63             change residue V.toSubring (algebraMap k V.toSubring a) = _
64             rw [← ResidueField.algebraMap_eq]
65             exact (IsScalarTower.algebraMap_apply k V.toSubring
66               (ResidueField V.toSubring) a).symm
67       have hmem : φ (algebraMap k C a) ∈ L := by
68         rw [hbase]
69         exact L.algebraMap_mem a
70       have harg : (⟨(algebraMap k K) a, C.algebraMap_mem a⟩ : C) = algebraMap k C a := by
71         apply Subtype.ext
72         rfl
73       simpa [harg] using hmem
74     · intro a b haC hbC ha hb
75       change φ ((⟨a, haC⟩ : C) + ⟨b, hbC⟩) ∈ L
76       rw [map_add]
77       exact add_mem ha hb
78     · intro a b haC hbC ha hb
79       change φ ((⟨a, haC⟩ : C) * ⟨b, hbC⟩) ∈ L
80       rw [map_mul]
81       exact mul_mem ha hb
82   let cL : C →+* L := φ.codRestrict L.toSubring hφL
83   letI : Algebra C V.toSubring := cV.toAlgebra
84   letI : Algebra C (ResidueField V.toSubring) := φ.toAlgebra
85   letI : IsScalarTower C V.toSubring K :=
86     IsScalarTower.of_algebraMap_eq (fun _ => rfl)
87   letI : IsScalarTower C V.toSubring (ResidueField V.toSubring) := by
88     constructor
89     intro x y z
90     change ρ (cV x) * ρ y * z = ρ (cV x) * (ρ y * z)
91     exact mul_assoc _ _ _
92   letI : IsScalarTower C C V.toSubring :=
93     IsScalarTower.of_algebraMap_eq (fun _ => rfl)
94   letI : IsScalarTower C C (ResidueField V.toSubring) :=
95     IsScalarTower.of_algebraMap_eq (fun _ => rfl)
96   letI : Algebra C L := cL.toAlgebra
97   letI : IsScalarTower C L (ResidueField V.toSubring) := by
98     apply IsScalarTower.of_algebraMap_eq
99     intro x
100     rfl
101   have hmap : ∀ a : A, IsIntegral C (ρ ⟨a, hAV a⟩) := by
102     intro a
103     let av : V.toSubring := ⟨a, hAV a⟩
104     have haK : IsIntegral C (a : K) := by simpa [C] using hint a
105     have haV : IsIntegral C av := by
106       apply (isIntegral_algebraMap_iff (R := C) (A := V.toSubring)
107         (B := K)).mp
108       change IsIntegral C (a : K)
109       exact haK
110     let f : V.toSubring →ₐ[C] ResidueField V.toSubring :=
111       { ρ with
112         commutes' := by
113           intro x
114           rfl }
115     exact IsIntegral.map f haV
116   refine ⟨fun z => ?_⟩
117   obtain ⟨a, b, hb, hz⟩ := hsurj z
118   have ha := hmap a
119   have hbin := hmap b
120   have hbp : ρ ⟨b, hAV b⟩ ≠ 0 := by
121     intro hzero
122     have hnotmax : (⟨b, hAV b⟩ : V.toSubring) ∉ maximalIdeal V.toSubring := by
123       intro hmem
124       exact hb ((hp b).mp hmem)
125     exact ((residue_ne_zero_iff_isUnit _).2
126       ((notMem_maximalIdeal).1 hnotmax)) hzero
127   have haL : IsIntegral L (ρ ⟨a, hAV a⟩) := IsIntegral.tower_top ha
128   have hbL : IsIntegral L (ρ ⟨b, hAV b⟩) := IsIntegral.tower_top hbin
129   have h_ainv : IsAlgebraic L (ρ ⟨a, hAV a⟩) := haL.isAlgebraic
130   have h_binv : IsAlgebraic L (ρ ⟨b, hAV b⟩) := hbL.isAlgebraic
131   rw [hz]
132   exact h_ainv.mul h_binv.inv

136 theorem stage4'_isAlgebraic_of_normalized_column
137     {k K : Type u} [Field k] [Field K] [Algebra k K]
138     (V : ValuationSubring K) [IsLocalRing V.toSubring] [Algebra k V.toSubring]
139     {r : ℕ} (y : Fin r → K) (j : Fin (r + 1)) (hj : j ≠ 0) (s₀ : K)
140     (c : Fin (r + 1) → V.toSubring) (hc : ∀ a, (c a : K) = (Fin.cases 1 y a : K) * s₀)
141     (hcj : c j = 1)
142     (halg : Algebra.IsAlgebraic
143       (IntermediateField.adjoin k (Set.range fun a => residue V.toSubring (c a)) :
144         IntermediateField k (ResidueField V.toSubring)) (ResidueField V.toSubring))
145     (s : K) (q : Fin (r + 1) → V.toSubring) (hq : ∀ a, (q a : K) = s * Fin.cases 1 y a)
146     (hq1 : ∃ j', q j' = 1) (hq0 : ¬ IsUnit (q 0)) :
147     Algebra.IsAlgebraic
148       (IntermediateField.adjoin k
149         (Set.range fun b : Fin r => residue V.toSubring (q (Fin.succ b))) :
150         IntermediateField k (ResidueField V.toSubring))
151       (ResidueField V.toSubring) := by
152   classical
153   let ρ : V.toSubring →+* ResidueField V.toSubring := residue V.toSubring
154   let Lc : IntermediateField k (ResidueField V.toSubring) :=
155     IntermediateField.adjoin k (Set.range fun a => ρ (c a))
156   let Lq : IntermediateField k (ResidueField V.toSubring) :=
157     IntermediateField.adjoin k (Set.range fun b : Fin r => ρ (q (Fin.succ b)))
158   let lam : ResidueField V.toSubring := ρ (q j)
159   have hcjK : (Fin.cases 1 y j : K) * s₀ = 1 := by
160     have := congrArg (fun z : V.toSubring => (z : K)) hcj
161     simpa [hc j] using this
162   have hprodK (a : Fin (r + 1)) : (q a : K) = (q j : K) * (c a : K) := by
163     rw [hq a, hq j, hc a]
164     calc
165       s * Fin.cases 1 y a = (s * Fin.cases 1 y a) * 1 := by rw [mul_one]
166       _ = (s * Fin.cases 1 y a) * ((Fin.cases 1 y j) * s₀) := by rw [hcjK]
167       _ = (s * Fin.cases 1 y j) * ((Fin.cases 1 y a) * s₀) := by ring
168   have hprodV (a : Fin (r + 1)) : q a = q j * c a := by
169     apply Subtype.ext
170     exact hprodK a
171   have hprod (a : Fin (r + 1)) : ρ (q a) = lam * ρ (c a) := by
172     rw [hprodV a, map_mul]
173   obtain ⟨j', hj'⟩ := hq1
174   have hlam : lam ≠ 0 := by
175     intro hzero
176     have hqj' : IsUnit (q j') := by rw [hj']; exact isUnit_one
177     have hnz : ρ (q j') ≠ 0 := (residue_ne_zero_iff_isUnit _).2 hqj'
178     apply hnz
179     rw [hprod j', hzero, zero_mul]
180   obtain ⟨jt, hjt⟩ := Fin.exists_succ_eq_of_ne_zero hj
181   have hlammem : lam ∈ Lq := by
182     change ρ (q j) ∈ Lq
183     rw [← hjt]
184     exact IntermediateField.subset_adjoin k _ ⟨jt, rfl⟩
185   have hq0res : ρ (q 0) = 0 := by
186     by_contra hnz
187     exact hq0 ((residue_ne_zero_iff_isUnit _).1 hnz)
188   have hc0res : ρ (c 0) = 0 := by
189     have h := hprod 0
190     rw [hq0res] at h
191     exact (mul_eq_zero.mp h.symm).resolve_left hlam
192   have hctail (b : Fin r) : ρ (c (Fin.succ b)) = lam⁻¹ * ρ (q (Fin.succ b)) := by
193     calc
194       ρ (c (Fin.succ b)) = lam⁻¹ * (lam * ρ (c (Fin.succ b))) := by
195         rw [← mul_assoc, inv_mul_cancel₀ hlam, one_mul]
196       _ = lam⁻¹ * ρ (q (Fin.succ b)) := by rw [← hprod]
197   have hcLq : ∀ a : Fin (r + 1), ρ (c a) ∈ Lq := by
198     intro a
199     refine Fin.cases ?_ (fun b => ?_) a
200     · rw [hc0res]
201       exact Lq.zero_mem
202     · rw [hctail]
203       exact Lq.mul_mem (Lq.inv_mem hlammem)
204         (IntermediateField.subset_adjoin k _ ⟨b, rfl⟩)
205   have hLcLq : Lc ≤ Lq := by
206     intro z hz
207     apply IntermediateField.adjoin_induction (F := k)
208       (E := ResidueField V.toSubring)
209       (s := Set.range fun a => ρ (c a))
210       (p := fun x hx => x ∈ Lq)
211     · intro x hx
212       rcases hx with ⟨a, rfl⟩
213       exact hcLq a
214     · intro a
215       exact Lq.algebraMap_mem a
216     · intro a b _ _ ha hb
217       simpa only [map_add] using Lq.add_mem ha hb
218     · intro a _ ha
219       simpa only [map_inv₀] using Lq.inv_mem ha
220     · intro a b _ _ ha hb
221       simpa only [map_mul] using Lq.mul_mem ha hb
222     · exact hz
223   letI : Algebra Lc Lq := (IntermediateField.inclusion hLcLq).toAlgebra
224   letI : IsScalarTower Lc Lq (ResidueField V.toSubring) :=
225     IsScalarTower.of_algebraMap_eq (fun _ => rfl)
226   exact ⟨fun z =>
227     (Algebra.IsAlgebraic.isAlgebraic z).extendScalars
228       (IntermediateField.inclusion_injective hLcLq)⟩
