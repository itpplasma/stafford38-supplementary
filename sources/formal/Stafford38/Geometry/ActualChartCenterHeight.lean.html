15 /-- After a localization center has been identified with the contraction of a
16 canonical normalization center, the dimension-theoretic conclusion follows
17 from the finite-type curve height theorem. -/
18 theorem normalized_center_height_one_of_localized_curve
19     {E : Type u} {R : Type v} {A : Type w} {F : Type x} {C : Type y}
20     [Field E] [CommRing R] [IsDomain R] {S : Submonoid R}
21     [CommRing A] [IsDomain A] [Algebra R A] [IsLocalization S A]
22     [Algebra E A] [Algebra.FiniteType E A]
23     [Field F] [Algebra A F] [IsFractionRing A F]
24     [Algebra E F] [IsScalarTower E A F]
25     [CommRing C]
26     (e : R ≃+* C)
27     (p : PrimeSpectrum A) [hp : p.asIdeal.IsMaximal]
28     (J : Ideal C)
29     (hcenter : J.comap e.toRingHom = Ideal.under R p.asIdeal)
30     (h0 : J.comap e.toRingHom ≠ ⊥)
31     (htrdeg : Algebra.trdeg E F = 1) : J.height = 1 := by
32   letI : p.asIdeal.IsMaximal := hp
33   have h0' : Ideal.under R p.asIdeal ≠ ⊥ := by
34     rw [← hcenter]
35     exact h0
36   have hheight :=
37     Stafford38.Geometry.FiniteTypeCurveHeight.localized_maximal_center_height_eq_one_of_fraction_field_trdeg_one
38       (E := E) (R := R) (A := A) (L := F) (S := S) htrdeg p h0'
39   calc
40     J.height = (J.comap e.toRingHom).height :=
41       (RingEquiv.height_comap e J).symm
42     _ = (Ideal.under R p.asIdeal).height := by rw [hcenter]
43     _ = 1 := hheight
44 
45 end Stafford38.Geometry.ActualChartCenterHeight
