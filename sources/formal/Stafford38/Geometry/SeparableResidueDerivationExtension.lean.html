8 # Extending residue-field derivations through a separable extension
9 
10 Let `k -> E -> K` be a tower of fields and suppose that `K/E` is separable.
11 Formal etaleness identifies the Kahler differentials of `K/k` with the base
12 change to `K` of the differentials of `E/k`.  Dualizing gives an exact linear
13 equivalence
14 
15 `Der_k(K, K) = Der_k(E, K)`.
16 
17 Thus every `k`-derivation of the relative coefficient field, after mapping its
18 values into the boundary residue field, extends uniquely to that residue
19 field.  This is the derivation-extension part of the higher-dimensional
20 boundary-chart producer.
21 
22 This file does not construct a coefficient-field section of a complete DVR,
23 identify a completion with `K[[t]]`, or extend these derivations continuously
24 to power series.  Those remain separate inputs to the completed-chart step.
25 -/

41 /-- Extend a `k`-derivation `E -> K` uniquely through the separable field
42 extension `K/E`.  The construction is the dual of the formally-etale base
43 change equivalence for Kahler differentials. -/
44 noncomputable def extendDerivation
45     (D : Derivation k E K) : Derivation k K K := by
46   letI : Algebra.FormallyEtale E K :=
47     Algebra.FormallyEtale.of_isSeparable E K
48   let base : K ⊗[E] KaehlerDifferential k E →ₗ[K] K :=
49     D.liftKaehlerDifferential.liftBaseChange K
50   let pull : KaehlerDifferential k K →ₗ[K] K :=
51     base.comp
52       (KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale
53         k E K).symm.toLinearMap
54   exact KaehlerDifferential.linearMapEquivDerivation k K pull
55 
56 /-- Restricting the extended derivation to `E` recovers the supplied
57 `E -> K` derivation exactly. -/
58 @[simp]
59 theorem extendDerivation_compAlgebraMap
60     (D : Derivation k E K) :
61     (extendDerivation k E K D).compAlgebraMap E = D := by
62   letI : Algebra.FormallyEtale E K :=
63     Algebra.FormallyEtale.of_isSeparable E K
64   apply Derivation.ext
65   intro e
66   simp [extendDerivation,
67     KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale_symm_D_algebraMap,
68     Derivation.liftKaehlerDifferential_comp_D]
69 
70 /-- A `k`-derivation of `K` is determined by its restriction to `E` when
71 `K/E` is separable. -/
72 theorem derivation_ext_of_compAlgebraMap_eq
73     {D₁ D₂ : Derivation k K K}
74     (h : D₁.compAlgebraMap E = D₂.compAlgebraMap E) :
75     D₁ = D₂ := by
76   letI : Algebra.FormallyEtale E K :=
77     Algebra.FormallyEtale.of_isSeparable E K
78   let e := KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale k E K
79   have hbase :
80       (D₁.liftKaehlerDifferential.restrictScalars E).comp
81           (KaehlerDifferential.map k k E K) =
82         (D₂.liftKaehlerDifferential.restrictScalars E).comp
83           (KaehlerDifferential.map k k E K) := by
84     apply Derivation.liftKaehlerDifferential_unique
85     apply Derivation.ext
86     intro x
87     simpa [KaehlerDifferential.map_D,
88       Derivation.liftKaehlerDifferential_comp_D] using
89       Derivation.congr_fun h x
90   have hpull :
91       D₁.liftKaehlerDifferential.comp e.toLinearMap =
92         D₂.liftKaehlerDifferential.comp e.toLinearMap := by
93     apply LinearMap.ext
94     intro z
95     induction z using TensorProduct.induction_on with
96     | zero => simp
97     | add x y hx hy => simp only [map_add, hx, hy]
98     | tmul a x =>
99         simp only [e,
100           KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale_apply,
101           KaehlerDifferential.mapBaseChange_tmul,
102           LinearMap.comp_apply, LinearEquiv.coe_coe, LinearMap.map_smul]
103         exact congrArg (a • ·) (LinearMap.congr_fun hbase x)
104   apply Derivation.ext
105   intro x
106   have hmaps : D₁.liftKaehlerDifferential = D₂.liftKaehlerDifferential := by
107     apply LinearMap.ext
108     intro w
109     obtain ⟨z, rfl⟩ := e.surjective w
110     exact LinearMap.congr_fun hpull z
111   simpa [Derivation.liftKaehlerDifferential_comp_D] using
112     LinearMap.congr_fun hmaps (KaehlerDifferential.D k K x)

114 /-- Existence and uniqueness in the direct form used by the boundary tower. -/
115 theorem existsUnique_derivation_extension
116     (D : Derivation k E K) :
117     ∃! D' : Derivation k K K, D'.compAlgebraMap E = D := by
118   refine ⟨extendDerivation k E K D,
119     extendDerivation_compAlgebraMap k E K D, ?_⟩
120   intro D' hD'
121   apply derivation_ext_of_compAlgebraMap_eq k E K
122   rw [hD', extendDerivation_compAlgebraMap]
123 
124 /-- Restriction of derivations along `E -> K`, as a `K`-linear map. -/
125 def restrictDerivation :
126     Derivation k K K →ₗ[K] Derivation k E K where
127   toFun D := D.compAlgebraMap E
128   map_add' _ _ := by ext; rfl
129   map_smul' _ _ := by ext; rfl
130 
131 /-- Restriction is a linear equivalence for a separable field extension.  In
132 particular, finite derivation frames can be transported without losing linear
133 relations. -/
134 noncomputable def derivationRestrictionEquiv :
135     Derivation k K K ≃ₗ[K] Derivation k E K :=
136   LinearEquiv.ofBijective (restrictDerivation k E K) ⟨
137     (fun _ _ h ↦ derivation_ext_of_compAlgebraMap_eq k E K h),
138     (fun D ↦ ⟨extendDerivation k E K D,
139       extendDerivation_compAlgebraMap k E K D⟩)⟩
140 
141 @[simp]
142 theorem derivationRestrictionEquiv_apply
143     (D : Derivation k K K) :
144     derivationRestrictionEquiv k E K D = D.compAlgebraMap E :=
145   rfl

147 /-- Specialization to a derivation whose values initially lie in `E`. -/
148 noncomputable def extendCoefficientDerivation
149     (D : Derivation k E E) : Derivation k K K :=
150   extendDerivation k E K ((Algebra.linearMap E K).compDer D)

152 @[simp]
153 theorem extendCoefficientDerivation_compAlgebraMap
154     (D : Derivation k E E) :
155     (extendCoefficientDerivation k E K D).compAlgebraMap E =
