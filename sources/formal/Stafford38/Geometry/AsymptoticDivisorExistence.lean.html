13 # Coordinate-zero local model for a boundary place
14 
15 For a transcendental coordinate `x`, evaluation embeds the DVR
16 `k[X]_(X)` in the ambient field, and its local parameter maps to `x`. This
17 file also records unrestricted valuation-ring extension and the strict order
18 gap for `[x:x²]`. The finite-type rank-one extension is assembled by
19 `RelativeRetainedBoundaryPlace.exists_data_of_fg_charZero`; the visible
20 normalized frame is assembled by `GeneralDivisorialVisibleFrame`. The scope
21 here is the one-coordinate local model, not the projective-chart comparison.
22 -/

35 /-- Over an algebraically closed ground field, "nonconstant" is exactly the
36 input needed below: an element outside the scalar image is transcendental. -/
37 theorem transcendental_of_not_mem_range_algebraMap
38     {K : Type v} [Field K] [Algebra k K] [IsAlgClosed k]
39     {x : K} (hx : x ∉ Set.range (algebraMap k K)) :
40     Transcendental k x := by
41   change ¬IsAlgebraic k x
42   intro halg
43   let L : IntermediateField k K := IntermediateField.adjoin k {x}
44   letI : Algebra.IsAlgebraic k L :=
45     IntermediateField.isAlgebraic_adjoin (K := k) (L := K) (S := {x}) (by
46       intro y hy
47       simpa only [Set.mem_singleton_iff] using hy ▸ halg.isIntegral)
48   have hL : L = ⊥ :=
49     IntermediateField.eq_bot_of_isAlgClosed_of_isAlgebraic L
50   apply hx
51   rw [← IntermediateField.mem_bot, ← hL]
52   exact IntermediateField.subset_adjoin k {x} (Set.mem_singleton x)

54 /-- The prime ideal `(X)` in the one-variable polynomial ring. -/
55 def coordinateZeroPrime : Ideal (Polynomial k) := Ideal.span {Polynomial.X}
56 
57 instance coordinateZeroPrime_isPrime : (coordinateZeroPrime k).IsPrime := by
58   exact (Ideal.span_singleton_prime Polynomial.X_ne_zero).2 Polynomial.prime_X

60 /-- The local ring of the affine line at the origin. -/
61 abbrev CoordinateZeroLocalRing := Localization.AtPrime (coordinateZeroPrime k)
62 
63 instance coordinateZeroLocalRing_isDiscreteValuationRing :
64     IsDiscreteValuationRing (CoordinateZeroLocalRing k) := by
65   apply IsLocalization.AtPrime.isDiscreteValuationRing_of_dedekind_domain
66     (Polynomial k) (P := coordinateZeroPrime k)
67   · rw [coordinateZeroPrime, ne_eq, Ideal.span_singleton_eq_bot]
68     exact Polynomial.X_ne_zero

70 /-- Evaluation at a transcendental element extends from `k[X]` to the local
71 ring at `(X)`, because every polynomial outside `(X)` is nonzero and hence a
72 unit in the ambient field. -/
73 def coordinateZeroLocalMap
74     {K : Type v} [Field K] [Algebra k K]
75     (x : K) (hx : Transcendental k x) :
76     CoordinateZeroLocalRing k →+* K :=
77   IsLocalization.lift (S := CoordinateZeroLocalRing k)
78     (M := (coordinateZeroPrime k).primeCompl)
79     (g := (Polynomial.aeval x).toRingHom) fun y ↦ by
80       rw [isUnit_iff_ne_zero]
81       intro hy
82       have hy0 : (y : Polynomial k) = 0 := by
83         apply transcendental_iff_injective.mp hx
84         simpa using hy
85       exact y.2 (hy0 ▸ (coordinateZeroPrime k).zero_mem)

88 theorem coordinateZeroLocalMap_algebraMap_X
89     {K : Type v} [Field K] [Algebra k K]
90     (x : K) (hx : Transcendental k x) :
91     coordinateZeroLocalMap k x hx
92         (algebraMap k[X] (CoordinateZeroLocalRing k) Polynomial.X) = x := by
93   rw [coordinateZeroLocalMap, IsLocalization.lift_eq]
94   exact Polynomial.aeval_X x

96 /-- The coordinate-zero map embeds the source local DVR. The relative
97 finite-type extension is constructed in
98 `RelativeRetainedBoundaryPlace.exists_data_of_fg_charZero`. -/
99 theorem coordinateZeroLocalMap_injective
100     {K : Type v} [Field K] [Algebra k K]
101     (x : K) (hx : Transcendental k x) :
102     Function.Injective (coordinateZeroLocalMap k x hx) := by
103   rw [coordinateZeroLocalMap, IsLocalization.lift_injective_iff]
104   intro a b
105   have hloc := IsLocalization.injective (CoordinateZeroLocalRing k)
106     (coordinateZeroPrime k).primeCompl_le_nonZeroDivisors
107   have heval := transcendental_iff_injective.mp hx
108   constructor
109   · intro hab
110     exact congrArg (Polynomial.aeval x) (hloc hab)
111   · intro hab
112     exact congrArg (algebraMap (Polynomial k) (CoordinateZeroLocalRing k))
113       (heval hab)

115 /-- The source local line at `x = 0` is a DVR and gives the strict order gap
116 for `[x:x²]`. The relative extension to the full finitely generated field,
117 with its coefficient field retained, is provided by
118 `RelativeRetainedBoundaryPlace.exists_data_of_fg_charZero`. -/
119 theorem exists_coordinateZeroDVR_projectiveOrderGap
120     {K : Type v} [Field K] [Algebra k K]
121     (x : K) (hx : Transcendental k x) :
122     let R := CoordinateZeroLocalRing k
123     let q : R := algebraMap (Polynomial k) R Polynomial.X
124     ∃ (uniformizer : R) (a r b : ℕ) (u₀ ur u₁ : Rˣ),
125       coordinateZeroLocalMap k x hx q = x ∧
126       Irreducible uniformizer ∧
127       0 < a ∧ 0 < r ∧ b = a + r ∧ a < b ∧
128       q = (u₀ : R) * uniformizer ^ a ∧
129       q = (ur : R) * uniformizer ^ r ∧
130       u₁ = u₀ * ur ∧
131       q * q = (u₁ : R) * uniformizer ^ b := by
132   let R := CoordinateZeroLocalRing k
133   let q : R := algebraMap (Polynomial k) R Polynomial.X
134   have hq_ne : q ≠ 0 := by
135     intro hq
136     apply Polynomial.X_ne_zero (R := k)
137     apply IsLocalization.injective R
138       (coordinateZeroPrime k).primeCompl_le_nonZeroDivisors
139     simpa only [q, map_zero] using hq
140   have hq_vanish : q ∈ maximalIdeal R := by
141     exact (IsLocalization.AtPrime.to_map_mem_maximal_iff
142       R (coordinateZeroPrime k) Polynomial.X).2
143         (Ideal.mem_span_singleton_self Polynomial.X)
144   obtain ⟨uniformizer, huniformizer⟩ :=
145     IsDiscreteValuationRing.exists_irreducible R
146   obtain ⟨a, r, b, u₀, ur, u₁, ha, hr, hb, hab,
147       hq₀, hratio, hu₁, hq₁⟩ :=
148     ProjectiveDivisorOrderGap.exists_uniformizer_strict_orderGap
149       uniformizer huniformizer q q (q * q)
150       hq_ne hq_ne hq_vanish hq_vanish rfl
151   exact ⟨uniformizer, a, r, b, u₀, ur, u₁,
152     coordinateZeroLocalMap_algebraMap_X k x hx, huniformizer,
153     ha, hr, hb, hab, hq₀, hratio, hu₁, hq₁⟩

155 /-- The complete scheme-free boundary-place package: a valuation subring of
156 the ambient function field together with a *local* factorization of
157 `k[X]_(X) → K`.  Locality records that the centre really lies over `(X)`;
158 `factor_commutes` records that no abstract replacement of the coordinate map
159 has been made. -/
160 structure BoundaryValuationData
161     {K : Type v} [Field K] [Algebra k K]
162     (x : K) (hx : Transcendental k x) where
163   valuation : ValuationSubring K
164   factor : CoordinateZeroLocalRing k →+* valuation.toSubring
165   factor_isLocal : IsLocalHom factor
166   factor_commutes :
167     valuation.toSubring.subtype.comp factor = coordinateZeroLocalMap k x hx
168 
169 /-- A rank-one refinement of the coordinate boundary valuation. This
170 structure records a DVR centered at the chosen nonzero coordinate; existence
171 under finite-type characteristic-zero hypotheses is constructed by
172 `RelativeRetainedBoundaryPlace.exists_data_of_fg_charZero`. -/
173 structure DiscreteBoundaryRefinement
174     {K : Type v} [Field K] [Algebra k K]
175     (x : K) where
176   valuation : ValuationSubring K
177   isDiscrete : IsDiscreteValuationRing valuation.toSubring
178   coordinate : valuation.toSubring
179   coordinate_eq : (coordinate : K) = x
180   coordinate_ne : coordinate ≠ 0
181   coordinate_nonunit : ¬IsUnit coordinate
182 
183 /-- Zorn's valuation-ring theorem produces the complete local factorization
184 of the coordinate-zero local ring inside the function field. -/
185 theorem exists_boundaryValuationData
186     {K : Type v} [Field K] [Algebra k K]
187     (x : K) (hx : Transcendental k x) :
188     Nonempty (BoundaryValuationData k x hx) := by
189   let f := coordinateZeroLocalMap k x hx
190   obtain ⟨V, hV, hlocal⟩ := IsLocalRing.exists_factor_valuationRing f
191   exact ⟨{
192     valuation := V
193     factor := f.codRestrict V.toSubring hV
194     factor_isLocal := hlocal
195     factor_commutes := rfl
196   }⟩

198 /-- A supplied discrete boundary refinement yields the strict order gap for
199 the rational projective pair `[x:x²]`. The finite-type construction of such
200 retained places is in `RelativeRetainedBoundaryPlace`. -/
201 theorem exists_projectiveOrderGap_of_discreteBoundaryRefinement
202     {K : Type v} [Field K] [Algebra k K]
203     (x : K) (D : DiscreteBoundaryRefinement k x) :
204     let R := D.valuation.toSubring
205     ∃ (uniformizer : R) (a r b : ℕ) (u₀ ur u₁ : Rˣ),
206       Irreducible uniformizer ∧
207       0 < a ∧ 0 < r ∧ b = a + r ∧ a < b ∧
208       D.coordinate = (u₀ : R) * uniformizer ^ a ∧
209       D.coordinate = (ur : R) * uniformizer ^ r ∧
210       u₁ = u₀ * ur ∧
211       D.coordinate * D.coordinate = (u₁ : R) * uniformizer ^ b := by
212   let R := D.valuation.toSubring
213   letI : IsDiscreteValuationRing R := D.isDiscrete
214   obtain ⟨uniformizer, huniformizer⟩ :=
215     IsDiscreteValuationRing.exists_irreducible R
216   have hvanish : D.coordinate ∈ maximalIdeal R := by
217     rw [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff]
218     exact D.coordinate_nonunit
219   obtain ⟨a, r, b, u₀, ur, u₁, ha, hr, hb, hab,
220       hq₀, hratio, hu₁, hq₁⟩ :=
221     ProjectiveDivisorOrderGap.exists_uniformizer_strict_orderGap
222       uniformizer huniformizer
223       D.coordinate D.coordinate (D.coordinate * D.coordinate)
224       D.coordinate_ne D.coordinate_ne hvanish hvanish rfl
225   exact ⟨uniformizer, a, r, b, u₀, ur, u₁, huniformizer,
226     ha, hr, hb, hab, hq₀, hratio, hu₁, hq₁⟩

228 /-- A nonconstant coordinate has a scheme-free boundary valuation.
229 
230 The valuation subring is obtained by dominating `k[X]_(X)` inside `K`.
231 The first conclusion says that the chosen coordinate lies at the centre;
232 the second says its inverse has a pole there. -/
233 theorem exists_boundaryValuationSubring
234     {K : Type v} [Field K] [Algebra k K]
235     (x : K) (hx : Transcendental k x) :
236     ∃ V : ValuationSubring K,
237       ∃ xV : V.toSubring,
238         (xV : K) = x ∧
239         x⁻¹ ∉ V ∧
240         ¬IsUnit xV := by
241   let R := CoordinateZeroLocalRing k
242   let f : R →+* K := coordinateZeroLocalMap k x hx
243   obtain ⟨D⟩ := exists_boundaryValuationData k x hx
244   let V := D.valuation
245   let fV : R →+* V.toSubring := D.factor
246   letI hlocalV : IsLocalHom fV := D.factor_isLocal
247   let xR : R := algebraMap k[X] R Polynomial.X
248   have hxR : xR ∈ maximalIdeal R := by
249     exact (IsLocalization.AtPrime.to_map_mem_maximal_iff
250       R (coordinateZeroPrime k) Polynomial.X).2
251         (Ideal.mem_span_singleton_self Polynomial.X)
252   have hfx : f xR = x := by
253     exact coordinateZeroLocalMap_algebraMap_X k x hx
254   let xV : V.toSubring := fV xR
255   have hxV : (xV : K) = x := by
256     have hcomm := D.factor_commutes
257     have := DFunLike.congr_fun hcomm xR
258     exact this.trans hfx
259   have hxR_nonunit : ¬IsUnit xR := mem_nonunits_iff.mp hxR
260   have hxnonunit : ¬IsUnit xV := by
261     intro hu
262     apply hxR_nonunit
263     exact @IsLocalHom.map_nonunit _ _ _ _ _ _ fV hlocalV xR hu
264   have hxinv : x⁻¹ ∉ V := by
265     intro hxinvV
266     have hxunit : IsUnit xV := by
267       apply isUnit_iff_exists_inv.mpr
268       refine ⟨⟨x⁻¹, hxinvV⟩, ?_⟩
269       apply Subtype.ext
270       change (xV : K) * x⁻¹ = (1 : K)
271       rw [hxV]
272       exact mul_inv_cancel₀ (by
273         intro hx0
274         apply hx
275         simpa [hx0] using (isAlgebraic_zero : IsAlgebraic k (0 : K)))
276     exact hxnonunit hxunit
277   exact ⟨V, xV, hxV, hxinv, hxnonunit⟩

279 /-- Algebraically closed form: a nonconstant coordinate directly produces
280 the boundary valuation, with no separately supplied transcendence proof. -/
281 theorem exists_boundaryValuationSubring_of_nonconstant
282     {K : Type v} [Field K] [Algebra k K] [IsAlgClosed k]
283     (x : K) (hx : x ∉ Set.range (algebraMap k K)) :
284     ∃ V : ValuationSubring K,
285       ∃ xV : V.toSubring,
286         (xV : K) = x ∧
287         x⁻¹ ∉ V ∧
288         ¬IsUnit xV :=
289   exists_boundaryValuationSubring k x
290     (transcendental_of_not_mem_range_algebraMap k hx)

292 /-- Coordinate-ring form of the construction.  If an affine coordinate is
293 a unit (the algebraic meaning of avoiding its zero hyperplane) and remains
294 nonconstant in a function field, then there is a boundary valuation which
295 contains the coordinate but excludes the image of its *regular inverse*.
296 
297 For a prime affine ideal one applies this to its quotient domain and its
298 fraction field; the unit is supplied by `I + (x) = ⊤`. -/
299 theorem exists_boundaryValuationSubring_of_affineUnit
300     [IsAlgClosed k]
301     {A : Type*} [CommRing A] [IsDomain A] [Algebra k A]
302     {K : Type v} [Field K] [Algebra k K]
303     (ι : A →ₐ[k] K) (u : Aˣ)
304     (hu : ι (u : A) ∉ Set.range (algebraMap k K)) :
305     ∃ V : ValuationSubring K,
306       ∃ xV : V.toSubring,
307         (xV : K) = ι (u : A) ∧
308         ι (u⁻¹ : Aˣ) ∉ V ∧
309         ¬IsUnit xV := by
310   obtain ⟨V, xV, hxV, hxinv, hxnonunit⟩ :=
311     exists_boundaryValuationSubring_of_nonconstant k (ι (u : A)) hu
312   refine ⟨V, xV, hxV, ?_, hxnonunit⟩
313   have hprod : ι (u : A) * ι (u⁻¹ : Aˣ) = 1 := by
314     rw [← map_mul]
315     simp
316   have hinv : ι (u⁻¹ : Aˣ) = (ι (u : A))⁻¹ :=
317     eq_inv_of_mul_eq_one_right hprod
318   rw [hinv]
319   exact hxinv
