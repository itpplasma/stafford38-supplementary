37 /-- Convert the relevant fields of the full smooth local axis-lift output into
38 the exact paper-facing direct-summand input.  `hCB` and `haxis` are output by
39 the lift; `hqzero` and `hu₀` are its zeroth-coordinate factorization and unit
40 coefficient.  The dimensions are independent: the ambient projective space
41 has dimension `n`, while the `d+2` split columns give `dimY=d+1`. -/
42 theorem directSummandInput_of_tiltedAxisLiftFields
43     {n d : ℕ}
44     (I : Ideal (MvPolynomial (Fin n) k))
45     (q : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)
46     (α : Fin d → k)
47     (chart : Fin (n + 1)) (axis : Fin n)
48     (a : ℕ)
49     (u₀ : MvPowerSeries (Fin (d + 1)) k)
50     (hqchart : q chart = 1)
51     (hqzero : tiltedArc (k := k) α (q 0) =
52       (PowerSeries.X : PowerSeries k) ^ a * tiltedArc (k := k) α u₀)
53     (hu₀ : PowerSeries.constantCoeff (tiltedArc (k := k) α u₀) ≠ 0)
54     (tau : Fin (n + 1) → PowerSeries k)
55     (C : Matrix (FormalTangentColumn (Fin d)) (Fin (n + 1)) (PowerSeries k))
56     (hCB : C * formalTangentMatrix
57       (fun i => tiltedArc (k := k) α (q i))
58       (tiltedTransverseDerivativeMatrix (k := k) α q) tau = 1)
59     (haxis : ∀ j, PowerSeries.constantCoeff
60       (formalTangentMatrix
61         (fun i => tiltedArc (k := k) α (q i))
62         (tiltedTransverseDerivativeMatrix (k := k) α q) tau axis.succ j) = 0)
63     (hprime : I.IsPrime)
64     (hclosure : dehomogenizedPoint
65       (laurentColumn (fun i => tiltedArc (k := k) α (q i))) ∈
66       MvPolynomial.zeroLocus (LaurentSeries k)
67         (I.map (scalarPolynomialMap
68           (k := k) (K := LaurentSeries k) (Fin n))))
69     (hsmooth : SmoothAffinePoint
70       (I.map (scalarPolynomialMap
71         (k := k) (K := LaurentSeries k) (Fin n)))
72       (dehomogenizedPoint
73         (laurentColumn (fun i => tiltedArc (k := k) α (q i)))))
74     (hcone : genericFibre (K := LaurentSeries k)
75       (normalizedTangentLattice
76         (fun i => tiltedArc (k := k) α (q i))
77         (tiltedTransverseDerivativeMatrix (k := k) α q) tau) =
78       projectiveTangentCone
79         (laurentColumn (fun i => tiltedArc (k := k) α (q i)))
80         (zariskiTangentSpace
81           (dehomogenizedPoint
82             (laurentColumn (fun i => tiltedArc (k := k) α (q i))))
83           (I.map (scalarPolynomialMap
84             (k := k) (K := LaurentSeries k) (Fin n))))) :
85     Nonempty (DirectSummandInput (dimY := d + 1) I
86       (fun i => tiltedArc (k := k) α (q i))
87       (normalizedTangentLattice
88         (fun i => tiltedArc (k := k) α (q i))
89         (tiltedTransverseDerivativeMatrix (k := k) α q) tau)) := by
90   let qα : Fin (n + 1) → PowerSeries k :=
91     fun i => tiltedArc (k := k) α (q i)
92   let Zα : Matrix (Fin (n + 1)) (Fin d) (PowerSeries k) :=
93     tiltedTransverseDerivativeMatrix (k := k) α q
94   let Lα : Submodule (PowerSeries k) (Fin (n + 1) → PowerSeries k) :=
95     normalizedTangentLattice qα Zα tau
96   have hqchartα : qα chart = 1 := by
97     dsimp [qα]
98     rw [hqchart]
99     simp [tiltedArc]
100   have hq0 : qα 0 ≠ 0 := by
101     rw [show qα 0 =
102       (PowerSeries.X : PowerSeries k) ^ a * tiltedArc (k := k) α u₀ by
103         simpa [qα] using hqzero]
104     apply mul_ne_zero
105     · exact pow_ne_zero a PowerSeries.X_ne_zero
106     · intro hu
107       apply hu₀
108       rw [hu]
109       simp
110   have hq0Laurent : laurentColumn qα 0 ≠ 0 :=
111     laurentColumn_ne_zero_of_ne_zero qα hq0
112   have harc : FormalProjectiveArcInClosure I qα := by
113     refine ⟨⟨chart, ?_⟩, ?_⟩
114     · rw [hqchartα]
115       exact isUnit_one
116     · exact (projectiveClosureAtZero_iff
117         (I.map (scalarPolynomialMap
118           (k := k) (K := LaurentSeries k) (Fin n)))
119         (laurentColumn qα) hq0Laurent).mpr hclosure
120   have hCBα : C * formalTangentMatrix qα Zα tau = 1 := by
121     simpa [qα, Zα] using hCB
122   have haxisα : ∀ j, PowerSeries.constantCoeff
123       (formalTangentMatrix qα Zα tau axis.succ j) = 0 := by
124     simpa [qα, Zα] using haxis
125   have hconeα : genericFibre (K := LaurentSeries k) Lα =
126       projectiveTangentCone (laurentColumn qα)
127         (zariskiTangentSpace (dehomogenizedPoint (laurentColumn qα))
128           (I.map (scalarPolynomialMap
129             (k := k) (K := LaurentSeries k) (Fin n)))) := by
130     simpa [qα, Zα, Lα] using hcone
131   refine ⟨⟨axis,
132     normalizedTangentLattice_isComplemented qα Zα tau C hCBα,
133     ?_, hprime, hq0, harc, hsmooth, hconeα,
134     normalizedTangentLattice_residue_axis qα Zα tau axis haxisα⟩⟩
135   rw [normalizedTangentLattice_finrank qα Zα tau C hCBα]
136   simp [FormalTangentColumn]
137   omega

139 /-- The full local avoidance/lift theorem supplies the selected arc and its
140 split columns.  This wrapper packages them as the existing exact input once
141 the caller supplies the actual affine equations and the two generic
142 geometric assertions on the selected arc.  No second conclusion-carrying
143 record is introduced. -/
144 theorem exists_directSummandInput_of_tilted_local_axis_lift
145     [Infinite k] [CharZero k]
146     {n d : ℕ}
147     (I : Ideal (MvPolynomial (Fin n) k))
148     (q : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)
149     (bad : MvPowerSeries (Fin (d + 1)) k) (hbad : bad ≠ 0)
150     (rows : Fin d ↪ Fin (n + 1))
151     (chart : Fin (n + 1)) (axis : Fin n)
152     (a b : ℕ)
153     (u₀ u₁ : MvPowerSeries (Fin (d + 1)) k)
154     (hqchart : q chart = 1)
155     (ha : 0 < a) (hab : a < b)
156     (hqzero : q 0 = (MvPowerSeries.X 0) ^ a * u₀)
157     (hu₀ : MvPowerSeries.constantCoeff u₀ ≠ 0)
158     (hqaxis : q axis.succ = (MvPowerSeries.X 0) ^ b * u₁)
159     (hu₁ : MvPowerSeries.constantCoeff u₁ ≠ 0)
160     (hminor : MvPowerSeries.constantCoeff
161       (selectedMinor (localTransverseDerivativeMatrix (k := k) q) rows).det ≠ 0)
162     (hprime : I.IsPrime)
163     (hclosure : ∀ α : Fin d → k, tiltedArc (k := k) α bad ≠ 0 →
164       dehomogenizedPoint
165           (laurentColumn (fun i => tiltedArc (k := k) α (q i))) ∈
166         MvPolynomial.zeroLocus (LaurentSeries k)
167           (I.map (scalarPolynomialMap
168             (k := k) (K := LaurentSeries k) (Fin n))))
169     (hsmooth : ∀ α : Fin d → k,
170       tiltedArc (k := k) α bad ≠ 0 →
171       SmoothAffinePoint
172         (I.map (scalarPolynomialMap
173           (k := k) (K := LaurentSeries k) (Fin n)))
174         (dehomogenizedPoint
175           (laurentColumn (fun i => tiltedArc (k := k) α (q i)))))
176     (hcone : ∀ (α : Fin d → k), tiltedArc (k := k) α bad ≠ 0 →
177       ∀ (lambda : Fin d → PowerSeries k) (c : ℕ)
178         (tau : Fin (n + 1) → PowerSeries k)
179         (C : Matrix (FormalTangentColumn (Fin d))
180           (Fin (n + 1)) (PowerSeries k)),
181       (∀ i, PowerSeries.derivative (R := k) (tiltedArc (k := k) α (q i)) -
182           (tiltedTransverseDerivativeMatrix (k := k) α q).mulVec lambda i =
183         (PowerSeries.X : PowerSeries k) ^ c * tau i) →
184       C * formalTangentMatrix
185           (fun i => tiltedArc (k := k) α (q i))
186           (tiltedTransverseDerivativeMatrix (k := k) α q) tau = 1 →
187       (∀ j, PowerSeries.constantCoeff
188         (formalTangentMatrix
189           (fun i => tiltedArc (k := k) α (q i))
190           (tiltedTransverseDerivativeMatrix (k := k) α q) tau
191           axis.succ j) = 0) →
192       genericFibre (K := LaurentSeries k)
193           (normalizedTangentLattice
194             (fun i => tiltedArc (k := k) α (q i))
195             (tiltedTransverseDerivativeMatrix (k := k) α q) tau) =
196         projectiveTangentCone
197           (laurentColumn (fun i => tiltedArc (k := k) α (q i)))
198           (zariskiTangentSpace
199             (dehomogenizedPoint
200               (laurentColumn (fun i => tiltedArc (k := k) α (q i))))
201             (I.map (scalarPolynomialMap
202               (k := k) (K := LaurentSeries k) (Fin n))))) :
203     ∃ α : Fin d → k, ∃ tau : Fin (n + 1) → PowerSeries k,
204       Nonempty (DirectSummandInput (dimY := d + 1) I
205         (fun i => tiltedArc (k := k) α (q i))
206         (normalizedTangentLattice
207           (fun i => tiltedArc (k := k) α (q i))
208           (tiltedTransverseDerivativeMatrix (k := k) α q) tau)) := by
209   obtain ⟨α, hbadArc, hqzeroArc, hu₀Arc, _hqaxisArc, _hu₁Arc,
210       _hminorArc, _lambda, _c, tau, C, _ell, _hlambda, _hselected,

216   exact ⟨α, tau,
217     directSummandInput_of_tiltedAxisLiftFields
