10 /-!
11 # Divisorial visible frames for arbitrary prime affine components
12 
13 An invertible, transcendental coordinate on a prime affine component gives
14 a normalized visible divisor frame. This removes the canonical Weyl-support
15 hypotheses from the boundary producer. Identification of the resulting Laurent
16 direction with the smooth projective conormal closure is performed downstream
17 in the general asymptotic-conormal construction.
18 -/

53 and the same visible frame, under the ground algebra structure induced by the
54 retained coefficient map. -/

338 /-- The original interface is the projection of the richer Lane-C witness. -/
339 theorem generalDivisorialVisibleFrameExistence
340     {k : Type u} [Field k] [CharZero k]
341     {m : ℕ} (hm : 0 < m)
342     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
343     (hunit : ∃ g : MvPolynomial (Fin m) k,
344       MvPolynomial.X ⟨0, hm⟩ * g - 1 ∈ P.asIdeal)
345     (htrans : Transcendental k
346       (componentCoordinate P ⟨0, hm⟩)) :
347     HasNormalizedCompatibleVisibleFrame P hm := by
348   obtain ⟨w⟩ :=
349     generalDivisorialVisibleFrameWithResidueAlgebraicity hm P hunit htrans
350   let C := w.column
351   let F := w.differential.core
352   letI : Algebra (CoordinateZeroLocalRing C.W.coefficientField)
353       (ComponentFractionField P) := C.W.ambientAlgebra
354   let V := C.W.place.valuation.toSubring
355   letI : IsDiscreteValuationRing V := C.W.place.isDiscrete
356   exact ⟨C.W, C.chart, C.q, C.scale, C.scale_ne, C.chart_one, C.q0_ne,
357     C.q_commonScale, C.q_parameter,
358     ⟨F.D, F.D_Q0, F.D_Q1, F.D_Q⟩⟩
