8 /-!
9 # Ground coefficients in the retained completed chart
10 
11 The retained DVR construction carries a canonical ground-field map into its
12 residue field.  With that map chosen as the `Algebra` structure, the explicit
13 coefficient map obtained by passing through the valuation ring, its completion,
14 and Laurent series is exactly the ground map used by the terminal geometric
15 consumer.
16 
17 The choice of `Algebra` structure is part of the statement.  No equality with
18 an unrelated ground-field embedding of the residue field is asserted.
19 -/

46 /-- The ground-field structure on the retained residue field induced by the
47 actual coefficient map through the valuation ring. -/
48 def retainedResidueGroundAlgebra
49     [CharZero k]
50     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
51     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
52     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
53         (ComponentFractionField P) := W.ambientAlgebra
54     let V := W.place.valuation.toSubring
55     letI : IsDiscreteValuationRing V := W.place.isDiscrete
56     letI : Algebra W.coefficientField V :=
57       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
58     Algebra k (ResidueField V) := by
59   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
60       (ComponentFractionField P) := W.ambientAlgebra
61   let V := W.place.valuation.toSubring
62   letI : IsDiscreteValuationRing V := W.place.isDiscrete
63   letI : Algebra W.coefficientField V :=
64     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
65   exact ((residue V).comp (retainedComponentCoefficientMap P i W)).toAlgebra

67 /-- Ground coefficients transported through the retained valuation ring become
68 the corresponding constant power series for the induced residue-field map. -/
69 theorem retainedToCompletedPowerSeries_ground
70     [CharZero k]
71     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
72     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
73     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
74         (ComponentFractionField P) := W.ambientAlgebra
75     let V := W.place.valuation.toSubring
76     letI : IsDiscreteValuationRing V := W.place.isDiscrete
77     letI : Algebra W.coefficientField V :=
78       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
79     letI : Algebra k (ResidueField V) :=
80       retainedResidueGroundAlgebra P i W
81     ∀ c : k,
82       retainedToCompletedPowerSeries W
83           (retainedComponentCoefficientMap P i W c) =
84         PowerSeries.C (R := ResidueField V)
85           (algebraMap k (ResidueField V) c) := by
86   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
87       (ComponentFractionField P) := W.ambientAlgebra
88   let V := W.place.valuation.toSubring
89   letI : IsDiscreteValuationRing V := W.place.isDiscrete
90   letI : Algebra W.coefficientField V :=
91     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
92   letI : Algebra k (ResidueField V) :=
93     retainedResidueGroundAlgebra P i W
94   dsimp only
95   intro c
96   change W.completedPowerSeriesEquiv.symm
97       (algebraMap V (AdicCompletion (maximalIdeal V) V)
98         (relativeCoefficientMap W.coefficientField W.place
99           (algebraMap k W.coefficientField c))) = _
100   apply W.completedPowerSeriesEquiv.toEquiv.symm_apply_eq.mpr
101   change algebraMap V (AdicCompletion (maximalIdeal V) V)
102       (relativeCoefficientMap W.coefficientField W.place
103         (algebraMap k W.coefficientField c)) =
104     completedDVRPowerSeriesMap W.coefficientField V
105         (relativeResidue_isSeparable W.coefficientField W.place)
106       (PowerSeries.C (R := ResidueField V)
107         (residue V
108           (relativeCoefficientMap W.coefficientField W.place
109             (algebraMap k W.coefficientField c))))
110   rw [completedDVRPowerSeriesMap_C]
111   change algebraMap V (AdicCompletion (maximalIdeal V) V)
112       (algebraMap W.coefficientField V
113         (algebraMap k W.coefficientField c)) =
114     completedCoefficientSection W.coefficientField V
115       (relativeResidue_isSeparable W.coefficientField W.place)
116       (algebraMap W.coefficientField (ResidueField V)
117         (algebraMap k W.coefficientField c))
118   calc
119     algebraMap V (AdicCompletion (maximalIdeal V) V)
120         (algebraMap W.coefficientField V
121           (algebraMap k W.coefficientField c)) =
122       algebraMap W.coefficientField
123           (AdicCompletion (maximalIdeal V) V)
124           (algebraMap k W.coefficientField c) :=
125         IsScalarTower.algebraMap_apply W.coefficientField V
126           (AdicCompletion (maximalIdeal V) V)
127           (algebraMap k W.coefficientField c)
128     _ = completedCoefficientSection W.coefficientField V
129         (relativeResidue_isSeparable W.coefficientField W.place)
130         (algebraMap W.coefficientField (ResidueField V)
131           (algebraMap k W.coefficientField c)) :=
132       ((completedCoefficientSection W.coefficientField V
133         (relativeResidue_isSeparable W.coefficientField W.place)).commutes
134           (algebraMap k W.coefficientField c)).symm

136 /-- With the residue-induced `Algebra` structure, the explicit retained
137 Laurent coefficient map agrees with the canonical ground map through the
138 checked coefficient-tower compatibility. -/
139 theorem retainedLaurentCoefficientMap_eq_groundLaurentMap
140     [CharZero k]
141     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
142     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
143     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
144         (ComponentFractionField P) := W.ambientAlgebra
145     let V := W.place.valuation.toSubring
146     letI : IsDiscreteValuationRing V := W.place.isDiscrete
147     letI : Algebra W.coefficientField V :=
148       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
149     letI : Algebra k (ResidueField V) :=
150       retainedResidueGroundAlgebra P i W
151     retainedLaurentCoefficientMap P i W =
152       groundLaurentMap (k := k) (K := ResidueField V) := by
153   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
154       (ComponentFractionField P) := W.ambientAlgebra
155   let V := W.place.valuation.toSubring
156   letI : IsDiscreteValuationRing V := W.place.isDiscrete
157   letI : Algebra W.coefficientField V :=
158     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
159   letI : Algebra k (ResidueField V) :=
160     retainedResidueGroundAlgebra P i W
161   apply RingHom.ext
162   intro c
163   rw [← groundLaurentComposite_eq_algebraMap (k := k) (K := ResidueField V)]
164   change algebraMap (PowerSeries (ResidueField V))
165       (LaurentSeries (ResidueField V))
166       (retainedToCompletedPowerSeries W
167         (retainedComponentCoefficientMap P i W c)) =
168     algebraMap (ResidueField V) (LaurentSeries (ResidueField V))
169       (algebraMap k (ResidueField V) c)
170   rw [retainedToCompletedPowerSeries_ground P i W]
171   rfl

173 /-- The finite retained equation package therefore has exactly the coefficient
174 map required by the terminal residue-extension chart consumer. -/
175 theorem retainedComponentGroundEquationPackage
176     [CharZero k]
177     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
178     (I : Ideal (MvPolynomial (Fin m) k)) (hIP : I ≤ P.asIdeal)
179     (i : Fin m)
180     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
181     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
182         (ComponentFractionField P) := W.ambientAlgebra
183     let V := W.place.valuation.toSubring
184     letI : IsDiscreteValuationRing V := W.place.isDiscrete
185     letI : Algebra W.coefficientField V :=
186       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
187     letI : Algebra k (ResidueField V) :=
188       retainedResidueGroundAlgebra P i W
189     ∀ (q : Fin (m + 1) → V) (scale : ComponentFractionField P),
190       (∀ a, (q a : ComponentFractionField P) =
191         scale * componentProjectivePoint P a) →
192       Nonempty (EquationPackage I
193         (groundLaurentMap (k := k) (K := ResidueField V))
194         (fun a ↦ algebraMap (PowerSeries (ResidueField V))
195           (LaurentSeries (ResidueField V))
196             (retainedToCompletedPowerSeries W (q a)))) := by
197   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
198       (ComponentFractionField P) := W.ambientAlgebra
199   let V := W.place.valuation.toSubring
200   letI : IsDiscreteValuationRing V := W.place.isDiscrete
201   letI : Algebra W.coefficientField V :=
202     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
203   letI : Algebra k (ResidueField V) :=
204     retainedResidueGroundAlgebra P i W
205   dsimp only
206   intro q scale hq
207   simpa only [retainedLaurentCoefficientMap_eq_groundLaurentMap P i W] using
208     (retainedComponentEquationPackage P I hIP i W q scale hq)
