7 /-!
8 # Coisotropic sets and their vanishing ideals
9 
10 Closed affine sets are represented as common zero loci of polynomial ideals.
11 The vanishing ideal is radical, and its base contraction describes the base
12 projection of a closed fibre-conical set.
13 -/

32 abbrev CoordinateAxisClosureInput {k : Type*} [Field k] [IsAlgClosed k]
33     [CharZero k] :=
34   ∀ {m : ℕ} (hm : 0 < m) (I : PrimeSpectrum (MvPolynomial (Fin m) k)),
35     (∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) →
36     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
37       MvPolynomial.zeroLocus k
38         (MvPolynomial.vanishingIdeal k
39           (smoothConormalFibreProjection I.asIdeal))

41 theorem vanishingIdeal_isRadical (W : Set (PhaseVar n → k)) :
42     (MvPolynomial.vanishingIdeal k W).IsRadical := by
43   intro p hp
44   obtain ⟨m, hm⟩ := Ideal.mem_radical_iff.mp hp
45   intro q hq
46   have h := hm q hq
47   simpa only [map_pow] using eq_zero_of_pow_eq_zero
48     (show (MvPolynomial.aeval q p) ^ m = 0 by simpa only [map_pow] using h)

50 theorem zeroLocus_vanishingIdeal_of_algebraic_closed
51     (W : Set (PhaseVar n → k))
52     (hclosed : ∃ L : Ideal (SymbolRing k n), W = MvPolynomial.zeroLocus k L) :
53     MvPolynomial.zeroLocus k (MvPolynomial.vanishingIdeal k W) = W := by
54   obtain ⟨L, rfl⟩ := hclosed
55   apply Set.Subset.antisymm
56   · exact MvPolynomial.zeroLocus_anti_mono (MvPolynomial.le_vanishingIdeal_zeroLocus L)
57   · exact MvPolynomial.zeroLocus_vanishingIdeal_le _

59 theorem baseProjection_eq_zeroLocus_comap
60     (W : Set (PhaseVar n → k))
61     (hclosed : ∃ L : Ideal (SymbolRing k n), W = MvPolynomial.zeroLocus k L)
62     (hhom : (MvPolynomial.vanishingIdeal k W).IsHomogeneous
63       (orderDecomposition (k := k) (n := n))) :
64     {y : Fin n → k | ∃ ξ : Fin n → k, Sum.elim y ξ ∈ W} =
65       MvPolynomial.zeroLocus k ((MvPolynomial.vanishingIdeal k W).comap baseLift) := by
66   ext y
67   constructor
68   · rintro ⟨ξ, hξ⟩ f hf
69     have h := hf (Sum.elim y ξ) hξ
70     simpa [baseLift, MvPolynomial.eval_rename, Function.comp_def] using h
71   · intro hy
72     refine ⟨0, ?_⟩
73     rw [← zeroLocus_vanishingIdeal_of_algebraic_closed W hclosed]
74     have hzero := zeroSection_commonZero_of_isHomogeneous
75       (MvPolynomial.vanishingIdeal k W) hhom y hy
76     have hpoint : Sum.elim y (fun _ : Fin n => (0 : k)) = zeroSectionPoint y := by
77       funext i
78       rcases i with i | i <;> rfl
79     change Sum.elim y (fun _ : Fin n => (0 : k)) ∈
80       MvPolynomial.zeroLocus k (MvPolynomial.vanishingIdeal k W)
81     rw [hpoint]
82     exact hzero

84 /-- For every minimal component of the base projection, the closure of its
85 smooth conormal bundle lies in the original closed fibre-conical coisotropic
86 set.  This is the set-level form of the component containment used in the
87 manuscript. -/
88 theorem smoothConormalClosure_minimalPrime_subset_of_isFibreConical
89     (W : Set (PhaseVar n → k))
90     (hclosed : ∃ L : Ideal (SymbolRing k n),
91       W = MvPolynomial.zeroLocus k L)
92     (hW : FibreConicalVanishingIdeal.IsFibreConical W)
93     (hpoisson : ∀ f ∈ MvPolynomial.vanishingIdeal k W,
94       ∀ g ∈ MvPolynomial.vanishingIdeal k W,
95         poissonBracket f g ∈ MvPolynomial.vanishingIdeal k W)
96     (P : Ideal (MvPolynomial (Fin n) k))
97     (hP : P ∈ ((MvPolynomial.vanishingIdeal k W).comap baseLift).minimalPrimes) :
98     MvPolynomial.zeroLocus k
99         (MvPolynomial.vanishingIdeal k
100           (restrictedEquationConormalLocus P {y | SmoothAffinePoint P y})) ⊆ W := by
101   let J : Ideal (SymbolRing k n) := MvPolynomial.vanishingIdeal k W
102   have hrad : J.IsRadical := by
103     simpa only [J] using vanishingIdeal_isRadical W
104   have hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := n)) := by
105     simpa only [J] using
106       FibreConicalVanishingIdeal.vanishingIdeal_isHomogeneous_of_isFibreConical W hW
107   have hbasePoisson : IsBaseRelativePoisson J := by
108     intro f hf g hg
109     exact hpoisson (baseLift f) hf g hg
110   rw [← zeroLocus_vanishingIdeal_of_algebraic_closed W hclosed]
111   exact smoothConormalClosure_minimalPrime_subset_zeroLocus
112     J hrad hhom hbasePoisson P hP

114 /-- A nonempty closed fibre-conical coisotropic set cannot avoid the distinguished
115 base-coordinate hyperplane while lying in a fibre polynomial hypersurface
116 transverse to the distinguished axis.
117 
118 The ideal used by the argument is the actual vanishing ideal of `W`; its
119 radicality is automatic, fibre homogeneity is derived from `hW`, and
120 coisotropy means the exact self-involutivity condition `{I(W), I(W)} ⊆ I(W)`. -/
121 theorem exists_zero_base_coordinate_of_isFibreConical_using
122     {k : Type*} [Field k] [IsAlgClosed k] [CharZero k]
123     (axisClosure : CoordinateAxisClosureInput (k := k))
124     {m : ℕ} (hm : 0 < m)
125     (W : Set (PhaseVar m → k))
126     (hnonempty : W.Nonempty)
127     (hclosed : ∃ L : Ideal (SymbolRing k m),
128       W = MvPolynomial.zeroLocus k L)
129     (hW : FibreConicalVanishingIdeal.IsFibreConical W)
130     (hpoisson : ∀ f ∈ MvPolynomial.vanishingIdeal k W,
131       ∀ g ∈ MvPolynomial.vanishingIdeal k W,
132         poissonBracket f g ∈ MvPolynomial.vanishingIdeal k W)
133     (P : MvPolynomial (Fin m) k)
134     (hP : fibreLift P ∈ MvPolynomial.vanishingIdeal k W)
135     (haxis : MvPolynomial.eval
136       (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) P ≠ 0) :
137     ∃ q ∈ W, q (.inl ⟨0, hm⟩) = 0 := by
138   let J : Ideal (SymbolRing k m) := MvPolynomial.vanishingIdeal k W
139   have hproper : J ≠ ⊤ := by
140     intro htop
141     obtain ⟨q, hq⟩ := hnonempty
142     have hqzero : q ∈ MvPolynomial.zeroLocus k J := by
143       exact MvPolynomial.zeroLocus_vanishingIdeal_le W hq
144     rw [htop] at hqzero
145     simpa using hqzero
146   have hrad : J.IsRadical := by
147     simpa only [J] using vanishingIdeal_isRadical W
148   have hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := m)) := by
149     simpa only [J] using
150       FibreConicalVanishingIdeal.vanishingIdeal_isHomogeneous_of_isFibreConical W hW
151   have hbasePoisson : IsBaseRelativePoisson J := by
152     intro f hf g hg
153     exact hpoisson (baseLift f) hf g hg
154   obtain ⟨q, hq, hqcoord⟩ := exists_zero_base_coordinate_using axisClosure
155     hm J hproper hrad hhom hbasePoisson P hP haxis
156   refine ⟨q, ?_, hqcoord⟩
157   rw [← zeroLocus_vanishingIdeal_of_algebraic_closed W hclosed]
158   exact hq
