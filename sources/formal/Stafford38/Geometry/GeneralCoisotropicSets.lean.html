9 /-!
10 # Coisotropic sets and their vanishing ideals
11 
12 Closed affine sets are represented as common zero loci of polynomial ideals.
13 The vanishing ideal is radical, and its base contraction describes the base
14 projection of a closed fibre-conical set.
15 -/

34 abbrev CoordinateAxisClosureInput {k : Type*} [Field k] [IsAlgClosed k]
35     [CharZero k] :=
36   ∀ {m : ℕ} (hm : 0 < m) (I : PrimeSpectrum (MvPolynomial (Fin m) k)),
37     (∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) →
38     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
39       MvPolynomial.zeroLocus k
40         (MvPolynomial.vanishingIdeal k
41           (smoothConormalFibreProjection I.asIdeal))

43 theorem vanishingIdeal_isRadical (W : Set (PhaseVar n → k)) :
44     (MvPolynomial.vanishingIdeal k W).IsRadical := by
45   intro p hp
46   obtain ⟨m, hm⟩ := Ideal.mem_radical_iff.mp hp
47   intro q hq
48   have h := hm q hq
49   simpa only [map_pow] using eq_zero_of_pow_eq_zero
50     (show (MvPolynomial.aeval q p) ^ m = 0 by simpa only [map_pow] using h)

52 theorem zeroLocus_vanishingIdeal_of_algebraic_closed
53     (W : Set (PhaseVar n → k))
54     (hclosed : ∃ L : Ideal (SymbolRing k n), W = MvPolynomial.zeroLocus k L) :
55     MvPolynomial.zeroLocus k (MvPolynomial.vanishingIdeal k W) = W := by
56   obtain ⟨L, rfl⟩ := hclosed
57   apply Set.Subset.antisymm
58   · exact MvPolynomial.zeroLocus_anti_mono (MvPolynomial.le_vanishingIdeal_zeroLocus L)
59   · exact MvPolynomial.zeroLocus_vanishingIdeal_le _

61 theorem baseProjection_eq_zeroLocus_comap
62     (W : Set (PhaseVar n → k))
63     (hclosed : ∃ L : Ideal (SymbolRing k n), W = MvPolynomial.zeroLocus k L)
64     (hhom : (MvPolynomial.vanishingIdeal k W).IsHomogeneous
65       (orderDecomposition (k := k) (n := n))) :
66     {y : Fin n → k | ∃ ξ : Fin n → k, Sum.elim y ξ ∈ W} =
67       MvPolynomial.zeroLocus k ((MvPolynomial.vanishingIdeal k W).comap baseLift) := by
68   ext y
69   constructor
70   · rintro ⟨ξ, hξ⟩ f hf
71     have h := hf (Sum.elim y ξ) hξ
72     simpa [baseLift, MvPolynomial.eval_rename, Function.comp_def] using h
73   · intro hy
74     refine ⟨0, ?_⟩
75     rw [← zeroLocus_vanishingIdeal_of_algebraic_closed W hclosed]
76     have hzero := zeroSection_commonZero_of_isHomogeneous
77       (MvPolynomial.vanishingIdeal k W) hhom y hy
78     have hpoint : Sum.elim y (fun _ : Fin n => (0 : k)) = zeroSectionPoint y := by
79       funext i
80       rcases i with i | i <;> rfl
81     change Sum.elim y (fun _ : Fin n => (0 : k)) ∈
82       MvPolynomial.zeroLocus k (MvPolynomial.vanishingIdeal k W)
83     rw [hpoint]
84     exact hzero

86 /-- For every minimal component of the base projection, the closure of its
87 smooth conormal bundle lies in the original closed fibre-conical coisotropic
88 set.  This is the set-level form of the component containment used in the
89 manuscript. -/
90 theorem smoothConormalClosure_minimalPrime_subset_of_isFibreConical
91     (W : Set (PhaseVar n → k))
92     (hclosed : ∃ L : Ideal (SymbolRing k n),
93       W = MvPolynomial.zeroLocus k L)
94     (hW : FibreConicalVanishingIdeal.IsFibreConical W)
95     (hpoisson : ∀ f ∈ MvPolynomial.vanishingIdeal k W,
96       ∀ g ∈ MvPolynomial.vanishingIdeal k W,
97         poissonBracket f g ∈ MvPolynomial.vanishingIdeal k W)
98     (P : Ideal (MvPolynomial (Fin n) k))
99     (hP : P ∈ ((MvPolynomial.vanishingIdeal k W).comap baseLift).minimalPrimes) :
100     MvPolynomial.zeroLocus k
101         (MvPolynomial.vanishingIdeal k
102           (restrictedEquationConormalLocus P {y | SmoothAffinePoint P y})) ⊆ W := by
103   let J : Ideal (SymbolRing k n) := MvPolynomial.vanishingIdeal k W
104   have hrad : J.IsRadical := by
105     simpa only [J] using vanishingIdeal_isRadical W
106   have hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := n)) := by
107     simpa only [J] using
108       FibreConicalVanishingIdeal.vanishingIdeal_isHomogeneous_of_isFibreConical W hW
109   have hbasePoisson : IsBaseRelativePoisson J := by
110     intro f hf g hg
111     exact hpoisson (baseLift f) hf g hg
112   rw [← zeroLocus_vanishingIdeal_of_algebraic_closed W hclosed]
113   exact smoothConormalClosure_minimalPrime_subset_zeroLocus
114     J hrad hhom hbasePoisson P hP

116 /-- A nonempty closed fibre-conical coisotropic set cannot avoid the distinguished
117 base-coordinate hyperplane while lying in a fibre polynomial hypersurface
118 transverse to the distinguished axis.
119 
120 The ideal used by the argument is the actual vanishing ideal of `W`; its
121 radicality is automatic, fibre homogeneity is derived from `hW`, and
122 coisotropy means the exact self-involutivity condition `{I(W), I(W)} ⊆ I(W)`. -/
123 theorem exists_zero_base_coordinate_of_isFibreConical_using
124     {k : Type*} [Field k] [IsAlgClosed k] [CharZero k]
125     (axisClosure : CoordinateAxisClosureInput (k := k))
126     {m : ℕ} (hm : 0 < m)
127     (W : Set (PhaseVar m → k))
128     (hnonempty : W.Nonempty)
129     (hclosed : ∃ L : Ideal (SymbolRing k m),
130       W = MvPolynomial.zeroLocus k L)
131     (hW : FibreConicalVanishingIdeal.IsFibreConical W)
132     (hpoisson : ∀ f ∈ MvPolynomial.vanishingIdeal k W,
133       ∀ g ∈ MvPolynomial.vanishingIdeal k W,
134         poissonBracket f g ∈ MvPolynomial.vanishingIdeal k W)
135     (P : MvPolynomial (Fin m) k)
136     (hP : fibreLift P ∈ MvPolynomial.vanishingIdeal k W)
137     (haxis : MvPolynomial.eval
138       (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) P ≠ 0) :
139     ∃ q ∈ W, q (.inl ⟨0, hm⟩) = 0 := by
140   let J : Ideal (SymbolRing k m) := MvPolynomial.vanishingIdeal k W
141   have hproper : J ≠ ⊤ := by
142     intro htop
143     obtain ⟨q, hq⟩ := hnonempty
144     have hqzero : q ∈ MvPolynomial.zeroLocus k J := by
145       exact MvPolynomial.zeroLocus_vanishingIdeal_le W hq
146     rw [htop] at hqzero
147     simpa using hqzero
148   have hrad : J.IsRadical := by
149     simpa only [J] using vanishingIdeal_isRadical W
150   have hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := m)) := by
151     simpa only [J] using
152       FibreConicalVanishingIdeal.vanishingIdeal_isHomogeneous_of_isFibreConical W hW
153   have hbasePoisson : IsBaseRelativePoisson J := by
154     intro f hf g hg
155     exact hpoisson (baseLift f) hf g hg
156   obtain ⟨q, hq, hqcoord⟩ := exists_zero_base_coordinate_using axisClosure
157     hm J hproper hrad hhom hbasePoisson P hP haxis
158   refine ⟨q, ?_, hqcoord⟩
159   rw [← zeroLocus_vanishingIdeal_of_algebraic_closed W hclosed]
160   exact hq
