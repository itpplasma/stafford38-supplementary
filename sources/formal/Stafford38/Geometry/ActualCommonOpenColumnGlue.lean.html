23 /-- The actual localization maps identify a column transported from the
24 chart algebra with the same column transported through the point-local ring.
25 The only coordinate equality used here is equality in the original ring `A`;
26 there is no separately assumed common-open/point-local compatibility. -/
27 theorem pointLocal_commonOpen_column
28     {Q : Type u} [CommRing Q]
29     {A : Type v} [CommRing A] [Algebra Q A]
30     {ι : Type z}
31     (M : Ideal A) [M.IsPrime]
32     (f : Q) (e : Localization.Away f ≃ₐ[Q]
33       Localization.Away (algebraMap Q A f)) (g : Q)
34     (qQ : ι → Q) (qA : ι → A)
35     (hA : ∀ i, qA i = algebraMap Q A (qQ i))
36     (qT : ι → Localization.AtPrime M)
37     (hT : ∀ i, qT i = algebraMap A (Localization.AtPrime M) (qA i)) :
38     let Cq := genericOpenRing M f e
39     let U := genericOpenExtraAwayB M f e g
40     letI : Algebra Q Cq := inferInstance
41     letI : Algebra Cq U := inferInstance
42     letI : Algebra Q U := Algebra.compHom U (algebraMap Q Cq)
43     ∀ i, pointLocalToCommonOpen (A := A) M f e g (qT i) =
44       algebraMap Q U (qQ i) := by
45   dsimp
46   let Cq := genericOpenRing M f e
47   let U := genericOpenExtraAwayB M f e g
48   letI : Algebra Q Cq := inferInstance
49   letI : Algebra Cq U := inferInstance
50   letI : Algebra Q U := Algebra.compHom U (algebraMap Q Cq)
51   have hroute := pointLocalToCommonOpen_comp_algebraMap (A := A) M f e g
52   intro i
53   rw [hT i]
54   calc
55     pointLocalToCommonOpen (A := A) M f e g
56         (algebraMap A (Localization.AtPrime M) (qA i)) =
57       genericOpenExtraAwayBMap M f e g (qA i) := by
58         have hi := congrArg (fun φ : A →+* U => φ (qA i)) hroute
59         simpa only [RingHom.comp_apply] using hi
60     _ = genericOpenExtraAwayBMap M f e g (algebraMap Q A (qQ i)) :=
61       congrArg (genericOpenExtraAwayBMap M f e g) (hA i)
62     _ = algebraMap Q U (qQ i) := by
63       change genericOpenExtraAwayBMap M f e g (algebraMap Q A (qQ i)) =
64         algebraMap Cq U (algebraMap Q Cq (qQ i))
65       exact genericOpenExtraAwayBMap_algebraMap M f e g (qQ i)

67 /-- The unique inclusion of an affine chart element into its integral closure
68 is recovered from its value in the common component fraction field. -/
69 theorem actualColumn_eq_chartImage
70     {k F : Type u} [Field k] [Field F] [Algebra k F]
71     (Q : Subalgebra k F) {ι : Type z}
72     (qQ : ι → Q)
73     (qB : ι → integralClosure Q.toSubring F)
74     (v : ι → F)
75     (hqQ : ∀ i, (qQ i : F) = v i)
76     (hqB : ∀ i, ((qB i : integralClosure Q.toSubring F) : F) = v i) :
77     ∀ i, qB i = chartSubalgebraToIntegralClosure Q (qQ i) := by
78   intro i
79   apply Subtype.ext
80   exact (hqB i).trans (hqQ i).symm

82 /-- For one actual integral-closure column, the chart-algebra and point-local
83 routes to the same common open agree. The two F-valued specifications are
84 the existing same-witness producers' outputs; this theorem derives the
85 intervening equality in the integral closure by subtype extensionality. -/
86 theorem actual_integral_column_commonOpen_eq_pointLocal
87     {k F : Type u} [Field k] [Field F] [Algebra k F]
88     (Q : Subalgebra k F) {ι : Type z}
89     (M : Ideal (integralClosure Q.toSubring F)) [M.IsPrime]
90     (f : Q)
91     (e : Localization.Away f ≃ₐ[Q]
92       Localization.Away
93         (algebraMap Q (integralClosure Q.toSubring F) f))
94     (g : Q)
95     (qQ : ι → Q)
96     (qB : ι → integralClosure Q.toSubring F)
97     (v : ι → F)
98     (hqQ : ∀ i, (qQ i : F) = v i)
99     (hqB : ∀ i, ((qB i : integralClosure Q.toSubring F) : F) = v i)
100     (qC : ι → genericOpenExtraAwayB M f e g)
101     (hqC : ∀ i,
102       algebraMap Q (genericOpenExtraAwayB M f e g) (qQ i) = qC i)
103     (qT : ι → Localization.AtPrime M)
104     (hqT : ∀ i,
105       qT i = algebraMap (integralClosure Q.toSubring F)
106         (Localization.AtPrime M) (qB i)) :
107     ∀ i, qC i = pointLocalToCommonOpen
108       (A := integralClosure Q.toSubring F) M f e g (qT i) := by
109   classical
110   let B := integralClosure Q.toSubring F
111   let qToB : Q →+* B := chartSubalgebraToIntegralClosure Q
112   letI : Algebra Q B := qToB.toAlgebra
113   let Cq := genericOpenRing M f e
114   let U := genericOpenExtraAwayB M f e g
115   letI : Algebra Q Cq := inferInstance
116   letI : Algebra Cq U := inferInstance
117   letI : Algebra Q U := Algebra.compHom U (algebraMap Q Cq)
118   have hqB' : ∀ i, qB i = algebraMap Q B (qQ i) := by
119     intro i
120     calc
121       qB i = qToB (qQ i) := actualColumn_eq_chartImage Q qQ qB v hqQ hqB i
122       _ = algebraMap Q B (qQ i) := rfl
123   have hroute := pointLocal_commonOpen_column
124     (A := B) M f e g qQ qB hqB' qT hqT
125   intro i
126   calc
127     qC i = algebraMap Q U (qQ i) := (hqC i).symm
128     _ = pointLocalToCommonOpen (A := B) M f e g (qT i) := (hroute i).symm
129 
130 end
