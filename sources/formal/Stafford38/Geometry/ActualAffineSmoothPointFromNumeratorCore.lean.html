27 /-- A nonzero homogenized numerator at a nonzero projective coordinate gives
28 an actual smooth point of the original affine quotient, after dehomogenizing
29 through the given quotient map. -/
30 theorem smoothAffinePoint_of_homogenized_eval
31     {k L : Type u} [Field k] [IsAlgClosed k] [CharZero k]
32     [Field L] [Algebra k L] {n : ℕ}
33     (I : Ideal (MvPolynomial (Fin n) k)) [I.IsPrime]
34     (p : MvPolynomial (Fin n) k)
35     (fbar : MvPolynomial (Fin n) k ⧸ I)
36     (hrep : Ideal.Quotient.mk I p = fbar)
37     (φ : (MvPolynomial (Fin n) k ⧸ I) →ₐ[k] L)
38     (q : Fin (n + 1) → L)
39     (hdehom : dehomogenizedPoint q =
40       fun i => φ (Ideal.Quotient.mk I (MvPolynomial.X i)))
41     (hq₀ : q 0 ≠ 0)
42     (hsmooth : Algebra.Smooth k (Localization.Away fbar))
43     (hnumerator : MvPolynomial.eval q
44       (MvPolynomial.map (algebraMap k L) (homogenizeAtZero p)) ≠ 0) :
45     SmoothAffinePoint (k := L)
46       (I.map (scalarPolynomialMap (k := k) (K := L) (Fin n)))
47       (dehomogenizedPoint q) := by
48   let φP : MvPolynomial (Fin n) k →ₐ[k] L :=
49     φ.comp (Ideal.Quotient.mkₐ k I)
50   have hφeval : ∀ r : MvPolynomial (Fin n) k,
51       MvPolynomial.eval₂ (algebraMap k L)
52         (fun i => φP (MvPolynomial.X i)) r = φP r := by
53     intro r
54     have hunique := MvPolynomial.aeval_unique φP
55     have hpoint : φP = MvPolynomial.aeval (fun i => φP (MvPolynomial.X i)) := by
56       convert hunique using 1 <;> rfl
57     have heq := congrArg
58       (fun ψ : MvPolynomial (Fin n) k →ₐ[k] L => ψ r) hpoint.symm
59     simpa only [MvPolynomial.aeval_def] using heq
60   have hdehom_eval :
61       MvPolynomial.eval (dehomogenizedPoint q)
62         (MvPolynomial.map (algebraMap k L) p) = φP p := by
63     rw [hdehom]
64     rw [← MvPolynomial.eval₂_eq_eval_map]
65     change MvPolynomial.eval₂ (algebraMap k L)
66       (fun i => φP (MvPolynomial.X i)) p = φP p
67     exact hφeval p
68   have havoid : φ fbar ≠ 0 := by
69     rw [map_homogenizeAtZero] at hnumerator
70     rw [eval_homogenizeAtZero_eq_pow_mul_dehomogenizedPoint
71       (MvPolynomial.map (algebraMap k L) p) q hq₀,
72       hdehom_eval,
73       show φP p = φ (Ideal.Quotient.mk I p) from rfl,
74       hrep] at hnumerator
75     exact (mul_ne_zero_iff.mp hnumerator).2
76   rw [hdehom]
77   exact smoothAffinePoint_of_quotient_point_avoiding_smooth_away
78     (k := k) (L := L) I fbar hsmooth φ havoid

81 end
