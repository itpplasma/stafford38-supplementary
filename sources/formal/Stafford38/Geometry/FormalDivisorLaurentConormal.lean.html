8 /-!
9 # From a formal divisor tangent to a Laurent conormal point
10 
11 This file is the local algebraic bridge between the formal-divisor tangent
12 construction and Laurent conormal specialization.  Projective coordinates,
13 an annihilating row, and the divisor and normalized-transverse tangent columns
14 are embedded from power series into Laurent series.  The position column of
15 the formal tangent matrix supplies the projective Euler relation; its other
16 columns supply the projective tangent annihilation relation.  The existing
17 dehomogenization theorem then gives an equation-conormal point once the base
18 equations and the exact affine tangent-space equality are supplied.
19 
20 No normalization or divisor chart is constructed here.  In particular this
21 file makes no global closure, coisotropy, or Gabber claim.
22 -/

38 /-- Coefficientwise embedding of a power-series projective column into
39 Laurent series. -/
40 def laurentColumn (v : Fin (n + 1) → PowerSeries k) :
41     Fin (n + 1) → LaurentSeries k :=
42   fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (v i)
43 
44 /-- The divisor and normalized-transverse columns of the formal tangent
45 matrix, embedded in Laurent series.  The position column is deliberately
46 omitted: it is used separately as the projective-point Euler relation. -/
47 def laurentNonpositionTangentMatrix { κ : Type* }
48     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
49     (tau : Fin (n + 1) → PowerSeries k) :
50     Matrix (Fin (n + 1)) (κ ⊕ Unit) (LaurentSeries k)
51   | i, Sum.inl j =>
52       algebraMap (PowerSeries k) (LaurentSeries k) (Z i j)
53   | i, Sum.inr _ =>
54       algebraMap (PowerSeries k) (LaurentSeries k) (tau i)
55 
56 /-- The position component of an exact formal tangent annihilation becomes
57 the projective point-annihilation relation over Laurent series. -/
58 theorem laurentColumn_dot_eq_zero_of_formalTangent_rowMul
59     { κ : Type* } [Fintype κ]
60     (q ell : Fin (n + 1) → PowerSeries k)
61     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
62     (tau : Fin (n + 1) → PowerSeries k)
63     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0) :
64     ∑ i, laurentColumn ell i * laurentColumn q i = 0 := by
65   have hposition := congrFun hrow (Sum.inl ())
66   simp only [Pi.zero_apply, rowMul, formalTangentMatrix] at hposition
67   simpa [laurentColumn, map_sum] using
68     congrArg (algebraMap (PowerSeries k) (LaurentSeries k)) hposition
69 
70 /-- The non-position components of an exact formal tangent annihilation
71 become annihilation of all embedded divisor and normalized-transverse
72 columns. -/
73 theorem laurentNonposition_rowMul_eq_zero_of_formalTangent_rowMul
74     { κ : Type* } [Fintype κ]
75     (q ell : Fin (n + 1) → PowerSeries k)
76     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
77     (tau : Fin (n + 1) → PowerSeries k)
78     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0) :
79     rowMul (laurentColumn ell) (laurentNonpositionTangentMatrix Z tau) = 0 := by
80   funext column
81   rcases column with j | _
82   · have hj := congrFun hrow (Sum.inr (Sum.inl j))
83     simp only [Pi.zero_apply, rowMul, formalTangentMatrix] at hj
84     simpa [rowMul, laurentColumn, laurentNonpositionTangentMatrix, map_sum] using
85       congrArg (algebraMap (PowerSeries k) (LaurentSeries k)) hj
86   · have htau := congrFun hrow (Sum.inr (Sum.inr ()))
87     simp only [Pi.zero_apply, rowMul, formalTangentMatrix] at htau
88     simpa [rowMul, laurentColumn, laurentNonpositionTangentMatrix, map_sum] using
89       congrArg (algebraMap (PowerSeries k) (LaurentSeries k)) htau
90 
91 /-- The power-series embedding in Laurent series is nonzero on every nonzero
92 power series. -/
93 theorem laurentColumn_ne_zero_of_ne_zero
94     (q : Fin (n + 1) → PowerSeries k) (hq0 : q 0 ≠ 0) :
95     laurentColumn q 0 ≠ 0 := by
96   intro hzero
97   apply hq0
98   apply HahnSeries.ofPowerSeries_injective (Γ := ℤ)
99   simpa [laurentColumn] using hzero

101 /-- The complete conditional local bridge.  Exact formal row annihilation
102 supplies both projective relations.  Given the base equations and the exact
103 dehomogenized tangent-space equality, the resulting Laurent phase point lies
104 in the equation-defined conormal locus. -/
105 theorem laurentPhasePoint_mem_equationConormalLocus
106     { κ : Type* } [Fintype κ]
107     (I : Ideal (MvPolynomial (Fin n) (LaurentSeries k)))
108     (q ell : Fin (n + 1) → PowerSeries k)
109     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
110     (tau : Fin (n + 1) → PowerSeries k)
111     (hq0 : q 0 ≠ 0)
112     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
113     (hbase : ∀ f ∈ I,
114       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0)
115     (htangent :
116       dehomogenizedTangentSpan (laurentColumn q)
117           (laurentNonpositionTangentMatrix Z tau) =
118         zariskiTangentSpace (dehomogenizedPoint (laurentColumn q)) I) :
119     Sum.elim (dehomogenizedPoint (laurentColumn q))
120         (fun i ↦ laurentColumn ell i.succ) ∈
121       equationConormalLocus I := by
122   exact phasePoint_mem_equationConormalLocus_of_projective_row
123     I (laurentColumn q) (laurentColumn ell)
124       (laurentNonpositionTangentMatrix Z tau)
125       (laurentColumn_ne_zero_of_ne_zero q hq0)
126       (laurentColumn_dot_eq_zero_of_formalTangent_rowMul q ell Z tau hrow)
127       (laurentNonposition_rowMul_eq_zero_of_formalTangent_rowMul
128         q ell Z tau hrow)
129       hbase htangent

131 /-- The residue of the affine covector is exactly the tail of the residue of
132 the formal projective annihilating row.  This is the fibre-residue equality
133 consumed by `LaurentConormalDirection`. -/
134 theorem residueColumn_tail
135     (ell : Fin (n + 1) → PowerSeries k) :
136     residueColumn (fun i : Fin n ↦ ell i.succ) =
137       fun i ↦ residueColumn ell i.succ := by
138   rfl
139 
140 /-- Package the conormal membership together with the exact residue relation
141 for its regular fibre coordinates. -/
142 theorem laurentPhasePoint_mem_equationConormalLocus_and_residue
143     { κ : Type* } [Fintype κ]
144     (I : Ideal (MvPolynomial (Fin n) (LaurentSeries k)))
145     (q ell : Fin (n + 1) → PowerSeries k)
146     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
147     (tau : Fin (n + 1) → PowerSeries k)
148     (hq0 : q 0 ≠ 0)
149     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
150     (hbase : ∀ f ∈ I,
151       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0)
152     (htangent :
153       dehomogenizedTangentSpan (laurentColumn q)
154           (laurentNonpositionTangentMatrix Z tau) =
155         zariskiTangentSpace (dehomogenizedPoint (laurentColumn q)) I) :
156     Sum.elim (dehomogenizedPoint (laurentColumn q))
157           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (ell i.succ)) ∈
158         equationConormalLocus I ∧
159       residueColumn (fun i : Fin n ↦ ell i.succ) =
160         fun i ↦ residueColumn ell i.succ := by
161   exact ⟨laurentPhasePoint_mem_equationConormalLocus
162     I q ell Z tau hq0 hrow hbase htangent, residueColumn_tail ell⟩
163 
164 /-- Direct handoff to `LaurentConormalDirection` for a scalar-extended base
165 ideal.  The possibly singular dehomogenized position is retained over Laurent
166 series, while the regular fibre is the power-series tail of `ell`. -/
167 theorem residue_tail_mem_laurentEquationConormalDirectionClosure
168     { κ : Type* } [Fintype κ]
169     (I : Ideal (MvPolynomial (Fin n) k))
170     (q ell : Fin (n + 1) → PowerSeries k)
171     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
172     (tau : Fin (n + 1) → PowerSeries k)
173     (hq0 : q 0 ≠ 0)
174     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
175     (hbase : ∀ f ∈
176         I.map (scalarPolynomialMap
177           (k := k) (K := LaurentSeries k) (Fin n)),
178       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0)
179     (htangent :
180       dehomogenizedTangentSpan (laurentColumn q)
181           (laurentNonpositionTangentMatrix Z tau) =
182         zariskiTangentSpace (dehomogenizedPoint (laurentColumn q))
183           (I.map (scalarPolynomialMap
184             (k := k) (K := LaurentSeries k) (Fin n)))) :
185     residueColumn (fun i : Fin n ↦ ell i.succ) ∈
186       laurentEquationConormalDirectionClosure I := by
187   apply residue_mem_laurentEquationConormalDirectionClosure
188     I (dehomogenizedPoint (laurentColumn q)) (fun i : Fin n ↦ ell i.succ)
189   exact laurentPhasePoint_mem_equationConormalLocus
190     (I.map (scalarPolynomialMap
191       (k := k) (K := LaurentSeries k) (Fin n)))
192     q ell Z tau hq0 hrow hbase htangent
