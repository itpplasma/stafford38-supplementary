40 noncomputable def actualParameterOutput
41     {k : Type u} [Field k] [CharZero k]
42     {m : ℕ} (hm : 0 < m)
43     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
44     (w : GeneralDivisorialVisibleFrameWitness hm P) : Prop := by
45   let C := w.column
46   let W := C.W
47   let F := ComponentFractionField P
48   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
49   let U := W.place.valuation
50   let V := U.toSubring
51   letI : IsDiscreteValuationRing V := W.place.isDiscrete
52   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
53   letI : Algebra W.coefficientField V :=
54     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
55   letI : Algebra k V := C.groundCoeff.toAlgebra
56   letI : Algebra V F := V.subtype.toAlgebra
57   letI : IsScalarTower k V F := C.groundTower
58   let κ := ResidueField V
59   let retained := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
60   letI : Algebra k κ := retained
61   have hretainedMap : @algebraMap k κ _ _ retained =
62       (residue V).comp C.groundCoeff := by
63     simpa [C, W, V, κ, C.groundCoeff_eq_retained] using w.retainedGroundMap
64   letI : IsScalarTower k V κ := by
65     exact IsScalarTower.of_algebraMap_eq fun c => by
66       have h := RingHom.congr_fun hretainedMap c
67       change algebraMap k κ c = residue V (C.groundCoeff c)
68       exact h
69   let eChart := chartAffineCoordinateEquiv C.chart
70   let Q : Subalgebra k F := Algebra.adjoin k (Set.range (chartGenericPoint P C.chart eChart))
71   let hQV : Q.toSubring ≤ V := by
72     intro z hz
73     exact Stafford38.Geometry.ActualChartValuationImage.chartGenericPointSubalgebra_le_valuationSubring
74       hm P w z hz
75   let B := integralClosure Q.toSubring F
76   let S := B.toSubring
77   let center := integralClosureCenter Q.toSubring U hQV
78   let eB := ActualChartNormalizationCenter.subalgebraToSubringRingEquiv B
79   let centerB := center.comap (eB : B →+* S)
80   letI : center.IsPrime := integralClosureCenter_isPrime Q.toSubring U hQV
81   letI : centerB.IsPrime := Ideal.IsPrime.comap (eB : B →+* S)
82   exact ∀ (t : Set κ), t.Finite →
83       IsTranscendenceBasis k ((↑) : t → κ) →
84       ∀ (index : t → Fin m), Function.Injective index →
85         (∀ z : t, IsLocalRing.residue V (C.q (eChart (index z)).1) = (z : κ)) →
86           ∃ (e : Localization.AtPrime centerB ≃+* V) (s : B)
87             (den : centerB.primeCompl) (u₀ u₁ : Localization.AtPrime centerB),
88             (∀ b : B, ((e (algebraMap B _ b) : V) : F) = (b : F)) ∧
89             s ≠ 0 ∧ s ∈ centerB ∧
90             IsLocalization.mk' _ s den = e.symm
91               (Stafford38.Geometry.CompletedDVRPowerSeries.chosenUniformizer V) ∧
92             Ideal.span {algebraMap B (Localization.AtPrime centerB) s} =
93               maximalIdeal (Localization.AtPrime centerB) ∧
94             IsUnit u₀ ∧ IsUnit u₁ ∧
95             algebraMap B (Localization.AtPrime centerB)
96                 (ActualOptionColumnBinding.actualNormalizedProjectiveColumnInIntegralClosure hm P w 0) =
97               (algebraMap B _ s) ^ w.differential.core.D.a * u₀ ∧
98             algebraMap B (Localization.AtPrime centerB)
99                 (ActualOptionColumnBinding.actualNormalizedProjectiveColumnInIntegralClosure hm P w
100                   (Fin.succ ⟨0, hm⟩)) =
101               (algebraMap B _ s) ^ (w.differential.core.D.a + w.differential.core.D.e) * u₁

136     intro z hz

145   letI : centerB.IsPrime := Ideal.IsPrime.comap (eB : B →+* S)

163     let eqv := ActualChartNormalizationCenter.subalgebraToSubringRingEquiv Q
164     refine ⟨eqv a, eqv b, ?_⟩
165     change z = (eqv a : F) / (eqv b : F)

234       _ = (qB (Fin.succ ⟨0, hm⟩) : F) := by rfl
235       _ = (C.q (Fin.succ ⟨0, hm⟩) : F) := hqB (Fin.succ ⟨0, hm⟩)

253   exact hcore

255 /-- The canonical selected residue basis and the actual retained-order
256 producer coexist over one witness. The latter is universal in the selected
257 basis inputs, so it can be instantiated at precisely the `t/index/hres` from
258 the former without making a second selection. -/
259 theorem selected_basis_and_parameter_output
260     {k : Type u} [Field k] [CharZero k]
261     {m : ℕ} (hm : 0 < m)
262     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
263     (w : GeneralDivisorialVisibleFrameWitness hm P) :
264     let C := w.column
265     let W := C.W
266     let F := ComponentFractionField P
267     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
268     let V := W.place.valuation.toSubring
269     letI : IsDiscreteValuationRing V := W.place.isDiscrete
270     letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
271     letI : Algebra W.coefficientField V :=
272       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
273     let κ := ResidueField V
274     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
275     (∃ (t : Set κ), t.Finite ∧
276       IsTranscendenceBasis k ((↑) : t → κ) ∧
277       ∃ (index : t → Fin m), Function.Injective index ∧
278         ∀ z : t, residue V
279           (C.q ((chartAffineCoordinateEquiv C.chart) (index z)).1) = (z : κ)) ∧
280     actualParameterOutput hm P w := by
281   constructor
282   · exact ActualSelectedResidueBasis.exists_actual_selected_residue_basis hm P w
283   · exact exists_actual_parameter_with_retained_orders hm P w
