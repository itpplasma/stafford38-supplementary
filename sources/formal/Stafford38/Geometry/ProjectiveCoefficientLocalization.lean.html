20 /-- Generic coefficient-to-polynomial localization with an arbitrary set of coefficient
21 variables. The `none` coordinate is the distinguished polynomial variable; `some i` are the
22 coefficient variables. -/
23 abbrev OptionSource (σ : Type w) := MvPolynomial (Option σ) k
24 
25 abbrev OptionCoeff (σ : Type w) := MvPolynomial σ k

31 /-- The coordinate-preserving map `k[a,zᵢ] → E[a]`, with the distinguished parameter indexed by
32 `none` and the selected coefficient variables indexed by `some i`. -/
33 def optionCoordinateMap : OptionSource (k := k) σ →ₐ[k] Polynomial E :=
34   (Polynomial.mapAlgHom (IsScalarTower.toAlgHom k (OptionCoeff (k := k) σ) E)).comp
35     (MvPolynomial.optionEquivLeft k σ).toAlgHom
36 
37 @[simp] theorem optionCoordinateMap_X_none :
38     optionCoordinateMap (k := k) (E := E) (σ := σ) (MvPolynomial.X none) =
39       (Polynomial.X : Polynomial E) := by
40   simp [optionCoordinateMap]

42 @[simp] theorem optionCoordinateMap_X_some (i : σ) :
43     optionCoordinateMap (k := k) (E := E) (σ := σ) (MvPolynomial.X (some i)) =
44       Polynomial.C (algebraMap (OptionCoeff (k := k) σ) E (MvPolynomial.X i)) := by
45   simp [optionCoordinateMap]

47 /-- Invert exactly the non-zero-divisor coefficient polynomials, and no power of the distinguished
48 variable. -/

50   Submonoid.comap (MvPolynomial.optionEquivLeft k σ).toRingHom
51     (Submonoid.map Polynomial.C (nonZeroDivisors (OptionCoeff (k := k) σ)))

55 /-- The generic option-indexed coordinate map is localization at the coefficient polynomials. -/
56 theorem optionCoordinateMap_isLocalization :
57     letI : Algebra (OptionSource (k := k) σ) (Polynomial E) :=
58       (optionCoordinateMap (k := k) (E := E) (σ := σ)).toRingHom.toAlgebra
59     IsLocalization (optionCoordinateDenominators (k := k) (σ := σ)) (Polynomial E) := by
60   letI : Algebra (OptionSource (k := k) σ) (Polynomial E) :=
61     (optionCoordinateMap (k := k) (E := E) (σ := σ)).toRingHom.toAlgebra
62   let A := OptionCoeff (k := k) σ
63   let M₀ : Submonoid A := nonZeroDivisors A
64   let M₁ : Submonoid (Polynomial A) := Submonoid.map Polynomial.C M₀
65   let e : OptionSource (k := k) σ ≃ₐ[k] Polynomial A := MvPolynomial.optionEquivLeft k σ
66   let N : Submonoid (OptionSource (k := k) σ) := Submonoid.comap e.toRingHom M₁
67   have hN : N = optionCoordinateDenominators (k := k) (σ := σ) := rfl
68   letI : Algebra (Polynomial A) (Polynomial E) := Polynomial.algebra A E
69   have hcoord (x : OptionSource (k := k) σ) :
70       optionCoordinateMap (k := k) (E := E) (σ := σ) x =
71         algebraMap (Polynomial A) (Polynomial E) (e x) := rfl
72   have hloc : IsLocalization M₁ (Polynomial E) := by
73     dsimp [M₁, M₀]
74     exact Polynomial.isLocalization (nonZeroDivisors A) E
75   rw [← hN, isLocalization_iff]
76   refine ⟨?_, ⟨?_, ?_⟩⟩
77   · rintro ⟨x, hx⟩
78     change IsUnit ((optionCoordinateMap (k := k) (E := E) (σ := σ)).toRingHom x)
79     have hx' : e x ∈ M₁ := hx
80     change IsUnit (algebraMap (Polynomial A) (Polynomial E) (e x))
81     exact IsLocalization.map_units (Polynomial E) ⟨e x, hx'⟩
82   · intro z
83     obtain ⟨⟨a, b⟩, hab⟩ := IsLocalization.surj (M := M₁) z
84     refine ⟨⟨e.symm a, ⟨e.symm b, ?_⟩⟩, ?_⟩
85     · change e (e.symm b) ∈ M₁
86       simpa only [e.apply_symm_apply] using b.property
87     · change z * optionCoordinateMap (k := k) (E := E) (σ := σ) (e.symm b) =
88         optionCoordinateMap (k := k) (E := E) (σ := σ) (e.symm a)
89       rw [hcoord, e.apply_symm_apply, hcoord, e.apply_symm_apply]
90       exact hab
91   · intro x y hxy
92     have hxy' : algebraMap (Polynomial A) (Polynomial E) (e x) =
93         algebraMap (Polynomial A) (Polynomial E) (e y) := hxy
94     obtain ⟨c, hc⟩ := IsLocalization.exists_of_eq (M := M₁) hxy'
95     refine ⟨⟨e.symm c, ?_⟩, ?_⟩
96     · change e (e.symm c) ∈ M₁
97       simpa only [e.apply_symm_apply] using c.property
98     · apply e.injective
99       simpa using hc

101 /-- The generic option-indexed map is formally étale because it is a localization. -/
102 theorem optionCoordinateMap_formallyEtale :
103     letI : Algebra (OptionSource (k := k) σ) (Polynomial E) :=
104       (optionCoordinateMap (k := k) (E := E) (σ := σ)).toRingHom.toAlgebra
105     Algebra.FormallyEtale (OptionSource (k := k) σ) (Polynomial E) := by
106   letI : Algebra (OptionSource (k := k) σ) (Polynomial E) :=
107     (optionCoordinateMap (k := k) (E := E) (σ := σ)).toRingHom.toAlgebra
108   letI : IsLocalization (optionCoordinateDenominators (k := k) (σ := σ)) (Polynomial E) :=
109     optionCoordinateMap_isLocalization (k := k) (E := E) (σ := σ)
110   exact Algebra.FormallyEtale.of_isLocalization (optionCoordinateDenominators (k := k) (σ := σ))

121 /-- Reindex the legacy `Fin (d+1)` presentation as `Option (Fin d)`, with zero sent to `none`. -/
122 def finToOptionEquiv : Source (k := k) d ≃ₐ[k] OptionSource (k := k) (Fin d) :=
123   MvPolynomial.renameEquiv k (_root_.finSuccEquiv d)

125 /-- The original finite-coordinate map is the generic option-indexed map transported along
126 `Fin (d+1) ≃ Option (Fin d)`. -/
127 def coordinateMap : Source (k := k) d →ₐ[k] Polynomial E :=
128   (optionCoordinateMap (k := k) (E := E) (σ := Fin d)).comp
129     (finToOptionEquiv (k := k) d).toAlgHom

156 /-- The legacy finite-index localization theorem is transported from the single generic producer. -/
157 theorem coordinateMap_isLocalization :
158     letI : Algebra (Source k d) (Polynomial E) :=
159       (coordinateMap (k := k) d).toRingHom.toAlgebra
160     IsLocalization (coordinateDenominators (k := k) d) (Polynomial E) := by
161   letI : Algebra (Source k d) (Polynomial E) :=
162     (coordinateMap (k := k) d).toRingHom.toAlgebra
163   let e := finToOptionEquiv (k := k) d
164   letI : Algebra (OptionSource (k := k) (Fin d)) (Polynomial E) :=
165     (optionCoordinateMap (k := k) (E := E) (σ := Fin d)).toRingHom.toAlgebra
166   have hden :
167       (coordinateDenominators (k := k) d).map e.toRingEquiv =
168         optionCoordinateDenominators (k := k) (σ := Fin d) := by
169     ext x
170     constructor
171     · rintro ⟨y, hy, rfl⟩
172       change e y ∈ optionCoordinateDenominators (k := k) (σ := Fin d) at hy
173       exact hy
174     · intro hx
175       refine ⟨e.symm x, ?_, e.apply_symm_apply x⟩
176       change e (e.symm x) ∈ optionCoordinateDenominators (k := k) (σ := Fin d)
177       simpa using hx
178   have hlocOption :
179       IsLocalization (optionCoordinateDenominators (k := k) (σ := Fin d)) (Polynomial E) := by
180     exact optionCoordinateMap_isLocalization (k := k) (E := E) (σ := Fin d)
181   exact IsLocalization.of_ringEquiv_left e.toRingEquiv hden (by
182     intro x
183     change coordinateMap (k := k) d x =
184       optionCoordinateMap (k := k) (E := E) (σ := Fin d) (e x)
185     rfl)

190     letI : Algebra (Source k d) (Polynomial E) :=
