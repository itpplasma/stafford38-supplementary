155 /-- A left inverse of the augmented projective matrix makes the affine
156 dehomogenizations of its nonposition columns independent in the `q₀` chart. -/
157 theorem linearIndependent_dehomogenizedTangentColumns
158     {m : ℕ} (hm : 0 < m) {δ : Type*} [Fintype δ]
159     (q : Fin (m + 1) → LaurentSeries κ)
160     (B : Matrix (Fin (m + 1)) δ (LaurentSeries κ))
161     (hq0 : q 0 ≠ 0)
162     (hinjective : Function.Injective
163       (augmentedProjectiveMatrix q B).mulVec) :
164     LinearIndependent (LaurentSeries κ)
165       (fun j ↦ dehomogenizedTangentColumn q (fun i ↦ B i j)) := by
166   let e := zeroChartIndexEquiv (m := m)
167   have hchart := linearIndependent_chartDehomogenizedTangentColumns
168     (chart := (0 : Fin (m + 1))) q B hq0 hinjective
169   let reindex : (ChartAffineIndex (Fin (m + 1)) 0 → LaurentSeries κ) →ₗ[
170       LaurentSeries κ] (Fin m → LaurentSeries κ) := {
171     toFun := fun f i ↦ f (e i)
172     map_add' := by intro f g; ext i; rfl
173     map_smul' := by intro a f; ext i; rfl }
174   have hReindex : Function.Injective reindex := by
175     intro f g h
176     funext i
177     have hi := congrFun h (e.symm i)
178     simpa [reindex] using hi
179   have hkernel : LinearMap.ker reindex = ⊥ :=
180     LinearMap.ker_eq_bot.mpr hReindex
181   have h := hchart.map' reindex hkernel
182   change LinearIndependent (LaurentSeries κ)
183     (fun j ↦ reindex (chartDehomogenizedTangentColumn 0 q
184       (fun i ↦ B i j))) at h
185   have hcomp :
186       (fun j ↦ reindex (chartDehomogenizedTangentColumn 0 q
187         (fun i ↦ B i j))) =
188         (fun j ↦ dehomogenizedTangentColumn q (fun i ↦ B i j)) := by
189     funext j
190     funext i
191     simp [reindex, e, zeroChartIndexEquiv,
192       chartDehomogenizedTangentColumn, dehomogenizedTangentColumn]
193   rw [hcomp] at h
194   exact h

282     fun i ↦ algebraMap (PowerSeries κ) L (PowerSeries.derivative (R := κ) (q i))

310         rw [hfactor i, map_mul]
