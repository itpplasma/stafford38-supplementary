8 /-!
9 # The pure-momentum axis obstruction
10 
11 The normalized canonical order symbol is homogeneous of ordinary total degree:
12 it is order-homogeneous and its proved fibre-only property removes every
13 weight-zero coordinate exponent.  Its value at the distinguished pure-momentum
14 axis is therefore its normalized pure coefficient, namely one.
15 
16 Consequently that axis cannot belong to any checked affine zero locus contained
17 in the principal-symbol hypersurface.  Applying the existing conditional
18 equation-conormal-closure containment gives the corresponding contradiction.
19 This file proves no conormal-axis producer and no Gabber theorem.
20 -/

42 /-- An order-homogeneous fibre-only symbol is homogeneous for ordinary total
43 degree. -/
44 theorem isHomogeneous_of_orderHomogeneous_of_isFibreOnly
45     {P : SymbolRing k n}
46     (horder : P.IsWeightedHomogeneous (@orderWeight n) N)
47     (hfibre : IsFibreOnly k P) :
48     P.IsHomogeneous N := by
49   intro m hm
50   have horder' := horder hm
51   rw [Finsupp.weight_apply] at horder'
52   rw [← horder']
53   apply Finsupp.sum_congr
54   intro i hi
55   rcases i with i | i
56   · simp [orderWeight, fibreWeight, hfibre m hm i]
57   · simp [orderWeight, fibreWeight]

72 /-- A homogeneous polynomial with pure-axis coefficient one does not vanish at
73 that axis. -/
74 theorem eval_axis_eq_one_of_pureCoefficient_one
75     (t : PhaseVar n) {P : SymbolRing k n}
76     (hP : P.IsHomogeneous N)
77     (hpure : MvPolynomial.coeff (Finsupp.single t N) P = 1) :
78     MvPolynomial.eval (axisPoint k t) P = 1 := by
79   classical
80   rw [MvPolynomial.eval_eq]
81   rw [Finset.sum_eq_single (Finsupp.single t N)]
82   · change AddMonoidAlgebra.coeff P (Finsupp.single t N) = 1 at hpure
83     rw [hpure, one_mul]
84     by_cases hN : N = 0
85     · subst N
86       simp
87     · rw [Finsupp.support_single_ne_zero t hN]
88       simp [axisPoint]
89   · intro m hm hne
90     have hmcoeff : MvPolynomial.coeff m P ≠ 0 :=
91       MvPolynomial.mem_support_iff.mp hm
92     have hexists : ∃ i ∈ m.support, i ≠ t := by
93       by_contra hnot
94       have hsubset : m.support ⊆ {t} := by
95         intro i hi
96         simp only [Finset.mem_singleton]
97         by_contra hit
98         exact hnot ⟨i, hi, hit⟩
99       have hmform : m = Finsupp.single t (m t) :=
100         Finsupp.support_subset_singleton.mp hsubset
101       have hdegree : m.degree = N := by
102         rw [Finsupp.degree_eq_weight_one]
103         exact hP hmcoeff
104       rw [hmform, Finsupp.degree_single] at hdegree
105       apply hne
106       rw [hmform, hdegree]
107     rcases hexists with ⟨i, hi, hit⟩
108     rw [Finset.prod_eq_zero hi, mul_zero]
109     rw [axisPoint, if_neg hit, zero_pow]
110     exact Finsupp.mem_support_iff.mp hi
111   · intro hnot
112     exact (hnot (MvPolynomial.mem_support_iff.mpr (hpure.trans_ne one_ne_zero))).elim

114 /-- Direct set-level obstruction: if a checked affine locus lies in `V(P)`,
115 axis membership contradicts homogeneity and pure coefficient one. -/
116 theorem false_of_axis_mem_of_subset_principal_zeroLocus
117     (t : PhaseVar n) {P : SymbolRing k n}
118     (hP : P.IsHomogeneous N)
119     (hpure : MvPolynomial.coeff (Finsupp.single t N) P = 1)
120     {S : Set (PhaseVar n → k)}
121     (hS : S ⊆ MvPolynomial.zeroLocus k (Ideal.span ({P} : Set (SymbolRing k n))))
122     (haxis : axisPoint k t ∈ S) : False := by
123   have hzero := (MvPolynomial.mem_zeroLocus_iff.mp (hS haxis)) P
124     (Ideal.subset_span (Set.mem_singleton P))
125   rw [MvPolynomial.aeval_eq_eval,
126     eval_axis_eq_one_of_pureCoefficient_one t hP hpure] at hzero
127   exact one_ne_zero hzero

179 /-- Exact equation-conormal connector.  Under the already isolated
180 base-relative Poisson hypothesis, membership of the distinguished axis in the
181 equation-conormal closure is impossible. -/
182 theorem false_of_canonical_pureMomentumAxis_mem_equationConormalClosure
183     [CharZero k]
184     (n N : ℕ) {d : PresentedWeyl k (n + 1)}
185     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
186     (hJ : IsBaseRelativePoisson
187       (reducedOrderSupportIdeal k
188         (canonicalRightIdeal (presentedCoordinate k n) d N)))
189     (haxis : axisPoint k (.inr (0 : Fin (n + 1))) ∈
190       equationConormalClosure
191         (reducedOrderBaseIdeal k
192           (canonicalRightIdeal (presentedCoordinate k n) d N))) : False := by
193   apply false_of_axis_mem_of_subset_principal_zeroLocus
194     (.inr (0 : Fin (n + 1)))
195     (canonical_orderPrincipalComponent_isHomogeneous n N hd)
196     (canonical_orderPrincipalComponent_pureMomentumCoefficient k n N hd)
197     (S := equationConormalClosure
198       (reducedOrderBaseIdeal k
199         (canonicalRightIdeal (presentedCoordinate k n) d N)))
200   · exact (equationConormalClosure_subset_reducedOrderSupport
201       (canonicalRightIdeal (presentedCoordinate k n) d N) hJ).trans
202         (canonical_reducedOrderSupport_zeroLocus_subset_principal_zeroLocus
203           n N hd)
