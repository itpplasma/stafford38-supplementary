10 /-!
11 # The pure-momentum axis obstruction
12 
13 The normalized canonical order symbol is homogeneous of ordinary total degree:
14 it is order-homogeneous and its proved fibre-only property removes every
15 weight-zero coordinate exponent.  Its value at the distinguished pure-momentum
16 axis is therefore its normalized pure coefficient, namely one.
17 
18 Consequently that axis cannot belong to any checked affine zero locus contained
19 in the principal-symbol hypersurface.  Applying the existing conditional
20 equation-conormal-closure containment gives the corresponding contradiction.
21 This file proves no conormal-axis producer and no Gabber theorem.
22 -/

44 /-- An order-homogeneous fibre-only symbol is homogeneous for ordinary total
45 degree. -/
46 theorem isHomogeneous_of_orderHomogeneous_of_isFibreOnly
47     {P : SymbolRing k n}
48     (horder : P.IsWeightedHomogeneous (@orderWeight n) N)
49     (hfibre : IsFibreOnly k P) :
50     P.IsHomogeneous N := by
51   intro m hm
52   have horder' := horder hm
53   rw [Finsupp.weight_apply] at horder'
54   rw [← horder']
55   apply Finsupp.sum_congr
56   intro i hi
57   rcases i with i | i
58   · simp [orderWeight, fibreWeight, hfibre m hm i]
59   · simp [orderWeight, fibreWeight]

65     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
66     (presentedPrincipalComponent k orderWeight N d).IsHomogeneous N := by

74 /-- A homogeneous polynomial with pure-axis coefficient one does not vanish at
75 that axis. -/
76 theorem eval_axis_eq_one_of_pureCoefficient_one
77     (t : PhaseVar n) {P : SymbolRing k n}
78     (hP : P.IsHomogeneous N)
79     (hpure : MvPolynomial.coeff (Finsupp.single t N) P = 1) :
80     MvPolynomial.eval (axisPoint k t) P = 1 := by
81   classical
82   rw [MvPolynomial.eval_eq]
83   rw [Finset.sum_eq_single (Finsupp.single t N)]
84   · change AddMonoidAlgebra.coeff P (Finsupp.single t N) = 1 at hpure
85     rw [hpure, one_mul]
86     by_cases hN : N = 0
87     · subst N
88       simp
89     · rw [Finsupp.support_single_ne_zero t hN]
90       simp [axisPoint]
91   · intro m hm hne
92     have hmcoeff : MvPolynomial.coeff m P ≠ 0 :=
93       MvPolynomial.mem_support_iff.mp hm
94     have hexists : ∃ i ∈ m.support, i ≠ t := by
95       by_contra hnot
96       have hsubset : m.support ⊆ {t} := by
97         intro i hi
98         simp only [Finset.mem_singleton]
99         by_contra hit
100         exact hnot ⟨i, hi, hit⟩
101       have hmform : m = Finsupp.single t (m t) :=
102         Finsupp.support_subset_singleton.mp hsubset
103       have hdegree : m.degree = N := by
104         rw [Finsupp.degree_eq_weight_one]
105         exact hP hmcoeff
106       rw [hmform, Finsupp.degree_single] at hdegree
107       apply hne
108       rw [hmform, hdegree]
109     rcases hexists with ⟨i, hi, hit⟩
110     rw [Finset.prod_eq_zero hi, mul_zero]
111     rw [axisPoint, if_neg hit, zero_pow]
112     exact Finsupp.mem_support_iff.mp hi
113   · intro hnot
114     exact (hnot (MvPolynomial.mem_support_iff.mpr (hpure.trans_ne one_ne_zero))).elim

116 /-- Direct set-level obstruction: if a checked affine locus lies in `V(P)`,
117 axis membership contradicts homogeneity and pure coefficient one. -/
118 theorem false_of_axis_mem_of_subset_principal_zeroLocus
119     (t : PhaseVar n) {P : SymbolRing k n}
120     (hP : P.IsHomogeneous N)
121     (hpure : MvPolynomial.coeff (Finsupp.single t N) P = 1)
122     {S : Set (PhaseVar n → k)}
123     (hS : S ⊆ MvPolynomial.zeroLocus k (Ideal.span ({P} : Set (SymbolRing k n))))
124     (haxis : axisPoint k t ∈ S) : False := by
125   have hzero := (MvPolynomial.mem_zeroLocus_iff.mp (hS haxis)) P
126     (Ideal.subset_span (Set.mem_singleton P))
127   rw [MvPolynomial.aeval_eq_eval,
128     eval_axis_eq_one_of_pureCoefficient_one t hP hpure] at hzero
129   exact one_ne_zero hzero

181 /-- Exact equation-conormal connector.  Under the already isolated
182 base-relative Poisson hypothesis, membership of the distinguished axis in the
183 equation-conormal closure is impossible. -/
184 theorem false_of_canonical_pureMomentumAxis_mem_equationConormalClosure
185     [CharZero k]
186     (n N : ℕ) {d : WeylAlg k (n + 1)}
187     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)

191     (haxis : axisPoint k (.inr (0 : Fin (n + 1))) ∈
192       equationConormalClosure
193         (reducedOrderBaseIdeal k
194           (canonicalRightIdeal (presentedCoordinate k n) d N))) : False := by
195   apply false_of_axis_mem_of_subset_principal_zeroLocus
196     (.inr (0 : Fin (n + 1)))
197     (canonical_orderPrincipalComponent_isHomogeneous n N hd)
198     (canonical_orderPrincipalComponent_pureMomentumCoefficient k n N hd)

203       (canonicalRightIdeal (presentedCoordinate k n) d N) hJ).trans
204         (canonical_reducedOrderSupport_zeroLocus_subset_principal_zeroLocus
205           n N hd)
