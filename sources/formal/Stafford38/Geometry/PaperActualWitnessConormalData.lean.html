56 /-- Assemble the actual retained witness into the conditional matrix theorem.
57 The tangent bound, point kernel, base equations, derivative columns, and strict
58 order gap are all derived from this same witness. -/
59 theorem exists_regularizedOneRowConormalData_of_actual_witness
60     {k : Type u} [Field k] [CharZero k] {m : ℕ} (hm : 0 < m)
61     (I : Ideal (MvPolynomial (Fin m) k)) (hI : I.IsRadical)
62     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
63     (hP : P.asIdeal ∈ I.minimalPrimes)
64     (w : GeneralDivisorialVisibleFrameWitness hm P)
65     (hunit : ∃ g : MvPolynomial (Fin m) k,
66       MvPolynomial.X ⟨0, hm⟩ * g - 1 ∈ P.asIdeal) :
67     let C := w.column
68     let W := C.W
69     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
70       (ComponentFractionField P) := W.ambientAlgebra
71     let V := W.place.valuation.toSubring
72     letI : IsDiscreteValuationRing V := W.place.isDiscrete
73     letI : Algebra W.coefficientField V :=
74       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
75     let κ := ResidueField V
76     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
77     Nonempty (RegularizedOneRowConormalData (k := k) (K := κ) hm I
78       (fun a ↦ retainedToCompletedPowerSeries W (C.q a))) := by
79   classical
80   let C := w.column
81   let W := C.W
82   let F := ComponentFractionField P
83   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
84   let V := W.place.valuation.toSubring
85   letI : IsDiscreteValuationRing V := W.place.isDiscrete
86   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
87   letI : Algebra W.coefficientField V :=
88     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
89   let κ := ResidueField V
90   letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
91   letI : CharZero κ :=
92     charZero_of_injective_algebraMap (FaithfulSMul.algebraMap_injective k κ)
93   let E := LaurentSeries κ
94   letI : Algebra k E := (groundLaurentMap (k := k) (K := κ)).toAlgebra
95   let i : Fin m := ⟨0, hm⟩
96   let q : Fin (m + 1) → PowerSeries κ :=
97     fun a ↦ retainedToCompletedPowerSeries W (C.q a)
98   let ell := retainedLaurentLift P i W
99   let y : Fin m → E := dehomogenizedPoint (laurentColumn q)
100   obtain ⟨rows, D, _hdual, hminorCoeff⟩ :=
101     columns_of_visible_witness hm P w
102   let Z := coefficientwiseTangentMatrix q D
103   have hZ : Z = coefficientwiseTangentMatrix q D := rfl
104   have hqchart : q C.chart = 1 := by
105     dsimp [q]
106     rw [C.chart_one]
107     exact map_one (retainedToCompletedPowerSeries W)
108   obtain ⟨a, b, u₀, u₁, ha, hab, hqzero, hu₀, hqaxis⟩ :=
109     order_gap_of_actual_witness hm P w hunit
110   have hminor : PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0 := by
111     change PowerSeries.constantCoeff
112       (selectedMinor (coefficientwiseTangentMatrix q D) rows).det ≠ 0
113     rw [hminorCoeff]
114     norm_num
115   have hground := retainedLaurentLift_comp_algebraMap P i W
116   have hpoint := retainedLaurentLift_componentCoordinate P i W C.q C.scale
117     C.q_commonScale C.q0_ne
118   have hychart : y = dehomogenizedPoint (laurentColumn q) := rfl
119   have hbaseP : ∀ f ∈ P.asIdeal,
120       MvPolynomial.eval₂ (algebraMap k E)
121         (dehomogenizedPoint (laurentColumn q)) f = 0 := by
122     intro f hf
123     have hvanish := component_equations_vanish_after_point_transport
124       P ell hground y hpoint (f := f) hf
125     simpa [y] using hvanish
126   have hkernel := component_generic_kernel_after_point_transport
127     P ell (RingHom.injective ell) hground y hpoint
128   have hy : ∀ f,
129       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q))
130         (MvPolynomial.map (algebraMap k E) f) = 0 ↔ f ∈ P.asIdeal := by
131     intro f
132     rw [← hychart]
133     exact hkernel f
134   have hbound : Module.finrank E
135       (zariskiTangentSpace (dehomogenizedPoint (laurentColumn q))
136         (P.asIdeal.map (MvPolynomial.map (algebraMap k E)))) ≤
137       Module.finrank κ (Ω[κ⁄k]) + 1 := by
138     change tangent_finrank_goal hm P w
139     exact tangent_finrank_le_residue_add_one_of_witness hm P w
140   exact exists_regularizedOneRowConormalData_of_actual_columns
141     hm I hI P hP q Z rows C.chart a b u₀ u₁ D hZ hqchart ha hab
142     hqzero hu₀ hqaxis hminor hbaseP hbound hy
