25 /-- The actual normalized projective witness has a strict order gap with
26 respect to any supplied irreducible uniformizer of its retained DVR. The
27 ratio is the retained place parameter; in particular, this theorem does not
28 identify the projective denominator `q₀` with the uniformizer. -/
29 theorem exists_actual_witness_strict_orderGap
30     {k : Type u} [Field k] [CharZero k]
31     {m : ℕ} (hm : 0 < m)
32     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
33     (w : GeneralDivisorialVisibleFrameWitness hm P) :
34     letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
35       (ComponentFractionField P) := w.column.W.ambientAlgebra
36     let C := w.column
37     let W := C.W
38     let V := W.place.valuation.toSubring
39     letI : IsDiscreteValuationRing V := W.place.isDiscrete
40     letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
41     ∀ (π : V), Irreducible π →
42       ∃ (a r b : ℕ) (u₀ ur u₁ : Vˣ),
43         0 < a ∧ 0 < r ∧ b = a + r ∧ a < b ∧
44         C.q 0 = (u₀ : V) * π ^ a ∧
45         W.place.parameter = (ur : V) * π ^ r ∧
46         u₁ = u₀ * ur ∧
47         C.q (Fin.succ ⟨0, hm⟩) = (u₁ : V) * π ^ b := by
48   classical
49   dsimp only
50   intro π hπ
51   let C := w.column
52   let W := C.W
53   let F := ComponentFractionField P
54   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
55   let V := W.place.valuation.toSubring
56   letI : IsDiscreteValuationRing V := W.place.isDiscrete
57   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
58   have hq₀max : C.q 0 ∈ maximalIdeal V :=
59     GeneralDivisorialVisibleFrame.witness_q0_mem_maximal hm P w
60   have hratioMax : W.place.parameter ∈ maximalIdeal V := by
61     apply (IsLocalRing.mem_maximalIdeal _).2
62     exact mem_nonunits_iff.mpr W.place.parameter_nonunit
63   exact ProjectiveDivisorOrderGap.exists_uniformizer_strict_orderGap
64     π hπ (C.q 0) W.place.parameter (C.q (Fin.succ ⟨0, hm⟩))
65     C.q0_ne W.place.parameter_ne hq₀max hratioMax C.q_parameter
66 
67 
68 end
