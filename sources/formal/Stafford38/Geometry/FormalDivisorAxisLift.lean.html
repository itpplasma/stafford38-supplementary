8 # The axis lift for a formal divisor tangent
9 
10 This file completes the local power-series part of the formal-divisor
11 construction.  If the distinguished axis coordinate and every divisor-tangent
12 entry in that row vanish to order at least `b`, while the common factor removed
13 from the corrected transverse derivative has order at most `a - 1 < b - 1`,
14 then the normalized transverse tangent has zero axis residue.  Consequently
15 the pure axis covector annihilates every reduced tangent column.
16 
17 The residue-rank theorem from `FormalDivisorTangent` supplies a power-series
18 left inverse.  Retraction correction then produces an exact annihilator whose
19 residue is the pure axis covector.  No normalization, projective divisor,
20 formal-chart construction, or global conormal-closure statement is made here.
21 -/

37 /-! ## The strict axis-order calculation -/
38 
39 /-- The factor removed from the corrected transverse derivative has strictly
40 smaller order than the axis row.  Therefore the normalized transverse tangent
41 still vanishes at the closed point in the axis coordinate. -/
42 theorem constantCoeff_normalizedTransverse_axis_eq_zero
43     {ι : Type v} {κ : Type w} [Fintype κ]
44     (q : ι → PowerSeries k)
45     (Z : Matrix ι κ (PowerSeries k))
46     (lambda : κ → PowerSeries k)
47     (tau : ι → PowerSeries k)
48     (axis : ι) (a b c : ℕ) (u₁ : PowerSeries k)
49     (ha : 0 < a)
50     (hab : a < b)
51     (hc : c ≤ a - 1)
52     (hqaxis : q axis = (PowerSeries.X : PowerSeries k) ^ b * u₁)
53     (hZaxis : ∀ j, ∃ w : PowerSeries k,
54       Z axis j = (PowerSeries.X : PowerSeries k) ^ b * w)
55     (hfactor :
56       PowerSeries.derivative (R := k) (q axis) - Z.mulVec lambda axis =
57         (PowerSeries.X : PowerSeries k) ^ c * tau axis) :
58     PowerSeries.constantCoeff (tau axis) = 0 := by
59   have hcb : c < b := by omega
60   have hcsuccb : c + 1 < b := by omega
61   have hderivative :
62       PowerSeries.coeff c (PowerSeries.derivative (R := k) (q axis)) = 0 := by
63     rw [PowerSeries.coeff_derivative, hqaxis]
64     rw [PowerSeries.coeff_X_pow_mul']
65     simp [Nat.not_le_of_gt hcsuccb]
66   have hcombination :
67       PowerSeries.coeff c (Z.mulVec lambda axis) = 0 := by
68     change PowerSeries.coeff c (∑ j, Z axis j * lambda j) = 0
69     rw [map_sum]
70     apply Finset.sum_eq_zero
71     intro j hj
72     obtain ⟨w, hw⟩ := hZaxis j
73     rw [hw, mul_assoc]
74     rw [PowerSeries.coeff_X_pow_mul']
75     simp [Nat.not_le_of_gt hcb]
76   have hcorrected :
77       PowerSeries.coeff c
78         (PowerSeries.derivative (R := k) (q axis) - Z.mulVec lambda axis) = 0 := by
79     rw [map_sub, hderivative, hcombination, sub_zero]
80   have hnormalized :
81       PowerSeries.coeff c
82           ((PowerSeries.X : PowerSeries k) ^ c * tau axis) =
83         PowerSeries.constantCoeff (tau axis) := by
84     simpa [PowerSeries.coeff_zero_eq_constantCoeff_apply, Nat.add_comm] using
85       (PowerSeries.coeff_X_pow_mul (tau axis) c 0)
86   rw [← hnormalized, ← hfactor]
87   exact hcorrected

89 /-! ## The axis covector on the assembled tangent matrix -/
90 
91 /-- Vanishing of the position, divisor, and normalized transverse entries in
92 the axis row gives residue-zero for every assembled tangent column. -/
93 theorem constantCoeff_formalTangentMatrix_axis_eq_zero
94     {ι : Type v} {κ : Type w}
95     (q : ι → PowerSeries k)
96     (Z : Matrix ι κ (PowerSeries k))
97     (tau : ι → PowerSeries k)
98     (axis : ι) (b : ℕ) (u₁ : PowerSeries k)
99     (hb : 0 < b)
100     (hqaxis : q axis = (PowerSeries.X : PowerSeries k) ^ b * u₁)
101     (hZaxis : ∀ j, ∃ w : PowerSeries k,
102       Z axis j = (PowerSeries.X : PowerSeries k) ^ b * w)
103     (htauaxis : PowerSeries.constantCoeff (tau axis) = 0) :
104     ∀ column : FormalTangentColumn κ,
105       PowerSeries.constantCoeff
106         (formalTangentMatrix q Z tau axis column) = 0 := by
107   intro column
108   rcases column with (_ | (j | _))
109   · rw [formalTangentMatrix, hqaxis]
110     obtain ⟨m, rfl⟩ := Nat.exists_eq_succ_of_ne_zero (Nat.ne_of_gt hb)
111     simp [pow_succ]
112   · rw [formalTangentMatrix]
113     obtain ⟨w, hw⟩ := hZaxis j
114     rw [hw]
115     obtain ⟨m, rfl⟩ := Nat.exists_eq_succ_of_ne_zero (Nat.ne_of_gt hb)
116     simp [pow_succ]
117   · simpa [formalTangentMatrix] using htauaxis

119 /-- The pure axis covector annihilates the residue of the assembled tangent
120 matrix. -/
121 theorem axisRow_residue_formalTangentMatrix_eq_zero
122     {ι : Type v} {κ : Type w} [Fintype ι] [DecidableEq ι]
123     (q : ι → PowerSeries k)
124     (Z : Matrix ι κ (PowerSeries k))
125     (tau : ι → PowerSeries k)
126     (axis : ι)
127     (haxis : ∀ column : FormalTangentColumn κ,
128       PowerSeries.constantCoeff
129         (formalTangentMatrix q Z tau axis column) = 0) :
130     rowMul (axisRow (k := k) axis)
131       (fun i column => PowerSeries.constantCoeff
132         (formalTangentMatrix q Z tau i column)) = 0 := by
133   funext column
134   simp [rowMul, axisRow, haxis column]

138 /-- A supplied correction constructs the axis annihilator and split tangent
139 lattice. Constant corrections use this owner without a matrix inverse. -/
140 theorem exists_formalDivisorAxisLift_of_selected_correction
141     [CharZero k]
142     {ι : Type v} {κ : Type w} [Fintype ι] [Fintype κ]
143     [DecidableEq ι] [DecidableEq κ]
144     (q : ι → PowerSeries k)
145     (Z : Matrix ι κ (PowerSeries k))
146     (rows : κ ↪ ι) (chart zero axis : ι)
147     (a b : ℕ) (u₀ u₁ : PowerSeries k) (lambda : κ → PowerSeries k)
148     (hqchart : q chart = 1)
149     (hZchart : ∀ j, Z chart j = 0)
150     (ha : 0 < a)
151     (hab : a < b)
152     (hqzero : q zero = (PowerSeries.X : PowerSeries k) ^ a * u₀)
153     (hu₀ : PowerSeries.constantCoeff u₀ ≠ 0)
154     (hZzero : ∀ j, ∃ w : PowerSeries k,
155       Z zero j = (PowerSeries.X : PowerSeries k) ^ a * w)
156     (hqaxis : q axis = (PowerSeries.X : PowerSeries k) ^ b * u₁)
157     (hZaxis : ∀ j, ∃ w : PowerSeries k,
158       Z axis j = (PowerSeries.X : PowerSeries k) ^ b * w)
159     (hselected : ∀ j, PowerSeries.derivative (R := k) (q (rows j)) =
160       Z.mulVec lambda (rows j))
161     (hminor :
162       PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0) :
163     ∃ (c : ℕ) (tau : ι → PowerSeries k)
164       (C : Matrix (FormalTangentColumn κ) ι (PowerSeries k))
165       (ell : ι → PowerSeries k),
166       tau chart = 0 ∧
167       (∀ j, tau (rows j) = 0) ∧
168       c ≤ a - 1 ∧
169       (∀ i,
170         PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
171           (PowerSeries.X : PowerSeries k) ^ c * tau i) ∧
172       (∃ i, PowerSeries.constantCoeff (tau i) ≠ 0) ∧
173       PowerSeries.constantCoeff (tau axis) = 0 ∧
174       (∀ column : FormalTangentColumn κ,
175         PowerSeries.constantCoeff
176           (formalTangentMatrix q Z tau axis column) = 0) ∧
177       C * formalTangentMatrix q Z tau = 1 ∧
178       rowMul ell (formalTangentMatrix q Z tau) = 0 ∧
179       residueColumn ell = axisRow (k := k) axis := by
180   obtain ⟨c, tau, htauchart, htauselected, hc, hfactor, hprimitive, hinjective⟩ :=
181     exists_formalDivisorTangent_residue_injective_of_selected_correction
182       q Z rows chart zero a u₀ lambda hqchart hZchart ha hqzero hu₀
183         hZzero hselected hminor
184   have htauaxis : PowerSeries.constantCoeff (tau axis) = 0 :=
185     constantCoeff_normalizedTransverse_axis_eq_zero
186       q Z lambda tau axis a b c u₁ ha hab hc hqaxis hZaxis (hfactor axis)
187   have hb : 0 < b := lt_trans ha hab
188   have haxiscolumns : ∀ column : FormalTangentColumn κ,
189       PowerSeries.constantCoeff
190         (formalTangentMatrix q Z tau axis column) = 0 :=
191     constantCoeff_formalTangentMatrix_axis_eq_zero
192       q Z tau axis b u₁ hb hqaxis hZaxis htauaxis
193   obtain ⟨C, hCB⟩ :=
194     powerSeries_exists_leftInverse_of_residue_mulVec_injective
195       (formalTangentMatrix q Z tau) hinjective
196   let a₀ : ι → k := axisRow (k := k) axis
197   let ell : ι → PowerSeries k :=
198     annihilatorLift (constantColumn a₀)
199       (formalTangentMatrix q Z tau) C
200   have hresidueInput : residueColumn (constantColumn a₀) = a₀ :=
201     residueColumn_constantColumn a₀
202   have haxisRow :
203       rowMul a₀
204         (fun i column => PowerSeries.constantCoeff
205           (formalTangentMatrix q Z tau i column)) = 0 :=
206     axisRow_residue_formalTangentMatrix_eq_zero
207       q Z tau axis haxiscolumns
208   have hell :
209       rowMul ell (formalTangentMatrix q Z tau) = 0 ∧
210         residueColumn ell = a₀ := by
211     simpa [ell] using
212       (powerSeries_annihilatorLift_spec
213         (constantColumn a₀) a₀ (formalTangentMatrix q Z tau) C
214           hCB hresidueInput haxisRow)
215   exact ⟨c, tau, C, ell, htauchart,
216     htauselected, hc, hfactor, hprimitive, htauaxis, haxiscolumns,
217     hCB, hell.1, by simpa [a₀] using hell.2⟩

219 /-- The complete local formal-divisor axis lift.
220 
221 The correction and primitive normalization are constructed from the formal
222 chart data.  The strict axis-order hypotheses force the pure axis covector to
223 annihilate the reduced tangent matrix.  A left inverse derived from the same
224 matrix then gives an exact power-series annihilator specializing to that
225 covector.
226 -/
227 theorem exists_formalDivisorAxisLift
228     [CharZero k]
229     {ι : Type v} {κ : Type w} [Fintype ι] [Fintype κ]
230     [DecidableEq ι] [DecidableEq κ]
231     (q : ι → PowerSeries k)
232     (Z : Matrix ι κ (PowerSeries k))
233     (rows : κ ↪ ι) (chart zero axis : ι)
234     (a b : ℕ) (u₀ u₁ : PowerSeries k)
235     (hqchart : q chart = 1)
236     (hZchart : ∀ j, Z chart j = 0)
237     (ha : 0 < a)
238     (hab : a < b)
239     (hqzero : q zero = (PowerSeries.X : PowerSeries k) ^ a * u₀)
240     (hu₀ : PowerSeries.constantCoeff u₀ ≠ 0)
241     (hZzero : ∀ j, ∃ w : PowerSeries k,
242       Z zero j = (PowerSeries.X : PowerSeries k) ^ a * w)
243     (hqaxis : q axis = (PowerSeries.X : PowerSeries k) ^ b * u₁)
244     (hZaxis : ∀ j, ∃ w : PowerSeries k,
245       Z axis j = (PowerSeries.X : PowerSeries k) ^ b * w)
246     (hminor :
247       PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0) :
248     ∃ (lambda : κ → PowerSeries k) (c : ℕ)
249       (tau : ι → PowerSeries k)
250       (C : Matrix (FormalTangentColumn κ) ι (PowerSeries k))
251       (ell : ι → PowerSeries k),
252       lambda = correctionCoefficients Z rows
253         (fun i => PowerSeries.derivative (R := k) (q i)) ∧
254       (∀ j,
255         PowerSeries.derivative (R := k) (q (rows j)) =
256           Z.mulVec lambda (rows j)) ∧
257       tau chart = 0 ∧
258       (∀ j, tau (rows j) = 0) ∧
259       c ≤ a - 1 ∧
260       (∀ i,
261         PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
262           (PowerSeries.X : PowerSeries k) ^ c * tau i) ∧
263       (∃ i, PowerSeries.constantCoeff (tau i) ≠ 0) ∧
264       PowerSeries.constantCoeff (tau axis) = 0 ∧
265       (∀ column : FormalTangentColumn κ,
266         PowerSeries.constantCoeff
267           (formalTangentMatrix q Z tau axis column) = 0) ∧
268       C * formalTangentMatrix q Z tau = 1 ∧
269       rowMul ell (formalTangentMatrix q Z tau) = 0 ∧
270       residueColumn ell = axisRow (k := k) axis := by
271   let v : ι → PowerSeries k := fun i => PowerSeries.derivative (R := k) (q i)
272   let lambda : κ → PowerSeries k := correctionCoefficients Z rows v
273   have hselected : ∀ j, v (rows j) = Z.mulVec lambda (rows j) := by
274     intro j
275     exact sub_eq_zero.mp (correctedVector_selectedRow_eq_zero Z rows v hminor j)
276   obtain ⟨c, tau, C, ell, htauchart, htauselected, hc, hfactor, hprimitive,
277       htauaxis, haxiscolumns, hCB, hell, hresidue⟩ :=
278     exists_formalDivisorAxisLift_of_selected_correction
279       q Z rows chart zero axis a b u₀ u₁ lambda hqchart hZchart ha hab
280         hqzero hu₀ hZzero hqaxis hZaxis hselected hminor
281   exact ⟨lambda, c, tau, C, ell, rfl, hselected, htauchart, htauselected,
282     hc, hfactor, hprimitive, htauaxis, haxiscolumns, hCB, hell, hresidue⟩
