26 private lemma chart_of_valuation
27     {r : ℕ} (Y : Fin (r + 1) → K) (W : ValuationSubring K) (hY : Y 0 = 1) :
28     ∃ (j : Fin (r + 1)) (s : K), s ≠ 0 ∧ Y j * s = 1 ∧
29       ∀ a, Y a * s ∈ W.toSubring := by
30   classical
31   let S : Set (Fin (r + 1)) := {a | Y a ≠ 0}
32   have hS : S.Nonempty := ⟨0, by simp [S, hY]⟩
33   letI : LE (Fin (r + 1)) :=
34     ⟨fun a b => Y b ≠ 0 ∧ Y a / Y b ∈ W.toSubring⟩
35   have htrans : IsTrans (Fin (r + 1)) LE.le := by
36     constructor
37     intro a b c hab hbc
38     rcases hab with ⟨hb, hab⟩
39     rcases hbc with ⟨hc, hbc⟩
40     refine ⟨hc, ?_⟩
41     have := W.toSubring.mul_mem hab hbc
42     rw [div_eq_mul_inv] at hab hbc ⊢
43     simpa [div_eq_mul_inv, mul_assoc, hb] using this
44   letI : IsTrans (Fin (r + 1)) LE.le := htrans
45   obtain ⟨j, hjS, hjmax⟩ := S.toFinite.exists_maximalFor id S hS
46   have hj0 : Y j ≠ 0 := hjS
47   let s := (Y j)⁻¹
48   refine ⟨j, s, inv_ne_zero hj0, ?_, ?_⟩
49   · simp [s, hj0]
50   · intro a
51     by_cases ha : Y a = 0
52     · rw [ha, zero_mul]
53       exact W.zero_mem
54     have haS : a ∈ S := ha
55     rcases W.mem_or_inv_mem (Y a / Y j) with h | h
56     · simpa [s, div_eq_mul_inv, mul_comm] using h
57     · have h' : Y a / Y j ∈ W.toSubring := by
58         apply (hjmax haS ⟨ha, ?_⟩).2
59         simpa [div_eq_mul_inv, mul_comm] using h
60       simpa [s, div_eq_mul_inv, mul_comm] using h'
61 
62 end Chart

67 private lemma normalized_generates_top
68     {r : ℕ} (y : Fin r → K) (i : Fin r) (s : K) (hs : s ≠ 0)
69     (hgen : IntermediateField.adjoin k (Set.range y) = ⊤) :
70     IntermediateField.adjoin k
71         (Set.range (fun a : Fin (r + 1) => (Fin.cases 1 y a : K) * s) ∪ {y i}) = ⊤ := by
72   apply top_unique
73   rw [← hgen]
74   apply IntermediateField.adjoin_le_iff.2
75   intro a ha
76   rcases ha with ⟨a, rfl⟩
77   apply IntermediateField.mem_adjoin_iff_div.mpr
78   refine ⟨(Fin.cases 1 y (Fin.succ a) : K) * s, ?_, s, ?_, ?_⟩
79   · exact Algebra.subset_adjoin (R := k) (A := K)
80       (Set.mem_union_left _ (Set.mem_range_self _))
81   · have h0 : (Fin.cases 1 y (0 : Fin (r + 1)) : K) * s ∈
82         Algebra.adjoin k
83           (Set.range (fun a : Fin (r + 1) => (Fin.cases 1 y a : K) * s) ∪ {y i}) :=
84       Algebra.subset_adjoin (R := k) (A := K)
85         (Set.mem_union_left _ (Set.mem_range_self (0 : Fin (r + 1))))
86     simpa only [Fin.cases_zero, one_mul] using h0
87   · simp only [Fin.cases_succ]
88     exact (eq_div_iff hs).2 rfl
89 
90 end Generation

96 private theorem exists_valuationSubring_of_transcendental
97     (x : K) (hx : Transcendental k x) :
98     ∃ W : ValuationSubring K, (∀ c : k, algebraMap k K c ∈ W) ∧ x ∈ W ∧ x⁻¹ ∉ W := by
99   classical
100   have hx0 : x ≠ 0 := fun h => hx (h ▸ isAlgebraic_zero)
101   let R : Subring K := (Algebra.adjoin k ({x} : Set K)).toSubring
102   have hxR : x ∈ R := Algebra.subset_adjoin (Set.mem_singleton x)
103   let I : Ideal R := Ideal.span {⟨x, hxR⟩}
104   have hI : I ≠ ⊤ := by
105     intro htop
106     have h1 : (1 : R) ∈ I := htop ▸ Submodule.mem_top
107     rw [Ideal.mem_span_singleton'] at h1
108     obtain ⟨g, hg⟩ := h1
109     have hg' : (g : K) * x = 1 := by
110       have := congrArg Subtype.val hg
111       simpa using this
112     obtain ⟨q, hq⟩ : ∃ q : Polynomial k, Polynomial.aeval x q = (g : K) := by
113       have hmem : (g : K) ∈ Algebra.adjoin k ({x} : Set K) := g.2
114       rw [Algebra.adjoin_singleton_eq_range_aeval] at hmem
115       exact (AlgHom.mem_range _).1 hmem
116     apply hx
117     refine ⟨q * Polynomial.X - 1, ?_, ?_⟩
118     · intro h0
119       have hc := congrArg (fun p : Polynomial k => p.coeff 0) h0
120       simp [Polynomial.mul_coeff_zero] at hc
121     · simp [map_sub, map_mul, hq, hg']
122   obtain ⟨W, hRW, hIW⟩ := Ideal.image_subset_nonunits_valuationSubring I hI
123   have hxW : x ∈ W := hRW hxR
124   refine ⟨W, fun c => hRW (Subalgebra.algebraMap_mem _ c), hxW, ?_⟩
125   intro hinv
126   have hxn : x ∈ W.nonunits := hIW ⟨⟨x, hxR⟩, Ideal.mem_span_singleton_self _, rfl⟩
127   rw [ValuationSubring.mem_nonunits_iff_exists_mem_maximalIdeal] at hxn
128   obtain ⟨hxW', hxm⟩ := hxn
129   rw [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff] at hxm
130   exact hxm (isUnit_iff_exists_inv.2
131     ⟨⟨x⁻¹, hinv⟩, Subtype.ext (mul_inv_cancel₀ hx0)⟩)
132 
133 end Pivot

137 theorem stage2_exists_chart_normalization
138     {k K : Type u} [Field k] [CharZero k] [Field K] [Algebra k K]
139     {r : ℕ} (y : Fin r → K) (i : Fin r)
140     (hgen : IntermediateField.adjoin k (Set.range y) = ⊤) (hx : Transcendental k (y i)) :
141     ∃ (j : Fin (r + 1)) (s : K) (A : Subalgebra k K),
142       s ≠ 0 ∧ (Fin.cases 1 y j : K) * s = 1 ∧
143       (∀ a, (Fin.cases 1 y a : K) * s ∈ A) ∧ y i ∈ A ∧
144       Algebra.FiniteType k A ∧ IsIntegrallyClosedIn A K ∧ IsFractionRing A K ∧
145       (∀ a : A, IsIntegral (Algebra.adjoin k
146         (Set.range (fun a : Fin (r + 1) => (Fin.cases 1 y a : K) * s) ∪ {y i})) (a : K)) ∧
147       (y i)⁻¹ ∉ A := by
148   classical
149   obtain ⟨W, hkW, hxiW, hxiInvW⟩ := exists_valuationSubring_of_transcendental (y i) hx
150   let Y : Fin (r + 1) → K := Fin.cases 1 y
151   obtain ⟨j, s, hs, hjs, hYsW⟩ := chart_of_valuation Y W (by simp [Y])
152   let C₀ : Subalgebra k K :=
153     Algebra.adjoin k (Set.range (fun a : Fin (r + 1) => Y a * s) ∪ {y i})
154   let SW : Subalgebra k K :=
155     { W.toSubring with algebraMap_mem' := hkW }
156   have hC₀W : C₀ ≤ SW := by
157     apply Algebra.adjoin_le
158     rintro z (⟨a, rfl⟩ | rfl)
159     · exact hYsW a
160     · exact hxiW
161   have htop : IntermediateField.adjoin k
162       (Set.range (fun a : Fin (r + 1) => Y a * s) ∪ {y i}) = ⊤ :=
163     normalized_generates_top y i s hs hgen
164   letI hC₀fr : IsFractionRing C₀ K := IsFractionRing.of_field C₀ K fun z => by
165     have hz : z ∈ IntermediateField.adjoin k
166         (Set.range (fun a : Fin (r + 1) => Y a * s) ∪ {y i}) := by
167       rw [htop]
168       exact Set.mem_univ z
169     rw [IntermediateField.mem_adjoin_iff_div] at hz
170     rcases hz with ⟨a, ha, b, hb, hab⟩
171     exact ⟨⟨a, ha⟩, ⟨b, hb⟩, hab⟩
172   let A₀ : Subalgebra C₀ K := integralClosure C₀ K
173   let A : Subalgebra k K := A₀.restrictScalars k
174   have hC₀A : C₀ ≤ A := fun z hz => A₀.algebraMap_mem ⟨z, hz⟩
175   letI : IsIntegrallyClosedIn W.toSubring K :=
176     inferInstanceAs (IsIntegrallyClosedIn W K)
177   have hA₀W : A₀.toSubring ≤ W.toSubring := by
178     change (integralClosure C₀ K).toSubring ≤ W.toSubring
179     apply (Subring.integralClosure_subring_le_iff (S := C₀) (T := W.toSubring)).2
180     exact hC₀W
181   have hyA : y i ∈ A := by
182     exact A₀.algebraMap_mem ⟨y i, Algebra.subset_adjoin (Set.mem_union_right _ (Set.mem_singleton _))⟩
183   have hnormA : ∀ a, Y a * s ∈ A := by
184     intro a
185     exact A₀.algebraMap_mem
186       ⟨Y a * s, Algebra.subset_adjoin (Set.mem_union_left _ (Set.mem_range_self a))⟩
187   have hinvA : (y i)⁻¹ ∉ A := fun h => hxiInvW (hA₀W h)
188   haveI hC₀ft : Algebra.FiniteType k C₀ := Algebra.FiniteType.adjoin_of_finite
189     (Set.Finite.union (Set.finite_range _) (Set.finite_singleton _))
190   let D : Subalgebra C₀ (FractionRing C₀) := integralClosure C₀ (FractionRing C₀)
191   haveI hDfin : Module.Finite C₀ D :=
192     finite_normalization_of_fg_domain k C₀ D
193   let e : FractionRing C₀ ≃ₐ[C₀] K :=
194     IsLocalization.algEquiv C₀⁰ (FractionRing C₀) K
195   haveI hA₀fin : Module.Finite C₀ A₀ :=
196     Module.Finite.equiv e.mapIntegralClosure.toLinearEquiv
197   have hsub : A.toSubring = A₀.toSubring := rfl
198   let er : A ≃+* A₀ := RingEquiv.subringCongr hsub
199   letI : Algebra C₀ A :=
200     (er.symm.toRingHom.comp (algebraMap C₀ A₀)).toAlgebra
201   let ea : A₀ ≃ₐ[C₀] A := AlgEquiv.ofRingEquiv (f := er.symm) (fun _ => rfl)
202   letI : IsScalarTower k C₀ A := IsScalarTower.of_algebraMap_eq fun _ => by
203     apply Subtype.ext
204     rfl
205   letI : IsScalarTower C₀ A K := IsScalarTower.of_algebraMap_eq fun _ => rfl
206   haveI hAfin : Module.Finite C₀ A := Module.Finite.equiv ea.toLinearEquiv
207   have hAft : Algebra.FiniteType k A :=
208     Algebra.FiniteType.trans hC₀ft (inferInstance : Algebra.FiniteType C₀ A)
209   have hAic : IsIntegrallyClosedIn A K := by
210     apply Subring.isIntegrallyClosedIn_iff.mpr
211     intro x hx
212     change x ∈ integralClosure C₀ K
213     exact isIntegral_trans (R := C₀) (A := A) x hx
214   have hAfr : IsFractionRing A K := IsFractionRing.of_field A K fun z => by
215     obtain ⟨a, b, hb, hab⟩ := IsFractionRing.div_surjective C₀ z
216     refine ⟨⟨a, hC₀A a.2⟩, ⟨b, hC₀A b.2⟩, ?_⟩
217     change z = (a : K) / (b : K)
218     exact hab.symm
219   refine ⟨j, s, A, hs, hjs, hnormA, hyA, hAft, hAic, hAfr, ?_, hinvA⟩
220   intro a
221   exact a.2
