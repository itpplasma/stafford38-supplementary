14 /-- Every prime in a formally étale principal open has a formally étale
15 local ring for the original coordinate-algebra map. -/
16 theorem formallyEtale_atPrime_of_away
17     {R B : Type*} [CommRing R] [CommRing B] [Algebra R B]
18     (f : B) (M : Ideal B) [M.IsPrime]
19     (hfM : f ∉ M)
20     [Algebra.FormallyEtale R (Localization.Away f)] :
21     Algebra.FormallyEtale R (Localization.AtPrime M) := by
22   have hopen : ↑(PrimeSpectrum.basicOpen f) ⊆ Algebra.etaleLocus R B :=
23     (Algebra.basicOpen_subset_etaleLocus_iff (R := R) (A := B)).mpr inferInstance
24   exact hopen (show (⟨M, inferInstance⟩ : PrimeSpectrum B) ∈ PrimeSpectrum.basicOpen f from hfM)

26 /-- Choose a ground point on the divisor where both the original coordinate
27 map is étale and an additional regularity function does not vanish. -/
28 theorem exists_standardEtale_neighborhood_and_formallyEtale_ground_point_avoiding
29     {k R B : Type*} [Field k] [IsAlgClosed k] [CommRing R] [CommRing B]
30     [Algebra k R] [Algebra R B] [Algebra k B] [IsScalarTower k R B]
31     [Algebra.FiniteType k B] [Algebra.FinitePresentation R B]
32     (p : Ideal B) [p.IsPrime] (g : B) (hgp : g ∉ p)
33     (hEt : Algebra.FormallyEtale R (Localization.AtPrime p)) :
34     ∃ (f : B) (M : Ideal B) (hM : M.IsMaximal),
35       f ∉ p ∧ p ≤ M ∧ f ∉ M ∧ g ∉ M ∧
36       Algebra.IsStandardEtale R (Localization.Away f) ∧
37       Nonempty ((B ⧸ M) ≃ₐ[k] k) ∧
38       (letI : M.IsMaximal := hM
39        Algebra.FormallyEtale R (Localization.AtPrime M)) := by
40   let : Algebra.IsEtaleAt R p := hEt
41   obtain ⟨f, hfp, hstd⟩ := Algebra.IsEtaleAt.exists_isStandardEtale (R := R) p
42   have hfgp : f * g ∉ p := by
43     intro h
44     rcases (inferInstance : p.IsPrime).mem_or_mem h with hf | hg
45     · exact hfp hf
46     · exact hgp hg
47   obtain ⟨M, hM, hpM, hfgM, hres⟩ :=
48     Stafford38.Geometry.ClosedPointOnDivisor.exists_ground_closed_point_over_prime_avoiding
49       (k := k) (R := B) p inferInstance (f * g) hfgp
50   have hfM : f ∉ M := fun hf => hfgM (M.mul_mem_right g hf)
51   have hgM : g ∉ M := fun hg => hfgM (M.mul_mem_left f hg)
52   let : M.IsMaximal := hM
53   let : Algebra.IsStandardEtale R (Localization.Away f) := hstd
54   let : Algebra.FormallyEtale R (Localization.Away f) := inferInstance
55   exact ⟨f, M, hM, hfp, hpM, hfM, hgM, hstd, hres,
56     formallyEtale_atPrime_of_away f M hfM⟩

58 /-- A formally etale local map from the prescribed coordinate algebra has a standard-etale
59 neighborhood of the divisor prime, and that neighborhood contains a ground-rational closed
60 point when the target is finite type over an algebraically closed field.  The coordinate map is
61 the original algebra map `R → B`; this lemma does not alter or choose a replacement chart. -/
62 theorem exists_standardEtale_neighborhood_and_ground_point
63     {k R B : Type*} [Field k] [IsAlgClosed k] [CommRing R] [CommRing B]
64     [Algebra k R] [Algebra R B] [Algebra k B] [IsScalarTower k R B]
65     [Algebra.FiniteType k B] [Algebra.FinitePresentation R B]
66     (p : Ideal B) [p.IsPrime]
67     (hEt : Algebra.FormallyEtale R (Localization.AtPrime p)) :
68     ∃ (f : B) (M : Ideal B), f ∉ p ∧ M.IsMaximal ∧ p ≤ M ∧ f ∉ M ∧
69       Algebra.IsStandardEtale R (Localization.Away f) ∧
70       Nonempty ((B ⧸ M) ≃ₐ[k] k) := by
71   have h1p : (1 : B) ∉ p := Ideal.one_notMem p
72   obtain ⟨f, M, hM, hfp, hpM, hfM, _, hstd, hres, _⟩ :=
73     exists_standardEtale_neighborhood_and_formallyEtale_ground_point_avoiding
74       (k := k) (R := R) (B := B) p 1 h1p hEt
75   exact ⟨f, M, hfp, hM, hpM, hfM, hstd, hres⟩
