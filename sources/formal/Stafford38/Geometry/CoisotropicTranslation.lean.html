8 /-!
9 # Poisson closure and vertical translation
10 
11 This file proves a polynomial statement used by the coisotropic part of the
12 Stafford argument.  If a Poisson ideal vanishes at a point of the zero section
13 and contains a polynomial in the base variables, then it also vanishes after
14 the fibre coordinate is translated by any scalar multiple of the differential
15 of that base polynomial at the point.
16 
17 Only polynomial rings, ideals, partial derivatives, and evaluation maps occur
18 here.  In particular, no variety, characteristic support, conormal bundle, or
19 geometric integration theorem is represented by these declarations.
20 -/

30 /-- Embed a polynomial in the base coordinates into the phase-space ring. -/
31 def baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n :=
32   MvPolynomial.rename Sum.inl
33 
34 /-- Evaluate the differential of a base polynomial at a base point. -/
35 def differentialAt (y : Fin n → k) (f : MvPolynomial (Fin n) k) (i : Fin n) : k :=
36   MvPolynomial.eval y (MvPolynomial.pderiv i f)
37 
38 /-- The polynomial obtained by evaluating a phase-space polynomial along the
39 arbitrary affine fibre line `(y, xi + t v)`. -/
40 def affineFibreLinePolynomial
41     (y xi v : Fin n → k) (g : SymbolRing k n) : Polynomial k :=
42   MvPolynomial.aeval
43     (Sum.elim (fun i => Polynomial.C (y i))
44       (fun i => Polynomial.C (xi i) + Polynomial.C (v i) * Polynomial.X)) g
45 
46 /-- The point on the affine fibre line `(y, xi + t v)`. -/
47 def affineFibreTranslatePoint
48     (y xi v : Fin n → k) (t : k) : PhaseVar n → k
49   | Sum.inl i => y i
50   | Sum.inr i => xi i + t * v i
51 
52 @[simp] theorem affineFibreTranslatePoint_inl
53     (y xi v : Fin n → k) (t : k) (i : Fin n) :
54     affineFibreTranslatePoint y xi v t (Sum.inl i) = y i := rfl
55 
56 @[simp] theorem affineFibreTranslatePoint_inr
57     (y xi v : Fin n → k) (t : k) (i : Fin n) :
58     affineFibreTranslatePoint y xi v t (Sum.inr i) = xi i + t * v i := rfl

60 /-- The zero-section point over `y`. -/
61 def zeroSectionPoint (y : Fin n → k) : PhaseVar n → k
62   | Sum.inl i => y i
63   | Sum.inr _ => 0

65 /-- The zero-fibre specialization of `affineFibreTranslatePoint`, translating
66 `(y, 0)` by `t df_y`. -/
67 abbrev differentialTranslatePoint
68     (y : Fin n → k) (f : MvPolynomial (Fin n) k) (t : k) : PhaseVar n → k :=
69   affineFibreTranslatePoint y (fun _ => 0) (differentialAt y f) t

71 theorem differentialTranslatePoint_eq_affineFibreTranslatePoint_zero
72     (y : Fin n → k) (f : MvPolynomial (Fin n) k) (t : k) :
73     differentialTranslatePoint y f t =
74       affineFibreTranslatePoint y (fun _ => 0) (differentialAt y f) t := rfl

76 /-- The polynomial line `t ↦ g(y, t v)` in the fibre over `y`, the

86 /-- Directional derivative in the fibre variables. -/
87 def verticalDeriv (v : Fin n → k) (g : SymbolRing k n) : SymbolRing k n :=
88   ∑ i, MvPolynomial.C (v i) * MvPolynomial.pderiv (Sum.inr i) g

99     verticalDeriv v (g * h) = verticalDeriv v g * h + g * verticalDeriv v h := by

158 /-- Evaluating the affine fibre-line polynomial at time `t` is evaluation at
159 `(y, xi + t v)`. -/
160 theorem eval_affineFibreLinePolynomial (y xi v : Fin n → k)
161     (g : SymbolRing k n) (t : k) :
162     Polynomial.eval t (affineFibreLinePolynomial y xi v g) =
163       MvPolynomial.eval (Sum.elim y (fun i => xi i + t * v i)) g := by
164   change Polynomial.aeval t
165       (MvPolynomial.aeval (Sum.elim
166         (fun i => Polynomial.C (y i))
167         (fun i => Polynomial.C (xi i) + Polynomial.C (v i) * Polynomial.X)) g) = _
168   rw [MvPolynomial.comp_aeval_apply]
169   change MvPolynomial.aeval _ g = MvPolynomial.aeval _ g
170   apply DFunLike.congr_fun
171   apply MvPolynomial.algHom_ext
172   intro i
173   rcases i with i | i
174   · rw [MvPolynomial.aeval_X, MvPolynomial.aeval_X]
175     simp
176   · rw [MvPolynomial.aeval_X, MvPolynomial.aeval_X]
177     simp only [Sum.elim_inr, map_add, Polynomial.aeval_C, Polynomial.aeval_mul,
178       Polynomial.aeval_X]
179     simp
180     ring

182 theorem derivative_fibreLinePolynomial
183     (y v : Fin n → k) (g : SymbolRing k n) :
184     Polynomial.derivative (fibreLinePolynomial y v g) =
185       fibreLinePolynomial y v (verticalDeriv v g) := by
186   induction g using MvPolynomial.induction_on with
187   | C a => simp [verticalDeriv]
188   | add p q hp hq =>
189       simp only [fibreLinePolynomial_add, Polynomial.derivative_add, hp, hq]
190       rw [verticalDeriv_add, fibreLinePolynomial_add]
191   | mul_X p i hp =>
192       rcases i with i | i
193       · simp only [fibreLinePolynomial_mul, fibreLinePolynomial_X_base, Polynomial.derivative_mul,
194           Polynomial.derivative_C, zero_mul, add_zero, hp]
195         simp [verticalDeriv_mul]
196       · simp only [fibreLinePolynomial_mul, fibreLinePolynomial_X_fibre, Polynomial.derivative_mul,
197           Polynomial.derivative_C, Polynomial.derivative_X, zero_mul, zero_add, mul_one, hp]
198         simp [verticalDeriv_mul]

264 /-- Iteration of bracketing on the left by one base equation. -/
265 def hamiltonIter (f : MvPolynomial (Fin n) k) : ℕ → SymbolRing k n → SymbolRing k n
266   | 0, g => g
267   | m + 1, g => poissonBracket (baseLift f) (hamiltonIter f m g)
268 
269 theorem iterate_derivative_fibreLinePolynomial
270     (y : Fin n → k) (f : MvPolynomial (Fin n) k)
271     (g : SymbolRing k n) (m : ℕ) :
272     (Polynomial.derivative^[m])
273         (fibreLinePolynomial y (differentialAt y f) g) =
274       fibreLinePolynomial y (differentialAt y f) (hamiltonIter f m g) := by
275   induction m with
276   | zero => rfl
277   | succ m hm =>
278       rw [Function.iterate_succ_apply', hm, derivative_fibreLinePolynomial,
279         ← fibreLinePolynomial_poissonBracket_baseLift]
280       rfl
281 
282 theorem hamiltonIter_mem
283     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
284     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
285     (g : SymbolRing k n) (hg : g ∈ J) :
286     ∀ m, hamiltonIter f m g ∈ J := by
287   intro m
288   induction m with
289   | zero => exact hg
290   | succ m hm => exact hJ (baseLift f) hf (hamiltonIter f m g)

292 theorem polynomial_eq_zero_of_eval_iterate_derivative_zero
293     [CharZero k]
294     (q : Polynomial k)
295     (h : ∀ m, Polynomial.eval 0 ((Polynomial.derivative^[m]) q) = 0) :
296     q = 0 := by
297   ext m
298   have hm0 : ((Polynomial.derivative^[m]) q).coeff 0 = 0 := by
299     rw [Polynomial.coeff_zero_eq_eval_zero]
300     exact h m
301   have hm : (Nat.factorial m : k) * q.coeff m = 0 := by
302     simpa only [Polynomial.coeff_iterate_derivative, zero_add,
303       Nat.descFactorial_self, nsmul_eq_mul] using hm0
304   have hc : q.coeff m = 0 :=
305     (mul_eq_zero.mp hm).resolve_left (by exact_mod_cast Nat.factorial_ne_zero m)
306   simpa using hc

310     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)

321 /-! ## The exact translation statement -/
322 
323 /--
324 A zero-section common zero of a Poisson ideal remains a common zero after
325 vertical translation by `t df_y`, for every scalar `t`, whenever the base
326 polynomial `f` belongs to the ideal.
327 
328 The radical hypothesis customary in the geometric application is unnecessary
329 for this polynomial implication; see the wrapper below.
330 -/
331 theorem zeroSection_stable_under_differential_translation
332     [CharZero k]
333     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
334     (y : Fin n → k)
335     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
336     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
337     (g : SymbolRing k n) (hg : g ∈ J) (t : k) :
338     MvPolynomial.eval (differentialTranslatePoint y f t) g = 0 := by
339   have heval :
340       MvPolynomial.eval (differentialTranslatePoint y f t) g =
341         MvPolynomial.eval (Sum.elim y (fun i => t * differentialAt y f i)) g := by
342     apply MvPolynomial.eval₂_congr
343     intro i c hi hc
344     rcases i with i | i <;> simp [differentialTranslatePoint, affineFibreTranslatePoint]
345   rw [heval, ← eval_fibreLinePolynomial]
346   rw [fibreLinePolynomial_eq_zero_of_poisson J hJ y hzero f hf g hg]
347   exact Polynomial.eval_zero
348 
349 /-- The same statement with the radical hypothesis carried explicitly. -/
350 theorem radicalPoisson_zeroSection_stable_under_differential_translation
351     [CharZero k]
352     (J : Ideal (SymbolRing k n)) (_hJrad : J.IsRadical) (hJ : IsPoisson J)
353     (y : Fin n → k)
354     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
355     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
356     (g : SymbolRing k n) (hg : g ∈ J) (t : k) :
357     MvPolynomial.eval (differentialTranslatePoint y f t) g = 0 :=
358   zeroSection_stable_under_differential_translation J hJ y hzero f hf g hg t
