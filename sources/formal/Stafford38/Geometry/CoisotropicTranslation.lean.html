6 /-!
7 # Poisson closure and vertical translation
8 
9 This file proves a polynomial statement used by the coisotropic part of the
10 Stafford argument.  If a Poisson ideal vanishes at a point of the zero section
11 and contains a polynomial in the base variables, then it also vanishes after
12 the fibre coordinate is translated by any scalar multiple of the differential
13 of that base polynomial at the point.
14 
15 Only polynomial rings, ideals, partial derivatives, and evaluation maps occur
16 here.  In particular, no variety, characteristic support, conormal bundle, or
17 geometric integration theorem is represented by these declarations.
18 -/

28 /-- Embed a polynomial in the base coordinates into the phase-space ring. -/
29 def baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n :=
30   MvPolynomial.rename Sum.inl
31 
32 /-- Evaluate the differential of a base polynomial at a base point. -/
33 def differentialAt (y : Fin n → k) (f : MvPolynomial (Fin n) k) (i : Fin n) : k :=
34   MvPolynomial.eval y (MvPolynomial.pderiv i f)
35 
36 /-- The polynomial obtained by evaluating a phase-space polynomial along the
37 arbitrary affine fibre line `(y, xi + t v)`. -/
38 def affineFibreLinePolynomial
39     (y xi v : Fin n → k) (g : SymbolRing k n) : Polynomial k :=
40   MvPolynomial.aeval
41     (Sum.elim (fun i => Polynomial.C (y i))
42       (fun i => Polynomial.C (xi i) + Polynomial.C (v i) * Polynomial.X)) g
43 
44 /-- The point on the affine fibre line `(y, xi + t v)`. -/
45 def affineFibreTranslatePoint
46     (y xi v : Fin n → k) (t : k) : PhaseVar n → k
47   | Sum.inl i => y i
48   | Sum.inr i => xi i + t * v i
49 
50 @[simp] theorem affineFibreTranslatePoint_inl
51     (y xi v : Fin n → k) (t : k) (i : Fin n) :
52     affineFibreTranslatePoint y xi v t (Sum.inl i) = y i := rfl

54 @[simp] theorem affineFibreTranslatePoint_inr
55     (y xi v : Fin n → k) (t : k) (i : Fin n) :
56     affineFibreTranslatePoint y xi v t (Sum.inr i) = xi i + t * v i := rfl

58 /-- The zero-section point over `y`. -/
59 def zeroSectionPoint (y : Fin n → k) : PhaseVar n → k
60   | Sum.inl i => y i
61   | Sum.inr _ => 0

63 /-- The zero-fibre specialization of `affineFibreTranslatePoint`, translating
64 `(y, 0)` by `t df_y`. -/
65 abbrev differentialTranslatePoint
66     (y : Fin n → k) (f : MvPolynomial (Fin n) k) (t : k) : PhaseVar n → k :=
67   affineFibreTranslatePoint y (fun _ => 0) (differentialAt y f) t

69 theorem differentialTranslatePoint_eq_affineFibreTranslatePoint_zero
70     (y : Fin n → k) (f : MvPolynomial (Fin n) k) (t : k) :
71     differentialTranslatePoint y f t =
72       affineFibreTranslatePoint y (fun _ => 0) (differentialAt y f) t := rfl

74 /-- The polynomial line `t ↦ g(y, t v)` in the fibre over `y`, the

84 /-- Directional derivative in the fibre variables. -/
85 def verticalDeriv (v : Fin n → k) (g : SymbolRing k n) : SymbolRing k n :=
86   ∑ i, MvPolynomial.C (v i) * MvPolynomial.pderiv (Sum.inr i) g

156 /-- Evaluating the affine fibre-line polynomial at time `t` is evaluation at
157 `(y, xi + t v)`. -/
158 theorem eval_affineFibreLinePolynomial (y xi v : Fin n → k)
159     (g : SymbolRing k n) (t : k) :
160     Polynomial.eval t (affineFibreLinePolynomial y xi v g) =
161       MvPolynomial.eval (Sum.elim y (fun i => xi i + t * v i)) g := by
162   change Polynomial.aeval t
163       (MvPolynomial.aeval (Sum.elim
164         (fun i => Polynomial.C (y i))
165         (fun i => Polynomial.C (xi i) + Polynomial.C (v i) * Polynomial.X)) g) = _
166   rw [MvPolynomial.comp_aeval_apply]
167   change MvPolynomial.aeval _ g = MvPolynomial.aeval _ g
168   apply DFunLike.congr_fun
169   apply MvPolynomial.algHom_ext
170   intro i
171   rcases i with i | i
172   · rw [MvPolynomial.aeval_X, MvPolynomial.aeval_X]
173     simp
174   · rw [MvPolynomial.aeval_X, MvPolynomial.aeval_X]
175     simp only [Sum.elim_inr, map_add, Polynomial.aeval_C, Polynomial.aeval_mul,
176       Polynomial.aeval_X]
177     simp
178     ring

180 theorem derivative_fibreLinePolynomial
181     (y v : Fin n → k) (g : SymbolRing k n) :
182     Polynomial.derivative (fibreLinePolynomial y v g) =
183       fibreLinePolynomial y v (verticalDeriv v g) := by
184   induction g using MvPolynomial.induction_on with
185   | C a => simp [verticalDeriv]
186   | add p q hp hq =>
187       simp only [fibreLinePolynomial_add, Polynomial.derivative_add, hp, hq]
188       rw [verticalDeriv_add, fibreLinePolynomial_add]
189   | mul_X p i hp =>
190       rcases i with i | i
191       · simp only [fibreLinePolynomial_mul, fibreLinePolynomial_X_base, Polynomial.derivative_mul,
192           Polynomial.derivative_C, zero_mul, add_zero, hp]
193         simp [verticalDeriv_mul]
194       · simp only [fibreLinePolynomial_mul, fibreLinePolynomial_X_fibre, Polynomial.derivative_mul,
195           Polynomial.derivative_C, Polynomial.derivative_X, zero_mul, zero_add, mul_one, hp]
196         simp [verticalDeriv_mul]

262 /-- Iteration of bracketing on the left by one base equation. -/
263 def hamiltonIter (f : MvPolynomial (Fin n) k) : ℕ → SymbolRing k n → SymbolRing k n
264   | 0, g => g
265   | m + 1, g => poissonBracket (baseLift f) (hamiltonIter f m g)
266 
267 theorem iterate_derivative_fibreLinePolynomial
268     (y : Fin n → k) (f : MvPolynomial (Fin n) k)
269     (g : SymbolRing k n) (m : ℕ) :
270     (Polynomial.derivative^[m])
271         (fibreLinePolynomial y (differentialAt y f) g) =
272       fibreLinePolynomial y (differentialAt y f) (hamiltonIter f m g) := by
273   induction m with
274   | zero => rfl
275   | succ m hm =>
276       rw [Function.iterate_succ_apply', hm, derivative_fibreLinePolynomial,
277         ← fibreLinePolynomial_poissonBracket_baseLift]
278       rfl

280 theorem hamiltonIter_mem
281     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
282     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
283     (g : SymbolRing k n) (hg : g ∈ J) :
284     ∀ m, hamiltonIter f m g ∈ J := by
285   intro m
286   induction m with
287   | zero => exact hg
288   | succ m hm => exact hJ (baseLift f) hf (hamiltonIter f m g)

290 theorem polynomial_eq_zero_of_eval_iterate_derivative_zero
291     [CharZero k]
292     (q : Polynomial k)
293     (h : ∀ m, Polynomial.eval 0 ((Polynomial.derivative^[m]) q) = 0) :
294     q = 0 := by
295   ext m
296   have hm0 : ((Polynomial.derivative^[m]) q).coeff 0 = 0 := by
297     rw [Polynomial.coeff_zero_eq_eval_zero]
298     exact h m
299   have hm : (Nat.factorial m : k) * q.coeff m = 0 := by
300     simpa only [Polynomial.coeff_iterate_derivative, zero_add,
301       Nat.descFactorial_self, nsmul_eq_mul] using hm0
302   have hc : q.coeff m = 0 :=
303     (mul_eq_zero.mp hm).resolve_left (by exact_mod_cast Nat.factorial_ne_zero m)
304   simpa using hc

319 /-! ## The exact translation statement -/
320 
321 /--
322 A zero-section common zero of a Poisson ideal remains a common zero after
323 vertical translation by `t df_y`, for every scalar `t`, whenever the base
324 polynomial `f` belongs to the ideal.
325 
326 The radical hypothesis customary in the geometric application is unnecessary
327 for this polynomial implication; see the wrapper below.
328 -/
329 theorem zeroSection_stable_under_differential_translation
330     [CharZero k]
331     (J : Ideal (SymbolRing k n)) (hJ : IsPoisson J)
332     (y : Fin n → k)
333     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
334     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
335     (g : SymbolRing k n) (hg : g ∈ J) (t : k) :
336     MvPolynomial.eval (differentialTranslatePoint y f t) g = 0 := by
337   have heval :
338       MvPolynomial.eval (differentialTranslatePoint y f t) g =
339         MvPolynomial.eval (Sum.elim y (fun i => t * differentialAt y f i)) g := by
340     apply MvPolynomial.eval₂_congr
341     intro i c hi hc
342     rcases i with i | i <;> simp [differentialTranslatePoint, affineFibreTranslatePoint]
343   rw [heval, ← eval_fibreLinePolynomial]
344   rw [fibreLinePolynomial_eq_zero_of_poisson J hJ y hzero f hf g hg]
345   exact Polynomial.eval_zero
346 
347 /-- The same statement with the radical hypothesis carried explicitly. -/
348 theorem radicalPoisson_zeroSection_stable_under_differential_translation
349     [CharZero k]
350     (J : Ideal (SymbolRing k n)) (_hJrad : J.IsRadical) (hJ : IsPoisson J)
351     (y : Fin n → k)
352     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
353     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
354     (g : SymbolRing k n) (hg : g ∈ J) (t : k) :
355     MvPolynomial.eval (differentialTranslatePoint y f t) g = 0 :=
356   zeroSection_stable_under_differential_translation J hJ y hzero f hf g hg t
