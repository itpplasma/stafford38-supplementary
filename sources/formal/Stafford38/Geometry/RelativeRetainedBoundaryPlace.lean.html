7 # Relative retained boundary places
8 
9 The divisorial construction for a transcendental element can retain the
10 actual local map from the coordinate DVR `E[X]_(X)`.  This is stronger than
11 the bare `DiscreteBoundaryRefinement`: it preserves the coefficient field and
12 the algebra structure needed by the completed-DVR coefficient section.
13 
14 No projective normalization, completed projective coordinates, or tangent
15 frame is constructed here.
16 -/

40 /-- A retained boundary place for `x`, including the relative coefficient
41 field and the exact ambient algebra structure under which the local
42 coordinate maps to `x`. -/
43 structure Data
44     (k K : Type u) [Field k] [Field K] [Algebra k K]
45     (x : K) where
46   coefficientField : IntermediateField k K
47   coordinate_transcendental : Transcendental coefficientField x
48   ambientAlgebra : Algebra (SourceDVR coefficientField) K
49   coefficientTower :
50     letI : Algebra (SourceDVR coefficientField) K := ambientAlgebra
51     IsScalarTower coefficientField (SourceDVR coefficientField) K
52   coordinate_eq :
53     letI : Algebra (SourceDVR coefficientField) K := ambientAlgebra
54     algebraMap (SourceDVR coefficientField) K
55         (algebraMap (Polynomial coefficientField)
56           (SourceDVR coefficientField) Polynomial.X) = x
57   place :
58     letI : Algebra (SourceDVR coefficientField) K := ambientAlgebra
59     RetainedDVRPlace (SourceDVR coefficientField) (L := K)
60       (algebraMap (Polynomial coefficientField)
61         (SourceDVR coefficientField) Polynomial.X)
62 
63 /-- The retained parameter is the selected function-field coordinate. -/
64 theorem Data.parameter_eq_coordinate
65     {k K : Type u} [Field k] [Field K] [Algebra k K]
66     {x : K} (W : Data k K x) :
67     letI : Algebra (SourceDVR W.coefficientField) K := W.ambientAlgebra
68     (W.place.parameter : K) = x := by
69   letI : Algebra (SourceDVR W.coefficientField) K := W.ambientAlgebra
70   exact W.place.parameter_eq.trans W.coordinate_eq
71 
72 /-- Forgetting the retained source map recovers the earlier discrete boundary
73 refinement, with the same valuation ring and parameter. -/
74 def Data.toDiscreteBoundaryRefinement
75     {k K : Type u} [Field k] [Field K] [Algebra k K]
76     {x : K} (W : Data k K x) :
77     letI : Algebra (SourceDVR W.coefficientField) K := W.ambientAlgebra
78     DiscreteBoundaryRefinement k x := by
79   letI : Algebra (SourceDVR W.coefficientField) K := W.ambientAlgebra
80   exact {
81     valuation := W.place.valuation
82     isDiscrete := W.place.isDiscrete
83     coordinate := W.place.parameter
84     coordinate_eq := W.parameter_eq_coordinate
85     coordinate_ne := W.place.parameter_ne
86     coordinate_nonunit := W.place.parameter_nonunit
87   }

89 /-- The completion of the retained component DVR is literally a power-series
90 ring over its residue field.  This is the completion, not the original DVR. -/
91 def Data.completedPowerSeriesEquiv
92     {k K : Type u} [Field k] [CharZero k]
93     [Field K] [Algebra k K]
94     {x : K} (W : Data k K x) :
95     letI : Algebra (SourceDVR W.coefficientField) K := W.ambientAlgebra
96     let V := W.place.valuation.toSubring
97     letI : IsDiscreteValuationRing V := W.place.isDiscrete
98     letI : Algebra W.coefficientField V :=
99       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
100     PowerSeries (ResidueField V) ≃+*
101       AdicCompletion (IsLocalRing.maximalIdeal V) V := by
102   letI : Algebra (SourceDVR W.coefficientField) K := W.ambientAlgebra
103   let V := W.place.valuation.toSubring
104   letI : IsDiscreteValuationRing V := W.place.isDiscrete
105   letI : Algebra W.coefficientField V :=
106     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
107   exact completedDVRPowerSeriesEquivOfSurjective W.coefficientField
108     V
109     (relativeResidue_isSeparable W.coefficientField W.place)
110     (completedDVRPowerSeriesMap_surjective W.coefficientField
111       V
112       (relativeResidue_isSeparable W.coefficientField W.place))

114 /-- A finitely generated characteristic-zero function field and a selected
115 transcendental element admit a retained relative boundary place. -/
116 theorem exists_data_of_fg_charZero
117     (k : Type u) [Field k] [CharZero k]
118     {K : Type u} [Field K] [Algebra k K]
119     (hfg : (⊤ : IntermediateField k K).FG)
120     (x : K) (hx : Transcendental k x) :
121     Nonempty (Data k K x) := by
122   obtain ⟨s, E, F, hxs, hs, hE, hF, hrestrict, hxE, hxF,
123       hfin, hsep⟩ :=
124     exists_relative_finite_separable_tower_of_charZero k hfg x hx
125   subst F
126   let R := SourceDVR E
127   let L := FractionRing R
128   let F := IntermediateField.adjoin E ({x} : Set K)
129   let e : L ≃ₐ[E] F := coordinateLocalFractionEquivAdjoin E x hxE
130   letI : IsScalarTower E R L :=
131     IsScalarTower.of_algebraMap_eq fun _ => rfl
132   let algRF : Algebra R F :=
133     (e.toRingEquiv.toRingHom.comp (algebraMap R L)).toAlgebra
134   letI : Algebra R F := algRF
135   let towerERF : IsScalarTower E R F :=
136     IsScalarTower.of_algebraMap_eq fun a => by
137       change algebraMap E F a = e (algebraMap R L (algebraMap E R a))
138       rw [← IsScalarTower.algebraMap_apply E R L]
139       exact (e.commutes a).symm
140   letI : IsScalarTower E R F := towerERF
141   let eR : L ≃ₐ[R] F :=
142     { e.toRingEquiv with
143       commutes' := fun _ => rfl }
144   letI : IsFractionRing R F :=
145     IsLocalization.isLocalization_of_algEquiv (nonZeroDivisors R) eR
146   let algRK : Algebra R K :=
147     ((algebraMap F K).comp (algebraMap R F)).toAlgebra
148   letI : Algebra R K := algRK
149   let coefficientTower : IsScalarTower E R K :=
150     IsScalarTower.of_algebraMap_eq fun a => by
151       change algebraMap E K a =
152         algebraMap F K (algebraMap R F (algebraMap E R a))
153       rw [← IsScalarTower.algebraMap_apply E R F]
154       exact IsScalarTower.algebraMap_apply E F K a
155   letI : IsScalarTower E R K := coefficientTower
156   letI : IsScalarTower R F K :=
157     IsScalarTower.of_algebraMap_eq fun _ => rfl
158   letI : FiniteDimensional F K := hfin
159   letI : Algebra.IsSeparable F K := hsep
160   let q : R := algebraMap (Polynomial E) R Polynomial.X
161   have hq_ne : q ≠ 0 := by
162     intro hq
163     apply Polynomial.X_ne_zero (R := E)
164     apply IsLocalization.injective R
165       (coordinateZeroPrime E).primeCompl_le_nonZeroDivisors
166     simpa only [q, map_zero] using hq
167   have hq_nonunit : ¬IsUnit q := by
168     rw [← mem_nonunits_iff, ← mem_maximalIdeal]
169     exact (IsLocalization.AtPrime.to_map_mem_maximal_iff
170       R (coordinateZeroPrime E) Polynomial.X).2
171         (Ideal.mem_span_singleton_self Polynomial.X)
172   have hcoordinate : algebraMap R K q = x := by
173     change (((e (algebraMap R L q) : F) : K)) = x
174     exact coordinateLocalFractionEquivAdjoin_X E x hxE
175   obtain ⟨D⟩ :=
176     exists_retainedDVRPlace (A := R) (F := F) (L := K)
177       q hq_ne hq_nonunit
178   exact ⟨{
179     coefficientField := E
180     coordinate_transcendental := hxE
181     ambientAlgebra := algRK
182     coefficientTower := coefficientTower
183     coordinate_eq := hcoordinate
184     place := D
185   }⟩
