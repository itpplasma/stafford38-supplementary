22 /-- On the transcendental-coordinate branch, the original geometric
23 avoidance hypothesis supplies exactly the polynomial inverse required by
24 the retained visible-frame producer. The witness is produced from that same
25 inverse and the same original prime. -/
26 theorem exists_inverse_and_visible_frame_of_coordinate_avoidance
27     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
28     {m : ℕ} (hm : 0 < m)
29     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
30     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k P.asIdeal,
31       y ⟨0, hm⟩ ≠ 0)
32     (htrans : Transcendental k (componentCoordinate P ⟨0, hm⟩)) :
33     ∃ g : MvPolynomial (Fin m) k,
34       MvPolynomial.X ⟨0, hm⟩ * g - 1 ∈ P.asIdeal ∧
35       Nonempty (GeneralDivisorialVisibleFrameWitness hm P) := by
36   obtain ⟨g, hg⟩ :=
37     exists_coordinate_inverse_of_avoidance P.asIdeal ⟨0, hm⟩ havoid
38   exact ⟨g, hg,
39       generalDivisorialVisibleFrameWithResidueAlgebraicity hm P ⟨g, hg⟩ htrans⟩

41 /-- Separate the direct constant-coordinate case from the transcendental
42 retained-witness case using only the original prime and zero-locus avoidance.
43 This is the pre-endpoint split consumed by the same-witness affine-fibre
44 producer. -/
45 theorem coordinate_axis_or_visible_frame_of_avoidance
46     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
47     {m : ℕ} (hm : 0 < m)
48     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
49     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k P.asIdeal,
50       y ⟨0, hm⟩ ≠ 0) :
51     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
52       MvPolynomial.zeroLocus k
53         (MvPolynomial.vanishingIdeal k
54           (Stafford38.Geometry.ProjectiveConormalDirections.smoothConormalFibreProjection
55             P.asIdeal)) ∨
56       Nonempty (GeneralDivisorialVisibleFrameWitness hm P) := by
57   by_cases halg : IsAlgebraic k (componentCoordinate P ⟨0, hm⟩)
58   · exact Or.inl
59       (coordinate_axis_mem_smooth_fibre_closure_of_coordinate_algebraic
60         hm P halg)
61   · have htrans : Transcendental k (componentCoordinate P ⟨0, hm⟩) := halg
62     obtain ⟨g, hg, hw⟩ :=
63       exists_inverse_and_visible_frame_of_coordinate_avoidance
64         hm P havoid htrans
65     exact Or.inr hw
