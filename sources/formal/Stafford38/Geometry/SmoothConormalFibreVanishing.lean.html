24 theorem fibreLift_mem_vanishingIdeal_equationConormal
25     {k : Type*} [Field k] [IsAlgClosed k] {n : ℕ}
26     (I : Ideal (MvPolynomial (Fin n) k)) (hI : I.IsPrime)
27     (P : MvPolynomial (Fin n) k)
28     (hP : P ∈ MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I)) :
29     fibreLift P ∈ MvPolynomial.vanishingIdeal k (equationConormalLocus I) := by
30   intro q hq
31   have hclosure : q ∈ equationConormalClosure I := by
32     intro f hf
33     exact hf q hq
34   rw [← equationConormalClosure_smoothAffine_eq I hI] at hclosure
35   apply hclosure
36   intro r hr
37   have hproj : (fun i => r (.inr i)) ∈ smoothConormalFibreProjection I :=
38     ⟨fun i => r (.inl i), hr.2, hr.1.2⟩
39   have hv := hP _ hproj
40   simpa [fibreLift, MvPolynomial.eval_rename, Function.comp_def] using hv

44 end
