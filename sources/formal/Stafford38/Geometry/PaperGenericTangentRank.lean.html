12 # Generic Kähler rank at a retained divisorial place
13 
14 A finite differential lattice over a DVR has generic rank at most one more
15 than the dimension of the residue-field cotangent space.  The proof uses the
16 actual Kähler module of the fraction field, reduction modulo the divisor, and
17 Nakayama; it does not assume a tangent-dimension estimate.
18 -/

31 /-- A finite family spanning a finite-dimensional vector space contains a
32 subfamily indexed by its finrank which still spans. -/
33 theorem exists_spanning_subfamily_of_span_eq_top
34     {K M ι : Type*} [Field K] [AddCommGroup M] [Module K M]
35     [FiniteDimensional K M] [Fintype ι]
36     (v : ι → M)
37     (hspan : Submodule.span K (Set.range v) = ⊤) :
38     ∃ rows : Fin (Module.finrank K M) ↪ ι,
39       Submodule.span K (Set.range fun i ↦ v (rows i)) = ⊤ := by
40   classical
41   let b := Module.Basis.ofSpan (K := K)
42     (s := Set.range v) hspan.ge
43   letI : Fintype ((linearIndepOn_empty K (id : M → M)).extend
44       (Set.empty_subset (Set.range v))) := Fintype.ofFinite _
45   let e : Fin (Module.finrank K M) ≃
46       ((linearIndepOn_empty K (id : M → M)).extend
47         (Set.empty_subset (Set.range v))) :=
48     Fintype.equivOfCardEq (by
49       rw [Fintype.card_fin, ← Module.finrank_eq_card_basis b])
50   have hb_range (a : Fin (Module.finrank K M)) : b (e a) ∈ Set.range v :=
51     Module.Basis.ofSpan_subset (K := K) (V := M) hspan.ge ⟨e a, rfl⟩
52   let row (a : Fin (Module.finrank K M)) : ι := Classical.choose (hb_range a)
53   have hrow (a : Fin (Module.finrank K M)) : v (row a) = b (e a) :=
54     Classical.choose_spec (hb_range a)
55   have hrows : Function.Injective row := by
56     intro a a' haa'
57     apply e.injective
58     apply b.injective
59     rw [← hrow a, ← hrow a', haa']
60   let rows : Fin (Module.finrank K M) ↪ ι := ⟨row, hrows⟩
61   refine ⟨rows, ?_⟩
62   have hrange : Set.range (fun a : Fin (Module.finrank K M) ↦ v (rows a)) =
63       Set.range b := by
64     ext z
65     constructor
66     · rintro ⟨a, rfl⟩
67       exact ⟨e a, (hrow a).symm⟩
68     · rintro ⟨j, rfl⟩
69       refine ⟨e.symm j, ?_⟩
70       change v (row (e.symm j)) = b j
71       rw [hrow, e.apply_symm_apply]
72   rw [hrange, b.span_eq]
73 
74 end
75 
76 end Stafford38.Geometry.PaperGenericTangentRank

120 theorem visibleFrame_le_selectedCoordinateFrame
121     {k V F : Type*} [Field k] [CharZero k] [CommRing V] [IsLocalRing V]
122     [Field F] [Algebra k V] [Algebra k F] [Algebra V F]
123     [IsScalarTower k V F]
124     {ι : Type x} [Fintype ι]
125     (D : VisibleDivisorFrame (V := V) (KaehlerDifferential.D k F) ι)
126     (hmax : maximalIdeal V = Ideal.span {D.t})
127     (himage : ∀ ω : Ω[V⁄k],
128       KaehlerDifferential.map k k V F ω ∈ D.W)
129     {s : ℕ} (rows : Fin s → ι)
130     (hspan : Submodule.span (ResidueField V)
131       (Set.range fun i : Fin s ↦
132         KaehlerDifferential.D k (ResidueField V)
133           (residue V (D.Q (rows i)))) = ⊤) :
134     D.W ≤ coordinateFrame (KaehlerDifferential.D k F) D.t
135       (fun i : Fin s ↦ D.Q (rows i)) := by
136   classical
137   let φ : Ω[V⁄k] →ₗ[V] Ω[F⁄k] := KaehlerDifferential.map k k V F
138   let ψ : Ω[V⁄k] →ₗ[V] Ω[ResidueField V⁄k] :=
139     KaehlerDifferential.map k k V (ResidueField V)
140   let C0 : Submodule V (Ω[V⁄k]) :=
141     Submodule.span V (Set.range fun i : Fin s ⊕ Unit ↦
142       match i with
143       | Sum.inl j => KaehlerDifferential.D k V (D.Q (rows j))
144       | Sum.inr _ => KaehlerDifferential.D k V D.t)
145   let C : Submodule V (Ω[F⁄k]) :=
146     coordinateFrame (KaehlerDifferential.D k F) D.t
147       (fun i : Fin s ↦ D.Q (rows i))
148   let S : Submodule (ResidueField V) (Ω[ResidueField V⁄k]) :=
149     Submodule.span (ResidueField V)

151         KaehlerDifferential.D k (ResidueField V)
152           (residue V (D.Q (rows i))))
153   have hmapC0 : ∀ η ∈ C0, φ η ∈ C := by
154     intro η hη
155     induction hη using Submodule.span_induction with
156     | mem z hz =>
157         obtain ⟨i, rfl⟩ := hz
158         cases i with

161               (KaehlerDifferential.D k V (D.Q (rows i))) ∈ C
162             rw [KaehlerDifferential.map_D]
163             exact Submodule.mem_sup_right
164               (Submodule.subset_span ⟨i, rfl⟩)
165         | inr j =>
166             cases j
167             change KaehlerDifferential.map k k V F
168               (KaehlerDifferential.D k V D.t) ∈ C
169             rw [KaehlerDifferential.map_D]
170             exact Submodule.mem_sup_left
171               (Submodule.subset_span (Set.mem_singleton _))
172     | zero => exact C.zero_mem
173     | add _ _ _ _ hx hy =>
174         rw [map_add]
175         exact C.add_mem hx hy
176     | smul a _ _ hx =>
177         simpa only [LinearMap.map_smul, IsScalarTower.algebraMap_smul] using
178           C.smul_mem a hx
179   have hlift : ∀ z ∈ S, ∃ η ∈ C0, ψ η = z := by
180     intro z hz
181     induction hz using Submodule.span_induction with
182     | mem z hz =>
183         obtain ⟨i, rfl⟩ := hz
184         refine ⟨KaehlerDifferential.D k V (D.Q (rows i)), ?_, ?_⟩
185         · exact Submodule.subset_span ⟨Sum.inl i, rfl⟩
186         · change KaehlerDifferential.map k k V (ResidueField V)
187             (KaehlerDifferential.D k V (D.Q (rows i))) = _
188           rw [KaehlerDifferential.map_D, ResidueField.algebraMap_eq]
189     | zero => exact ⟨0, C0.zero_mem, map_zero _⟩
190     | add _ _ _ _ hx hy =>
191         obtain ⟨x, hxC, hxmap⟩ := hx
192         obtain ⟨y, hyC, hymap⟩ := hy
193         exact ⟨x + y, C0.add_mem hxC hyC, by rw [map_add, hxmap, hymap]⟩
194     | smul a z₀ _ hx =>
195         obtain ⟨x, hxC, hxmap⟩ := hx
196         obtain ⟨a', ha'⟩ := IsLocalRing.residue_surjective a
197         refine ⟨a' • x, C0.smul_mem a' hxC, ?_⟩
198         rw [map_smul, hxmap]
199         change (algebraMap V (ResidueField V) a') • z₀ = a • z₀
200         rw [ResidueField.algebraMap_eq, ha']
201   have hq : ∀ j : ι,
202       KaehlerDifferential.D k F (algebraMap V F (D.Q j)) ∈
203         C ⊔ Ideal.span {D.t} • D.W := by
204     intro j
205     have hres : ψ (KaehlerDifferential.D k V (D.Q j)) ∈ S := by
206       change KaehlerDifferential.map k k V (ResidueField V)
207         (KaehlerDifferential.D k V (D.Q j)) ∈ S
208       rw [KaehlerDifferential.map_D, ResidueField.algebraMap_eq]
209       change KaehlerDifferential.D k (ResidueField V)
210         (residue V (D.Q j)) ∈
211           Submodule.span (ResidueField V)
212             (Set.range fun i : Fin s ↦
213               KaehlerDifferential.D k (ResidueField V)
214                 (residue V (D.Q (rows i))))
215       rw [hspan]
216       trivial
217     obtain ⟨η, hηC, hη⟩ := hlift _ hres
218     have hker : KaehlerDifferential.D k V (D.Q j) - η ∈ LinearMap.ker ψ := by
219       apply LinearMap.mem_ker.mpr
220       simp only [map_sub]
221       exact sub_eq_zero.mpr hη.symm
222     rw [ker_kaehler_residue_map_eq D.t hmax] at hker
223     obtain ⟨δ, hδ, μ, hμ, hsum⟩ := Submodule.mem_sup.mp hker
224     have hdtC0 : KaehlerDifferential.D k V D.t ∈ C0 :=
225       Submodule.subset_span ⟨Sum.inr (), rfl⟩
226     have hδC0 : δ ∈ C0 := by
227       have hle : Submodule.span V {KaehlerDifferential.D k V D.t} ≤ C0 := by
228         rw [Submodule.span_le]
229         intro z hz
230         rw [Set.mem_singleton_iff] at hz
231         simpa [hz] using hdtC0
232       exact hle hδ
233     have hδC : φ δ ∈ C := hmapC0 δ hδC0
234     have hμW : φ μ ∈ Ideal.span {D.t} • D.W := by
235       rw [hmax] at hμ
236       refine Submodule.smul_induction_on hμ ?_ ?_
237       · intro a ha ω hω
238         rw [map_smul]
239         exact Submodule.smul_mem_smul ha (himage ω)
240       · intro x y hx hy
241         rw [map_add]
242         exact add_mem hx hy
243     have hdecomp :
244         KaehlerDifferential.D k F (algebraMap V F (D.Q j)) - φ η = φ δ + φ μ := by
245       have := congrArg φ hsum
246       simpa [φ, KaehlerDifferential.map_D] using this.symm
247     have hηC' : φ η ∈ C := hmapC0 η hηC
248     have heq : KaehlerDifferential.D k F (algebraMap V F (D.Q j)) =
249         φ η + φ δ + φ μ := by
250       calc
251         _ = (_ - φ η) + φ η := by abel
252         _ = (φ δ + φ μ) + φ η := by rw [hdecomp]
253         _ = φ η + φ δ + φ μ := by abel
254     have hsumC : φ η + φ δ + φ μ ∈ C ⊔ Ideal.span {D.t} • D.W := by
255       have hηδ : φ η + φ δ ∈ C := C.add_mem hηC' hδC
256       exact add_mem (Submodule.mem_sup_left hηδ)
257         (Submodule.mem_sup_right hμW)
258     rw [heq]
259     exact hsumC
260   have hall : coordinateFrame (KaehlerDifferential.D k F) D.t D.Q ≤
261       C ⊔ Ideal.span {D.t} • D.W := by
262     refine sup_le ?_ ?_
263     · rw [Submodule.span_le, Set.singleton_subset_iff]
264       have hdtC : KaehlerDifferential.D k F (algebraMap V F D.t) ∈ C :=
265         Submodule.mem_sup_left (Submodule.subset_span (Set.mem_singleton _))
266       exact Submodule.mem_sup_left hdtC
267     · rw [Submodule.span_le]
268       rintro _ ⟨j, rfl⟩
269       exact hq j
270   have hvisible : D.W ≤ C ⊔ Ideal.span {D.t} • D.W := by
271     exact D.visible.trans (sup_le hall le_sup_right)
272   exact Submodule.le_of_le_smul_of_le_jacobson_bot D.W_fg
273     D.parameterIdeal_le_jacobson hvisible

275 /-- At an actual retained DVR on a finitely-generated component, the generic
276 Kähler rank is at most one plus the residue-field Kähler rank.  The numerator
277 coordinates and the nonzero projective denominator generate the component
278 function field: their ratio family is assumed to be its explicit generic
279 coordinate family.  The denominator differential is supplied by its actual
280 factorization `t^a * u` in the visible frame. -/
281 theorem componentFractionField_kaehler_finrank_le_residue_add_one
282     {k V F : Type u} [Field k] [CharZero k] [CommRing V] [IsLocalRing V]
283     [Field F] [Algebra k V] [Algebra k F] [Algebra V F]
284     [IsScalarTower k V F]
285     {ι : Type x} [Fintype ι]
286     (D : VisibleDivisorFrame (V := V) (KaehlerDifferential.D k F) ι)
287     (hmax : maximalIdeal V = Ideal.span {D.t})
288     (himage : ∀ ω : Ω[V⁄k],
289       KaehlerDifferential.map k k V F ω ∈ D.W)
290     (halg : Algebra.IsAlgebraic
291       (IntermediateField.adjoin k
292         (Set.range fun j : ι ↦ residue V (D.Q j)) :
293           IntermediateField k (ResidueField V)) (ResidueField V))
294     (x : ι → F)
295     (hcoordinate : ∀ j, x j =
296       algebraMap V F (D.Q j) / algebraMap V F D.Q₀)
297     (hgen : IntermediateField.adjoin k (Set.range x) = ⊤) :
298     Module.finrank F (Ω[F⁄k]) ≤
299       Module.finrank (ResidueField V) (Ω[ResidueField V⁄k]) + 1 := by
300   classical
301   let κ := ResidueField V
302   obtain ⟨rows, hspanRows⟩ :=
303     exists_residue_kaehler_basis_rows_of_algebraic_generators
304       (fun j : ι ↦ residue V (D.Q j)) (by simpa [κ] using halg)
305   letI : FiniteDimensional κ (Ω[κ⁄k]) := by
306     letI : FiniteDimensional κ
307         (Submodule.span κ
308           (Set.range fun i ↦ KaehlerDifferential.D k κ
309             (residue V (D.Q (rows i))))) :=
310       FiniteDimensional.span_of_finite κ (Set.finite_range _)
311     letI : FiniteDimensional κ (⊤ : Submodule κ (Ω[κ⁄k])) :=
312       (LinearEquiv.ofEq _ _ hspanRows).finiteDimensional
313     exact Submodule.topEquiv.finiteDimensional
314   have hW := visibleFrame_le_selectedCoordinateFrame
315     D hmax himage rows (by simpa [κ] using hspanRows)
316   let G : Fin (Module.finrank κ (Ω[κ⁄k])) ⊕ Unit → Ω[F⁄k] := fun i ↦
317     match i with
318     | Sum.inl j => KaehlerDifferential.D k F
319         (algebraMap V F (D.Q (rows j)))
320     | Sum.inr _ => KaehlerDifferential.D k F (algebraMap V F D.t)
321   let T : Submodule F (Ω[F⁄k]) := Submodule.span F (Set.range G)
322   have hC : coordinateFrame (KaehlerDifferential.D k F) D.t
323       (fun j : Fin (Module.finrank κ (Ω[κ⁄k])) ↦ D.Q (rows j)) ≤
324         T.restrictScalars V := by
325     unfold coordinateFrame T
326     refine sup_le ?_ ?_
327     · rw [Submodule.span_le, Set.singleton_subset_iff]
328       exact Submodule.subset_span ⟨Sum.inr (), rfl⟩
329     · rw [Submodule.span_le]
330       rintro _ ⟨j, rfl⟩
331       exact Submodule.subset_span ⟨Sum.inl j, rfl⟩
332   have hW_T : D.W ≤ T.restrictScalars V := hW.trans hC
333   have hq0 : KaehlerDifferential.D k F (algebraMap V F D.Q₀) ∈ D.W := by
334     have h := himage (KaehlerDifferential.D k V D.Q₀)
335     rw [KaehlerDifferential.map_D] at h
336     exact h
337   have hcoordinateD : ∀ j : ι,
338       KaehlerDifferential.D k F (x j) ∈ T := by
339     intro j
340     rw [hcoordinate j, Derivation.leibniz_div]
341     have hnum : KaehlerDifferential.D k F
342         (algebraMap V F (D.Q j)) ∈ T :=
343       hW_T (D.dQ_mem j)
344     have hden : KaehlerDifferential.D k F
345         (algebraMap V F D.Q₀) ∈ T := hW_T hq0
346     exact T.smul_mem _ (T.sub_mem (T.smul_mem _ hnum)
347       (T.smul_mem _ hden))
348   have hcoordinateSpan :=
349     Stafford38.Geometry.KaehlerSpanSeparableAdjoin.kaehler_span_of_intermediateField_adjoin_eq_top
350       (k := k) (E := F) x hgen
351   have hTtop : T = ⊤ := by
352     apply top_unique
353     rw [← hcoordinateSpan, Submodule.span_le]
354     rintro _ ⟨j, rfl⟩
355     exact hcoordinateD j
356   have hTfin : Module.finrank F T ≤
357       Fintype.card (Fin (Module.finrank κ (Ω[κ⁄k])) ⊕ Unit) := by
358     simpa [T, Set.finrank] using (finrank_range_le_card G)
359   calc
360     Module.finrank F (Ω[F⁄k]) = Module.finrank F (⊤ : Submodule F (Ω[F⁄k])) :=
361       (finrank_top F (Ω[F⁄k])).symm
362     _ = Module.finrank F T := by rw [hTtop]
363     _ ≤ Fintype.card (Fin (Module.finrank κ (Ω[κ⁄k])) ⊕ Unit) := hTfin
364     _ = Module.finrank κ (Ω[κ⁄k]) + 1 := by simp

366 /-- The scalar extension of a generic-point Kähler relation is still an
367 equation conormal at the same generic point after extending the coefficient
368 field.  This is the bridge needed to compare tangent dimensions over the
369 completed Laurent field with Kähler rank over the component function field. -/
370 theorem mapped_generic_kaehler_relation_mem_affineConormalSpace
371     {k F E : Type u} [Field k] [Field F] [Field E]
372     [Algebra k F] [Algebra k E] [Algebra F E] [IsScalarTower k F E]
373     {m : ℕ} {I : Ideal (MvPolynomial (Fin m) k)} [I.IsPrime]
374     [Algebra (MvPolynomial (Fin m) k ⧸ I) F]
375     [IsScalarTower k (MvPolynomial (Fin m) k ⧸ I) F]
376     [IsFractionRing (MvPolynomial (Fin m) k ⧸ I) F]
377     (xi : Fin m → F)
378     (hxi : ∑ i, xi i • KaehlerDifferential.D k F
379       (GenericPointKaehlerConormal.genericPoint I F i) = 0) :
380     AffineConormalSpan.coordinateCovector
381         (fun i ↦ algebraMap F E (xi i)) ∈
382       AffineConormalSpan.affineConormalSpace
383         (fun i ↦ algebraMap F E
384           (GenericPointKaehlerConormal.genericPoint I F i))
385         (I.map (MvPolynomial.map (algebraMap k E))) := by
386   classical
387   let yF := GenericPointKaehlerConormal.genericPoint I F
388   let yE : Fin m → E := fun i ↦ algebraMap F E (yF i)
389   let IF : Ideal (MvPolynomial (Fin m) F) :=
390     GenericPointKaehlerConormal.extendedIdeal I F
391   let IE : Ideal (MvPolynomial (Fin m) E) :=
392     I.map (MvPolynomial.map (algebraMap k E))
393   let φ : MvPolynomial (Fin m) F →+* MvPolynomial (Fin m) E :=
394     MvPolynomial.map (algebraMap F E)
395   have hcomp : φ.comp (MvPolynomial.map (algebraMap k F)) =
396       MvPolynomial.map (algebraMap k E) := by
397     apply MvPolynomial.ringHom_ext
398     · intro a
399       change MvPolynomial.map (algebraMap F E)
400           (MvPolynomial.map (algebraMap k F) (MvPolynomial.C a)) =
401         MvPolynomial.map (algebraMap k E) (MvPolynomial.C a)
402       rw [MvPolynomial.map_C, MvPolynomial.map_C]
403       simpa only [MvPolynomial.map_C] using congrArg MvPolynomial.C
404         (IsScalarTower.algebraMap_apply k F E a).symm
405     · intro i
406       change MvPolynomial.map (algebraMap F E)
407           (MvPolynomial.map (algebraMap k F) (MvPolynomial.X i)) =
408         MvPolynomial.map (algebraMap k E) (MvPolynomial.X i)
409       simp
410   have hmapIdeal : IF.map φ = IE := by
411     dsimp [IF, IE, GenericPointKaehlerConormal.extendedIdeal, φ]
412     rw [Ideal.map_map]
413     rw [hcomp]
414   have hdiff (f : MvPolynomial (Fin m) F) (i : Fin m) :
415       Stafford38.Geometry.CoisotropicTranslation.differentialAt yE (φ f) i =
416         algebraMap F E
417           (Stafford38.Geometry.CoisotropicTranslation.differentialAt yF f i) := by
418     simp only [Stafford38.Geometry.CoisotropicTranslation.differentialAt, φ,
419       MvPolynomial.pderiv_map, MvPolynomial.eval_map]
420     change MvPolynomial.eval₂ (algebraMap F E) (algebraMap F E ∘ yF)
421         (MvPolynomial.pderiv i f) =
422       algebraMap F E (MvPolynomial.eval yF (MvPolynomial.pderiv i f))
423     exact (MvPolynomial.eval₂_comp (algebraMap F E) yF
424       (MvPolynomial.pderiv i f)).symm
425   have hFconormal :=
426     GenericPointKaehlerConormal.coordinateCovector_mem_affineConormalSpace_of_kaehler_sum_eq_zero
427       xi (by simpa [yF] using hxi)
428   obtain ⟨c, hc⟩ :=
429     AffineConormalSpan.coordinate_mem_affineConormalSpace_exists_finsupp
430       yF xi IF hFconormal
431   have hcomponent (i : Fin m) : algebraMap F E (xi i) =
432       ∑ f ∈ c.support, algebraMap F E (c f) *
433         Stafford38.Geometry.CoisotropicTranslation.differentialAt yE (φ f.1) i := by
434     rw [hc i, map_sum]
435     apply Finset.sum_congr rfl
