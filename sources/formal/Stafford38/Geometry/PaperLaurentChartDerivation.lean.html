11 /-!
12 # Derivation tangencies in the affine chart of a retained Laurent arc
13 
14 The retained divisor construction supplies projective coordinates in a
15 one-variable completion.  Their affine ratios need only exist in its Laurent
16 fraction field.  This file extends actual power-series derivations through
17 the localization and computes their affine chart vectors by the quotient
18 rule.  In particular, it does not identify the projective column with its
19 dehomogenization.
20 -/

58     (D : Derivation k (PowerSeries κ) (PowerSeries κ)) :
59     letI : Module k (LaurentSeries κ) := Algebra.toModule
60     letI : SMul k (LaurentSeries κ) :=
61       (Algebra.toModule : Module k (LaurentSeries κ)).toSMul
62     letI : IsScalarTower k (PowerSeries κ) (LaurentSeries κ) :=
63       algebraModule_powerSeriesLaurentTower
64     Derivation k (LaurentSeries κ) (LaurentSeries κ) := by
65   letI : Module k (LaurentSeries κ) := Algebra.toModule
66   letI : SMul k (LaurentSeries κ) :=
67     (Algebra.toModule : Module k (LaurentSeries κ)).toSMul
68   letI : IsScalarTower k (PowerSeries κ) (LaurentSeries κ) :=
69     algebraModule_powerSeriesLaurentTower
70   exact extendDerivation k (PowerSeries κ) (LaurentSeries κ)

92 theorem extendPowerSeriesDerivation_on_powerSeries
93     (D : Derivation k (PowerSeries κ) (PowerSeries κ))
94     (f : PowerSeries κ) :
95     letI : Module k (LaurentSeries κ) := Algebra.toModule
96     letI : SMul k (LaurentSeries κ) :=
97       (Algebra.toModule : Module k (LaurentSeries κ)).toSMul
98     letI : IsScalarTower k (PowerSeries κ) (LaurentSeries κ) :=
99       algebraModule_powerSeriesLaurentTower
100     extendPowerSeriesDerivation D
101         (algebraMap (PowerSeries κ) (LaurentSeries κ) f) =
102       algebraMap (PowerSeries κ) (LaurentSeries κ) (D f) := by
103   letI : Module k (LaurentSeries κ) := Algebra.toModule
104   letI : SMul k (LaurentSeries κ) :=
105     (Algebra.toModule : Module k (LaurentSeries κ)).toSMul
106   letI : IsScalarTower k (PowerSeries κ) (LaurentSeries κ) :=
107     algebraModule_powerSeriesLaurentTower
108   have h := DFunLike.congr_fun (extendPowerSeriesDerivation_comp D) f
109   change extendPowerSeriesDerivation D
110       (algebraMap (PowerSeries κ) (LaurentSeries κ) f) = _ at h
111   simpa [LinearMap.compDer] using h

113 /-- The derivative of the dehomogenized ratio `q_(i+1) / q_0` is the
114 dehomogenization of the projective derivative column. -/
115 theorem extendPowerSeriesDerivation_dehomogenizedPoint
116     (D : Derivation k (PowerSeries κ) (PowerSeries κ))
117     (q : Fin (n + 1) → PowerSeries κ) (hq0 : q 0 ≠ 0) (i : Fin n) :
118     letI : Module k (LaurentSeries κ) := Algebra.toModule
119     letI : SMul k (LaurentSeries κ) :=
120       (Algebra.toModule : Module k (LaurentSeries κ)).toSMul
121     letI : IsScalarTower k (PowerSeries κ) (LaurentSeries κ) :=
122       algebraModule_powerSeriesLaurentTower
123     extendPowerSeriesDerivation D
124         (dehomogenizedPoint (laurentColumn q) i) =
125       dehomogenizedTangentColumn (laurentColumn q)
126         (laurentColumn (fun a ↦ D (q a))) i := by
127   letI : Module k (LaurentSeries κ) := Algebra.toModule
128   letI : SMul k (LaurentSeries κ) :=
129     (Algebra.toModule : Module k (LaurentSeries κ)).toSMul
130   letI : IsScalarTower k (PowerSeries κ) (LaurentSeries κ) :=
131     algebraModule_powerSeriesLaurentTower
132   let E := extendPowerSeriesDerivation D
133   have hq (a : Fin (n + 1)) :
134       E (laurentColumn q a) = laurentColumn (fun b ↦ D (q b)) a := by
135     exact extendPowerSeriesDerivation_on_powerSeries D (q a)
136   have hden : laurentColumn q 0 ≠ 0 :=
137     laurentColumn_ne_zero_of_ne_zero q hq0
138   change E (laurentColumn q i.succ / laurentColumn q 0) = _
139   rw [Derivation.leibniz_div E]
140   rw [hq i.succ, hq 0]
141   simp only [dehomogenizedTangentColumn_apply, laurentColumn, smul_eq_mul]
142   field_simp [hden]

144 /-- Any derivation direction of the completed chart is tangent to the actual
145 affine equation ideal at the Laurent point, provided those equations vanish
146 there. -/
147 theorem derivation_chartVector_mem_zariskiTangentSpace
148     {n : ℕ} (I : Ideal (MvPolynomial (Fin n) k))
149     (q : Fin (n + 1) → PowerSeries κ) (hq0 : q 0 ≠ 0)
150     (D : Derivation k (PowerSeries κ) (PowerSeries κ))
151     (hbase : ∀ f ∈ I,
152       MvPolynomial.eval₂ (algebraMap k (LaurentSeries κ))
153         (dehomogenizedPoint (laurentColumn q)) f = 0) :
154     dehomogenizedTangentColumn (laurentColumn q)
155         (laurentColumn (fun a ↦ D (q a))) ∈
156         zariskiTangentSpace (dehomogenizedPoint (laurentColumn q))
157         (I.map (MvPolynomial.map (algebraMap k (LaurentSeries κ)))) := by
158   letI : Module k (LaurentSeries κ) := Algebra.toModule
159   letI : SMul k (LaurentSeries κ) :=
160     (Algebra.toModule : Module k (LaurentSeries κ)).toSMul
161   letI : IsScalarTower k (PowerSeries κ) (LaurentSeries κ) :=
162     algebraModule_powerSeriesLaurentTower
163   have htan := derivationVector_mem_zariskiTangentSpace_of_eval₂_eq_zero
164     (k := k) (S := LaurentSeries κ) (extendPowerSeriesDerivation D)
165     I (dehomogenizedPoint (laurentColumn q)) hbase
166   have hEq :
167       (fun i ↦ extendPowerSeriesDerivation D
168         (dehomogenizedPoint (laurentColumn q) i)) =
169       dehomogenizedTangentColumn (laurentColumn q)
170         (laurentColumn (fun a ↦ D (q a))) := by
171     funext i
172     exact extendPowerSeriesDerivation_dehomogenizedPoint D q hq0 i
173   rw [← hEq]
174   exact htan

185     dehomogenizedTangentColumn (laurentColumn q)
