22 /-- Extend a derivation across a formally étale algebra by transporting its
23 Kähler functional through the canonical cotangent base-change equivalence. -/
24 noncomputable def extendDerivation (D : Derivation k A L) : Derivation k U L := by
25   let d : Ω[A⁄k] →ₗ[A] L := D.liftKaehlerDifferential
26   let T : U ⊗[A] Ω[A⁄k] →ₗ[U] L := d.liftBaseChange U
27   let E : U ⊗[A] Ω[A⁄k] ≃ₗ[U] Ω[U⁄k] :=
28     KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale k A U
29   exact KaehlerDifferential.linearMapEquivDerivation k U (T.comp E.symm.toLinearMap)
30 
31 /-- The extended derivation restricts to the original derivation on `A`. -/
32 theorem extendDerivation_algebraMap (D : Derivation k A L) (a : A) :
33     extendDerivation (k := k) (A := A) (U := U) (L := L) D
34       (algebraMap A U a) = D a := by

36     KaehlerDifferential.linearMapEquivDerivation_apply_apply,
37     KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale_symm_D_algebraMap,
38     LinearMap.liftBaseChange_tmul,
