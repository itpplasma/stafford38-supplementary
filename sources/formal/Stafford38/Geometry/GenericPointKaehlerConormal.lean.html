11 /-!
12 # Kähler relations at the generic point are affine conormal covectors
13 
14 Let `A = k[X_1, …, X_m] ⧸ I` be a prime affine component with fraction
15 field `F`, and let `y_i ∈ F` be the images of the coordinates (the generic
16 point of the component).  If a coefficient vector `ξ ∈ F^m` satisfies the
17 Kähler relation `∑ ξ_i d y_i = 0` in `Ω[F⁄k]`, then the coordinate covector
18 `v ↦ ∑ ξ_i v_i` lies in the equation-defined affine conormal space of the
19 extended ideal `I·F[X]` at `y`.
20 
21 The proof turns a tangent vector `v` into a derivation `k[X] → F`, descends
22 it to `A`, extends it through `Ω[A⁄k] → Ω[F⁄k]` (which is a localization of
23 modules), and evaluates the resulting `F`-linear functional on the relation.
24 
25 Not proved here: nothing about places, residue fields, boundary charts,
26 smoothness, or geometric conormal bundles.  Only linear algebra of Kähler
27 differentials and derivations over the fraction field of a quotient ring.
28 -/

50 /-- The underlying linear map of a derivation vanishing on `I`, descended to
51 the quotient `P ⧸ I`. -/
52 def quotientDerivationLinear (D : Derivation k P F) (hD : ∀ f ∈ I, D f = 0) :
53     (P ⧸ I) →ₗ[k] F :=
54   (Submodule.liftQ (I.restrictScalars k) D.toLinearMap (by
55       intro f hf
56       exact hD f hf)).comp
57     (Submodule.Quotient.restrictScalarsEquiv k I).symm.toLinearMap
58 
59 theorem quotientDerivationLinear_mk (D : Derivation k P F) (hD : ∀ f ∈ I, D f = 0)
60     (f : P) :
61     quotientDerivationLinear D hD (Ideal.Quotient.mk I f) = D f :=
62   rfl
63 
64 /-- A derivation `P → F` vanishing on `I` descends to a derivation `P ⧸ I → F`,
65 provided the `P`-action on `F` factors through `P ⧸ I`. -/
66 def quotientDerivation [IsScalarTower P (P ⧸ I) F]
67     (D : Derivation k P F) (hD : ∀ f ∈ I, D f = 0) :
68     Derivation k (P ⧸ I) F where
69   toLinearMap := quotientDerivationLinear D hD
70   map_one_eq_zero' := by
71     have h1 : (1 : P ⧸ I) = Ideal.Quotient.mk I 1 := by simp
72     rw [h1]
73     change quotientDerivationLinear D hD (Ideal.Quotient.mk I 1) = 0
74     rw [quotientDerivationLinear_mk, Derivation.map_one_eq_zero]
75   leibniz' a b := by
76     obtain ⟨p, rfl⟩ := Ideal.Quotient.mk_surjective a
77     obtain ⟨q, rfl⟩ := Ideal.Quotient.mk_surjective b
78     change quotientDerivationLinear D hD (Ideal.Quotient.mk I p * Ideal.Quotient.mk I q) =
79       Ideal.Quotient.mk I p • quotientDerivationLinear D hD (Ideal.Quotient.mk I q) +
80         Ideal.Quotient.mk I q • quotientDerivationLinear D hD (Ideal.Quotient.mk I p)
81     rw [← map_mul, quotientDerivationLinear_mk, quotientDerivationLinear_mk,
82       quotientDerivationLinear_mk, Derivation.leibniz]
83     rw [← Ideal.Quotient.algebraMap_eq, ← IsScalarTower.algebraMap_smul (P ⧸ I) p,
84       ← IsScalarTower.algebraMap_smul (P ⧸ I) q]
85 
86 theorem quotientDerivation_mk [IsScalarTower P (P ⧸ I) F]
87     (D : Derivation k P F) (hD : ∀ f ∈ I, D f = 0) (f : P) :
88     quotientDerivation D hD (Ideal.Quotient.mk I f) = D f :=
89   quotientDerivationLinear_mk D hD f
90 
91 end Descent

101 /-- The generic point of the component: images of the coordinates in `F`. -/
102 def genericPoint (I : Ideal (MvPolynomial (Fin m) k)) (F : Type*) [Field F]
103     [Algebra (MvPolynomial (Fin m) k ⧸ I) F] : Fin m → F :=
104   fun i ↦ algebraMap (MvPolynomial (Fin m) k ⧸ I) F (Ideal.Quotient.mk I (X i))
105 
106 /-- The equation ideal extended to `F[X]`. -/
107 def extendedIdeal (I : Ideal (MvPolynomial (Fin m) k)) (F : Type*) [Field F]
108     [Algebra k F] : Ideal (MvPolynomial (Fin m) F) :=
109   I.map (MvPolynomial.map (algebraMap k F))

111 /-- Evaluation of `MvPolynomial.mkDerivation` through partial derivatives,
112 when the polynomial ring acts on `F` through evaluation at `y`. -/
113 theorem mkDerivation_eq_sum_pderiv [Algebra (MvPolynomial (Fin m) k) F]
114     [IsScalarTower k (MvPolynomial (Fin m) k) F]
115     (y v : Fin m → F)
116     (halg : ∀ p : MvPolynomial (Fin m) k, algebraMap (MvPolynomial (Fin m) k) F p = aeval y p)
117     (f : MvPolynomial (Fin m) k) :
118     mkDerivation k v f = ∑ i, aeval y (pderiv i f) * v i := by
119   classical
120   induction f using MvPolynomial.induction_on with
121   | C a =>
122     rw [← MvPolynomial.algebraMap_eq, Derivation.map_algebraMap]
123     simp
124   | add p q hp hq =>
125     simp only [map_add, hp, hq, add_mul, Finset.sum_add_distrib]
126   | mul_X p n hp =>
127     have h1 : ∑ i, aeval y (pderiv i (p * X n)) * v i =
128         aeval y p * v n + y n * ∑ i, aeval y (pderiv i p) * v i := by
129       have h2 : ∀ i, aeval y (pderiv i (p * X n)) * v i =
130           aeval y p * (if n = i then v i else 0) + y n * (aeval y (pderiv i p) * v i) := by
131         intro i
132         simp only [pderiv_mul, map_add, map_mul, aeval_X, pderiv_X, Pi.single_apply]
133         split_ifs <;> simp <;> ring
134       simp only [h2, Finset.sum_add_distrib, ← Finset.mul_sum, Finset.sum_ite_eq,
135         Finset.mem_univ, if_true]
136     rw [h1, Derivation.leibniz, mkDerivation_X, hp, Algebra.smul_def, Algebra.smul_def, halg, halg,
137       aeval_X]

139 /-- Nonzero-divisors of `A` act invertibly on the fraction field `F`. -/
140 theorem isUnit_algebraMap_end
141     (s : nonZeroDivisors (MvPolynomial (Fin m) k ⧸ I)) :
142     IsUnit (algebraMap (MvPolynomial (Fin m) k ⧸ I)
143       (Module.End (MvPolynomial (Fin m) k ⧸ I) F) s) := by
144   rw [Module.End.isUnit_iff]
145   have hfun : ⇑(algebraMap (MvPolynomial (Fin m) k ⧸ I)
146       (Module.End (MvPolynomial (Fin m) k ⧸ I) F) s) = fun x ↦ (s : MvPolynomial (Fin m) k ⧸ I) • x :=
147     funext fun _ ↦ rfl
148   rw [hfun]
149   have := IsLocalization.smul_bijective (M := nonZeroDivisors (MvPolynomial (Fin m) k ⧸ I)) F s
150   simpa [Submonoid.smul_def] using this

152 /-- **Main theorem.**  A Kähler relation among the generic-point coordinates
153 places the corresponding coordinate covector in the affine conormal space of
154 the extended ideal at the generic point. -/
155 theorem coordinateCovector_mem_affineConormalSpace_of_kaehler_sum_eq_zero
156     (xi : Fin m → F)
157     (h : ∑ i, xi i • KaehlerDifferential.D k F (genericPoint I F i) = 0) :
158     coordinateCovector xi ∈ affineConormalSpace (genericPoint I F) (extendedIdeal I F) := by
159   classical
160   set A := MvPolynomial (Fin m) k ⧸ I
161   set y := genericPoint I F
162   rw [affineConormalSpace, Submodule.mem_dualAnnihilator]
163   intro v hv
164   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator] at hv
165   have htan : ∀ f ∈ extendedIdeal I F, ∑ i, differentialAt y f i * v i = 0 := by
166     intro f hf
167     have := hv (differentialCovector y f) (Submodule.subset_span ⟨⟨f, hf⟩, rfl⟩)
168     simpa using this
169   -- `P`-algebra structure on `F` through the quotient and the fraction field.
170   let _ : Algebra (MvPolynomial (Fin m) k) F :=
171     ((algebraMap A F).comp (algebraMap (MvPolynomial (Fin m) k) A)).toAlgebra
172   have : IsScalarTower (MvPolynomial (Fin m) k) A F := by
173     refine ⟨fun p a x ↦ ?_⟩
174     obtain ⟨q, rfl⟩ := Ideal.Quotient.mk_surjective a
175     rw [← Ideal.Quotient.mk_eq_mk, ← Submodule.Quotient.mk_smul, Ideal.Quotient.mk_eq_mk,
176       smul_eq_mul, Algebra.smul_def, Algebra.smul_def, Algebra.smul_def, map_mul, map_mul,
177       mul_assoc]
178     rfl
179   have : IsScalarTower k (MvPolynomial (Fin m) k) F :=
180     IsScalarTower.of_algebraMap_eq fun a ↦ by
181       change algebraMap k F a = algebraMap A F (algebraMap (MvPolynomial (Fin m) k) A (algebraMap k _ a))
182       rw [← IsScalarTower.algebraMap_apply k (MvPolynomial (Fin m) k) A,
183         ← IsScalarTower.algebraMap_apply k A F]
184   have halg : ∀ p : MvPolynomial (Fin m) k, algebraMap (MvPolynomial (Fin m) k) F p = aeval y p := by
185     intro p
186     have : (IsScalarTower.toAlgHom k (MvPolynomial (Fin m) k) F) = aeval y := by
187       apply MvPolynomial.algHom_ext
188       intro i
189       rw [IsScalarTower.coe_toAlgHom', aeval_X]
190       rfl
191     exact (AlgHom.congr_fun this p)
192   -- The tangent vector as a derivation `P → F`.
193   let DP : Derivation k (MvPolynomial (Fin m) k) F := mkDerivation k v
194   have hDP : ∀ f, DP f = ∑ i, aeval y (pderiv i f) * v i :=
195     mkDerivation_eq_sum_pderiv y v halg
196   have hDPI : ∀ f ∈ I, DP f = 0 := by
197     intro f hf
198     rw [hDP]
199     have hmem : MvPolynomial.map (algebraMap k F) f ∈ extendedIdeal I F :=
200       Ideal.mem_map_of_mem _ hf
201     have := htan _ hmem
202     convert this using 2 with i
203     rw [differentialAt, pderiv_map, eval_map, aeval_def]
204   -- Descend to `A`.
205   let DA : Derivation k A F := quotientDerivation DP hDPI
206   have hDA : ∀ i, DA (Ideal.Quotient.mk I (X i)) = v i := by
207     intro i
208     rw [quotientDerivation_mk]
209     exact mkDerivation_X k v i
210   -- Lift through Kähler differentials of `A`.
211   let ψ : Ω[A⁄k] →ₗ[A] F := DA.liftKaehlerDifferential
212   -- Extend through the localization `Ω[A⁄k] → Ω[F⁄k]`.
213   let φ : Ω[F⁄k] →ₗ[A] F :=
214     IsLocalizedModule.lift (nonZeroDivisors A) (KaehlerDifferential.map k k A F) ψ
215       isUnit_algebraMap_end
216   have hφ : ∀ i, φ (KaehlerDifferential.D k F (y i)) = v i := by
217     intro i
218     have hyi : y i = algebraMap A F (Ideal.Quotient.mk I (X i)) := rfl
219     rw [hyi, ← KaehlerDifferential.map_D k k A F, IsLocalizedModule.lift_apply]
220     change DA.liftKaehlerDifferential (KaehlerDifferential.D k A _) = _
221     rw [Derivation.liftKaehlerDifferential_comp_D, hDA]
222   let Φ : Ω[F⁄k] →ₗ[F] F := LinearMap.extendScalarsOfIsLocalization (nonZeroDivisors A) F φ
223   have hΦ := congrArg Φ h
224   rw [map_sum, map_zero] at hΦ
225   simp only [map_smul, LinearMap.extendScalarsOfIsLocalization_apply', Φ, hφ, smul_eq_mul] at hΦ
226   simpa using hΦ
227 
228 end Main
