9 # Contraction of the localized residue kernel
10 
11 The normalization-center calculation is made before coefficient
12 localization.  Once the residue map of the localization is known to restrict
13 to the retained residue map on the normalization, this file identifies the
14 contracted kernel without adding any geometric assumptions.
15 -/

21 /-- Contracting the kernel of a map on an algebra is the kernel of its
22 restriction to the original algebra. -/
23 theorem under_ker_eq_ker_comp
24     {B : Type u} {A : Type v} {K : Type w}
25     [CommSemiring B] [CommSemiring A] [CommSemiring K] [Algebra B A]
26     (ρ : A →+* K) :
27     Ideal.under B (RingHom.ker ρ) =
28       RingHom.ker (ρ.comp (algebraMap B A)) := by
29   ext b
30   simp [Ideal.under_def]
31 
32 /-- If a localized residue map extends the retained residue map, its kernel
33 contracts to the original retained-residue kernel. -/
34 theorem under_ker_eq_ker_of_restriction
35     {B : Type u} {A : Type v} {K : Type w}
36     [CommSemiring B] [CommSemiring A] [CommSemiring K] [Algebra B A]
37     (ρA : A →+* K) (ρB : B →+* K)
38     (hρ : ρA.comp (algebraMap B A) = ρB) :
39     Ideal.under B (RingHom.ker ρA) = RingHom.ker ρB := by
40   rw [under_ker_eq_ker_comp, hρ]
41
