28 `Fin (d+1) ≃ Option (Fin d)` equivalence. -/
29 def localToFinSuccPowerSeries (M : Ideal B) [M.IsMaximal]
30     (eM : (B ⧸ M) ≃ₐ[k] k)
31     [Algebra.FormallyEtale R (Localization.AtPrime M)] :
32     Localization.AtPrime M →ₐ[k] MvPowerSeries (Fin (d + 1)) k :=
33   (MvPowerSeries.renameEquiv k (_root_.finSuccEquiv d).symm).toAlgHom.comp
34     (Stafford38.Geometry.PrescribedGroundPointPowerSeriesMap.localToPowerSeries
35       (σ := Option (Fin d)) M eM)

64 /-- Combine the actual divisor-prime unit-power factorizations with the chosen
65 ground point and its prescribed completion chart. The only open avoided at the
66 ground point for the two factors is the common denominator open supplied by
67 `PaperUnitPowerFactorization`; later bad-locus avoidance remains a separate
68 tilt choice. -/
69 theorem exists_prescribed_ground_point_chart_with_unit_powers
70     [IsAlgClosed k] [IsDomain B] [Algebra.FiniteType k B]
71     (P : Ideal B) [P.IsPrime] (s q₀ q₁ : B)
72     (hs : s = algebraMap R B (MvPolynomial.X none)) (hsP : s ∈ P)
73     (hEt : Algebra.FormallyEtale R (Localization.AtPrime P))
74     (e₀ e₁ : ℕ) (hOrders : 0 < e₀ ∧ e₀ < e₁)
75     (u₀ u₁ : Localization.AtPrime P)
76     (hu₀ : IsUnit u₀) (hu₁ : IsUnit u₁)
77     (h₀ : algebraMap B (Localization.AtPrime P) q₀ =
78       (algebraMap B (Localization.AtPrime P) s) ^ e₀ * u₀)
79     (h₁ : algebraMap B (Localization.AtPrime P) q₁ =
80       (algebraMap B (Localization.AtPrime P) s) ^ e₁ * u₁) :
81     letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
82     ∃ (fUnit fEtale : B) (M : Ideal B) (hM : M.IsMaximal),
83       letI : M.IsMaximal := hM
84       ∃ (eM : (B ⧸ M) ≃ₐ[k] k)
85         (hEtM : Algebra.FormallyEtale R (Localization.AtPrime M))
86         (v₀ v₁ : Localization.Away fUnit),
87         fUnit ∉ P ∧ fUnit ∉ M ∧ P ≤ M ∧
88         fEtale ∉ P ∧ fEtale ∉ M ∧
89         Algebra.IsStandardEtale R (Localization.Away fEtale) ∧
90         IsUnit v₀ ∧ IsUnit v₁ ∧
91         algebraMap B (Localization.Away fUnit) q₀ =
92           (algebraMap B (Localization.Away fUnit) s) ^ e₀ * v₀ ∧
93         algebraMap B (Localization.Away fUnit) q₁ =
94           (algebraMap B (Localization.Away fUnit) s) ^ e₁ * v₁ ∧
95         (0 < e₀ ∧ e₀ < e₁) ∧
96         (letI : Algebra.FormallyEtale R (Localization.AtPrime M) := hEtM
97          let chart := localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
98          ∃ (U₀ U₁ : MvPowerSeries (Fin (d + 1)) k),
99            IsUnit U₀ ∧ IsUnit U₁ ∧
100            MvPowerSeries.constantCoeff U₀ ≠ 0 ∧
101            MvPowerSeries.constantCoeff U₁ ≠ 0 ∧
102            chart (algebraMap B (Localization.AtPrime M) s) =
103              MvPowerSeries.X (0 : Fin (d + 1)) ∧
104            (∀ i : Fin d,
105              chart (algebraMap B (Localization.AtPrime M)
106                (algebraMap R B (MvPolynomial.X (some i)))) =
107                MvPowerSeries.C (residueCoordinates (σ := Option (Fin d)) M eM (some i)) +
108                  MvPowerSeries.X i.succ) ∧
109            chart (algebraMap B (Localization.AtPrime M) q₀) =
110              MvPowerSeries.X (0 : Fin (d + 1)) ^ e₀ * U₀ ∧
111            chart (algebraMap B (Localization.AtPrime M) q₁) =
112              MvPowerSeries.X (0 : Fin (d + 1)) ^ e₁ * U₁) := by
113   classical
114   letI : Algebra.EssFiniteType R B := Algebra.EssFiniteType.of_comp k R B
115   letI : Algebra.FiniteType R B :=
116     Algebra.FiniteType.of_restrictScalars_finiteType k R B
117   letI : Algebra.FinitePresentation R B :=
118     Algebra.FinitePresentation.of_finiteType.mp inferInstance
119   obtain ⟨_d₀, _n₀, _d₁, _n₁, fUnit, _hd₀, _hn₀, _hd₁, _hn₁, hfUnitP,
120       _h₀cleared, _h₁cleared, v₀, v₁, hv₀, hv₁, hAway₀, hAway₁⟩ :=
121     Stafford38.Geometry.PaperUnitPowerFactorization.exists_common_away_unit_power_factorizations
122       P s q₀ q₁ e₀ e₁ u₀ u₁ hu₀ hu₁ h₀ h₁
123   obtain ⟨fEtale, M, hM, hfEtaleP, hPM, hfEtaleM, hfUnitM, hStandard,
124       hResidue, hEtM⟩ :=
125     Stafford38.Geometry.PrescribedEtaleGroundPoint.exists_standardEtale_neighborhood_and_formallyEtale_ground_point_avoiding
126       (k := k) (B := B) P fUnit hfUnitP hEt
127   letI : M.IsMaximal := hM
128   let eM : (B ⧸ M) ≃ₐ[k] k := Classical.choice hResidue
129   let φ : Localization.Away fUnit →+* Localization.AtPrime M :=
130     IsLocalization.Away.lift fUnit
131       (IsLocalization.map_units (Localization.AtPrime M) (M := M.primeCompl)
132         ⟨fUnit, hfUnitM⟩)
133   let chart : Localization.AtPrime M →ₐ[k] MvPowerSeries (Fin (d + 1)) k :=
134     localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
135   let ψ : Localization.Away fUnit →+* MvPowerSeries (Fin (d + 1)) k :=
136     chart.toRingHom.comp φ
137   have hφ : φ.comp (algebraMap B (Localization.Away fUnit)) =
138       algebraMap B (Localization.AtPrime M) := by
139     ext b
140     simp [φ]
141   have hψ : ψ.comp (algebraMap B (Localization.Away fUnit)) =
142       chart.toRingHom.comp (algebraMap B (Localization.AtPrime M)) := by
143     calc
144       ψ.comp (algebraMap B (Localization.Away fUnit)) =
145           chart.toRingHom.comp (φ.comp (algebraMap B (Localization.Away fUnit))) := rfl
146       _ = chart.toRingHom.comp (algebraMap B (Localization.AtPrime M)) :=
147         congrArg (fun f => chart.toRingHom.comp f) hφ
148   have hsM : s ∈ M := hPM hsP
149   have hBeta : residueCoordinates (σ := Option (Fin d)) M eM none = 0 := by
150     unfold residueCoordinates
151     rw [Ideal.Quotient.mkₐ_eq_mk]
152     have hmk : Ideal.Quotient.mk M (algebraMap R B (MvPolynomial.X none)) = 0 :=
153       Ideal.Quotient.eq_zero_iff_mem.mpr (by simpa [hs] using hsM)
154     rw [hmk]
155     simp
156   have hAxis : chart (algebraMap B (Localization.AtPrime M) s) =
157       MvPowerSeries.X (0 : Fin (d + 1)) := by
158     simpa [hs, hBeta] using
159       localToFinSuccPowerSeries_none (k := k) (B := B) (d := d) M eM
160   have hCoordinates : ∀ i : Fin d,
161       chart (algebraMap B (Localization.AtPrime M)
162         (algebraMap R B (MvPolynomial.X (some i)))) =
163       MvPowerSeries.C (residueCoordinates (σ := Option (Fin d)) M eM (some i)) +
164         MvPowerSeries.X i.succ := by
165     intro i
166     exact localToFinSuccPowerSeries_some (k := k) (B := B) (d := d) M eM i
167   have hψs : ψ (algebraMap B (Localization.Away fUnit) s) =
168       chart (algebraMap B (Localization.AtPrime M) s) := DFunLike.congr_fun hψ s
169   let U₀ : MvPowerSeries (Fin (d + 1)) k := ψ v₀
170   let U₁ : MvPowerSeries (Fin (d + 1)) k := ψ v₁
171   have hU₀ : IsUnit U₀ := IsUnit.map ψ hv₀
172   have hU₁ : IsUnit U₁ := IsUnit.map ψ hv₁
173   have hConst₀ : MvPowerSeries.constantCoeff U₀ ≠ 0 := by
174     have h := MvPowerSeries.isUnit_constantCoeff U₀ hU₀
175     exact (isUnit_iff_ne_zero.mp h)
176   have hConst₁ : MvPowerSeries.constantCoeff U₁ ≠ 0 := by
177     have h := MvPowerSeries.isUnit_constantCoeff U₁ hU₁
178     exact (isUnit_iff_ne_zero.mp h)
179   have hPower₀ : chart (algebraMap B (Localization.AtPrime M) q₀) =
180       MvPowerSeries.X (0 : Fin (d + 1)) ^ e₀ * U₀ := by
181     calc
182       chart (algebraMap B (Localization.AtPrime M) q₀) =
183           ψ (algebraMap B (Localization.Away fUnit) q₀) := (DFunLike.congr_fun hψ q₀).symm
184       _ = ψ ((algebraMap B (Localization.Away fUnit) s) ^ e₀ * v₀) :=
185         congrArg ψ hAway₀
186       _ = ψ (algebraMap B (Localization.Away fUnit) s) ^ e₀ * ψ v₀ := by
187         simp only [map_mul, map_pow]
188       _ = MvPowerSeries.X (0 : Fin (d + 1)) ^ e₀ * U₀ := by
189         rw [hψs, hAxis]
190   have hPower₁ : chart (algebraMap B (Localization.AtPrime M) q₁) =
191       MvPowerSeries.X (0 : Fin (d + 1)) ^ e₁ * U₁ := by
192     calc
193       chart (algebraMap B (Localization.AtPrime M) q₁) =
194           ψ (algebraMap B (Localization.Away fUnit) q₁) := (DFunLike.congr_fun hψ q₁).symm
195       _ = ψ ((algebraMap B (Localization.Away fUnit) s) ^ e₁ * v₁) :=
196         congrArg ψ hAway₁
197       _ = ψ (algebraMap B (Localization.Away fUnit) s) ^ e₁ * ψ v₁ := by
198         simp only [map_mul, map_pow]
199       _ = MvPowerSeries.X (0 : Fin (d + 1)) ^ e₁ * U₁ := by
200         rw [hψs, hAxis]
201   refine ⟨fUnit, fEtale, M, hM, ?_⟩
202   letI : M.IsMaximal := hM
203   refine ⟨eM, hEtM, v₀, v₁, hfUnitP, hfUnitM, hPM, hfEtaleP, hfEtaleM,
204     hStandard, hv₀, hv₁, hAway₀, hAway₁, hOrders, ?_⟩
205   letI : Algebra.FormallyEtale R (Localization.AtPrime M) := hEtM
206   exact ⟨U₀, U₁, hU₀, hU₁, hConst₀, hConst₁, hAxis, hCoordinates, hPower₀, hPower₁⟩

208 /-- Canonical reindexing of the genuine local completion preserves injectivity. -/
209 theorem localToFinSuccPowerSeries_injective
210     (M : Ideal B) [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
211     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) (Localization.AtPrime M)] :
212     Function.Injective (localToFinSuccPowerSeries (k := k) (d := d) M eM) := by
213   exact (MvPowerSeries.renameEquiv k (_root_.finSuccEquiv d).symm).injective.comp
214     (PrescribedGroundPointPowerSeriesMap.localToPowerSeries_injective
215       (σ := Option (Fin d)) M eM)

217 /-- A nonzero element of the actual domain remains nonzero in the canonical
218 renamed completion, even if its closed-point residue is zero. -/
219 theorem localToFinSuccPowerSeries_algebraMap_ne_zero
220     [IsDomain B] (M : Ideal B) [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
221     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k) (Localization.AtPrime M)]
222     (b : B) (hb : b ≠ 0) :
223     localToFinSuccPowerSeries (k := k) (d := d) M eM
224       (algebraMap B (Localization.AtPrime M) b) ≠ 0 := by
225   have hinj := (localToFinSuccPowerSeries_injective (d := d) M eM).comp
226     (IsLocalization.injective (Localization.AtPrime M)
227       (Ideal.primeCompl_le_nonZeroDivisors M))
228   intro hz
229   exact hb (hinj (by simpa using hz))
