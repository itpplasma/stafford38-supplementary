27 /-- Evaluating the retained normalized chart map and then including its values
28 in the component function field gives exactly the generic chart-ratio map. -/
29 theorem componentChartEquationQuotientMap_mk_comp_genericEval
30     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
31     {V : Type u} [CommRing V]
32     (coeff : k →+* V)
33     (ι : V →+* ComponentFractionField P)
34     (hι : Function.Injective ι)
35     (hcoeff : ι.comp coeff = algebraMap k (ComponentFractionField P))
36     (q : Fin (m + 1) → V) (chart : Fin (m + 1))
37     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
38     (hqchart : q chart = 1)
39     (scale : ComponentFractionField P)
40     (hq : ∀ a, ι (q a) = scale * componentProjectivePoint P a) :
41     ∀ a : MvPolynomial (Fin m) k,
42       ι (componentChartEquationQuotientMap P coeff ι hι hcoeff q chart e
43         hqchart scale hq (Ideal.Quotient.mk (componentChartEquationIdeal P chart e) a)) =
44         MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
45           (chartGenericPoint P chart e) a := by
46   let qaff : Fin m → V := fun i ↦ q (e i).1
47   let evalV : MvPolynomial (Fin m) k →+* V := MvPolynomial.eval₂Hom coeff qaff
48   let evalF : MvPolynomial (Fin m) k →+* ComponentFractionField P :=
49     MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
50       (chartGenericPoint P chart e)
51   have hcoords : ∀ i, ι (qaff i) = chartGenericPoint P chart e i := by
52     intro i
53     simpa [qaff, chartGenericPoint] using
54       chart_coordinates_agree_with_generic_point P ι q chart e hqchart scale hq i
55   have heval : ∀ a, ι (evalV a) = evalF a := by
56     intro a
57     calc
58       ι (evalV a) = (ι.comp evalV) a := rfl
59       _ = (MvPolynomial.eval₂Hom (ι.comp coeff) (fun i ↦ ι (qaff i))) a := by
60         rw [MvPolynomial.comp_eval₂Hom]
61       _ = evalF a := by
62         rw [hcoeff, funext hcoords]
63   intro a
64   calc
65     ι (componentChartEquationQuotientMap P coeff ι hι hcoeff q chart e
66         hqchart scale hq (Ideal.Quotient.mk (componentChartEquationIdeal P chart e) a)) =
67         ι (evalV a) := rfl
68     _ = evalF a := heval a

70 /-- The retained normalized chart map into a subring of the component function field
71 is injective: after inclusion into the function field it is the generic chart evaluation,
72 whose kernel is exactly the dehomogenized homogeneous component ideal. -/
73 theorem componentChartEquationQuotientMap_injective
74     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
75     {V : Type u} [CommRing V]
76     (coeff : k →+* V)
77     (ι : V →+* ComponentFractionField P)
78     (hι : Function.Injective ι)
79     (hcoeff : ι.comp coeff = algebraMap k (ComponentFractionField P))
80     (q : Fin (m + 1) → V) (chart : Fin (m + 1))
81     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
82     (hqchart : q chart = 1)
83     (scale : ComponentFractionField P)
84     (hq : ∀ a, ι (q a) = scale * componentProjectivePoint P a) :
85     Function.Injective
86       (componentChartEquationQuotientMap P coeff ι hι hcoeff q chart e
87         hqchart scale hq) := by
88   let I := componentChartEquationIdeal P chart e
89   let qaff : Fin m → V := fun i ↦ q (e i).1
90   let evalV : MvPolynomial (Fin m) k →+* V := MvPolynomial.eval₂Hom coeff qaff
91   let evalF : MvPolynomial (Fin m) k →+* ComponentFractionField P :=
92     MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
93       (chartGenericPoint P chart e)
94   let f : MvPolynomial (Fin m) k ⧸ I →+* V :=
95     componentChartEquationQuotientMap P coeff ι hι hcoeff q chart e hqchart scale hq
96   have hchart : componentProjectivePoint P chart ≠ 0 := by
97     have hscale : scale * componentProjectivePoint P chart = 1 := by
98       simpa [hqchart] using (hq chart).symm
99     intro hz
100     rw [hz, mul_zero] at hscale
101     exact one_ne_zero hscale.symm
102   intro x y hxy
103   obtain ⟨a, rfl⟩ := Ideal.Quotient.mk_surjective x
104   obtain ⟨b, rfl⟩ := Ideal.Quotient.mk_surjective y
105   have hxyV : evalV a = evalV b := by
106     have hma : f (Ideal.Quotient.mk I a) = evalV a := by
107       rfl
108     have hmb : f (Ideal.Quotient.mk I b) = evalV b := by
109       rfl
110     change f (Ideal.Quotient.mk I a) = f (Ideal.Quotient.mk I b) at hxy
111     rw [hma, hmb] at hxy
112     exact hxy
113   have hxyF : evalF a = evalF b := by
114     calc
115       evalF a = ι (f (Ideal.Quotient.mk I a)) := by
116         exact (componentChartEquationQuotientMap_mk_comp_genericEval
117           P coeff ι hι hcoeff q chart e hqchart scale hq a).symm
118       _ = ι (f (Ideal.Quotient.mk I b)) := congrArg ι hxy
119       _ = evalF b :=
120         componentChartEquationQuotientMap_mk_comp_genericEval
121           P coeff ι hι hcoeff q chart e hqchart scale hq b
122   have hdiff : evalF (a - b) = 0 := by
123     rw [map_sub, hxyF, sub_self]
124   have hmem : a - b ∈ I := by
125     change a - b ∈ componentChartEquationIdeal P chart e
126     rw [componentChartEquationIdeal_eq_genericEvalKer P chart e hchart]
127     exact RingHom.mem_ker.mpr hdiff
128   exact Ideal.Quotient.eq.mpr hmem

130 /-- The quotient map built from a retained Lane-C witness is injective for every
131 normalized projective coordinate column. -/
132 theorem retainedComponentChartEquationQuotientMap_injective
133     [CharZero k]
134     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
135     (W : Stafford38.Geometry.RelativeRetainedBoundaryPlace.Data k
136       (ComponentFractionField P) (componentCoordinate P i))
137     (chart : Fin (m + 1))
138     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart) :
139     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
140         (ComponentFractionField P) := W.ambientAlgebra
141     letI : IsDiscreteValuationRing W.place.valuation.toSubring := W.place.isDiscrete
142     letI : Algebra W.coefficientField W.place.valuation.toSubring :=
143       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
144     ∀ (q : Fin (m + 1) → W.place.valuation.toSubring)
145       (hqchart : q chart = 1) (scale : ComponentFractionField P)
146       (hq : ∀ a, (q a : ComponentFractionField P) =
147         scale * componentProjectivePoint P a),
148       Function.Injective
149         (retainedComponentChartEquationQuotientMap P i W chart e q
150           hqchart scale hq) := by
151   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
152       (ComponentFractionField P) := W.ambientAlgebra
153   let V := W.place.valuation.toSubring
154   letI : IsDiscreteValuationRing V := W.place.isDiscrete
155   letI : Algebra W.coefficientField V :=
156     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
157   letI : IsScalarTower W.coefficientField
158       (CoordinateZeroLocalRing W.coefficientField)
159       (ComponentFractionField P) := W.coefficientTower
160   intro q hqchart scale hq
161   let coeff : k →+* V :=
162     ComponentProjectiveClosureNormalization.retainedComponentCoefficientMap P i W
163   have hcoeff : V.subtype.comp coeff =
164       algebraMap k (ComponentFractionField P) := by
165     ext c
166     change ((relativeCoefficientMap
167         W.coefficientField W.place (algebraMap k W.coefficientField c) : V) :
168         ComponentFractionField P) = algebraMap k (ComponentFractionField P) c
169     calc
170       ((relativeCoefficientMap
171           W.coefficientField W.place (algebraMap k W.coefficientField c) : V) :
172           ComponentFractionField P) =
173           algebraMap W.coefficientField (ComponentFractionField P)
174             (algebraMap k W.coefficientField c) :=
175         DFunLike.congr_fun
176           (relativeCoefficientMap_commutes W.coefficientField W.place)
177           (algebraMap k W.coefficientField c)
178       _ = algebraMap k (ComponentFractionField P) c :=
179         IsScalarTower.algebraMap_apply k W.coefficientField
180           (ComponentFractionField P) c
181   simpa [retainedComponentChartEquationQuotientMap] using
182     (componentChartEquationQuotientMap_injective P coeff V.subtype
183       Subtype.val_injective hcoeff q chart e hqchart scale hq)
