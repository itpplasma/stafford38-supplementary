42 /-- Ratios in any nonzero projective chart generate the same component function
43 field as the original affine coordinate family. -/
44 theorem chartGenericPoint_adjoin_eq_top
45     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
46     (chart : Fin (m + 1))
47     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
48     (hchart : componentProjectivePoint P chart ≠ 0) :
49     IntermediateField.adjoin k (Set.range (chartGenericPoint P chart e)) = ⊤ := by
50   let L := IntermediateField.adjoin k (Set.range (chartGenericPoint P chart e))
51   apply top_unique
52   rw [← Stafford38.Geometry.ExactDivisorialVisibleFrameExistence.componentCoordinate_adjoin_eq_top P]
53   apply IntermediateField.adjoin_le_iff.2
54   rintro x ⟨j, rfl⟩
55   change componentCoordinate P j ∈ L
56   rcases Fin.eq_zero_or_eq_succ chart with hzero | ⟨c, hchartEq⟩
57   · subst chart
58     let ij := e.symm ⟨j.succ, Fin.succ_ne_zero j⟩
59     have hij : (e ij).1 = j.succ := by
60       have := Equiv.apply_symm_apply e ⟨j.succ, Fin.succ_ne_zero j⟩
61       exact congrArg Subtype.val this
62     have heval : chartGenericPoint P 0 e ij = componentCoordinate P j := by
63       simp [chartGenericPoint, componentProjectivePoint, ij, hij]
64     rw [← heval]
65     exact IntermediateField.subset_adjoin k _ (Set.mem_range_self ij)
66   · subst chart
67     let i0 := e.symm ⟨0, (Fin.succ_ne_zero c).symm⟩
68     have hi0 : (e i0).1 = 0 := by
69       have := Equiv.apply_symm_apply e ⟨0, (Fin.succ_ne_zero c).symm⟩
70       exact congrArg Subtype.val this
71     have hr0 : chartGenericPoint P c.succ e i0 =
72         (componentCoordinate P c)⁻¹ := by
73       simp [chartGenericPoint, componentProjectivePoint, hi0]
74     have hr0mem : chartGenericPoint P c.succ e i0 ∈ L :=
75       IntermediateField.subset_adjoin k _ (Set.mem_range_self i0)
76     have hxcnz : componentCoordinate P c ≠ 0 := by
77       simpa [componentProjectivePoint] using hchart
78     by_cases hjc : j = c
79     · subst j
80       have hxc : componentCoordinate P c =
81           (chartGenericPoint P c.succ e i0)⁻¹ := by
82         rw [hr0]
83         simp [hxcnz]
84       rw [hxc]
85       exact L.inv_mem hr0mem
86     · have hne : j.succ ≠ c.succ := by simpa [Fin.succ_inj] using hjc
87       let ij := e.symm ⟨j.succ, hne⟩
88       have hij : (e ij).1 = j.succ := by
89         have := Equiv.apply_symm_apply e ⟨j.succ, hne⟩
90         exact congrArg Subtype.val this
91       have hrj : chartGenericPoint P c.succ e ij =
92           componentCoordinate P j / componentCoordinate P c := by
93         simp [chartGenericPoint, componentProjectivePoint, hij]
94       have hrjmem : chartGenericPoint P c.succ e ij ∈ L :=
95         IntermediateField.subset_adjoin k _ (Set.mem_range_self ij)
96       have hxratio : componentCoordinate P j =
97           chartGenericPoint P c.succ e ij / chartGenericPoint P c.succ e i0 := by
98         rw [hrj, hr0]
99         field_simp [hxcnz]
100       rw [hxratio]
101       exact L.div_mem hrjmem hr0mem

103 /-- Reindexing the normalized projective-column residues from the complement
104 of an arbitrary chart to the fixed tail `Fin.succ` changes only the residue
105 of coordinate zero (which is assumed zero) and the chart coordinate (which
106 is one). Thus both finite families generate the same residue subfield. -/
107 theorem chartCoordinateResidue_adjoin_eq_tail
108     {V : Type*} [CommRing V] [IsLocalRing V]
109     [Algebra k V] [Algebra k (ResidueField V)]
110     [IsScalarTower k V (ResidueField V)]
111     (q : Fin (m + 1) → V) (chart : Fin (m + 1))
112     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
113     (hqchart : q chart = 1)
114     (hq0res : residue V (q 0) = 0) :
115     IntermediateField.adjoin k
116         (Set.range fun i : Fin m ↦ residue V (q (e i).1)) =
117       IntermediateField.adjoin k
118         (Set.range fun j : Fin m ↦ residue V (q (Fin.succ j))) := by
119   let Lchart : IntermediateField k (ResidueField V) :=
120     IntermediateField.adjoin k
121       (Set.range fun i : Fin m ↦ residue V (q (e i).1))
122   let Ltail : IntermediateField k (ResidueField V) :=
123     IntermediateField.adjoin k
124       (Set.range fun j : Fin m ↦ residue V (q (Fin.succ j)))
125   apply le_antisymm
126   · apply IntermediateField.adjoin_le_iff.mpr
127     rintro z ⟨i, rfl⟩
128     change residue V (q (e i).1) ∈ Ltail
129     rcases Fin.eq_zero_or_eq_succ (e i).1 with hzero | ⟨j, hj⟩
130     · rw [hzero, hq0res]
131       exact Ltail.zero_mem
132     · rw [hj]
133       exact IntermediateField.subset_adjoin k _ ⟨j, rfl⟩
134   · apply IntermediateField.adjoin_le_iff.mpr
135     rintro z ⟨j, rfl⟩
136     change residue V (q (Fin.succ j)) ∈ Lchart
137     by_cases hjchart : Fin.succ j = chart
138     · rw [hjchart, hqchart]
139       simp only [map_one]
140       exact Lchart.one_mem
141     · let i := e.symm ⟨Fin.succ j, hjchart⟩
142       have hi : e i = ⟨Fin.succ j, hjchart⟩ := Equiv.apply_symm_apply e _
143       have hidx : (e i).1 = Fin.succ j := congrArg Subtype.val hi
144       refine IntermediateField.subset_adjoin k _ ⟨i, ?_⟩
145       change residue V (q (e i).1) = residue V (q (Fin.succ j))
146       rw [hidx]

148 /-- Algebraicity over all tail-coordinate residues transfers to any projective
149 chart's affine residue-coordinate family. -/
150 theorem chartCoordinateResidue_algebraic_of_tail
151     {V : Type*} [CommRing V] [IsLocalRing V]
152     [Algebra k V] [Algebra k (ResidueField V)]
153     [IsScalarTower k V (ResidueField V)]
154     (q : Fin (m + 1) → V) (chart : Fin (m + 1))
155     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
156     (hqchart : q chart = 1)
157     (hq0res : residue V (q 0) = 0)
158     (halg : Algebra.IsAlgebraic
159       (IntermediateField.adjoin k
160         (Set.range fun j : Fin m ↦ residue V (q (Fin.succ j))) :
161         IntermediateField k (ResidueField V))
162       (ResidueField V)) :
163     Algebra.IsAlgebraic
164       (IntermediateField.adjoin k
165         (Set.range fun i : Fin m ↦ residue V (q (e i).1)) :
166         IntermediateField k (ResidueField V))
167       (ResidueField V) := by
168   have hfield := chartCoordinateResidue_adjoin_eq_tail
169     (k := k) (m := m) (V := V) q chart e hqchart hq0res
170   exact hfield.symm ▸ halg

172 /-- An algebraic residue field generated by a finite family has a
173 transcendence basis selected from that family. This is the finite residue
174 basis needed before inverting the corresponding chart-coordinate lifts. -/
175 theorem exists_transcendenceBasis_subset_of_intermediateAdjoin_algebraic
176     {κ : Type*} [Field κ] [Algebra k κ]
177     (s : Set κ)
178     (halg : Algebra.IsAlgebraic (IntermediateField.adjoin k s) κ) :
179     ∃ t : Set κ, t ⊆ s ∧ IsTranscendenceBasis k ((↑) : t → κ) := by
180   let A := Algebra.adjoin k s
181   let E := IntermediateField.adjoin k s
182   letI : Algebra A E := (Subalgebra.inclusion
183     (IntermediateField.algebra_adjoin_le_adjoin k s)).toAlgebra
184   letI : Algebra E κ := E.toSubalgebra.toAlgebra
185   letI : Algebra A κ := A.val.toAlgebra
186   have hTower : IsScalarTower A E κ :=
187     Subalgebra.inclusion.isScalarTower_right
188       (IntermediateField.algebra_adjoin_le_adjoin k s) κ
189   have hAEalg : Algebra.IsAlgebraic A E :=
190     Stafford38.Geometry.RelativeDivisorialTower.intermediateAdjoin_isAlgebraic_over_algebraAdjoin
191       k s
192   letI : Algebra.IsAlgebraic A E := hAEalg
193   letI : Algebra.IsAlgebraic E κ := halg
194   have hAκ : Algebra.IsAlgebraic A κ :=
195     @Algebra.IsAlgebraic.trans A E κ _ _ _ _ _ _ hTower _ hAEalg halg
196   have hAκ' : Algebra.IsAlgebraic (Algebra.adjoin k s) κ := by
197     simpa [A] using hAκ
198   exact @exists_isTranscendenceBasis_subset k κ inferInstance inferInstance
199     inferInstance inferInstance inferInstance s hAκ'

201 /-- In the retained projective chart, algebraicity of the tail-residue field
202 selects a transcendence basis from the actual affine chart-coordinate
203 residues, once the normalized denominator has residue zero. -/
204 theorem exists_chartCoordinate_residue_transcendenceBasis
205     {V : Type*} [CommRing V] [IsLocalRing V]
206     [Algebra k V] [Algebra k (ResidueField V)]
207     [IsScalarTower k V (ResidueField V)]
208     (q : Fin (m + 1) → V) (chart : Fin (m + 1))
209     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
210     (hqchart : q chart = 1) (hq0res : residue V (q 0) = 0)
211     (halg : Algebra.IsAlgebraic
212       (IntermediateField.adjoin k
213         (Set.range fun j : Fin m ↦ residue V (q (Fin.succ j))) :
214         IntermediateField k (ResidueField V))
215       (ResidueField V)) :
216     ∃ t : Set (ResidueField V),
217       t ⊆ Set.range (fun i : Fin m ↦ residue V (q (e i).1)) ∧

221   exact exists_transcendenceBasis_subset_of_intermediateAdjoin_algebraic

247 /-- The chart equation quotient is the affine generic chart subalgebra inside
248 its component function field; this is an algebraic affine presentation. -/
249 noncomputable def componentChartEquationQuotient_equiv_genericSubalgebra
250     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
251     (chart : Fin (m + 1))
252     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
253     (hchart : componentProjectivePoint P chart ≠ 0) :
254     (MvPolynomial (Fin m) k ⧸ componentChartEquationIdeal P chart e) ≃ₐ[k]
255       chartGenericPointSubalgebra P chart e := by
256   let evalA : MvPolynomial (Fin m) k →ₐ[k] ComponentFractionField P :=
257     MvPolynomial.aeval (chartGenericPoint P chart e)
258   have heval : (evalA : MvPolynomial (Fin m) k →+* ComponentFractionField P) =
259       MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
260         (chartGenericPoint P chart e) := rfl
261   have hker : componentChartEquationIdeal P chart e =
262       RingHom.ker (evalA : MvPolynomial (Fin m) k →+* ComponentFractionField P) := by
263     rw [heval]
264     exact componentChartEquationIdeal_eq_genericEvalKer P chart e hchart
265   have hrange : Algebra.adjoin k (Set.range (chartGenericPoint P chart e)) = evalA.range := by
266     simpa [evalA] using Algebra.adjoin_range_eq_range_aeval k
267       (chartGenericPoint P chart e)
268   exact (Ideal.quotientEquivAlgOfEq k hker).trans
269     ((Ideal.quotientKerEquivRange evalA).trans
270       (Subalgebra.equivOfEq evalA.range
271         (Algebra.adjoin k (Set.range (chartGenericPoint P chart e))) hrange.symm))

325 /-- The actual affine generic chart domain has the component function field as
326 its fraction field. -/
327 theorem chartGenericPointSubalgebra_isFractionRing
328     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
329     (chart : Fin (m + 1))
330     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
331     (hchart : componentProjectivePoint P chart ≠ 0) :
332     IsFractionRing (chartGenericPointSubalgebra P chart e)
333       (ComponentFractionField P) := by
334   let A := Algebra.adjoin k (Set.range (chartGenericPoint P chart e))
335   have hgen := chartGenericPoint_adjoin_eq_top P chart e hchart
336   refine IsFractionRing.of_field A (ComponentFractionField P) fun z => ?_
337   have hz : z ∈ (⊤ : IntermediateField k (ComponentFractionField P)) := Set.mem_univ z
338   rw [← hgen] at hz
339   rw [IntermediateField.mem_adjoin_iff_div] at hz
340   rcases hz with ⟨a, ha, b, hb, hab⟩
341   exact ⟨⟨a, ha⟩, ⟨b, hb⟩, hab⟩

343 /-- The exact dehomogenized chart equation quotient embeds in the component
344 function field and has that field as its fraction field. -/
345 theorem componentChartEquationQuotient_isFractionRing
346     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
347     (chart : Fin (m + 1))
348     (e : Fin m ≃ ChartAffineIndex (Fin (m + 1)) chart)
349     (hchart : componentProjectivePoint P chart ≠ 0) :
350     let R := MvPolynomial (Fin m) k ⧸ componentChartEquationIdeal P chart e
351     letI : Algebra R (ComponentFractionField P) :=
352       (componentChartEquationGenericPointMap P chart e hchart).toRingHom.toAlgebra
353     IsFractionRing R (ComponentFractionField P) := by
354   let A := Algebra.adjoin k (Set.range (chartGenericPoint P chart e))
355   let eqv := componentChartEquationQuotient_equiv_genericSubalgebra P chart e hchart
356   let f := componentChartEquationGenericPointMap P chart e hchart
357   letI : IsFractionRing A (ComponentFractionField P) :=
358     chartGenericPointSubalgebra_isFractionRing P chart e hchart
359   letI : Algebra (MvPolynomial (Fin m) k ⧸ componentChartEquationIdeal P chart e)
360       (ComponentFractionField P) := f.toRingHom.toAlgebra
361   have hinj : Function.Injective f := by
362     intro x y h
363     obtain ⟨a, rfl⟩ := Ideal.Quotient.mk_surjective x
364     obtain ⟨b, rfl⟩ := Ideal.Quotient.mk_surjective y
365     have hEval : MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
366         (chartGenericPoint P chart e) a =
367       MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
368         (chartGenericPoint P chart e) b := by
369       dsimp only [f] at h
370       simpa only [componentChartEquationGenericPointMap_mk] using h
371     apply Ideal.Quotient.eq.mpr
372     change (MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
373         (chartGenericPoint P chart e)) a =
374       (MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
375         (chartGenericPoint P chart e)) b at hEval
376     have hdiff : MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
377         (chartGenericPoint P chart e) (a - b) = 0 := by
378       change (MvPolynomial.eval₂Hom (algebraMap k (ComponentFractionField P))
379         (chartGenericPoint P chart e)) (a - b) = 0
380       rw [map_sub, hEval, sub_self]
381     rw [componentChartEquationIdeal_eq_genericEvalKer P chart e hchart]
382     exact RingHom.mem_ker.mpr hdiff
383   letI : FaithfulSMul
384       (MvPolynomial (Fin m) k ⧸ componentChartEquationIdeal P chart e)
385       (ComponentFractionField P) :=
386     (faithfulSMul_iff_algebraMap_injective _ _).mpr hinj
387   refine IsFractionRing.of_field
388     (MvPolynomial (Fin m) k ⧸ componentChartEquationIdeal P chart e)
389     (ComponentFractionField P) fun z => ?_
390   obtain ⟨a, b, hb, hab⟩ := IsFractionRing.div_surjective A z
391   refine ⟨eqv.symm a, eqv.symm b, ?_⟩
392   have hfa : f (eqv.symm a) = (a : ComponentFractionField P) := by
393     change ((eqv (eqv.symm a) : A) : ComponentFractionField P) = _
394     exact congrArg Subtype.val (eqv.apply_symm_apply a)
395   have hfb : f (eqv.symm b) = (b : ComponentFractionField P) := by
396     change ((eqv (eqv.symm b) : A) : ComponentFractionField P) = _
397     exact congrArg Subtype.val (eqv.apply_symm_apply b)
398   calc
399     z = (a : ComponentFractionField P) / (b : ComponentFractionField P) := hab.symm
400     _ = f (eqv.symm a) / f (eqv.symm b) := by rw [hfa, hfb]
