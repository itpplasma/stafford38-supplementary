8 /-!
9 # Correcting and normalizing a formal divisor tangent
10 
11 This file supplies the first constructive part of the formal-divisor argument.
12 An invertible selected minor of the divisor-tangent matrix gives regular
13 correction coefficients which kill the selected rows of a transverse
14 derivative.  If one distinguished coordinate has positive order `a`, while
15 the divisor-tangent columns have order at least `a` there, characteristic zero
16 leaves a nonzero coefficient in degree `a - 1` after correction.  Taking the
17 least nonzero coefficient of the corrected vector then produces a common
18 power of `X` and a primitive normalized vector.
19 
20 Every output is constructed from the displayed equations.  No residue-rank,
21 saturation, splitting, or left-inverse hypothesis is assumed.
22 -/

35 /-! ## The first nonzero derivative coefficient -/
36 
37 /-- Differentiating `X^(a+1) u` leaves the expected nonzero leading
38 coefficient in degree `a`. -/
39 theorem coeff_derivative_X_pow_succ_mul
40     [CharZero k] (u : PowerSeries k) (a : ℕ) :
41     PowerSeries.coeff a
42         (PowerSeries.derivative (R := k)
43           ((PowerSeries.X : PowerSeries k) ^ (a + 1) * u)) =
44       PowerSeries.constantCoeff u * ((a + 1 : ℕ) : k) := by
45   rw [PowerSeries.coeff_derivative]
46   congr 1
47   simpa [PowerSeries.coeff_zero_eq_constantCoeff_apply, Nat.add_comm] using
48     (PowerSeries.coeff_X_pow_mul u (a + 1) 0)
49   simp
50 
51 /-- In characteristic zero the leading derivative coefficient above cannot
52 vanish when the leading coefficient of `u` is nonzero. -/
53 theorem coeff_derivative_X_pow_succ_mul_ne_zero
54     [CharZero k] (u : PowerSeries k) (a : ℕ)
55     (hu : PowerSeries.constantCoeff u ≠ 0) :
56     PowerSeries.coeff a
57         (PowerSeries.derivative (R := k)
58           ((PowerSeries.X : PowerSeries k) ^ (a + 1) * u)) ≠ 0 := by
59   rw [coeff_derivative_X_pow_succ_mul]
60   exact mul_ne_zero hu (Nat.cast_ne_zero.mpr (Nat.succ_ne_zero a))
61 
62 /-- The same leading-coefficient statement in the geometric indexing:
63 `q = X^a u` with `a > 0` has a nonzero derivative coefficient in degree
64 `a - 1`. -/
65 theorem coeff_derivative_X_pow_mul_ne_zero
66     [CharZero k] (q u : PowerSeries k) (a : ℕ)
67     (ha : 0 < a)
68     (hq : q = (PowerSeries.X : PowerSeries k) ^ a * u)
69     (hu : PowerSeries.constantCoeff u ≠ 0) :
70     PowerSeries.coeff (a - 1) (PowerSeries.derivative (R := k) q) ≠ 0 := by
71   obtain ⟨b, rfl⟩ := Nat.exists_eq_succ_of_ne_zero (Nat.ne_of_gt ha)
72   simpa [hq] using coeff_derivative_X_pow_succ_mul_ne_zero u b hu

74 /-! ## Selected-minor correction -/
75 
76 /-- Coefficients obtained by solving the selected square subsystem. -/
77 def correctionCoefficients
78     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
79     (Z : Matrix ι κ (PowerSeries k)) (rows : κ ↪ ι)
80     (v : ι → PowerSeries k) : κ → PowerSeries k :=
81   ((selectedMinor Z rows)⁻¹).mulVec fun j => v (rows j)
82 
83 /-- Subtract the divisor-tangent combination determined by the selected
84 minor. -/
85 def correctedVector
86     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
87     (Z : Matrix ι κ (PowerSeries k)) (rows : κ ↪ ι)
88     (v : ι → PowerSeries k) : ι → PowerSeries k :=
89   fun i => v i - Z.mulVec (correctionCoefficients Z rows v) i
90 
91 /-- A selected minor with nonzero residue determinant eliminates every
92 selected row exactly over the power-series ring. -/
93 theorem correctedVector_selectedRow_eq_zero
94     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
95     (Z : Matrix ι κ (PowerSeries k)) (rows : κ ↪ ι)
96     (v : ι → PowerSeries k)
97     (hminor :
98       PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0) :
99     ∀ j, correctedVector Z rows v (rows j) = 0 := by
100   have hunit : IsUnit (selectedMinor Z rows).det := by
101     rw [PowerSeries.isUnit_iff_constantCoeff]
102     exact isUnit_iff_ne_zero.mpr hminor
103   have hinv :
104       selectedMinor Z rows * (selectedMinor Z rows)⁻¹ = 1 :=
105     Matrix.mul_nonsing_inv _ hunit
106   intro j
107   have hsolve :
108       (selectedMinor Z rows).mulVec (correctionCoefficients Z rows v) =
109         fun a => v (rows a) := by
110     rw [correctionCoefficients, Matrix.mulVec_mulVec, hinv]
111     exact Matrix.one_mulVec _
112   change v (rows j) -
113       (selectedMinor Z rows).mulVec (correctionCoefficients Z rows v) j = 0
114   rw [hsolve]
115   exact sub_self _

117 /-! ## Primitive common-power normalization -/
118 
119 /-- A vector with a displayed nonzero coefficient has a least common
120 `X`-order.  Dividing every component by that common power produces a
121 primitive vector: at least one component has nonzero constant coefficient.
122 
123 No finiteness assumption on the coordinate type is needed; well-ordering of
124 the coefficient degree supplies the minimum. -/
125 theorem exists_primitive_commonPower_of_coeff_ne_zero
126     {ι : Type v} (T : ι → PowerSeries k) (i₀ : ι) (bound : ℕ)
127     (hcoeff : PowerSeries.coeff bound (T i₀) ≠ 0) :
128     ∃ c : ℕ, c ≤ bound ∧
129       ∃ B : ι → PowerSeries k,
130         (∀ i, T i = (PowerSeries.X : PowerSeries k) ^ c * B i) ∧
131         ∃ i, PowerSeries.constantCoeff (B i) ≠ 0 := by
132   classical
133   let p : ℕ → Prop := fun d => ∃ i, PowerSeries.coeff d (T i) ≠ 0
134   have hp : ∃ d, p d := ⟨bound, i₀, hcoeff⟩
135   let c : ℕ := Nat.find hp
136   have hc_le : c ≤ bound := Nat.find_min' hp ⟨i₀, hcoeff⟩
137   have hbelow : ∀ i m, m < c → PowerSeries.coeff m (T i) = 0 := by
138     intro i m hm
139     by_contra hne
140     apply Nat.find_min hp (m := m)
141     · simpa [c] using hm
142     · exact ⟨i, hne⟩
143   have hdvd : ∀ i, (PowerSeries.X : PowerSeries k) ^ c ∣ T i := by
144     intro i
145     exact PowerSeries.X_pow_dvd_iff.mpr fun m hm => hbelow i m hm
146   choose B hB using hdvd
147   refine ⟨c, hc_le, B, ?_, ?_⟩
148   · intro i
149     exact hB i
150   · obtain ⟨i, hi⟩ := Nat.find_spec hp
151     refine ⟨i, ?_⟩
152     have hcoefficient :
153         PowerSeries.coeff c (T i) =
154           PowerSeries.constantCoeff (B i) := by
155       rw [hB i]
156       simpa [PowerSeries.coeff_zero_eq_constantCoeff_apply, Nat.add_comm] using
157         (PowerSeries.coeff_X_pow_mul (B i) c 0)
158     rwa [hcoefficient] at hi

160 /-! ## The explicit normalized tangent matrix -/
161 
162 /-- Columns of the formal tangent matrix: position, divisor tangents, and the
163 normalized transverse tangent. -/
164 abbrev FormalTangentColumn (κ : Type*) := Unit ⊕ (κ ⊕ Unit)
165 
166 /-- Assemble the position, divisor-tangent, and normalized transverse columns
167 without making any rank assertion. -/
168 def formalTangentMatrix
169     {ι : Type v} {κ : Type w}
170     (q : ι → PowerSeries k) (Z : Matrix ι κ (PowerSeries k))
171     (tau : ι → PowerSeries k) :
172     Matrix ι (FormalTangentColumn κ) (PowerSeries k)
173   | i, Sum.inl _ => q i
174   | i, Sum.inr (Sum.inl j) => Z i j
175   | i, Sum.inr (Sum.inr _) => tau i
176 
177 /-- The chart row, selected divisor minor, and one primitive normalized entry
178 give an explicit pivot proof that the normalized tangent matrix has full
179 column rank after reduction modulo `X`. -/
180 theorem residue_formalTangentMatrix_mulVec_injective
181     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
182     (q : ι → PowerSeries k) (Z : Matrix ι κ (PowerSeries k))
183     (tau : ι → PowerSeries k) (rows : κ ↪ ι) (chart : ι)
184     (hqchart : q chart = 1)
185     (hZchart : ∀ j, Z chart j = 0)
186     (htauchart : tau chart = 0)
187     (htauselected : ∀ j, tau (rows j) = 0)
188     (hminor :
189       PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0)
190     (hprimitive : ∃ i, PowerSeries.constantCoeff (tau i) ≠ 0) :
191     Function.Injective
192       (residueMatrix (formalTangentMatrix q Z tau)).mulVec := by
193   classical
194   intro c₁ c₂ hc
195   apply sub_eq_zero.mp
196   let c : FormalTangentColumn κ → k := c₁ - c₂
197   have hc₀ :
198       (residueMatrix (formalTangentMatrix q Z tau)).mulVec c = 0 := by
199     change (residueMatrix (formalTangentMatrix q Z tau)).mulVec (c₁ - c₂) = 0
200     rw [Matrix.mulVec_sub, hc, sub_self]
201   have hposition : c (Sum.inl ()) = 0 := by
202     have hchart := congrFun hc₀ chart
203     simpa [Matrix.mulVec, dotProduct, residueMatrix, formalTangentMatrix,
204       hqchart, hZchart, htauchart] using hchart
205   let zc : κ → k := fun j => c (Sum.inr (Sum.inl j))
206   have hdivisor : (selectedMinor (residueMatrix Z) rows).mulVec zc = 0 := by
207     funext j
208     have hrow := congrFun hc₀ (rows j)
209     simpa [Matrix.mulVec, dotProduct, selectedMinor, residueMatrix,
210       formalTangentMatrix, zc, hposition, htauselected] using hrow
211   have hminor_residue : (selectedMinor (residueMatrix Z) rows).det ≠ 0 := by
212     rw [← constantCoeff_selectedMinor_det]
213     exact hminor
214   have hdivisor_zero : zc = 0 := by
215     have hinjective :
216         Function.Injective (selectedMinor (residueMatrix Z) rows).mulVec := by
217       apply Matrix.mulVec_injective_iff_isUnit.mpr
218       rw [Matrix.isUnit_iff_isUnit_det]
219       exact isUnit_iff_ne_zero.mpr hminor_residue
220     apply hinjective
221     simpa using hdivisor
222   obtain ⟨pivot, hpivot⟩ := hprimitive
223   have htransverse : c (Sum.inr (Sum.inr ())) = 0 := by
224     have hp := congrFun hc₀ pivot
225     have hz : ∀ j, c (Sum.inr (Sum.inl j)) = 0 := by
226       intro j
227       exact congrFun hdivisor_zero j
228     simpa [Matrix.mulVec, dotProduct, residueMatrix, formalTangentMatrix,
229       hposition, hz, hpivot] using hp
230   have hc_zero : c = 0 := by
231     funext column
232     rcases column with (_ | (j | _))
233     · exact hposition
234     · exact congrFun hdivisor_zero j
235     · exact htransverse
236   simpa [c] using hc_zero

261 /-- A supplied correction killing the selected rows admits the same primitive
262 normalization. The noncancellation calculation has one owner; this theorem
263 requires no inverse matrix and applies directly to constant corrections. -/
264 theorem exists_primitive_formalDivisorTangent_of_selected_correction
265     [CharZero k]
266     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
267     (q : ι → PowerSeries k)
268     (Z : Matrix ι κ (PowerSeries k))
269     (rows : κ ↪ ι) (zero : ι)
270     (a : ℕ) (u₀ : PowerSeries k) (lambda : κ → PowerSeries k)
271     (ha : 0 < a)
272     (hqzero : q zero = (PowerSeries.X : PowerSeries k) ^ a * u₀)
273     (hu₀ : PowerSeries.constantCoeff u₀ ≠ 0)
274     (hZzero : ∀ j, ∃ w : PowerSeries k,
275       Z zero j = (PowerSeries.X : PowerSeries k) ^ a * w)
276     (hselected : ∀ j, PowerSeries.derivative (R := k) (q (rows j)) =
277       Z.mulVec lambda (rows j)) :
278     ∃ (c : ℕ) (tau : ι → PowerSeries k),
279       (∀ j, tau (rows j) = 0) ∧
280       c ≤ a - 1 ∧
281       (∀ i, PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
282         (PowerSeries.X : PowerSeries k) ^ c * tau i) ∧
283       ∃ i, PowerSeries.constantCoeff (tau i) ≠ 0 := by
284   let v : ι → PowerSeries k := fun i => PowerSeries.derivative (R := k) (q i)
285   let T : ι → PowerSeries k := fun i => v i - Z.mulVec lambda i
286   have hselectedZero : ∀ j, T (rows j) = 0 := by
287     intro j
288     exact sub_eq_zero.mpr (hselected j)
289   have hderiv : PowerSeries.coeff (a - 1) (v zero) ≠ 0 := by
290     exact coeff_derivative_X_pow_mul_ne_zero (q zero) u₀ a ha hqzero hu₀
291   have hcombination :
292       PowerSeries.coeff (a - 1) (Z.mulVec lambda zero) = 0 := by
293     change PowerSeries.coeff (a - 1)
294       (∑ j, Z zero j * lambda j) = 0
295     rw [map_sum]
296     apply Finset.sum_eq_zero
297     intro j hj
298     obtain ⟨w, hw⟩ := hZzero j
299     rw [hw, mul_assoc]
300     rw [PowerSeries.coeff_X_pow_mul']
301     simp [Nat.not_le_of_gt (Nat.sub_lt ha Nat.zero_lt_one)]
302   have hTcoeff : PowerSeries.coeff (a - 1) (T zero) ≠ 0 := by
303     change PowerSeries.coeff (a - 1)
304       (v zero - Z.mulVec lambda zero) ≠ 0
305     rw [map_sub, hcombination, sub_zero]
306     exact hderiv
307   obtain ⟨c, hc, tau, hfactor, hprimitive⟩ :=
308     exists_primitive_commonPower_of_coeff_ne_zero T zero (a - 1) hTcoeff
309   refine ⟨c, tau, ?_, hc, hfactor, hprimitive⟩
310   intro j
311   have hz : (PowerSeries.X : PowerSeries k) ^ c * tau (rows j) = 0 := by
312     rw [← hfactor (rows j), hselectedZero j]
313   exact (mul_eq_zero.mp hz).resolve_left (pow_ne_zero c PowerSeries.X_ne_zero)

315 /-- The explicit correction-and-normalization block used in the formal
316 divisor tangent construction.
317 
318 The selected minor constructs `lambda` and kills its selected rows.  The
319 distinguished coordinate equation and divisibility of the corresponding row
320 of `Z` guarantee that the correction cannot cancel the first nonzero
321 derivative coefficient.  Hence the corrected transverse derivative admits a
322 primitive common-power normalization of order at most `a - 1`.
323 -/
324 theorem exists_corrected_primitive_formalDivisorTangent
325     [CharZero k]
326     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
327     (q : ι → PowerSeries k)
328     (Z : Matrix ι κ (PowerSeries k))
329     (rows : κ ↪ ι) (zero : ι)
330     (a : ℕ) (u₀ : PowerSeries k)
331     (ha : 0 < a)
332     (hqzero : q zero = (PowerSeries.X : PowerSeries k) ^ a * u₀)
333     (hu₀ : PowerSeries.constantCoeff u₀ ≠ 0)
334     (hZzero : ∀ j, ∃ w : PowerSeries k,
335       Z zero j = (PowerSeries.X : PowerSeries k) ^ a * w)
336     (hminor :
337       PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0) :
338     ∃ (lambda : κ → PowerSeries k) (c : ℕ)
339       (tau : ι → PowerSeries k),
340       lambda = correctionCoefficients Z rows
341         (fun i => PowerSeries.derivative (R := k) (q i)) ∧
342       (∀ j,
343         PowerSeries.derivative (R := k) (q (rows j)) =
344           Z.mulVec lambda (rows j)) ∧
345       (∀ j, tau (rows j) = 0) ∧
346       c ≤ a - 1 ∧
347       (∀ i,
348         PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
349           (PowerSeries.X : PowerSeries k) ^ c * tau i) ∧
350       ∃ i, PowerSeries.constantCoeff (tau i) ≠ 0 := by
351   let v : ι → PowerSeries k := fun i => PowerSeries.derivative (R := k) (q i)
352   let lambda : κ → PowerSeries k := correctionCoefficients Z rows v
353   have hselected : ∀ j, v (rows j) = Z.mulVec lambda (rows j) := by
354     intro j
355     exact sub_eq_zero.mp (correctedVector_selectedRow_eq_zero Z rows v hminor j)
356   obtain ⟨c, tau, htauselected, hc, hfactor, hprimitive⟩ :=
357     exists_primitive_formalDivisorTangent_of_selected_correction
358       q Z rows zero a u₀ lambda ha hqzero hu₀ hZzero hselected
359   exact ⟨lambda, c, tau, rfl, hselected, htauselected, hc, hfactor, hprimitive⟩

361 /-- A prescribed selected-row correction constructs the primitive tangent
362 matrix and proves its residue columns independent. -/
363 theorem exists_formalDivisorTangent_residue_injective_of_selected_correction
364     [CharZero k]
365     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
366     (q : ι → PowerSeries k)
367     (Z : Matrix ι κ (PowerSeries k))
368     (rows : κ ↪ ι) (chart zero : ι)
369     (a : ℕ) (u₀ : PowerSeries k) (lambda : κ → PowerSeries k)
370     (hqchart : q chart = 1)
371     (hZchart : ∀ j, Z chart j = 0)
372     (ha : 0 < a)
373     (hqzero : q zero = (PowerSeries.X : PowerSeries k) ^ a * u₀)
374     (hu₀ : PowerSeries.constantCoeff u₀ ≠ 0)
375     (hZzero : ∀ j, ∃ w : PowerSeries k,
376       Z zero j = (PowerSeries.X : PowerSeries k) ^ a * w)
377     (hselected : ∀ j, PowerSeries.derivative (R := k) (q (rows j)) =
378       Z.mulVec lambda (rows j))
379     (hminor :
380       PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0) :
381     ∃ (c : ℕ) (tau : ι → PowerSeries k),
382       tau chart = 0 ∧
383       (∀ j, tau (rows j) = 0) ∧
384       c ≤ a - 1 ∧
385       (∀ i,
386         PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
387           (PowerSeries.X : PowerSeries k) ^ c * tau i) ∧
388       (∃ i, PowerSeries.constantCoeff (tau i) ≠ 0) ∧
389       Function.Injective
390         (residueMatrix (formalTangentMatrix q Z tau)).mulVec := by
391   obtain ⟨c, tau, htauselected, hc, hfactor, hprimitive⟩ :=
392     exists_primitive_formalDivisorTangent_of_selected_correction
393       q Z rows zero a u₀ lambda ha hqzero hu₀ hZzero hselected
394   have hderivative_chart : PowerSeries.derivative (R := k) (q chart) = 0 := by
395     simp [hqchart]
396   have hcombination_chart : Z.mulVec lambda chart = 0 := by
397     simp [Matrix.mulVec, dotProduct, hZchart]
398   have htauchart : tau chart = 0 := by
399     have hz := hfactor chart
400     rw [hderivative_chart, hcombination_chart, sub_zero] at hz
401     have hproduct : (PowerSeries.X : PowerSeries k) ^ c * tau chart = 0 :=
402       hz.symm
403     exact (mul_eq_zero.mp hproduct).resolve_left
404       (pow_ne_zero c PowerSeries.X_ne_zero)
405   have hinjective : Function.Injective
406       (residueMatrix (formalTangentMatrix q Z tau)).mulVec :=
407     residue_formalTangentMatrix_mulVec_injective
408       q Z tau rows chart hqchart hZchart htauchart htauselected
409         hminor hprimitive
410   exact ⟨c, tau, htauchart, htauselected,
411     hc, hfactor, hprimitive, hinjective⟩

413 /-- With an explicit affine chart row, the corrected primitive tangent above
414 assembles into a matrix whose residue columns are injective.  This is the
415 complete local formal-divisor producer: residue injectivity is a conclusion,
416 not an input. -/
417 theorem exists_formalDivisorTangent_residue_injective
418     [CharZero k]
419     {ι : Type v} {κ : Type w} [Fintype κ] [DecidableEq κ]
420     (q : ι → PowerSeries k)
421     (Z : Matrix ι κ (PowerSeries k))
422     (rows : κ ↪ ι) (chart zero : ι)
423     (a : ℕ) (u₀ : PowerSeries k)
424     (hqchart : q chart = 1)
425     (hZchart : ∀ j, Z chart j = 0)
426     (ha : 0 < a)
427     (hqzero : q zero = (PowerSeries.X : PowerSeries k) ^ a * u₀)
428     (hu₀ : PowerSeries.constantCoeff u₀ ≠ 0)
429     (hZzero : ∀ j, ∃ w : PowerSeries k,
430       Z zero j = (PowerSeries.X : PowerSeries k) ^ a * w)
431     (hminor :
432       PowerSeries.constantCoeff (selectedMinor Z rows).det ≠ 0) :
433     ∃ (lambda : κ → PowerSeries k) (c : ℕ)
434       (tau : ι → PowerSeries k),
435       lambda = correctionCoefficients Z rows
436         (fun i => PowerSeries.derivative (R := k) (q i)) ∧
437       (∀ j,
438         PowerSeries.derivative (R := k) (q (rows j)) =
439           Z.mulVec lambda (rows j)) ∧
440       tau chart = 0 ∧
441       (∀ j, tau (rows j) = 0) ∧
442       c ≤ a - 1 ∧
443       (∀ i,
444         PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
445           (PowerSeries.X : PowerSeries k) ^ c * tau i) ∧
446       (∃ i, PowerSeries.constantCoeff (tau i) ≠ 0) ∧
447       Function.Injective
448         (residueMatrix (formalTangentMatrix q Z tau)).mulVec := by
449   let v : ι → PowerSeries k := fun i => PowerSeries.derivative (R := k) (q i)
450   let lambda : κ → PowerSeries k := correctionCoefficients Z rows v
451   have hselected : ∀ j, v (rows j) = Z.mulVec lambda (rows j) := by
452     intro j
453     exact sub_eq_zero.mp (correctedVector_selectedRow_eq_zero Z rows v hminor j)
454   obtain ⟨c, tau, htauchart, htauselected, hc, hfactor, hprimitive, hinjective⟩ :=
455     exists_formalDivisorTangent_residue_injective_of_selected_correction
456       q Z rows chart zero a u₀ lambda hqchart hZchart ha hqzero hu₀
457         hZzero hselected hminor
458   exact ⟨lambda, c, tau, rfl, hselected, htauchart, htauselected,
459     hc, hfactor, hprimitive, hinjective⟩
