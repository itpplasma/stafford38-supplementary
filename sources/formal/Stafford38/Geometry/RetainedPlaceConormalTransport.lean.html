8 /-!
9 # Transport of a divisor-tangent conormal row to the completed chart
10 
11 Let `P` be a prime component of affine `m`-space, `F` its function field,
12 `W` a retained boundary place for a transcendental component coordinate,
13 `V` its valuation ring and `Kres` its residue field.  Given
14 
15 * normalized projective coordinates `q : Fin (m + 1) → V` of the generic
16   point (the output of
17   `exists_normalizedProjectivePoint_relativeRetainedBoundaryPlace`),
18 * a visible divisor frame `D` on those coordinates (the input of the
19   divisor-tangent lattice lemma), and
20 * the generic-point Kähler bridge `hbridge` (a relation in `Ω[F⁄k]` among
21   the differentials of the coordinates yields an equation-defined conormal
22   covector over `F`; proved in a sibling file and taken here as an explicit
23   hypothesis),
24 
25 this file produces the `RegularizedOneRowConormalData` consumed by
26 `CanonicalNonconstantFiniteGradientProductionProof`, with the residue field
27 carrying the residue-induced ground structure
28 `retainedResidueGroundAlgebra`.
29 
30 What is proved (all trust-zero, no axioms beyond the standard three):
31 
32 1. `divisorFrame_kaehler_relation`: the lattice lemma gives a covector
33    `xi = e₀ - ∑ c_j e_j` with `c_j ∈ V` and `∑ xi_i • d y_i = 0`.
34 2. `coordinateCovector_map_mem_affineConormalSpace`: equation-defined
35    conormal membership transports along any ring homomorphism of fields.
36 3. `retainedLaurentLift`: the ring homomorphism `F →+* LaurentSeries Kres`
37    extending the retained completion, with its ground identification
38    `retainedLaurentLift_algebraMap_ground`.
39 4. `retainedLaurentLift_componentCoordinate`: the transported generic point
40    is the dehomogenized completed projective column.
41 5. `annihilation_identity`: the exact `V`-identity giving the projective
42    annihilation row.
43 6. `regularizedOneRowConormalData_of_transport`: the assembly over an
44    abstract local ring `V ⊆ F` with an injective completion map killing the
45    residue of nonunits and a compatible Laurent lift of `F`.
46 7. `exists_regularizedOneRowConormalData`: the instantiation at a retained
47    boundary place; the residue field carries `retainedResidueGroundAlgebra`.
48 
49 What is not proved here: the existence of a visible divisor frame `D`, and
50 the bridge `hbridge`; both are explicit hypotheses.
51 -/

88 /-- `differentialAt` commutes with coefficient transport. -/
89 theorem differentialAt_map (ℓ : F →+* L) (y : Fin n → F)
90     (f : MvPolynomial (Fin n) F) (i : Fin n) :
91     ℓ (differentialAt y f i) =
92       differentialAt (ℓ ∘ y) (MvPolynomial.map ℓ f) i := by

95     MvPolynomial.coe_eval₂Hom, MvPolynomial.eval₂_comp_left]

98 /-- A coordinate covector given by a finite gradient identity lies in the
99 equation-defined conormal space. -/
100 theorem coordinateCovector_mem_of_gradient_identity
101     (I : Ideal (MvPolynomial (Fin n) F)) (y xi : Fin n → F)
102     {r : ℕ} (equations : Fin r → MvPolynomial (Fin n) F)
103     (hmem : ∀ j, equations j ∈ I) (coefficients : Fin r → F)
104     (hxi : ∀ i, xi i = ∑ j, coefficients j * differentialAt y (equations j) i) :
105     coordinateCovector xi ∈ affineConormalSpace y I := by
106   rw [affineConormalSpace_eq_equationCovectorSpan]
107   have hcov : coordinateCovector xi =
108       ∑ j, coefficients j • differentialCovector y (equations j) := by
109     apply LinearMap.ext
110     intro v
111     simp only [coordinateCovector_apply, LinearMap.sum_apply, LinearMap.smul_apply,
112       differentialCovector_apply, smul_eq_mul, Finset.mul_sum, hxi, Finset.sum_mul]
113     rw [Finset.sum_comm]
114     refine Finset.sum_congr rfl fun i _ ↦ Finset.sum_congr rfl fun j _ ↦ ?_
115     ring
116   rw [hcov]
117   refine Submodule.sum_mem _ fun j _ ↦ Submodule.smul_mem _ _ ?_
118   exact Submodule.subset_span ⟨⟨equations j, hmem j⟩, rfl⟩

120 /-- Equation-defined conormal membership transports along a ring
121 homomorphism of fields, with the ideal mapped along the same homomorphism. -/
122 theorem coordinateCovector_map_mem_affineConormalSpace
123     (ℓ : F →+* L) (I : Ideal (MvPolynomial (Fin n) F)) (y xi : Fin n → F)
124     (hxi : coordinateCovector xi ∈ affineConormalSpace y I) :
125     coordinateCovector (ℓ ∘ xi) ∈
126       affineConormalSpace (ℓ ∘ y) (I.map (MvPolynomial.map ℓ)) := by
127   obtain ⟨r, equations, coefficients, hgradient⟩ :=
128     exists_fin_gradient_identity_of_mem_affineConormalSpace I y xi hxi
129   refine coordinateCovector_mem_of_gradient_identity _ (ℓ ∘ y) (ℓ ∘ xi)
130     (fun j ↦ MvPolynomial.map ℓ (equations j).1)
131     (fun j ↦ Ideal.mem_map_of_mem _ (equations j).2)
132     (fun j ↦ ℓ (coefficients j)) fun i ↦ ?_
133   simp only [Function.comp_apply, hgradient i, map_sum, map_mul, differentialAt_map]
134 
135 end CovectorTransport

148 /-- The lattice lemma, rewritten as a single vanishing relation for the
149 covector `e_{j₀'} - ∑ c_j e_j` where `c_j = t^(a+e-1) b_j ∈ V`. -/
150 theorem divisorFrame_kaehler_relation
151     (D : VisibleDivisorFrame (V := V) d ι)
152     (hinj : Function.Injective (algebraMap V F))
153     (y : ι → F) (i₀ : ι)
154     (hy : ∀ j, algebraMap V F (D.Q j) / algebraMap V F D.Q₀ = y j)
155     (hy₁ : algebraMap V F D.Q₁ / algebraMap V F D.Q₀ = y i₀) :
156     ∃ b : ι → V,
157       ∑ j, ((if j = i₀ then 1 else 0) -
158         algebraMap V F (D.t ^ (D.a + D.e - 1) * b j)) • d (y j) = 0 := by
159   obtain ⟨b, hb⟩ := D.exists_coefficients hinj
160   refine ⟨b, ?_⟩
161   simp only [hy, hy₁] at hb
162   simp only [sub_smul, Finset.sum_sub_distrib, ite_smul, one_smul, zero_smul,
163     Finset.sum_ite_eq', Finset.mem_univ, if_true, algebraMap_smul]
164   rw [← hb, sub_self]
165 
166 end KaehlerRelation

170 section Annihilation
171 
172 variable {V : Type*} [CommRing V] {ι : Type*} [Fintype ι] [DecidableEq ι]
173 
174 /-- The row `(ell₀, xi_j)` annihilates the projective column `(q₀, q_j)`.
175 Here `q₀ = t^a u`, `q_{i₀} = q₀ p`, `u uinv = 1`, `c_j = t^(a+e-1) b_j`,
176 `xi_j = δ_{j i₀} - c_j`, and `ell₀ = ∑ t^(e-1) b_j q_j uinv - p`. -/
177 theorem annihilation_identity
178     (t u uinv p : V) (a e : ℕ) (he : 1 ≤ e) (hu : u * uinv = 1)
179     (q₀ : V) (qs : ι → V) (i₀ : ι) (b : ι → V)
180     (hq₀ : q₀ = t ^ a * u) (hq₁ : qs i₀ = q₀ * p) :
181     ((∑ j, t ^ (e - 1) * b j * qs j * uinv) - p) * q₀ +
182       ∑ j, ((if j = i₀ then 1 else 0) - t ^ (a + e - 1) * b j) * qs j = 0 := by
183   obtain ⟨e', rfl⟩ := Nat.exists_eq_add_of_le' he
184   simp only [Nat.add_sub_cancel, sub_mul, Finset.sum_sub_distrib, ite_mul, one_mul,
185     zero_mul, Finset.sum_ite_eq', Finset.mem_univ, if_true, hq₁, Finset.sum_mul]
186   have hpow : ∀ j, t ^ (a + (e' + 1) - 1) * b j * qs j =
187       t ^ e' * b j * qs j * uinv * q₀ := by
188     intro j
189     rw [show a + (e' + 1) - 1 = e' + a by omega, pow_add, hq₀]
190     linear_combination (-(t ^ e' * b j * qs j * t ^ a)) * hu
191   simp only [hpow]
192   ring
193 
194 end Annihilation

434 /-- Abstract assembly of the one-row datum.  `φ` is the completion of `V`
435 (injective, killing the residue of nonunits), `ℓ` the compatible extension

440     (hm : 0 < m) (I : Ideal (MvPolynomial (Fin m) k)) (y : Fin m → F)
441     (hIy : ∀ g ∈ I, MvPolynomial.eval₂ (algebraMap k F) y g = 0)
442     (hinj : Function.Injective (algebraMap V F))
443     (φ : V →+* PowerSeries Kres) (hφinj : Function.Injective φ)
444     (hφconst : ∀ v : V, ¬IsUnit v → PowerSeries.constantCoeff (R := Kres) (φ v) = 0)
445     (ℓ : F →+* LaurentSeries Kres)
446     (hℓ : ∀ v : V, ℓ (algebraMap V F v) =
447       algebraMap (PowerSeries Kres) (LaurentSeries Kres) (φ v))
448     (hground : ℓ.comp (algebraMap k F) = groundLaurentMap (k := k) (K := Kres))
449     (q : Fin (m + 1) → V) (hq0 : q 0 ≠ 0)
450     (hdiv : ∀ j, algebraMap V F (q j.succ) / algebraMap V F (q 0) = y j)
451     (p : V) (hratio : q (Fin.succ ⟨0, hm⟩) = q 0 * p)
452     (D : VisibleDivisorFrame (V := V) (KaehlerDifferential.D k F) (Fin m))
453     (hQ₀ : D.Q₀ = q 0) (hQ₁ : D.Q₁ = q (Fin.succ ⟨0, hm⟩))
454     (hQ : ∀ j, D.Q j = q (Fin.succ j))
455     (hbridge : ∀ xi : Fin m → F,
456       ∑ j, xi j • KaehlerDifferential.D k F (y j) = 0 →
457       coordinateCovector xi ∈ affineConormalSpace y
458         (I.map (MvPolynomial.map (algebraMap k F)))) :
459     Nonempty (RegularizedOneRowConormalData (k := k) (K := Kres) hm I
460       (fun a ↦ φ (q a))) := by
461   let i₀ : Fin m := ⟨0, hm⟩
462   let ψ : PowerSeries Kres →+* LaurentSeries Kres :=
463     algebraMap (PowerSeries Kres) (LaurentSeries Kres)
464   have hpoint : ℓ ∘ y = dehomogenizedPoint (laurentColumn fun a ↦ φ (q a)) := by
465     funext j
466     simp only [Function.comp_apply, dehomogenizedPoint, laurentColumn, ← hdiv, map_div₀, hℓ]
467   obtain ⟨b, hb⟩ := divisorFrame_kaehler_relation D hinj y i₀
468     (fun j ↦ by rw [hQ j, hQ₀]; exact hdiv j) (by rw [hQ₁, hQ₀]; exact hdiv i₀)
469   let c : Fin m → V := fun j ↦ D.t ^ (D.a + D.e - 1) * b j
470   let xiV : Fin m → V := fun j ↦ (if j = i₀ then 1 else 0) - c j
471   let xi : Fin m → F := fun j ↦ (if j = i₀ then 1 else 0) - algebraMap V F (c j)
472   have hxi : ∀ j, algebraMap V F (xiV j) = xi j := by
473     intro j
474     simp only [xiV, xi, map_sub]
475     split_ifs <;> simp
476   have hconormalF : coordinateCovector xi ∈ affineConormalSpace y
477       (I.map (MvPolynomial.map (algebraMap k F))) := hbridge xi hb
478   have hcnonunit : ∀ j, ¬IsUnit (c j) := by
479     intro j
480     have hpos : 0 < D.a + D.e - 1 := by
481       have := D.one_le_a; have := D.one_le_e; omega
482     have hmem : c j ∈ maximalIdeal V :=
483       Ideal.mul_mem_right _ _ (Ideal.pow_mem_of_mem _ D.t_mem _ hpos)
484     exact mem_nonunits_iff.1 ((mem_maximalIdeal _).1 hmem)
485   let uinv : V := ↑D.u_unit.unit⁻¹
486   have hu : D.u * uinv = 1 := D.u_unit.mul_val_inv
487   let ell₀ : V := (∑ j, D.t ^ (D.e - 1) * b j * q j.succ * uinv) - p
488   have hann : ell₀ * q 0 + ∑ j, xiV j * q j.succ = 0 :=
489     annihilation_identity D.t D.u uinv p D.a D.e D.one_le_e hu
490       (q 0) (fun j ↦ q j.succ) i₀ b (hQ₀.symm.trans D.Q₀_eq) hratio
491   let ell : Fin (m + 1) → PowerSeries Kres :=
492     Fin.cases (φ ell₀) (fun j ↦ φ (xiV j))
493   have hcol : (fun i ↦ laurentColumn ell i.succ) = ℓ ∘ xi := by
494     funext i
495     simp only [laurentColumn, ell, Fin.cases_succ, Function.comp_apply, ← hxi, hℓ]
496   have hground' : ℓ.comp (algebraMap k F) = algebraMap k (LaurentSeries Kres) :=
497     hground
498   have hideal : (I.map (MvPolynomial.map (algebraMap k F))).map (MvPolynomial.map ℓ) =
499       I.map (groundPolynomialMap (k := k) (K := Kres) (Fin m)) := by
500     rw [Ideal.map_map]
501     congr 1
502     refine RingHom.ext fun f ↦ ?_
503     simp only [RingHom.comp_apply, MvPolynomial.map_map, groundPolynomialMap,
504       ScalarExtensionPoints.scalarPolynomialMap, hground, groundLaurentMap]
505   refine ⟨{
506     ell := ell
507     q_origin_ne := ?_
508     projective_annihilation := ?_
509     base_vanish := ?_
510     conormal := ?_
511     residue_axis := ?_ }⟩
512   · intro h
513     exact hq0 (hφinj (h.trans (map_zero _).symm))
514   · have h := congrArg (ψ.comp φ) hann
515     simp only [RingHom.comp_apply, map_add, map_mul, map_sum, map_zero] at h
516     simpa only [laurentColumn, Fin.sum_univ_succ, ell, Fin.cases_zero, Fin.cases_succ]
517       using h
518   · intro f hf
519     rw [← hpoint]
520     have hle : I.map (groundPolynomialMap (k := k) (K := Kres) (Fin m)) ≤
521         RingHom.ker (MvPolynomial.eval (ℓ ∘ y)) := by
522       refine Ideal.map_le_iff_le_comap.mpr fun g hg ↦ ?_
523       rw [Ideal.mem_comap, RingHom.mem_ker, groundPolynomialMap,
524         ScalarExtensionPoints.scalarPolynomialMap, ← hground',
525         MvPolynomial.eval_map, ← MvPolynomial.eval₂_comp_left, hIy g hg, map_zero]
526     exact hle hf
527   · rw [hcol, ← hpoint, ← hideal]
528     exact coordinateCovector_map_mem_affineConormalSpace ℓ _ _ xi hconormalF
529   · funext j
530     simp only [residueColumn, ell, Fin.cases_succ, xiV, map_sub, hφconst (c j) (hcnonunit j),
531       sub_zero]
532     split_ifs <;> simp
533
