10 /-!
11 # Transport of a divisor-tangent conormal row to the completed chart
12 
13 Let `P` be a prime component of affine `m`-space, `F` its function field,
14 `W` a retained boundary place for a transcendental component coordinate,
15 `V` its valuation ring and `Kres` its residue field.  Given
16 
17 * normalized projective coordinates `q : Fin (m + 1) → V` of the generic
18   point (the output of
19   `exists_normalizedProjectivePoint_relativeRetainedBoundaryPlace`),
20 * a visible divisor frame `D` on those coordinates (the input of the
21   divisor-tangent lattice lemma), and
22 * the generic-point Kähler bridge `hbridge` (a relation in `Ω[F⁄k]` among
23   the differentials of the coordinates yields an equation-defined conormal
24   covector over `F`; proved in a sibling file and taken here as an explicit
25   hypothesis),
26 
27 this file produces the `RegularizedOneRowConormalData` consumed by
28 `CanonicalNonconstantFiniteGradientProductionProof`, with the residue field
29 carrying the residue-induced ground structure
30 `retainedResidueGroundAlgebra`.
31 
32 What is proved (all trust-zero, no axioms beyond the standard three):
33 
34 1. `divisorFrame_kaehler_relation`: the lattice lemma gives a covector
35    `xi = e₀ - ∑ c_j e_j` with `c_j ∈ V` and `∑ xi_i • d y_i = 0`.
36 2. `coordinateCovector_map_mem_affineConormalSpace`: equation-defined
37    conormal membership transports along any ring homomorphism of fields.
38 3. `retainedLaurentLift`: the ring homomorphism `F →+* LaurentSeries Kres`
39    extending the retained completion, with its ground identification
40    `retainedLaurentLift_algebraMap_ground`.
41 4. `retainedLaurentLift_componentCoordinate`: the transported generic point
42    is the dehomogenized completed projective column.
43 5. `annihilation_identity`: the exact `V`-identity giving the projective
44    annihilation row.
45 6. `regularizedOneRowConormalData_of_transport`: the assembly over an
46    abstract local ring `V ⊆ F` with an injective completion map killing the
47    residue of nonunits and a compatible Laurent lift of `F`.
48 7. `exists_regularizedOneRowConormalData`: the instantiation at a retained
49    boundary place; the residue field carries `retainedResidueGroundAlgebra`.
50 
51 What is not proved here: the existence of a visible divisor frame `D`, and
52 the bridge `hbridge`; both are explicit hypotheses.
53 -/

91 theorem differentialAt_map (ℓ : F →+* L) (y : Fin n → F)
92     (f : MvPolynomial (Fin n) F) (i : Fin n) :
93     ℓ (differentialAt y f i) =
94       differentialAt (ℓ ∘ y) (MvPolynomial.map ℓ f) i := by
95   unfold differentialAt

97     MvPolynomial.coe_eval₂Hom, MvPolynomial.eval₂_comp_left]
98   rfl

100 /-- A coordinate covector given by a finite gradient identity lies in the
101 equation-defined conormal space. -/
102 theorem coordinateCovector_mem_of_gradient_identity
103     (I : Ideal (MvPolynomial (Fin n) F)) (y xi : Fin n → F)
104     {r : ℕ} (equations : Fin r → MvPolynomial (Fin n) F)
105     (hmem : ∀ j, equations j ∈ I) (coefficients : Fin r → F)
106     (hxi : ∀ i, xi i = ∑ j, coefficients j * differentialAt y (equations j) i) :
107     coordinateCovector xi ∈ affineConormalSpace y I := by
108   rw [affineConormalSpace_eq_equationCovectorSpan]
109   have hcov : coordinateCovector xi =
110       ∑ j, coefficients j • differentialCovector y (equations j) := by
111     apply LinearMap.ext
112     intro v
113     simp only [coordinateCovector_apply, LinearMap.sum_apply, LinearMap.smul_apply,
114       differentialCovector_apply, smul_eq_mul, Finset.mul_sum, hxi, Finset.sum_mul]
115     rw [Finset.sum_comm]
116     refine Finset.sum_congr rfl fun i _ ↦ Finset.sum_congr rfl fun j _ ↦ ?_
117     ring
118   rw [hcov]
119   refine Submodule.sum_mem _ fun j _ ↦ Submodule.smul_mem _ _ ?_
120   exact Submodule.subset_span ⟨⟨equations j, hmem j⟩, rfl⟩

122 /-- Equation-defined conormal membership transports along a ring
123 homomorphism of fields, with the ideal mapped along the same homomorphism. -/
124 theorem coordinateCovector_map_mem_affineConormalSpace
125     (ℓ : F →+* L) (I : Ideal (MvPolynomial (Fin n) F)) (y xi : Fin n → F)
126     (hxi : coordinateCovector xi ∈ affineConormalSpace y I) :
127     coordinateCovector (ℓ ∘ xi) ∈
128       affineConormalSpace (ℓ ∘ y) (I.map (MvPolynomial.map ℓ)) := by
129   obtain ⟨r, equations, coefficients, hgradient⟩ :=
130     exists_fin_gradient_identity_of_mem_affineConormalSpace I y xi hxi
131   refine coordinateCovector_mem_of_gradient_identity _ (ℓ ∘ y) (ℓ ∘ xi)
132     (fun j ↦ MvPolynomial.map ℓ (equations j).1)
133     (fun j ↦ Ideal.mem_map_of_mem _ (equations j).2)
134     (fun j ↦ ℓ (coefficients j)) fun i ↦ ?_
135   simp only [Function.comp_apply, hgradient i, map_sum, map_mul, differentialAt_map]
136 
137 end CovectorTransport

150 /-- The lattice lemma, rewritten as a single vanishing relation for the
151 covector `e_{j₀'} - ∑ c_j e_j` where `c_j = t^(a+e-1) b_j ∈ V`. -/
152 theorem divisorFrame_kaehler_relation
153     (D : VisibleDivisorFrame (V := V) d ι)
154     (hinj : Function.Injective (algebraMap V F))
155     (y : ι → F) (i₀ : ι)
156     (hy : ∀ j, algebraMap V F (D.Q j) / algebraMap V F D.Q₀ = y j)
157     (hy₁ : algebraMap V F D.Q₁ / algebraMap V F D.Q₀ = y i₀) :
158     ∃ b : ι → V,
159       ∑ j, ((if j = i₀ then 1 else 0) -
160         algebraMap V F (D.t ^ (D.a + D.e - 1) * b j)) • d (y j) = 0 := by
161   obtain ⟨b, hb⟩ := D.exists_coefficients hinj
162   refine ⟨b, ?_⟩
163   simp only [hy, hy₁] at hb
164   simp only [sub_smul, Finset.sum_sub_distrib, ite_smul, one_smul, zero_smul,
165     Finset.sum_ite_eq', Finset.mem_univ, if_true, algebraMap_smul]
166   rw [← hb, sub_self]
167 
168 end KaehlerRelation

172 section Annihilation
173 
174 variable {V : Type*} [CommRing V] {ι : Type*} [Fintype ι] [DecidableEq ι]
175 
176 /-- The row `(ell₀, xi_j)` annihilates the projective column `(q₀, q_j)`.
177 Here `q₀ = t^a u`, `q_{i₀} = q₀ p`, `u uinv = 1`, `c_j = t^(a+e-1) b_j`,
178 `xi_j = δ_{j i₀} - c_j`, and `ell₀ = ∑ t^(e-1) b_j q_j uinv - p`. -/
179 theorem annihilation_identity
180     (t u uinv p : V) (a e : ℕ) (he : 1 ≤ e) (hu : u * uinv = 1)
181     (q₀ : V) (qs : ι → V) (i₀ : ι) (b : ι → V)
182     (hq₀ : q₀ = t ^ a * u) (hq₁ : qs i₀ = q₀ * p) :
183     ((∑ j, t ^ (e - 1) * b j * qs j * uinv) - p) * q₀ +
184       ∑ j, ((if j = i₀ then 1 else 0) - t ^ (a + e - 1) * b j) * qs j = 0 := by
185   obtain ⟨e', rfl⟩ := Nat.exists_eq_add_of_le' he
186   simp only [Nat.add_sub_cancel, sub_mul, Finset.sum_sub_distrib, ite_mul, one_mul,
187     zero_mul, Finset.sum_ite_eq', Finset.mem_univ, if_true, hq₁, Finset.sum_mul]
188   have hpow : ∀ j, t ^ (a + (e' + 1) - 1) * b j * qs j =
189       t ^ e' * b j * qs j * uinv * q₀ := by
190     intro j
191     rw [show a + (e' + 1) - 1 = e' + a by omega, pow_add, hq₀]
192     linear_combination (-(t ^ e' * b j * qs j * t ^ a)) * hu
193   simp only [hpow]
194   ring
195 
196 end Annihilation

436 /-- Abstract assembly of the one-row datum.  `φ` is the completion of `V`

443     (hIy : ∀ g ∈ I, MvPolynomial.eval₂ (algebraMap k F) y g = 0)

452     (hdiv : ∀ j, algebraMap V F (q j.succ) / algebraMap V F (q 0) = y j)
453     (p : V) (hratio : q (Fin.succ ⟨0, hm⟩) = q 0 * p)
454     (D : VisibleDivisorFrame (V := V) (KaehlerDifferential.D k F) (Fin m))
455     (hQ₀ : D.Q₀ = q 0) (hQ₁ : D.Q₁ = q (Fin.succ ⟨0, hm⟩))
456     (hQ : ∀ j, D.Q j = q (Fin.succ j))

459       coordinateCovector xi ∈ affineConormalSpace y
460         (I.map (MvPolynomial.map (algebraMap k F)))) :
461     Nonempty (RegularizedOneRowConormalData (k := k) (K := Kres) hm I
462       (fun a ↦ φ (q a))) := by
463   let i₀ : Fin m := ⟨0, hm⟩
464   let ψ : PowerSeries Kres →+* LaurentSeries Kres :=
465     algebraMap (PowerSeries Kres) (LaurentSeries Kres)
466   have hpoint : ℓ ∘ y = dehomogenizedPoint (laurentColumn fun a ↦ φ (q a)) := by
467     funext j
468     simp only [Function.comp_apply, dehomogenizedPoint, laurentColumn, ← hdiv, map_div₀, hℓ]
469   obtain ⟨b, hb⟩ := divisorFrame_kaehler_relation D hinj y i₀
470     (fun j ↦ by rw [hQ j, hQ₀]; exact hdiv j) (by rw [hQ₁, hQ₀]; exact hdiv i₀)
471   let c : Fin m → V := fun j ↦ D.t ^ (D.a + D.e - 1) * b j
472   let xiV : Fin m → V := fun j ↦ (if j = i₀ then 1 else 0) - c j
473   let xi : Fin m → F := fun j ↦ (if j = i₀ then 1 else 0) - algebraMap V F (c j)
474   have hxi : ∀ j, algebraMap V F (xiV j) = xi j := by
475     intro j
476     simp only [xiV, xi, map_sub]
477     split_ifs <;> simp
478   have hconormalF : coordinateCovector xi ∈ affineConormalSpace y
479       (I.map (MvPolynomial.map (algebraMap k F))) := hbridge xi hb
480   have hcnonunit : ∀ j, ¬IsUnit (c j) := by
481     intro j
482     have hpos : 0 < D.a + D.e - 1 := by
483       have := D.one_le_a; have := D.one_le_e; omega
484     have hmem : c j ∈ maximalIdeal V :=
485       Ideal.mul_mem_right _ _ (Ideal.pow_mem_of_mem _ D.t_mem _ hpos)
486     exact mem_nonunits_iff.1 ((mem_maximalIdeal _).1 hmem)
487   let uinv : V := ↑D.u_unit.unit⁻¹
488   have hu : D.u * uinv = 1 := D.u_unit.mul_val_inv
489   let ell₀ : V := (∑ j, D.t ^ (D.e - 1) * b j * q j.succ * uinv) - p
490   have hann : ell₀ * q 0 + ∑ j, xiV j * q j.succ = 0 :=
491     annihilation_identity D.t D.u uinv p D.a D.e D.one_le_e hu
492       (q 0) (fun j ↦ q j.succ) i₀ b (hQ₀.symm.trans D.Q₀_eq) hratio
493   let ell : Fin (m + 1) → PowerSeries Kres :=
494     Fin.cases (φ ell₀) (fun j ↦ φ (xiV j))
495   have hcol : (fun i ↦ laurentColumn ell i.succ) = ℓ ∘ xi := by
496     funext i
497     simp only [laurentColumn, ell, Fin.cases_succ, Function.comp_apply, ← hxi, hℓ]
498   have hground' : ℓ.comp (algebraMap k F) = algebraMap k (LaurentSeries Kres) :=
499     hground
500   have hideal : (I.map (MvPolynomial.map (algebraMap k F))).map (MvPolynomial.map ℓ) =
501       I.map (scalarPolynomialMap (k := k) (K := LaurentSeries Kres) (Fin m)) := by
502     rw [Ideal.map_map]
503     congr 1
504     refine RingHom.ext fun f ↦ ?_
505     simp only [RingHom.comp_apply, MvPolynomial.map_map, scalarPolynomialMap,
506       hground, groundLaurentMap]
507   refine ⟨{
508     ell := ell
509     q_origin_ne := ?_
510     projective_annihilation := ?_
511     base_vanish := ?_
512     conormal := ?_
513     residue_axis := ?_ }⟩
514   · intro h
515     exact hq0 (hφinj (h.trans (map_zero _).symm))
516   · have h := congrArg (ψ.comp φ) hann
517     simp only [RingHom.comp_apply, map_add, map_mul, map_sum, map_zero] at h
518     simpa only [laurentColumn, Fin.sum_univ_succ, ell, Fin.cases_zero, Fin.cases_succ]
519       using h
520   · intro f hf
521     rw [← hpoint]
522     have hle : I.map (scalarPolynomialMap (k := k) (K := LaurentSeries Kres) (Fin m)) ≤
523         RingHom.ker (MvPolynomial.eval (ℓ ∘ y)) := by
524       refine Ideal.map_le_iff_le_comap.mpr fun g hg ↦ ?_
525       rw [Ideal.mem_comap, RingHom.mem_ker, scalarPolynomialMap, ← hground',
526         MvPolynomial.eval_map, ← MvPolynomial.eval₂_comp_left, hIy g hg, map_zero]
527     exact hle hf
528   · rw [hcol, ← hpoint, ← hideal]
529     exact coordinateCovector_map_mem_affineConormalSpace ℓ _ _ xi hconormalF
530   · funext j
531     simp only [residueColumn, ell, Fin.cases_succ, xiV, map_sub, hφconst (c j) (hcnonunit j),
532       sub_zero]
533     split_ifs <;> simp
