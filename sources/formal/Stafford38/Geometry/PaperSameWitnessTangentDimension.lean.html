41 /-- Tangent rank at the actual Laurent point of the same retained chart is at
42 most one plus the Kähler rank of its residue field. -/
43 def tangent_finrank_goal
44     {k : Type u} [Field k] [CharZero k]
45     {m : ℕ} (hm : 0 < m)
46     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
47     (w : GeneralDivisorialVisibleFrameWitness hm P) : Prop :=
48     let C := w.column
49     let W := C.W
50     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
51         (ComponentFractionField P) := W.ambientAlgebra
52     let V := W.place.valuation.toSubring
53     letI : IsDiscreteValuationRing V := W.place.isDiscrete
54     letI : Algebra W.coefficientField V :=
55       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
56     let κ := ResidueField V
57     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
58     let E := LaurentSeries κ
59     letI : Algebra k E := (groundLaurentMap (k := k) (K := κ)).toAlgebra
60     letI : SMul k E := (inferInstance : Algebra k E).toSMul
61     letI : Module k E := (inferInstance : Algebra k E).toModule
62     Module.finrank E
63         (zariskiTangentSpace
64           (dehomogenizedPoint
65             (laurentColumn fun a ↦ retainedToCompletedPowerSeries W (C.q a)))
66           (P.asIdeal.map
67             (scalarPolynomialMap (k := k) (K := E) (Fin m)))) ≤
68       Module.finrank κ (Ω[κ⁄k]) + 1

70 theorem tangent_finrank_le_residue_add_one_of_witness
71     {k : Type u} [Field k] [CharZero k]
72     {m : ℕ} (hm : 0 < m)
73     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
74     (w : GeneralDivisorialVisibleFrameWitness hm P) :
75     tangent_finrank_goal hm P w := by
76   dsimp [tangent_finrank_goal]
77   let C := w.column
78   let W := C.W
79   let F := ComponentFractionField P
80   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
81   let V := W.place.valuation.toSubring
82   let i₀ : Fin m := ⟨0, hm⟩
83   letI : IsDiscreteValuationRing V := W.place.isDiscrete
84   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
85   letI : Algebra W.coefficientField V :=
86     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
87   let κ := ResidueField V
88   letI : Algebra k κ := retainedResidueGroundAlgebra P i₀ W
89   let E := LaurentSeries κ
90   letI : Algebra k E := (groundLaurentMap (k := k) (K := κ)).toAlgebra
91   letI : SMul k E := (inferInstance : Algebra k E).toSMul
92   letI : Module k E := (inferInstance : Algebra k E).toModule
93   let ell : F →+* E := retainedLaurentLift P i₀ W
94   let y : Fin m → E := dehomogenizedPoint
95     (laurentColumn fun a ↦ retainedToCompletedPowerSeries W (C.q a))
96   have hground : ell.comp (algebraMap k F) =
97       groundLaurentMap (k := k) (K := κ) := by
98     dsimp [ell]
99     exact retainedLaurentLift_comp_algebraMap P i₀ W
100   have hpoint : ell ∘ componentCoordinate P = y := by
101     dsimp [ell, y]
102     exact retainedLaurentLift_componentCoordinate P i₀ W C.q C.scale
103       C.q_commonScale C.q0_ne
104   have hy : y = dehomogenizedPoint
105       (laurentColumn fun a ↦ retainedToCompletedPowerSeries W (C.q a)) := rfl
106   letI : Algebra F E := ell.toAlgebra
107   letI : Algebra k F := inferInstance
108   letI : SMul k F := (inferInstance : Algebra k F).toSMul
109   letI : Module k F := (inferInstance : Algebra k F).toModule
110   letI : IsScalarTower k F E := IsScalarTower.of_algebraMap_eq fun c ↦ by
111     have h := congrArg (fun f : k →+* E ↦ f c) hground
112     change (groundLaurentMap (k := k) (K := κ)) c =
113       ell (algebraMap k F c)
114     exact h.symm
115   have hfieldRank : Module.finrank F (Ω[F⁄k]) ≤
116       Module.finrank κ (Ω[κ⁄k]) + 1 :=
117     component_field_rank_of_witness hm P w
118   have hgenPoint : IntermediateField.adjoin k
119       (Set.range (GenericPointKaehlerConormal.genericPoint P.asIdeal F)) = ⊤ := by
120     change IntermediateField.adjoin k
121       (Set.range (GenericPointKaehlerConormal.genericPoint P.asIdeal
122         (ComponentFractionField P))) = ⊤
123     have hcoordinates :
124         GenericPointKaehlerConormal.genericPoint P.asIdeal
125           (ComponentFractionField P) = componentCoordinate P := by
126       funext j
127       rfl
128     rw [hcoordinates]
129     exact componentCoordinate_adjoin_eq_top P
130   have hgenericRank : Module.finrank E
131       (AffineConormalSpan.zariskiTangentSpace
132         (fun i : Fin m ↦ algebraMap F E
133           (GenericPointKaehlerConormal.genericPoint P.asIdeal F i))
134         (P.asIdeal.map (MvPolynomial.map (algebraMap k E)))) ≤
135       Module.finrank F (Ω[F⁄k]) :=
136     generic_tangent_finrank_le_component_kaehler_finrank
137       (I := P.asIdeal) (hgen := hgenPoint)
138   have hchartPoint :
139       (fun j : Fin m ↦
140         algebraMap F E (GenericPointKaehlerConormal.genericPoint P.asIdeal F j)) =
141       dehomogenizedPoint
142         (laurentColumn fun a ↦ retainedToCompletedPowerSeries W (C.q a)) := by
143     calc
144       (fun j : Fin m ↦
145           algebraMap F E (GenericPointKaehlerConormal.genericPoint P.asIdeal F j)) = y := by
146         funext j
147         change ell (componentCoordinate P j) = y j
148         exact congrFun hpoint j
149       _ = dehomogenizedPoint
150           (laurentColumn fun a ↦ retainedToCompletedPowerSeries W (C.q a)) := hy
151   rw [hchartPoint] at hgenericRank
152   have hbound := hgenericRank.trans hfieldRank
153   change Nat.le _ _ at hbound
154   change Nat.le _ _
155   exact hbound
