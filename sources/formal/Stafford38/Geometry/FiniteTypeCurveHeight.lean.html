14 /-- In a finite-type domain over a field, transcendence degree one of an actual
15 fraction field bounds the height of every maximal ideal by one. -/
16 theorem maximal_ideal_height_le_one_of_fraction_field_trdeg_one
17     {E A L : Type*} [Field E] [CommRing A] [IsDomain A] [Algebra E A]
18     [Algebra.FiniteType E A] [Field L] [Algebra A L] [IsFractionRing A L]
19     [Algebra E L] [IsScalarTower E A L]
20     (htrdeg : Algebra.trdeg E L = 1)
21     (p : PrimeSpectrum A) [p.asIdeal.IsMaximal] : p.asIdeal.height ≤ 1 := by
22   obtain ⟨f, _hinj, hint⟩ :=
23     Stafford38.Geometry.FiniteTypeCurveNormalization.exists_injective_integral_polynomial_of_isFractionRing_trdeg_one
24       (E := E) (A := A) (L := L) htrdeg
25   exact Stafford38.Geometry.IntegralPolynomialExtensionDimension.height_le_one_of_integral_polynomial_map
26     f hint p

28 /-- A maximal center in a localization of a finite-type curve has height at
29 most one after contraction to the original domain. -/
30 theorem localized_maximal_center_height_le_one_of_fraction_field_trdeg_one
31     {E R A L : Type*}
32     [Field E] [CommRing R] [IsDomain R] {S : Submonoid R}
33     [CommRing A] [IsDomain A]
34     [Algebra R A] [IsLocalization S A]
35     [Algebra E A] [Algebra.FiniteType E A]
36     [Field L] [Algebra A L] [IsFractionRing A L]
37     [Algebra E L] [IsScalarTower E A L]
38     (htrdeg : Algebra.trdeg E L = 1)
39     (q : PrimeSpectrum A) [q.asIdeal.IsMaximal] :
40     (Ideal.under R q.asIdeal).height ≤ 1 := by
41   rw [IsLocalization.height_under (R := R) (S := S) (A := A) q.asIdeal]
42   exact maximal_ideal_height_le_one_of_fraction_field_trdeg_one
43     (E := E) (A := A) (L := L) htrdeg q

45 /-- If the contracted center is nonzero, the preceding bound is sharp. -/
46 theorem localized_maximal_center_height_eq_one_of_fraction_field_trdeg_one
47     {E R A L : Type*}
48     [Field E] [CommRing R] [IsDomain R] {S : Submonoid R}
49     [CommRing A] [IsDomain A]
50     [Algebra R A] [IsLocalization S A]
51     [Algebra E A] [Algebra.FiniteType E A]
52     [Field L] [Algebra A L] [IsFractionRing A L]
53     [Algebra E L] [IsScalarTower E A L]
54     (htrdeg : Algebra.trdeg E L = 1)
55     (q : PrimeSpectrum A) [q.asIdeal.IsMaximal]
56     (h0 : Ideal.under R q.asIdeal ≠ ⊥) :
57     (Ideal.under R q.asIdeal).height = 1 := by
58   let I := Ideal.under R q.asIdeal
59   have hle : I.height ≤ 1 :=
60     localized_maximal_center_height_le_one_of_fraction_field_trdeg_one
61       (E := E) (R := R) (A := A) (L := L) (S := S) htrdeg q
62   have hne : I.height ≠ 0 := by
63     intro h
64     apply h0
65     exact Ideal.height_eq_zero_iff_eq_bot.mp h
66   have hge : 1 ≤ I.height := (Order.one_le_iff_ne_zero).2 hne
67   exact le_antisymm hle hge
68 
69 end Stafford38.Geometry.FiniteTypeCurveHeight
