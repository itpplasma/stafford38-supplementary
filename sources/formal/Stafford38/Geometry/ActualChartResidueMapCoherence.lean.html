14 /-- The actual residue map on the selected coefficient localization agrees
15 with the residue map induced by the chosen coefficient-field embedding. The
16 proof checks equality on the original polynomial ring, using the recorded
17 generator identities at each stage of the same chart/normalization diagram. -/
18 theorem residue_map_coherence
19     {k : Type u} {P : Type v} {E : Type w} {R : Type x} {Q : Type y}
20     {B : Type z} {L : Type a} {V : Type b} {κ : Type c}
21     [CommSemiring k] [CommSemiring P] [IsDomain P]
22     [CommSemiring E] [Algebra P E] [IsLocalization (nonZeroDivisors P) E]
23     [Algebra k P] [Algebra k E] [IsScalarTower k P E]
24     [CommSemiring R] [Algebra k R]
25     [CommSemiring Q] [Algebra k Q]
26     [CommSemiring B] [Algebra R B]
27     [CommSemiring L] [Algebra k L]
28     [CommSemiring V] [Algebra k V]
29     [CommSemiring κ] [Algebra k κ]
30     (eA : P ≃ₐ[k] R)
31     (iA : R →ₐ[k] E) (fEV : E →ₐ[k] V) (fA : R →ₐ[k] V)
32     (g : E →ₐ[k] κ) (fEL : E →ₐ[k] L)
33     (inclBL : B →+* L) (fLoc : L →+* V) (fB : B →+* V)
34     (j : R →+* B) (r0Q : R →ₐ[k] Q) (qV : Q →ₐ[k] V)
35     (rhoV : V →+* κ) (rhoL : L →+* κ)
36     (hELgen : ∀ p : P,
37       fEL (algebraMap P E p) = inclBL (j (eA p)))
38     (hLoc : fLoc.comp inclBL = fB)
39     (hRhoL : rhoL = rhoV.comp fLoc)
40     (hcomp : fB.comp j = (qV.comp r0Q).toRingHom)
41     (hAchart : fA.comp eA.toAlgHom = qV.comp (r0Q.comp eA.toAlgHom))
42     (hfE : (fEV.comp iA).toRingHom = fA.toRingHom)
43     (hiA : ∀ p : P, iA (eA p) = algebraMap P E p)
44     (hresE : rhoV.comp fEV.toRingHom = g.toRingHom) :
45     rhoL.comp fEL.toRingHom = g.toRingHom := by
46   apply IsLocalization.ringHom_ext (M := nonZeroDivisors P)
47   apply RingHom.ext
48   intro p
49   change rhoL (fEL (algebraMap P E p)) = g (algebraMap P E p)
50   calc
51     rhoL (fEL (algebraMap P E p)) =
52         rhoV (fLoc (fEL (algebraMap P E p))) := by
53       rw [hRhoL]
54       rfl
55     _ = rhoV (fLoc (inclBL (j (eA p)))) :=
56       congrArg (fun z : L => rhoV (fLoc z)) (hELgen p)
57     _ = rhoV (fB (j (eA p))) := by
58       exact congrArg rhoV (RingHom.congr_fun hLoc (j (eA p)))
59     _ = rhoV (qV (r0Q (eA p))) := by
60       exact congrArg rhoV (RingHom.congr_fun hcomp (eA p))
61     _ = rhoV (fA (eA p)) := by
62       exact congrArg rhoV (AlgHom.congr_fun hAchart p).symm
63     _ = rhoV (fEV (iA (eA p))) := by
64       exact congrArg rhoV (RingHom.congr_fun hfE (eA p)).symm
65     _ = rhoV (fEV (algebraMap P E p)) := by
66       exact congrArg (fun z : E => rhoV (fEV z)) (hiA p)
67     _ = g (algebraMap P E p) := by
68       exact RingHom.congr_fun hresE (algebraMap P E p)
69 
70 end Stafford38.Geometry.ActualChartResidueMapCoherence
