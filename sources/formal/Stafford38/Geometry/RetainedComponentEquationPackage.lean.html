6 /-!
7 # Finite retained component equations
8 
9 A finite generating family of an affine ideal contained in a prime component
10 is homogenized over the ground field and then transported through the explicit
11 retained Laurent coefficient map.  The resulting homogeneous equations vanish
12 on every retained common-scale completion, and their zeroth-chart
13 dehomogenizations generate exactly the scalar-extended affine ideal.
14 
15 No ambient algebra structure on the residue field is used, and no tangent data
16 are constructed here.
17 -/

47 /-- Mapping coefficients commutes with zeroth-chart dehomogenization. -/
48 theorem projectiveDehomogenize_map
49     {S : Type u} [Field S] (coeff : k →+* S)
50     (H : MvPolynomial (Fin (m + 1)) k) :
51     projectiveDehomogenize (n := m)
52         (MvPolynomial.map coeff H) =
53       MvPolynomial.map coeff (projectiveDehomogenize H) := by
54   simp only [projectiveDehomogenize,
55     Stafford38.Geometry.ProjectiveChartCoordinates.dehomogenizeProjectiveChart_zero]
56   change MvPolynomial.bind₁ (Fin.cases 1 fun j ↦ MvPolynomial.X j)
57       (MvPolynomial.map coeff H) =
58     MvPolynomial.map coeff
59       (MvPolynomial.bind₁ (Fin.cases 1 fun j ↦ MvPolynomial.X j) H)
60   rw [MvPolynomial.map_bind₁]
61   apply congrArg (fun g ↦
62     MvPolynomial.bind₁ g (MvPolynomial.map coeff H))
63   funext a
64   refine Fin.cases ?_ (fun j ↦ ?_) a <;> simp

66 /-- Mapping coefficients therefore commutes with dehomogenizing the standard
67 ground-field homogenization. -/
68 theorem projectiveDehomogenize_map_homogenizeAtZero
69     {S : Type u} [Field S] (coeff : k →+* S)
70     (f : MvPolynomial (Fin m) k) :
71     projectiveDehomogenize (n := m)
72         (MvPolynomial.map coeff (homogenizeAtZero f)) =
73       MvPolynomial.map coeff f := by
74   rw [projectiveDehomogenize_map,
75     projectiveDehomogenize_homogenizeAtZero]

77 /-- The mapped homogenizations of a finite family dehomogenize to precisely
78 the mapped ideal spanned by that family. -/
79 theorem dehomogenizedEquationIdeal_mapped_homogenizations
80     {S : Type u} [Field S] (coeff : k →+* S) {r : ℕ}
81     (generators : Fin r → MvPolynomial (Fin m) k) :
82     dehomogenizedEquationIdeal
83         (fun j ↦ MvPolynomial.map coeff
84           (homogenizeAtZero (generators j))) =
85       (Ideal.span (Set.range generators)).map (MvPolynomial.map coeff) := by
86   rw [dehomogenizedEquationIdeal,
87     Stafford38.Geometry.ProjectiveChartCoordinates.dehomogenizedEquationIdeal]
88   simp_rw [projectiveDehomogenize_map_homogenizeAtZero coeff]
89   rw [Ideal.map_span]
90   congr 1
91   ext x
92   constructor
93   · rintro ⟨j, rfl⟩
94     exact ⟨generators j, ⟨j, rfl⟩, rfl⟩
95   · rintro ⟨_, ⟨j, rfl⟩, rfl⟩
96     exact ⟨j, rfl⟩

98 /-- A finite homogeneous equation package over a displayed coefficient map.
99 Its last field records exact generation, not merely containment. -/
100 structure EquationPackage
101     {S : Type u} [Field S]
102     (I : Ideal (MvPolynomial (Fin m) k)) (coeff : k →+* S)
103     (q : Fin (m + 1) → S) where
104   equationCount : ℕ
105   equations : Fin equationCount → MvPolynomial (Fin (m + 1)) S
106   degree : Fin equationCount → ℕ
107   homogeneous : ∀ j, (equations j).IsHomogeneous (degree j)
108   equations_vanish : ∀ j, MvPolynomial.eval q (equations j) = 0
109   dehomogenizedEquationIdeal_eq :
110     dehomogenizedEquationIdeal equations =
111       I.map (MvPolynomial.map coeff)

113 /-- Exact finite equation package obtained from ground-field generators of
114 `I`.  The coefficient map in both the equations and the ideal equality is the
115 displayed retained Laurent coefficient map. -/
116 theorem retainedComponentEquationPackage
117     [CharZero k]
118     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
119     (I : Ideal (MvPolynomial (Fin m) k)) (hIP : I ≤ P.asIdeal)
120     (i : Fin m)
121     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
122     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
123         (ComponentFractionField P) := W.ambientAlgebra
124     let V := W.place.valuation.toSubring
125     letI : IsDiscreteValuationRing V := W.place.isDiscrete
126     letI : Algebra W.coefficientField V :=
127       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
128     ∀ (q : Fin (m + 1) → V) (scale : ComponentFractionField P),
129       (∀ a, (q a : ComponentFractionField P) =
130         scale * componentProjectivePoint P a) →
131       Nonempty (EquationPackage I
132         (retainedLaurentCoefficientMap P i W)
133         (fun a ↦ algebraMap (PowerSeries (ResidueField V))
134           (LaurentSeries (ResidueField V))
135             (retainedToCompletedPowerSeries W (q a)))) := by
136   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
137       (ComponentFractionField P) := W.ambientAlgebra
138   let V := W.place.valuation.toSubring
139   letI : IsDiscreteValuationRing V := W.place.isDiscrete
140   letI : Algebra W.coefficientField V :=
141     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
142   dsimp only
143   intro q scale hq
144   obtain ⟨r, generators, hgenerators⟩ :=
145     Submodule.fg_iff_exists_fin_generating_family.mp
146       (IsNoetherian.noetherian (I : Submodule _ _))
147   let coeff : k →+* LaurentSeries (ResidueField V) :=
148     retainedLaurentCoefficientMap P i W
149   let equations : Fin r →
150       MvPolynomial (Fin (m + 1)) (LaurentSeries (ResidueField V)) :=
151     fun j ↦ MvPolynomial.map coeff
152       (homogenizeAtZero (generators j))
153   let degree : Fin r → ℕ :=
154     fun j ↦ (generators j).totalDegree
155   have hgenerator_mem : ∀ j, generators j ∈ I := by
156     intro j
157     rw [← hgenerators]
158     exact Submodule.subset_span (Set.mem_range_self j)
159   refine ⟨{
160     equationCount := r
161     equations := equations
162     degree := degree
163     homogeneous := ?_
164     equations_vanish := ?_
165     dehomogenizedEquationIdeal_eq := ?_ }⟩
166   · intro j
167     exact (homogenizeAtZero_isHomogeneous
168       (generators j)).map coeff
169   · intro j
170     change MvPolynomial.eval _
171       (MvPolynomial.map coeff
172         (homogenizeAtZero
173           (generators j))) = 0
174     rw [MvPolynomial.eval_map]
175     exact retainedLaurent_eval₂_eq_zero_of_commonScale P i W
176       (homogenizeAtZero_isHomogeneous
177         (generators j))
178       (homogenizeAtZero_mem_componentProjectiveClosureIdeal P
179         (hIP (hgenerator_mem j))) q scale hq
180   · change dehomogenizedEquationIdeal
181         (fun j ↦ MvPolynomial.map coeff
182           (homogenizeAtZero (generators j))) =
183       I.map (MvPolynomial.map coeff)
184     rw [dehomogenizedEquationIdeal_mapped_homogenizations]
185     change Ideal.map (MvPolynomial.map coeff)
186         (Submodule.span _ (Set.range generators)) =
187       I.map (MvPolynomial.map coeff)
188     rw [hgenerators]
