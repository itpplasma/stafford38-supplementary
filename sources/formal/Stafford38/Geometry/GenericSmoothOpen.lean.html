9 # A generic smooth principal open
10 
11 For a finite-type domain over a perfect field, the generic localization is a
12 finitely generated field extension and hence formally smooth.  Mathlib's
13 smooth-locus spreading theorem then supplies a nonzero principal open which
14 is smooth.  This file deliberately makes no Jacobian or conormal
15 identification.
16 -/

22 variable {k A : Type*} {n : ℕ} [Field k] [PerfectField k] [CommRing A]
23   [IsDomain A] [Algebra k A] [Algebra.FiniteType k A]
24   [Algebra.FinitePresentation k A]

26 theorem exists_nonzero_smooth_away
27     : ∃ f : A, f ≠ 0 ∧ Algebra.Smooth k (Localization.Away f) := by
28   letI : Algebra.IsSmoothAt k (⊥ : Ideal A) := by
29     change Algebra.FormallySmooth k (Localization.AtPrime (⊥ : Ideal A))
30     letI : Field (Localization.AtPrime (⊥ : Ideal A)) :=
31       IsField.toField (by
32         rw [IsLocalRing.isField_iff_maximalIdeal_eq]
33         rw [← Localization.AtPrime.map_eq_maximalIdeal]
34         simp)
35     exact Algebra.FormallySmooth.of_perfectField
36       (K := k) (L := Localization.AtPrime (⊥ : Ideal A))
37   obtain ⟨f, hf, hsmooth⟩ :=
38     Algebra.IsSmoothAt.exists_notMem_smooth k (⊥ : Ideal A)
39   exact ⟨f, by simpa using hf, hsmooth⟩
40 
41 theorem exists_nonzero_smooth_away_quotient
42     (P : Ideal (MvPolynomial (Fin n) k)) [P.IsPrime] :
43     ∃ f : (MvPolynomial (Fin n) k ⧸ P), f ≠ 0 ∧
44       Algebra.Smooth k (Localization.Away f) := by
45   let A := MvPolynomial (Fin n) k ⧸ P
46   letI : Algebra k A := Ideal.Quotient.algebra k
47   letI : Algebra.FinitePresentation k A := by
48     dsimp [A]
49     exact Algebra.FinitePresentation.quotient P.fg_of_isNoetherianRing
50   simpa [A] using (exists_nonzero_smooth_away (k := k) (A := A))
