8 # Projective residue frames and the generic tangent rank bound
9 
10 This file extracts two consequences which are already forced by the completed
11 projective/DVR certificate but were previously left as consumer hypotheses.
12 
13 First, full column rank of an augmented projective matrix `[q | B]` implies
14 linear independence of the affine columns obtained by differentiating in any
15 chart where `q` is nonzero.  The chart is arbitrary: the completed certificate
16 does not identify its normalized chart coordinate with coordinate zero.
17 
18 Second, the stored Laurent tangent inclusion bounds the generic tangent
19 dimension by the number of nonposition columns, namely `tangentCount + 1`.
20 This is a generic-fibre statement.  No equality with an arc-derivation frame
21 and no closed-boundary tangent bound is asserted; either conclusion needs an
22 additional geometric comparison.
23 -/

51 /-- Adjoin the projective position column to a family of tangent columns. -/
52 def augmentedProjectiveMatrix {ι : Type v} {κ : Type w}
53     (q : ι → K) (B : Matrix ι κ K) : Matrix ι (Unit ⊕ κ) K
54   | i, Sum.inl _ => q i
55   | i, Sum.inr j => B i j
56 
57 /-- The nonposition part of the formal power-series tangent matrix. -/
58 def nonpositionPowerSeriesMatrix {k : Type u} {ι : Type v} {κ : Type w}
59     [Field k] (Z : Matrix ι κ (PowerSeries k))
60     (tau : ι → PowerSeries k) : Matrix ι (κ ⊕ Unit) (PowerSeries k)
61   | i, Sum.inl j => Z i j
62   | i, Sum.inr _ => tau i

64 /-- Independent projective position and tangent columns remain independent
65 after passing to affine tangent vectors in any nonvanishing chart. -/
66 theorem linearIndependent_chartDehomogenizedTangentColumns
67     {ι : Type v} {κ : Type w} [Fintype κ]
68     (chart : ι) (q : ι → K) (B : Matrix ι κ K)
69     (hq : q chart ≠ 0)
70     (hinjective : Function.Injective (augmentedProjectiveMatrix q B).mulVec) :
71     LinearIndependent K
72       (fun j ↦ chartDehomogenizedTangentColumn chart q (fun i ↦ B i j)) := by
73   classical
74   apply Fintype.linearIndependent_iff.mpr
75   intro c hc j
76   let s : K := -(∑ a, c a * B chart a) / q chart
77   let c' : Unit ⊕ κ → K
78     | Sum.inl _ => s
79     | Sum.inr a => c a
80   have hc' : (augmentedProjectiveMatrix q B).mulVec c' = 0 := by
81     funext i
82     simp only [Pi.zero_apply, Matrix.mulVec, dotProduct,
83       augmentedProjectiveMatrix, c', Fintype.sum_sum_type,
84       Fintype.sum_unique, one_mul]
85     change q i * s + ∑ a, B i a * c a = 0
86     by_cases hi : i = chart
87     · subst i
88       rw [show s = -(∑ a, c a * B chart a) / q chart by rfl,
89         mul_div_cancel₀ _ hq]
90       simp [mul_comm]
91     · have hcoordinate := congrFun hc ⟨i, hi⟩
92       simp only [Pi.zero_apply] at hcoordinate
93       simp [chartDehomogenizedTangentColumn] at hcoordinate
94       simp_rw [← mul_div_assoc] at hcoordinate
95       rw [← Finset.sum_div] at hcoordinate
96       simp only [s]
97       field_simp [hq] at hcoordinate ⊢
98       have hcoordinate' :
99           q chart * (∑ a, B i a * c a) -
100             q i * (∑ a, B chart a * c a) = 0 := by
101         simpa [mul_sub, Finset.sum_sub_distrib, Finset.sum_mul,
102           ← Finset.mul_sum, mul_comm, mul_left_comm, mul_assoc] using hcoordinate
103       have hchartSum : (∑ a, c a * B chart a) =
104           ∑ a, B chart a * c a := by
105         apply Finset.sum_congr rfl
106         intro a ha
107         exact mul_comm _ _
108       rw [hchartSum]
109       linear_combination hcoordinate'
110   have hc'_zero : c' = 0 := by
111     apply hinjective
112     simpa using hc'
113   exact congrFun hc'_zero (Sum.inr j)

115 /-- Entrywise reduction of a power-series left inverse remains a left inverse
116 of the residue matrix. -/
117 theorem residueMatrix_mul_eq_one_of_mul_eq_one
118     {k : Type u} [Field k] {ι : Type v} {κ : Type w}
119     [Fintype ι] [Fintype κ] [DecidableEq κ]
120     (C : Matrix κ ι (PowerSeries k)) (B : Matrix ι κ (PowerSeries k))
121     (hCB : C * B = 1) :
122     residueMatrix C * residueMatrix B = 1 := by
123   ext i j
124   have hij := congrArg (PowerSeries.constantCoeff)
125     (congrFun (congrFun hCB i) j)
126   simpa [Matrix.mul_apply, Matrix.one_apply, residueMatrix, map_sum] using hij
127 
128 /-- A power-series left inverse therefore gives full column rank after
129 reduction modulo the uniformizer. -/
130 theorem residueMatrix_mulVec_injective_of_leftInverse
131     {k : Type u} [Field k] {ι : Type v} {κ : Type w}
132     [Fintype ι] [Fintype κ] [DecidableEq κ]
133     (C : Matrix κ ι (PowerSeries k)) (B : Matrix ι κ (PowerSeries k))
134     (hCB : C * B = 1) :
135     Function.Injective (residueMatrix B).mulVec := by
136   intro x y hxy
137   have h := congrArg (residueMatrix C).mulVec hxy
138   rw [Matrix.mulVec_mulVec, Matrix.mulVec_mulVec,
139     residueMatrix_mul_eq_one_of_mul_eq_one C B hCB,
140     Matrix.one_mulVec] at h
141   simpa using h

143 /-- The completed certificate's split formal matrix produces a genuinely
144 independent affine residue frame in its normalized projective chart. -/
145 theorem completedBoundaryChart_residueFrame_linearIndependent
146     {k : Type u} [Field k] [CharZero k]
147     {m : ℕ} (hm : 0 < m)
148     (I : Ideal (MvPolynomial (Fin m) k))
149     (W : CompletedProjectiveBoundaryChart k m hm I) :
150     LinearIndependent k
151       (fun j : (Fin W.tangentCount ⊕ Unit) ↦
152         chartDehomogenizedTangentColumn W.chart
153           (fun i ↦ PowerSeries.constantCoeff (W.q i))
154           (fun i ↦ PowerSeries.constantCoeff
155             (nonpositionPowerSeriesMatrix W.Z W.tau i j))) := by
156   apply linearIndependent_chartDehomogenizedTangentColumns W.chart
157   · simp [W.q_chart]
158   · have hinj := residueMatrix_mulVec_injective_of_leftInverse
159       W.C (formalTangentMatrix W.q W.Z W.tau) W.left_inverse
160     have hmatrix :
161         residueMatrix (formalTangentMatrix W.q W.Z W.tau) =
162           augmentedProjectiveMatrix
163             (fun i ↦ PowerSeries.constantCoeff (W.q i))
164             (fun i j ↦ PowerSeries.constantCoeff
165               (nonpositionPowerSeriesMatrix W.Z W.tau i j)) := by
166       ext i j
167       rcases j with (_ | (j | _)) <;>
168         rfl
169     rw [hmatrix] at hinj
170     exact hinj

172 /-- The stored generic tangent inclusion gives the exact numerical bound by
173 the number of supplied nonposition columns. -/
174 theorem completedBoundaryChart_genericTangent_finrank_le
175     {k : Type u} [Field k] [CharZero k]
176     {m : ℕ} (hm : 0 < m)
177     (I : Ideal (MvPolynomial (Fin m) k))
178     (W : CompletedProjectiveBoundaryChart k m hm I) :
179     Module.finrank (LaurentSeries k)
180         (zariskiTangentSpace (dehomogenizedPoint (laurentColumn W.q))
181           (I.map (Stafford38.Geometry.ScalarExtensionPoints.scalarPolynomialMap
182             (k := k) (K := LaurentSeries k) (Fin m)))) ≤
183       W.tangentCount + 1 := by
184   calc
185     Module.finrank (LaurentSeries k)
186         (zariskiTangentSpace (dehomogenizedPoint (laurentColumn W.q))
187           (I.map (Stafford38.Geometry.ScalarExtensionPoints.scalarPolynomialMap
188             (k := k) (K := LaurentSeries k) (Fin m)))) ≤
189         Module.finrank (LaurentSeries k)
190           (dehomogenizedTangentSpan (laurentColumn W.q)
191             (laurentNonpositionTangentMatrix W.Z W.tau)) :=
192       Submodule.finrank_mono W.tangent_inclusion
193     _ ≤ Fintype.card (Fin W.tangentCount ⊕ Unit) :=
194       finrank_range_le_card
195         (fun j ↦ dehomogenizedTangentColumn (laurentColumn W.q)
196           (fun i ↦ laurentNonpositionTangentMatrix W.Z W.tau i j))
197     _ = W.tangentCount + 1 := by simp
