31 /-- The normalization of the selected affine chart becomes the selected chart
32 itself after inverting one nonzero element of the chart domain. -/
33 theorem actual_selected_normalization_is_away_equiv
34     {k : Type u} [Field k] [CharZero k] {m : ℕ} {hm : 0 < m}
35     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
36     (w : GeneralDivisorialVisibleFrameWitness hm P) :
37     ∃ f : actualSelectedChartAlgebra P w, f ≠ 0 ∧
38       Nonempty
39         (Localization.Away f ≃ₐ[actualSelectedChartAlgebra P w]
40           Localization.Away
41             (algebraMap (actualSelectedChartAlgebra P w)
42               (actualSelectedNormalization P w) f)) := by
43   let Q := actualSelectedChartAlgebra P w
44   let F := ComponentFractionField P
45   let B := actualSelectedNormalization P w
46   letI : Algebra Q F := Q.val.toAlgebra
47   let echart := chartAffineCoordinateEquiv w.column.chart
48   have hchart : componentProjectivePoint P w.column.chart ≠ 0 := by
49     have hscale : w.column.scale * componentProjectivePoint P w.column.chart = 1 := by
50       simpa [w.column.chart_one] using (w.column.q_commonScale w.column.chart).symm
51     intro hz
52     rw [hz, mul_zero] at hscale
53     exact one_ne_zero hscale.symm
54   letI : Algebra Q.toSubring F := Q.toSubring.subtype.toAlgebra
55   have hQfrac : IsFractionRing Q.toSubring F := by
56     letI : IsFractionRing Q F :=
57       chartGenericPointSubalgebra_isFractionRing P w.column.chart echart hchart
58     letI : FaithfulSMul Q.toSubring F :=
59       (faithfulSMul_iff_algebraMap_injective Q.toSubring F).mpr Subtype.val_injective
60     apply IsFractionRing.of_field Q.toSubring F
61     intro z
62     obtain ⟨a, b, hb, hz⟩ := IsFractionRing.div_surjective Q z
63     let eqv := subalgebraToSubringRingEquiv Q
64     refine ⟨eqv a, eqv b, ?_⟩
65     change z = (eqv a : F) / (eqv b : F)
66     calc
67       z = (a : F) / (b : F) := hz.symm
68       _ = (eqv a : F) / (eqv b : F) := by rfl
69   letI : IsFractionRing Q.toSubring F := hQfrac
70   letI : Algebra k Q.toSubring := Q.algebra
71   have hQfinite : Algebra.FiniteType k Q.toSubring := by
72     change Algebra.FiniteType k Q
73     exact chartGenericPointSubalgebra_finiteType P w.column.chart echart
74   letI : Algebra.FiniteType k Q.toSubring := hQfinite
75   letI : Algebra Q.toSubring B := B.algebra
76   have hfiniteSubring : Module.Finite Q.toSubring B := by
77     change Module.Finite Q.toSubring (integralClosure Q.toSubring F)
78     exact finite_integralClosure_in_fractionField k Q.toSubring F
79   let eqv := subalgebraToSubringRingEquiv Q
80   letI : Algebra Q.toSubring Q := eqv.symm.toRingHom.toAlgebra
81   letI : Algebra Q B := (chartSubalgebraToIntegralClosure Q).toAlgebra
82   have hTowerSubring : IsScalarTower Q.toSubring Q B := by
83     refine IsScalarTower.of_algebraMap_eq fun r => ?_
84     apply Subtype.ext
85     rfl
86   letI : IsScalarTower Q.toSubring Q B := hTowerSubring
87   letI : Module.Finite Q.toSubring B := hfiniteSubring
88   have hfinite : Module.Finite Q B :=
89     Module.Finite.of_restrictScalars_finite Q.toSubring Q B
90   letI : IsFractionRing Q F :=
91     chartGenericPointSubalgebra_isFractionRing P w.column.chart
92       (chartAffineCoordinateEquiv w.column.chart) hchart
93   letI : Algebra B F := B.val.toAlgebra
94   have hTowerQ : IsScalarTower Q B F := by
95     refine IsScalarTower.of_algebraMap_eq fun q => ?_
96     change (q : F) = ((chartSubalgebraToIntegralClosure Q q : B) : F)
97     rw [coe_chartSubalgebraToIntegralClosure]
98   exact @exists_nonzero_away_equiv_of_finite Q _ _ F _ _ _ B _ _ _
99     hTowerQ hfinite Subtype.val_injective
