15 /-- A ring map taking the adic ideal onto the target ideal induces a map of
16 adic completions. -/
17 def mapOfRingHom (f : A →+* B) (I : Ideal A) (J : Ideal B)
18     (hJ : J = I.map f) : AdicCompletion I A →+* AdicCompletion J B := by
19   classical
20   let stage : (n : ℕ) → AdicCompletion I A →+* B ⧸ J ^ n := fun n =>
21     (Ideal.quotientMap (J ^ n) f (by
22       have hpow : (I ^ n).map f = J ^ n := by rw [Ideal.map_pow, hJ]
23       intro a ha
24       rw [← hpow]
25       exact Ideal.mem_map_of_mem f ha)).comp (AdicCompletion.evalₐ I n).toRingHom
26   have hstage : ∀ {m n : ℕ} (hle : m ≤ n),
27       (Ideal.Quotient.factorPow J hle).comp (stage n) = stage m := by
28     intro m n hle
29     ext x
30     apply AdicCompletion.induction_on I A x
31     intro c
32     simp only [stage, RingHom.comp_apply]
33     change Ideal.Quotient.mk (J ^ m) (f (c n)) =
34       Ideal.Quotient.mk (J ^ m) (f (c m))
35     apply Ideal.Quotient.eq.mpr
36     have hc : c m - c n ∈ I ^ m := by
37       have hc' := SModEq.sub_mem.mp (c.property hle)
38       simpa [smul_eq_mul, Ideal.mul_top] using hc'
39     have hneg : c n - c m ∈ I ^ m := by simpa using (I ^ m).neg_mem hc
40     have hmap : f (c n) - f (c m) ∈ (I ^ m).map f := by
41       simpa only [map_sub] using Ideal.mem_map_of_mem f hneg
42     have hpow : (I ^ m).map f = J ^ m := by rw [Ideal.map_pow, hJ]
43     rw [hpow] at hmap
44     exact hmap
45   exact AdicCompletion.liftRingHom J stage hstage

47 theorem eval_mapOfRingHom (f : A →+* B) (I : Ideal A) (J : Ideal B)
48     (hJ : J = I.map f) (x : AdicCompletion I A) (n : ℕ) :
49     AdicCompletion.evalₐ J n (mapOfRingHom f I J hJ x) =
50       Ideal.quotientMap (J ^ n) f (by
51         have hpow : (I ^ n).map f = J ^ n := by rw [Ideal.map_pow, hJ]
52         intro a ha
53         rw [← hpow]
54         exact Ideal.mem_map_of_mem f ha) (AdicCompletion.evalₐ I n x) := by
55   classical
56   unfold mapOfRingHom
57   simp only [AdicCompletion.evalₐ_liftRingHom]
58   rfl

61 theorem mapOfRingHom_apply_of (f : A →+* B) (I : Ideal A) (J : Ideal B)
62     (hJ : J = I.map f) (a : A) :
63     mapOfRingHom f I J hJ (AdicCompletion.of I A a) =
64       AdicCompletion.of J B (f a) := by
65   classical
66   apply AdicCompletion.ext_evalₐ
67   intro n
68   rw [eval_mapOfRingHom]
69   simp

71 theorem mapOfRingHom_comp {C : Type*} [CommRing C]
72     (f : A →+* B) (g : B →+* C) (I : Ideal A) (J : Ideal B) (K : Ideal C)
73     (hJ : J = I.map f) (hK : K = J.map g) :
74     (mapOfRingHom g J K hK).comp (mapOfRingHom f I J hJ) =
75       mapOfRingHom (g.comp f) I K (by rw [hK, hJ, Ideal.map_map]) := by
76   have hIK : K = I.map (g.comp f) := by rw [hK, hJ, Ideal.map_map]
77   apply RingHom.ext
78   intro x
79   apply AdicCompletion.ext_evalₐ
80   intro n
81   change AdicCompletion.evalₐ K n
82       (mapOfRingHom g J K hK (mapOfRingHom f I J hJ x)) = _
83   rw [eval_mapOfRingHom, eval_mapOfRingHom]
84   let qf : A ⧸ I ^ n →+* B ⧸ J ^ n :=
85     Ideal.quotientMap (J ^ n) f (by
86       have hpow : (I ^ n).map f = J ^ n := by rw [Ideal.map_pow, hJ]
87       intro a ha; rw [← hpow]; exact Ideal.mem_map_of_mem f ha)
88   let qg : B ⧸ J ^ n →+* C ⧸ K ^ n :=
89     Ideal.quotientMap (K ^ n) g (by
90       have hpow : (J ^ n).map g = K ^ n := by rw [Ideal.map_pow, hK]
91       intro b hb; rw [← hpow]; exact Ideal.mem_map_of_mem g hb)
92   let qcomp : A ⧸ I ^ n →+* C ⧸ K ^ n :=
93     Ideal.quotientMap (K ^ n) (g.comp f) (by
94       have hpow : (I ^ n).map (g.comp f) = K ^ n := by rw [Ideal.map_pow, hIK]
95       intro a ha; rw [← hpow]; exact Ideal.mem_map_of_mem (g.comp f) ha)
96   rw [eval_mapOfRingHom (g.comp f) I K hIK x n]
97   change qg (qf (AdicCompletion.evalₐ I n x)) =
98     qcomp (AdicCompletion.evalₐ I n x)
99   have hq : qg.comp qf = qcomp := by
100     apply RingHom.ext
101     intro y
102     induction y using Quotient.inductionOn' with
103     | h a =>
104       change Ideal.Quotient.mk (K ^ n) (g (f a)) =
105         Ideal.Quotient.mk (K ^ n) ((g.comp f) a)
106       simp
107   exact RingHom.congr_fun hq _
108 
109 end
110 end Stafford38.Geometry.AdicCompletionMap
