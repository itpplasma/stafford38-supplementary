7 /-!
8 # Density of the conormal over the smooth affine locus
9 
10 For a prime affine variety over an algebraically closed field, its genuine
11 Mathlib smooth points contain a nonempty principal open.  Consequently the
12 equation conormal over those smooth points has the same algebraic closure as
13 the full equation conormal.
14 -/

23 variable {k : Type*} [Field k] [IsAlgClosed k] {n : ℕ}
24 
25 def SmoothAffinePoint (I : Ideal (MvPolynomial (Fin n) k))
26     (y : Fin n → k) : Prop :=
27   ∃ e : (MvPolynomial (Fin n) k ⧸ I) →ₐ[k] k,
28     (∀ p, e (Ideal.Quotient.mk I p) = MvPolynomial.aeval y p) ∧
29     let hprime : (RingHom.ker e.toRingHom).IsPrime := RingHom.ker_isPrime e
30     (⟨RingHom.ker e.toRingHom, hprime⟩ :
31       PrimeSpectrum (MvPolynomial (Fin n) k ⧸ I)) ∈
32       Algebra.smoothLocus k (MvPolynomial (Fin n) k ⧸ I)
33 
34 theorem smoothAffinePoint_mem_zeroLocus
35     (I : Ideal (MvPolynomial (Fin n) k)) :
36     {y | SmoothAffinePoint I y} ⊆ MvPolynomial.zeroLocus k I := by
37   intro y hy p hp
38   obtain ⟨e, heval, _⟩ := hy
39   have hmk : Ideal.Quotient.mk I p = 0 :=
40     Ideal.Quotient.eq_zero_iff_mem.mpr hp
41   rw [← heval p, hmk, map_zero]
42 
43 theorem equationConormalClosure_smoothAffine_eq
44     (I : Ideal (MvPolynomial (Fin n) k)) (hI : I.IsPrime) :
45     MvPolynomial.zeroLocus k
46         (MvPolynomial.vanishingIdeal k
47           (restrictedEquationConormalLocus I {y | SmoothAffinePoint I y})) =
48       equationConormalClosure I := by
49   letI : I.IsPrime := hI
50   letI : Algebra.FinitePresentation k (MvPolynomial (Fin n) k ⧸ I) :=
51     Algebra.FinitePresentation.quotient I.fg_of_isNoetherianRing
52   obtain ⟨fbar, hfbar, hsmooth⟩ := exists_nonzero_smooth_away_quotient I
53   obtain ⟨f, rfl⟩ := Ideal.Quotient.mk_surjective fbar
54   have hf : f ∉ I := by
55     intro h
56     exact hfbar (Ideal.Quotient.eq_zero_iff_mem.mpr h)
57   apply equationConormalClosure_restricted_eq I hI f hf
58     {y | SmoothAffinePoint I y}
59     (smoothAffinePoint_mem_zeroLocus I)
60   intro y hyI hyf
61   let e : (MvPolynomial (Fin n) k ⧸ I) →ₐ[k] k :=
62     Ideal.Quotient.liftₐ I (MvPolynomial.aeval y) (by
63       intro p hp
64       exact hyI p hp)
65   refine ⟨e, ?_, ?_⟩
66   · intro p
67     simp [e]
68   · have hopen : (⟨RingHom.ker e.toRingHom, RingHom.ker_isPrime e⟩ :
69         PrimeSpectrum (MvPolynomial (Fin n) k ⧸ I)) ∈
70         PrimeSpectrum.basicOpen (Ideal.Quotient.mk I f) := by
71       rw [PrimeSpectrum.mem_basicOpen]
72       intro hker
73       have hezero : e (Ideal.Quotient.mk I f) = 0 := hker
74       simp [e] at hezero
75       exact hyf hezero
76     exact (Algebra.basicOpen_subset_smoothLocus_iff_smooth.mpr hsmooth) hopen
