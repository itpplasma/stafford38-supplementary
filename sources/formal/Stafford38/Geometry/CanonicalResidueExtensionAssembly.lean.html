10 /-!
11 # Canonical geometry over the boundary residue field
12 
13 The residue field `K` of a projective boundary divisor need not equal the
14 ground field `k`.  The completed arc therefore lives in `K[[t]]`, while the
15 canonical Weyl operator, its principal symbol, and the contracted base ideal
16 remain defined over `k`.
17 
18 This file repairs the terminal rank split at that interface.  In ambient rank
19 one the existing direct argument stays over `k`.  In higher rank the producer
20 is allowed to choose an arbitrary extension field `K/k` and returns the
21 Laurent conormal point actually consumed by the symbol contradiction.  It may
22 obtain that point directly from a constant-coordinate component or from a
23 completed boundary chart.  Symbol vanishing is requested over the same chosen
24 extension, and the contradiction is completed there.  No map `K → k`
25 occurs.
26 
27 The results are conditional assemblers.  They do not produce the
28 higher-dimensional conormal axis, prove post-extension Poisson/Gabber
29 integrability, prove noncharacteristic restriction, or prove the canonical
30 coordinate-preimage hypothesis.
31 -/

64 /-- Symbol control after extension to the Laurent field of an arbitrary
65 residue extension `K/k`.
66 
67 This is the exact geometric-support input used below.  It is deliberately not
68 called a theorem of Gabber: a caller must still prove it for the concrete
69 canonical quotient and the particular extension selected by boundary
70 geometry. -/
71 def CanonicalResidueExtensionSymbolControl : Prop :=
72   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
73     (K : Type u) [Field K] [Algebra k K]
74     (n N : ℕ) (d : WeylAlg k (n + 1)),
75     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
76       ∃ P : MvPolynomial (Fin (n + 1)) k,
77         fibreLift P = presentedPrincipalComponent k orderWeight N d ∧
78         ∀ q ∈ groundEquationConormalLocus (k := k) (K := K)
79             (reducedOrderBaseIdeal k
80               (canonicalRightIdeal (presentedCoordinate k n) d N)),
81           MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
82             (fibreLift P) = 0
83 
84 /-- Higher-dimensional production of the exact conormal-axis endpoint used by
85 the symbol contradiction, over its natural existential residue extension.
86 
87 The strict inequality excludes ambient rank one, which is handled directly.
88 In higher rank this interface deliberately does not prescribe how the point is
89 produced: the constant-coordinate branch may take `K = k`, while the
90 nonconstant branch may extract it from a completed boundary chart over a
91 genuine residue extension. -/
92 def HigherDimensionalCanonicalResidueExtensionConormalAxisProduction : Prop :=
93   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
94     (n N : ℕ) (d : WeylAlg k (n + 1)),
95     0 < n →
96     0 < N →
97     IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
98     Disjoint
99       (orderCharacteristicSupport k
100         (canonicalRightIdeal (presentedCoordinate k n) d N))
101       (PrimeSpectrum.zeroLocus
102         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
103           Set (SymbolRing k (n + 1)))) →
104     (orderCharacteristicSupport k
105       (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty →
106     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
107       (y : Fin (n + 1) → LaurentSeries K)
108       (xi : Fin (n + 1) → PowerSeries K),
109       Sum.elim y
110           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
111         groundEquationConormalLocus (k := k) (K := K)
112           (reducedOrderBaseIdeal k
113             (canonicalRightIdeal (presentedCoordinate k n) d N)) ∧
114       residueColumn xi =
115         (fun i : Fin (n + 1) ↦
116           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)

118 /-! ## One chart gives the terminal contradiction over `K` -/
119 
120 /-- The canonical principal symbol evaluates to one on the pure first-fibre
121 axis before and after extension from `k` to `K`. -/
122 theorem canonical_fibrePolynomial_eval_extensionAxis
123     {k K : Type u} [Field k] [Field K] [Algebra k K]
124     {n N : ℕ} {d : WeylAlg k (n + 1)}
125     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
126     (P : MvPolynomial (Fin (n + 1)) k)
127     (hP : fibreLift P = presentedPrincipalComponent k orderWeight N d) :
128     MvPolynomial.eval₂ (algebraMap k K)
129         (fun i : Fin (n + 1) ↦
130           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) P = 1 := by
131   have hcanonical :=
132     canonical_orderPrincipalComponent_eval_pureMomentumAxis n N hd
133   have hsplit :
134       Sum.elim (fun _ : Fin (n + 1) ↦ (0 : k))
135           (fun i : Fin (n + 1) ↦
136             if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) =
137         axisPoint k (.inr (0 : Fin (n + 1))) := by
138     funext i
139     rcases i with i | i
140     · simp [axisPoint]
141     · simp [axisPoint] <;> rfl
142   rw [← hP] at hcanonical
143   have heval := eval₂_fibreLift (K := k) P
144     (fun _ : Fin (n + 1) ↦ (0 : k))
145     (fun i : Fin (n + 1) ↦
146       if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)
147   rw [hsplit] at heval
148   have hground : MvPolynomial.eval
149       (fun i : Fin (n + 1) ↦
150         if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) P = 1 := by
151     simpa only [← MvPolynomial.aeval_def, MvPolynomial.aeval_eq_eval] using
152       heval.symm.trans hcanonical
153   rw [eval₂_extensionAxis_eq_algebraMap_eval_axis
154     (K := K) (Nat.zero_lt_succ n) P, hground]
155   exact map_one (algebraMap k K)

157 /-- A residue-extension completed chart, together with canonical symbol
158 vanishing over the same Laurent field, is contradictory.  The conclusion is
159 proved in `K`; no residue-field descent or retraction is used. -/
160 theorem false_of_canonical_completedProjectiveBoundaryChartOver
161     {k K : Type u} [Field k] [CharZero k]
162     [Field K] [Algebra k K]
163     (n N : ℕ) (d : WeylAlg k (n + 1))
164     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
165     (P : MvPolynomial (Fin (n + 1)) k)
166     (hP : fibreLift P = presentedPrincipalComponent k orderWeight N d)
167     (W : CompletedProjectiveBoundaryChartOver
168       (k := k) (K := K) (n + 1) (Nat.zero_lt_succ n)
169       (reducedOrderBaseIdeal k
170         (canonicalRightIdeal (presentedCoordinate k n) d N)))
171     (hvanishes :
172       ∀ q ∈ groundEquationConormalLocus (k := k) (K := K)
173           (reducedOrderBaseIdeal k
174             (canonicalRightIdeal (presentedCoordinate k n) d N)),
175         MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
176           (fibreLift P) = 0) : False := by
177   letI : CharZero K :=
178     charZero_of_injective_algebraMap (algebraMap k K).injective
179   obtain ⟨y, xi, hgeneric, hresidue⟩ :=
180     exists_conormalAxis_of_completedProjectiveBoundaryChartOver
181       (k := k) (K := K) (Nat.zero_lt_succ n)
182       (reducedOrderBaseIdeal k
183         (canonicalRightIdeal (presentedCoordinate k n) d N)) W
184   exact false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
185     (k := k) (K := K)
186     (reducedOrderBaseIdeal k
187       (canonicalRightIdeal (presentedCoordinate k n) d N))
188     P
189     (fun i : Fin (n + 1) ↦
190       if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)
191     y xi hgeneric hresidue hvanishes
192     (canonical_fibrePolynomial_eval_extensionAxis
193       (K := K) hd P hP)

197 /-- Conditional support vanishing with the corrected coefficient fields.
198 
199 The rank-one branch uses the existing direct ambient conormal theorem over
200 `k`.  In the successor branch, conormal-axis production chooses `K/k`; symbol
201 control and that point are then consumed over precisely that `K`.  The theorem
202 never specializes a `K`-valued point to `k[[t]]`. -/
203 theorem algebraicallyClosedCanonicalSupportVanishing_of_residueExtension_rankSplit
204     (hunit : CanonicalStrictUnitCoordinatePreimage.{u})
205     (hcontrol : CanonicalResidueExtensionSymbolControl.{u})
206     (hproduction :
207       HigherDimensionalCanonicalResidueExtensionConormalAxisProduction.{u}) :
208     AlgebraicallyClosedCanonicalSupportVanishing.{u} := by
209   intro k _ _ _ n N d hN hd
210   let I := canonicalRightIdeal (presentedCoordinate k n) d N
211   have hdisjoint : Disjoint
212       (orderCharacteristicSupport k I)
213       (PrimeSpectrum.zeroLocus
214         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
215           Set (SymbolRing k (n + 1)))) :=
216     canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
217       k n N (hunit k n N d hN hd)
218   by_contra hsupp
219   have hnonempty : (orderCharacteristicSupport k I).Nonempty :=
220     Set.nonempty_iff_ne_empty.mpr hsupp
221   cases n with
222   | zero =>
223       obtain ⟨P, hP, hvanishes⟩ := hcontrol k k 0 N d hN hd
224       obtain ⟨y, xi, hgeneric, hresidue⟩ :=
225         exists_canonical_rankOne_laurentConormalAxis I hdisjoint hnonempty
226       apply false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
227         (k := k) (K := k) (reducedOrderBaseIdeal k I) P
228           (fun _ : Fin 1 ↦ (1 : k)) y xi
229       · simpa [groundEquationConormalLocus, scalarPolynomialMap,
230           groundLaurentMap, scalarPolynomialMap] using hgeneric
231       · exact hresidue
232       · exact hvanishes
233       · have haxis := canonical_fibrePolynomial_eval_extensionAxis
234           (K := k) hd P hP
235         have hfun : (fun _ : Fin 1 ↦ (1 : k)) =
236             (fun i : Fin 1 ↦
237               if i = ⟨0, Nat.zero_lt_succ 0⟩ then 1 else 0) := by
238           funext i
239           fin_cases i
240           simp
241         rw [hfun]
242         exact haxis
243   | succ n =>
244       obtain ⟨K, fieldK, algebraK, y, xi, hgeneric, hresidue⟩ :=
245         hproduction k (n + 1) N d (Nat.zero_lt_succ n)
246           hN hd hdisjoint hnonempty
247       letI : Field K := fieldK
248       letI : Algebra k K := algebraK
249       obtain ⟨P, hP, hvanishes⟩ :=
250         hcontrol k K (n + 1) N d hN hd
251       exact false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
252         (k := k) (K := K)
253         (reducedOrderBaseIdeal k
254           (canonicalRightIdeal (presentedCoordinate k (n + 1)) d N))
255         P
256         (fun i : Fin ((n + 1) + 1) ↦
257           if i = ⟨0, Nat.zero_lt_succ (n + 1)⟩ then 1 else 0)
258         y xi hgeneric hresidue hvanishes
259         (canonical_fibrePolynomial_eval_extensionAxis
260           (K := K) hd P hP)
