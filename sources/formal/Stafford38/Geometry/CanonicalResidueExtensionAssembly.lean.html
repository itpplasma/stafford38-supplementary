8 /-!
9 # Canonical geometry over the boundary residue field
10 
11 The residue field `K` of a projective boundary divisor need not equal the
12 ground field `k`.  The completed arc therefore lives in `K[[t]]`, while the
13 canonical Weyl operator, its principal symbol, and the contracted base ideal
14 remain defined over `k`.
15 
16 This file repairs the terminal rank split at that interface.  In ambient rank
17 one the existing direct argument stays over `k`.  In higher rank the producer
18 is allowed to choose an arbitrary extension field `K/k` and returns the
19 Laurent conormal point actually consumed by the symbol contradiction.  It may
20 obtain that point directly from a constant-coordinate component or from a
21 completed boundary chart.  Symbol vanishing is requested over the same chosen
22 extension, and the contradiction is completed there.  No map `K → k`
23 occurs.
24 
25 The results are conditional assemblers.  They do not produce the
26 higher-dimensional conormal axis, prove post-extension Poisson/Gabber
27 integrability, prove noncharacteristic restriction, or prove the canonical
28 coordinate-preimage hypothesis.
29 -/

62 /-- Symbol control after extension to the Laurent field of an arbitrary
63 residue extension `K/k`.
64 
65 This is the exact geometric-support input used below.  It is deliberately not
66 called a theorem of Gabber: a caller must still prove it for the concrete
67 canonical quotient and the particular extension selected by boundary
68 geometry. -/
69 def CanonicalResidueExtensionSymbolControl : Prop :=
70   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
71     (K : Type u) [Field K] [Algebra k K]
72     (n N : ℕ) (d : PresentedWeyl k (n + 1)),
73     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
74       ∃ P : MvPolynomial (Fin (n + 1)) k,
75         fibreLift P = presentedPrincipalComponent k orderWeight N d ∧
76         ∀ q ∈ groundEquationConormalLocus (k := k) (K := K)
77             (reducedOrderBaseIdeal k
78               (canonicalRightIdeal (presentedCoordinate k n) d N)),
79           MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
80             (fibreLift P) = 0
81 
82 /-- Higher-dimensional production of the exact conormal-axis endpoint used by
83 the symbol contradiction, over its natural existential residue extension.
84 
85 The strict inequality excludes ambient rank one, which is handled directly.
86 In higher rank this interface deliberately does not prescribe how the point is
87 produced: the constant-coordinate branch may take `K = k`, while the
88 nonconstant branch may extract it from a completed boundary chart over a
89 genuine residue extension. -/
90 def HigherDimensionalCanonicalResidueExtensionConormalAxisProduction : Prop :=
91   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
92     (n N : ℕ) (d : PresentedWeyl k (n + 1)),
93     0 < n →
94     0 < N →
95     IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
96     Disjoint
97       (orderCharacteristicSupport k
98         (canonicalRightIdeal (presentedCoordinate k n) d N))
99       (PrimeSpectrum.zeroLocus
100         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
101           Set (SymbolRing k (n + 1)))) →
102     (orderCharacteristicSupport k
103       (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty →
104     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
105       (y : Fin (n + 1) → LaurentSeries K)
106       (xi : Fin (n + 1) → PowerSeries K),
107       Sum.elim y
108           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
109         groundEquationConormalLocus (k := k) (K := K)
110           (reducedOrderBaseIdeal k
111             (canonicalRightIdeal (presentedCoordinate k n) d N)) ∧
112       residueColumn xi =
113         (fun i : Fin (n + 1) ↦
114           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)

116 /-! ## One chart gives the terminal contradiction over `K` -/
117 
118 /-- The canonical principal symbol evaluates to one on the pure first-fibre
119 axis before and after extension from `k` to `K`. -/
120 theorem canonical_fibrePolynomial_eval_extensionAxis
121     {k K : Type u} [Field k] [Field K] [Algebra k K]
122     {n N : ℕ} {d : PresentedWeyl k (n + 1)}
123     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
124     (P : MvPolynomial (Fin (n + 1)) k)
125     (hP : fibreLift P = presentedPrincipalComponent k orderWeight N d) :
126     MvPolynomial.eval₂ (algebraMap k K)
127         (fun i : Fin (n + 1) ↦
128           if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) P = 1 := by
129   have hcanonical :=
130     canonical_orderPrincipalComponent_eval_pureMomentumAxis n N hd
131   have hsplit :
132       Sum.elim (fun _ : Fin (n + 1) ↦ (0 : k))
133           (fun i : Fin (n + 1) ↦
134             if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) =
135         axisPoint k (.inr (0 : Fin (n + 1))) := by
136     funext i
137     rcases i with i | i
138     · simp [axisPoint]
139     · simp [axisPoint] <;> rfl
140   rw [← hP] at hcanonical
141   have heval := eval₂_fibreLift (K := k) P
142     (fun _ : Fin (n + 1) ↦ (0 : k))
143     (fun i : Fin (n + 1) ↦
144       if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)
145   rw [hsplit] at heval
146   have hground : MvPolynomial.eval
147       (fun i : Fin (n + 1) ↦
148         if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0) P = 1 := by
149     simpa only [← MvPolynomial.aeval_def, MvPolynomial.aeval_eq_eval] using
150       heval.symm.trans hcanonical
151   rw [eval₂_extensionAxis_eq_algebraMap_eval_axis
152     (K := K) (Nat.zero_lt_succ n) P, hground]
153   exact map_one (algebraMap k K)

155 /-- A residue-extension completed chart, together with canonical symbol
156 vanishing over the same Laurent field, is contradictory.  The conclusion is
157 proved in `K`; no residue-field descent or retraction is used. -/
158 theorem false_of_canonical_completedProjectiveBoundaryChartOver
159     {k K : Type u} [Field k] [CharZero k]
160     [Field K] [Algebra k K]
161     (n N : ℕ) (d : PresentedWeyl k (n + 1))
162     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
163     (P : MvPolynomial (Fin (n + 1)) k)
164     (hP : fibreLift P = presentedPrincipalComponent k orderWeight N d)
165     (W : CompletedProjectiveBoundaryChartOver
166       (k := k) (K := K) (n + 1) (Nat.zero_lt_succ n)
167       (reducedOrderBaseIdeal k
168         (canonicalRightIdeal (presentedCoordinate k n) d N)))
169     (hvanishes :
170       ∀ q ∈ groundEquationConormalLocus (k := k) (K := K)
171           (reducedOrderBaseIdeal k
172             (canonicalRightIdeal (presentedCoordinate k n) d N)),
173         MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
174           (fibreLift P) = 0) : False := by
175   letI : CharZero K :=
176     charZero_of_injective_algebraMap (algebraMap k K).injective
177   obtain ⟨y, xi, hgeneric, hresidue⟩ :=
178     exists_conormalAxis_of_completedProjectiveBoundaryChartOver
179       (k := k) (K := K) (Nat.zero_lt_succ n)
180       (reducedOrderBaseIdeal k
181         (canonicalRightIdeal (presentedCoordinate k n) d N)) W
182   exact false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
183     (k := k) (K := K)
184     (reducedOrderBaseIdeal k
185       (canonicalRightIdeal (presentedCoordinate k n) d N))
186     P
187     (fun i : Fin (n + 1) ↦
188       if i = ⟨0, Nat.zero_lt_succ n⟩ then 1 else 0)
189     y xi hgeneric hresidue hvanishes
190     (canonical_fibrePolynomial_eval_extensionAxis
191       (K := K) hd P hP)

193 /-! ## Rank-split terminal assembly -/

195 /-- Conditional support vanishing with the corrected coefficient fields.
196 
197 The rank-one branch uses the existing direct ambient conormal theorem over
198 `k`.  In the successor branch, conormal-axis production chooses `K/k`; symbol
199 control and that point are then consumed over precisely that `K`.  The theorem
200 never specializes a `K`-valued point to `k[[t]]`. -/
201 theorem algebraicallyClosedCanonicalSupportVanishing_of_residueExtension_rankSplit
202     (hunit : CanonicalStrictUnitCoordinatePreimage.{u})
203     (hcontrol : CanonicalResidueExtensionSymbolControl.{u})
204     (hproduction :
205       HigherDimensionalCanonicalResidueExtensionConormalAxisProduction.{u}) :
206     AlgebraicallyClosedCanonicalSupportVanishing.{u} := by
207   intro k _ _ _ n N d hN hd
208   let I := canonicalRightIdeal (presentedCoordinate k n) d N
209   have hdisjoint : Disjoint
210       (orderCharacteristicSupport k I)
211       (PrimeSpectrum.zeroLocus
212         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
213           Set (SymbolRing k (n + 1)))) :=
214     canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
215       k n N (hunit k n N d hN hd)
216   by_contra hsupp
217   have hnonempty : (orderCharacteristicSupport k I).Nonempty :=
218     Set.nonempty_iff_ne_empty.mpr hsupp
219   cases n with
220   | zero =>
221       obtain ⟨P, hP, hvanishes⟩ := hcontrol k k 0 N d hN hd
222       obtain ⟨y, xi, hgeneric, hresidue⟩ :=
223         exists_canonical_rankOne_laurentConormalAxis I hdisjoint hnonempty
224       apply false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
225         (k := k) (K := k) (reducedOrderBaseIdeal k I) P
226           (fun _ : Fin 1 ↦ (1 : k)) y xi
227       · simpa [groundEquationConormalLocus, groundPolynomialMap,
228           groundLaurentMap, scalarPolynomialMap] using hgeneric
229       · exact hresidue
230       · exact hvanishes
231       · have haxis := canonical_fibrePolynomial_eval_extensionAxis
232           (K := k) hd P hP
233         have hfun : (fun _ : Fin 1 ↦ (1 : k)) =
234             (fun i : Fin 1 ↦
235               if i = ⟨0, Nat.zero_lt_succ 0⟩ then 1 else 0) := by
236           funext i
237           fin_cases i
238           simp
239         rw [hfun]
240         exact haxis
241   | succ n =>
242       obtain ⟨K, fieldK, algebraK, y, xi, hgeneric, hresidue⟩ :=
243         hproduction k (n + 1) N d (Nat.zero_lt_succ n)
244           hN hd hdisjoint hnonempty
245       letI : Field K := fieldK
246       letI : Algebra k K := algebraK
247       obtain ⟨P, hP, hvanishes⟩ :=
248         hcontrol k K (n + 1) N d hN hd
249       exact false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
250         (k := k) (K := K)
251         (reducedOrderBaseIdeal k
252           (canonicalRightIdeal (presentedCoordinate k (n + 1)) d N))
253         P
254         (fun i : Fin ((n + 1) + 1) ↦
255           if i = ⟨0, Nat.zero_lt_succ (n + 1)⟩ then 1 else 0)
256         y xi hgeneric hresidue hvanishes
257         (canonical_fibrePolynomial_eval_extensionAxis
258           (K := K) hd P hP)
