11 /-!
12 # Discrete boundary places for affine components
13 
14 An affine component function field is finitely generated over the ground
15 field.  Therefore every transcendental component coordinate admits the
16 discrete boundary refinement constructed by the relative divisorial tower.
17 -/

40 /-- The function field of a prime affine component is finitely generated as a
41 field extension of the ground field. -/
42 theorem componentFunctionField_fg
43     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
44     (⊤ : IntermediateField k
45       (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal))).FG := by
46   exact top_fg_of_finiteType_fractionRing k
47     (MvPolynomial (Fin m) k ⧸ P.asIdeal)
48     (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal))

50 /-- The affine component point in homogeneous coordinates. -/
51 abbrev componentProjectivePoint
52     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
53   Fin (m + 1) → FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal) :=
54   insertProjectiveChart (0 : Fin (m + 1)) (zerothChartEquiv (m := m))
55     (fun i ↦ componentCoordinate P i)

58     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
59     componentProjectivePoint P = Fin.cases 1 (fun i ↦ componentCoordinate P i) :=
60   insertProjectiveChart_zero _

74 /-- The complete affine coordinate family can be scaled into the retained
75 valuation ring with one projective coordinate equal to one. The selected
76 coordinate remains the retained parameter times the homogeneous zeroth
77 coordinate. -/
78 theorem exists_normalizedProjectivePoint_relativeRetainedBoundaryPlace
79     [CharZero k]
80     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
81     (hi : Transcendental k (componentCoordinate P i)) :
82     ∃ W : Data k (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal))
83         (componentCoordinate P i),
84       letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
85           (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)) :=
86         W.ambientAlgebra
87       ∃ (chart : Fin (m + 1))
88         (q : Fin (m + 1) → W.place.valuation.toSubring)
89         (scale : FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)),
90         scale ≠ 0 ∧ q chart = 1 ∧ q 0 ≠ 0 ∧
91         (∀ a, (q a : FractionRing
92             (MvPolynomial (Fin m) k ⧸ P.asIdeal)) =
93           scale * componentProjectivePoint P a) ∧
94         q (Fin.succ i) = q 0 * W.place.parameter := by
95   obtain ⟨W⟩ :=
96     exists_relativeRetainedBoundaryPlace_componentCoordinate P i hi
97   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
98       (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)) :=
99     W.ambientAlgebra
100   obtain ⟨chart, q, scale, hscale, hchart, hq⟩ :=
101     exists_normalized_projective_lift W.place.valuation
102       (componentProjectivePoint P) ⟨0, by simp [componentProjectivePoint]⟩
103   have hqzero : q 0 ≠ 0 := by
104     intro hzero
105     apply hscale
106     have h := hq 0
107     rw [hzero] at h
108     simpa [componentProjectivePoint] using h.symm
109   refine ⟨W, chart, q, scale, hscale, hchart, hqzero, hq, ?_⟩
110   apply Subtype.ext
111   change (q (Fin.succ i) : FractionRing
112       (MvPolynomial (Fin m) k ⧸ P.asIdeal)) =
113     (q 0 : FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)) *
114       (W.place.parameter : FractionRing
115         (MvPolynomial (Fin m) k ⧸ P.asIdeal))
116   rw [hq, hq, W.parameter_eq_coordinate]
117   simp [componentProjectivePoint]

119 /-- A transcendental coordinate on a prime affine component has a genuine
120 discrete valuation place centred at coordinate zero.  It is the forgetful
121 image of the retained relative place above. -/
122 theorem exists_discreteBoundaryRefinement_componentCoordinate
123     [CharZero k]
124     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
125     (hi : Transcendental k (componentCoordinate P i)) :
126     Nonempty (DiscreteBoundaryRefinement k (componentCoordinate P i)) := by
127   obtain ⟨W⟩ :=
128     exists_relativeRetainedBoundaryPlace_componentCoordinate P i hi
129   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
130       (FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)) :=
131     W.ambientAlgebra
132   exact ⟨W.toDiscreteBoundaryRefinement⟩
