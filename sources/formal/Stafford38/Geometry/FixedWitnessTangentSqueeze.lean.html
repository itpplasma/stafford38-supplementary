6 /-!
7 # A finite family determines the tangent space at full rank
8 
9 If a finite linearly independent family consists of Zariski tangent vectors
10 and the tangent space has dimension at most the size of the family, then the
11 tangent space is exactly their span.  This is only the finite-dimensional
12 linear-algebra consumer; it does not construct tangent vectors or establish a
13 smoothness or dimension bound.
14 -/
15 
16 namespace Stafford38.Geometry.FixedWitnessTangentSqueeze
17 
18 open Stafford38.Geometry.AffineConormalSpan
19 
20 noncomputable section
21 
22 variable {K : Type*} [Field K] {n : ℕ}
23 
24 /-- A fixed independent tangent family spans the whole Zariski tangent space
25 once the tangent dimension is bounded by the cardinality of that family. -/
26 theorem fixedWitnessTangentSqueeze
27     {κ : Type*} [Fintype κ]
28     (I : Ideal (MvPolynomial (Fin n) K))
29     (y : Fin n → K)
30     (b : κ → AffineTangentVector K n)
31     (hb : ∀ j, b j ∈ zariskiTangentSpace y I)
32     (hindependent : LinearIndependent K b)
33     (hfinrank : Module.finrank K (zariskiTangentSpace y I) ≤ Fintype.card κ) :
34     zariskiTangentSpace y I = Submodule.span K (Set.range b) := by
35   classical
36   have hspan_le :
37       Submodule.span K (Set.range b) ≤ zariskiTangentSpace y I := by
38     refine Submodule.span_le.mpr ?_
39     rintro _ ⟨j, rfl⟩
40     exact hb j
41   have hrank :
42       Module.finrank K (zariskiTangentSpace y I) ≤
43         Module.finrank K (Submodule.span K (Set.range b)) := by
44     rw [finrank_span_eq_card hindependent]
45     exact hfinrank
46   exact (Submodule.eq_of_le_of_finrank_le hspan_le hrank).symm
47 
48 /-- The reverse inclusion needed by the conormal consumer. -/
49 theorem fixedWitnessTangent_le_span
50     {κ : Type*} [Fintype κ]
51     (I : Ideal (MvPolynomial (Fin n) K))
52     (y : Fin n → K)
53     (b : κ → AffineTangentVector K n)
54     (hb : ∀ j, b j ∈ zariskiTangentSpace y I)
55     (hindependent : LinearIndependent K b)
56     (hfinrank : Module.finrank K (zariskiTangentSpace y I) ≤ Fintype.card κ) :
57     zariskiTangentSpace y I ≤ Submodule.span K (Set.range b) := by
58   rw [fixedWitnessTangentSqueeze I y b hb hindependent hfinrank]
59 
60 #print axioms fixedWitnessTangentSqueeze
61 #print axioms fixedWitnessTangent_le_span
62 
63 end

65 end Stafford38.Geometry.FixedWitnessTangentSqueeze
