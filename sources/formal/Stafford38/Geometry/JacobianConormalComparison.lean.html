9 # Comparing Jacobian and equation-defined affine conormals
10 
11 An externally supplied smooth locus can be compared with the repository's
12 equation-defined conormal locus once its tangent space is identified with the
13 common kernel of a family of Jacobian rows.  This file isolates that exact
14 bridge.
15 
16 The equations in the Jacobian family are required to belong to the ambient
17 ideal.  Under that hypothesis, their conormal space is contained in the
18 conormal space defined using all equations.  Consequently, if the supplied
19 tangent space agrees pointwise with the Jacobian kernel, its conormal locus
20 embeds into `equationConormalLocus`, and its field-valued algebraic closure
21 hull embeds into `equationConormalClosure`.
22 
23 No smoothness criterion, scheme tangent-space construction, geometric-point
24 comparison, algebraic closedness, or base change is asserted here.  Those are
25 inputs to the comparison theorem, not consequences of it.
26 -/

40 variable {k : Type*} [Field k] {n : ℕ}

42 /-- The span of the Jacobian rows of a chosen family of equations at `y`. -/
43 def jacobianCovectorSpan {ι : Type*} (y : Fin n → k)
44     (equations : ι → MvPolynomial (Fin n) k) :
45     Submodule k (Module.Dual k (AffineTangentVector k n)) :=
46   Submodule.span k (Set.range fun i ↦ differentialCovector y (equations i))
47 
48 /-- The tangent space cut out by the chosen Jacobian rows. -/
49 def jacobianTangentSpace {ι : Type*} (y : Fin n → k)
50     (equations : ι → MvPolynomial (Fin n) k) :
51     Submodule k (AffineTangentVector k n) :=
52   (jacobianCovectorSpan y equations).dualCoannihilator
53 
54 /-- The conormal space obtained as the annihilator of the Jacobian kernel. -/
55 def jacobianConormalSpace {ι : Type*} (y : Fin n → k)
56     (equations : ι → MvPolynomial (Fin n) k) :
57     Submodule k (Module.Dual k (AffineTangentVector k n)) :=
58   (jacobianTangentSpace y equations).dualAnnihilator
59 
60 /-- In finite-dimensional affine space, taking the annihilator of the
61 Jacobian kernel recovers the span of the Jacobian rows. -/
62 theorem jacobianConormalSpace_eq_jacobianCovectorSpan {ι : Type*}
63     (y : Fin n → k) (equations : ι → MvPolynomial (Fin n) k) :
64     jacobianConormalSpace y equations = jacobianCovectorSpan y equations := by
65   classical
66   exact Subspace.dualCoannihilator_dualAnnihilator_eq
67 
68 /-- Jacobian rows coming from equations in `I` span a subspace of the
69 covectors generated by all equations in `I`. -/
70 theorem jacobianCovectorSpan_le_equationCovectorSpan {ι : Type*}
71     (y : Fin n → k) (I : Ideal (MvPolynomial (Fin n) k))
72     (equations : ι → MvPolynomial (Fin n) k)
73     (hequations : ∀ i, equations i ∈ I) :
74     jacobianCovectorSpan y equations ≤ equationCovectorSpan y I := by
75   apply Submodule.span_le.mpr
76   rintro _ ⟨i, rfl⟩
77   apply Submodule.subset_span
78   exact ⟨⟨equations i, hequations i⟩, rfl⟩
79 
80 /-- The conormal computed from any chosen equations in `I` embeds in the
81 equation-defined affine conormal computed from the whole ideal. -/
82 theorem jacobianConormalSpace_le_affineConormalSpace {ι : Type*}
83     (y : Fin n → k) (I : Ideal (MvPolynomial (Fin n) k))
84     (equations : ι → MvPolynomial (Fin n) k)
85     (hequations : ∀ i, equations i ∈ I) :
86     jacobianConormalSpace y equations ≤ affineConormalSpace y I := by
87   rw [jacobianConormalSpace_eq_jacobianCovectorSpan,
88     affineConormalSpace_eq_equationCovectorSpan]
89   exact jacobianCovectorSpan_le_equationCovectorSpan y I equations hequations
90 
91 /-- The conormal locus of a supplied set of base points and supplied tangent
92 spaces.  For a smooth locally closed locus, callers instantiate `Y` with its
93 field-valued points and `T` with its scheme-theoretic tangent spaces. -/
94 def tangentConormalLocus
95     (Y : Set (Fin n → k))
96     (T : (Fin n → k) → Submodule k (AffineTangentVector k n)) :
97     Set (PhaseVar n → k) :=
98   {q | (fun i ↦ q (Sum.inl i)) ∈ Y ∧
99     coordinateCovector (fun i ↦ q (Sum.inr i)) ∈
100       (T (fun i ↦ q (Sum.inl i))).dualAnnihilator}
101 
102 /-- Exact pointwise smooth-conormal comparison.
103 
104 The caller supplies two geometric facts: points of `Y` satisfy `I`, and the
105 supplied tangent space on `Y` is the kernel of the displayed Jacobian rows.
106 The latter is the explicit Jacobian/tangent hypothesis normally furnished by
107 smoothness. -/
108 theorem tangentConormalLocus_subset_equationConormalLocus_of_eq_jacobian
109     {ι : Type*}
110     (I : Ideal (MvPolynomial (Fin n) k))
111     (Y : Set (Fin n → k))
112     (T : (Fin n → k) → Submodule k (AffineTangentVector k n))
113     (equations : ι → MvPolynomial (Fin n) k)
114     (hequations : ∀ i, equations i ∈ I)
115     (hY : ∀ y ∈ Y, ∀ f ∈ I, MvPolynomial.eval y f = 0)
116     (hT : ∀ y ∈ Y, T y = jacobianTangentSpace y equations) :
117     tangentConormalLocus Y T ⊆ equationConormalLocus I := by
118   intro q hq
119   let y : Fin n → k := fun i ↦ q (Sum.inl i)
120   let ξ : Fin n → k := fun i ↦ q (Sum.inr i)
121   refine ⟨hY y hq.1, ?_⟩
122   apply jacobianConormalSpace_le_affineConormalSpace y I equations hequations
123   change coordinateCovector ξ ∈ jacobianConormalSpace y equations
124   have hq' := hq.2
125   change coordinateCovector ξ ∈ (T y).dualAnnihilator at hq'
126   rw [hT y hq.1] at hq'
127   exact hq'
128 
129 /-- Closure-level form of the comparison: the field-valued algebraic closure
130 hull of the supplied tangent conormal locus lies in the equation-defined
131 conormal closure. -/
132 theorem tangentConormalClosure_subset_equationConormalClosure_of_eq_jacobian
133     {ι : Type*}
134     (I : Ideal (MvPolynomial (Fin n) k))
135     (Y : Set (Fin n → k))
136     (T : (Fin n → k) → Submodule k (AffineTangentVector k n))
137     (equations : ι → MvPolynomial (Fin n) k)
138     (hequations : ∀ i, equations i ∈ I)
139     (hY : ∀ y ∈ Y, ∀ f ∈ I, MvPolynomial.eval y f = 0)
140     (hT : ∀ y ∈ Y, T y = jacobianTangentSpace y equations) :
141     MvPolynomial.zeroLocus k
142         (MvPolynomial.vanishingIdeal k (tangentConormalLocus Y T)) ⊆
143       equationConormalClosure I := by
144   rw [equationConormalClosure]
145   exact MvPolynomial.zeroLocus_anti_mono
146     (MvPolynomial.vanishingIdeal_anti_mono
147       (tangentConormalLocus_subset_equationConormalLocus_of_eq_jacobian
148         I Y T equations hequations hY hT))
149 
150 /-- Direct characteristic-support consumer.  Once the tangent data on `Y`
151 are identified with Jacobian kernels for equations in the contracted base
152 ideal, the closure of their conormal locus lies in reduced order support under
153 the existing base-relative Gabber fragment. -/
154 theorem tangentConormalClosure_subset_reducedOrderSupport_of_eq_jacobian
155     [CharZero k]
156     {ι : Type*}
157     (W : RightIdeal (WeylAlg k n))
158     (hJ : IsBaseRelativePoisson (reducedOrderSupportIdeal k W))
159     (Y : Set (Fin n → k))
160     (T : (Fin n → k) → Submodule k (AffineTangentVector k n))
161     (equations : ι → MvPolynomial (Fin n) k)
162     (hequations : ∀ i, equations i ∈ reducedOrderBaseIdeal k W)
163     (hY : ∀ y ∈ Y, ∀ f ∈ reducedOrderBaseIdeal k W,
164       MvPolynomial.eval y f = 0)
165     (hT : ∀ y ∈ Y, T y = jacobianTangentSpace y equations) :
166     MvPolynomial.zeroLocus k
167         (MvPolynomial.vanishingIdeal k (tangentConormalLocus Y T)) ⊆
168       MvPolynomial.zeroLocus k (reducedOrderSupportIdeal k W) := by
169   exact (tangentConormalClosure_subset_equationConormalClosure_of_eq_jacobian
170     (reducedOrderBaseIdeal k W) Y T equations hequations hY hT).trans
171       (equationConormalClosure_subset_reducedOrderSupport W hJ)
