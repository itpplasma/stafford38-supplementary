8 /-!
9 # Finite-gradient certificates over a boundary residue field
10 
11 A boundary divisor can have residue field `K` strictly larger than the ground
12 field `k`.  The completed arc and its conormal covector then naturally have
13 coefficients in `PowerSeries K`, while the affine ideal and the fibre symbol
14 remain defined over `k`.
15 
16 This file gives the finite-gradient certificate in that coefficient tower and
17 connects it directly to the residue-extension fibre-symbol contradiction.  No
18 map `K → k`, algebraic-closedness of `K`, normalization, boundary divisor, or
19 global certificate construction is assumed.
20 -/

41 /-- A residue-field finite-gradient certificate gives exactly the
42 `K((t))`-valued equation-conormal point needed by residue specialization. -/
43 theorem exists_groundConormalAxis_of_finiteGradientBoundaryCertificateOver
44     {m : ℕ} (hm : 0 < m)
45     (I : Ideal (MvPolynomial (Fin m) k))
46     (W : FiniteGradientBoundaryCertificateOver
47       (k := k) (K := K) m hm I) :
48     ∃ (y : Fin m → LaurentSeries K)
49       (xi : Fin m → PowerSeries K),
50       Sum.elim y
51           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
52         groundEquationConormalLocus (k := k) (K := K) I ∧
53       residueColumn xi =
54         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
55   let Iext := I.map (groundPolynomialMap (k := k) (K := K) (Fin m))
56   let y := dehomogenizedPoint (laurentColumn W.q)
57   let xi : Fin m → PowerSeries K := fun i ↦ W.ell i.succ
58   have hconormal :
59       coordinateCovector (fun i ↦ laurentColumn W.ell i.succ) ∈
60         affineConormalSpace y Iext := by
61     exact coordinateCovector_mem_affineConormalSpace_of_finite_gradient
62       Iext y (fun i ↦ laurentColumn W.ell i.succ)
63         W.equations W.coefficients W.gradient_identity
64   refine ⟨y, xi, ?_, W.residue_axis⟩
65   refine ⟨?_, ?_⟩
66   · simpa [Iext, y, groundEquationConormalLocus] using W.base_vanish
67   · simpa [xi, laurentColumn, y, Iext] using hconormal

69 /-- Evaluation on the coordinate axis commutes with extension from `k` to
70 `K`.  This is the only scalar-extension fact needed for the terminal value
71 `P(axis)=1`. -/
72 theorem eval₂_extensionAxis_eq_algebraMap_eval_axis
73     {m : ℕ} (hm : 0 < m) (P : MvPolynomial (Fin m) k) :
74     MvPolynomial.eval₂ (algebraMap k K)
75         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) P =
76       algebraMap k K
77         (MvPolynomial.eval
78           (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) P) := by
79   have hpoint :
80       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then (1 : K) else 0) =
81         fun i ↦ algebraMap k K
82           (if i = ⟨0, hm⟩ then (1 : k) else 0) := by
83     funext i
84     split <;> simp_all
85   rw [hpoint]
86   exact (MvPolynomial.map_eval₂Hom
87     (RingHom.id k)
88     (fun i : Fin m ↦ if i = ⟨0, hm⟩ then (1 : k) else 0)
89     (algebraMap k K) P).symm

91 /-- Direct terminal contradiction from a residue-field finite-gradient
92 certificate.  The symbol and its axis value stay over `k`; the completed arc
93 and all gradient coefficients may live over an arbitrary extension `K/k`. -/
94 theorem false_of_finiteGradientBoundaryCertificateOver
95     {m : ℕ} (hm : 0 < m)
96     (I : Ideal (MvPolynomial (Fin m) k))
97     (P : MvPolynomial (Fin m) k)
98     (W : FiniteGradientBoundaryCertificateOver
99       (k := k) (K := K) m hm I)
100     (hvanishes :
101       ∀ q ∈ groundEquationConormalLocus (k := k) (K := K) I,
102         MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
103           (fibreLift P) = 0)
104     (haxis : MvPolynomial.eval
105       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) P = 1) : False := by
106   obtain ⟨y, xi, hgeneric, hresidue⟩ :=
107     exists_groundConormalAxis_of_finiteGradientBoundaryCertificateOver
108       (k := k) (K := K) hm I W
109   apply false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
110     (k := k) (K := K) I P
111       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0)
112       y xi hgeneric hresidue hvanishes
113   rw [eval₂_extensionAxis_eq_algebraMap_eval_axis (K := K) hm P, haxis]
114   exact map_one (algebraMap k K)
