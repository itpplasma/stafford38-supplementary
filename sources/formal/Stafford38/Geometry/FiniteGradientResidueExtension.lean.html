10 /-!
11 # Finite-gradient certificates over a boundary residue field
12 
13 A boundary divisor can have residue field `K` strictly larger than the ground
14 field `k`.  The completed arc and its conormal covector then naturally have
15 coefficients in `PowerSeries K`, while the affine ideal and the fibre symbol
16 remain defined over `k`.
17 
18 This file gives the finite-gradient certificate in that coefficient tower and
19 connects it directly to the residue-extension fibre-symbol contradiction.  No
20 map `K → k`, algebraic-closedness of `K`, normalization, boundary divisor, or
21 global certificate construction is assumed.
22 -/

43 /-- A residue-field finite-gradient certificate gives exactly the
44 `K((t))`-valued equation-conormal point needed by residue specialization. -/
45 theorem exists_groundConormalAxis_of_finiteGradientBoundaryCertificateOver
46     {m : ℕ} (hm : 0 < m)
47     (I : Ideal (MvPolynomial (Fin m) k))
48     (W : FiniteGradientBoundaryCertificateOver
49       (k := k) (K := K) m hm I) :
50     ∃ (y : Fin m → LaurentSeries K)
51       (xi : Fin m → PowerSeries K),
52       Sum.elim y
53           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
54         groundEquationConormalLocus (k := k) (K := K) I ∧
55       residueColumn xi =
56         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
57   let Iext := I.map (scalarPolynomialMap (k := k) (K := LaurentSeries K) (Fin m))
58   let y := dehomogenizedPoint (laurentColumn W.q)
59   let xi : Fin m → PowerSeries K := fun i ↦ W.ell i.succ
60   have hconormal :
61       coordinateCovector (fun i ↦ laurentColumn W.ell i.succ) ∈
62         affineConormalSpace y Iext := by
63     exact coordinateCovector_mem_affineConormalSpace_of_finite_gradient
64       Iext y (fun i ↦ laurentColumn W.ell i.succ)
65         W.equations W.coefficients W.gradient_identity
66   refine ⟨y, xi, ?_, W.residue_axis⟩
67   refine ⟨?_, ?_⟩
68   · simpa [Iext, y, groundEquationConormalLocus] using W.base_vanish
69   · simpa [xi, laurentColumn, y, Iext] using hconormal

71 /-- Evaluation on the coordinate axis commutes with extension from `k` to
72 `K`.  This is the only scalar-extension fact needed for the terminal value
73 `P(axis)=1`. -/
74 theorem eval₂_extensionAxis_eq_algebraMap_eval_axis
75     {m : ℕ} (hm : 0 < m) (P : MvPolynomial (Fin m) k) :
76     MvPolynomial.eval₂ (algebraMap k K)
77         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) P =
78       algebraMap k K
79         (MvPolynomial.eval
80           (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) P) := by
81   have hpoint :
82       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then (1 : K) else 0) =
83         fun i ↦ algebraMap k K
84           (if i = ⟨0, hm⟩ then (1 : k) else 0) := by
85     funext i
86     split <;> simp_all
87   rw [hpoint]
88   exact (MvPolynomial.map_eval₂Hom
89     (RingHom.id k)
90     (fun i : Fin m ↦ if i = ⟨0, hm⟩ then (1 : k) else 0)
91     (algebraMap k K) P).symm

93 /-- Direct terminal contradiction from a residue-field finite-gradient
94 certificate.  The symbol and its axis value stay over `k`; the completed arc
95 and all gradient coefficients may live over an arbitrary extension `K/k`. -/
96 theorem false_of_finiteGradientBoundaryCertificateOver
97     {m : ℕ} (hm : 0 < m)
98     (I : Ideal (MvPolynomial (Fin m) k))
99     (P : MvPolynomial (Fin m) k)
100     (W : FiniteGradientBoundaryCertificateOver
101       (k := k) (K := K) m hm I)
102     (hvanishes :
103       ∀ q ∈ groundEquationConormalLocus (k := k) (K := K) I,
104         MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q
105           (fibreLift P) = 0)
106     (haxis : MvPolynomial.eval
107       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) P = 1) : False := by
108   obtain ⟨y, xi, hgeneric, hresidue⟩ :=
109     exists_groundConormalAxis_of_finiteGradientBoundaryCertificateOver
110       (k := k) (K := K) hm I W
111   apply false_of_ground_fibreOnly_symbol_one_on_residue_and_vanishing
112     (k := k) (K := K) I P
113       (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0)
114       y xi hgeneric hresidue hvanishes
115   rw [eval₂_extensionAxis_eq_algebraMap_eval_axis (K := K) hm P, haxis]
116   exact map_one (algebraMap k K)
