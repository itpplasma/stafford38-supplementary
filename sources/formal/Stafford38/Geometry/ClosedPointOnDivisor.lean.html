13 /-- Every principal open meeting a prime closed set in a Jacobson spectrum
14 contains a closed point of the ambient spectrum. -/
15 theorem exists_maximal_over_prime_avoiding
16     {R : Type*} [CommRing R] [IsJacobsonRing R]
17     (p : Ideal R) (hp : p.IsPrime) (f : R) (hf : f ∉ p) :
18     ∃ M : Ideal R, M.IsMaximal ∧ p ≤ M ∧ f ∉ M := by
19   classical
20   have hJ : p.jacobson = p := IsJacobsonRing.out inferInstance hp.isRadical
21   by_contra h
22   apply hf
23   rw [← hJ, Ideal.jacobson, Ideal.mem_sInf]
24   intro M hM
25   by_contra hnot
26   exact h ⟨M, hM.2, hM.1, hnot⟩

28 /-- The residue field at a closed point of a finite-type algebra over an
29 algebraically closed field is the ground field as an algebra. -/
30 theorem quotient_algEquiv_ground_of_maximal
31     {k R : Type*} [Field k] [IsAlgClosed k]
32     [CommRing R] [Algebra k R] [Algebra.FiniteType k R]
33     (M : Ideal R) [M.IsMaximal] :
34     Nonempty ((R ⧸ M) ≃ₐ[k] k) := by
35   letI : Field (R ⧸ M) := Ideal.Quotient.field M
36   letI : Module.Finite k (R ⧸ M) :=
37     finite_of_finite_type_of_isJacobsonRing k (R ⧸ M)
38   exact ⟨(AlgEquiv.ofBijective (Algebra.ofId k (R ⧸ M))
39     IsAlgClosed.algebraMap_bijective_of_isIntegral).symm⟩

41 /-- A dense principal open on a prime closed subset of a finite-type affine model
42 contains a ground-field closed point. The smoothness and parameter conditions
43 must still be proved to hold on the chosen open. -/
44 theorem exists_ground_closed_point_over_prime_avoiding
45     {k R : Type*} [Field k] [IsAlgClosed k]
46     [CommRing R] [Algebra k R] [Algebra.FiniteType k R]
47     (p : Ideal R) (hp : p.IsPrime) (f : R) (hf : f ∉ p) :
48     ∃ (M : Ideal R) (hM : M.IsMaximal), p ≤ M ∧ f ∉ M ∧
49       (letI : M.IsMaximal := hM; Nonempty ((R ⧸ M) ≃ₐ[k] k)) := by
50   letI : IsJacobsonRing R := isJacobsonRing_of_finiteType (A := k) (B := R)
51   obtain ⟨M, hM, hpM, hfM⟩ := exists_maximal_over_prime_avoiding p hp f hf
52   letI : M.IsMaximal := hM
53   exact ⟨M, hM, hpM, hfM, quotient_algEquiv_ground_of_maximal M⟩
