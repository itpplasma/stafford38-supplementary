23 /-- Essential finite type over the polynomial coordinate ring makes the selected
24 prime localization Noetherian. -/
25 theorem localizationAtPrime_isNoetherianRing
26     [Algebra.EssFiniteType R B] (M : Ideal B) [M.IsPrime] :
27     IsNoetherianRing (Localization.AtPrime M) := by
28   letI : IsNoetherianRing B := Algebra.EssFiniteType.isNoetherianRing R B
29   exact IsLocalization.isNoetherianRing M.primeCompl _ (inferInstance : IsNoetherianRing B)

31 /-- Completion at the maximal ideal of this local Noetherian ring is separated,
32 so its canonical map is injective. -/
33 theorem localizationCompletion_of_injective
34     [Algebra.EssFiniteType R B] (M : Ideal B) [M.IsPrime] :
35     Function.Injective
36       (AdicCompletion.of (IsLocalRing.maximalIdeal (Localization.AtPrime M))
37         (Localization.AtPrime M)) := by
38   letI : IsNoetherianRing (Localization.AtPrime M) :=
39     localizationAtPrime_isNoetherianRing (k := k) (σ := σ) (B := B) M
40   exact AdicCompletion.of_injective _ _

42 /-- A nonzero element of the finite-type domain remains nonzero after localization,
43 completion, and transport to the prescribed formal-series model. The point may
44 lie on its zero set; localization injectivity and separatedness give the claim. -/
45 theorem prescribedSeries_image_ne_zero
46     (M : Ideal B) [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
47     [Algebra.EssFiniteType R B]
48     [Algebra.FormallyEtale R (Localization.AtPrime M)]
49     [IsDomain B] (b : B) (hb : b ≠ 0) :
50     (powerSeriesCompletionAtGroundPoint (σ := σ) M eM).symm
51       (AdicCompletion.of
52         (IsLocalRing.maximalIdeal (Localization.AtPrime M))
53         (Localization.AtPrime M)
54         (algebraMap B (Localization.AtPrime M) b)) ≠ 0 := by
55   let E := powerSeriesCompletionAtGroundPoint (σ := σ) M eM
56   letI : IsNoetherianRing (Localization.AtPrime M) :=
57     localizationAtPrime_isNoetherianRing (k := k) (σ := σ) (B := B) M
58   have hinj : Function.Injective (algebraMap B (Localization.AtPrime M)) :=
59     IsLocalization.injective (Localization.AtPrime M)
60       (Ideal.primeCompl_le_nonZeroDivisors M)
61   intro hseries
62   have hcompleted :
63       AdicCompletion.of (IsLocalRing.maximalIdeal (Localization.AtPrime M))
64           (Localization.AtPrime M) (algebraMap B (Localization.AtPrime M) b) = 0 := by
65     have h := congrArg E hseries
66     simpa [E] using h
67   have hlocal : algebraMap B (Localization.AtPrime M) b = 0 :=
68     (localizationCompletion_of_injective (k := k) (σ := σ) (B := B) M).eq_iff.mp
69       hcompleted
70   exact hb (hinj (by simpa using hlocal))
