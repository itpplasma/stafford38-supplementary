36 /-- The place at infinity of the paper's proof of Theorem 5.7: the valuation
37 ring and chart of Lemma 5.9, the discrete valuation ring `V = A_{𝔭_D}` of
38 Lemma 5.10, and a coefficient field `E` over which `κ` is finite (the input to
39 the coefficient field of the completion, Matsumura Thm. 28.3). -/
40 theorem divisorialVisibleFrameExistence : DivisorialVisibleFrameExistence.{u} := by
41   intro k K _ _ _ _ r y i hgen hx
42   -- Lemma 5.9.
43   obtain ⟨W, j, hkW, -, hxW, hj, hqW⟩ := Stafford38.Paper.Section5.lemma_5_9 y i hx
44   -- Lemma 5.10.
45   obtain ⟨L⟩ := Stafford38.Paper.Section5.lemma_5_10 y i hgen hx W j hkW hxW hj hqW
46   let A := L.A
47   let p := L.p
48   let V : ValuationSubring K := L.V
49   have hVdvr : IsDiscreteValuationRing V.toSubring := L.isDVR
50   letI : IsLocalRing V.toSubring := hVdvr.toIsLocalRing
51   have hAV : ∀ a : A, (a : K) ∈ V.toSubring := fun a => ofField_mem_algebraMap_v A p L.place a
52   have hpV : ∀ a : A, (⟨a, hAV a⟩ : V.toSubring) ∈ maximalIdeal V.toSubring ↔ a ∈ p :=
53     fun a => ofField_mem_maximalIdeal_iff_v A p L.place a
54   have hsurj : ∀ z : ResidueField V.toSubring, ∃ (a b : A), b ∉ p ∧
55       z = residue V.toSubring ⟨a, hAV a⟩ / residue V.toSubring ⟨b, hAV b⟩ :=
56     fun z => ofField_residue_surjective_v A p L.place z
57   have hy0 : y i ≠ 0 := fun h => hx (h ▸ isAlgebraic_zero)
58   have hxV : y i ∈ V.toSubring := hAV ⟨y i, L.x_mem⟩
59   have hxm : (⟨y i, hxV⟩ : V.toSubring) ∈ maximalIdeal V.toSubring :=
60     (hpV ⟨y i, L.x_mem⟩).2 L.x_mem_p
61   obtain ⟨E, hEV, hEfin⟩ := stage5_of_verified A p V hAV hpV hsurj
62   letI : Algebra E V.toSubring := (coeffHom E V hEV).toAlgebra
63   have htrans : Transcendental E (y i) :=
64     stage6_transcendental_of_mem_maximalIdeal E V hEV (y i) hy0 hxV hxm
65   have hk : ∀ c : k, algebraMap k K c ∈ V.toSubring := fun c => hAV (algebraMap k A c)
66   let cod : Algebra k V.toSubring := ((algebraMap k K).codRestrict _ hk).toAlgebra
67   let ground : Algebra k V.toSubring := (groundHom E V hEV).toAlgebra
68   have hcod_ground : cod = ground := by
69     apply Algebra.algebra_ext
70     intro c
71     apply Subtype.ext
72     rfl
73   letI : Algebra k V.toSubring := ground
74   letI : IsScalarTower k V.toSubring K :=
75     IsScalarTower.of_algebraMap_eq fun c => by
76       change algebraMap k K c = algebraMap k K c
77       rfl
78   have hkaehler : Module.Finite V.toSubring (Ω[V.toSubring⁄k]) := by
79     have h := L.kaehler_finite
80     change (letI : Algebra k V.toSubring := cod; Module.Finite V.toSubring (Ω[V.toSubring⁄k])) at h
81     rw [hcod_ground] at h
82     exact h
83   -- The normalized column `q_a = x̄_a / x̄_j` of Lemma 5.10.
84   let s : K := (Fin.cases 1 y j : K)⁻¹
85   have hjs : (Fin.cases 1 y j : K) * s = 1 := mul_inv_cancel₀ hj
86   let c : Fin (r + 1) → V.toSubring := fun a =>
87     ⟨(Fin.cases 1 y a : K) / Fin.cases 1 y j, hAV ⟨_, L.q_mem a⟩⟩
88   have hc : ∀ a, (c a : K) = (Fin.cases 1 y a : K) * s := fun a => div_eq_mul_inv _ _
89   have hcj : c j = 1 := Subtype.ext (div_self hj)
90   have halg : Algebra.IsAlgebraic
91       (IntermediateField.adjoin k (Set.range fun a => residue V.toSubring (c a)) :
92         IntermediateField k (ResidueField V.toSubring))
93       (ResidueField V.toSubring) := by
94     have h := L.residue_algebraic
95     change (letI : Algebra k V.toSubring := cod; Algebra.IsAlgebraic
96       (IntermediateField.adjoin k (Set.range fun a => residue V.toSubring (c a)) :
97         IntermediateField k (ResidueField V.toSubring))
98       (ResidueField V.toSubring)) at h
99     rw [hcod_ground] at h
100     exact h

102   intro s' q hq hq1 hq0

105     exfalso
106     have hsone : s = 1 := by simpa [hj0] using hjs
107     have hprod : ∀ a, q a = q 0 * c a := by
108       intro a

117       rw [hprod j', map_mul, hq0res, zero_mul]
