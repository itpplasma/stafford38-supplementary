37 /-- Every equation-defined tangent vector factors through an actual formally
38 étale polynomial chart.  Its coordinates are a linear combination of the
39 parameter-coordinate derivations of that chart.  The source chart algebra `C`
40 may be a further localization of a local étale chart, provided both formal
41 étale structures are retained. -/
42 theorem tangentVector_eq_sum_parameterDerivations
43     (q v : Fin n → L)
44     (hq : ∀ i, q i = algebraMap C L
45       (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
46         (Ideal.Quotient.mk I (MvPolynomial.X i))))
47     (hv : v ∈ Stafford38.Geometry.AffineConormalSpan.zariskiTangentSpace q
48       (I.map (MvPolynomial.map (algebraMap k L))))
49     : ∃ coeff : σ → L, ∀ i,
50       v i = ∑ j, coeff j *
51         Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
52           (k := k) (σ := σ) (B := C) (L := L) j
53           (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
54             (Ideal.Quotient.mk I (MvPolynomial.X i)) ) := by
55   have hi : ∀ i, algebraMap (MvPolynomial (Fin n) k) L
56       (MvPolynomial.X i) = q i := by
57     intro i
58     calc
59       algebraMap (MvPolynomial (Fin n) k) L (MvPolynomial.X i) =
60           algebraMap (MvPolynomial (Fin n) k ⧸ I) L
61             (algebraMap (MvPolynomial (Fin n) k)
62               (MvPolynomial (Fin n) k ⧸ I) (MvPolynomial.X i)) :=
63         IsScalarTower.algebraMap_apply (MvPolynomial (Fin n) k)
64           (MvPolynomial (Fin n) k ⧸ I) L (MvPolynomial.X i)
65       _ = algebraMap C L (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
66           (Ideal.Quotient.mk I (MvPolynomial.X i))) := by
67         rw [IsScalarTower.algebraMap_apply (MvPolynomial (Fin n) k ⧸ I) C L]
68         simp only [Ideal.Quotient.algebraMap_eq]
69       _ = q i := (hq i).symm
70   have halgHom : IsScalarTower.toAlgHom k (MvPolynomial (Fin n) k) L =
71       MvPolynomial.aeval q := by
72     apply MvPolynomial.algHom_ext
73     intro i
74     rw [IsScalarTower.coe_toAlgHom', MvPolynomial.aeval_X]
75     exact hi i
76   have halg : ∀ p : MvPolynomial (Fin n) k,
77       algebraMap (MvPolynomial (Fin n) k) L p = MvPolynomial.aeval q p := by
78     intro p
79     exact congrArg (fun f : MvPolynomial (Fin n) k →ₐ[k] L => f p) halgHom
80   let D : Derivation k (MvPolynomial (Fin n) k) L := MvPolynomial.mkDerivation k v
81   have hDI : ∀ f ∈ I, D f = 0 := by
82     intro f hf
83     exact Stafford38.Geometry.EtaleTangentKernel.mkDerivation_vanishes_of_mem_zariskiTangentSpace
84       I q v halg hv f hf
85   let DA : Derivation k (MvPolynomial (Fin n) k ⧸ I) L :=
86     Stafford38.Geometry.GenericPointKaehlerConormal.quotientDerivation D hDI
87   let DC : Derivation k C L :=
88     Stafford38.Geometry.EtaleDerivationExtension.extendDerivation DA
89   refine ⟨fun j => DC
90       (algebraMap (MvPolynomial (σ) k) C (MvPolynomial.X j)), ?_⟩
91   intro i
92   have hcomp : DC (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
93       (Ideal.Quotient.mk I (MvPolynomial.X i))) =
94       D (MvPolynomial.X i) := by
95     calc
96       DC (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
97         (Ideal.Quotient.mk I (MvPolynomial.X i))) =
98           DA (Ideal.Quotient.mk I (MvPolynomial.X i)) :=
99         Stafford38.Geometry.EtaleDerivationExtension.extendDerivation_algebraMap DA _
100       _ = D (MvPolynomial.X i) :=
101         Stafford38.Geometry.GenericPointKaehlerConormal.quotientDerivation_mk D hDI _
102   have hcomp' : D (MvPolynomial.X i) =
103       DC (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
104         (Ideal.Quotient.mk I (MvPolynomial.X i))) := by
105     simpa [D, MvPolynomial.mkDerivation_X] using hcomp.symm
106   have hsum_eval :
107       (∑ j, DC (algebraMap (MvPolynomial (σ) k) C
108         (MvPolynomial.X j)) •
109           Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
110             (k := k) (σ := σ) (B := C) (L := L) j)
111         (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
112           (Ideal.Quotient.mk I (MvPolynomial.X i))) =
113       ∑ j, DC (algebraMap (MvPolynomial (σ) k) C
114         (MvPolynomial.X j)) *
115           Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
116             (k := k) (σ := σ) (B := C) (L := L) j
117           (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
118             (Ideal.Quotient.mk I (MvPolynomial.X i)) ) := by
119     change Derivation.coeFnAddMonoidHom (∑ j, _) _ = _
120     rw [map_sum]
121     simp [Derivation.coeFnAddMonoidHom, smul_eq_mul]
122   have hdecomp :=
123     Stafford38.Geometry.EtaleCotangentBasis.derivation_eq_sum_coordinateDerivations
124       (k := k) (σ := σ) (B := C) (L := L) DC
125   calc
126     v i = D (MvPolynomial.X i) := by simp [D]
127     _ = DC (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
128         (Ideal.Quotient.mk I (MvPolynomial.X i))) := hcomp'
129     _ = (∑ j, (DC (algebraMap (MvPolynomial (σ) k) C
130         (MvPolynomial.X j)) •
131           Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
132             (k := k) (σ := σ) (B := C) (L := L) j))
133           (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
134             (Ideal.Quotient.mk I (MvPolynomial.X i))) := by
135       exact congrArg (fun E : Derivation k C L =>
136         E (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
137           (Ideal.Quotient.mk I (MvPolynomial.X i)))) hdecomp
138     _ = ∑ j, DC (algebraMap (MvPolynomial (σ) k) C
139         (MvPolynomial.X j)) *
140           Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
141             (k := k) (σ := σ) (B := C) (L := L) j
142           (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
143             (Ideal.Quotient.mk I (MvPolynomial.X i))) := hsum_eval

146 /-- The genuine parameter derivations of a common étale chart span exactly
147  the equation-defined tangent space of the original affine presentation.
148  The ambient number of coordinates and the number of parameters are independent. -/
149 theorem zariskiTangentSpace_eq_span_parameterDerivations
150     [Algebra (MvPolynomial (Fin n) k) C]
151     [IsScalarTower k (MvPolynomial (Fin n) k) C]
152     [IsScalarTower (MvPolynomial (Fin n) k)
153       (MvPolynomial (Fin n) k ⧸ I) C]
154     [IsScalarTower (MvPolynomial (Fin n) k) C L]
155     (q : Fin n → L)
156     (hq : ∀ i, q i = algebraMap C L
157       (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
158         (Ideal.Quotient.mk I (MvPolynomial.X i)))) :
159     Stafford38.Geometry.AffineConormalSpan.zariskiTangentSpace q
160       (I.map (MvPolynomial.map (algebraMap k L))) =
161       Submodule.span L (Set.range fun j : σ => fun i : Fin n =>
162         Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
163           (k := k) (σ := σ) (B := C) (L := L) j
164           (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
165             (Ideal.Quotient.mk I (MvPolynomial.X i)))) := by
166   classical
167   let cols : σ → Fin n → L := fun j i =>
168     Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
169       (k := k) (σ := σ) (B := C) (L := L) j
170       (algebraMap (MvPolynomial (Fin n) k ⧸ I) C
171         (Ideal.Quotient.mk I (MvPolynomial.X i)))
172   change _ = Submodule.span L (Set.range cols)
173   have halg : ∀ p : MvPolynomial (Fin n) k,
174       algebraMap (MvPolynomial (Fin n) k) L p = MvPolynomial.aeval q p := by
175     have hhom : IsScalarTower.toAlgHom k (MvPolynomial (Fin n) k) L =
176         MvPolynomial.aeval q := by
177       apply MvPolynomial.algHom_ext
178       intro i
179       rw [IsScalarTower.coe_toAlgHom', MvPolynomial.aeval_X, hq i]
180       rw [IsScalarTower.algebraMap_apply (MvPolynomial (Fin n) k) C L,
181         IsScalarTower.algebraMap_apply (MvPolynomial (Fin n) k)
182           (MvPolynomial (Fin n) k ⧸ I) C]
183       simp only [Ideal.Quotient.algebraMap_eq]
184     intro p
185     exact congrArg (fun f : MvPolynomial (Fin n) k →ₐ[k] L => f p) hhom
186   have hI : ∀ f ∈ I, algebraMap (MvPolynomial (Fin n) k) C f = 0 := by
187     intro f hf
188     rw [IsScalarTower.algebraMap_apply (MvPolynomial (Fin n) k)
189       (MvPolynomial (Fin n) k ⧸ I) C]
190     simp only [Ideal.Quotient.algebraMap_eq]
191     rw [Ideal.Quotient.eq_zero_iff_mem.mpr hf, map_zero]
192   apply le_antisymm
193   · intro v hv
194     obtain ⟨coeff, hcoeff⟩ := tangentVector_eq_sum_parameterDerivations (σ := σ) q v hq hv
195     have hvrepr : v = ∑ j, coeff j • cols j := by
196       ext i
197       simpa [cols, Finset.sum_apply, Pi.smul_apply, smul_eq_mul] using hcoeff i
198     rw [hvrepr]
199     apply Submodule.sum_mem
200     intro j _
201     exact Submodule.smul_mem _ _ (Submodule.subset_span ⟨j, rfl⟩)
202   · apply Submodule.span_le.mpr
203     rintro _ ⟨j, rfl⟩
204     have h := Stafford38.Geometry.EtaleTangentKernel.chartDerivation_mem_zariskiTangentSpace
205       (Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
206         (k := k) (σ := σ) (B := C) (L := L) j) I q halg hI
207     have heq : cols j = (fun i =>
208         (Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
209           (k := k) (σ := σ) (B := C) (L := L) j).compAlgebraMap
210             (MvPolynomial (Fin n) k) (MvPolynomial.X i)) := by
211       ext i
212       simp only [cols, Derivation.compAlgebraMap_apply,
213         IsScalarTower.algebraMap_apply (MvPolynomial (Fin n) k)
214           (MvPolynomial (Fin n) k ⧸ I) C, Ideal.Quotient.algebraMap_eq]
215     exact heq ▸ h
