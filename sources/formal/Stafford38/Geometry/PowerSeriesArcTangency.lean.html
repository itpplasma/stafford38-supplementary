7 /-!
8 # Tangency of the formal-arc velocity
9 
10 If a power-series arc annihilates an affine ideal coefficientwise, its formal
11 derivative and the residues of arbitrary derivation directions are Zariski
12 tangent vectors at the constant point.  The result is deliberately
13 independent of any projective closure or normalization: it consumes an actual
14 power-series arc and produces actual tangent data for the scalar-extended
15 base ideal.
16 
17 The file does not construct the arc or identify its tangent data with a
18 projective component.  Those chart, normalization, frame-independence, and
19 dimension arguments remain separate geometric obligations.
20 -/

37 /-- Chain rule for a derivation of a field-valued polynomial evaluation. -/
38 theorem derivation_eval₂
39     {S : Type*} [CommSemiring S] [Algebra k S] [Module k S]
40     (D : Derivation k S S)
41     {m : ℕ} (f : MvPolynomial (Fin m) k)
42     (q : Fin m → S) :
43     D (MvPolynomial.eval₂ (algebraMap k S) q f) =
44       ∑ i, MvPolynomial.eval₂ (algebraMap k S) q
45           (MvPolynomial.pderiv i f) * D (q i) := by
46   induction f using MvPolynomial.induction_on with
47   | C a => simp [PowerSeries.algebraMap_apply]
48   | add f g hf hg =>
49       simp only [MvPolynomial.eval₂_add, D.map_add, map_add, hf, hg,
50         Finset.sum_add_distrib, add_mul]
51   | mul_X f i hf =>
52       simp only [MvPolynomial.eval₂_mul, D.leibniz, smul_eq_mul, hf,
53         MvPolynomial.pderiv_mul, map_add, add_mul, Finset.sum_add_distrib,
54         MvPolynomial.eval₂_add, MvPolynomial.eval₂_X,
55         MvPolynomial.eval₂_zero, MvPolynomial.pderiv_X,
56         Pi.single_apply]
57       simp [Finset.mul_sum, Finset.sum_ite_eq', eq_comm, mul_comm,
58         mul_left_comm, mul_assoc]
59       have hsecond :
60           (∑ x : Fin m,
61             MvPolynomial.eval₂ (algebraMap k S) q f *
62               (MvPolynomial.eval₂ (algebraMap k S) q
63                 (if i = x then 1 else 0) * D (q x))) =
64             MvPolynomial.eval₂ (algebraMap k S) q f * D (q i) := by
65         classical
66         rw [← Finset.mul_sum]
67         congr 1
68         simpa using
69           (Finset.sum_eq_single (s := Finset.univ)
70             (f := fun x : Fin m ↦
71               MvPolynomial.eval₂ (algebraMap k S)
72                   q (if i = x then 1 else 0) * D (q x)) i
73             (by
74               intro b hb hbi
75               simp [hbi, Ne.symm hbi])
76             (by
77               intro hi
78               exact (hi (Finset.mem_univ i)).elim))
79       rw [hsecond]
80       ring

82 /-! ## Derivation directions of extended ideal loci -/
83 
84 /-- A derivation of an evaluation point is tangent to the scalar extension of
85 an ideal that vanishes at that point.  This is the coefficient-direction
86 analogue of the power-series arc theorem below; it is stated for an arbitrary
87 field extension and an arbitrary derivation, so no chart or completeness
88 hypothesis is hidden in the result. -/
89 theorem derivationVector_mem_zariskiTangentSpace_of_eval₂_eq_zero
90     {S : Type*} [Field S] [Algebra k S] [Module k S]
91     (D : Derivation k S S)
92     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) k))
93     (q : Fin m → S)
94     (hq : ∀ f ∈ I,
95       MvPolynomial.eval₂ (algebraMap k S) q f = 0) :
96     (fun i ↦ D (q i)) ∈
97       zariskiTangentSpace q
98         (I.map (MvPolynomial.map (algebraMap k S))) := by
99   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator]
100   intro φ hφ
101   induction hφ using Submodule.span_induction with
102   | mem f hf =>
103       rcases hf with ⟨g, rfl⟩
104       have hmap :
105           ∀ h : MvPolynomial (Fin m) S,
106             h ∈ I.map (MvPolynomial.map (algebraMap k S)) →
107               MvPolynomial.eval q h = 0 ∧
108                 differentialCovector q h (fun i ↦ D (q i)) = 0 := by
109         intro h hh
110         rw [Ideal.map] at hh
111         induction hh using Submodule.span_induction with
112         | mem h hh =>
113             rcases hh with ⟨g, hg, rfl⟩
114             constructor
115             · rw [MvPolynomial.eval_map]
116               exact hq g hg
117             · have hderiv := derivation_eval₂ D g q
118               rw [hq g hg] at hderiv
119               simpa [differentialCovector, differentialAt,
120                 MvPolynomial.pderiv_map, MvPolynomial.eval_map] using
121                 hderiv.symm
122         | zero =>
123             simp [differentialCovector, differentialAt]
124         | add h₁ h₂ hh₁ hh₂ ih₁ ih₂ =>
125             constructor
126             · rw [MvPolynomial.eval_add, ih₁.1, ih₂.1, add_zero]
127             · have hi := congrArg₂ (· + ·) ih₁.2 ih₂.2
128               change ∑ i, MvPolynomial.eval q
129                   (MvPolynomial.pderiv i (h₁ + h₂)) * D (q i) = 0
130               simp only [map_add, MvPolynomial.eval_add, add_mul]
131               rw [Finset.sum_add_distrib]
132               simpa [differentialCovector, differentialAt] using hi
133         | smul a h hh ih =>
134             constructor
135             · simpa [smul_eq_mul, ih.1]
136             · have ihD :
137                   ∑ i, MvPolynomial.eval q
138                       (MvPolynomial.pderiv i h) * D (q i) = 0 := by
139                 simpa [differentialCovector, differentialAt] using ih.2
140               have hprod :
141                   differentialCovector q (a * h) (fun i ↦ D (q i)) =
142                     MvPolynomial.eval q h *
143                         differentialCovector q a (fun i ↦ D (q i)) +
144                       MvPolynomial.eval q a *
145                   differentialCovector q h (fun i ↦ D (q i)) := by
146                 simp only [differentialCovector_apply, differentialAt,
147                   MvPolynomial.pderiv_mul,
148                   MvPolynomial.eval_add, MvPolynomial.eval_mul, add_mul,
149                   Finset.sum_add_distrib]
150                 simp [Finset.mul_sum, Finset.sum_mul, mul_assoc, mul_comm,
151                   mul_left_comm]
152               rw [show (a • h : MvPolynomial (Fin m) S) = a * h by rfl]
153               rw [hprod]
154               simp [ih.1, ih.2]
155       exact hmap g.1 g.2 |>.2
156   | zero => simp
157   | add φ ψ hφ hψ ihφ ihψ =>
158       simpa [ihφ, ihψ]
159   | smul a φ hφ ihφ =>
160       simpa [ihφ]

162 /-- Constant-term evaluation commutes with evaluating a base-field
163 polynomial in a power-series point. -/
164 theorem constantCoeff_eval₂
165     {K : Type u} [Field K] [Algebra k K]
166     {m : ℕ} (f : MvPolynomial (Fin m) k)
167     (q : Fin m → PowerSeries K) :
168     PowerSeries.constantCoeff
169         (MvPolynomial.eval₂ (algebraMap k (PowerSeries K)) q f) =
170       MvPolynomial.eval₂ (algebraMap k K)
171         (fun i ↦ PowerSeries.constantCoeff (q i)) f := by
172   induction f using MvPolynomial.induction_on with
173   | C a =>
174       simp only [MvPolynomial.eval₂_C]
175       rw [PowerSeries.algebraMap_apply, PowerSeries.constantCoeff_C]
176   | add f g hf hg => simp [hf, hg]
177   | mul_X f i hf => simp [hf]

179 /-- The residue of an arbitrary derivation direction on an annihilating
180 power-series arc is tangent to the scalar-extended base ideal.  The
181 derivation need not preserve the coefficient field: the ideal-span argument
182 handles the coefficient terms because every base equation already vanishes
183 along the arc. -/
184 theorem residueDerivationVector_mem_zariskiTangentSpace_of_eval₂_eq_zero
185     {K : Type u} [Field K] [Algebra k K]
186     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) k))
187     (q : Fin m → PowerSeries K)
188     (D : Derivation k (PowerSeries K) (PowerSeries K))
189     (hq : ∀ f ∈ I,
190       MvPolynomial.eval₂ (algebraMap k (PowerSeries K)) q f = 0) :
191     (fun i ↦ PowerSeries.constantCoeff (D (q i))) ∈
192       zariskiTangentSpace
193         (fun i ↦ PowerSeries.constantCoeff (q i))
194         (I.map (MvPolynomial.map (algebraMap k K))) := by
195   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator]
196   intro φ hφ
197   induction hφ using Submodule.span_induction with
198   | mem f hf =>
199       rcases hf with ⟨g, rfl⟩
200       have hmap :
201           ∀ h : MvPolynomial (Fin m) K,
202             h ∈ I.map (MvPolynomial.map (algebraMap k K)) →
203               MvPolynomial.eval
204                   (fun i ↦ PowerSeries.constantCoeff (q i)) h = 0 ∧
205                 differentialCovector
206                     (fun i ↦ PowerSeries.constantCoeff (q i)) h
207                     (fun i ↦ PowerSeries.constantCoeff (D (q i))) = 0 := by
208         intro h hh
209         rw [Ideal.map] at hh
210         induction hh using Submodule.span_induction with
211         | mem h hh =>
212             rcases hh with ⟨g, hg, rfl⟩
213             have hzero := congrArg (PowerSeries.constantCoeff) (hq g hg)
214             rw [constantCoeff_eval₂] at hzero
215             constructor
216             · rw [MvPolynomial.eval_map]
217               exact hzero
218             · have hderiv := derivation_eval₂ D g q
219               rw [hq g hg] at hderiv
220               have hconst :=
221                 congrArg (PowerSeries.constantCoeff) hderiv.symm
222               simpa [differentialCovector, differentialAt,
223                 MvPolynomial.pderiv_map, MvPolynomial.eval_map,
224                 constantCoeff_eval₂] using hconst
225         | zero =>
226             simp [differentialCovector, differentialAt]
227         | add h₁ h₂ hh₁ hh₂ ih₁ ih₂ =>
228             constructor
229             · rw [MvPolynomial.eval_add, ih₁.1, ih₂.1, add_zero]
230             · have hi := congrArg₂ (· + ·) ih₁.2 ih₂.2
231               change ∑ i, MvPolynomial.eval
232                   (fun i ↦ PowerSeries.constantCoeff (q i))
233                   (MvPolynomial.pderiv i (h₁ + h₂)) *
234                     PowerSeries.constantCoeff (D (q i)) = 0
235               simp only [map_add, MvPolynomial.eval_add, add_mul]
236               rw [Finset.sum_add_distrib]
237               simpa [differentialCovector, differentialAt] using hi
238         | smul a h hh ih =>
239             constructor
240             · simpa [smul_eq_mul, ih.1]
241             · have hprod :
242                   differentialCovector
243                       (fun i ↦ PowerSeries.constantCoeff (q i))
244                       (a * h)
245                       (fun i ↦ PowerSeries.constantCoeff (D (q i))) =
246                     MvPolynomial.eval
247                         (fun i ↦ PowerSeries.constantCoeff (q i)) h *
248                         differentialCovector
249                           (fun i ↦ PowerSeries.constantCoeff (q i)) a
250                           (fun i ↦ PowerSeries.constantCoeff (D (q i))) +
251                       MvPolynomial.eval
252                           (fun i ↦ PowerSeries.constantCoeff (q i)) a *
253                         differentialCovector
254                           (fun i ↦ PowerSeries.constantCoeff (q i)) h
255                           (fun i ↦ PowerSeries.constantCoeff (D (q i))) := by
256                 simp only [differentialCovector_apply, differentialAt,
257                   MvPolynomial.pderiv_mul, MvPolynomial.eval_add,
258                   MvPolynomial.eval_mul, add_mul, Finset.sum_add_distrib]
259                 simp [Finset.mul_sum, Finset.sum_mul, mul_assoc, mul_comm,
260                   mul_left_comm]
261               rw [show (a • h : MvPolynomial (Fin m) K) = a * h by rfl]
262               rw [hprod]
263               simp [ih.1, ih.2]
264       exact hmap g.1 g.2 |>.2
265   | zero => simp
266   | add φ ψ hφ hψ ihφ ihψ =>
267       simpa [ihφ, ihψ]
268   | smul a φ hφ ihφ =>
269       simpa [ihφ]

271 /-- The uniformizer member of the power-series frame is tangent after
272 residue specialization whenever the arc annihilates the base ideal. -/
273 theorem residueFrameVector_uniformizer_mem_zariskiTangentSpace_of_eval₂_eq_zero
274     {K : Type u} [Field K] [Algebra k K]
275     {m κ : ℕ} (I : Ideal (MvPolynomial (Fin m) k))
276     (q : Fin m → PowerSeries K)
277     (D : Fin κ → Derivation k K K)
278     (hq : ∀ f ∈ I,
279       MvPolynomial.eval₂ (algebraMap k (PowerSeries K)) q f = 0) :
280     residueFrameVector (k := k) (K := K) q D none ∈
281       zariskiTangentSpace
282         (fun i ↦ PowerSeries.constantCoeff (q i))
283         (I.map (MvPolynomial.map (algebraMap k K))) := by
284   have hD :=
285     residueDerivationVector_mem_zariskiTangentSpace_of_eval₂_eq_zero
286       (k := k) (K := K) I q
287         (uniformizerDerivation (k := k) (K := K)) hq
288   change (fun i ↦ PowerSeries.constantCoeff
289       (uniformizerDerivation (k := k) (K := K) (q i))) ∈ _
290   exact hD

315 /-- Formal chain rule for evaluating a polynomial along a power-series arc. -/
316 theorem derivative_eval_map
317     {m : ℕ} (f : MvPolynomial (Fin m) k)
318     (q : Fin m → PowerSeries k) :
319     PowerSeries.derivative (R := k)
320         (MvPolynomial.eval q (MvPolynomial.map (PowerSeries.C) f)) =
321       ∑ i, MvPolynomial.eval q
322           (MvPolynomial.map (PowerSeries.C) (MvPolynomial.pderiv i f)) *
323         PowerSeries.derivative (R := k) (q i) := by
324   induction f using MvPolynomial.induction_on with
325   | C a => simp
326   | add f g hf hg =>
327       simp only [map_add, MvPolynomial.eval_add, Derivation.map_add,
328         hf, hg, Finset.sum_add_distrib, add_mul]
329   | mul_X f i hf =>
330       simp only [map_mul, MvPolynomial.eval_mul, Derivation.leibniz,
331         smul_eq_mul, map_add, hf, MvPolynomial.pderiv_mul, add_mul,
332         Finset.sum_add_distrib]
333       simp [Pi.single_apply, Finset.mul_sum, mul_comm, mul_left_comm,
334         mul_assoc]

353 /-- The formal-arc velocity annihilates the differential of every polynomial
354 that vanishes identically along the arc. -/
355 theorem differentialCovector_arcVelocity_eq_zero_of_eval_eq_zero
356     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) k))
357     (q : Fin m → PowerSeries k)
358     (hq : ∀ f ∈ I,
359       MvPolynomial.eval q (MvPolynomial.map (PowerSeries.C) f) = 0) :
360     ∀ f ∈ I,
361       differentialCovector (fun i ↦ PowerSeries.constantCoeff (q i)) f
362         (powerSeriesFirstCoefficient q) = 0 := by
363   intro f hf
364   have hderiv := derivative_eval_map f q
365   rw [hq f hf, map_zero] at hderiv
366   have hconst :
367       (0 : k) =
368         ∑ i, PowerSeries.constantCoeff
369             (MvPolynomial.eval q
370               (MvPolynomial.map (PowerSeries.C)
371                 (MvPolynomial.pderiv i f))) *
372           PowerSeries.constantCoeff
373             (PowerSeries.derivative (R := k) (q i)) := by
374     simpa only [map_zero, map_sum, map_mul] using
375       congrArg (PowerSeries.constantCoeff) hderiv
376   simp_rw [residue_eval_map] at hconst
377   have hres : residueColumn q =
378       (fun i ↦ PowerSeries.constantCoeff (q i)) := rfl
379   rw [hres] at hconst
380   simpa [differentialCovector, differentialAt, powerSeriesFirstCoefficient,
381     ← PowerSeries.coeff_zero_eq_constantCoeff,
382     PowerSeries.coeff_mul, Finset.sum_apply, Finset.sum_mul] using hconst.symm

384 /-- An actual power-series arc supplies the uniformizer tangent vector needed
385 by the higher-dimensional conormal consumer. -/
386 theorem arcVelocity_mem_zariskiTangentSpace_of_eval_eq_zero
387     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) k))
388     (q : Fin m → PowerSeries k)
389     (hq : ∀ f ∈ I,
390       MvPolynomial.eval q (MvPolynomial.map (PowerSeries.C) f) = 0) :
391     powerSeriesFirstCoefficient q ∈
392       zariskiTangentSpace
393         (fun i ↦ PowerSeries.constantCoeff (q i)) I := by
394   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator]
395   intro φ hφ
396   induction hφ using Submodule.span_induction with
397   | mem f hf =>
398       rcases hf with ⟨g, rfl⟩
399       exact differentialCovector_arcVelocity_eq_zero_of_eval_eq_zero I q hq
400         g.1 g.2
401   | zero => simp
402   | add φ ψ hφ hψ ihφ ihψ =>
403       have ihφ' : φ (fun i ↦
404           PowerSeries.constantCoeff (PowerSeries.derivative (R := k) (q i))) = 0 := by
405         simpa only [← arcVelocity_eq_constantCoeff_derivative q] using ihφ
406       have ihψ' : ψ (fun i ↦
407           PowerSeries.constantCoeff (PowerSeries.derivative (R := k) (q i))) = 0 := by
408         simpa only [← arcVelocity_eq_constantCoeff_derivative q] using ihψ
409       simp [ihφ', ihψ']
410   | smul a φ hφ ihφ =>
411       have ihφ' : φ (fun i ↦
412           PowerSeries.constantCoeff (PowerSeries.derivative (R := k) (q i))) = 0 := by
413         simpa only [← arcVelocity_eq_constantCoeff_derivative q] using ihφ
414       simp [ihφ']
