48 /-- A strict DVR order gap remains strict in its chosen power-series
49 completion. The witness-specific dependent local algebra is kept out of this
50 generic step. -/
51 theorem completed_order_gap
52     {R κ : Type u} [CommRing R] [IsDomain R]
53     [IsDiscreteValuationRing R] [Field κ]
54     (ret : R →+* PowerSeries κ) (t : R)
55     (ht : ret t = (PowerSeries.X : PowerSeries κ))
56     (huniformizer : Irreducible t)
57     (q₀ ratio q₁ : R) (hq₀_ne : q₀ ≠ 0) (hratio_ne : ratio ≠ 0)
58     (hq₀_vanish : q₀ ∈ maximalIdeal R)
59     (hratio_vanish : ratio ∈ maximalIdeal R)
60     (hq₁ : q₁ = q₀ * ratio) :
61     ∃ (a b : ℕ) (u₀ u₁ : PowerSeries κ),
62       0 < a ∧ a < b ∧
63       ret q₀ = (PowerSeries.X : PowerSeries κ) ^ a * u₀ ∧
64       PowerSeries.constantCoeff u₀ ≠ 0 ∧
65       ret q₁ = (PowerSeries.X : PowerSeries κ) ^ b * u₁ := by
66   obtain ⟨a, _r, b, u₀v, _ur, u₁v, ha, _hr, _hb, hab,
67       hq₀v, _hratio, _hu₁, hq₁v⟩ :=
68     exists_uniformizer_strict_orderGap t huniformizer q₀ ratio q₁
69       hq₀_ne hratio_ne hq₀_vanish hratio_vanish hq₁
70   let u₀ : PowerSeries κ := ret (u₀v : R)
71   let u₁ : PowerSeries κ := ret (u₁v : R)
72   have hq₀series : ret q₀ = PowerSeries.X ^ a * u₀ := by
73     calc
74       ret q₀ = ret ((u₀v : R) * t ^ a) := congrArg ret hq₀v
75       _ = ret (u₀v : R) * PowerSeries.X ^ a := by rw [map_mul, map_pow, ht]
76       _ = PowerSeries.X ^ a * u₀ := by simp [u₀, mul_comm]
77   have hu₀unit : IsUnit u₀ := by
78     change IsUnit (ret (u₀v : R))
79     exact u₀v.isUnit.map ret
80   have hu₀coeff : PowerSeries.constantCoeff u₀ ≠ 0 :=
81     (PowerSeries.isUnit_iff_constantCoeff.mp hu₀unit).ne_zero
82   have hq₁series : ret q₁ = PowerSeries.X ^ b * u₁ := by
83     calc
84       ret q₁ = ret ((u₁v : R) * t ^ b) := congrArg ret hq₁v
85       _ = ret (u₁v : R) * PowerSeries.X ^ b := by rw [map_mul, map_pow, ht]
86       _ = PowerSeries.X ^ b * u₁ := by simp [u₁, mul_comm]
87   exact ⟨a, b, u₀, u₁, ha, hab, hq₀series, hu₀coeff, hq₁series⟩

89 /-- The same normalized retained column has the strict denominator/axis
90 order gap in its completed chart. -/
91 theorem order_gap_of_actual_witness
92     (hm : 0 < m) (P : PrimeSpectrum (MvPolynomial (Fin m) k))
93     (w : GeneralDivisorialVisibleFrameWitness hm P)
94     (hunit : ∃ g : MvPolynomial (Fin m) k,
95       MvPolynomial.X ⟨0, hm⟩ * g - 1 ∈ P.asIdeal) :
96     letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
97       (ComponentFractionField P) := w.column.W.ambientAlgebra
98     letI : IsDiscreteValuationRing
99       w.column.W.place.valuation.toSubring := w.column.W.place.isDiscrete
100     ∃ (a b : ℕ)
101       (u₀ u₁ : PowerSeries
102         (ResidueField (w.column.W.place.valuation.toSubring))),
103       0 < a ∧ a < b ∧
104       retainedToCompletedPowerSeries w.column.W (w.column.q 0) =
105         (PowerSeries.X : PowerSeries
106           (ResidueField (w.column.W.place.valuation.toSubring))) ^ a * u₀ ∧
107       PowerSeries.constantCoeff u₀ ≠ 0 ∧
108       retainedToCompletedPowerSeries w.column.W
109         (w.column.q (Fin.succ ⟨0, hm⟩)) =
110         (PowerSeries.X : PowerSeries
111           (ResidueField (w.column.W.place.valuation.toSubring))) ^ b * u₁ := by
112   let i : Fin m := ⟨0, hm⟩
113   let C := w.column
114   let W := C.W
115   let K := ComponentFractionField P
116   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) K := W.ambientAlgebra
117   let V := W.place.valuation.toSubring
118   letI : IsDiscreteValuationRing V := W.place.isDiscrete
119   letI : Algebra W.coefficientField V :=
120     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
121   let κ := ResidueField V
122   letI : Algebra k κ := retainedResidueGroundAlgebra P i W
123   obtain ⟨g, hg⟩ := hunit
124   let phi : MvPolynomial (Fin m) k →+* K :=
125     (algebraMap (MvPolynomial (Fin m) k ⧸ P.asIdeal) K).comp
126       (Ideal.Quotient.mk P.asIdeal)
127   have hphiC : phi.comp MvPolynomial.C = algebraMap k K := by
128     ext c
129     exact IsScalarTower.algebraMap_apply k
130       (MvPolynomial (Fin m) k ⧸ P.asIdeal) K c
131   have hphiX : ∀ j, phi (MvPolynomial.X j) = componentCoordinate P j := by
132     intro j
133     rfl
134   have hpoly : phi g = MvPolynomial.eval₂ (algebraMap k K)
135       (fun j ↦ componentCoordinate P j) g := by
136     rw [MvPolynomial.map_mvPolynomial_eq_eval₂ phi g]
137     change MvPolynomial.eval₂Hom (phi.comp MvPolynomial.C)
138         (fun j ↦ phi (MvPolynomial.X j)) g =
139       MvPolynomial.eval₂Hom (algebraMap k K)
140         (fun j ↦ componentCoordinate P j) g
141     apply MvPolynomial.eval₂Hom_congr hphiC
142     · funext j
143       exact hphiX j
144     · rfl
145   have hinverse : componentCoordinate P i *
146       MvPolynomial.eval₂ (algebraMap k K)
147         (fun j ↦ componentCoordinate P j) g = 1 := by
148     have hzero : phi (MvPolynomial.X i * g - 1) = 0 := by
149       have hmk : Ideal.Quotient.mk P.asIdeal
150           (MvPolynomial.X i * g - 1) = 0 :=
151         Ideal.Quotient.eq_zero_iff_mem.mpr hg
152       simpa [phi] using (congrArg
153         (algebraMap (MvPolynomial (Fin m) k ⧸ P.asIdeal) K) hmk)
154     rw [map_sub, map_mul, map_one, sub_eq_zero, hphiX, hpoly] at hzero
155     exact hzero
156   have hq₀nonunit : ¬ IsUnit (C.q 0) :=
157     normalized_denominator_nonunit_of_polynomial_inverse
158       (V := W.place.valuation) (coeff := C.groundCoeff)
159       C.groundCoeff_commutes
160       (x := fun j ↦ componentCoordinate P j)
161       (qzero := C.q 0) (q := fun j ↦ C.q (Fin.succ j))
162       (scale := C.scale) (i := i) (parameter := W.place.parameter) (g := g)
163       (by simpa [componentProjectivePoint] using C.q_commonScale 0)
164       (by intro j; simpa [componentProjectivePoint] using
165         C.q_commonScale (Fin.succ j))
166       W.parameter_eq_coordinate W.place.parameter_nonunit hinverse
167   have hq₀max : C.q 0 ∈ maximalIdeal V := by
168     rw [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff]
169     exact hq₀nonunit
170   have hparammax : W.place.parameter ∈ maximalIdeal V := by
171     rw [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff]
172     exact W.place.parameter_nonunit
173   have hcompleted := completed_order_gap
174     (retainedToCompletedPowerSeries W) (chosenUniformizer V)
175     (retainedToCompletedPowerSeries_chosenUniformizer W)
176     (chosenUniformizer_irreducible V)
177     (C.q 0) W.place.parameter (C.q (Fin.succ i)) C.q0_ne
178     W.place.parameter_ne hq₀max hparammax C.q_parameter
179   simpa [C, W, i, V, κ] using hcompleted
180 
181 end
