12 /-- Localize the common generic open at the image of a chart denominator.
13 The denominator is mapped through the actual `B`-algebra structure. -/
14 abbrev genericOpenExtraAwayB
15     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
16     (M : Ideal B) [M.IsPrime] (f : Q)
17     (e : Localization.Away f ≃ₐ[Q]
18       Localization.Away (algebraMap Q B f)) (g : Q) : Type u :=
19   Localization.Away
20     (genericOpenBMap M f e (algebraMap Q B g))

45 /-- Formal étaleness is transported first from the actual local ring `B_M`
46 to the common open, then through the extra localization. -/
47 theorem formallyEtale_genericOpenExtraAway_of_pointLocal
48     {R : Type*} [CommRing R]
49     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
50     [Algebra R B]
51     (M : Ideal B) [M.IsPrime] (f : Q)
52     (e : Localization.Away f ≃ₐ[Q]
53       Localization.Away (algebraMap Q B f)) (g : Q)
54     [Algebra.FormallyEtale R (Localization.AtPrime M)] :
55     let Cq := genericOpenRing M f e
56     let C := Localization.Away (genericOpenBMap M f e (algebraMap Q B g))
57     letI : Algebra R Cq :=
58       ((algebraMap B (genericOpenRing M f e)).comp (algebraMap R B)).toAlgebra
59     letI : Algebra R C := Algebra.compHom C (algebraMap R Cq)
60     Algebra.FormallyEtale R C := by
61   dsimp
62   let Cq := genericOpenRing M f e
63   let C := Localization.Away (genericOpenBMap M f e (algebraMap Q B g))
64   letI : Algebra R Cq :=
65     ((algebraMap B Cq).comp (algebraMap R B)).toAlgebra
66   letI : IsScalarTower R B Cq := IsScalarTower.of_algebraMap_eq (fun _ => rfl)
67   letI : Algebra R C := Algebra.compHom C (algebraMap R Cq)
68   letI : Algebra Cq C := inferInstance
69   letI : SMul Cq C := (inferInstance : Algebra Cq C).toSMul
70   letI : SMul R C := (Algebra.compHom C (algebraMap R Cq)).toSMul
71   letI : IsScalarTower R Cq C := IsScalarTower.of_algebraMap_eq (fun _ => rfl)
72   letI : Algebra.FormallyEtale Cq C :=
73     Algebra.FormallyEtale.of_isLocalization
74       (Submonoid.powers (genericOpenBMap M f e (algebraMap Q B g)))
75   letI : Algebra.FormallyEtale R Cq :=
76     formallyEtale_genericOpenRing_of_pointLocal M f e
77   exact Algebra.FormallyEtale.comp R Cq C

79 def genericArcToGenericOpenExtraAwayB
80     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
81     {L : Type*} [Field L]
82     (M : Ideal B) [M.IsPrime] (f : Q)
83     (e : Localization.Away f ≃ₐ[Q]
84       Localization.Away (algebraMap Q B f))
85     (ρ : B →+* L)
86     (hf : IsUnit (ρ (algebraMap Q B f)))
87     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b)) (g : Q)
88     (hg : IsUnit (ρ (algebraMap Q B g))) :
89     genericOpenExtraAwayB M f e g →+* L := by
90   let ψ := genericArcToGenericOpen M f e ρ hf hunitM
91   have hval : ψ (genericOpenBMap M f e (algebraMap Q B g)) =
92       ρ (algebraMap Q B g) := by
93     exact genericArcToGenericOpen_apply M f e ρ hf hunitM (algebraMap Q B g)
94   have hψ : IsUnit
95       (ψ (genericOpenBMap M f e (algebraMap Q B g))) := by
96     rw [hval]
97     exact hg
98   exact Localization.awayLift ψ
99     (genericOpenBMap M f e (algebraMap Q B g)) hψ

101 /-- The chart-denominator localization of the common open agrees, as a
102 `B`-algebra, with the same localization of the point-local presentation. -/
103 noncomputable def genericOpenExtraAwayEquiv_pointLocalAwayB
104     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
105     (M : Ideal B) [M.IsPrime] (f : Q)
106     (e : Localization.Away f ≃ₐ[Q]
107       Localization.Away (algebraMap Q B f)) (g : Q) :
108     genericOpenExtraAwayB M f e g ≃ₐ[B] pointLocalExtraAwayB M f g := by
109   let Cq := genericOpenRing M f e
110   let Cl := pointLocalAwayOpen M f
111   let b := algebraMap Q B g
112   let eB : Cq ≃ₐ[B] Cl := genericOpenEquiv_pointLocalAway_overB M f e
113   let N : Submonoid Cq := Submonoid.powers (genericOpenBMap M f e b)
114   let T : Submonoid Cl := Submonoid.powers (algebraMap B Cl b)
115   let C : Type u := Localization.Away
116     (genericOpenBMap M f e (algebraMap Q B g))
117   letI : SMul B C := (Algebra.compHom C (genericOpenBMap M f e)).toSMul
118   have hTower : IsScalarTower B Cq (genericOpenExtraAwayB M f e g) := by
119     change IsScalarTower B Cq C
120     exact ⟨fun b' c x => by
121       change (b' • c) • x = genericOpenBMap M f e b' • (c • x)
122       rw [Algebra.smul_def]
123       exact mul_smul _ _ _⟩
124   letI : IsScalarTower B Cq (genericOpenExtraAwayB M f e g) := hTower
125   have hmap : Submonoid.map eB.toMonoidHom N = T := by
126     rw [Submonoid.map_powers]
127     change Submonoid.powers (eB (algebraMap B Cq b)) = _
128     rw [eB.commutes b]
129   exact IsLocalization.algEquivOfAlgEquiv
130     (A := B) (R := Cq) (M := N) (genericOpenExtraAwayB M f e g)
131     (P := Cl) (T := T) (pointLocalExtraAwayB M f g) eB hmap
