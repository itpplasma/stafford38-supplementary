15 /-- The map from the finite integral chart algebra into the generic chart,
16 transported through the actual `Q_f ≃ B_f` equivalence. -/
17 def genericPointMap
18     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
19     (f : Q) (e : Localization.Away f ≃ₐ[Q]
20       Localization.Away (algebraMap Q B f)) : B →+* Localization.Away f :=
21   (e.symm.toAlgHom.toRingHom).comp
22     (algebraMap B (Localization.Away (algebraMap Q B f)))
23 
24 /-- Denominators corresponding to elements of `B` outside the selected prime. -/
25 def genericOpenDenominators
26     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
27     (M : Ideal B) [M.IsPrime] (f : Q)
28     (e : Localization.Away f ≃ₐ[Q]
29       Localization.Away (algebraMap Q B f)) : Submonoid (Localization.Away f) :=
30   M.primeCompl.map (genericPointMap f e)

32 /-- The common generic open: the actual finite-birational open `B_f`, further
33 localized at the elements that are units at `B_M`, represented on `Q_f`. -/
34 abbrev genericOpenRing
35     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
36     (M : Ideal B) [M.IsPrime] (f : Q)
37     (e : Localization.Away f ≃ₐ[Q]
38       Localization.Away (algebraMap Q B f)) : Type u :=
39   Localization (genericOpenDenominators M f e)

41 theorem formallyEtale_genericOpenRing
42     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
43     (M : Ideal B) [M.IsPrime] (f : Q)
44     (e : Localization.Away f ≃ₐ[Q]
45       Localization.Away (algebraMap Q B f)) :
46     Algebra.FormallyEtale Q (genericOpenRing M f e) := by
47   letI : Algebra.FormallyEtale Q (Localization.Away f) :=
48     Algebra.FormallyEtale.of_isLocalization (Submonoid.powers f)
49   letI : Algebra.FormallyEtale (Localization.Away f) (genericOpenRing M f e) :=
50     Algebra.FormallyEtale.of_isLocalization (genericOpenDenominators M f e)
51   exact Algebra.FormallyEtale.comp Q (Localization.Away f) (genericOpenRing M f e)

53 /-- The selected local ring maps to the common generic open. This map inverts
54 every element outside `M` by construction; it does not require `f ∉ M`. -/
55 def localPointToGenericOpen
56     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
57     (M : Ideal B) [M.IsPrime] (f : Q)
58     (e : Localization.Away f ≃ₐ[Q]
59       Localization.Away (algebraMap Q B f)) :
60     Localization.AtPrime M →+* genericOpenRing M f e := by
61   let φ : B →+* genericOpenRing M f e :=
62     (algebraMap (Localization.Away f) (genericOpenRing M f e)).comp
63       (genericPointMap f e)
64   have hunit : ∀ s : M.primeCompl, IsUnit (φ s.1) := by
65     intro s
66     have hden : genericPointMap f e s.1 ∈ genericOpenDenominators M f e :=
67       ⟨s.1, s.2, rfl⟩
68     simpa [φ] using
69       (IsLocalization.map_units (genericOpenRing M f e)
70         ⟨genericPointMap f e s.1, hden⟩)
71   exact IsLocalization.lift hunit

73 /-- The same generic arc induces the direct map from the selected local ring. -/
74 def localPointToField
75     {B : Type v} [CommRing B] {L : Type*} [Field L]
76     (M : Ideal B) [M.IsPrime] (ρ : B →+* L)
77     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b)) :
78     Localization.AtPrime M →+* L :=
79   IsLocalization.lift (M := M.primeCompl) (g := ρ) (fun s => hunitM s.1 s.2)

81 @[simp] theorem localPointToField_apply
82     {B : Type v} [CommRing B] {L : Type*} [Field L]
83     (M : Ideal B) [M.IsPrime] (ρ : B →+* L)
84     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b)) (b : B) :
85     localPointToField M ρ hunitM (algebraMap B (Localization.AtPrime M) b) = ρ b := by

88 /-- A generic field-valued point of the integral chart algebra extends across
89 this same common open. The only additional requirement is that the generic
90 finite-birational denominator remain nonzero; it may vanish at the closed
91 point `M`. -/
92 def genericArcToGenericOpen
93     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
94     {L : Type*} [Field L]
95     (M : Ideal B) [M.IsPrime] (f : Q)
96     (e : Localization.Away f ≃ₐ[Q]
97       Localization.Away (algebraMap Q B f))
98     (ρ : B →+* L)
99     (hf : IsUnit (ρ (algebraMap Q B f)))
100     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b)) :
101     genericOpenRing M f e →+* L := by
102   let ρBf : Localization.Away (algebraMap Q B f) →+* L :=
103     IsLocalization.lift (M := Submonoid.powers (algebraMap Q B f)) (g := ρ) (by
104       intro s
105       rcases (Submonoid.mem_powers_iff _ _).mp s.2 with ⟨n, hn⟩
106       rw [← hn]
107       simpa only [map_pow] using hf.pow n)
108   let ρQf : Localization.Away f →+* L := ρBf.comp e.toAlgHom.toRingHom
109   have hunit : ∀ s : genericOpenDenominators M f e, IsUnit (ρQf s.1) := by
110     intro s
111     obtain ⟨b, hbM, hsb⟩ := s.2
112     have hpoint : ρQf s.1 = ρ b := by
113       rw [← hsb]
114       simp [ρQf, ρBf, genericPointMap, IsLocalization.lift_eq]
115     rw [hpoint]
116     exact hunitM b (by simpa using hbM)
117   exact IsLocalization.lift hunit

119 @[simp] theorem localPointToGenericOpen_apply
120     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
121     (M : Ideal B) [M.IsPrime] (f : Q)
122     (e : Localization.Away f ≃ₐ[Q]
123       Localization.Away (algebraMap Q B f)) (b : B) :
124     localPointToGenericOpen M f e (algebraMap B (Localization.AtPrime M) b) =
125       (algebraMap (Localization.Away f) (genericOpenRing M f e))
126       (genericPointMap f e b) := by

143 /-- The generic-open map and the point-local map induce the same arc on the
144 selected local ring. This does not require `f ∉ M`. -/
145 theorem genericArcToGenericOpen_comp_localPointToGenericOpen
146     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
147     {L : Type*} [Field L]
148     (M : Ideal B) [M.IsPrime] (f : Q)
149     (e : Localization.Away f ≃ₐ[Q]
150       Localization.Away (algebraMap Q B f))
151     (ρ : B →+* L)
152     (hf : IsUnit (ρ (algebraMap Q B f)))
153     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b)) :
154     (genericArcToGenericOpen M f e ρ hf hunitM).comp
155       (localPointToGenericOpen M f e) = localPointToField M ρ hunitM := by
156   apply IsLocalization.ringHom_ext M.primeCompl
157   ext b
158   simp [genericArcToGenericOpen_apply, localPointToGenericOpen_apply,
159     localPointToField_apply]

161 @[simp] theorem genericArcToGenericOpen_base
162     {Q : Type u} [CommRing Q] {B : Type v} [CommRing B] [Algebra Q B]
163     {L : Type*} [Field L]
164     (M : Ideal B) [M.IsPrime] (f : Q)
165     (e : Localization.Away f ≃ₐ[Q]
166       Localization.Away (algebraMap Q B f))
167     (ρ : B →+* L)
168     (hf : IsUnit (ρ (algebraMap Q B f)))
169     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b)) (q : Q) :
170     genericArcToGenericOpen M f e ρ hf hunitM
171       ((algebraMap (Localization.Away f) (genericOpenRing M f e))
172         (algebraMap Q (Localization.Away f) q)) = ρ (algebraMap Q B q) := by
173   have hpoint : genericPointMap f e (algebraMap Q B q) =
174       algebraMap Q (Localization.Away f) q := by
175     change e.symm (algebraMap B (Localization.Away (algebraMap Q B f))
176       (algebraMap Q B q)) = algebraMap Q (Localization.Away f) q
177     rw [← IsScalarTower.algebraMap_apply Q B (Localization.Away (algebraMap Q B f))]
178     exact e.symm.commutes q
179   rw [← hpoint]
180   exact genericArcToGenericOpen_apply M f e ρ hf hunitM (algebraMap Q B q)
