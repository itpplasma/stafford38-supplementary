17 /-- After extending to Laurent series, replacing the normalized transverse
18 column by the raw derivative preserves the span with the position and
19 selected divisor-tangent columns. The correction identity is first mapped
20 from power series, and the nonzero power of `X` is used as a Laurent scalar. -/
21 theorem mapped_formalTangentColumns_span_eq_of_corrected_derivative
22     {n : ℕ} {κ : Type*} [Fintype κ]
23     (q : Fin (n + 1) → PowerSeries k)
24     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
25     (tau : Fin (n + 1) → PowerSeries k)
26     (lambda : κ → PowerSeries k) (c : ℕ)
27     (hfactor : ∀ i,
28       PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
29         (PowerSeries.X : PowerSeries k) ^ c * tau i) :
30     Submodule.span (LaurentSeries k)
31         (Set.range (fun j : FormalTangentColumn κ =>
32           fun i => algebraMap (PowerSeries k) (LaurentSeries k)
33             (formalTangentMatrix q Z tau i j))) =
34       Submodule.span (LaurentSeries k)
35         (Set.range (fun j : FormalTangentColumn κ =>
36           fun i => algebraMap (PowerSeries k) (LaurentSeries k)
37             (formalTangentMatrix q Z (fun i => PowerSeries.derivative (R := k) (q i)) i j))) := by
38   classical
39   let A := algebraMap (PowerSeries k) (LaurentSeries k)
40   let fixed : Option κ → (Fin (n + 1) → LaurentSeries k) := fun a =>
41     match a with
42     | none => fun i => A (q i)
43     | some j => fun i => A (Z i j)
44   let coeff : Option κ → LaurentSeries k := fun a =>
45     match a with
46     | none => 0
47     | some j => A (lambda j)
48   let corrected : Fin (n + 1) → LaurentSeries k := fun i => A (tau i)
49   let raw : Fin (n + 1) → LaurentSeries k :=
50     fun i => A (PowerSeries.derivative (R := k) (q i))
51   have hx : A (PowerSeries.X : PowerSeries k) ≠ 0 := by
52     simpa [A] using
53       (IsFractionRing.injective (PowerSeries k) (LaurentSeries k)).ne
54         (PowerSeries.X_ne_zero (R := k))
55   have hc : A ((PowerSeries.X : PowerSeries k) ^ c) ≠ 0 := by
56     rw [map_pow]
57     exact pow_ne_zero c hx
58   have hfactorL : raw - (∑ a, coeff a • fixed a) =
59       A ((PowerSeries.X : PowerSeries k) ^ c) • corrected := by
60     funext i
61     simp only [Pi.sub_apply, Pi.smul_apply, smul_eq_mul, Finset.sum_apply]
62     change A (PowerSeries.derivative (R := k) (q i)) -
63         (∑ a, coeff a * fixed a i) =
64       A ((PowerSeries.X : PowerSeries k) ^ c) * A (tau i)
65     have hi := congrArg A (hfactor i)
66     simpa [coeff, fixed, Matrix.mulVec, dotProduct, map_sum, map_mul,
67       mul_comm] using hi
68   have hspan :=
69     Stafford38.Geometry.CorrectedVelocitySpan.span_map_range_union_singleton_eq_of_corrected_velocity
70       (LinearMap.id) fixed coeff corrected raw
71       (A ((PowerSeries.X : PowerSeries k) ^ c)) hc hfactorL
72   have hcolsTau :
73       Set.range (fun j : FormalTangentColumn κ =>
74         fun i => A (formalTangentMatrix q Z tau i j)) =
75         Set.range fixed ∪ {corrected} := by
76     ext v
77     constructor
78     · rintro ⟨j, rfl⟩
79       cases j with
80       | inl u => exact Or.inl ⟨none, rfl⟩
81       | inr j =>
82         cases j with
83         | inl j => exact Or.inl ⟨some j, rfl⟩
84         | inr u => exact Or.inr (Set.mem_singleton _)
85     · intro hv
86       rcases hv with ⟨a, ha⟩ | hv
87       · rcases a with _ | j
88         · exact ⟨Sum.inl (), ha⟩
89         · exact ⟨Sum.inr (Sum.inl j), ha⟩
90       · rcases Set.mem_singleton_iff.mp hv with rfl
91         exact ⟨Sum.inr (Sum.inr ()), rfl⟩
92   have hcolsRaw :
93       Set.range (fun j : FormalTangentColumn κ =>
94         fun i => A (formalTangentMatrix q Z
95           (fun i => PowerSeries.derivative (R := k) (q i)) i j)) =
96         Set.range fixed ∪ {raw} := by
97     ext v
98     constructor
99     · rintro ⟨j, rfl⟩
100       cases j with
101       | inl u => exact Or.inl ⟨none, rfl⟩
102       | inr j =>
103         cases j with
104         | inl j => exact Or.inl ⟨some j, rfl⟩
105         | inr u => exact Or.inr (Set.mem_singleton _)
106     · intro hv
107       rcases hv with ⟨a, ha⟩ | hv
108       · rcases a with _ | j
109         · exact ⟨Sum.inl (), ha⟩
110         · exact ⟨Sum.inr (Sum.inl j), ha⟩
111       · rcases Set.mem_singleton_iff.mp hv with rfl
112         exact ⟨Sum.inr (Sum.inr ()), rfl⟩
113   rw [hcolsTau, hcolsRaw]
114   exact hspan

116 /-- The corrected transverse generator and raw derivative give the same
117 Laurent generic fibre: this composes the column-span replacement with the
118 canonical generic-fibre/column dictionary. -/
119 theorem genericFibre_normalizedTangentLattice_eq_span_derivative
120     {n : ℕ} {κ : Type*} [Fintype κ]
121     (q : Fin (n + 1) → PowerSeries k)
122     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
123     (tau : Fin (n + 1) → PowerSeries k)
124     (lambda : κ → PowerSeries k) (c : ℕ)
125     (hfactor : ∀ i,
126       PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
127         (PowerSeries.X : PowerSeries k) ^ c * tau i) :
128     Stafford38.Geometry.GeneralTangentLimitCriterion.genericFibre
129         (K := LaurentSeries k)
130         (Stafford38.Geometry.PaperDivisorTangent.normalizedTangentLattice q Z tau) =
131       Submodule.span (LaurentSeries k)
132         (Set.range (fun j : FormalTangentColumn κ =>
133           fun i => algebraMap (PowerSeries k) (LaurentSeries k)
134             (formalTangentMatrix q Z
135               (fun i => PowerSeries.derivative (R := k) (q i)) i j))) := by
136   let B := formalTangentMatrix q Z tau
137   have hsourceSpan :
138       Submodule.span (PowerSeries k)
139           (Set.range (fun j : FormalTangentColumn κ => fun i => B i j)) =
140         Stafford38.Geometry.PaperDivisorTangent.normalizedTangentLattice q Z tau := by
141     change Submodule.span (PowerSeries k)
142         (Set.range (fun j : FormalTangentColumn κ => fun i => B i j)) =
143       LinearMap.range B.mulVecLin
144     exact (Matrix.range_mulVecLin B).symm
145   have hgeneric :=
146     Stafford38.Geometry.GeneralTangentLimitCriterion.genericFibre_eq_span_matrixColumns
147       (K := LaurentSeries k)
148       (L := Stafford38.Geometry.PaperDivisorTangent.normalizedTangentLattice q Z tau)
149       B hsourceSpan
150   rw [hgeneric]
151   exact mapped_formalTangentColumns_span_eq_of_corrected_derivative
152     q Z tau lambda c hfactor
