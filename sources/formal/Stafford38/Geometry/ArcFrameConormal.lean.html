7 /-!
8 # From an annihilating arc frame to an affine conormal covector
9 
10 The power-series arc file proves that the uniformizer and coefficient-field
11 residue directions are tangent to the scalar-extended equation ideal.  The
12 continuous-frame file separately proves that a finite independent family with
13 the correct tangent-dimension bound spans the tangent space.  This file joins
14 those two facts and records the exact local tangent-to-conormal bridge.
15 
16 The result is deliberately affine and local.  It does not construct a
17 divisorial chart, identify an arc with a projective component, or descend a
18 `K[[t]]` chart to `k[[t]]`.  Those are still global production obligations for
19 `CanonicalAsymptoticLaurentProducer`.
20 -/

34 /-! ## The frame-span bridge -/
35 
36 /--
37 An actual annihilating power-series arc makes the complete residue frame
38 tangent.  If that finite frame is independent and its cardinality bounds the
39 tangent dimension, the equation-defined tangent space is contained in its
40 span.
41 
42 The coefficient directions are arbitrary `k`-derivations of the residue
43 field; no coefficient-field section of a geometric DVR is inferred here.
44 -/
45 theorem tangent_le_span_residueFrame_of_annihilatingArc
46     {m : ℕ} {κ : Type v} [Fintype κ]
47     (I : Ideal (MvPolynomial (Fin m) k))
48     (q : Fin m → PowerSeries K)
49     (D : κ → Derivation k K K)
50     (hq : ∀ f ∈ I,
51       MvPolynomial.eval₂ (algebraMap k (PowerSeries K)) q f = 0)
52     (hindependent : LinearIndependent K (residueFrameVector q D))
53     (hfinrank : Module.finrank K
54         (zariskiTangentSpace
55           (fun i ↦ PowerSeries.constantCoeff (q i))
56           (I.map (MvPolynomial.map (algebraMap k K)))) ≤
57       Fintype.card (FrameIndex κ)) :
58     zariskiTangentSpace
59         (fun i ↦ PowerSeries.constantCoeff (q i))
60         (I.map (MvPolynomial.map (algebraMap k K))) ≤
61       Submodule.span K (Set.range (residueFrameVector q D)) := by
62   apply tangent_le_span_residueFrame
63     (k := k) (K := K)
64     (I.map (MvPolynomial.map (algebraMap k K))) q D ?_
65     hindependent hfinrank
66   intro j
67   cases j with
68   | none =>
69       have hD :=
70         residueDerivationVector_mem_zariskiTangentSpace_of_eval₂_eq_zero
71           (k := k) (K := K) I q
72             (uniformizerDerivation (k := k) (K := K)) hq
73       change (fun i ↦ PowerSeries.constantCoeff
74           (uniformizerDerivation (k := k) (K := K) (q i))) ∈ _
75       exact hD
76   | some j =>
77       have hD :=
78         residueDerivationVector_mem_zariskiTangentSpace_of_eval₂_eq_zero
79           (k := k) (K := K) I q
80             (coefficientwiseDerivation (D j)) hq
81       change (fun i ↦ PowerSeries.constantCoeff
82           (coefficientwiseDerivation (D j) (q i))) ∈ _
83       exact hD

85 /-! ## The tangent-to-conormal bridge -/
86 
87 /--
88 If a covector annihilates every residue-frame direction of an annihilating
89 arc, then the frame-span bridge places that covector in the affine conormal
90 space of the scalar-extended equation ideal.
91 
92 The hypothesis is an explicit finite sum, so the theorem does not hide a
93 projective tangent comparison or a smoothness assertion.
94 -/
95 theorem coordinateCovector_mem_affineConormalSpace_of_annihilatingArcFrame
96     {m : ℕ} {κ : Type v} [Fintype κ]
97     (I : Ideal (MvPolynomial (Fin m) k))
98     (q : Fin m → PowerSeries K)
99     (D : κ → Derivation k K K)
100     (xi : Fin m → K)
101     (hq : ∀ f ∈ I,
102       MvPolynomial.eval₂ (algebraMap k (PowerSeries K)) q f = 0)
103     (hframe : ∀ j,
104       ∑ i, xi i * residueFrameVector q D j i = 0)
105     (hindependent : LinearIndependent K (residueFrameVector q D))
106     (hfinrank : Module.finrank K
107         (zariskiTangentSpace
108           (fun i ↦ PowerSeries.constantCoeff (q i))
109           (I.map (MvPolynomial.map (algebraMap k K)))) ≤
110       Fintype.card (FrameIndex κ)) :
111     coordinateCovector xi ∈
112       affineConormalSpace
113         (fun i ↦ PowerSeries.constantCoeff (q i))
114         (I.map (MvPolynomial.map (algebraMap k K))) := by
115   have htangent := tangent_le_span_residueFrame_of_annihilatingArc
116     (k := k) (K := K) I q D hq hindependent hfinrank
117   rw [affineConormalSpace, Submodule.mem_dualAnnihilator]
118   intro v hv
119   have hspan : ∀ w,
120       w ∈ Submodule.span K (Set.range (residueFrameVector q D)) →
121         coordinateCovector xi w = 0 := by
122     intro w hw
123     induction hw using Submodule.span_induction with
124     | mem w hw =>
125         rcases hw with ⟨j, rfl⟩
126         simpa [coordinateCovector_apply] using hframe j
127     | zero =>
128         simp
129     | add w z hw hz ihw ihz =>
130         rw [map_add, ihw, ihz, add_zero]
131     | smul a w hw ih =>
132         rw [map_smul, ih, smul_zero]
133   exact hspan v (htangent hv)
