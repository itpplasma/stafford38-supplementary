10 /-!
11 # Laurent specialization of projected conormal directions
12 
13 A Laurent-valued phase point need not specialize to a finite phase point: its
14 base coordinates may have poles.  Its fibre coordinates can nevertheless be
15 regular power series.  This file records exactly the consequence that survives:
16 the residue of those fibre coordinates belongs to the ground-coefficient
17 vanishing hull of the projected fibre image.
18 
19 The construction is then specialized to the scalar-extended
20 `equationConormalLocus`.  The final theorem is the corresponding direct
21 polynomial obstruction for a ground-field symbol using only fibre variables.
22 
23 No full phase-space closure, compatibility of closure with base change,
24 normalization chart, asymptotic-conormal producer, or Gabber theorem is proved
25 here.
26 -/

40 /-- Forget the base coordinate of a set of phase points. -/
41 def fibreImage (S : Set (PhaseVar n → K)) : Set (Fin n → K) :=
42   (fun q i ↦ q (Sum.inr i)) '' S
43 
44 /-- The ground-field affine vanishing hull of the fibre directions of a set
45 of extension-valued phase points.  Both projection and extension-valued
46 vanishing are explicit; this is not a full phase-space closure. -/
47 def extensionFibreClosure (S : Set (PhaseVar n → K)) : Set (Fin n → k) :=
48   MvPolynomial.zeroLocus k
49     (extensionValuedVanishingIdeal (k := k) (K := K) (fibreImage S))

51 /-- Evaluating a ground-field polynomial on power-series coordinates and then
52 embedding in Laurent series is the same as evaluating directly on the
53 embedded Laurent coordinates. -/
54 theorem algebraMap_eval_powerSeries_eq_eval₂_laurent
55     (P : MvPolynomial (Fin n) k) (xi : Fin n → PowerSeries k) :
56     algebraMap (PowerSeries k) (LaurentSeries k)
57         (MvPolynomial.eval xi (MvPolynomial.map (PowerSeries.C) P)) =
58       MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
59         (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) P := by
60   have hcoeff :
61       (algebraMap (PowerSeries k) (LaurentSeries k)).comp
62           (PowerSeries.C) =
63         algebraMap k (LaurentSeries k) := by
64     ext r
65     simp [LaurentSeries.algebraMap_apply]
66   rw [MvPolynomial.eval_map, MvPolynomial.eval₂_comp_left, hcoeff]
67   rfl

69 /-- A Laurent-valued phase point whose fibre coordinates are regular power
70 series specializes, after forgetting its possibly singular base coordinate,
71 to the ground-field extension-valued vanishing hull of the projected fibre
72 image. -/
73 theorem residue_mem_extensionFibreClosure_of_laurent_generic
74     (S : Set (PhaseVar n → LaurentSeries k))
75     (y : Fin n → LaurentSeries k)
76     (xi : Fin n → PowerSeries k)
77     (hgeneric :
78       Sum.elim y
79           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈ S) :
80     residueColumn xi ∈
81       extensionFibreClosure (k := k) (K := LaurentSeries k) S := by
82   rw [extensionFibreClosure, MvPolynomial.mem_zeroLocus_iff]
83   intro P hP
84   have hzeroLaurent :
85       MvPolynomial.eval₂ (algebraMap k (LaurentSeries k))
86           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) P = 0 := by
87     apply (mem_extensionValuedVanishingIdeal_iff
88       (k := k) (K := LaurentSeries k) (fibreImage S) P).mp hP
89     exact ⟨Sum.elim y
90       (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)),
91       hgeneric, rfl⟩
92   have hzeroPowerSeries :
93       MvPolynomial.eval xi (MvPolynomial.map (PowerSeries.C) P) = 0 := by
94     apply HahnSeries.ofPowerSeries_injective (Γ := ℤ)
95     change algebraMap (PowerSeries k) (LaurentSeries k)
96         (MvPolynomial.eval xi (MvPolynomial.map (PowerSeries.C) P)) =
97       algebraMap (PowerSeries k) (LaurentSeries k) 0
98     rw [map_zero, algebraMap_eval_powerSeries_eq_eval₂_laurent]
99     exact hzeroLaurent
100   exact residue_eval_eq_zero_of_eval_map_eq_zero P xi hzeroPowerSeries

102 /-- The projected Laurent-direction hull of the scalar-extended
103 equation-defined conormal locus. -/
104 def laurentEquationConormalDirectionClosure
105     (I : Ideal (MvPolynomial (Fin n) k)) : Set (Fin n → k) :=
106   extensionFibreClosure (k := k) (K := LaurentSeries k)
107     (equationConormalLocus
108       (I.map
109         (scalarPolynomialMap
110           (k := k) (K := LaurentSeries k) (Fin n))))

115     (I : Ideal (MvPolynomial (Fin n) k))
116     (y : Fin n → LaurentSeries k)
117     (xi : Fin n → PowerSeries k)

133 /-- Embed a ground-field polynomial in fibre variables as a fibre-only phase
134 symbol. -/
135 def fibreLift (P : MvPolynomial (Fin n) k) : SymbolRing k n :=
136   MvPolynomial.rename Sum.inr P
137 
138 /-- Evaluation of a fibre-only phase symbol ignores the base coordinate. -/
139 theorem eval₂_fibreLift
140     (P : MvPolynomial (Fin n) k)
141     (y xi : Fin n → K) :
142     MvPolynomial.eval₂ (algebraMap k K) (Sum.elim y xi) (fibreLift P) =
143       MvPolynomial.eval₂ (algebraMap k K) xi P := by
144   rw [fibreLift, MvPolynomial.eval₂_rename]
145   rfl

147 /-- Direct projected-direction contradiction.  A fibre-only ground-field
148 symbol cannot both evaluate to one on the residue axis and vanish on a
149 Laurent generic conormal direction whose fibre coordinates have that residue.
150 
151 The conormal hypothesis is used only to place the generic fibre in the
152 projected set; no specialization of its base coordinate is asserted. -/
153 theorem false_of_fibreOnly_symbol_one_on_residue_and_vanishes_on_laurentConormal
154     (I : Ideal (MvPolynomial (Fin n) k))
155     (P : MvPolynomial (Fin n) k)
156     (axis : Fin n → k)
157     (y : Fin n → LaurentSeries k)
158     (xi : Fin n → PowerSeries k)
159     (hgeneric :
160       Sum.elim y
161           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈
162         equationConormalLocus
163           (I.map
164             (scalarPolynomialMap
165               (k := k) (K := LaurentSeries k) (Fin n))))
166     (hresidue : residueColumn xi = axis)
167     (hvanishes :
168       ∀ q ∈ equationConormalLocus
169           (I.map
170             (scalarPolynomialMap
171               (k := k) (K := LaurentSeries k) (Fin n))),
172         MvPolynomial.eval₂ (algebraMap k (LaurentSeries k)) q
173           (fibreLift P) = 0)
174     (haxis : MvPolynomial.eval axis P = 1) : False := by
175   let S : Set (PhaseVar n → LaurentSeries k) :=
176     equationConormalLocus
177       (I.map
178         (scalarPolynomialMap
179           (k := k) (K := LaurentSeries k) (Fin n)))
180   have hP :
181       P ∈ extensionValuedVanishingIdeal
182         (k := k) (K := LaurentSeries k) (fibreImage S) := by
183     rw [mem_extensionValuedVanishingIdeal_iff]
184     intro v hv
185     rcases hv with ⟨q, hq, rfl⟩
186     rw [← eval₂_fibreLift P (fun i ↦ q (Sum.inl i))]
187     have hsplit :
188         Sum.elim (fun i ↦ q (Sum.inl i)) (fun i ↦ q (Sum.inr i)) = q := by
189       funext i
190       rcases i with i | i <;> rfl
191     rw [hsplit]
192     exact hvanishes q hq
193   have hresidueInHull :
194       residueColumn xi ∈
195         extensionFibreClosure (k := k) (K := LaurentSeries k) S :=
196     residue_mem_extensionFibreClosure_of_laurent_generic
197       (S := S) y xi hgeneric
198   rw [extensionFibreClosure, MvPolynomial.mem_zeroLocus_iff] at hresidueInHull
199   have hzeroResidue := hresidueInHull P hP
200   rw [hresidue, MvPolynomial.aeval_eq_eval, haxis] at hzeroResidue
201   exact one_ne_zero hzeroResidue

203 /-! The declarations below make the scope boundary executable: projection
204 forgets arbitrary base coordinates, whereas membership in a full phase set
205 still retains them.  In particular no converse reconstructing a finite phase
206 specialization is available from `fibreImage`. -/
207 
208 theorem fibre_mem_fibreImage_of_phase_mem
209     (S : Set (PhaseVar n → K)) (q : PhaseVar n → K) (hq : q ∈ S) :
210     (fun i ↦ q (Sum.inr i)) ∈ fibreImage S :=
211   ⟨q, hq, rfl⟩
