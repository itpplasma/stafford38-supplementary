9 /-!
10 # Hamiltonian flows in the paper's proof
11 
12 The paper moves a common zero along the Hamiltonian flow of a base equation.
13 This file proves that statement directly from the base-relative Poisson
14 condition, by finite Taylor expansion on the affine fibre line.  The argument
15 does not assume translation stability as an axiom or as a prior theorem.
16 -/

163 /-- The paper's Hamiltonian map: base fixed and fibre translated by `t df_y`.
164 This is the arbitrary-start specialization of the shared affine fibre-point map. -/
165 abbrev paperHamiltonianFlowPoint (z : PhaseVar n → k)
166     (f : MvPolynomial (Fin n) k) (t : k) : PhaseVar n → k :=
167   Stafford38.Geometry.CoisotropicTranslation.affineFibreTranslatePoint
168     (fun i => z (Sum.inl i)) (fun i => z (Sum.inr i))
169     (differentialAt (fun j => z (Sum.inl j)) f) t

178 /-- Apply a finite family of the paper's commuting Hamiltonian translations.
179 Since each flow fixes the base coordinates, this is the canonical affine
180 fibre translation whose direction is the sum of the individual directions. -/
181 abbrev paperHamiltonianFlowsPoint {ι : Type*} [DecidableEq ι]
182     (z : PhaseVar n → k) (f : ι → MvPolynomial (Fin n) k)
183     (t : ι → k) (s : Finset ι) : PhaseVar n → k :=
184   affineFibreTranslatePoint
185     (fun i => z (Sum.inl i))
186     (fun i => z (Sum.inr i))
187     (fun i => s.sum (fun j =>
188       t j * differentialAt (fun l => z (Sum.inl l)) (f j) i))
189     1

217 /-- One Hamiltonian flow preserves a common zero of a base-relatively Poisson
218 ideal. The proof is finite Taylor on the affine fibre line through `z`. -/
219 theorem paperHamiltonianFlow_preserves_commonZero
220     [CharZero k]
221     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
222     (z : PhaseVar n → k)
223     (hzero : ∀ g ∈ J, MvPolynomial.eval z g = 0)
224     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
225     (g : SymbolRing k n) (hg : g ∈ J) (t : k) :
226     MvPolynomial.eval (paperHamiltonianFlowPoint z f t) g = 0 := by
227   let y : Fin n → k := fun i => z (Sum.inl i)
228   let xi : Fin n → k := fun i => z (Sum.inr i)
229   have hz : Sum.elim y xi = z := by
230     funext i
231     rcases i with i | i <;> simp [y, xi]
232   have hline : affineFibreLinePolynomial y xi (differentialAt y f) g = 0 := by
233     apply polynomial_eq_zero_of_eval_iterate_derivative_zero
234     intro m
235     rw [iterate_derivative_affineFibreLinePolynomial,
236       eval_zero_affineFibreLinePolynomial]
237     rw [hz]
238     exact hzero _ (hamiltonIter_mem_of_isBaseRelativePoisson J hJ f hf g hg m)
239   have heval :
240       MvPolynomial.eval (paperHamiltonianFlowPoint z f t) g =
241         Polynomial.eval t (affineFibreLinePolynomial y xi (differentialAt y f) g) := by
242     rw [eval_affineFibreLinePolynomial]
243     apply MvPolynomial.eval₂_congr
244     intro i c hi hc
245     rcases i with i | i <;> simp [paperHamiltonianFlowPoint, affineFibreTranslatePoint, y, xi]
246   rw [heval, hline]
247   exact Polynomial.eval_zero

249 /-- Any finite composition preserves a common zero, from an arbitrary starting
250 covector. -/
251 theorem paperHamiltonianFlows_preserve_commonZero
252     [CharZero k] {ι : Type*} [DecidableEq ι]
253     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
254     (z : PhaseVar n → k)
255     (hzero : ∀ g ∈ J, MvPolynomial.eval z g = 0)
256     (f : ι → MvPolynomial (Fin n) k)
257     (hf : ∀ i, baseLift (f i) ∈ J) (t : ι → k) (s : Finset ι) :
258     ∀ g ∈ J, MvPolynomial.eval (paperHamiltonianFlowsPoint z f t s) g = 0 := by
259   induction s using Finset.induction_on with
260   | empty =>
261       intro g hg
262       have hpoint : paperHamiltonianFlowsPoint z f t ∅ = z := by
263         funext q
264         rcases q with q | q <;> simp [paperHamiltonianFlowsPoint]
265       rw [hpoint]
266       exact hzero g hg
267   | @insert i s hi ih =>
268       intro g hg
269       rw [← paperHamiltonianFlowPoint_paperHamiltonianFlowsPoint_insert
270         z f t s i hi]
271       exact paperHamiltonianFlow_preserves_commonZero J hJ
272         (paperHamiltonianFlowsPoint z f t s) ih (f i) (hf i) g hg (t i)

274 /-- The affine conormal consumer using the paper's finite successive-flow
275 argument. Every covector in the equation-defined conormal space is a finite
276 linear combination of equation differentials. -/
277 theorem affineConormal_coordinatePoint_isCommonZero
278     [CharZero k]
279     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
280     (y : Fin n → k)
281     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
282     (I : Ideal (MvPolynomial (Fin n) k))
283     (hlift : ∀ f : I, baseLift f.1 ∈ J)
284     (xi : Fin n → k)
285     (hxi : coordinateCovector xi ∈ affineConormalSpace y I) :
286     ∀ g ∈ J, MvPolynomial.eval (Sum.elim y xi) g = 0 := by
287   classical
288   obtain ⟨c, hc⟩ :=
289     coordinate_mem_affineConormalSpace_exists_finsupp y xi I hxi
290   let ι := {f : I // f ∈ c.support}
291   let equations : ι → MvPolynomial (Fin n) k := fun f => f.1.1
292   let parameters : ι → k := fun f => c f.1
293   have hflows :
294       paperHamiltonianFlowsPoint (zeroSectionPoint y) equations parameters
295         Finset.univ = differentialCombinationPoint y parameters equations := by
296     funext q
297     rcases q with i | i
298     · simp [zeroSectionPoint, paperHamiltonianFlowsPoint,
299         differentialCombinationPoint]
300     · simp [paperHamiltonianFlowsPoint, differentialCombinationPoint,
301         parameters, equations, zeroSectionPoint, differentialAt_baseLinearCombination]
302   have hpoint : differentialCombinationPoint y parameters equations = Sum.elim y xi := by
303     funext q
304     rcases q with i | i
305     · rfl
306     · rw [differentialCombinationPoint_inr]
307       simp only [parameters, equations, Sum.elim_inr]
308       rw [hc i]
309       exact (Finset.sum_subtype c.support (by simp)
310         (fun f => c f * differentialAt y f.1 i)).symm
311   rw [← hpoint, ← hflows]
312   exact paperHamiltonianFlows_preserve_commonZero J hJ (zeroSectionPoint y)
313     hzero equations (fun f => hlift f.1) parameters Finset.univ
