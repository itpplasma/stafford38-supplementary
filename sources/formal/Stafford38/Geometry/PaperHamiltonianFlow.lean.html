7 /-!
8 # Hamiltonian flows in the paper's proof
9 
10 The paper moves a common zero along the Hamiltonian flow of a base equation.
11 This file proves that statement directly from the base-relative Poisson
12 condition, by finite Taylor expansion on the affine fibre line.  The argument
13 does not assume translation stability as an axiom or as a prior theorem.
14 -/

161 /-- The paper's Hamiltonian map: base fixed and fibre translated by `t df_y`.
162 This is the arbitrary-start specialization of the shared affine fibre-point map. -/
163 abbrev paperHamiltonianFlowPoint (z : PhaseVar n → k)
164     (f : MvPolynomial (Fin n) k) (t : k) : PhaseVar n → k :=
165   Stafford38.Geometry.CoisotropicTranslation.affineFibreTranslatePoint
166     (fun i => z (Sum.inl i)) (fun i => z (Sum.inr i))
167     (differentialAt (fun j => z (Sum.inl j)) f) t

176 /-- Apply a finite family of the paper's commuting Hamiltonian translations.
177 Since each flow fixes the base coordinates, this is the canonical affine
178 fibre translation whose direction is the sum of the individual directions. -/
179 abbrev paperHamiltonianFlowsPoint {ι : Type*} [DecidableEq ι]
180     (z : PhaseVar n → k) (f : ι → MvPolynomial (Fin n) k)
181     (t : ι → k) (s : Finset ι) : PhaseVar n → k :=
182   affineFibreTranslatePoint
183     (fun i => z (Sum.inl i))
184     (fun i => z (Sum.inr i))
185     (fun i => s.sum (fun j =>
186       t j * differentialAt (fun l => z (Sum.inl l)) (f j) i))
187     1

215 /-- One Hamiltonian flow preserves a common zero of a base-relatively Poisson
216 ideal. The proof is finite Taylor on the affine fibre line through `z`. -/
217 theorem paperHamiltonianFlow_preserves_commonZero
218     [CharZero k]
219     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
220     (z : PhaseVar n → k)
221     (hzero : ∀ g ∈ J, MvPolynomial.eval z g = 0)
222     (f : MvPolynomial (Fin n) k) (hf : baseLift f ∈ J)
223     (g : SymbolRing k n) (hg : g ∈ J) (t : k) :
224     MvPolynomial.eval (paperHamiltonianFlowPoint z f t) g = 0 := by
225   let y : Fin n → k := fun i => z (Sum.inl i)
226   let xi : Fin n → k := fun i => z (Sum.inr i)
227   have hz : Sum.elim y xi = z := by
228     funext i
229     rcases i with i | i <;> simp [y, xi]
230   have hline : affineFibreLinePolynomial y xi (differentialAt y f) g = 0 := by
231     apply polynomial_eq_zero_of_eval_iterate_derivative_zero
232     intro m
233     rw [iterate_derivative_affineFibreLinePolynomial,
234       eval_zero_affineFibreLinePolynomial]
235     rw [hz]
236     exact hzero _ (hamiltonIter_mem_of_isBaseRelativePoisson J hJ f hf g hg m)
237   have heval :
238       MvPolynomial.eval (paperHamiltonianFlowPoint z f t) g =
239         Polynomial.eval t (affineFibreLinePolynomial y xi (differentialAt y f) g) := by
240     rw [eval_affineFibreLinePolynomial]
241     apply MvPolynomial.eval₂_congr
242     intro i c hi hc
243     rcases i with i | i <;> simp [paperHamiltonianFlowPoint, affineFibreTranslatePoint, y, xi]
244   rw [heval, hline]
245   exact Polynomial.eval_zero

247 /-- Any finite composition preserves a common zero, from an arbitrary starting
248 covector. -/
249 theorem paperHamiltonianFlows_preserve_commonZero
250     [CharZero k] {ι : Type*} [DecidableEq ι]
251     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
252     (z : PhaseVar n → k)
253     (hzero : ∀ g ∈ J, MvPolynomial.eval z g = 0)
254     (f : ι → MvPolynomial (Fin n) k)
255     (hf : ∀ i, baseLift (f i) ∈ J) (t : ι → k) (s : Finset ι) :
256     ∀ g ∈ J, MvPolynomial.eval (paperHamiltonianFlowsPoint z f t s) g = 0 := by
257   induction s using Finset.induction_on with
258   | empty =>
259       intro g hg
260       have hpoint : paperHamiltonianFlowsPoint z f t ∅ = z := by
261         funext q
262         rcases q with q | q <;> simp [paperHamiltonianFlowsPoint]
263       rw [hpoint]
264       exact hzero g hg
265   | @insert i s hi ih =>
266       intro g hg
267       rw [← paperHamiltonianFlowPoint_paperHamiltonianFlowsPoint_insert
268         z f t s i hi]
269       exact paperHamiltonianFlow_preserves_commonZero J hJ
270         (paperHamiltonianFlowsPoint z f t s) ih (f i) (hf i) g hg (t i)

272 /-- The affine conormal consumer using the paper's finite successive-flow
273 argument. Every covector in the equation-defined conormal space is a finite
274 linear combination of equation differentials. -/
275 theorem affineConormal_coordinatePoint_isCommonZero
276     [CharZero k]
277     (J : Ideal (SymbolRing k n)) (hJ : IsBaseRelativePoisson J)
278     (y : Fin n → k)
279     (hzero : ∀ g ∈ J, MvPolynomial.eval (zeroSectionPoint y) g = 0)
280     (I : Ideal (MvPolynomial (Fin n) k))
281     (hlift : ∀ f : I, baseLift f.1 ∈ J)
282     (xi : Fin n → k)
283     (hxi : coordinateCovector xi ∈ affineConormalSpace y I) :
284     ∀ g ∈ J, MvPolynomial.eval (Sum.elim y xi) g = 0 := by
285   classical
286   obtain ⟨c, hc⟩ :=
287     coordinate_mem_affineConormalSpace_exists_finsupp y xi I hxi
288   let ι := {f : I // f ∈ c.support}
289   let equations : ι → MvPolynomial (Fin n) k := fun f => f.1.1
290   let parameters : ι → k := fun f => c f.1
291   have hflows :
292       paperHamiltonianFlowsPoint (zeroSectionPoint y) equations parameters
293         Finset.univ = differentialCombinationPoint y parameters equations := by
294     funext q
295     rcases q with i | i
296     · simp [zeroSectionPoint, paperHamiltonianFlowsPoint,
297         differentialCombinationPoint]
298     · simp [paperHamiltonianFlowsPoint, differentialCombinationPoint,
299         parameters, equations, zeroSectionPoint, differentialAt_baseLinearCombination]
300   have hpoint : differentialCombinationPoint y parameters equations = Sum.elim y xi := by
301     funext q
302     rcases q with i | i
303     · rfl
304     · rw [differentialCombinationPoint_inr]
305       simp only [parameters, equations, Sum.elim_inr]
306       rw [hc i]
307       exact (Finset.sum_subtype c.support (by simp)
308         (fun f => c f * differentialAt y f.1 i)).symm
309   rw [← hpoint, ← hflows]
310   exact paperHamiltonianFlows_preserve_commonZero J hJ (zeroSectionPoint y)
311     hzero equations (fun f => hlift f.1) parameters Finset.univ
