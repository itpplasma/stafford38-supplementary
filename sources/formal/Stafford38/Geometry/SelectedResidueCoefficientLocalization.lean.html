25 /-- Restrict the retained ground map to the actual affine chart algebra. -/
26 def chartAlgebraToValuation
27     (Q : Subalgebra k F) (V : ValuationSubring F)
28     [IsLocalRing V.toSubring]
29     [Algebra k V.toSubring]
30     (hground : ∀ c : k,
31       (algebraMap k V.toSubring c : F) = algebraMap k F c)
32     (hQV : Q.toSubring ≤ V.toSubring) : Q →ₐ[k] V.toSubring := {
33   toRingHom := RingHom.codRestrict Q.val.toRingHom V.toSubring (fun q => hQV q.property)
34   commutes' := by
35     intro c
36     apply Subtype.ext
37     simpa using (hground c).symm

40 /-- The coefficient subalgebra generated by the selected chart-coordinate
41 lifts, sent to the same retained residue field by reduction. -/
42 noncomputable def selectedCoefficientResidueMap
43     (Q : Subalgebra k F) (V : ValuationSubring F)
44     [IsLocalRing V.toSubring]
45     [Algebra k V.toSubring]
46     (hground : ∀ c : k,
47       (algebraMap k V.toSubring c : F) = algebraMap k F c)
48     [Algebra k (ResidueField V.toSubring)]
49     [IsScalarTower k V.toSubring (ResidueField V.toSubring)]
50     (hQV : Q.toSubring ≤ V.toSubring) (t : Set (ResidueField V.toSubring))
51     (x : t → Q) :
52     let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
53     A →ₐ[k] ResidueField V.toSubring := by
54   let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
55   let ρ : V.toSubring →ₐ[k] ResidueField V.toSubring := {
56     toRingHom := residue V.toSubring
57     commutes' := by
58       intro c
59       rw [← ResidueField.algebraMap_eq]
60       exact (IsScalarTower.algebraMap_apply k V.toSubring
61         (ResidueField V.toSubring) c).symm
62   }
63   exact ρ.comp ((chartAlgebraToValuation Q V hground hQV).comp (Subalgebra.val A))

65 /-- Lifts of a transcendence basis remain algebraically independent in the
66 chart algebra, since a polynomial relation would reduce to the same residue
67 relation. -/
68 theorem selected_lifts_algebraicallyIndependent
69     (Q : Subalgebra k F) (V : ValuationSubring F)
70     [IsLocalRing V.toSubring]
71     [Algebra k V.toSubring]
72     (hground : ∀ c : k,
73       (algebraMap k V.toSubring c : F) = algebraMap k F c)
74     [Algebra k (ResidueField V.toSubring)]
75     [IsScalarTower k V.toSubring (ResidueField V.toSubring)]
76     (hQV : Q.toSubring ≤ V.toSubring)
77     {t : Set (ResidueField V.toSubring)}
78     (htb : IsTranscendenceBasis k ((↑) : t → ResidueField V.toSubring))
79     (x : t → Q)
80     (hx : ∀ z, residue V.toSubring
81       ⟨(x z : F), hQV (x z).property⟩ = (z : ResidueField V.toSubring)) :
82     AlgebraicIndependent k (fun z : t => x z) := by
83   have hAIκ : AlgebraicIndependent k ((↑) : t → ResidueField V.toSubring) := htb.1
84   let xV : t → V.toSubring := fun z => ⟨(x z : F), hQV (x z).property⟩
85   have hxV : ∀ z, residue V.toSubring (xV z) = (z : ResidueField V.toSubring) := hx
86   letI : Algebra V.toSubring F := V.toSubring.subtype.toAlgebra
87   letI : IsScalarTower k V.toSubring F :=
88     IsScalarTower.of_algebraMap_eq (fun c => (hground c).symm)
89   have hinjVF : Function.Injective (algebraMap V.toSubring F) := by
90     change Function.Injective (fun a : V.toSubring => (a : F))
91     exact Subtype.val_injective
92   have hAI_F :=
93     Stafford38.Geometry.ResidueLiftIndependence.algebraicIndependent_of_residue
94       (V := V.toSubring) (K := F) hinjVF
95       xV (by
96         convert hAIκ using 1
97         funext z
98         exact hxV z)
99   have hAIx : AlgebraicIndependent k (fun z : t => (x z : F)) := by
100     convert hAI_F using 1
101     funext z
102     rfl
103   rw [algebraicIndependent_iff_injective_aeval]
104   intro p q hpq
105   have hmap (r : MvPolynomial t k) :
106       algebraMap Q F (MvPolynomial.aeval x r) =
107         MvPolynomial.aeval (fun z : t => (x z : F)) r := by
108     rw [MvPolynomial.map_aeval, MvPolynomial.aeval_eq_eval₂Hom]
109     have hcoeff : (algebraMap Q F).comp (algebraMap k Q) = algebraMap k F := by
110       ext c
111       rfl
112     rw [hcoeff]
113     rfl
114   apply (algebraicIndependent_iff_injective_aeval.mp hAIx)
115   calc
116     MvPolynomial.aeval (fun z : t => (x z : F)) p = algebraMap Q F (MvPolynomial.aeval x p) :=
117       (hmap p).symm
118     _ = algebraMap Q F (MvPolynomial.aeval x q) := congrArg (algebraMap Q F) hpq
119     _ = MvPolynomial.aeval (fun z : t => (x z : F)) q := hmap q

121 /-- Reduction is injective on the subalgebra generated by the chosen lifts. -/
122 theorem selectedCoefficientResidueMap_injective
123     (Q : Subalgebra k F) (V : ValuationSubring F)
124     [IsLocalRing V.toSubring]
125     [Algebra k V.toSubring]
126     [Algebra k (ResidueField V.toSubring)]
127     [IsScalarTower k V.toSubring (ResidueField V.toSubring)]
128     (hQV : Q.toSubring ≤ V.toSubring)
129     (hground : ∀ c : k,
130       (algebraMap k V.toSubring c : F) = algebraMap k F c)
131     {t : Set (ResidueField V.toSubring)}
132     (htb : IsTranscendenceBasis k ((↑) : t → ResidueField V.toSubring))
133     (x : t → Q)
134     (hx : ∀ z, residue V.toSubring
135       ⟨(x z : F), hQV (x z).property⟩ = (z : ResidueField V.toSubring)) :
136     let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
137     Function.Injective (selectedCoefficientResidueMap Q V hground hQV t x) ∧
138       (selectedCoefficientResidueMap Q V hground hQV t x).range =
139         Algebra.adjoin k (Set.range (fun z : t => (z : ResidueField V.toSubring))) := by
140   classical
141   let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
142   let ψ : A →ₐ[k] ResidueField V.toSubring :=
143     selectedCoefficientResidueMap Q V hground hQV t x
144   let evQ : MvPolynomial t k →ₐ[k] Q := MvPolynomial.aeval x
145   have hevQ : Function.Injective evQ := by
146     rw [← algebraicIndependent_iff_injective_aeval]
147     exact selected_lifts_algebraicallyIndependent Q V hground hQV htb x hx
148   have hrange : evQ.range = A := (Algebra.adjoin_range_eq_range_aeval k x).symm
149   let eA : MvPolynomial t k ≃ₐ[k] A :=
150     (AlgEquiv.ofInjective evQ hevQ).trans (Subalgebra.equivOfEq evQ.range A hrange)
151   have heA (p : MvPolynomial t k) : (eA p : Q) = evQ p := by
152     simp [eA]
153   have hcomp : ψ.comp eA.toAlgHom =
154       MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring)) := by
155     ext z
156     change ψ (eA (MvPolynomial.X z)) =
157       MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring))
158         (MvPolynomial.X z)
159     rw [MvPolynomial.aeval_X]
160     change residue V.toSubring
161       (chartAlgebraToValuation Q V hground hQV
162         ((eA (MvPolynomial.X z) : A) : Q)) =
163         (z : ResidueField V.toSubring)
164     have hmapx : chartAlgebraToValuation Q V hground hQV (x z) =
165         (⟨(x z : F), hQV (x z).property⟩ : V.toSubring) := by
166       apply Subtype.ext
167       rfl
168     rw [heA, MvPolynomial.aeval_X]
169     change residue V.toSubring
170       (chartAlgebraToValuation Q V hground hQV (x z)) =
171         (z : ResidueField V.toSubring)
172     rw [hmapx]
173     exact hx z
174   have hevalκ : Function.Injective
175       (MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring)) :
176         MvPolynomial t k →ₐ[k] ResidueField V.toSubring) :=
177     (algebraicIndependent_iff_injective_aeval.mpr htb.1)
178   constructor
179   · change Function.Injective ψ
180     intro a b hab
181     obtain ⟨p, hp⟩ := eA.surjective a
182     obtain ⟨q, hq⟩ := eA.surjective b
183     have hpq : MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring)) p =
184         MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring)) q := by
185       calc
186         _ = ψ (eA p) := (AlgHom.congr_fun hcomp p).symm
187         _ = ψ a := congrArg ψ hp
188         _ = ψ b := hab
189         _ = ψ (eA q) := congrArg ψ hq.symm
190         _ = _ := AlgHom.congr_fun hcomp q
191     have hpq' : p = q := hevalκ hpq
192     calc
193       a = eA p := hp.symm
194       _ = eA q := congrArg eA hpq'
195       _ = b := hq
196   · have hrangeMaps : ψ.range = (ψ.comp eA.toAlgHom).range := by
197       apply Subalgebra.ext
198       intro y
199       constructor
200       · rintro ⟨a, rfl⟩
201         obtain ⟨p, rfl⟩ := eA.surjective a
202         exact ⟨p, rfl⟩
203       · rintro ⟨p, hp⟩
204         exact ⟨eA p, hp⟩
205     calc
206       ψ.range = (ψ.comp eA.toAlgHom).range := hrangeMaps
207       _ = (MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring))).range := by
208         rw [hcomp]
209       _ = Algebra.adjoin k (Set.range (fun z : t => (z : ResidueField V.toSubring))) :=
210         (Algebra.adjoin_range_eq_range_aeval k _).symm

212 /-- The polynomial fraction field on the selected residue coordinates maps
213 into the retained valuation ring. Its residue is the canonical embedding of
214 the field generated by those coordinates. -/
215 theorem exists_selected_fraction_field_maps
216     (Q : Subalgebra k F) (V : ValuationSubring F)
217     [IsLocalRing V.toSubring]
218     [Algebra k V.toSubring]
219     (hground : ∀ c : k,
220       (algebraMap k V.toSubring c : F) = algebraMap k F c)
221     [Algebra k (ResidueField V.toSubring)]
222     [IsScalarTower k V.toSubring (ResidueField V.toSubring)]
223     (hQV : Q.toSubring ≤ V.toSubring)
224     {t : Set (ResidueField V.toSubring)}
225     (htb : IsTranscendenceBasis k ((↑) : t → ResidueField V.toSubring))
226     (x : t → Q)
227     (hx : ∀ z, residue V.toSubring
228       ⟨(x z : F), hQV (x z).property⟩ = (z : ResidueField V.toSubring)) :
229     let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
230     let E := FractionRing (MvPolynomial t k)
231     ∃ (eA : MvPolynomial t k ≃ₐ[k] A)
232       (fA : A →ₐ[k] V.toSubring)
233       (iA : A →ₐ[k] E)
234       (g : E →ₐ[k] ResidueField V.toSubring)
235       (fE : E →ₐ[k] V.toSubring),
236       Function.Injective g ∧
237       g.fieldRange = IntermediateField.adjoin k
238         (Set.range (fun z : t => (z : ResidueField V.toSubring))) ∧
239       (∀ z : t, ((eA (MvPolynomial.X z) : A) : Q) = x z) ∧
240       (∀ z : t, fA (eA (MvPolynomial.X z)) =
241         (⟨(x z : F), hQV (x z).property⟩ : V.toSubring)) ∧
242       (∀ z : t, g (algebraMap (MvPolynomial t k) E (MvPolynomial.X z)) =
243         (z : ResidueField V.toSubring)) ∧
244       (∀ p : MvPolynomial t k,
245         iA (eA p) = algebraMap (MvPolynomial t k) E p) ∧
246       (residue V.toSubring).comp fE.toRingHom = g.toRingHom ∧
247       fE.toRingHom.comp iA.toRingHom = fA.toRingHom := by
248   classical
249   let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
250   let P := MvPolynomial t k
251   let E := FractionRing P
252   letI : IsDomain A := inferInstance
253   letI : Field E := FractionRing.field P
254   let qV : Q →ₐ[k] V.toSubring := chartAlgebraToValuation Q V hground hQV
255   let evQ : P →ₐ[k] Q := MvPolynomial.aeval x
256   have hevQ : Function.Injective evQ := by
257     rw [← algebraicIndependent_iff_injective_aeval]
258     exact selected_lifts_algebraicallyIndependent Q V hground hQV htb x hx
259   have hrange : evQ.range = A := (Algebra.adjoin_range_eq_range_aeval k x).symm
260   let eA : P ≃ₐ[k] A :=
261     (AlgEquiv.ofInjective evQ hevQ).trans (Subalgebra.equivOfEq evQ.range A hrange)
262   have heA (p : P) : (eA p : Q) = evQ p := by simp [eA]
263   have heAvar (z : t) : ((eA (MvPolynomial.X z) : A) : Q) = x z := by
264     rw [heA]
265     exact MvPolynomial.aeval_X x z
266   let iA : A →ₐ[k] E := (IsScalarTower.toAlgHom k P E).comp eA.symm.toAlgHom
267   let fA : A →ₐ[k] V.toSubring := qV.comp (Subalgebra.val A)
268   let evV : P →ₐ[k] V.toSubring := qV.comp evQ
269   let evκ : P →ₐ[k] ResidueField V.toSubring :=
270     MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring))
271   let ρ : V.toSubring →ₐ[k] ResidueField V.toSubring := {
272     toRingHom := residue V.toSubring
273     commutes' := by
274       intro c
275       rw [← ResidueField.algebraMap_eq]
276       exact (IsScalarTower.algebraMap_apply k V.toSubring
277         (ResidueField V.toSubring) c).symm
278   }
279   have hρev : ρ.comp evV = evκ := by
280     ext z
281     dsimp [evV, evQ, ρ, evκ]
282     simp only [AlgHom.comp_apply, MvPolynomial.aeval_X]
283     change residue V.toSubring (qV (x z)) = (z : ResidueField V.toSubring)
284     have hmapx : qV (x z) =
285         (⟨(x z : F), hQV (x z).property⟩ : V.toSubring) := by
286       apply Subtype.ext
287       rfl
288     rw [hmapx]
289     exact hx z
290   have hevκ : Function.Injective evκ :=
291     (algebraicIndependent_iff_injective_aeval.mpr htb.1)
292   have hevV : Function.Injective evV := by
293     intro p q hpq
294     apply hevκ
295     calc
296       evκ p = ρ (evV p) := (AlgHom.congr_fun hρev p).symm
297       _ = ρ (evV q) := congrArg ρ hpq
298       _ = evκ q := AlgHom.congr_fun hρev q
299   have hunitV : ∀ s : nonZeroDivisors P, IsUnit (evV s.1) := by
300     intro s
301     apply (IsLocalRing.notMem_maximalIdeal).1
302     intro hs
303     have hres0 : ρ (evV s.1) = 0 :=
304       (IsLocalRing.residue_eq_zero_iff (evV s.1)).2 hs
305     have heval0 : evκ s.1 = 0 := by
306       rw [← AlgHom.congr_fun hρev s.1]
307       exact hres0
308     have hp0 : s.1 = 0 := hevκ (by simpa using heval0)
309     exact (nonZeroDivisors.ne_zero s.2) hp0
310   have hunitκ : ∀ s : nonZeroDivisors P, IsUnit (evκ s.1) := by
311     intro s
312     apply isUnit_iff_ne_zero.mpr
313     intro hs
314     exact (nonZeroDivisors.ne_zero s.2) (hevκ (by simpa using hs))
315   have hg : Function.Injective (IsFractionRing.liftAlgHom hevκ : E →ₐ[k]
316       ResidueField V.toSubring) := by
317     change Function.Injective (IsLocalization.lift hunitκ)
318     rw [IsLocalization.lift_injective_iff hunitκ]
319     intro p q
320     constructor
321     · intro hpq
322       exact congrArg evκ ((FaithfulSMul.algebraMap_injective P E) hpq)
323     · intro hpq
324       exact congrArg (algebraMap P E) (hevκ hpq)
325   let fE : E →ₐ[k] V.toSubring :=
326     IsLocalization.liftAlgHom (A := k) (M := nonZeroDivisors P) hunitV
327   let g : E →ₐ[k] ResidueField V.toSubring := IsFractionRing.liftAlgHom hevκ
328   have hfieldRange : g.fieldRange = IntermediateField.adjoin k
329       (Set.range (fun z : t => (z : ResidueField V.toSubring))) := by
330     apply IsFractionRing.liftAlgHom_fieldRange_eq_of_range_eq hevκ
331     exact (Algebra.adjoin_range_eq_range_aeval k
332       (fun z : t => (z : ResidueField V.toSubring))).symm
333   have hfEbase : fE.comp (IsScalarTower.toAlgHom k P E) = evV := by
334     apply AlgHom.coe_ringHom_injective
335     apply RingHom.ext
336     intro p
337     change fE (algebraMap P E p) = evV p
338     change IsLocalization.lift hunitV (algebraMap P E p) = evV p
339     exact IsLocalization.lift_eq hunitV p
340   have hgbase : g.comp (IsScalarTower.toAlgHom k P E) = evκ := by
341     apply AlgHom.coe_ringHom_injective
342     apply RingHom.ext
343     intro p
344     change g (algebraMap P E p) = evκ p
345     rw [IsFractionRing.liftAlgHom_apply, IsFractionRing.lift_algebraMap hevκ]
346     rfl
347   have hgvar (z : t) : g (algebraMap P E (MvPolynomial.X z)) =
348       (z : ResidueField V.toSubring) := by
349     have h := AlgHom.congr_fun hgbase (MvPolynomial.X z)
350     change g (algebraMap P E (MvPolynomial.X z)) = evκ (MvPolynomial.X z) at h
351     calc
352       g (algebraMap P E (MvPolynomial.X z)) = evκ (MvPolynomial.X z) := h
353       _ = (z : ResidueField V.toSubring) := by
354         change MvPolynomial.aeval (fun z : t => (z : ResidueField V.toSubring))
355           (MvPolynomial.X z) = _
356         rw [MvPolynomial.aeval_X]
357   have hfAvar (z : t) : fA (eA (MvPolynomial.X z)) =
358       (⟨(x z : F), hQV (x z).property⟩ : V.toSubring) := by
359     apply Subtype.ext
360     simpa [fA, qV, chartAlgebraToValuation] using
361       congrArg (fun y : Q => (y : F)) (heAvar z)
362   have hiA : ∀ p : P, iA (eA p) = algebraMap P E p := by
363     intro p
364     simp [iA]
365   have hred : ρ.comp fE = g := by
366     apply IsLocalization.algHom_ext (W := nonZeroDivisors P)
367     calc
368       (ρ.comp fE).comp (IsScalarTower.toAlgHom k P E) = ρ.comp evV := by
369         rw [AlgHom.comp_assoc, hfEbase]
370       _ = evκ := hρev
371       _ = g.comp (IsScalarTower.toAlgHom k P E) := hgbase.symm
372   have hcomp : fE.comp iA = fA := by
373     apply AlgHom.ext
374     intro a
375     dsimp [iA, fA]
376     change fE (algebraMap P E (eA.symm a)) = qV ((a : A) : Q)
377     change IsLocalization.lift hunitV (algebraMap P E (eA.symm a)) =
378       qV ((a : A) : Q)
379     rw [IsLocalization.lift_eq hunitV]
380     have hback : evQ (eA.symm a) = (a : Q) := by
381       calc
382         evQ (eA.symm a) = (eA (eA.symm a) : Q) := (heA _).symm
383         _ = (a : Q) := congrArg (fun b : A => (b : Q)) (eA.apply_symm_apply a)
384     change evV (eA.symm a) = qV ((a : A) : Q)
385     exact congrArg qV hback
386   exact ⟨eA, fA, iA, g, fE, hg, hfieldRange, heAvar, hfAvar, hgvar, hiA,
387     congrArg AlgHom.toRingHom hred, congrArg AlgHom.toRingHom hcomp⟩
