17 theorem exists_injective_integral_polynomial_of_ring_trdeg_one
18     {E : Type u} {A : Type v} [Field E] [CommRing A] [IsDomain A] [Algebra E A]
19     [Algebra.FiniteType E A]
20     (htrdeg : Algebra.trdeg E A = 1) :
21     ∃ f : Polynomial E →ₐ[E] A, Function.Injective f ∧ f.IsIntegral := by
22   obtain ⟨s, g, hg, hgi⟩ := exists_integral_inj_algHom_of_fg E A
23   let P := MvPolynomial (Fin s) E
24   letI : Algebra P A := g.toRingHom.toAlgebra
25   letI : IsScalarTower E P A := IsScalarTower.of_algHom g
26   letI : FaithfulSMul E P :=
27     (faithfulSMul_iff_algebraMap_injective E P).mpr <| by
28       change Function.Injective (MvPolynomial.C : E → P)
29       exact MvPolynomial.C_injective _ _
30   letI : FaithfulSMul P A :=
31     (faithfulSMul_iff_algebraMap_injective P A).mpr <| by
32       change Function.Injective (g : P →+* A)
33       exact hg
34   have hInt : Algebra.IsIntegral P A := by
35     refine ⟨fun x => ?_⟩
36     exact hgi x
37   letI : Algebra.IsIntegral P A := hInt
38   letI : Algebra.IsAlgebraic P A := Algebra.IsIntegral.isAlgebraic
39   have hPA : Algebra.trdeg P A = 0 := trdeg_eq_zero
40   have hEq : Cardinal.lift.{v} (Algebra.trdeg E P) =
41       Cardinal.lift.{u} (Algebra.trdeg E A) := by
42     have h : Cardinal.lift.{v} (Algebra.trdeg E P) +
43         Cardinal.lift.{u} (Algebra.trdeg P A) =
44         Cardinal.lift.{u} (Algebra.trdeg E A) := lift_trdeg_add_eq E P A
45     simpa [hPA] using h
46   have hsCard : Cardinal.lift.{v} (Algebra.trdeg E P) = 1 := by
47     calc
48       _ = Cardinal.lift.{u} (Algebra.trdeg E A) := hEq
49       _ = 1 := by rw [htrdeg]; simp
50   have hs : s = 1 := by
51     simpa [P, MvPolynomial.trdeg_of_isDomain, Cardinal.lift_mk_fin] using hsCard
52   subst s
53   let e : MvPolynomial (Fin 1) E ≃ₐ[E] Polynomial E :=
54     MvPolynomial.uniqueAlgEquiv E (Fin 1)
55   let f : Polynomial E →ₐ[E] A := g.comp e.symm.toAlgHom
56   refine ⟨f, ?_, ?_⟩
57   · exact hg.comp e.symm.injective
58   · have he : (e.symm : Polynomial E →+* MvPolynomial (Fin 1) E).IsIntegral :=
59       RingHom.isIntegral_of_surjective _ e.symm.surjective
60     change (f : Polynomial E →+* A).IsIntegral
61     exact RingHom.IsIntegral.trans _ _ he hgi

63 /-- A domain and its fraction field have the same transcendence degree over the base field. -/
64 theorem trdeg_fractionRing_eq_trdeg
65     {E : Type u} {A : Type v} [Field E] [CommRing A] [IsDomain A] [Algebra E A] :
66     Algebra.trdeg E (FractionRing A) = Algebra.trdeg E A := by
67   letI : Algebra.IsAlgebraic A (FractionRing A) :=
68     (IsFractionRing.comap_isAlgebraic_iff
69       (A := A) (K := FractionRing A) (C := FractionRing A)).2 inferInstance
70   have hzero : Algebra.trdeg A (FractionRing A) = 0 := trdeg_eq_zero
71   have h : Algebra.trdeg E A + Algebra.trdeg A (FractionRing A) =
72       Algebra.trdeg E (FractionRing A) := trdeg_add_eq E A
73   rw [hzero, add_zero] at h
74   exact h.symm

76 /-- The same equality for any chosen realization of the fraction field. -/
77 theorem lift_trdeg_isFractionRing_eq
78     {E : Type u} {A : Type v} {L : Type w} [Field E] [CommRing A] [IsDomain A]
79     [Algebra E A] [Field L] [Algebra A L] [IsFractionRing A L]
80     [Algebra E L] [IsScalarTower E A L] :
81     Cardinal.lift.{w} (Algebra.trdeg E A) =
82       Cardinal.lift.{v} (Algebra.trdeg E L) := by
83   letI : Algebra.IsAlgebraic A L :=
84     (IsFractionRing.comap_isAlgebraic_iff (A := A) (K := L) (C := L)).2 inferInstance
85   have hzero : Algebra.trdeg A L = 0 := trdeg_eq_zero
86   have h : Cardinal.lift.{w} (Algebra.trdeg E A) +
87       Cardinal.lift.{v} (Algebra.trdeg A L) =
88       Cardinal.lift.{v} (Algebra.trdeg E L) := lift_trdeg_add_eq E A L
89   simpa [hzero] using h

91 /-- Noether normalization for an affine domain whose fraction field has transcendence degree one. -/
92 theorem exists_injective_integral_polynomial_of_fractionRing_trdeg_one
93     {E : Type u} {A : Type v} [Field E] [CommRing A] [IsDomain A] [Algebra E A]
94     [Algebra.FiniteType E A]
95     (htrdeg : Algebra.trdeg E (FractionRing A) = 1) :
96     ∃ f : Polynomial E →ₐ[E] A, Function.Injective f ∧ f.IsIntegral := by
97   have hA : Algebra.trdeg E A = 1 :=
98     (trdeg_fractionRing_eq_trdeg (E := E) (A := A)).symm.trans htrdeg
99   exact exists_injective_integral_polynomial_of_ring_trdeg_one hA

101 /-- Noether normalization when the fraction field is given by an arbitrary fraction-ring model. -/
102 theorem exists_injective_integral_polynomial_of_isFractionRing_trdeg_one
103     {E : Type u} {A : Type v} {L : Type w} [Field E] [CommRing A] [IsDomain A]
104     [Algebra E A] [Algebra.FiniteType E A] [Field L] [Algebra A L]
105     [IsFractionRing A L] [Algebra E L] [IsScalarTower E A L]
106     (htrdeg : Algebra.trdeg E L = 1) :
107     ∃ f : Polynomial E →ₐ[E] A, Function.Injective f ∧ f.IsIntegral := by
108   have hLift : Cardinal.lift.{w} (Algebra.trdeg E A) = 1 := by
109     calc
110       _ = Cardinal.lift.{v} (Algebra.trdeg E L) :=
111         lift_trdeg_isFractionRing_eq (E := E) (A := A) (L := L)
112       _ = 1 := by rw [htrdeg]; simp
113   have hA : Algebra.trdeg E A = 1 := Cardinal.lift_eq_one.mp hLift
114   exact exists_injective_integral_polynomial_of_ring_trdeg_one hA
