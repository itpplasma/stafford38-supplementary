18 /-- An integral closure inside the ambient field is contained in every
19 valuation subring that contains the original domain. -/
20 theorem integralClosure_subring_le_valuationSubring
21     {K : Type u} [Field K] (R : Subring K) (V : ValuationSubring K)
22     (hRV : R ≤ V.toSubring) :
23     (integralClosure R K).toSubring ≤ V.toSubring := by
24   letI : IsIntegrallyClosedIn V.toSubring K := inferInstanceAs (IsIntegrallyClosedIn V K)
25   exact Subring.integralClosure_subring_le_iff.mpr hRV
26 
27 /-- The center on the integral closure is the contraction of the retained
28 valuation ring's maximal ideal along the canonical inclusion of the closure. -/
29 def integralClosureCenter
30     {K : Type u} [Field K] (R : Subring K) (V : ValuationSubring K)
31     (hRV : R ≤ V.toSubring) :
32     Ideal (integralClosure R K).toSubring := by
33   let hBV : (integralClosure R K).toSubring ≤ V.toSubring :=
34     integralClosure_subring_le_valuationSubring R V hRV
35   exact (maximalIdeal V.toLocalSubring.toSubring).comap
36     (Subring.inclusion hBV)
37 
38 /-- The contracted ideal is literally the kernel of the map to the residue
39 field of the valuation ring. -/
40 theorem integralClosureCenter_eq_residueKernel
41     {K : Type u} [Field K] (R : Subring K) (V : ValuationSubring K)
42     (hRV : R ≤ V.toSubring) :
43     integralClosureCenter R V hRV =
44       RingHom.ker
45         ((Ideal.Quotient.mk (maximalIdeal V.toLocalSubring.toSubring)).comp
46           (Subring.inclusion
47             (integralClosure_subring_le_valuationSubring R V hRV))) := by
48   rw [integralClosureCenter, ← RingHom.comap_ker, Ideal.mk_ker]
49   rfl

51 theorem integralClosureCenter_isPrime
52     {K : Type u} [Field K] (R : Subring K) (V : ValuationSubring K)
53     (hRV : R ≤ V.toSubring) :
54     (integralClosureCenter R V hRV).IsPrime := by
55   let hBV : (integralClosure R K).toSubring ≤ V.toSubring :=
56     integralClosure_subring_le_valuationSubring R V hRV
57   have hmax : (maximalIdeal V.toLocalSubring.toSubring).IsMaximal :=
58     IsLocalRing.maximalIdeal.isMaximal _
59   change (maximalIdeal V.toLocalSubring.toSubring).comap
60       (Subring.inclusion hBV) |>.IsPrime
61   exact hmax.isPrime.comap _

63 /-- Any nonzero chart element vanishing at the valuation center makes the
64 normalization center nonzero. -/
65 theorem integralClosureCenter_ne_bot_of_nonzero_maximal
66     {K : Type u} [Field K] (R : Subring K) (V : ValuationSubring K)
67     (hRV : R ≤ V.toSubring) (r : R)
68     (hr0 : (r : K) ≠ 0)
69     (hrmax : (Subring.inclusion hRV r) ∈
70       maximalIdeal V.toLocalSubring.toSubring) :
71     integralClosureCenter R V hRV ≠ ⊥ := by
72   let hBV : (integralClosure R K).toSubring ≤ V.toSubring :=
73     integralClosure_subring_le_valuationSubring R V hRV
74   let b : (integralClosure R K).toSubring :=
75     ⟨(r : K), by
76       change IsIntegral R (algebraMap R K r)
77       exact isIntegral_algebraMap⟩
78   have hmap : Subring.inclusion hBV b = Subring.inclusion hRV r := by
79     apply Subtype.ext
80     rfl
81   have hbcenter : b ∈ integralClosureCenter R V hRV := by
82     change Subring.inclusion hBV b ∈ maximalIdeal V.toLocalSubring.toSubring
83     rw [hmap]
84     exact hrmax
85   intro hbot
86   have hb0 : b = 0 := by
87     apply Ideal.mem_bot.mp
88     simpa [hbot] using hbcenter
89   apply hr0
90   have hv := congrArg Subtype.val hb0
91   exact hv

93 /-- The center above is precisely the prime required by the standard
94 local-domination lemma. No codimension or height assertion is included. -/
95 theorem localRing_at_integralClosureCenter_le_valuationSubring
96     {K : Type u} [Field K] (R : Subring K) (V : ValuationSubring K)
97     (hRV : R ≤ V.toSubring)
98     (hp : (integralClosureCenter R V hRV).IsPrime) :
99     LocalSubring.ofPrime (integralClosure R K).toSubring
100         (integralClosureCenter R V hRV) ≤ V.toLocalSubring := by
101   let hBV : (integralClosure R K).toSubring ≤ V.toSubring :=
102     integralClosure_subring_le_valuationSubring R V hRV
103   letI : (integralClosureCenter R V hRV).IsPrime := hp
104   exact Stafford38.Geometry.localRing_at_contracted_maximalIdeal_le_valuationSubring
105     (integralClosure R K).toSubring V hBV
106     (integralClosureCenter R V hRV) rfl
