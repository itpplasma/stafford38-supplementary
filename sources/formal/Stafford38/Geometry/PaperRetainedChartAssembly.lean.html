11 # Generic-point equations for an actual retained chart
12 
13 The retained-place transport theorem identifies the point of its completed
14 projective column with the image of the component generic point.  These
15 abstract map lemmas extract the base equations and exact generic kernel from
16 that identification, without repeating the retained completion construction.
17 -/

44 /-- Lane-C algebraicity of the residue coordinates on the affine tail
45 produces the actual coefficientwise derivation columns and a unit selected
46 minor on the full retained projective column. -/
47 theorem exists_retained_projective_derivation_columns
48     [CharZero k]
49     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) (i : Fin m)
50     (W : Data k (ComponentFractionField P) (componentCoordinate P i)) :
51     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
52         (ComponentFractionField P) := W.ambientAlgebra
53     let V := W.place.valuation.toSubring
54     letI : IsDiscreteValuationRing V := W.place.isDiscrete
55     letI : Algebra W.coefficientField V :=
56       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
57     let κ := ResidueField V
58     letI : Algebra k κ := retainedResidueGroundAlgebra P i W
59     ∀ (q : Fin (m + 1) → V),
60       Algebra.IsAlgebraic
61         (IntermediateField.adjoin k
62           (Set.range fun j : Fin m ↦ residue V (q (Fin.succ j))) :
63             IntermediateField k κ) κ →
64       ∃ (rows : Fin (Module.finrank κ (Ω[κ⁄k])) ↪ Fin (m + 1))
65         (D : Fin (Module.finrank κ (Ω[κ⁄k])) → Derivation k κ κ),
66         (∀ i j, D j (PowerSeries.constantCoeff
67           (retainedToCompletedPowerSeries W (q (rows i)))) =
68             if i = j then 1 else 0) ∧
69         PowerSeries.constantCoeff
70           (selectedMinor
71             (coefficientwiseTangentMatrix
72               (fun a ↦ retainedToCompletedPowerSeries W (q a)) D) rows).det = 1 := by
73   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
74       (ComponentFractionField P) := W.ambientAlgebra
75   let V := W.place.valuation.toSubring
76   letI : IsDiscreteValuationRing V := W.place.isDiscrete
77   letI : Algebra W.coefficientField V :=
78     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
79   let κ := ResidueField V
80   letI : Algebra k κ := retainedResidueGroundAlgebra P i W
81   dsimp only
82   intro q halg
83   let qhat : Fin (m + 1) → PowerSeries κ :=
84     fun a ↦ retainedToCompletedPowerSeries W (q a)
85   let qbar : Fin m → κ := fun j ↦ residue V (q (Fin.succ j))
86   have hcoeff : ∀ j, PowerSeries.constantCoeff (qhat (Fin.succ j)) = qbar j := by
87     intro j
88     exact constantCoeff_retainedToCompleted_eq_residue W (q (Fin.succ j))
89   simpa [qhat] using
90     (exists_projective_derivation_frame_of_residue_coordinates
91       qhat qbar hcoeff (by simpa [qbar] using halg))

93 /-- Component equations remain zero after a field map, once its ground map and
94 its image of the generic coordinate point are identified. -/
95 theorem component_equations_vanish_after_point_transport
96     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
97     {L : Type u} [CommSemiring L] [Algebra k L]
98     (lift : ComponentFractionField P →+* L)
99     (hground : lift.comp (algebraMap k (ComponentFractionField P)) =
100       algebraMap k L)
101     (point : Fin m → L)
102     (hpoint : lift ∘ componentCoordinate P = point)
103     {f : MvPolynomial (Fin m) k} (hf : f ∈ P.asIdeal) :
104     MvPolynomial.eval₂ (algebraMap k L) point f = 0 := by
105   have hgeneric : MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
106       (componentCoordinate P) f = 0 := by
107     rw [← componentAffineGenericPointMap_eq_eval₂]
108     exact componentAffineGenericPointMap_eq_zero_of_mem P hf
109   have hmap := MvPolynomial.eval₂_comp_left lift
110     (algebraMap k (ComponentFractionField P)) (componentCoordinate P) f
111   rw [hgeneric, map_zero] at hmap
112   have htransport :
113       MvPolynomial.eval₂ (lift.comp (algebraMap k (ComponentFractionField P)))
114         (lift ∘ componentCoordinate P) f = 0 := hmap.symm
115   rw [← hground, ← hpoint]
116   exact htransport

118 /-- Evaluation at the transported generic point has exactly the component
119 prime as its kernel when the field map is injective. -/
120 theorem component_generic_kernel_after_point_transport
121     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
122     {L : Type u} [Field L] [Algebra k L]
123     (lift : ComponentFractionField P →+* L)
124     (hinjective : Function.Injective lift)
125     (hground : lift.comp (algebraMap k (ComponentFractionField P)) =
126       algebraMap k L)
127     (point : Fin m → L)
128     (hpoint : lift ∘ componentCoordinate P = point) :
129     ∀ f : MvPolynomial (Fin m) k,
130       MvPolynomial.eval point (MvPolynomial.map (algebraMap k L) f) = 0 ↔
131         f ∈ P.asIdeal := by
132   intro f
133   have heval : MvPolynomial.eval point (MvPolynomial.map (algebraMap k L) f) =
134       lift (componentAffineGenericPointMap P f) := by
135     rw [← hpoint, ← hground, MvPolynomial.eval_map,
136       ← MvPolynomial.eval₂_comp_left,
137       ← componentAffineGenericPointMap_eq_eval₂]
138   rw [heval, map_eq_zero_iff lift hinjective]
139   have hker : RingHom.ker (componentAffineGenericPointMap P) = P.asIdeal :=
140     componentAffineGenericPointMap_ker P
141   constructor
142   · intro hf
143     have hmem : f ∈ RingHom.ker (componentAffineGenericPointMap P) :=
144       RingHom.mem_ker.mpr hf
145     rw [hker] at hmem
146     exact hmem
147   · intro hf
148     have hmem : f ∈ RingHom.ker (componentAffineGenericPointMap P) := by
149       rw [hker]
150       exact hf
151     exact RingHom.mem_ker.mp hmem
