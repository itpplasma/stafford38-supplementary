25 variable {k : Type*} [Field k] [IsAlgClosed k] {n : ℕ}

27 def projectiveHomogeneousZeroLocus (P : MvPolynomial (Fin n) k) :
28     Set (Projectivization k (Fin n → k)) :=
29   {z | ∀ v : Fin n → k, ∀ hv : v ≠ 0,
30     z = Projectivization.mk k v hv → MvPolynomial.eval v P = 0}
31 
32 def projectiveHomogeneousClosure
33     (T : Set (Projectivization k (Fin n → k))) :
34     Set (Projectivization k (Fin n → k)) :=
35   {z | ∀ P : MvPolynomial (Fin n) k, ∀ d : ℕ,
36     P.IsHomogeneous d →
37     T ⊆ projectiveHomogeneousZeroLocus P →
38       z ∈ projectiveHomogeneousZeroLocus P}
39 
40 def smoothConormalDirectionSet
41     (I : Ideal (MvPolynomial (Fin n) k)) :
42     Set (Fin n → k) :=
43   {ξ | ∃ y : Fin n → k,
44     SmoothAffinePoint I y ∧ ξ ≠ 0 ∧
45       coordinateCovector ξ ∈ affineConormalSpace y I}
46 
47 def smoothConormalFibreProjection
48     (I : Ideal (MvPolynomial (Fin n) k)) : Set (Fin n → k) :=
49   {ξ | ∃ y : Fin n → k,
50     SmoothAffinePoint I y ∧
51       coordinateCovector ξ ∈ affineConormalSpace y I}
52 
53 def projectivizedDirectionSet (T : Set (Fin n → k)) :
54     Set (Projectivization k (Fin n → k)) :=
55   {z | ∃ v, v ∈ T ∧ ∃ hv : v ≠ 0, z = Projectivization.mk k v hv}

57 private theorem eval_smul_of_isHomogeneous
58     (P : MvPolynomial (Fin n) k) (d : ℕ) (hP : P.IsHomogeneous d)
59     (a : k) (v : Fin n → k) :
60     MvPolynomial.eval (a • v) P = a ^ d * MvPolynomial.eval v P := by
61   simp only [MvPolynomial.eval_eq']
62   rw [Finset.mul_sum]
63   apply Finset.sum_congr rfl
64   intro s hs
65   have hd : ∑ i, s i = d := by
66     rw [← s.sum_fintype (fun _ c => c) (fun _ => rfl)]
67     simpa [MvPolynomial.IsHomogeneous, Finsupp.weight_apply] using
68       hP (MvPolynomial.mem_support_iff.mp hs)
69   simp only [Pi.smul_apply, smul_eq_mul, mul_pow, Finset.prod_mul_distrib]
70   rw [Finset.prod_pow_eq_pow_sum, hd]
71   ring

73 theorem mk_mem_projectiveHomogeneousClosure_of_fibre_zeroLocus
74     (I : Ideal (MvPolynomial (Fin n) k)) (ξ : Fin n → k) (hξ : ξ ≠ 0)
75     (hvan : ξ ∈ MvPolynomial.zeroLocus k
76       (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I))) :
77     Projectivization.mk k ξ hξ ∈
78       projectiveHomogeneousClosure
79         (projectivizedDirectionSet (smoothConormalDirectionSet I)) := by
80   intro P d hP hT
81   obtain ⟨i, hi⟩ : ∃ i, ξ i ≠ 0 := by
82     simpa [funext_iff] using hξ
83   have hXPvan : MvPolynomial.X i * P ∈ MvPolynomial.vanishingIdeal k
84       (smoothConormalFibreProjection I) := by
85     rw [MvPolynomial.mem_vanishingIdeal_iff]
86     intro v hv
87     by_cases hv0 : v = 0
88     · subst v
89       simp
90     · have hproj : Projectivization.mk k v hv0 ∈
91           projectivizedDirectionSet (smoothConormalDirectionSet I) := by
92         rcases hv with ⟨y, hy, hc⟩
93         exact ⟨v, ⟨y, hy, hv0, hc⟩, hv0, rfl⟩
94       change MvPolynomial.eval v (MvPolynomial.X i * P) = 0
95       rw [MvPolynomial.eval_mul, MvPolynomial.eval_X]
96       rw [hT hproj v hv0 rfl, mul_zero]
97   intro v hv hz
98   obtain ⟨a, ha⟩ := (Projectivization.mk_eq_mk_iff k ξ v hξ hv).mp hz
99   have hξXP : MvPolynomial.eval ξ (MvPolynomial.X i * P) = 0 :=
100     (MvPolynomial.mem_zeroLocus_iff.mp hvan) _ hXPvan
101   have hξP : MvPolynomial.eval ξ P = 0 := by
102     rw [MvPolynomial.eval_mul, MvPolynomial.eval_X] at hξXP
103     exact (mul_eq_zero.mp hξXP).resolve_left hi
104   have hscaled := eval_smul_of_isHomogeneous P d hP (a : k) v
105   have ha' : (a : k) • v = ξ := by
106     rw [← Units.smul_def]
107     exact ha
108   rw [ha', hξP] at hscaled
109   exact (mul_eq_zero.mp hscaled.symm).resolve_left
110     (pow_ne_zero d (Units.ne_zero a))
111 
112 end
