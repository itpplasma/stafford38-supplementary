47 /- The common-scale equation is retained explicitly: this is the bridge used
48 by the homogeneous-equation transport theorem, rather than an opaque
49 membership assertion about a projective closure. -/
50 theorem exists_completed_normalized_projective_coordinates
51     {k : Type u} [Field k] [CharZero k] [IsAlgClosed k] {n : ℕ}
52     (I : RightIdeal (WeylAlg k n)) (i : Fin n)
53     (hdisjoint : Disjoint
54       (orderCharacteristicSupport k I)
55       (PrimeSpectrum.zeroLocus
56         ({MvPolynomial.X (.inl i)} : Set (SymbolRing k n))))
57     (P : PrimeSpectrum (MvPolynomial (Fin n) k))
58     (hBP : reducedOrderBaseIdeal k I ≤ P.asIdeal)
59     (hi : Transcendental k (componentCoordinate P i)) :
60     ∃ W : Data k (ComponentFractionField P) (componentCoordinate P i),
61       letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
62           (ComponentFractionField P) := W.ambientAlgebra
63       ∃ (chart : Fin (n + 1))
64         (q : Fin (n + 1) → W.place.valuation.toSubring)
65         (scale : ComponentFractionField P),
66         scale ≠ 0 ∧ q chart = 1 ∧ q 0 ≠ 0 ∧
67         (∀ a, (q a : ComponentFractionField P) =
68           scale * componentProjectivePoint P a) ∧
69         q (Fin.succ i) = q 0 * W.place.parameter ∧
70         (let V := W.place.valuation.toSubring
71          letI : IsDiscreteValuationRing V := W.place.isDiscrete
72          letI : Algebra W.coefficientField V :=
73            (relativeCoefficientMap W.coefficientField W.place).toAlgebra
74          let qhat := fun a ↦ retainedToCompletedPowerSeries W (q a)
75          let ratio := retainedToCompletedPowerSeries W W.place.parameter
76          qhat chart = 1 ∧ qhat 0 ≠ 0 ∧ ratio ≠ 0 ∧
77          PowerSeries.constantCoeff (qhat 0) = 0 ∧
78          PowerSeries.constantCoeff ratio = 0 ∧
79          qhat (Fin.succ i) = qhat 0 * ratio) := by
80   obtain ⟨W, chart, q, scale, hscale, hchart, hzero, hq, hratio⟩ :=
81     exists_normalizedProjectivePoint_relativeRetainedBoundaryPlace P i hi
82   letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
83       (ComponentFractionField P) := W.ambientAlgebra
84   have hzero_nonunit : ¬IsUnit (q 0) :=
85     normalizedComponentProjectivePoint_zero_nonunit
86       I i hdisjoint P hBP W q scale hq
87   refine ⟨W, chart, q, scale, hscale, hchart, hzero, hq, hratio, ?_⟩
88   let V := W.place.valuation.toSubring
89   letI : IsDiscreteValuationRing V := W.place.isDiscrete
90   letI : Algebra W.coefficientField V :=
91     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
92   exact retainedCompleted_projective_order_properties W (ι := Fin (n + 1))
93     (q := q) (chart := chart) (zero := 0) (axis := Fin.succ i)
94     hchart hzero hzero_nonunit hratio
95

97 
98 end
