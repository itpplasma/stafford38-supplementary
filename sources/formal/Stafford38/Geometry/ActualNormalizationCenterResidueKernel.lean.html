16 /-- A ring map from the integral closure into the retained valuation ring that
17 preserves the ambient-field value induces exactly the canonical contracted
18 residue-kernel center. -/
19 theorem residue_kernel_eq_canonical_center_comap
20     {F : Type u} [Field F]
21     (R : Subring F) (U : ValuationSubring F)
22     [IsLocalRing U.toSubring]
23     (hRV : R ≤ U.toSubring)
24     (eB : (integralClosure R F) ≃+* (integralClosure R F).toSubring)
25     (fB : (integralClosure R F) →+* U.toSubring)
26     (heB : ∀ b, ((eB b : (integralClosure R F).toSubring) : F) = b)
27     (hfB : ∀ b, ((fB b : U.toSubring) : F) = b) :
28     RingHom.ker ((residue U.toSubring).comp fB) =
29       (ProjectiveChartNormalizationCenter.integralClosureCenter R U hRV).comap
30         eB.toRingHom := by
31   let hBV :=
32     ProjectiveChartNormalizationCenter.integralClosure_subring_le_valuationSubring
33       R U hRV
34   let center := ProjectiveChartNormalizationCenter.integralClosureCenter R U hRV
35   ext b
36   change residue U.toSubring (fB b) = 0 ↔ eB b ∈ center
37   rw [IsLocalRing.residue_eq_zero_iff]
38   have hmap : Subring.inclusion hBV (eB b) = fB b := by
39     apply Subtype.ext
40     calc
41       ((eB b : (integralClosure R F).toSubring) : F) = b := heB b
42       _ = ((fB b : U.toSubring) : F) := (hfB b).symm
43   rw [← hmap]
44   change Subring.inclusion hBV (eB b) ∈ maximalIdeal U.toSubring ↔
45       eB b ∈ (maximalIdeal U.toLocalSubring.toSubring).comap
46         (Subring.inclusion hBV)
47   change Subring.inclusion hBV (eB b) ∈ maximalIdeal U.toSubring ↔
48       Subring.inclusion hBV (eB b) ∈ maximalIdeal U.toLocalSubring.toSubring
49   rfl
50 
51 end Stafford38.Geometry.ActualNormalizationCenterResidueKernel
