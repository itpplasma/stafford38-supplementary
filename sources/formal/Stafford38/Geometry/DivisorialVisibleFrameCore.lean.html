38 def DivisorialVisibleFrameExistence : Prop :=
39   ∀ (k K : Type u) [Field k] [CharZero k] [Field K] [Algebra k K]
40     (r : ℕ) (y : Fin r → K) (i : Fin r),
41     IntermediateField.adjoin k (Set.range y) = ⊤ →
42     Transcendental k (y i) →
43     ∃ (E : IntermediateField k K) (V : ValuationSubring K)
44       (hEV : ∀ z : E, (z : K) ∈ V.toSubring)
45       (hVdvr : IsDiscreteValuationRing V.toSubring)
46       (hxV : y i ∈ V.toSubring),
47       letI : IsLocalRing V.toSubring := hVdvr.toIsLocalRing
48       letI : Algebra E V.toSubring := (coeffHom E V hEV).toAlgebra
49       letI : Algebra k V.toSubring := (groundHom E V hEV).toAlgebra
50       Transcendental E (y i) ∧
51       (⟨y i, hxV⟩ : V.toSubring) ∈ maximalIdeal V.toSubring ∧
52       Module.Finite E (ResidueField V.toSubring) ∧
53       Module.Finite V.toSubring (Ω[V.toSubring⁄k]) ∧
54       (∀ (s : K) (q : Fin (r + 1) → V.toSubring),
55         (∀ a, (q a : K) = s * Fin.cases 1 y a) →
56         (∃ j, q j = 1) → ¬ IsUnit (q 0) →
57         Algebra.IsAlgebraic
58           (IntermediateField.adjoin k
59             (Set.range fun j : Fin r ↦ residue V.toSubring (q (Fin.succ j))) :
60             IntermediateField k (ResidueField V.toSubring))
61           (ResidueField V.toSubring))

130 theorem ofField_residue_surjective (A : Subalgebra k K) [IsFractionRing A K]
131     (p : Ideal A) [p.IsPrime]
132     (z : ResidueField (placeSubalgebra A p).toSubring) :
133     ∃ (a b : A), b ∉ p ∧
134       z = residue (placeSubalgebra A p).toSubring
135         ⟨(a : K), ofField_mem_algebraMap A p a⟩ /
136         residue (placeSubalgebra A p).toSubring
137           ⟨(b : K), ofField_mem_algebraMap A p b⟩ := by
138   obtain ⟨x, rfl⟩ := residue_surjective (R := (placeSubalgebra A p).toSubring) z
139   change ∃ (a b : A), b ∉ p ∧
140     residue (placeSubalgebra A p).toSubring x =
141       residue (placeSubalgebra A p).toSubring
142           ⟨(a : K), ofField_mem_algebraMap A p a⟩ /
143         residue (placeSubalgebra A p).toSubring
144           ⟨(b : K), ofField_mem_algebraMap A p b⟩
145   have hx : (x : K) ∈ placeSubalgebra A p := x.property
146   simp only [placeSubalgebra, Localization.subalgebra.ofField] at hx
147   obtain ⟨a, b, hb, hab⟩ := hx
148   refine ⟨a, b, ?_, ?_⟩
149   · exact hb
150   · have hbinv : (algebraMap A K b)⁻¹ ∈ placeSubalgebra A p := by
151       simp only [placeSubalgebra, Localization.subalgebra.ofField]
152       exact ⟨1, b, hb, by simp⟩
153     let binv : (placeSubalgebra A p).toSubring :=
154       ⟨(algebraMap A K b)⁻¹, hbinv⟩
155     have hxb : x = algebraMap A (placeSubalgebra A p) a * binv := by
156       apply Subtype.ext
157       simpa [binv] using hab
158     have hbA0 : b ≠ 0 := fun hbzero ↦ hb (hbzero ▸ p.zero_mem)
159     have hb0 : algebraMap A K b ≠ 0 := by
160       intro hbzero
161       apply hbA0
162       apply Subtype.ext
163       exact hbzero
164     have hbmul : binv * algebraMap A (placeSubalgebra A p) b = 1 := by
165       apply Subtype.ext
166       exact inv_mul_cancel₀ hb0
167     have hresinv : residue (placeSubalgebra A p).toSubring binv =
168         (residue (placeSubalgebra A p).toSubring
169           (algebraMap A (placeSubalgebra A p) b))⁻¹ := by
170       apply eq_inv_of_mul_eq_one_left
171       rw [← map_mul, hbmul, map_one]
172     calc
173       residue (placeSubalgebra A p).toSubring x =
174           residue (placeSubalgebra A p).toSubring
175             (algebraMap A (placeSubalgebra A p) a * binv) := by rw [hxb]
176       _ = residue (placeSubalgebra A p).toSubring
177             ⟨(a : K), ofField_mem_algebraMap A p a⟩ /
178           residue (placeSubalgebra A p).toSubring
179             ⟨(b : K), ofField_mem_algebraMap A p b⟩ := by
180         rw [map_mul, hresinv, div_eq_mul_inv]
181         rfl

215 theorem isDiscreteValuationRing_of_isLocalization_atPrime
216     {A : Type*} [CommRing A] [IsDomain A] [IsNoetherianRing A] [IsIntegrallyClosed A]
217     (p : Ideal A) [p.IsPrime] (h : p.height = 1)
218     (S : Type*) [CommRing S] [IsDomain S] [Algebra A S]
219     [IsLocalization.AtPrime S p] : IsDiscreteValuationRing S := by
220   have hpb : p ≠ ⊥ := Ideal.ne_bot_of_height_eq_one h
221   letI : IsLocalRing S := IsLocalization.AtPrime.isLocalRing S p
222   letI : IsNoetherianRing S := IsLocalization.isNoetherianRing p.primeCompl S inferInstance
223   letI : IsIntegrallyClosed S :=
224     isIntegrallyClosed_of_isLocalization S p.primeCompl p.primeCompl_le_nonZeroDivisors
225   have hnf : ¬ IsField S := IsLocalization.AtPrime.not_isField A hpb S
226   have hkd : Ring.KrullDimLE 1 S := by
227     rw [Ring.krullDimLE_iff, IsLocalization.AtPrime.ringKrullDim_eq_height p S, h]
228     norm_num
229   have h3 : IsIntegrallyClosed S ∧ ∃! P : Ideal S, P ≠ ⊥ ∧ P.IsPrime := by
230     refine ⟨inferInstance, IsLocalRing.maximalIdeal _,
231       ⟨IsLocalRing.isField_iff_maximalIdeal_eq.not.mp hnf, inferInstance⟩, ?_⟩
232     rintro P ⟨hPb, hPp⟩
233     exact IsLocalRing.eq_maximalIdeal (hPp.isMaximal_of_ne_bot hPb)
234   exact ((IsDiscreteValuationRing.TFAE S hnf).out 4 1).mp h3

236 theorem stage1_exists_valuationSubring_of_transcendental
237     {k K : Type*} [Field k] [Field K] [Algebra k K] (x : K)
238     (hx : Transcendental k x) :
239     ∃ W : ValuationSubring K, (∀ c : k, algebraMap k K c ∈ W) ∧ x ∈ W ∧ x⁻¹ ∉ W := by
240   classical
241   have hx0 : x ≠ 0 := fun h => hx (h ▸ isAlgebraic_zero)
242   let A : Subring K := (Algebra.adjoin k ({x} : Set K)).toSubring
243   have hxA : x ∈ A := Algebra.subset_adjoin (Set.mem_singleton x)
244   let I : Ideal A := Ideal.span {⟨x, hxA⟩}
245   have hI : I ≠ ⊤ := by
246     intro htop
247     have h1 : (1 : A) ∈ I := htop ▸ Submodule.mem_top
248     rw [Ideal.mem_span_singleton'] at h1
249     obtain ⟨g, hg⟩ := h1
250     have hg' : (g : K) * x = 1 := by
251       have := congrArg Subtype.val hg
252       simpa using this
253     obtain ⟨q, hq⟩ : ∃ q : Polynomial k, Polynomial.aeval x q = (g : K) := by
254       have hmem : (g : K) ∈ Algebra.adjoin k ({x} : Set K) := g.2
255       rw [Algebra.adjoin_singleton_eq_range_aeval] at hmem
256       exact (AlgHom.mem_range _).1 hmem
257     apply hx
258     refine ⟨q * Polynomial.X - 1, ?_, ?_⟩
259     · intro h0
260       have hc := congrArg (fun p : Polynomial k => p.coeff 0) h0
261       simp [Polynomial.mul_coeff_zero] at hc
262     · simp [map_sub, map_mul, hq, hg']
263   obtain ⟨W, hAW, hIW⟩ := Ideal.image_subset_nonunits_valuationSubring I hI
264   have hxW : x ∈ W := hAW hxA
265   refine ⟨W, fun c => hAW (Subalgebra.algebraMap_mem _ c), hxW, ?_⟩
266   intro hinv
267   have hxn : x ∈ W.nonunits := hIW
268     ⟨⟨x, hxA⟩, Ideal.mem_span_singleton_self _, rfl⟩
269   rw [ValuationSubring.mem_nonunits_iff_exists_mem_maximalIdeal] at hxn
270   obtain ⟨hxW', hxm⟩ := hxn
271   rw [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff] at hxm
272   exact hxm (isUnit_iff_exists_inv.2
273     ⟨⟨x⁻¹, hinv⟩, Subtype.ext (mul_inv_cancel₀ hx0)⟩)

296 theorem stage6_transcendental_of_mem_maximalIdeal
297     {k K : Type*} [Field k] [Field K] [Algebra k K]
298     (E : IntermediateField k K) (V : ValuationSubring K) [IsLocalRing V.toSubring]
299     (hEV : ∀ z : E, (z : K) ∈ V.toSubring) (x : K) (hx0 : x ≠ 0)
300     (hxV : x ∈ V.toSubring)
301     (hxm : (⟨x, hxV⟩ : V.toSubring) ∈ maximalIdeal V.toSubring) :
302     Transcendental E x := by
303   intro halg
304   have hint : IsIntegral E x := halg.isIntegral
305   set a : E := (minpoly E x).coeff 0 with ha
306   have ha0 : a ≠ 0 := minpoly.coeff_zero_ne_zero hint hx0
307   obtain ⟨q, hq⟩ := X_dvd_sub_C (p := minpoly E x)
308   have h1 : aeval x (minpoly E x - C a) = aeval x (X * q) := by rw [hq]
309   rw [map_sub, minpoly.aeval, aeval_C, zero_sub, map_mul, aeval_X] at h1
310   have hqV : aeval x q ∈ V.toSubring := by
311     rw [aeval_eq_sum_range]
312     refine Subring.sum_mem _ fun n _ ↦ ?_
313     rw [Algebra.smul_def]
314     exact Subring.mul_mem _ (hEV _) (Subring.pow_mem _ hxV n)
315   have haV : algebraMap E K a ∈ V.toSubring := hEV _
316   have hmem : (⟨algebraMap E K a, haV⟩ : V.toSubring) ∈ maximalIdeal V.toSubring := by
317     have : (⟨algebraMap E K a, haV⟩ : V.toSubring) =
318         -((⟨x, hxV⟩ : V.toSubring) * ⟨aeval x q, hqV⟩) := by
319       apply Subtype.ext
320       push_cast
321       exact neg_eq_iff_eq_neg.1 h1
322     rw [this]
323     exact (Submodule.neg_mem_iff _).2 (Ideal.mul_mem_right _ _ hxm)
324   have hu : IsUnit (⟨algebraMap E K a, haV⟩ : V.toSubring) := by
325     refine isUnit_iff_exists_inv.2 ⟨⟨(algebraMap E K a)⁻¹, ?_⟩, Subtype.ext ?_⟩
326     · have h : (algebraMap E K a)⁻¹ = ((a⁻¹ : E) : K) := by
327         first | simp | (rw [IntermediateField.coe_inv]; rfl)
328       rw [h]; exact hEV _
329     · exact mul_inv_cancel₀ ((_root_.map_ne_zero _).2 ha0)
330   exact (mem_nonunits_iff.1 ((IsLocalRing.mem_maximalIdeal _).1 hmem)) hu

369 def Stage2Obligation : Prop :=
370   ∀ {k K : Type u} [Field k] [CharZero k] [Field K] [Algebra k K]
371     {r : ℕ} (y : Fin r → K) (i : Fin r)
372     (hgen : IntermediateField.adjoin k (Set.range y) = ⊤)
373     (hx : Transcendental k (y i)),
374     ∃ (j : Fin (r + 1)) (s : K) (A : Subalgebra k K),
375       s ≠ 0 ∧ (Fin.cases 1 y j : K) * s = 1 ∧
376       (∀ a, (Fin.cases 1 y a : K) * s ∈ A) ∧ y i ∈ A ∧
377       Algebra.FiniteType k A ∧ IsIntegrallyClosedIn A K ∧ IsFractionRing A K ∧
378       (∀ a : A, IsIntegral (Algebra.adjoin k
379         (Set.range (fun a : Fin (r + 1) => (Fin.cases 1 y a : K) * s) ∪ {y i})) (a : K)) ∧
380       (y i)⁻¹ ∉ A
