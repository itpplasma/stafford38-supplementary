8 # A power-series tangent-lattice limit
9 
10 This file isolates an elementary power-series calculation used in a
11 Grassmannian limit.  A matrix `T` of tangent columns is presented as `X ^ m`
12 times a matrix `B`.  If one selected coordinate of every column of `T` has a
13 specified additional positive power of `X`, then the selected row of `B`
14 vanishes in the residue field.  Consequently the corresponding coordinate
15 covector annihilates the residue of `B`.
16 
17 If, as a separate hypothesis, the columns of `B` are split, the retraction
18 correction from `RetractionSpecialization` lifts that residue row to an exact
19 annihilator over `k[[X]]`.  This is only the tangent-lattice calculation.  It
20 does not construct the split lattice from `T`, nor does it construct an arc, a
21 normalization, a divisor at infinity, or a projective closure.
22 -/

32 variable {k : Type u} [Field k]

34 /-- The coordinate row selecting `axis`. -/
35 def axisRow {ι : Type*} [DecidableEq ι] (axis : ι) : ι → k :=
36   fun i => if i = axis then 1 else 0
37 
38 @[simp] theorem axisRow_apply_same {ι : Type*} [DecidableEq ι] (axis : ι) :
39     axisRow (k := k) axis axis = 1 := by
40   simp [axisRow]
41 
42 @[simp] theorem axisRow_apply_of_ne {ι : Type*} [DecidableEq ι]
43     {axis i : ι} (hi : i ≠ axis) : axisRow (k := k) axis i = 0 := by
44   simp [axisRow, hi]
45 
46 /-- `T` is presented as the common power `X ^ m` times `B`.
47 
48 No maximality of `m` and no splitting of `B` is asserted here.
49 -/
50 def HasCommonPowerPresentation {ι κ : Type*}
51     (T B : Matrix ι κ (PowerSeries k)) (m : ℕ) : Prop :=
52   ∀ i j, T i j = (PowerSeries.X : PowerSeries k) ^ m * B i j
53 
54 /-- The selected coordinate has order at least `m + gap + 1` in every column. -/
55 def HasSelectedOrderGap {ι κ : Type*}
56     (T : Matrix ι κ (PowerSeries k)) (axis : ι) (m gap : ℕ) : Prop :=
57   ∀ j, ∃ u : PowerSeries k,
58     T axis j = (PowerSeries.X : PowerSeries k) ^ (m + (gap + 1)) * u
59 
60 /--
61 Cancelling the common factor shows that the selected row of the normalized
62 matrix still contains a positive power of `X`.
63 -/
64 theorem normalized_selectedRow_factor
65     {ι κ : Type*} (T B : Matrix ι κ (PowerSeries k))
66     (axis : ι) (m gap : ℕ)
67     (hnorm : HasCommonPowerPresentation T B m)
68     (hgap : HasSelectedOrderGap T axis m gap) :
69     ∀ j, ∃ u : PowerSeries k,
70       B axis j = (PowerSeries.X : PowerSeries k) ^ (gap + 1) * u := by
71   intro j
72   obtain ⟨u, hu⟩ := hgap j
73   refine ⟨u, ?_⟩
74   have hpow :
75       (PowerSeries.X : PowerSeries k) ^ m * B axis j =
76         (PowerSeries.X : PowerSeries k) ^ m *
77           ((PowerSeries.X : PowerSeries k) ^ (gap + 1) * u) := by
78     rw [← hnorm axis j, hu, pow_add, mul_assoc]
79   exact mul_left_cancel₀ (pow_ne_zero m PowerSeries.X_ne_zero) hpow
80 
81 /-- The strict order gap makes the selected normalized row vanish modulo `X`. -/
82 theorem residue_normalized_selectedRow_eq_zero
83     {ι κ : Type*} (T B : Matrix ι κ (PowerSeries k))
84     (axis : ι) (m gap : ℕ)
85     (hnorm : HasCommonPowerPresentation T B m)
86     (hgap : HasSelectedOrderGap T axis m gap) :
87     ∀ j, PowerSeries.constantCoeff (B axis j) = 0 := by
88   intro j
89   obtain ⟨u, hu⟩ := normalized_selectedRow_factor T B axis m gap hnorm hgap j
90   rw [hu]
91   simp [pow_succ]
92 
93 /-- The selected coordinate covector annihilates the residue tangent columns. -/
94 theorem axisRow_residue_rowMul_eq_zero
95     {ι κ : Type*} [Fintype ι] [DecidableEq ι]
96     (T B : Matrix ι κ (PowerSeries k))
97     (axis : ι) (m gap : ℕ)
98     (hnorm : HasCommonPowerPresentation T B m)
99     (hgap : HasSelectedOrderGap T axis m gap) :
100     rowMul (axisRow (k := k) axis)
101       (fun i j => PowerSeries.constantCoeff (B i j)) = 0 := by
102   funext j
103   simp [rowMul, axisRow,
104     residue_normalized_selectedRow_eq_zero T B axis m gap hnorm hgap j]

106 /--
107 Tangent-lattice limit with a split power-series column matrix.
108 
109 The returned row annihilates `B` exactly over `k[[X]]` and specializes to the
110 selected coordinate covector.  Thus that residue covector is represented by a
111 genuine power-series annihilator, not merely by a row annihilating the special
112 fibre.  The left inverse `C` is an independent lattice-splitting hypothesis;
113 it is not obtained from the common-power presentation.
114 -/
115 theorem exists_annihilator_specializing_to_axis
116     {ι κ : Type*} [Fintype ι] [Fintype κ]
117     [DecidableEq ι] [DecidableEq κ]
118     (T B : Matrix ι κ (PowerSeries k))
119     (C : Matrix κ ι (PowerSeries k))
120     (axis : ι) (m gap : ℕ)
121     (hnorm : HasCommonPowerPresentation T B m)
122     (hgap : HasSelectedOrderGap T axis m gap)
123     (hCB : C * B = 1) :
124     ∃ a : ι → PowerSeries k,
125       rowMul a B = 0 ∧ residueColumn a = axisRow (k := k) axis := by
126   let a₀ : ι → k := axisRow (k := k) axis
127   let a : ι → PowerSeries k :=
128     annihilatorLift (constantColumn a₀) B C
129   refine ⟨a, ?_⟩
130   have hres : residueColumn (constantColumn a₀) = a₀ :=
131     residueColumn_constantColumn a₀
132   have haxis :
133       rowMul a₀ (fun i j => PowerSeries.constantCoeff (B i j)) = 0 :=
134     axisRow_residue_rowMul_eq_zero T B axis m gap hnorm hgap
135   simpa [a, a₀] using
136     (powerSeries_annihilatorLift_spec (constantColumn a₀) a₀ B C hCB hres haxis)
