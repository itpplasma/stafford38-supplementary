25 /-- A selected finite residue basis, lifted into a chart algebra inside a
26 valuation ring, produces relative transcendence degree one whenever the same
27 valuation has the total-degree bound and a nonzero maximal parameter. -/
28 theorem component_field_trdeg_one_of_selected_residue_basis
29     {k F : Type u} [Field k] [CharZero k] [Field F] [Algebra k F]
30     (Q : Subalgebra k F) (V : ValuationSubring F)
31     [IsLocalRing V.toSubring] [Algebra k V.toSubring]
32     [Algebra k (ResidueField V.toSubring)]
33     [IsScalarTower k V.toSubring (ResidueField V.toSubring)]
34     (hground : ∀ c : k,
35       (algebraMap k V.toSubring c : F) = algebraMap k F c)
36     (hQV : Q.toSubring ≤ V.toSubring)
37     {t : Set (ResidueField V.toSubring)}
38     (htFinite : t.Finite)
39     (htBasis : IsTranscendenceBasis k ((↑) : t → ResidueField V.toSubring))
40     (x : t → Q)
41     (hx : ∀ z : t, residue V.toSubring
42       ⟨(x z : F), hQV (x z).property⟩ = (z : ResidueField V.toSubring))
43     (hbound : Algebra.trdeg k F ≤ Algebra.trdeg k (ResidueField V.toSubring) + 1)
44     (parameter : V.toSubring) (hparameter0 : parameter ≠ 0)
45     (hparameterMax : parameter ∈ maximalIdeal V.toSubring) :
46     let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
47     ∃ (eA : MvPolynomial t k ≃ₐ[k] A)
48       (fA : A →ₐ[k] V.toSubring)
49       (iA : A →ₐ[k] FractionRing (MvPolynomial t k))
50       (g : FractionRing (MvPolynomial t k) →ₐ[k] ResidueField V.toSubring)
51       (fEV : FractionRing (MvPolynomial t k) →ₐ[k] V.toSubring)
52       (fEF : FractionRing (MvPolynomial t k) →ₐ[k] F),
53       Function.Injective g ∧
54       (∀ z : t, ((eA (MvPolynomial.X z) : A) : Q) = x z) ∧
55       (∀ z : t, fA (eA (MvPolynomial.X z)) =
56         (⟨(x z : F), hQV (x z).property⟩ : V.toSubring)) ∧
57       (∀ z : t, g (algebraMap (MvPolynomial t k)
58         (FractionRing (MvPolynomial t k)) (MvPolynomial.X z)) = (z : ResidueField V.toSubring)) ∧
59       (IsLocalRing.residue V.toSubring).comp fEV.toRingHom = g.toRingHom ∧
60       fEV.toRingHom.comp iA.toRingHom = fA.toRingHom ∧
61       (∀ p : MvPolynomial t k,
62         iA (eA p) = algebraMap (MvPolynomial t k) (FractionRing (MvPolynomial t k)) p) ∧
63       (∀ p : MvPolynomial t k,
64         fEF (algebraMap (MvPolynomial t k) (FractionRing (MvPolynomial t k)) p) =
65           (((eA p : A) : Q) : F)) ∧
66       letI : Algebra (FractionRing (MvPolynomial t k)) F := fEF.toRingHom.toAlgebra
67       Algebra.trdeg (FractionRing (MvPolynomial t k)) F = 1 := by
68   classical
69   let A : Subalgebra k Q := Algebra.adjoin k (Set.range x)
70   let κ := ResidueField V.toSubring
71   letI : Algebra V.toSubring F := V.toSubring.subtype.toAlgebra
72   letI : IsScalarTower k V.toSubring F := IsScalarTower.of_algebraMap_eq fun c =>
73     (hground c).symm
74   obtain ⟨eA, fA, iA, g, fEV, hg, hrange, hgenA, hfA, hgenG,
75       hiA, hresE, hfE⟩ :=
76     Stafford38.Geometry.SelectedResidueCoefficientLocalization.exists_selected_fraction_field_maps
77       Q V hground hQV htBasis x hx
78   let E := FractionRing (MvPolynomial t k)
79   letI : Field E := FractionRing.field (MvPolynomial t k)
80   letI : Algebra E κ := g.toRingHom.toAlgebra
81   have hvars : IsTranscendenceBasis k (fun z : t =>
82       algebraMap E κ (algebraMap (MvPolynomial t k) E (MvPolynomial.X z))) := by
83     change IsTranscendenceBasis k
84       (fun z : t => g (algebraMap (MvPolynomial t k) E (MvPolynomial.X z)))

94   letI : Algebra E V.toSubring := fEV.toRingHom.toAlgebra
95   let fEF : E →ₐ[k] F := {
96     toRingHom := (algebraMap V.toSubring F).comp fEV.toRingHom
97     commutes' := by
98       intro c
99       change algebraMap V.toSubring F (fEV (algebraMap k E c)) = algebraMap k F c
100       rw [fEV.commutes]

103   letI : Algebra E F := fEF.toRingHom.toAlgebra
104   letI : IsScalarTower E V.toSubring F :=
105     IsScalarTower.of_algebraMap_eq fun z => rfl
106   letI : IsScalarTower k E F := IsScalarTower.of_algebraMap_eq fun c =>
107     (fEF.commutes c).symm
108   letI : IsScalarTower k E κ := IsScalarTower.of_algebraMap_eq fun c =>
109     (g.commutes c).symm
110   letI : Fintype t := htFinite.fintype
111   let basis : ULift.{u} t → E := fun i =>
112     algebraMap (MvPolynomial t k) E (MvPolynomial.X i.down)
113   have hbasis : IsTranscendenceBasis k basis :=
114     standardVariables_isTranscendenceBasis_of_fintype
115   have hVF : Function.Injective (algebraMap V.toSubring F) := by
116     intro a b hab
117     change (a : F) = (b : F) at hab
118     exact Subtype.val_injective hab
119   let qV : Q →ₐ[k] V.toSubring :=
120     Stafford38.Geometry.SelectedResidueCoefficientLocalization.chartAlgebraToValuation
121       Q V hground hQV
122   let evQ : MvPolynomial t k →ₐ[k] Q := MvPolynomial.aeval x
123   have heAeval : ∀ p : MvPolynomial t k, (eA p : A) = evQ p := by
124     have hmaps : (Subalgebra.val A).comp eA.toAlgHom = evQ := by
125       apply MvPolynomial.algHom_ext
126       intro z
127       change ((eA (MvPolynomial.X z) : A) : Q) =
128         MvPolynomial.aeval x (MvPolynomial.X z)
129       rw [MvPolynomial.aeval_X]
130       exact hgenA z
131     intro p
132     exact AlgHom.congr_fun hmaps p
133   have hfAeval : ∀ p : MvPolynomial t k,
134       fA (eA p) = qV (evQ p) := by
135     have hmaps : fA.comp eA.toAlgHom = qV.comp evQ := by
136       apply MvPolynomial.algHom_ext
137       intro z
138       change fA (eA (MvPolynomial.X z)) =
139         qV (MvPolynomial.aeval x (MvPolynomial.X z))
140       rw [MvPolynomial.aeval_X]
141       rw [hfA z]
142       apply Subtype.ext
143       rfl
144     intro p
145     exact AlgHom.congr_fun hmaps p
146   have hgen : ∀ p : MvPolynomial t k,
147       fEF (algebraMap (MvPolynomial t k) E p) = (((eA p : A) : Q) : F) := by
148     intro p
149     have hcompose : fEV (iA (eA p)) = fA (eA p) := by
150       have h := congrArg (fun φ : A →+* V.toSubring => φ (eA p)) hfE
151       change fEV.toRingHom (iA.toRingHom (eA p)) = fA.toRingHom (eA p)
152       exact h
153     calc
154       fEF (algebraMap (MvPolynomial t k) E p) =
155           algebraMap V.toSubring F (fEV (algebraMap (MvPolynomial t k) E p)) := rfl
156       _ = algebraMap V.toSubring F (fEV (iA (eA p))) := by rw [hiA p]
157       _ = algebraMap V.toSubring F (fA (eA p)) := by rw [hcompose]
158       _ = (((eA p : A) : Q) : F) := by
159         rw [hfAeval p, heAeval p]
160         rfl
161   refine ⟨eA, fA, iA, g, fEV, fEF, hg, hgenA, hfA, hgenG,
162     hresE, hfE, hiA, hgen, ?_⟩
163   exact eq_one_of_same_total_bound_of_finite_basis_and_parameter
164       basis hbasis hVF parameter hparameter0 hparameterMax hκalg hbound

176 /-- The retained projective chart supplies the hypotheses of the generic
