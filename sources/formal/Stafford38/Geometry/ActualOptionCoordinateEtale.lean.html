19 /-- Formal etaleness of an actual option-indexed coordinate map follows from a
20 polynomial-parameter DVR model when the model's local ring is identified over
21 `Polynomial E` with the target local ring. The coordinate equations are imposed
22 on the transported local parameters themselves, so the conclusion concerns the
23 actual chart map, not just an abstract model map. -/
24 theorem formallyEtale_of_uniformizer_model
25     {k : Type u} [CommRing k] {σ : Type w}
26     {E : Type v} [Field E]
27     {S : Type x} [CommRing S] [IsDomain S]
28     [Algebra E S] [Algebra (Polynomial E) S]
29     [IsScalarTower E (Polynomial E) S]
30     [Algebra.FiniteType E S]
31     (p : Ideal S) [p.IsPrime]
32     (hX0 : algebraMap (Polynomial E) S (Polynomial.X : Polynomial E) ≠ 0)
33     (hparam : (Stafford38.Geometry.AsymptoticDivisorExistence.coordinateZeroPrime E).map
34         (algebraMap (Polynomial E) (Localization.AtPrime p)) =
35           IsLocalRing.maximalIdeal (Localization.AtPrime p))
36     (hsep : Algebra.IsSeparable E p.ResidueField)
37     [Algebra (MvPolynomial σ k) E] [Algebra k E]
38     [IsScalarTower k (MvPolynomial σ k) E]
39     [IsLocalization (nonZeroDivisors (MvPolynomial σ k)) E]
40     {B : Type y} [CommRing B] [Algebra k B]
41     {T : Type z} [CommRing T]
42     [Algebra k T] [Algebra (Polynomial E) T]
43     [IsScalarTower k (Polynomial E) T]
44     (ρ : B →ₐ[k] T)
45     (q : Option σ → B)
46     (e : Localization.AtPrime p ≃ₐ[Polynomial E] T)
47     (hnone : e (algebraMap (Polynomial E) (Localization.AtPrime p)
48         (Polynomial.X : Polynomial E)) = ρ (q none))
49     (hsome : ∀ i : σ,
50       e (algebraMap (Polynomial E) (Localization.AtPrime p)
51         (Polynomial.C (algebraMap (MvPolynomial σ k) E (MvPolynomial.X i)))) =
52           ρ (q (some i))) :
53     let f : MvPolynomial (Option σ) k →ₐ[k] T :=
54       MvPolynomial.aeval (fun i => ρ (q i))
55     letI : Algebra (MvPolynomial (Option σ) k) T := f.toRingHom.toAlgebra
56     Algebra.FormallyEtale (MvPolynomial (Option σ) k) T := by
57   have hmodel : Algebra.FormallyEtale (Polynomial E) (Localization.AtPrime p) :=
58     Stafford38.Geometry.DVRParameterSmoothness.formallyEtale_localization_of_polynomial_uniformizer
59       p hX0 hparam hsep
60   letI : Algebra.FormallyEtale (Polynomial E) (Localization.AtPrime p) := hmodel
61   have htarget : Algebra.FormallyEtale (Polynomial E) T :=
62     Algebra.FormallyEtale.of_equiv e
63   letI : Algebra.FormallyEtale (Polynomial E) T := htarget
64   let f : MvPolynomial (Option σ) k →ₐ[k] T :=
65     MvPolynomial.aeval (fun i => ρ (q i))
66   have hnone' : f (MvPolynomial.X none) =
67       algebraMap (Polynomial E) T (Polynomial.X : Polynomial E) := by
68     rw [show f (MvPolynomial.X none) = ρ (q none) by simp [f]]
69     rw [← hnone]
70     exact e.commutes (Polynomial.X : Polynomial E)
71   have hsome' : ∀ i : σ,
72       f (MvPolynomial.X (some i)) = algebraMap (Polynomial E) T
73         (Polynomial.C (algebraMap (MvPolynomial σ k) E (MvPolynomial.X i))) := by
74     intro i
75     rw [show f (MvPolynomial.X (some i)) = ρ (q (some i)) by simp [f]]
76     rw [← hsome i]
77     exact e.commutes (Polynomial.C
78       (algebraMap (MvPolynomial σ k) E (MvPolynomial.X i)))
79   exact formallyEtale_of_optionCoordinate_generator_images f hnone' hsome' htarget
