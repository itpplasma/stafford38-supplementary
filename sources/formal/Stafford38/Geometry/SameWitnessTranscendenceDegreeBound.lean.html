37 /-- The actual retained witness bounds the transcendence degree of its component field by one
38 more than that of its own residue field.  The chart-residue basis is selected from this same
39 witness's normalized projective coordinates. -/
40 theorem component_trdeg_le_residue_trdeg_add_one_of_witness
41     {k : Type u} [Field k] [CharZero k]
42     {m : ℕ} (hm : 0 < m)
43     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
44     (w : GeneralDivisorialVisibleFrameWitness hm P) :
45     let C := w.column
46     let W := C.W
47     let F := ComponentFractionField P
48     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
49     let V := W.place.valuation.toSubring
50     letI : IsDiscreteValuationRing V := W.place.isDiscrete
51     letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
52     letI : Algebra W.coefficientField V :=
53       (relativeCoefficientMap W.coefficientField W.place).toAlgebra
54     let κ := ResidueField V
55     letI : Algebra k κ := retainedResidueGroundAlgebra P ⟨0, hm⟩ W
56     Algebra.trdeg k F ≤ Algebra.trdeg k κ + 1 := by
57   classical
58   dsimp only
59   let C := w.column
60   let W := C.W
61   let F := ComponentFractionField P
62   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
63   let V := W.place.valuation.toSubring
64   let i₀ : Fin m := ⟨0, hm⟩
65   letI : IsDiscreteValuationRing V := W.place.isDiscrete
66   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
67   letI : Algebra W.coefficientField V :=
68     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
69   letI : Algebra k V := C.groundCoeff.toAlgebra
70   let κ := ResidueField V
71   let retained : Algebra k κ := retainedResidueGroundAlgebra P i₀ W
72   letI : Algebra k κ := retained
73   have hretainedMap :
74       @algebraMap k κ _ _ retained = (residue V).comp C.groundCoeff := by
75     simpa [C, W, V, κ, C.groundCoeff_eq_retained] using w.retainedGroundMap
76   letI : IsScalarTower k V κ := IsScalarTower.of_algebraMap_eq fun c => by
77     have h := RingHom.congr_fun hretainedMap c
78     change algebraMap k κ c = residue V (C.groundCoeff c)
79     exact h
80   obtain ⟨t, htFinite, _htSubset, htBasis⟩ :=
81     exists_finite_chartCoordinate_residue_transcendenceBasis
82       (k := k) (V := V) C.q C.chart (chartAffineCoordinateEquiv C.chart)
83       C.chart_one (witness_q0_residue_eq_zero hm P w) w.halg
84   let coordSet : Set F := Set.range (componentCoordinate P)
85   let A := Algebra.adjoin k coordSet
86   let E := IntermediateField.adjoin k coordSet
87   letI : Algebra A E := (Subalgebra.inclusion
88     (IntermediateField.algebra_adjoin_le_adjoin k coordSet)).toAlgebra
89   letI : Algebra E F := E.toSubalgebra.toAlgebra
90   letI : Algebra A F := A.val.toAlgebra
91   have hgen : E = ⊤ := componentCoordinate_adjoin_eq_top P
92   have hsurj : Function.Surjective (algebraMap E F) := by
93     intro x
94     have hx : x ∈ E := by rw [hgen]; simp
95     exact ⟨⟨x, hx⟩, rfl⟩
96   have hEF : Algebra.IsAlgebraic E F := by
97     constructor
98     intro x
99     obtain ⟨y, hy⟩ := hsurj x
100     rw [← hy]
101     exact isAlgebraic_algebraMap y
102   have hAE : Algebra.IsAlgebraic A E :=
103     Stafford38.Geometry.RelativeDivisorialTower.intermediateAdjoin_isAlgebraic_over_algebraAdjoin
104       k coordSet
105   have hTower : IsScalarTower A E F :=
106     Subalgebra.inclusion.isScalarTower_right
107       (IntermediateField.algebra_adjoin_le_adjoin k coordSet) F
108   have hAF : Algebra.IsAlgebraic A F :=
109     @Algebra.IsAlgebraic.trans A E F _ _ _ _ _ _ hTower _ hAE hEF
110   have hAF' : Algebra.IsAlgebraic (Algebra.adjoin k coordSet) F := by
111     simpa [A] using hAF
112   obtain ⟨s, hsSubset, hsBasis⟩ :=
113     @exists_isTranscendenceBasis_subset k F inferInstance inferInstance
114       inferInstance inferInstance inferInstance coordSet hAF'
115   have hsFinite : s.Finite := (Set.finite_range _).subset hsSubset
116   letI : Fintype s := hsFinite.fintype
117   letI : Fintype t := htFinite.fintype
118   have hrank : Module.finrank F (KaehlerDifferential k F) ≤
119       Module.finrank κ (KaehlerDifferential k κ) + 1 :=
120     component_field_rank_of_witness hm P w
121   rw [kaehlerFinrankOfTranscendenceBasis k F s (fun z => z) hsBasis,
122       kaehlerFinrankOfTranscendenceBasis k κ t (fun z => z) htBasis] at hrank
123   have hcard : (Fintype.card s : Cardinal) ≤ Fintype.card t + 1 := by
124     exact_mod_cast hrank
125   have hsCard : Algebra.trdeg k F = Fintype.card s := by
126     rw [← hsBasis.cardinalMk_eq_trdeg]
127     exact Cardinal.mk_fintype (↥s)
128   have htCard : Algebra.trdeg k κ = Fintype.card t := by
129     rw [← htBasis.cardinalMk_eq_trdeg]
130     exact Cardinal.mk_fintype (↥t)
131   calc
132     Algebra.trdeg k F = Fintype.card s := hsCard
133     _ ≤ Fintype.card t + 1 := hcard
134     _ = Algebra.trdeg k κ + 1 := by rw [← htCard]
135 
136 end
137 end Stafford38.Geometry.SameWitnessTranscendenceDegreeBound
