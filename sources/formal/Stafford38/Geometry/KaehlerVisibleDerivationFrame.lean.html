8 /-!
9 # Derivation frames visible in residue coordinates
10 
11 Let `K/k` be a field extension and let `qbar i : K` be a finite family whose
12 Kähler differentials span `Ω[K/k]`.  A basis can then be selected from that
13 family.  Its dual coordinate maps correspond, by the universal property of
14 Kähler differentials, to `k`-derivations of `K`.  These derivations evaluate
15 on the selected coordinates as the Kronecker delta.
16 
17 Applying the derivations coefficientwise to power-series lifts of the
18 coordinates gives a selected residue minor whose determinant has constant
19 coefficient exactly one.
20 
21 This file assumes the spanning hypothesis.  In particular, it does not claim
22 that an arbitrary retained DVR or boundary place supplies it.
23 -/

37 /-- If finitely many residue coordinates span the Kähler differentials, one
38 can select a basis among them and choose dual derivations. -/
39 theorem exists_visible_derivation_frame_of_kaehler_span
40     {iota : Type v} [Fintype iota]
41     [FiniteDimensional K (Ω[K⁄k])]
42     (qbar : iota → K)
43     (hspan :
44       Submodule.span K
45         (Set.range fun i ↦ KaehlerDifferential.D k K (qbar i)) = ⊤) :
46     ∃ (rows : Fin (Module.finrank K (Ω[K⁄k])) ↪ iota)
47       (D : Fin (Module.finrank K (Ω[K⁄k])) → Derivation k K K),
48       ∀ i j, D j (qbar (rows i)) = if i = j then 1 else 0 := by
49   let b := Module.Basis.ofSpan (K := K)
50     (s := Set.range fun i ↦ KaehlerDifferential.D k K (qbar i)) hspan.ge
51   letI : Fintype ((linearIndepOn_empty K (id : Ω[K⁄k] → Ω[K⁄k])).extend
52       (Set.empty_subset
53         (Set.range fun i ↦ KaehlerDifferential.D k K (qbar i)))) :=
54     Fintype.ofFinite _
55   let e : Fin (Module.finrank K (Ω[K⁄k])) ≃
56       ((linearIndepOn_empty K (id : Ω[K⁄k] → Ω[K⁄k])).extend
57         (Set.empty_subset
58           (Set.range fun i ↦ KaehlerDifferential.D k K (qbar i)))) :=
59     Fintype.equivOfCardEq (by
60       rw [Fintype.card_fin, ← Module.finrank_eq_card_basis b])
61   have hb_range (a : Fin (Module.finrank K (Ω[K⁄k]))) :
62       b (e a) ∈
63         Set.range fun i ↦ KaehlerDifferential.D k K (qbar i) :=
64     Module.Basis.ofSpan_subset (K := K) (V := Ω[K⁄k]) hspan.ge ⟨e a, rfl⟩
65   let row (a : Fin (Module.finrank K (Ω[K⁄k]))) : iota :=
66     Classical.choose (hb_range a)
67   have hrow (a : Fin (Module.finrank K (Ω[K⁄k]))) :
68       KaehlerDifferential.D k K (qbar (row a)) = b (e a) :=
69     Classical.choose_spec (hb_range a)
70   have hrow_injective : Function.Injective row := by
71     intro a₁ a₂ h
72     apply e.injective
73     apply b.injective
74     rw [← hrow a₁, ← hrow a₂, h]
75   let rows : Fin (Module.finrank K (Ω[K⁄k])) ↪ iota :=
76     ⟨row, hrow_injective⟩
77   let D (j : Fin (Module.finrank K (Ω[K⁄k]))) : Derivation k K K :=
78     KaehlerDifferential.linearMapEquivDerivation k K (b.coord (e j))
79   refine ⟨rows, D, ?_⟩
80   intro i j
81   rw [show qbar (rows i) = qbar (row i) by rfl]
82   rw [show D j (qbar (row i)) =
83       b.coord (e j) (KaehlerDifferential.D k K (qbar (row i))) by
84     simp [D]]
85   rw [hrow]
86   by_cases hij : i = j
87   · subst j
88     simp
89   · have heij : e i ≠ e j := fun h ↦ hij (e.injective h)
90     simp [heij, hij]

92 /-- Coefficientwise application of a family of residue-field derivations to
93 power-series coordinates. -/
94 noncomputable def coefficientwiseTangentMatrix
95     {iota kappa : Type*}
96     (q : iota → PowerSeries K) (D : kappa → Derivation k K K) :
97     Matrix iota kappa (PowerSeries K) :=
98   fun i j ↦ coefficientwiseDerivation (D j) (q i)
99 
100 @[simp]
101 theorem constantCoeff_coefficientwiseTangentMatrix
102     {iota kappa : Type*}
103     (q : iota → PowerSeries K) (D : kappa → Derivation k K K)
104     (i : iota) (j : kappa) :
105     PowerSeries.constantCoeff (coefficientwiseTangentMatrix q D i j) =
106       D j (PowerSeries.constantCoeff (q i)) := by
107   rw [coefficientwiseTangentMatrix,
108     ← PowerSeries.coeff_zero_eq_constantCoeff]
109   exact coeff_coefficientwiseDerivation (D j) (q i) 0

111 /-- Under the Kähler-span hypothesis, coefficientwise derivations furnish a
112 selected power-series minor with constant coefficient exactly one. -/
113 theorem exists_coefficientwise_selectedMinor_constantCoeff_eq_one_of_kaehler_span
114     {iota : Type v} [Fintype iota]
115     [FiniteDimensional K (Ω[K⁄k])]
116     (q : iota → PowerSeries K)
117     (hspan :
118       Submodule.span K
119         (Set.range fun i ↦ KaehlerDifferential.D k K
120           (PowerSeries.constantCoeff (q i))) = ⊤) :
121     ∃ (rows : Fin (Module.finrank K (Ω[K⁄k])) ↪ iota)
122       (D : Fin (Module.finrank K (Ω[K⁄k])) → Derivation k K K),
123       (∀ i j,
124         D j (PowerSeries.constantCoeff (q (rows i))) =
125           if i = j then 1 else 0) ∧
126       PowerSeries.constantCoeff
127         (selectedMinor (coefficientwiseTangentMatrix q D) rows).det = 1 := by
128   obtain ⟨rows, D, hD⟩ :=
129     exists_visible_derivation_frame_of_kaehler_span
130       (fun i ↦ PowerSeries.constantCoeff (q i)) hspan
131   refine ⟨rows, D, hD, ?_⟩
132   rw [constantCoeff_selectedMinor_det]
133   have hmatrix :
134       selectedMinor
135           (residueMatrix (coefficientwiseTangentMatrix q D)) rows =
136         (1 : Matrix (Fin (Module.finrank K (Ω[K⁄k])))
137           (Fin (Module.finrank K (Ω[K⁄k]))) K) := by
138     ext i j
139     simpa [selectedMinor, residueMatrix, Matrix.one_apply,
140       constantCoeff_coefficientwiseTangentMatrix] using hD i j
141   rw [hmatrix, Matrix.det_one]
