7 /-!
8 # The projective cone of an affine component
9 
10 For a prime affine component with function field `F`, this file evaluates
11 homogeneous coordinates on the generic cone point
12 
13 `(T, x₀ T, ..., x_{m-1} T) ∈ F[T]^(m+1)`.
14 
15 The kernel is a prime homogeneous-coordinate ideal, saturated by the zeroth
16 coordinate. Zeroth-chart dehomogenization recovers the affine component ideal
17 exactly.
18 -/

31 /-- The function field of the prime affine component `P`. -/
32 abbrev ComponentFractionField
33     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :=
34   FractionRing (MvPolynomial (Fin m) k ⧸ P.asIdeal)
35 
36 /-- Evaluation on the generic affine cone over a prime component:
37 `X₀ ↦ T` and `X_(j+1) ↦ x_j T`. -/
38 def componentProjectiveConeMap
39     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
40     MvPolynomial (Fin (m + 1)) k →+* Polynomial (ComponentFractionField P) :=
41   MvPolynomial.eval₂Hom
42     ((Polynomial.C : ComponentFractionField P →+* Polynomial
43       (ComponentFractionField P)).comp (algebraMap k (ComponentFractionField P)))
44     (Fin.cases Polynomial.X fun j ↦
45       Polynomial.C (componentCoordinate P j) * Polynomial.X)
46 
47 /-- The homogeneous-coordinate ideal of the generic projective cone point. -/
48 def componentProjectiveClosureIdeal
49     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
50     Ideal (MvPolynomial (Fin (m + 1)) k) :=
51   RingHom.ker (componentProjectiveConeMap P)
52 
53 /-- Evaluation at the generic point of the affine component. -/
54 def componentAffineGenericPointMap
55     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
56     MvPolynomial (Fin m) k →+* ComponentFractionField P :=
57   (algebraMap (MvPolynomial (Fin m) k ⧸ P.asIdeal)
58       (ComponentFractionField P)).comp
59     (Ideal.Quotient.mk P.asIdeal)

68 /-- Generic-point evaluation agrees with multivariate evaluation at the
69 component coordinates. -/
70 theorem componentAffineGenericPointMap_eq_eval₂
71     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
72     (f : MvPolynomial (Fin m) k) :
73     componentAffineGenericPointMap P f =
74       MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
75         (fun j ↦ componentCoordinate P j) f := by
76   rw [MvPolynomial.map_mvPolynomial_eq_eval₂
77     (componentAffineGenericPointMap P) f]
78   apply MvPolynomial.eval₂Hom_congr
79   · ext c
80     exact IsScalarTower.algebraMap_apply k
81       (MvPolynomial (Fin m) k ⧸ P.asIdeal) (ComponentFractionField P) c
82   · funext j
83     rfl
84   · rfl

86 /-- Every equation of the affine component vanishes at its generic point. -/
87 theorem componentAffineGenericPointMap_eq_zero_of_mem
88     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
89     {f : MvPolynomial (Fin m) k} (hf : f ∈ P.asIdeal) :
90     componentAffineGenericPointMap P f = 0 := by
91   change algebraMap (MvPolynomial (Fin m) k ⧸ P.asIdeal)
92       (ComponentFractionField P) (Ideal.Quotient.mk P.asIdeal f) = 0
93   rw [Ideal.Quotient.eq_zero_iff_mem.mpr hf, map_zero]
94 
95 /-- The generic-point map has exactly the prime component as its kernel. -/
96 theorem componentAffineGenericPointMap_ker
97     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
98     RingHom.ker (componentAffineGenericPointMap P) = P.asIdeal := by
99   apply le_antisymm
100   · intro f hf
101     have hzero : Ideal.Quotient.mk P.asIdeal f = 0 := by
102       apply IsFractionRing.injective
103         (MvPolynomial (Fin m) k ⧸ P.asIdeal) (ComponentFractionField P)
104       simpa only [map_zero, RingHom.mem_ker,
105         componentAffineGenericPointMap, RingHom.comp_apply] using hf
106     exact Ideal.Quotient.eq_zero_iff_mem.mp hzero
107   · intro f hf
108     exact componentAffineGenericPointMap_eq_zero_of_mem P hf

123 /-- The cone kernel is prime because its target is a polynomial ring over the
124 component function field. -/
125 theorem componentProjectiveClosureIdeal_isPrime
126     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
127     (componentProjectiveClosureIdeal P).IsPrime := by
128   exact RingHom.ker_isPrime (componentProjectiveConeMap P)

130 /-- The cone kernel is saturated by the homogeneous zeroth coordinate.  This
131 is the scheme-free saturation property needed at the projective boundary. -/
132 theorem componentProjectiveClosureIdeal_colon_zeroCoordinate
133     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) :
134     (componentProjectiveClosureIdeal P).colon
135         ({MvPolynomial.X 0} :
136           Set (MvPolynomial (Fin (m + 1)) k)) =
137       componentProjectiveClosureIdeal P := by
138   ext H
139   rw [Submodule.mem_colon_singleton]
140   change componentProjectiveConeMap P (H * MvPolynomial.X 0) = 0 ↔
141     componentProjectiveConeMap P H = 0
142   rw [map_mul, componentProjectiveConeMap_X_zero]
143   constructor
144   · intro h
145     apply Polynomial.ext
146     intro n
147     have hn := congrArg
148       (fun q : Polynomial (ComponentFractionField P) ↦ q.coeff (n + 1)) h
149     simpa using hn
150   · intro h
151     rw [h, zero_mul]

153 /-- Scaling every variable of a homogeneous multivariate polynomial by the
154 same scalar extracts the corresponding power of that scalar. -/
155 theorem eval₂_mul_common_of_isHomogeneous
156     {R S : Type*} [CommSemiring R] [CommSemiring S]
157     {σ : Type*} {p : MvPolynomial σ R} {d : ℕ}
158     (hp : p.IsHomogeneous d) (f : R →+* S) (x : σ → S) (t : S) :
159     MvPolynomial.eval₂ f (fun i ↦ x i * t) p =
160       MvPolynomial.eval₂ f x p * t ^ d := by
161   classical
162   rw [MvPolynomial.eval₂_eq, MvPolynomial.eval₂_eq, Finset.sum_mul]
163   apply Finset.sum_congr rfl
164   intro a ha
165   have hdegree : a.degree = d := by
166     rw [Finsupp.degree_eq_weight_one]
167     exact hp (MvPolynomial.mem_support_iff.mp ha)
168   rw [mul_assoc]
169   congr 1
170   simp_rw [mul_pow]
171   rw [Finset.prod_mul_distrib, Finset.prod_pow_eq_pow_sum]
172   change (∏ i ∈ a.support, x i ^ a i) * t ^ a.degree = _
173   rw [hdegree]

175 /-- Evaluation after zeroth-chart dehomogenization is evaluation at the
176 projective point `(1,y)`, allowing a larger target coefficient ring. -/
177 theorem eval₂_projectiveDehomogenize
178     {S : Type*} [CommSemiring S] (f : k →+* S) (y : Fin m → S)
179     (H : MvPolynomial (Fin (m + 1)) k) :
180     MvPolynomial.eval₂ f y
181         (Stafford38.Geometry.ProjectiveEquationFormalChart.projectiveDehomogenize H) =
182       MvPolynomial.eval₂ f (Fin.cases 1 y) H := by
183   rw [show
184       Stafford38.Geometry.ProjectiveEquationFormalChart.projectiveDehomogenize H =
185         MvPolynomial.bind₁
186           (Fin.cases 1 fun i ↦ MvPolynomial.X i) H by
187     simp [Stafford38.Geometry.ProjectiveEquationFormalChart.projectiveDehomogenize]]
188   change MvPolynomial.eval₂Hom f y
189       (MvPolynomial.bind₁ (Fin.cases 1 fun i ↦ MvPolynomial.X i) H) = _
190   rw [MvPolynomial.eval₂Hom_bind₁]
191   apply MvPolynomial.eval₂_congr
192   intro i
193   refine Fin.cases ?_ (fun j ↦ ?_) i <;> simp

195 /-- Evaluating a homogeneous projective polynomial on the generic cone point
196 is dehomogenized evaluation times the expected power of `T`. -/
197 theorem componentProjectiveConeMap_eq_eval₂_mul_X_pow_of_isHomogeneous
198     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
199     {H : MvPolynomial (Fin (m + 1)) k} {d : ℕ}
200     (hH : H.IsHomogeneous d) :
201     componentProjectiveConeMap P H =
202       Polynomial.C
203           (MvPolynomial.eval₂ (algebraMap k (ComponentFractionField P))
204             (Fin.cases 1 fun j ↦ componentCoordinate P j) H) *
205         Polynomial.X ^ d := by
206   change MvPolynomial.eval₂
207       ((Polynomial.C : ComponentFractionField P →+* Polynomial
208         (ComponentFractionField P)).comp (algebraMap k (ComponentFractionField P)))
209       (Fin.cases Polynomial.X fun j ↦
210         Polynomial.C (componentCoordinate P j) * Polynomial.X) H = _
211   let coords : Fin (m + 1) → ComponentFractionField P :=
212     Fin.cases 1 fun j ↦ componentCoordinate P j
213   have hvars :
214       (fun a ↦ Polynomial.C (coords a) * Polynomial.X) =
215         Fin.cases Polynomial.X
216           (fun j ↦ Polynomial.C (componentCoordinate P j) * Polynomial.X) := by
217     funext a
218     refine Fin.cases ?_ (fun j ↦ ?_) a
219     · simp [coords]
220     · simp [coords]
221   rw [← hvars]
222   have hscale := eval₂_mul_common_of_isHomogeneous hH
223     ((Polynomial.C : ComponentFractionField P →+* Polynomial
224       (ComponentFractionField P)).comp (algebraMap k (ComponentFractionField P)))
225     (fun a ↦ Polynomial.C (coords a)) Polynomial.X
226   rw [hscale]
227   congr 1
228   exact (MvPolynomial.map_eval₂Hom
229     (algebraMap k (ComponentFractionField P)) coords Polynomial.C H).symm

231 /-- The cone-map image of the standard homogenization is generic affine
232 evaluation times the expected power of `T`. -/
233 theorem componentProjectiveConeMap_homogenizeAtZero
234     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
235     (f : MvPolynomial (Fin m) k) :
236     componentProjectiveConeMap P (homogenizeAtZero f) =
237       Polynomial.C (componentAffineGenericPointMap P f) *
238         Polynomial.X ^ f.totalDegree := by
239   rw [componentProjectiveConeMap_eq_eval₂_mul_X_pow_of_isHomogeneous P
240     (homogenizeAtZero_isHomogeneous f)]
241   congr 1
242   rw [componentAffineGenericPointMap_eq_eval₂]
243   have heval := eval₂_projectiveDehomogenize
244     (algebraMap k (ComponentFractionField P))
245     (fun j ↦ componentCoordinate P j) (homogenizeAtZero f)
246   rw [projectiveDehomogenize_homogenizeAtZero] at heval
247   exact congrArg Polynomial.C heval.symm

249 /-- Setting the cone parameter to one recovers generic-point evaluation in
250 the zeroth affine chart. -/
251 theorem componentProjectiveConeMap_eval_one
252     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
253     (H : MvPolynomial (Fin (m + 1)) k) :
254     Polynomial.eval 1 (componentProjectiveConeMap P H) =
255       componentAffineGenericPointMap P
256         (Stafford38.Geometry.ProjectiveEquationFormalChart.projectiveDehomogenize H) := by
257   rw [componentAffineGenericPointMap_eq_eval₂]
258   rw [eval₂_projectiveDehomogenize]
259   change Polynomial.evalRingHom 1
260       (MvPolynomial.eval₂
261         ((Polynomial.C : ComponentFractionField P →+* Polynomial
262           (ComponentFractionField P)).comp
263             (algebraMap k (ComponentFractionField P)))
264         (Fin.cases Polynomial.X fun j ↦
265           Polynomial.C (componentCoordinate P j) * Polynomial.X) H) = _
266   rw [MvPolynomial.eval₂_comp_left]
267   change MvPolynomial.eval₂Hom _ _ H = MvPolynomial.eval₂Hom _ _ H
268   apply MvPolynomial.eval₂Hom_congr
269   · ext c
270     simp
271   · funext i
272     refine Fin.cases ?_ (fun j ↦ ?_) i <;> simp
273   · rfl

275 /-- Every homogeneous-coordinate equation in the cone kernel dehomogenizes
276 to an equation of the original affine component. -/
277 theorem projectiveDehomogenize_mem_of_mem_componentProjectiveClosureIdeal
278     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
279     {H : MvPolynomial (Fin (m + 1)) k}
280     (hH : H ∈ componentProjectiveClosureIdeal P) :
281     Stafford38.Geometry.ProjectiveEquationFormalChart.projectiveDehomogenize H ∈
282       P.asIdeal := by
283   rw [← componentAffineGenericPointMap_ker P, RingHom.mem_ker]
284   rw [← componentProjectiveConeMap_eval_one P H]
285   have hzero : componentProjectiveConeMap P H = 0 := by
286     exact hH
287   rw [hzero]
288   simp
