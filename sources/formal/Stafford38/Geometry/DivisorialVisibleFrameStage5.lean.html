21 theorem stage5_exists_coefficientField
22     {k K : Type u} [Field k] [Field K] [Algebra k K]
23     (A : Subalgebra k K) [Algebra.FiniteType k A] (p : Ideal A) [p.IsPrime]
24     (V : ValuationSubring K) [IsLocalRing V.toSubring]
25     (hAV : ∀ a : A, (a : K) ∈ V.toSubring)
26     (hp : ∀ a : A, (⟨a, hAV a⟩ : V.toSubring) ∈ maximalIdeal V.toSubring ↔ a ∈ p)
27     (hsurj : ∀ z : ResidueField V.toSubring, ∃ (a b : A), b ∉ p ∧
28       z = residue V.toSubring ⟨a, hAV a⟩ / residue V.toSubring ⟨b, hAV b⟩) :
29     ∃ (E : IntermediateField k K) (hEV : ∀ z : E, (z : K) ∈ V.toSubring),
30       letI : Algebra E V.toSubring := (coeffHom E V hEV).toAlgebra
31       Module.Finite E (ResidueField V.toSubring) := by
32   classical
33   let AV : A →+* V.toSubring :=
34     RingHom.codRestrict (Subalgebra.val A).toRingHom V.toSubring hAV
35   letI : Algebra k V.toSubring := (AV.comp (algebraMap k A)).toAlgebra
36   letI : Algebra k (ResidueField V.toSubring) :=
37     ((residue V.toSubring).comp (algebraMap k V.toSubring)).toAlgebra
38   letI : Nontrivial (A ⧸ p) :=
39     Ideal.Quotient.nontrivial_iff.mpr (inferInstance : p.IsPrime).ne_top
40   obtain ⟨s, g, hg, hgfin⟩ := exists_finite_inj_algHom_of_fg k (A ⧸ p)
41   choose a ha using fun i : Fin s ↦ Ideal.Quotient.mk_surjective (g (MvPolynomial.X i))
42   let t : Fin s → K := fun i ↦ (a i : K)
43   have heval : MvPolynomial.aeval t =
44       (Subalgebra.val A).comp (MvPolynomial.aeval a) := by
45     ext i
46     simp [t]
47   have hg_eval : g = (Ideal.Quotient.mkₐ k p).comp (MvPolynomial.aeval a) := by
48     ext i
49     simpa using (ha i).symm
50   have ht_independent : AlgebraicIndependent k t :=
51     algebraicIndependent_iff_injective_aeval.mpr fun f q hfq ↦ by
52       apply hg
53       rw [hg_eval]
54       apply congrArg (fun x : A ↦ Ideal.Quotient.mk p x)
55       apply Subtype.ext
56       exact (congrArg (fun e : MvPolynomial (Fin s) k →ₐ[k] K ↦ e f) heval).symm |>.trans
57         hfq |>.trans (congrArg (fun e : MvPolynomial (Fin s) k →ₐ[k] K ↦ e q) heval)
58   let E : IntermediateField k K := IntermediateField.adjoin k (Set.range t)
59   have eval_mem (f : MvPolynomial (Fin s) k) :
60       MvPolynomial.aeval t f ∈ V.toSubring := by
61     rw [heval]
62     exact hAV (MvPolynomial.aeval a f)
63   have eval_ne_p {f : MvPolynomial (Fin s) k} (hf : f ≠ 0) :
64       MvPolynomial.aeval a f ∉ p := by
65     intro hfp
66     have hgf : g f = 0 := by
67       rw [hg_eval]
68       exact Ideal.Quotient.eq_zero_iff_mem.mpr hfp
69     exact hf (hg (hgf.trans (map_zero g).symm))
70   have inv_eval_mem (f : MvPolynomial (Fin s) k) :
71       (MvPolynomial.aeval t f)⁻¹ ∈ V.toSubring := by
72     by_cases hf : f = 0
73     · simp [hf]
74     · let vf : V.toSubring :=
75         ⟨(MvPolynomial.aeval a f : A), hAV (MvPolynomial.aeval a f)⟩
76       have hvf : ((vf : V.toSubring) : K) = MvPolynomial.aeval t f := by
77         change A.val (MvPolynomial.aeval a f) = MvPolynomial.aeval t f
78         exact (congrArg (fun e : MvPolynomial (Fin s) k →ₐ[k] K ↦ e f) heval).symm
79       have hv_not_mem : vf ∉ maximalIdeal V.toSubring :=
80         (hp (MvPolynomial.aeval a f)).not.mpr (eval_ne_p hf)
81       have hv_unit : IsUnit vf := by
82         simpa only [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff, not_not] using hv_not_mem
83       obtain ⟨w, hw⟩ := hv_unit
84       have hinv : ((((w⁻¹ : (V.toSubring)ˣ) : V.toSubring) : K)) =
85           (MvPolynomial.aeval t f)⁻¹ := by
86         rw [← hvf, ← hw]
87         change (↑(Units.map V.toSubring.subtype.toMonoidHom w⁻¹) : K) =
88           (↑(Units.map V.toSubring.subtype.toMonoidHom w) : K)⁻¹
89         rw [map_inv]
90         exact Units.val_inv_eq_inv_val _
91       rw [← hinv]
92       exact ((w⁻¹ : (V.toSubring)ˣ) : V.toSubring).property
93   have hEV : ∀ z : E, (z : K) ∈ V.toSubring := by
94     rintro ⟨z, hz⟩
95     change z ∈ V.toSubring
96     obtain ⟨r, q, hz⟩ := (IntermediateField.mem_adjoin_range_iff k t z).mp hz
97     rw [hz, div_eq_mul_inv]
98     exact V.toSubring.mul_mem (eval_mem r) (inv_eval_mem q)
99   refine ⟨E, hEV, ?_⟩
100   letI : Algebra E V.toSubring := (coeffHom E V hEV).toAlgebra
101   let ρ : A →+* ResidueField V.toSubring := (residue V.toSubring).comp AV
102   have hpker : ∀ x ∈ p, ρ x = 0 := by
103     intro x hx
104     exact (IsLocalRing.residue_eq_zero_iff _).mpr ((hp x).mpr hx)
105   let ρbar : (A ⧸ p) →+* ResidueField V.toSubring := Ideal.Quotient.lift p ρ hpker
106   let et : Fin s → E := fun i ↦
107     ⟨t i, IntermediateField.subset_adjoin k (Set.range t) ⟨i, rfl⟩⟩
108   have halgebraMap_et (i : Fin s) :
109       algebraMap E (ResidueField V.toSubring) (et i) = ρ (a i) := by
110     rfl
111   haveI hkEV : IsScalarTower k E V.toSubring :=
112     IsScalarTower.of_algebraMap_eq fun _ ↦ rfl
113   haveI hEVκ : IsScalarTower E V.toSubring (ResidueField V.toSubring) :=
114     IsScalarTower.of_algebraMap_eq fun _ ↦ rfl
115   haveI hkEκ : IsScalarTower k E (ResidueField V.toSubring) :=
116     IsScalarTower.of_algebraMap_eq fun _ ↦ rfl
117   let ρbarK : (A ⧸ p) →ₐ[k] ResidueField V.toSubring :=
118     { toRingHom := ρbar
119       commutes' := fun c ↦ by
120         change ρbar (Ideal.Quotient.mk p (algebraMap k A c)) = _
121         rw [Ideal.Quotient.lift_mk]
122         rfl }
123   let P := MvPolynomial (Fin s) k
124   letI : Algebra P (A ⧸ p) := g.toRingHom.toAlgebra
125   letI : Algebra P E := (MvPolynomial.aeval et).toRingHom.toAlgebra
126   letI : Algebra P (ResidueField V.toSubring) :=
127     ((algebraMap E (ResidueField V.toSubring)).comp
128       (MvPolynomial.aeval et).toRingHom).toAlgebra
129   haveI hPEκ : IsScalarTower P E (ResidueField V.toSubring) :=
130     IsScalarTower.of_algebraMap_eq fun _ ↦ rfl
131   have hpoly : ρbarK.comp g =
132       (IsScalarTower.toAlgHom k E (ResidueField V.toSubring)).comp
133         (MvPolynomial.aeval et) := by
134     ext i
135     simp only [AlgHom.comp_apply, MvPolynomial.aeval_X]
136     change ρbar (g (MvPolynomial.X i)) = algebraMap E (ResidueField V.toSubring) (et i)
137     rw [show g (MvPolynomial.X i) = Ideal.Quotient.mk p (a i) from (ha i).symm,
138       Ideal.Quotient.lift_mk]
139     exact (halgebraMap_et i).symm
140   let ρbarP : (A ⧸ p) →ₐ[P] ResidueField V.toSubring :=
141     { toRingHom := ρbar
142       commutes' := fun f ↦ by
143         exact congrArg (fun e : MvPolynomial (Fin s) k →ₐ[k]
144           ResidueField V.toSubring ↦ e f) hpoly }
145   letI hPB : Module.Finite P (A ⧸ p) := hgfin
146   have residue_integral (x : A) : IsIntegral E (ρ x) := by
147     have hx : IsIntegral P (Ideal.Quotient.mk p x) := IsIntegral.of_finite P _
148     have hx' : IsIntegral E (ρbarP (Ideal.Quotient.mk p x)) :=
149       (hx.map ρbarP).tower_top
150     convert hx' using 1
151     change ρbar (Ideal.Quotient.mk p x) = ρ x
152     exact Ideal.Quotient.lift_mk p ρ hpker
153   obtain ⟨S, hS⟩ := (inferInstance : Algebra.FiniteType k A).out
154   let gen : S → ResidueField V.toSubring := fun x ↦ ρ x.1
155   let L : IntermediateField E (ResidueField V.toSubring) :=
156     IntermediateField.adjoin E (Set.range gen)
157   have rho_mem (x : A) : ρ x ∈ L := by
158     have hx : x ∈ Algebra.adjoin k (S : Set A) := by
159       rw [hS]
160       trivial
161     induction hx using Algebra.adjoin_induction with
162     | mem x hx =>
163         exact IntermediateField.subset_adjoin E (Set.range gen)
164           ⟨⟨x, hx⟩, rfl⟩
165     | algebraMap c =>
166         change ρ (algebraMap k A c) ∈ L
167         rw [show ρ (algebraMap k A c) =
168           algebraMap k (ResidueField V.toSubring) c from rfl,
169           IsScalarTower.algebraMap_apply k E (ResidueField V.toSubring)]
170         exact L.algebraMap_mem _
171     | add x y _ _ hx hy => exact L.add_mem hx hy
172     | mul x y _ _ hx hy => exact L.mul_mem hx hy
173   have hL : L = ⊤ := by
174     apply top_unique
175     intro z _
176     obtain ⟨x, y, hy, hz⟩ := hsurj z
177     rw [hz]
178     exact L.div_mem (rho_mem x) (rho_mem y)
179   have gen_integral (z : ResidueField V.toSubring) (hz : z ∈ Set.range gen) :
180       IsIntegral E z := by
181     obtain ⟨x, rfl⟩ := hz
182     exact residue_integral x.1
183   letI hfiniteL : Module.Finite E L :=
184     IntermediateField.finiteDimensional_adjoin gen_integral
185   let e : L ≃ₐ[E] ResidueField V.toSubring :=
186     (IntermediateField.equivOfEq hL).trans IntermediateField.topEquiv
187   exact Module.Finite.equiv e.toLinearEquiv
