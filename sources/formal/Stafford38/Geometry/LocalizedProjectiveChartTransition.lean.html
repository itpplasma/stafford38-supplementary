7 /-!
8 # Localized transition between projective affine charts
9 
10 The coordinate ring of the overlap between the zeroth affine chart and a
11 chosen chart is obtained by inverting the chosen-chart coordinate representing
12 `X₀ / X_chart`.  This file constructs the resulting transition homomorphism.
13 -/

27 /-! ## Zeroth-chart homogenization -/
28 
29 /-- Homogenize an affine polynomial in the zeroth projective coordinate by
30 padding each homogeneous component to the total degree of the polynomial. -/
31 def homogenizeAtZero {m : ℕ} (f : MvPolynomial (Fin m) K) :
32     MvPolynomial (Fin (m + 1)) K :=
33   ∑ i ∈ Finset.range (f.totalDegree + 1),
34     MvPolynomial.X 0 ^ (f.totalDegree - i) *
35       MvPolynomial.rename Fin.succ (MvPolynomial.homogeneousComponent i f)
36 
37 /-- The explicit homogenization is homogeneous of the expected degree. -/
38 theorem homogenizeAtZero_isHomogeneous {m : ℕ}
39     (f : MvPolynomial (Fin m) K) :
40     (homogenizeAtZero f).IsHomogeneous f.totalDegree := by
41   apply MvPolynomial.IsHomogeneous.sum
42   intro i hi
43   have hi_le : i ≤ f.totalDegree := Nat.le_of_lt_succ (Finset.mem_range.mp hi)
44   have hpow := MvPolynomial.isHomogeneous_X_pow
45     (R := K) (0 : Fin (m + 1)) (f.totalDegree - i)
46   have hcomponent :=
47     (MvPolynomial.homogeneousComponent_isHomogeneous i f).rename_isHomogeneous
48       (f := Fin.succ)
49   convert hpow.mul hcomponent using 1
50   omega

52 /-- Dehomogenizing the constructed projective polynomial in chart zero
53 recovers the original affine polynomial exactly. -/
54 theorem projectiveDehomogenize_homogenizeAtZero {m : ℕ}
55     (f : MvPolynomial (Fin m) K) :
56     Stafford38.Geometry.ProjectiveEquationFormalChart.projectiveDehomogenize
57         (homogenizeAtZero f) = f := by
58   rw [homogenizeAtZero]
59   simp only [map_sum, map_mul, map_pow]
60   have hcomponent : ∀ i,
61       projectiveDehomogenize
62           (MvPolynomial.rename Fin.succ
63             (MvPolynomial.homogeneousComponent i f)) =
64         MvPolynomial.homogeneousComponent i f := by
65     intro i
66     rw [show projectiveDehomogenize
67         (MvPolynomial.rename Fin.succ (MvPolynomial.homogeneousComponent i f)) =
68       MvPolynomial.bind₁ (Fin.cases 1 fun j ↦ MvPolynomial.X j)
69         (MvPolynomial.rename Fin.succ
70           (MvPolynomial.homogeneousComponent i f)) by
71           simp [projectiveDehomogenize]]
72     rw [MvPolynomial.bind₁_rename]
73     have hfun :
74         ((fun a : Fin (m + 1) ↦
75             Fin.cases 1 (fun j ↦ MvPolynomial.X (R := K) j) a) ∘
76             Fin.succ) = MvPolynomial.X (R := K) := by
77       funext j
78       simp
79     rw [hfun, MvPolynomial.bind₁_X_left]
80     rfl
81   simp_rw [hcomponent]
82   simp [projectiveDehomogenize]
83   exact MvPolynomial.sum_homogeneousComponent f

85 /-- Every affine ideal over a field has a finite homogeneous projective
86 equation family whose zeroth-chart dehomogenized ideal is exactly the given
87 ideal.  This is the algebraic homogenization step; it does not assert that the
88 resulting homogeneous ideal is saturated. -/
89 theorem exists_finite_homogeneous_equations_dehomogenizing_to
90     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) K)) :
91     ∃ (n : ℕ) (equations : Fin n → MvPolynomial (Fin (m + 1)) K)
92         (degree : Fin n → ℕ),
93       (∀ j, (equations j).IsHomogeneous (degree j)) ∧
94       dehomogenizedEquationIdeal equations = I := by
95   obtain ⟨n, generators, hgenerators⟩ :=
96     Submodule.fg_iff_exists_fin_generating_family.mp
97       (IsNoetherian.noetherian (I : Submodule _ _))
98   refine ⟨n, fun j ↦ homogenizeAtZero (generators j),
99     fun j ↦ (generators j).totalDegree, ?_, ?_⟩
100   · intro j
101     exact homogenizeAtZero_isHomogeneous (generators j)
102   · rw [dehomogenizedEquationIdeal,
103       Stafford38.Geometry.ProjectiveChartCoordinates.dehomogenizedEquationIdeal]
104     simp_rw [projectiveDehomogenize_homogenizeAtZero]
105     change Submodule.span _ (Set.range generators) = (I : Submodule _ _)
106     exact hgenerators

108 /-- The projective coordinate `X_a/X_chart`, represented in the polynomial
109 coordinate ring of the chosen affine chart. -/
110 def chartProjectiveCoordinate {m : ℕ} (chart a : Fin (m + 1)) :
111     MvPolynomial (Fin m) K :=
112   if h : a = chart then 1
113   else MvPolynomial.X ((chartAffineCoordinateEquiv chart).symm ⟨a, h⟩)
114 
115 /-- The chosen-chart polynomial representing `X₀/X_chart`. -/
116 def chartOverlapDenominator {m : ℕ} (chart : Fin (m + 1)) :
117     MvPolynomial (Fin m) K :=
118   chartProjectiveCoordinate chart 0
119 
120 /-- Coordinate ring of the overlap of the zeroth and chosen projective
121 affine charts. -/
122 abbrev ChartOverlapRing {m : ℕ} (chart : Fin (m + 1)) :=
123   Localization.Away (chartOverlapDenominator (K := K) chart)
124 
125 /-- The regular chart transition on the overlap.  The old zeroth-chart
126 coordinate `X_(i+1)/X₀` becomes
127 `(X_(i+1)/X_chart) / (X₀/X_chart)`. -/
128 def chartZeroToChosenOverlap {m : ℕ} (chart : Fin (m + 1)) :
129     MvPolynomial (Fin m) K →+* ChartOverlapRing (K := K) chart :=
130   MvPolynomial.eval₂Hom
131     (algebraMap K (ChartOverlapRing (K := K) chart))
132     (fun i ↦
133       algebraMap (MvPolynomial (Fin m) K)
134           (ChartOverlapRing (K := K) chart)
135           (chartProjectiveCoordinate chart i.succ) *
136         IsLocalization.Away.invSelf
137           (chartOverlapDenominator (K := K) chart))
138 
139 theorem chartZeroToChosenOverlap_X {m : ℕ} (chart : Fin (m + 1))
140     (i : Fin m) :
141     chartZeroToChosenOverlap (K := K) chart (MvPolynomial.X i) =
142       algebraMap (MvPolynomial (Fin m) K)
143           (ChartOverlapRing (K := K) chart)
144           (chartProjectiveCoordinate chart i.succ) *
145         IsLocalization.Away.invSelf
146           (chartOverlapDenominator (K := K) chart) := by
147   simp [chartZeroToChosenOverlap]
148 
149 theorem chartOverlapDenominator_mul_invSelf {m : ℕ}
150     (chart : Fin (m + 1)) :
151     algebraMap (MvPolynomial (Fin m) K)
152         (ChartOverlapRing (K := K) chart)
153         (chartOverlapDenominator (K := K) chart) *
154       IsLocalization.Away.invSelf
155         (chartOverlapDenominator (K := K) chart) = 1 := by
156   exact IsLocalization.Away.mul_invSelf _

160 /-- If `s * r = 1`, scaling every coordinate by `r` scales a homogeneous
161 polynomial of degree `d` by `s⁻ᵈ` in the cancellation-free form displayed
162 below.  This version works in the overlap localization, which need not be a
163 field. -/
164 theorem eval_eq_pow_mul_eval_mul_of_homogeneous
165     {R : Type*} [CommRing R] {n : ℕ}
166     (p : MvPolynomial (Fin n) R) (d : ℕ)
167     (hp : p.IsHomogeneous d) (q : Fin n → R) (s r : R)
168     (hsr : s * r = 1) :
169     MvPolynomial.eval q p =
170       s ^ d * MvPolynomial.eval (fun i ↦ r * q i) p := by
171   change p.eval₂ (RingHom.id R) q =
172     s ^ d * p.eval₂ (RingHom.id R) (fun i ↦ r * q i)
173   rw [MvPolynomial.eval₂_eq', MvPolynomial.eval₂_eq']
174   rw [Finset.mul_sum]
175   apply Finset.sum_congr rfl
176   intro exponent hexponent
177   have hdegree : ∑ i, exponent i = d := by
178     have hweight := hp (MvPolynomial.mem_support_iff.mp hexponent)
179     rw [Finsupp.weight_apply,
180       Finsupp.sum_fintype exponent
181         (fun i c ↦ c • (1 : Fin n → ℕ) i) (by simp)] at hweight
182     simpa only [Pi.one_apply, smul_eq_mul, mul_one] using hweight
183   have hcoordinate : ∀ i : Fin n, s * (r * q i) = q i := by
184     intro i
185     rw [← mul_assoc, hsr, one_mul]
186   have hproduct : (∏ i : Fin n, q i ^ exponent i) =
187       s ^ d * ∏ i : Fin n, (r * q i) ^ exponent i := by
188     calc
189       (∏ i : Fin n, q i ^ exponent i) =
190           ∏ i : Fin n, (s * (r * q i)) ^ exponent i := by
191             simp_rw [hcoordinate]
192       _ = (∏ i : Fin n, s ^ exponent i) *
193           ∏ i : Fin n, (r * q i) ^ exponent i := by
194             simp_rw [mul_pow]
195             rw [Finset.prod_mul_distrib]
196       _ = s ^ d * ∏ i : Fin n, (r * q i) ^ exponent i := by
197             rw [Finset.prod_pow_eq_pow_sum, hdegree]
198   rw [hproduct]
199   ring

203 /-- Mapping chosen-chart dehomogenization into the overlap ring is the same
204 as evaluating the projective polynomial at its chosen-chart coordinates. -/
205 theorem algebraMap_projectiveDehomogenizeAt {m : ℕ}
206     (chart : Fin (m + 1)) (p : MvPolynomial (Fin (m + 1)) K) :
207     algebraMap (MvPolynomial (Fin m) K)
208         (ChartOverlapRing (K := K) chart)
209         (projectiveDehomogenizeAt chart p) =
210       MvPolynomial.eval₂
211         (algebraMap K (ChartOverlapRing (K := K) chart))
212         (fun a ↦ algebraMap (MvPolynomial (Fin m) K)
213           (ChartOverlapRing (K := K) chart)
214           (chartProjectiveCoordinate chart a)) p := by
215   rw [show projectiveDehomogenizeAt chart p =
216       MvPolynomial.bind₁ (chartProjectiveCoordinate (K := K) chart) p by rfl]
217   rw [MvPolynomial.hom_bind₁]
218   change MvPolynomial.eval₂ _ _ p = MvPolynomial.eval₂ _ _ p
219   apply MvPolynomial.eval₂_congr
220   intro _ _ _ _
221   rfl

223 /-- Zeroth-chart dehomogenization followed by the overlap transition is
224 projective evaluation after dividing every chosen-chart coordinate by
225 `X₀/X_chart`. -/
226 theorem chartZeroToChosenOverlap_projectiveDehomogenize {m : ℕ}
227     (chart : Fin (m + 1)) (p : MvPolynomial (Fin (m + 1)) K) :
228     chartZeroToChosenOverlap (K := K) chart (projectiveDehomogenize p) =
229       MvPolynomial.eval₂
230         (algebraMap K (ChartOverlapRing (K := K) chart))
231         (fun a ↦
232           IsLocalization.Away.invSelf
233               (chartOverlapDenominator (K := K) chart) *
234             algebraMap (MvPolynomial (Fin m) K)
235               (ChartOverlapRing (K := K) chart)
236               (chartProjectiveCoordinate chart a)) p := by
237   rw [show projectiveDehomogenize p = MvPolynomial.bind₁
238       (Fin.cases 1 fun i ↦ MvPolynomial.X i) p by
239         simp [projectiveDehomogenize]]
240   rw [MvPolynomial.hom_bind₁]
241   have hcoeff :
242       (chartZeroToChosenOverlap (K := K) chart).comp MvPolynomial.C =
243         algebraMap K (ChartOverlapRing (K := K) chart) := by
244     ext k
245     simp [chartZeroToChosenOverlap]
246   rw [hcoeff]
247   apply MvPolynomial.eval₂_congr
248   intro a _ _ _
249   refine Fin.cases ?_ (fun i ↦ ?_) a
250   · rw [Fin.cases_zero, map_one]
251     calc
252       1 = algebraMap (MvPolynomial (Fin m) K)
253           (ChartOverlapRing (K := K) chart)
254           (chartOverlapDenominator (K := K) chart) *
255           IsLocalization.Away.invSelf
256             (chartOverlapDenominator (K := K) chart) :=
257         (chartOverlapDenominator_mul_invSelf (K := K) chart).symm
258       _ = IsLocalization.Away.invSelf
259             (chartOverlapDenominator (K := K) chart) *
260           algebraMap (MvPolynomial (Fin m) K)
261             (ChartOverlapRing (K := K) chart)
262             (chartProjectiveCoordinate chart 0) := mul_comm _ _
263   · rw [Fin.cases_succ, chartZeroToChosenOverlap_X]
264     exact mul_comm _ _

266 /-- Exact overlap transition for a homogeneous projective equation.  The two
267 dehomogenizations differ by the invertible degree-`d` power of
268 `X₀/X_chart`; hence they define the same localized equation ideal. -/
269 theorem localized_projectiveDehomogenizeAt_eq_pow_mul_transition
270     {m : ℕ} (chart : Fin (m + 1))
271     (p : MvPolynomial (Fin (m + 1)) K) (d : ℕ)
272     (hp : p.IsHomogeneous d) :
273     algebraMap (MvPolynomial (Fin m) K)
274         (ChartOverlapRing (K := K) chart)
275         (projectiveDehomogenizeAt chart p) =
276       algebraMap (MvPolynomial (Fin m) K)
277           (ChartOverlapRing (K := K) chart)
278           (chartOverlapDenominator (K := K) chart) ^ d *
279         chartZeroToChosenOverlap (K := K) chart
280           (projectiveDehomogenize p) := by
281   let L := ChartOverlapRing (K := K) chart
282   let q : Fin (m + 1) → L := fun a ↦
283     algebraMap (MvPolynomial (Fin m) K) L
284       (chartProjectiveCoordinate chart a)
285   let s : L := algebraMap (MvPolynomial (Fin m) K) L
286     (chartOverlapDenominator (K := K) chart)
287   let r : L := IsLocalization.Away.invSelf
288     (chartOverlapDenominator (K := K) chart)
289   have hsr : s * r = 1 :=
290     chartOverlapDenominator_mul_invSelf (K := K) chart
291   have hscale := eval_eq_pow_mul_eval_mul_of_homogeneous
292     (MvPolynomial.map (algebraMap K L) p) d
293     (hp.map (algebraMap K L)) q s r hsr
294   rw [algebraMap_projectiveDehomogenizeAt]
295   rw [chartZeroToChosenOverlap_projectiveDehomogenize]
296   simpa [MvPolynomial.eval_map, q, s, r] using hscale

298 /-- Homogenization followed by arbitrary-chart dehomogenization is the
299 localized transition of the original affine polynomial, up to the expected
300 invertible power of `X₀/X_chart`. -/
301 theorem localized_dehomogenizeAt_homogenizeAtZero
302     {m : ℕ} (chart : Fin (m + 1))
303     (f : MvPolynomial (Fin m) K) :
304     algebraMap (MvPolynomial (Fin m) K)
305         (ChartOverlapRing (K := K) chart)
306         (projectiveDehomogenizeAt chart (homogenizeAtZero f)) =
307       algebraMap (MvPolynomial (Fin m) K)
308           (ChartOverlapRing (K := K) chart)
309           (chartOverlapDenominator (K := K) chart) ^ f.totalDegree *
310         chartZeroToChosenOverlap (K := K) chart f := by
311   rw [localized_projectiveDehomogenizeAt_eq_pow_mul_transition
312     chart (homogenizeAtZero f) f.totalDegree
313       (homogenizeAtZero_isHomogeneous f)]
314   rw [projectiveDehomogenize_homogenizeAtZero]

316 /-- The zeroth-chart equations transported to the overlap ring. -/
317 def localizedZeroChartEquationIdeal {m : ℕ} {ι : Type*}
318     (chart : Fin (m + 1))
319     (equations : ι → MvPolynomial (Fin (m + 1)) K) :
320     Ideal (ChartOverlapRing (K := K) chart) :=
321   Ideal.span (Set.range fun j ↦
322     chartZeroToChosenOverlap (K := K) chart
323       (projectiveDehomogenize (equations j)))

325 /-- The chosen-chart equations mapped to the same overlap ring. -/
326 def localizedChosenChartEquationIdeal {m : ℕ} {ι : Type*}
327     (chart : Fin (m + 1))
328     (equations : ι → MvPolynomial (Fin (m + 1)) K) :
329     Ideal (ChartOverlapRing (K := K) chart) :=
330   Ideal.span (Set.range fun j ↦
331     algebraMap (MvPolynomial (Fin m) K)
332       (ChartOverlapRing (K := K) chart)
333       (projectiveDehomogenizeAt chart (equations j)))

335 /-- Homogeneous projective equations generate the same ideal after both
336 affine charts are restricted to their overlap.  This is the corrected ideal
337 statement: equality holds after the chart-transition map and localization,
338 not between the two untransformed affine polynomial ideals. -/
339 theorem localized_chartEquationIdeals_eq
340     {m : ℕ} {ι : Type*} (chart : Fin (m + 1))
341     (equations : ι → MvPolynomial (Fin (m + 1)) K)
342     (degree : ι → ℕ)
343     (homogeneous : ∀ j, (equations j).IsHomogeneous (degree j)) :
344     localizedZeroChartEquationIdeal chart equations =
345       localizedChosenChartEquationIdeal chart equations := by
346   apply le_antisymm
347   · apply Ideal.span_le.mpr
348     rintro _ ⟨j, rfl⟩
349     let s : ChartOverlapRing (K := K) chart :=
350       algebraMap (MvPolynomial (Fin m) K)
351         (ChartOverlapRing (K := K) chart)
352         (chartOverlapDenominator (K := K) chart)
353     have hchosen :
354         algebraMap (MvPolynomial (Fin m) K)
355             (ChartOverlapRing (K := K) chart)
356             (projectiveDehomogenizeAt chart (equations j)) ∈
357           localizedChosenChartEquationIdeal chart equations :=
358       Ideal.subset_span ⟨j, rfl⟩
359     have hrelation :=
360       localized_projectiveDehomogenizeAt_eq_pow_mul_transition
361         chart (equations j) (degree j) (homogeneous j)
362     have hmultiple :
363         chartZeroToChosenOverlap (K := K) chart
364               (projectiveDehomogenize (equations j)) * s ^ degree j ∈
365           localizedChosenChartEquationIdeal chart equations := by
366       rw [mul_comm, ← hrelation]
367       exact hchosen
368     have hs : IsUnit (s ^ degree j) :=
369       (IsLocalization.Away.algebraMap_isUnit
370         (chartOverlapDenominator (K := K) chart)).pow _
371     exact (Ideal.mul_unit_mem_iff_mem _ hs).mp hmultiple
372   · apply Ideal.span_le.mpr
373     rintro _ ⟨j, rfl⟩
374     change algebraMap (MvPolynomial (Fin m) K)
375         (ChartOverlapRing (K := K) chart)
376         (projectiveDehomogenizeAt chart (equations j)) ∈
377       localizedZeroChartEquationIdeal chart equations
378     rw [localized_projectiveDehomogenizeAt_eq_pow_mul_transition
379       chart (equations j) (degree j) (homogeneous j)]
380     exact Ideal.mul_mem_left _ _
381       (Ideal.subset_span ⟨j, rfl⟩)

383 /-- Finite homogenized equations for an affine ideal satisfy the corrected
384 global-to-local statement on every projective chart overlap. -/
385 theorem exists_finite_homogeneous_chartOverlapPresentation
386     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) K)) :
387     ∃ (n : ℕ) (equations : Fin n → MvPolynomial (Fin (m + 1)) K)
388         (degree : Fin n → ℕ),
389       (∀ j, (equations j).IsHomogeneous (degree j)) ∧
390       dehomogenizedEquationIdeal equations = I ∧
391       ∀ chart : Fin (m + 1),
392         localizedZeroChartEquationIdeal chart equations =
393           localizedChosenChartEquationIdeal chart equations := by
394   obtain ⟨n, equations, degree, homogeneous, hdehom⟩ :=
395     exists_finite_homogeneous_equations_dehomogenizing_to I
396   refine ⟨n, equations, degree, homogeneous, hdehom, ?_⟩
397   intro chart
398   exact localized_chartEquationIdeals_eq chart equations degree homogeneous
