22 /-- The retained order factorization makes the zeroth projective coordinate
23 nonzero in the chart index: its residue is zero, whereas the selected chart
24 coordinate has been normalized to one. Hence every actual witness chart is a
25 successor coordinate. -/
26 theorem exists_succ_chart_index
27     {k : Type u} [Field k] [CharZero k] {m : ℕ} (hm : 0 < m)
28     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
29     (w : GeneralDivisorialVisibleFrameWitness hm P) :
30     ∃ j : Fin m, w.column.chart = Fin.succ j := by
31   classical
32   let F := ComponentFractionField P
33   letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField) F :=
34     w.column.W.ambientAlgebra
35   let V := w.column.W.place.valuation.toSubring
36   letI : IsLocalRing V := w.column.W.place.isDiscrete.toIsLocalRing
37   have hq0res : residue V (w.column.q 0) = 0 :=
38     witness_q0_residue_eq_zero hm P w
39   have hchart : w.column.chart ≠ 0 := by
40     intro hz
41     have hq01 : w.column.q 0 = 1 := by
42       simpa [hz] using w.column.chart_one
43     have hres := congrArg (residue V) hq01
44     rw [hq0res, map_one] at hres
45     exact zero_ne_one hres
46   obtain ⟨j, hj⟩ := Fin.exists_succ_eq_of_ne_zero hchart
47   exact ⟨j, hj.symm⟩
48 
49 end
50 end Stafford38.Geometry.ActualWitnessSelectedChartIndex
