13 # Derivations commute with the canonical completed chart
14 
15 The completion map here is the prescribed map built from the canonical
16 completion equivalence, not a caller-supplied homomorphism.  Agreement on the
17 polynomial parameters and the checked Kähler-basis uniqueness theorem give
18 agreement on every element of the local ring.
19 -/

39 /-- For the actual prescribed completion chart, every polynomial-parameter
40 derivation commutes with completion.  Thus this is an identity of derivations
41 on the whole local ring, not only a formula on selected coordinates. -/
42 theorem localToPowerSeries_commutes_parameterDerivation
43     (M : Ideal A) [M.IsMaximal] (eM : (A ⧸ M) ≃ₐ[k] k)
44     [Algebra.FormallyEtale ChartBase (Localization.AtPrime M)]
45     (i : Fin (d + 1)) (b : Localization.AtPrime M) :
46     MvPowerSeries.pderiv i
47         (PrescribedGroundPointPowerSeriesMap.localToPowerSeries
48           (σ := Fin (d + 1)) M eM b) =
49       PrescribedGroundPointPowerSeriesMap.localToPowerSeries
50         (σ := Fin (d + 1)) M eM
51         (parameterDerivation (k := k) (d := d)
52           (B := Localization.AtPrime M) i b) := by
53   let T := Localization.AtPrime M
54   let Φ : T →ₐ[k] C :=
55     PrescribedGroundPointPowerSeriesMap.localToPowerSeries
56       (σ := Fin (d + 1)) M eM
57   letI : Algebra T C := Φ.toRingHom.toAlgebra
58   letI : IsScalarTower k T C := IsScalarTower.of_algebraMap_eq (fun c => by
59     change algebraMap k C c = Φ (algebraMap k T c)
60     exact (Φ.commutes c).symm)
61   have hΦ (j : Fin (d + 1)) :
62       Φ (algebraMap ChartBase T (MvPolynomial.X j)) =
63         MvPowerSeries.C (residueCoordinates (σ := Fin (d + 1)) M eM j) +
64           MvPowerSeries.X j := by
65     calc
66       Φ (algebraMap ChartBase T (MvPolynomial.X j)) =
67           Φ (algebraMap A T (algebraMap ChartBase A (MvPolynomial.X j))) := by
68         exact congrArg Φ (IsScalarTower.algebraMap_apply ChartBase A T _)
69       _ = originalAlgebraToPowerSeries (σ := Fin (d + 1)) M eM
70           (algebraMap ChartBase A (MvPolynomial.X j)) := rfl
71       _ = MvPowerSeries.C (residueCoordinates (σ := Fin (d + 1)) M eM j) +
72           MvPowerSeries.X j :=
73         originalAlgebraToPowerSeries_coordinate (σ := Fin (d + 1)) M eM j
74   let Dleft : Derivation k T C :=
75     (MvPowerSeries.pderiv i).compAlgebraMap T
76   let Φlin : T →ₗ[T] C :=
77     { toFun := Φ
78       map_add' := Φ.map_add
79       map_smul' := fun r x => by
80         change Φ (r * x) = Φ r * Φ x
81         exact Φ.map_mul r x }
82   let Dright : Derivation k T C :=
83     Φlin.compDer (parameterDerivation (k := k) (d := d) (B := T) i)
84   have hleft : ∀ j, Dleft (algebraMap ChartBase T (MvPolynomial.X j)) =
85       if j = i then (1 : C) else 0 := by
86     intro j
87     change MvPowerSeries.pderiv i
88       (Φ (algebraMap ChartBase T (MvPolynomial.X j))) = _
89     rw [hΦ j]
90     simp [MvPowerSeries.pderiv_X, Pi.single_apply, eq_comm]
91   have hright : ∀ j, Dright (algebraMap ChartBase T (MvPolynomial.X j)) =
92       if j = i then (1 : C) else 0 := by
93     intro j
94     change Φ (parameterDerivation (k := k) (d := d) (B := T) i
95       (algebraMap ChartBase T (MvPolynomial.X j))) = _
96     rw [parameterDerivation_apply_parameter]
97     simp
98   have hderiv : Dleft = Dright := by
99     calc
100       Dleft = EtaleCotangentBasis.derivationFromValues
101           (k := k) (σ := Fin (d + 1)) (B := T) (L := C)
102           (fun j => Dleft (algebraMap ChartBase T (MvPolynomial.X j))) :=
103         (EtaleCotangentBasis.derivation_eq_from_parameter_values Dleft).symm
104       _ = EtaleCotangentBasis.derivationFromValues
105           (k := k) (σ := Fin (d + 1)) (B := T) (L := C)
106           (fun j => Dright (algebraMap ChartBase T (MvPolynomial.X j))) := by
107         congr 1
108         funext j
109         rw [hleft j, hright j]
110       _ = Dright :=
111         EtaleCotangentBasis.derivation_eq_from_parameter_values Dright
112   change MvPowerSeries.pderiv i (Φ b) =
113     Φ (parameterDerivation (k := k) (d := d) (B := T) i b)
114   exact congrArg (fun D : Derivation k T C => D b) hderiv

116 /-- The same full-derivation identity for the actual `Option (Fin d)` chart
117 after its canonical `finSuccEquiv` reindexing.  The output coordinate `i`

120 theorem localToFinSuccPowerSeries_commutes_coordinateDerivation

123     [IsScalarTower k (MvPolynomial (Option (Fin d)) k) Aₒ]
124     [Algebra.EssFiniteType (MvPolynomial (Option (Fin d)) k) Aₒ]
125     (M : Ideal Aₒ) [M.IsMaximal] (eM : (Aₒ ⧸ M) ≃ₐ[k] k)
126     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k)

129     MvPowerSeries.pderiv i
130         (Stafford38.Geometry.PrescribedGroundPointUnitPowerChart.localToFinSuccPowerSeries

132       Stafford38.Geometry.PrescribedGroundPointUnitPowerChart.localToFinSuccPowerSeries
133         (k := k) (B := Aₒ) (d := d) M eM
134         (EtaleCotangentBasis.coordinateDerivation
135           (k := k) (σ := Option (Fin d))
136           (B := Localization.AtPrime M) (L := Localization.AtPrime M)
137           ((_root_.finSuccEquiv d) i) b) := by
138   let Rₒ := MvPolynomial (Option (Fin d)) k
139   let T := Localization.AtPrime M
140   let Ψ : T →ₐ[k] MvPowerSeries (Fin (d + 1)) k :=
141     Stafford38.Geometry.PrescribedGroundPointUnitPowerChart.localToFinSuccPowerSeries
142       (k := k) (B := Aₒ) (d := d) M eM
143   letI : Algebra T (MvPowerSeries (Fin (d + 1)) k) := Ψ.toRingHom.toAlgebra
144   letI : IsScalarTower k T (MvPowerSeries (Fin (d + 1)) k) :=
145     IsScalarTower.of_algebraMap_eq (fun c => by
146       change algebraMap k (MvPowerSeries (Fin (d + 1)) k) c = Ψ (algebraMap k T c)
147       exact (Ψ.commutes c).symm)
148   have hcoord (j : Option (Fin d)) :
149       Ψ (algebraMap Rₒ T (MvPolynomial.X j)) =
150         MvPowerSeries.C (residueCoordinates (σ := Option (Fin d)) M eM j) +
151           MvPowerSeries.X ((_root_.finSuccEquiv d).symm j) := by
152     cases j with
153     | none =>
154         calc
155           Ψ (algebraMap Rₒ T (MvPolynomial.X none)) =
156               Ψ (algebraMap Aₒ T (algebraMap Rₒ Aₒ (MvPolynomial.X none))) := by
157             exact congrArg Ψ (IsScalarTower.algebraMap_apply Rₒ Aₒ T _)
158           _ = Stafford38.Geometry.PrescribedGroundPointUnitPowerChart.localToFinSuccPowerSeries
159                 (k := k) (B := Aₒ) (d := d) M eM
160                 (algebraMap Aₒ T (algebraMap Rₒ Aₒ (MvPolynomial.X none))) := rfl
161           _ = _ :=
162             Stafford38.Geometry.PrescribedGroundPointUnitPowerChart.localToFinSuccPowerSeries_none
163               (k := k) (B := Aₒ) (d := d) M eM
164     | some j =>
165         calc
166           Ψ (algebraMap Rₒ T (MvPolynomial.X (some j))) =
167               Ψ (algebraMap Aₒ T (algebraMap Rₒ Aₒ (MvPolynomial.X (some j)))) := by
168             exact congrArg Ψ (IsScalarTower.algebraMap_apply Rₒ Aₒ T _)
169           _ = Stafford38.Geometry.PrescribedGroundPointUnitPowerChart.localToFinSuccPowerSeries
170                 (k := k) (B := Aₒ) (d := d) M eM
171                 (algebraMap Aₒ T (algebraMap Rₒ Aₒ (MvPolynomial.X (some j)))) := rfl
172           _ = _ := by
173             simpa using Stafford38.Geometry.PrescribedGroundPointUnitPowerChart.localToFinSuccPowerSeries_some
174               (k := k) (B := Aₒ) (d := d) M eM j
175   let Ψlin : T →ₗ[T] MvPowerSeries (Fin (d + 1)) k :=
176     { toFun := Ψ
177       map_add' := Ψ.map_add
178       map_smul' := fun r x => by
179         change Ψ (r * x) = Ψ r * Ψ x
180         exact Ψ.map_mul r x }
181   let Dleft : Derivation k T (MvPowerSeries (Fin (d + 1)) k) :=
182     (MvPowerSeries.pderiv i).compAlgebraMap T
183   let Dright : Derivation k T (MvPowerSeries (Fin (d + 1)) k) :=
184     Ψlin.compDer (EtaleCotangentBasis.coordinateDerivation
185       (k := k) (σ := Option (Fin d)) (B := T)
186       (L := T) ((_root_.finSuccEquiv d) i))
187   have hleft : ∀ j : Option (Fin d),
188       Dleft (algebraMap Rₒ T (MvPolynomial.X j)) =
189         if j = (_root_.finSuccEquiv d) i then (1 : MvPowerSeries (Fin (d + 1)) k) else 0 := by
190     intro j
191     change MvPowerSeries.pderiv i (Ψ (algebraMap Rₒ T (MvPolynomial.X j))) = _
192     rw [hcoord j]
193     by_cases h : i = (_root_.finSuccEquiv d).symm j
194     · subst i
195       simp [MvPowerSeries.pderiv_X, Pi.single_apply, eq_comm]
196     · have h' : j ≠ (_root_.finSuccEquiv d) i := by
197         intro hji
198         apply h
199         have heq := congrArg (_root_.finSuccEquiv d).symm hji
200         simpa using heq.symm
201       simp [MvPowerSeries.pderiv_X, Pi.single_apply, h, h']
202   have hright : ∀ j : Option (Fin d),
203       Dright (algebraMap Rₒ T (MvPolynomial.X j)) =
204         if j = (_root_.finSuccEquiv d) i then (1 : MvPowerSeries (Fin (d + 1)) k) else 0 := by
205     intro j
206     change Ψ (EtaleCotangentBasis.coordinateDerivation
207       (k := k) (σ := Option (Fin d)) (B := T)
208       (L := T) ((_root_.finSuccEquiv d) i)
209       (algebraMap Rₒ T (MvPolynomial.X j))) = _
210     rw [EtaleCotangentBasis.coordinateDerivation_apply_parameter]
211     simp
212   have hderiv : Dleft = Dright := by
213     calc
214       Dleft = EtaleCotangentBasis.derivationFromValues
215           (k := k) (σ := Option (Fin d)) (B := T)
216           (L := MvPowerSeries (Fin (d + 1)) k)
217           (fun j => Dleft (algebraMap Rₒ T (MvPolynomial.X j))) :=
218         (EtaleCotangentBasis.derivation_eq_from_parameter_values Dleft).symm
219       _ = EtaleCotangentBasis.derivationFromValues
220           (k := k) (σ := Option (Fin d)) (B := T)
221           (L := MvPowerSeries (Fin (d + 1)) k)
222           (fun j => Dright (algebraMap Rₒ T (MvPolynomial.X j))) := by
223         congr 1
224         funext j
225         rw [hleft j, hright j]
226       _ = Dright :=
227         EtaleCotangentBasis.derivation_eq_from_parameter_values Dright
228   calc
229     MvPowerSeries.pderiv i (Ψ b) = Dleft b := rfl
230     _ = Dright b := congrArg (fun D : Derivation k T
231       (MvPowerSeries (Fin (d + 1)) k) => D b) hderiv
232     _ = Ψ (EtaleCotangentBasis.coordinateDerivation
233         (k := k) (σ := Option (Fin d)) (B := T)
234         (L := T) ((_root_.finSuccEquiv d) i) b) := by
235       change (Ψlin.compDer (EtaleCotangentBasis.coordinateDerivation
236         (k := k) (σ := Option (Fin d)) (B := T)
237         (L := T) ((_root_.finSuccEquiv d) i))) b = _

240 /-- The canonical one-variable tilted arc differentiates every local function
241 by the corresponding weighted sum of the full chart parameter derivations.
242 This follows from the same Kähler-basis uniqueness theorem, using the actual
243 coordinate values of the prescribed arc. -/
244 theorem localToPowerSeriesArc_derivative_eq_parameter_sum
245     (M : Ideal A) [M.IsMaximal] (eM : (A ⧸ M) ≃ₐ[k] k)
246     [Algebra.FormallyEtale ChartBase (Localization.AtPrime M)]
247     (α : Fin d → k) (b : Localization.AtPrime M) :
248     PowerSeries.derivative (R := k)
249         (PrescribedGroundPointPowerSeriesMap.localToPowerSeriesArc
250           (M := M) (B := A) α eM b) =
251       PrescribedGroundPointPowerSeriesMap.localToPowerSeriesArc
252           (M := M) (B := A) α eM
253           (parameterDerivation (k := k) (d := d)
254             (B := Localization.AtPrime M) 0 b) +
255         ∑ j : Fin d, PowerSeries.C (α j) *
256           PrescribedGroundPointPowerSeriesMap.localToPowerSeriesArc
257             (M := M) (B := A) α eM
258             (parameterDerivation (k := k) (d := d)
259               (B := Localization.AtPrime M) j.succ b) := by
260   let T := Localization.AtPrime M
261   let Ψ : T →ₐ[k] PowerSeries k :=
262     PrescribedGroundPointPowerSeriesMap.localToPowerSeriesArc
263       (M := M) (B := A) α eM
264   letI : Algebra T (PowerSeries k) := Ψ.toRingHom.toAlgebra
265   letI : IsScalarTower k T (PowerSeries k) := IsScalarTower.of_algebraMap_eq
266     (fun c => by
267       change algebraMap k (PowerSeries k) c = Ψ (algebraMap k T c)
268       exact (Ψ.commutes c).symm)
269   have hΨ (j : Fin (d + 1)) :
270       Ψ (algebraMap ChartBase T (MvPolynomial.X j)) =
271         PowerSeries.C (residueCoordinates (σ := Fin (d + 1)) M eM j) +
272           PowerSeries.C ((Fin.cons 1 α : Fin (d + 1) → k) j) *
273             PowerSeries.X := by
274     calc
275       Ψ (algebraMap ChartBase T (MvPolynomial.X j)) =
276           Ψ (algebraMap A T (algebraMap ChartBase A (MvPolynomial.X j))) := by
277         exact congrArg Ψ (IsScalarTower.algebraMap_apply ChartBase A T _)
278       _ = originalAlgebraToPowerSeriesArc (M := M) (B := A) α eM
279           (algebraMap ChartBase A (MvPolynomial.X j)) := rfl
280       _ = _ := originalAlgebraToPowerSeriesArc_coordinate
281         (M := M) (B := A) α eM j
282   let Ψlin : T →ₗ[T] PowerSeries k :=
283     { toFun := Ψ
284       map_add' := Ψ.map_add
285       map_smul' := fun r x => by
286         change Ψ (r * x) = Ψ r * Ψ x
287         exact Ψ.map_mul r x }
288   let Dleft : Derivation k T (PowerSeries k) :=
289     (PowerSeries.derivative (R := k)).compAlgebraMap T
290   let Dright : Derivation k T (PowerSeries k) :=
291     (Ψlin.compDer (parameterDerivation (k := k) (d := d) (B := T) 0)) +
292       ∑ j : Fin d, PowerSeries.C (α j) •
293         (Ψlin.compDer (parameterDerivation (k := k) (d := d)
294           (B := T) j.succ))
295   have hleft : ∀ j, Dleft (algebraMap ChartBase T (MvPolynomial.X j)) =
296       PowerSeries.C ((Fin.cons 1 α : Fin (d + 1) → k) j) := by
297     intro j
298     change PowerSeries.derivative (R := k)
299       (Ψ (algebraMap ChartBase T (MvPolynomial.X j))) = _
300     rw [hΨ j]
301     simp [Derivation.map_add, Derivation.leibniz]
302   have hparam (i j : Fin (d + 1)) :
303       Ψ (parameterDerivation (k := k) (d := d) (B := T) i
304         (algebraMap ChartBase T (MvPolynomial.X j))) = if j = i then (1 : PowerSeries k) else 0 := by
305     change Ψ (parameterDerivation (k := k) (d := d) (B := T) i
306       (algebraMap ChartBase T (MvPolynomial.X j)) ) = _
307     rw [parameterDerivation_apply_parameter]
308     simp
309   have hright : ∀ j, Dright (algebraMap ChartBase T (MvPolynomial.X j)) =
310       PowerSeries.C ((Fin.cons 1 α : Fin (d + 1) → k) j) := by
311     intro j
312     change (Derivation.coeFnAddMonoidHom
313       ((Ψlin.compDer (parameterDerivation (k := k) (d := d) (B := T) 0)) +
314         ∑ l : Fin d, PowerSeries.C (α l) •
315           (Ψlin.compDer (parameterDerivation (k := k) (d := d) (B := T) l.succ))))
316       (algebraMap ChartBase T (MvPolynomial.X j)) = _
317     rw [map_add, map_sum]
318     simp [Derivation.smul_apply, hparam, Ψlin, eq_comm]
319     cases j using Fin.cases with
320     | zero =>
321         have hsum : ∑ l : Fin d,
322             (if (0 : Fin (d + 1)) = l.succ then PowerSeries.C (α l) else 0) = 0 := by
323           apply Finset.sum_eq_zero
324           intro l hl
325           have hne : (0 : Fin (d + 1)) ≠ l.succ := by
326             intro h
327             exact Fin.succ_ne_zero l h.symm
328           simp [hne]
329         rw [hsum]
330         simp
331     | succ j => simp [Fin.succ_inj]
332   have hderiv : Dleft = Dright := by
333     calc
334       Dleft = EtaleCotangentBasis.derivationFromValues
335           (k := k) (σ := Fin (d + 1)) (B := T) (L := PowerSeries k)
336           (fun j => Dleft (algebraMap ChartBase T (MvPolynomial.X j))) :=
337         (EtaleCotangentBasis.derivation_eq_from_parameter_values Dleft).symm
338       _ = EtaleCotangentBasis.derivationFromValues
339           (k := k) (σ := Fin (d + 1)) (B := T) (L := PowerSeries k)
340           (fun j => Dright (algebraMap ChartBase T (MvPolynomial.X j))) := by
341         congr 1
342         funext j
343         rw [hleft j, hright j]
344       _ = Dright :=
345         EtaleCotangentBasis.derivation_eq_from_parameter_values Dright
346   calc
347     PowerSeries.derivative (R := k) (Ψ b) = Dleft b := rfl
348     _ = Dright b := congrArg (fun D : Derivation k T (PowerSeries k) => D b) hderiv
349     _ = Ψ (parameterDerivation (k := k) (d := d) (B := T) 0 b) +
350         ∑ j : Fin d, PowerSeries.C (α j) *
351           Ψ (parameterDerivation (k := k) (d := d) (B := T) j.succ b) := by
352       change (Derivation.coeFnAddMonoidHom
353         ((Ψlin.compDer (parameterDerivation (k := k) (d := d) (B := T) 0)) +
354           ∑ j : Fin d, PowerSeries.C (α j) •
355             (Ψlin.compDer (parameterDerivation (k := k) (d := d) (B := T) j.succ)))) b = _
356       rw [map_add, map_sum]
357       simp [Derivation.smul_apply, Ψlin]
