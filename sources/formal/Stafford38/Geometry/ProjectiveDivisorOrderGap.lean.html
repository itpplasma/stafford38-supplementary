8 # The strict projective order gap at a divisor
9 
10 This file formalizes the first valuation-theoretic implication in the
11 normalization-at-infinity argument.  Let `q₀` be the projective denominator
12 and let `r = q₁ / q₀` be the distinguished affine coordinate in the local
13 ring at a prime divisor.  If both `q₀` and `r` vanish at the divisor, then a
14 uniformizer `ϖ` gives
15 
16 `q₀ = ϖ^a u₀`, `q₁ = ϖ^b u₁`, and `0 < a < b`.
17 
18 The power-series specialization feeds this strict order gap directly into
19 `FormalDivisorAxisLift`.  Thus its caller no longer has to supply the
20 integers `a,b`, their factorizations, or the inequality `a < b`.
21 
22 The inputs are the local DVR, its uniformizer, and the three coordinate
23 elements. The conclusions supply the factorizations and strict exponent
24 inequality used by the completed local argument.
25 -/

40 /-! ## The DVR order calculation -/
41 
42 /-- At a prime divisor, a nonzero projective denominator and a nonzero affine
43 coordinate which both vanish have strictly separated orders after
44 rehomogenization.
45 
46 The equation `q₁ = q₀ * ratio` is the local-ring form of
47 `ratio = X₁ / X₀`; unlike division in the fraction field, it retains the
48 integrality needed by the formal-divisor consumer. -/
49 theorem exists_uniformizer_strict_orderGap
50     {R : Type u} [CommRing R] [IsDomain R] [IsDiscreteValuationRing R]
51     (uniformizer : R) (huniformizer : Irreducible uniformizer)
52     (q₀ ratio q₁ : R)
53     (hq₀_ne : q₀ ≠ 0) (hratio_ne : ratio ≠ 0)
54     (hq₀_vanish : q₀ ∈ maximalIdeal R)
55     (hratio_vanish : ratio ∈ maximalIdeal R)
56     (hq₁ : q₁ = q₀ * ratio) :
57     ∃ (a r b : ℕ) (u₀ ur u₁ : Rˣ),
58       0 < a ∧ 0 < r ∧ b = a + r ∧ a < b ∧
59       q₀ = (u₀ : R) * uniformizer ^ a ∧
60       ratio = (ur : R) * uniformizer ^ r ∧
61       u₁ = u₀ * ur ∧
62       q₁ = (u₁ : R) * uniformizer ^ b := by
63   obtain ⟨a, u₀, hq₀⟩ :=
64     IsDiscreteValuationRing.eq_unit_mul_pow_irreducible hq₀_ne huniformizer
65   obtain ⟨r, ur, hratio⟩ :=
66     IsDiscreteValuationRing.eq_unit_mul_pow_irreducible hratio_ne huniformizer
67   have hq₀_nonunit : ¬IsUnit q₀ := by
68     simpa only [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff] using hq₀_vanish
69   have hratio_nonunit : ¬IsUnit ratio := by
70     simpa only [IsLocalRing.mem_maximalIdeal, mem_nonunits_iff] using hratio_vanish
71   have ha : 0 < a := by
72     apply Nat.pos_of_ne_zero
73     intro ha0
74     apply hq₀_nonunit
75     rw [hq₀, ha0, pow_zero, mul_one]
76     exact u₀.isUnit
77   have hr : 0 < r := by
78     apply Nat.pos_of_ne_zero
79     intro hr0
80     apply hratio_nonunit
81     rw [hratio, hr0, pow_zero, mul_one]
82     exact ur.isUnit
83   let b := a + r
84   let u₁ := u₀ * ur
85   refine ⟨a, r, b, u₀, ur, u₁, ha, hr, rfl, ?_, hq₀, hratio, rfl, ?_⟩
86   · simp only [b]
87     omega
88   · rw [hq₁, hq₀, hratio]
89     simp only [b, u₁, Units.val_mul, pow_add]
90     ring

92 /-! ## The completed local power-series form -/
93 
94 /-- The DVR implication above specialized to the completed divisor chart.
95 Vanishing is expressed by zero constant coefficient.  The output is exactly
96 the pair of factorizations and strict inequality consumed by
97 `exists_formalDivisorAxisLift`. -/
98 theorem exists_powerSeries_strict_projectiveOrderGap
99     {k : Type u} [Field k]
100     (q₀ ratio q₁ : PowerSeries k)
101     (hq₀_ne : q₀ ≠ 0) (hratio_ne : ratio ≠ 0)
102     (hq₀_vanish : PowerSeries.constantCoeff q₀ = 0)
103     (hratio_vanish : PowerSeries.constantCoeff ratio = 0)
104     (hq₁ : q₁ = q₀ * ratio) :
105     ∃ (a b : ℕ) (u₀ u₁ : PowerSeries k),
106       0 < a ∧ a < b ∧
107       q₀ = (PowerSeries.X : PowerSeries k) ^ a * u₀ ∧
108       PowerSeries.constantCoeff u₀ ≠ 0 ∧
109       q₁ = (PowerSeries.X : PowerSeries k) ^ b * u₁ ∧
110       PowerSeries.constantCoeff u₁ ≠ 0 := by
111   have hq₀_mem : q₀ ∈ IsLocalRing.maximalIdeal (PowerSeries k) := by
112     rw [← PowerSeries.ker_coeff_eq_max_ideal]
113     exact hq₀_vanish
114   have hratio_mem : ratio ∈ IsLocalRing.maximalIdeal (PowerSeries k) := by
115     rw [← PowerSeries.ker_coeff_eq_max_ideal]
116     exact hratio_vanish
117   obtain ⟨a, r, b, u₀, ur, u₁, ha, hr, hb, hab,
118       hq₀_factor, hratio_factor, hu₁, hq₁_factor⟩ :=
119     exists_uniformizer_strict_orderGap
120       (PowerSeries.X : PowerSeries k) PowerSeries.X_irreducible
121       q₀ ratio q₁ hq₀_ne hratio_ne hq₀_mem hratio_mem hq₁
122   have hu₀_const : PowerSeries.constantCoeff (u₀ : PowerSeries k) ≠ 0 := by
123     exact (PowerSeries.isUnit_iff_constantCoeff.mp u₀.isUnit).ne_zero
124   have hu₁_const : PowerSeries.constantCoeff (u₁ : PowerSeries k) ≠ 0 := by
125     exact (PowerSeries.isUnit_iff_constantCoeff.mp u₁.isUnit).ne_zero
126   refine ⟨a, b, (u₀ : PowerSeries k), (u₁ : PowerSeries k),
127     ha, hab, ?_, hu₀_const, ?_, hu₁_const⟩
128   · simpa [mul_comm] using hq₀_factor
129   · simpa [mul_comm] using hq₁_factor

131 /-! ## Direct handoff to the formal tangent producer -/
132 
133 /-- A completed projective-divisor chart with vanishing denominator and
134 vanishing affine ratio produces the exact formal tangent annihilator whose
135 residue is the distinguished projective axis.
136 
137 The tangent divisibility assumptions are coordinate-free order statements:
138 divisor-tangent derivatives of `q₀` and `q₁` remain divisible by those
139 coordinates.  The theorem converts them to the explicit uniformizer powers
140 required by `FormalDivisorAxisLift`. -/
141 theorem exists_formalDivisorAxisLift_of_projectiveRatio
142     {k : Type u} [Field k] [CharZero k]
143     {ι : Type v} {κ : Type w} [Fintype ι] [Fintype κ]
144     [DecidableEq ι] [DecidableEq κ]
145     (q : ι → PowerSeries k)
146     (Z : Matrix ι κ (PowerSeries k))
147     (rows : κ ↪ ι) (chart zero axis : ι)
148     (ratio : PowerSeries k)
149     (hqchart : q chart = 1)
150     (hZchart : ∀ j, Z chart j = 0)
151     (hqzero_ne : q zero ≠ 0)
152     (hratio_ne : ratio ≠ 0)
153     (hqzero_vanish : PowerSeries.constantCoeff (q zero) = 0)
154     (hratio_vanish : PowerSeries.constantCoeff ratio = 0)
155     (hqaxis : q axis = q zero * ratio)
156     (hZzero_dvd : ∀ j, q zero ∣ Z zero j)
157     (hZaxis_dvd : ∀ j, q axis ∣ Z axis j)
158     (hminor :
159       PowerSeries.constantCoeff
160         (selectedMinor Z rows).det ≠ 0) :
161     ∃ (tau : ι → PowerSeries k)
162       (C : Matrix (FormalTangentColumn κ) ι (PowerSeries k))
163       (ell : ι → PowerSeries k),
164       C * formalTangentMatrix q Z tau = 1 ∧
165       rowMul ell (formalTangentMatrix q Z tau) = 0 ∧
166       residueColumn ell = axisRow (k := k) axis := by
167   obtain ⟨a, b, u₀, u₁, ha, hab, hqzero_factor, hu₀,
168       hqaxis_factor, hu₁⟩ :=
169     exists_powerSeries_strict_projectiveOrderGap
170       (q zero) ratio (q axis) hqzero_ne hratio_ne
171         hqzero_vanish hratio_vanish hqaxis
172   have hZzero : ∀ j, ∃ w : PowerSeries k,
173       Z zero j = (PowerSeries.X : PowerSeries k) ^ a * w := by
174     intro j
175     obtain ⟨w, hw⟩ := hZzero_dvd j
176     refine ⟨u₀ * w, ?_⟩
177     rw [hw, hqzero_factor]
178     ring
179   have hZaxis : ∀ j, ∃ w : PowerSeries k,
180       Z axis j = (PowerSeries.X : PowerSeries k) ^ b * w := by
181     intro j
182     obtain ⟨w, hw⟩ := hZaxis_dvd j
183     refine ⟨u₁ * w, ?_⟩
184     rw [hw, hqaxis_factor]
185     ring
186   obtain ⟨lambda, c, tau, C, ell, _hlambda, _hselected,
187       _htauchart, _htauselected, _hc, _hfactor, _hprimitive,
188       _htauaxis, _haxiscolumns, hleft, hrow, hresidue⟩ :=
189     exists_formalDivisorAxisLift q Z rows chart zero axis a b u₀ u₁
190       hqchart hZchart ha hab hqzero_factor hu₀ hZzero
191         hqaxis_factor hZaxis hminor
192   exact ⟨tau, C, ell, hleft, hrow, hresidue⟩
