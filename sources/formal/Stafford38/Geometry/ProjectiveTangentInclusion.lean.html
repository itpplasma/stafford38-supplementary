6 /-!
7 # Projective tangent inclusion is enough for affine conormality
8 
9 The projective dehomogenization bridge does not need equality between the
10 Zariski tangent space and the span of the supplied dehomogenized columns.
11 It is enough that the Zariski tangent space is contained in that span: a
12 covector annihilating the larger space then annihilates the smaller one.
13 
14 This file propagates that weaker hypothesis through the existing formal
15 divisor and Laurent-direction consumers.  It constructs no projective chart,
16 normalization, tangent comparison, or conormal-closure theorem.
17 -/

34 /-- A projective annihilating row gives an affine conormal covector whenever
35 the Zariski tangent space is contained in the dehomogenized column span.
36 Equality of the two tangent spaces is not required. -/
37 theorem coordinateCovector_mem_affineConormalSpace_of_zariski_le_span
38     {κ : Type*}
39     (I : Ideal (MvPolynomial (Fin n) K))
40     (q ell : Fin (n + 1) → K) (B : Matrix (Fin (n + 1)) κ K)
41     (hq0 : q 0 ≠ 0)
42     (hq : ∑ i, ell i * q i = 0)
43     (hB : rowMul ell B = 0)
44     (htangent :
45       zariskiTangentSpace (dehomogenizedPoint q) I ≤
46         dehomogenizedTangentSpan q B) :
47     coordinateCovector (fun i ↦ ell i.succ) ∈
48       affineConormalSpace (dehomogenizedPoint q) I := by
49   rw [affineConormalSpace, Submodule.mem_dualAnnihilator]
50   intro v hv
51   have hann :=
52     coordinateCovector_mem_dehomogenizedTangentSpan_dualAnnihilator
53       q ell B hq0 hq hB
54   rw [Submodule.mem_dualAnnihilator] at hann
55   exact hann v (htangent hv)

57 /-- The weakened projective bridge reaches the equation-defined affine
58 conormal locus directly. -/
59 theorem phasePoint_mem_equationConormalLocus_of_zariski_le_span
60     {κ : Type*}
61     (I : Ideal (MvPolynomial (Fin n) K))
62     (q ell : Fin (n + 1) → K) (B : Matrix (Fin (n + 1)) κ K)
63     (hq0 : q 0 ≠ 0)
64     (hq : ∑ i, ell i * q i = 0)
65     (hB : rowMul ell B = 0)
66     (hbase : ∀ f ∈ I, MvPolynomial.eval (dehomogenizedPoint q) f = 0)
67     (htangent :
68       zariskiTangentSpace (dehomogenizedPoint q) I ≤
69         dehomogenizedTangentSpan q B) :
70     Sum.elim (dehomogenizedPoint q) (fun i ↦ ell i.succ) ∈
71       equationConormalLocus I := by
72   refine ⟨?_, ?_⟩
73   · simpa using hbase
74   · simpa using coordinateCovector_mem_affineConormalSpace_of_zariski_le_span
75       I q ell B hq0 hq hB htangent

79 /-- Formal row annihilation reaches the Laurent-valued equation-conormal
80 locus under tangent inclusion rather than tangent equality. -/
81 theorem laurentPhasePoint_mem_equationConormalLocus_of_zariski_le_span
82     {κ : Type*} [Fintype κ]
83     (I : Ideal (MvPolynomial (Fin n) (LaurentSeries k)))
84     (q ell : Fin (n + 1) → PowerSeries k)
85     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
86     (tau : Fin (n + 1) → PowerSeries k)
87     (hq0 : q 0 ≠ 0)
88     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
89     (hbase : ∀ f ∈ I,
90       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0)
91     (htangent :
92       zariskiTangentSpace (dehomogenizedPoint (laurentColumn q)) I ≤
93         dehomogenizedTangentSpan (laurentColumn q)
94           (laurentNonpositionTangentMatrix Z tau)) :
95     Sum.elim (dehomogenizedPoint (laurentColumn q))
96         (fun i ↦ laurentColumn ell i.succ) ∈
97       equationConormalLocus I := by
98   exact phasePoint_mem_equationConormalLocus_of_zariski_le_span
99     I (laurentColumn q) (laurentColumn ell)
100       (laurentNonpositionTangentMatrix Z tau)
101       (laurentColumn_ne_zero_of_ne_zero q hq0)
102       (laurentColumn_dot_eq_zero_of_formalTangent_rowMul q ell Z tau hrow)
103       (laurentNonposition_rowMul_eq_zero_of_formalTangent_rowMul
104         q ell Z tau hrow)
105       hbase htangent

107 /-- The weakened Laurent bridge retains the exact fibre-residue identity. -/
108 theorem laurentPhasePoint_mem_equationConormalLocus_and_residue_of_zariski_le_span
109     {κ : Type*} [Fintype κ]
110     (I : Ideal (MvPolynomial (Fin n) (LaurentSeries k)))
111     (q ell : Fin (n + 1) → PowerSeries k)
112     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
113     (tau : Fin (n + 1) → PowerSeries k)
114     (hq0 : q 0 ≠ 0)
115     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
116     (hbase : ∀ f ∈ I,
117       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0)
118     (htangent :
119       zariskiTangentSpace (dehomogenizedPoint (laurentColumn q)) I ≤
120         dehomogenizedTangentSpan (laurentColumn q)
121           (laurentNonpositionTangentMatrix Z tau)) :
122     Sum.elim (dehomogenizedPoint (laurentColumn q))
123           (fun i ↦ laurentColumn ell i.succ) ∈ equationConormalLocus I ∧
124       residueColumn (fun i : Fin n ↦ ell i.succ) =
125         fun i ↦ residueColumn ell i.succ := by
126   exact ⟨laurentPhasePoint_mem_equationConormalLocus_of_zariski_le_span
127     I q ell Z tau hq0 hrow hbase htangent, residueColumn_tail ell⟩

129 /-- The same inclusion hypothesis reaches the projected Laurent
130 equation-conormal direction closure required by the asymptotic consumer. -/
131 theorem residue_tail_mem_laurentEquationConormalDirectionClosure_of_zariski_le_span
132     {κ : Type*} [Fintype κ]
133     (I : Ideal (MvPolynomial (Fin n) k))
134     (q ell : Fin (n + 1) → PowerSeries k)
135     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
136     (tau : Fin (n + 1) → PowerSeries k)
137     (hq0 : q 0 ≠ 0)
138     (hrow : rowMul ell (formalTangentMatrix q Z tau) = 0)
139     (hbase : ∀ f ∈
140         I.map (scalarPolynomialMap
141           (k := k) (K := LaurentSeries k) (Fin n)),
142       MvPolynomial.eval (dehomogenizedPoint (laurentColumn q)) f = 0)
143     (htangent :
144       zariskiTangentSpace (dehomogenizedPoint (laurentColumn q))
145           (I.map (scalarPolynomialMap
146             (k := k) (K := LaurentSeries k) (Fin n))) ≤
147         dehomogenizedTangentSpan (laurentColumn q)
148           (laurentNonpositionTangentMatrix Z tau)) :
149     residueColumn (fun i : Fin n ↦ ell i.succ) ∈
150       laurentEquationConormalDirectionClosure I := by
151   apply residue_mem_laurentEquationConormalDirectionClosure
152     I (dehomogenizedPoint (laurentColumn q)) (fun i : Fin n ↦ ell i.succ)
153   exact laurentPhasePoint_mem_equationConormalLocus_of_zariski_le_span
154     (I.map (scalarPolynomialMap
155       (k := k) (K := LaurentSeries k) (Fin n)))
156     q ell Z tau hq0 hrow hbase htangent
