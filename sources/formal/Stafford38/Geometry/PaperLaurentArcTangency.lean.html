8 /-!
9 # Generic Laurent tangency from retained power-series equations
10 
11 If a polynomial ideal vanishes identically along a completed power-series
12 arc, applying any power-series derivation and then passing its chain-rule
13 identity to the Laurent fraction field gives an actual tangent vector at the
14 generic Laurent point.  In particular, the coefficientwise residue-field
15 derivations and the uniformizer derivative are tangent there.
16 -/

32 private theorem differentialCovector_mul
33     {m : ℕ} (y v : Fin m → LaurentSeries κ)
34     (f g : MvPolynomial (Fin m) (LaurentSeries κ)) :
35     differentialCovector y (f * g) v =
36       MvPolynomial.eval y f * differentialCovector y g v +
37       MvPolynomial.eval y g * differentialCovector y f v := by
38   rw [Stafford38.Geometry.AffineConormalSpan.differentialCovector_mul]
39   simp only [LinearMap.add_apply, LinearMap.smul_apply,
40     differentialCovector_apply, smul_eq_mul]

43 private theorem algebraMap_eval₂_eq_eval₂_algebraMap
44     {m : ℕ} (q : Fin m → PowerSeries κ)
45     (f : MvPolynomial (Fin m) k) :
46     algebraMap (PowerSeries κ) (LaurentSeries κ)
47         (MvPolynomial.eval₂ (algebraMap k (PowerSeries κ)) q f) =
48       MvPolynomial.eval₂ (algebraMap k (LaurentSeries κ))
49         (fun i ↦ algebraMap (PowerSeries κ) (LaurentSeries κ) (q i)) f := by
50   have hmap := MvPolynomial.eval₂_comp_left
51     (algebraMap (PowerSeries κ) (LaurentSeries κ))
52     (algebraMap k (PowerSeries κ)) q f
53   have hcoeff :
54       (algebraMap (PowerSeries κ) (LaurentSeries κ)).comp
55           (algebraMap k (PowerSeries κ)) = algebraMap k (LaurentSeries κ) := by
56     ext a
57     rw [HahnSeries.algebraMap_apply']
58     rfl
59   rw [hcoeff] at hmap
60   exact hmap

62 /-- A derivation of the retained power-series ring produces a tangent vector
63 at the generic Laurent point whenever the actual equations vanish on the
64 arc. -/
65 theorem powerSeriesDerivationVector_mem_genericZariskiTangentSpace
66     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) k))
67     (q : Fin m → PowerSeries κ)
68     (D : Derivation k (PowerSeries κ) (PowerSeries κ))
69     (hq : ∀ f ∈ I,
70       MvPolynomial.eval₂ (algebraMap k (PowerSeries κ)) q f = 0) :
71     (fun i ↦ algebraMap (PowerSeries κ) (LaurentSeries κ) (D (q i))) ∈
72       zariskiTangentSpace
73         (fun i ↦ algebraMap (PowerSeries κ) (LaurentSeries κ) (q i))
74         (I.map (MvPolynomial.map (algebraMap k (LaurentSeries κ)))) := by
75   let y : Fin m → LaurentSeries κ :=
76     fun i ↦ algebraMap (PowerSeries κ) (LaurentSeries κ) (q i)
77   let v : Fin m → LaurentSeries κ :=
78     fun i ↦ algebraMap (PowerSeries κ) (LaurentSeries κ) (D (q i))
79   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator]
80   intro φ hφ
81   have hmap : ∀ g : MvPolynomial (Fin m) (LaurentSeries κ),
82       g ∈ I.map (MvPolynomial.map (algebraMap k (LaurentSeries κ))) →
83       MvPolynomial.eval y g = 0 ∧ differentialCovector y g v = 0 := by
84     intro g hg
85     rw [Ideal.map] at hg
86     induction hg using Submodule.span_induction with
87     | mem g hg =>
88         rcases hg with ⟨f, hf, rfl⟩
89         have heval : MvPolynomial.eval y
90             (MvPolynomial.map (algebraMap k (LaurentSeries κ)) f) = 0 := by
91           rw [MvPolynomial.eval_map]
92           rw [← algebraMap_eval₂_eq_eval₂_algebraMap]
93           simpa using congrArg (algebraMap (PowerSeries κ) (LaurentSeries κ))
94             (hq f hf)
95         have hderiv := derivation_eval₂ D f q
96         rw [hq f hf, D.map_zero] at hderiv
97         have hsum : ∑ i : Fin m,
98             MvPolynomial.eval₂ (algebraMap k (PowerSeries κ)) q
99                 (MvPolynomial.pderiv i f) * D (q i) = 0 := hderiv.symm
100         have hsumL := congrArg (algebraMap (PowerSeries κ) (LaurentSeries κ)) hsum
101         simp only [map_zero, map_sum] at hsumL
102         have hdifferential : differentialCovector y
103             (MvPolynomial.map (algebraMap k (LaurentSeries κ)) f) v = 0 := by
104           unfold differentialCovector
105           simp only [differentialCovector_apply, differentialAt,
106             MvPolynomial.pderiv_map]
107           have hterms : ∀ i : Fin m,
108               MvPolynomial.eval y
109                   (MvPolynomial.map (algebraMap k (LaurentSeries κ))
110                     (MvPolynomial.pderiv i f)) * v i =
111                 algebraMap (PowerSeries κ) (LaurentSeries κ)
112                     (MvPolynomial.eval₂ (algebraMap k (PowerSeries κ)) q
113                       (MvPolynomial.pderiv i f) * D (q i)) := by
114             intro i
115             rw [MvPolynomial.eval_map]
116             rw [← algebraMap_eval₂_eq_eval₂_algebraMap]
117             simp [v]
118           change (∑ i : Fin m,
119             MvPolynomial.eval y
120               (MvPolynomial.map (algebraMap k (LaurentSeries κ))
121                 (MvPolynomial.pderiv i f)) * v i) = 0
122           simp_rw [hterms]
123           exact hsumL
124         exact ⟨heval, hdifferential⟩
125     | zero => simp [differentialCovector, differentialAt, map_zero]
126     | add f g hf hg ihf ihg =>
127         refine ⟨by rw [MvPolynomial.eval_add, ihf.1, ihg.1, add_zero], ?_⟩
128         change (∑ i : Fin m,
129           MvPolynomial.eval y (MvPolynomial.pderiv i (f + g)) * v i) = 0
130         simp only [map_add, MvPolynomial.eval_add, add_mul]
131         rw [Finset.sum_add_distrib]
132         have hif : (∑ i : Fin m,
133             MvPolynomial.eval y (MvPolynomial.pderiv i f) * v i) = 0 := by
134           simpa [differentialCovector, differentialAt] using ihf.2
135         have hig : (∑ i : Fin m,
136             MvPolynomial.eval y (MvPolynomial.pderiv i g) * v i) = 0 := by
137           simpa [differentialCovector, differentialAt] using ihg.2
138         rw [hif, hig]
139         simp
140     | smul a f hf ih =>
141         refine ⟨by simpa [MvPolynomial.smul_eq_C_mul, ih.1], ?_⟩
142         have hprod := differentialCovector_mul y v a f
143         change differentialCovector y (a * f) v = 0
144         rw [hprod, ih.1, ih.2]
145         simp
146   induction hφ using Submodule.span_induction with
147   | mem f hf =>
148       rcases hf with ⟨g, rfl⟩
149       exact (hmap g.1 g.2).2
150   | zero => simp
151   | add φ ψ hφ hψ ihφ ihψ =>
152       change φ v + ψ v = 0
153       rw [ihφ, ihψ]
154       simp
155   | smul a φ hφ ih =>
156       change a * φ v = 0
157       rw [ih]
158       simp

160 /-- Coefficientwise residue-field derivations are tangent at the generic
161 Laurent point of an actual retained arc. -/
162 theorem coefficientwiseDerivation_mem_genericZariskiTangentSpace
163     {m : ℕ} (I : Ideal (MvPolynomial (Fin m) k))
164     (q : Fin m → PowerSeries κ) (D : Derivation k κ κ)
165     (hq : ∀ f ∈ I,
166       MvPolynomial.eval₂ (algebraMap k (PowerSeries κ)) q f = 0) :
167     (fun i ↦ algebraMap (PowerSeries κ) (LaurentSeries κ)
168         (coefficientwiseDerivation D (q i))) ∈
169       zariskiTangentSpace
170         (fun i ↦ algebraMap (PowerSeries κ) (LaurentSeries κ) (q i))
171         (I.map (MvPolynomial.map (algebraMap k (LaurentSeries κ)))) :=
172   powerSeriesDerivationVector_mem_genericZariskiTangentSpace
173     I q (coefficientwiseDerivation D) hq
