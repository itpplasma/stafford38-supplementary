10 # Transport from a relative rational function field to the coordinate DVR
11 
12 This file identifies the fraction field of the coordinate-zero local ring
13 `E[X]_(X)` with the concrete simple transcendental field `E(x)`.  The
14 equivalence is compatible with evaluation at `x`; it therefore transports the
15 finite separable tower constructed in `RelativeDivisorialTower` to the exact
16 input expected by `DivisorialBoundaryExtension`.
17 -/

35 /-- Evaluation at a transcendental element identifies the polynomial
36 fraction field with the concrete intermediate field generated by that
37 element. -/
38 def polynomialFractionEquivAdjoin
39     (E : Type u) [Field E]
40     {K : Type v} [Field K] [Algebra E K]
41     (x : K) (hx : Transcendental E x) :
42     FractionRing (Polynomial E) ≃ₐ[E]
43       IntermediateField.adjoin E ({x} : Set K) := by
44   let g : Polynomial E →ₐ[E] K := Polynomial.aeval x
45   let i : FractionRing (Polynomial E) →ₐ[E] K :=
46     IsFractionRing.liftAlgHom
47       (K := FractionRing (Polynomial E))
48       (transcendental_iff_injective.mp hx)
49   exact (AlgEquiv.ofInjectiveField i).trans
50     (IntermediateField.equivOfEq
51       (IsFractionRing.liftAlgHom_fieldRange_eq_of_range_eq
52         (g := g)
53         (transcendental_iff_injective.mp hx)
54         (Algebra.adjoin_singleton_eq_range_aeval E x).symm))
55 
56 @[simp]
57 theorem polynomialFractionEquivAdjoin_algebraMap
58     (E : Type u) [Field E]
59     {K : Type v} [Field K] [Algebra E K]
60     (x : K) (hx : Transcendental E x) (p : Polynomial E) :
61     ((polynomialFractionEquivAdjoin E x hx
62         (algebraMap (Polynomial E) (FractionRing (Polynomial E)) p) :
63       IntermediateField.adjoin E ({x} : Set K)) : K) =
64       Polynomial.aeval x p := by
65   let g : Polynomial E →+* K := (Polynomial.aeval x).toRingHom
66   change IsFractionRing.lift (K := FractionRing (Polynomial E))
67       (transcendental_iff_injective.mp hx)
68       (algebraMap (Polynomial E) (FractionRing (Polynomial E)) p) =
69     Polynomial.aeval x p
70   exact IsFractionRing.lift_algebraMap
71     (transcendental_iff_injective.mp hx) p

73 /-- Localizing `E[X]` away from `(X)` does not change its fraction field.
74 Composing that canonical comparison with transcendental evaluation realizes
75 the result as the concrete field `E(x)`. -/
76 def coordinateLocalFractionEquivAdjoin
77     (E : Type u) [Field E]
78     {K : Type v} [Field K] [Algebra E K]
79     (x : K) (hx : Transcendental E x) :
80     FractionRing (CoordinateZeroLocalRing E) ≃ₐ[E]
81       IntermediateField.adjoin E ({x} : Set K) := by
82   letI : IsScalarTower E (CoordinateZeroLocalRing E)
83       (FractionRing (Polynomial E)) :=
84     IsScalarTower.of_algebraMap_eq fun a => by
85       rw [IsScalarTower.algebraMap_apply E (Polynomial E)
86         (CoordinateZeroLocalRing E)]
87       rw [IsScalarTower.algebraMap_apply E (Polynomial E)
88         (FractionRing (Polynomial E))]
89       exact IsScalarTower.algebraMap_apply (Polynomial E)
90         (CoordinateZeroLocalRing E) (FractionRing (Polynomial E))
91         (algebraMap E (Polynomial E) a)
92   let e₀ : FractionRing (CoordinateZeroLocalRing E) ≃ₐ[E]
93       FractionRing (Polynomial E) :=
94     (FractionRing.algEquiv (CoordinateZeroLocalRing E)
95       (FractionRing (Polynomial E))).restrictScalars E
96   exact e₀.trans (polynomialFractionEquivAdjoin E x hx)

98 /-- The fraction-field comparison sends the distinguished local coordinate
99 to the selected element `x`, not merely to an abstract generator. -/
100 @[simp]
101 theorem coordinateLocalFractionEquivAdjoin_X
102     (E : Type u) [Field E]
103     {K : Type v} [Field K] [Algebra E K]
104     (x : K) (hx : Transcendental E x) :
105     let R := CoordinateZeroLocalRing E
106     let L := FractionRing R
107     (((coordinateLocalFractionEquivAdjoin E x hx
108         (algebraMap R L
109           (algebraMap (Polynomial E) R Polynomial.X)) :
110       IntermediateField.adjoin E ({x} : Set K)) : K)) = x := by
111   dsimp only
112   letI : IsScalarTower E (CoordinateZeroLocalRing E)
113       (FractionRing (Polynomial E)) :=
114     IsScalarTower.of_algebraMap_eq fun a => by
115       rw [IsScalarTower.algebraMap_apply E (Polynomial E)
116         (CoordinateZeroLocalRing E)]
117       rw [IsScalarTower.algebraMap_apply E (Polynomial E)
118         (FractionRing (Polynomial E))]
119       exact IsScalarTower.algebraMap_apply (Polynomial E)
120         (CoordinateZeroLocalRing E) (FractionRing (Polynomial E))
121         (algebraMap E (Polynomial E) a)
122   rw [coordinateLocalFractionEquivAdjoin, AlgEquiv.trans_apply]
123   change ↑((polynomialFractionEquivAdjoin E x hx)
124     ((FractionRing.algEquiv (CoordinateZeroLocalRing E)
125       (FractionRing (Polynomial E)))
126       ((algebraMap (CoordinateZeroLocalRing E)
127         (FractionRing (CoordinateZeroLocalRing E)))
128         ((algebraMap (Polynomial E) (CoordinateZeroLocalRing E))
129           Polynomial.X)))) = x
130   rw [AlgEquiv.commutes]
131   rw [← IsScalarTower.algebraMap_apply (Polynomial E)
132     (CoordinateZeroLocalRing E) (FractionRing (Polynomial E))]
133   simpa only [Polynomial.aeval_X] using
134     polynomialFractionEquivAdjoin_algebraMap E x hx Polynomial.X

136 /-- A finite separable extension of the concrete field `E(x)` gives the exact
137 fraction-field action needed by the coordinate DVR theorem.  Both finiteness
138 and separability are transported across the displayed field equivalence; the
139 coordinate compatibility is discharged by
140 `coordinateLocalFractionEquivAdjoin_X`. -/
141 theorem exists_discreteBoundaryRefinement_of_adjoin
142     (E : Type u) [Field E]
143     {K : Type v} [Field K] [Algebra E K]
144     (x : K) (hx : Transcendental E x)
145     (hfin : FiniteDimensional
146       (IntermediateField.adjoin E ({x} : Set K)) K)
147     (hsep : Algebra.IsSeparable
148       (IntermediateField.adjoin E ({x} : Set K)) K) :
149     ∃ V : ValuationSubring K,
150       IsDiscreteValuationRing V.toSubring ∧
151       ∃ xV : V.toSubring,
152         (xV : K) = x ∧ xV ≠ 0 ∧ ¬IsUnit xV := by
153   let R := CoordinateZeroLocalRing E
154   let L := FractionRing R
155   let F := IntermediateField.adjoin E ({x} : Set K)
156   let e : L ≃ₐ[E] F := coordinateLocalFractionEquivAdjoin E x hx
157   letI : Algebra R F :=
158     (e.toRingEquiv.toRingHom.comp (algebraMap R L)).toAlgebra
159   let eR : L ≃ₐ[R] F :=
160     { e.toRingEquiv with
161       commutes' := fun _ => rfl }
162   letI : IsFractionRing R F :=
163     IsLocalization.isLocalization_of_algEquiv (nonZeroDivisors R) eR
164   letI : Algebra R K :=
165     ((algebraMap F K).comp (algebraMap R F)).toAlgebra
166   letI : IsScalarTower R F K :=
167     IsScalarTower.of_algebraMap_eq fun _ => rfl
168   letI : FiniteDimensional F K := hfin
169   letI : Algebra.IsSeparable F K := hsep
170   let q : R := algebraMap (Polynomial E) R Polynomial.X
171   have hq_ne : q ≠ 0 := by
172     intro hq
173     apply Polynomial.X_ne_zero (R := E)
174     apply IsLocalization.injective R
175       (coordinateZeroPrime E).primeCompl_le_nonZeroDivisors
176     simpa only [q, map_zero] using hq
177   have hq_nonunit : ¬IsUnit q := by
178     rw [← mem_nonunits_iff, ← mem_maximalIdeal]
179     exact (IsLocalization.AtPrime.to_map_mem_maximal_iff
180       R (coordinateZeroPrime E) Polynomial.X).2
181         (Ideal.mem_span_singleton_self Polynomial.X)
182   have hcoordinate : algebraMap R K q = x := by
183     change (((e (algebraMap R L q) : F) : K)) = x
184     exact coordinateLocalFractionEquivAdjoin_X E x hx
185   obtain ⟨V, hV, qV, hqV, hqV_ne, hqV_nonunit⟩ :=
186     exists_discreteValuationSubring_over_maximalIdeal
187       (A := R) (F := F) (L := K) q hq_ne hq_nonunit
188   exact ⟨V, hV, qV, hqV.trans hcoordinate, hqV_ne, hqV_nonunit⟩

190 /-- The relative tower constructed from a finitely generated characteristic-
191 zero function field therefore has an actual discrete boundary place above
192 the selected coordinate. -/
193 theorem exists_discreteBoundaryRefinement_of_fg_charZero
194     (k : Type u) [Field k] [CharZero k]
195     {K : Type v} [Field K] [Algebra k K]
196     (hfg : (⊤ : IntermediateField k K).FG)
197     (x : K) (hx : Transcendental k x) :
198     ∃ V : ValuationSubring K,
199       IsDiscreteValuationRing V.toSubring ∧
200       ∃ xV : V.toSubring,
201         (xV : K) = x ∧ xV ≠ 0 ∧ ¬IsUnit xV := by
202   obtain ⟨s, E, F, hxs, hs, hE, hF, hrestrict, hxE, hxF,
203       hfin, hsep⟩ :=
204     exists_relative_finite_separable_tower_of_charZero k hfg x hx
205   subst F
206   exact exists_discreteBoundaryRefinement_of_adjoin
207     E x hxE hfin hsep

209 /-- Structured form of the preceding theorem, ready for the downstream
210 projective order-gap consumer. -/
211 theorem exists_discreteBoundaryRefinement_of_fg_charZero_structured
212     (k : Type u) [Field k] [CharZero k]
213     {K : Type v} [Field K] [Algebra k K]
214     (hfg : (⊤ : IntermediateField k K).FG)
215     (x : K) (hx : Transcendental k x) :
216     Nonempty (DiscreteBoundaryRefinement k x) := by
217   obtain ⟨V, hV, xV, hxV, hxV_ne, hxV_nonunit⟩ :=
218     exists_discreteBoundaryRefinement_of_fg_charZero k hfg x hx
219   exact ⟨{
220     valuation := V
221     isDiscrete := hV
222     coordinate := xV
223     coordinate_eq := hxV
224     coordinate_ne := hxV_ne
225     coordinate_nonunit := hxV_nonunit
226   }⟩
