9 /-!
10 # Algebraic closure of the equation-defined affine conormal locus
11 
12 The equation-defined conormal locus over the contracted base zero set lies in
13 the reduced order support under the exact base-relative Gabber fragment. Since
14 that support is a polynomial zero locus, the algebraic closure hull of the
15 conormal locus lies there as well.
16 
17 This file uses the closure operator `V(I(S))` on field-valued affine points.
18 It does not identify a smooth scheme-theoretic conormal bundle or prove the
19 asymptotic conormal theorem at infinity.
20 -/

36 /-- Phase points whose base coordinate lies in `V(I)` and whose fibre
37 coordinate belongs to the equation-defined embedded conormal at that point. -/
38 def equationConormalLocus (I : Ideal (MvPolynomial (Fin n) k)) :
39     Set (PhaseVar n → k) :=
40   {q | (∀ f ∈ I,
41       MvPolynomial.eval (fun i => q (Sum.inl i)) f = 0) ∧
42     coordinateCovector (fun i => q (Sum.inr i)) ∈
43       affineConormalSpace (fun i => q (Sum.inl i)) I}
44 
45 /-- Algebraic closure hull of the equation-defined conormal locus. -/
46 def equationConormalClosure (I : Ideal (MvPolynomial (Fin n) k)) :
47     Set (PhaseVar n → k) :=
48   MvPolynomial.zeroLocus k
49     (MvPolynomial.vanishingIdeal k (equationConormalLocus I))
50 
51 /-- A subset of a polynomial zero locus has its algebraic closure hull in the
52 same zero locus. -/
53 theorem zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
54     (S : Set (PhaseVar n → k)) (J : Ideal (SymbolRing k n))
55     (h : S ⊆ MvPolynomial.zeroLocus k J) :
56     MvPolynomial.zeroLocus k (MvPolynomial.vanishingIdeal k S) ⊆
57       MvPolynomial.zeroLocus k J := by
58   apply MvPolynomial.zeroLocus_anti_mono
59   exact MvPolynomial.le_zeroLocus_iff_le_vanishingIdeal.mp h
60 
61 /-- Under the exact base-relative Gabber condition, every equation-defined
62 conormal point over the contracted base zero locus lies in the reduced order
63 support. -/
64 theorem equationConormalLocus_subset_reducedOrderSupport
65     [CharZero k]
66     (W : RightIdeal (WeylAlg k n))
67     (hJ : IsBaseRelativePoisson (reducedOrderSupportIdeal k W)) :
68     equationConormalLocus (reducedOrderBaseIdeal k W) ⊆
69       MvPolynomial.zeroLocus k (reducedOrderSupportIdeal k W) := by
70   intro q hq
71   let y : Fin n → k := fun i => q (Sum.inl i)
72   let ξ : Fin n → k := fun i => q (Sum.inr i)
73   have hs := reducedOrderBaseZero_affineConormal_coordinatePoint_isCommonZero
74     W hJ y hq.1 ξ hq.2
75   have hsplit : Sum.elim y ξ = q := by
76     funext i
77     rcases i with i | i <;> rfl
78   rw [← hsplit]
79   exact hs
80 
81 /-- The algebraic closure hull of the equation-defined conormal locus remains
82 inside the reduced order support. -/
83 theorem equationConormalClosure_subset_reducedOrderSupport
84     [CharZero k]
85     (W : RightIdeal (WeylAlg k n))
86     (hJ : IsBaseRelativePoisson (reducedOrderSupportIdeal k W)) :
87     equationConormalClosure (reducedOrderBaseIdeal k W) ⊆
88       MvPolynomial.zeroLocus k (reducedOrderSupportIdeal k W) := by
89   exact zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
90     (equationConormalLocus (reducedOrderBaseIdeal k W))
91     (reducedOrderSupportIdeal k W)
92     (equationConormalLocus_subset_reducedOrderSupport W hJ)
