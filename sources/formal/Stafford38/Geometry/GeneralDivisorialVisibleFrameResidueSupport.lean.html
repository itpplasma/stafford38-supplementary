26 do not duplicate or strengthen its data: they use the stored visible frame's

29 theorem witness_q0_mem_maximal
30     {k : Type u} [Field k] [CharZero k]
31     {m : ℕ} (hm : 0 < m)
32     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
33     (w : GeneralDivisorialVisibleFrameWitness hm P) :
34     letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
35         (ComponentFractionField P) := w.column.W.ambientAlgebra
36     let V := w.column.W.place.valuation.toSubring
37     letI : IsLocalRing V := w.column.W.place.isDiscrete.toIsLocalRing
38     w.column.q 0 ∈ maximalIdeal V := by
39   let C := w.column
40   letI : Algebra (CoordinateZeroLocalRing C.W.coefficientField)
41       (ComponentFractionField P) := C.W.ambientAlgebra
42   let V := C.W.place.valuation.toSubring
43   letI : IsLocalRing V := C.W.place.isDiscrete.toIsLocalRing
44   letI : Algebra C.W.coefficientField V :=
45     (relativeCoefficientMap C.W.coefficientField C.W.place).toAlgebra
46   letI : Algebra k V := C.groundCoeff.toAlgebra
47   letI : Algebra V (ComponentFractionField P) := V.subtype.toAlgebra
48   letI : IsScalarTower k V (ComponentFractionField P) := C.groundTower
49   let D := w.differential.core.D
50   have hpos : 0 < D.a := lt_of_lt_of_le Nat.zero_lt_one D.one_le_a
51   have hpow : D.t ^ D.a ∈ maximalIdeal V :=
52     Ideal.pow_mem_of_mem _ D.t_mem D.a hpos
53   change C.q 0 ∈ maximalIdeal V
54   rw [← w.differential.core.D_Q0, D.Q₀_eq]
55   exact Ideal.mul_mem_right _ _ hpow

57 theorem witness_q0_residue_eq_zero
58     {k : Type u} [Field k] [CharZero k]
59     {m : ℕ} (hm : 0 < m)
60     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
61     (w : GeneralDivisorialVisibleFrameWitness hm P) :
62     letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
63         (ComponentFractionField P) := w.column.W.ambientAlgebra
64     let V := w.column.W.place.valuation.toSubring
65     letI : IsLocalRing V := w.column.W.place.isDiscrete.toIsLocalRing
66     residue V (w.column.q 0) = 0 := by
67   letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
68       (ComponentFractionField P) := w.column.W.ambientAlgebra
69   let V := w.column.W.place.valuation.toSubring
70   letI : IsLocalRing V := w.column.W.place.isDiscrete.toIsLocalRing
71   exact (IsLocalRing.residue_eq_zero_iff _).2
72     (witness_q0_mem_maximal hm P w)
