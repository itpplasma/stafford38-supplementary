9 /-!
10 # Scalar extension and geometric points
11 
12 This file gives the coefficient-extension bridge for the field-valued affine
13 geometry used by the characteristic-support and conormal arguments.  All
14 polynomial maps, ideal maps/comaps, and evaluation maps are displayed
15 explicitly.
16 
17 For an ideal `J` over `k`, its geometric reduced extension is defined as the
18 radical of the extended ideal, not as the extension of `J.radical`.  The two
19 zero loci agree over every extension field.  No assertion that radicals,
20 closures, or ideal contractions commute with base change is made here.
21 -/

44 /-- Coefficient extension on a multivariable polynomial ring. -/
45 def scalarPolynomialMap (σ : Type*) :
46     MvPolynomial σ k →+* MvPolynomial σ K :=
47   MvPolynomial.map (algebraMap k K)
48 
49 /-- Coefficient extension is injective for a field extension. -/
50 theorem scalarPolynomialMap_injective (σ : Type*) :
51     Function.Injective (scalarPolynomialMap (k := k) (K := K) σ) :=
52   MvPolynomial.map_injective (algebraMap k K)
53     (FaithfulSMul.algebraMap_injective k K)

55 /-- Evaluation of a coefficient-extended polynomial at a `K`-point is
56 exactly `eval₂` with the structure map on coefficients. -/
57 theorem eval_scalarPolynomialMap (q : σ → K) (f : MvPolynomial σ k) :
58     MvPolynomial.eval q (scalarPolynomialMap (k := k) (K := K) σ f) =
59       MvPolynomial.eval₂ (algebraMap k K) q f := by
60   exact MvPolynomial.eval_map (algebraMap k K) q f

62 /-- Exact point semantics of ideal extension: a `K`-point kills the extended
63 ideal iff it kills the image of every member of the original ideal. -/
64 theorem mem_zeroLocus_map_iff
65     (J : Ideal (MvPolynomial σ k)) (q : σ → K) :
66     q ∈ MvPolynomial.zeroLocus K
67         (J.map (scalarPolynomialMap (k := k) (K := K) σ)) ↔
68       ∀ f ∈ J, MvPolynomial.eval₂ (algebraMap k K) q f = 0 := by
69   constructor
70   · intro hq f hf
71     have h := hq
72       (scalarPolynomialMap (k := k) (K := K) σ f)
73       (Ideal.mem_map_of_mem
74         (scalarPolynomialMap (k := k) (K := K) σ) hf)
75     simpa [eval_scalarPolynomialMap] using h
76   · intro hq
77     rw [Ideal.map, MvPolynomial.zeroLocus_span]
78     rintro _ ⟨f, hf, rfl⟩
79     simpa [eval_scalarPolynomialMap] using hq f hf

81 /-- Polynomials over `k` vanishing on a set of `K`-points.  This is the
82 contraction of the ordinary `K`-valued vanishing ideal. -/
83 def extensionValuedVanishingIdeal (S : Set (σ → K)) :
84     Ideal (MvPolynomial σ k) :=
85   (MvPolynomial.vanishingIdeal K S).comap
86     (scalarPolynomialMap (k := k) (K := K) σ)
87 
88 theorem mem_extensionValuedVanishingIdeal_iff
89     (S : Set (σ → K)) (f : MvPolynomial σ k) :
90     f ∈ extensionValuedVanishingIdeal (k := k) (K := K) S ↔
91       ∀ q ∈ S, MvPolynomial.eval₂ (algebraMap k K) q f = 0 := by
92   simp [extensionValuedVanishingIdeal, eval_scalarPolynomialMap]

94 /-- Exact Galois bridge between ideal extension and vanishing on
95 extension-valued points. -/
96 theorem subset_zeroLocus_map_iff_le_extensionValuedVanishingIdeal
97     (J : Ideal (MvPolynomial σ k)) (S : Set (σ → K)) :
98     S ⊆ MvPolynomial.zeroLocus K
99         (J.map (scalarPolynomialMap (k := k) (K := K) σ)) ↔
100       J ≤ extensionValuedVanishingIdeal (k := k) (K := K) S := by
101   constructor
102   · intro h f hf
103     rw [mem_extensionValuedVanishingIdeal_iff]
104     intro q hq
105     exact (mem_zeroLocus_map_iff J q).mp (h hq) f hf
106   · intro h q hq
107     rw [mem_zeroLocus_map_iff]
108     intro f hf
109     exact (mem_extensionValuedVanishingIdeal_iff S f).mp (h hf) q hq

111 /-- Taking a radical after scalar extension does not change the set of
112 `K`-valued zeros.  This does not identify that radical with the extension of
113 the ground-field radical. -/
114 theorem zeroLocus_radical (L : Ideal (MvPolynomial σ K)) :
115     MvPolynomial.zeroLocus K L.radical = MvPolynomial.zeroLocus K L := by
116   apply Set.Subset.antisymm
117   · exact MvPolynomial.zeroLocus_anti_mono Ideal.le_radical
118   · intro q hq f hf
119     exact MvPolynomial.radical_le_vanishingIdeal_zeroLocus L hf q hq

121 /-- The reduced geometric extension of an ideal.  Reduction is deliberately
122 performed after coefficient extension. -/
123 def geometricRadicalExtension (J : Ideal (MvPolynomial σ k)) :
124     Ideal (MvPolynomial σ K) :=
125   (J.map (scalarPolynomialMap (k := k) (K := K) σ)).radical

127 /-- Exact geometric-point semantics of the reduced extension. -/
128 theorem mem_zeroLocus_geometricRadicalExtension_iff
129     (J : Ideal (MvPolynomial σ k)) (q : σ → K) :
130     q ∈ MvPolynomial.zeroLocus K
131         (geometricRadicalExtension (k := k) (K := K) J) ↔
132       ∀ f ∈ J, MvPolynomial.eval₂ (algebraMap k K) q f = 0 := by
133   rw [geometricRadicalExtension, zeroLocus_radical,
134     mem_zeroLocus_map_iff]

136 /-- Coefficient extension commutes with the explicit embedding of base
137 coordinates into phase space. -/
138 theorem scalarPolynomialMap_baseLift (f : MvPolynomial (Fin n) k) :
139     scalarPolynomialMap (k := k) (K := K) (PhaseVar n) (baseLift f) =
140       baseLift (scalarPolynomialMap (k := k) (K := K) (Fin n) f) := by
141   exact MvPolynomial.map_rename (algebraMap k K) Sum.inl f

143 /-- The always-valid map/comap comparison for the base-coordinate square.
144 The orientation is important: extension of the old contraction is contained
145 in the contraction of the extended ideal.  The reverse inclusion would need
146 the flat Cartesian base-change/intersection theorem for this polynomial
147 square; no such equality is assumed. -/
148 theorem map_baseContraction_le_extendedBaseContraction
149     (J : Ideal (SymbolRing k n)) :
150     (J.comap
151         (baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom).map
152         (scalarPolynomialMap (k := k) (K := K) (Fin n)) ≤
153       (J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))).comap
154         (baseLift : MvPolynomial (Fin n) K →ₐ[K] SymbolRing K n).toRingHom := by
155   rw [Ideal.map_le_iff_le_comap]
156   intro f hf
157   change baseLift (scalarPolynomialMap (k := k) (K := K) (Fin n) f) ∈
158     J.map (scalarPolynomialMap (k := k) (K := K) (PhaseVar n))
159   rw [← scalarPolynomialMap_baseLift]
160   exact Ideal.mem_map_of_mem
161     (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) hf
162 
163 /-- The corresponding comparison remains valid when the extended phase
164 ideal is reduced after base change. -/
165 theorem map_baseContraction_le_geometricBaseContraction
166     (J : Ideal (SymbolRing k n)) :
167     (J.comap
168         (baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom).map
169         (scalarPolynomialMap (k := k) (K := K) (Fin n)) ≤
170       (geometricRadicalExtension (k := k) (K := K) J).comap
171         (baseLift : MvPolynomial (Fin n) K →ₐ[K] SymbolRing K n).toRingHom := by
172   exact (map_baseContraction_le_extendedBaseContraction J).trans
173     (Ideal.comap_mono Ideal.le_radical)

175 /-- A phase point of an extended ideal projects to a point of the contraction
176 of that extended ideal. -/
177 theorem baseProjection_mem_zeroLocus_comap
178     (L : Ideal (SymbolRing K n)) (q : PhaseVar n → K)
179     (hq : q ∈ MvPolynomial.zeroLocus K L) :
180     (fun i ↦ q (Sum.inl i)) ∈ MvPolynomial.zeroLocus K
181       (L.comap
182         (baseLift : MvPolynomial (Fin n) K →ₐ[K] SymbolRing K n).toRingHom) := by
183   intro f hf
184   have h := hq (baseLift f) hf
185   change MvPolynomial.eval q (MvPolynomial.rename Sum.inl f) = 0 at h
186   rw [MvPolynomial.eval_rename] at h
187   exact h

189 /-- Consequently, a geometric point of the reduced extension projects to a
190 zero of the scalar extension of the original base contraction.  This is the
191 strong unconditional one-way base-contraction theorem. -/
192 theorem geometricSupport_baseProjection_mem_groundContractionZeroLocus
193     (J : Ideal (SymbolRing k n)) (q : PhaseVar n → K)
194     (hq : q ∈ MvPolynomial.zeroLocus K
195       (geometricRadicalExtension (k := k) (K := K) J)) :
196     (fun i ↦ q (Sum.inl i)) ∈ MvPolynomial.zeroLocus K
197       ((J.comap
198           (baseLift : MvPolynomial (Fin n) k →ₐ[k] SymbolRing k n).toRingHom).map
199         (scalarPolynomialMap (k := k) (K := K) (Fin n))) := by
200   apply MvPolynomial.zeroLocus_anti_mono
201     (map_baseContraction_le_geometricBaseContraction J)
202   exact baseProjection_mem_zeroLocus_comap
203     (geometricRadicalExtension (k := k) (K := K) J) q hq

205 /-- Exact membership in the equation-defined conormal locus after extending
206 the equation ideal.  In particular, the base equations are evaluated by the
207 displayed `eval₂` map; no identification of closures across fields occurs. -/
208 theorem mem_equationConormalLocus_map_iff
209     (I : Ideal (MvPolynomial (Fin n) k)) (q : PhaseVar n → K) :
210     q ∈ equationConormalLocus
211         (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n))) ↔
212       (∀ f ∈ I,
213           MvPolynomial.eval₂ (algebraMap k K)
214             (fun i ↦ q (Sum.inl i)) f = 0) ∧
215         coordinateCovector (fun i ↦ q (Sum.inr i)) ∈
216           affineConormalSpace (fun i ↦ q (Sum.inl i))
217             (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n))) := by
218   rw [equationConormalLocus]
219   constructor
220   · intro h
221     refine ⟨?_, h.2⟩
222     exact (mem_zeroLocus_map_iff I (fun i ↦ q (Sum.inl i))).mp h.1
223   · intro h
224     refine ⟨?_, h.2⟩
225     exact (mem_zeroLocus_map_iff I (fun i ↦ q (Sum.inl i))).mpr h.1

227 /-- Partial derivatives commute with coefficient extension, with the
228 evaluation map at a geometric point shown explicitly. -/
229 theorem differentialAt_scalarPolynomialMap
230     (y : Fin n → K) (f : MvPolynomial (Fin n) k) (i : Fin n) :
231     differentialAt y
232         (scalarPolynomialMap (k := k) (K := K) (Fin n) f) i =
233       MvPolynomial.eval₂ (algebraMap k K) y
234         (MvPolynomial.pderiv i f) := by
235   change MvPolynomial.eval y
236       (MvPolynomial.pderiv i
237         (MvPolynomial.map (algebraMap k K) f)) = _
238   rw [MvPolynomial.pderiv_map, MvPolynomial.eval_map]

240 /-- Once a pointwise conormal containment has been proved over `K`, its
241 `K`-valued algebraic closure hull remains in the reduced geometric support.
242 Both closure and reduction are taken after base change. -/
243 theorem equationConormalClosure_map_subset_geometricRadicalExtension
244     (I : Ideal (MvPolynomial (Fin n) k))
245     (J : Ideal (SymbolRing k n))
246     (h : equationConormalLocus
247           (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n))) ⊆
248         MvPolynomial.zeroLocus K
249           (geometricRadicalExtension (k := k) (K := K) J)) :
250     equationConormalClosure
251         (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n))) ⊆
252       MvPolynomial.zeroLocus K
253         (geometricRadicalExtension (k := k) (K := K) J) := by
254   exact zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus

256       (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n))) )
257     (geometricRadicalExtension (k := k) (K := K) J) h

301 /-- A geometric zero of the actual contracted reduced support lifts to the
302 zero section of the actual geometric reduced support.  The proof transports
303 the ground-field degree-zero component through coefficient extension; it
304 does not identify the two base contractions. -/
305 theorem zeroSection_mem_of_mem_geometricReducedOrderBaseZeroSet
306     (W : RightIdeal (WeylAlg k n)) (y : Fin n → K)
307     (hy : ∀ f ∈ geometricReducedOrderBaseIdeal (k := k) (K := K) W,
308       MvPolynomial.eval y f = 0) :
309     zeroSectionPoint y ∈ MvPolynomial.zeroLocus K
310       (geometricReducedOrderSupportIdeal (k := k) (K := K) W) := by
311   rw [mem_zeroLocus_geometricReducedOrderSupportIdeal_iff]
312   intro P hP
313   obtain ⟨m, hm⟩ := (mem_reducedOrderSupportIdeal_iff k W P).mp hP
314   let f := fibreZeroSpecialization k (P ^ m)
315   have hcomponent :
316       (DirectSum.decompose
317           (orderDecomposition (k := k) (n := n)) (P ^ m) 0 :
318         SymbolRing k n) ∈ orderInitialIdeal k W :=
319     coe_mem_orderInitialIdeal_of_mem_orderSymbolRelation k W 0
320       (DirectSum.decompose
321         (orderDecomposition (k := k) (n := n)) (P ^ m) 0)
322       (decompose_mem_orderSymbolRelation_of_mem_orderInitialIdeal
323         k W (P ^ m) hm 0)
324   have hbaseLift : baseLift f ∈ reducedOrderSupportIdeal k W := by
325     rw [← zeroComponent_eq_baseLift_fibreZeroSpecialization]
326     exact orderInitialIdeal_le_reducedOrderSupportIdeal k W hcomponent
327   have hmapBase :
328       scalarPolynomialMap (k := k) (K := K) (Fin n) f ∈
329         geometricReducedOrderBaseIdeal (k := k) (K := K) W := by
330     change baseLift
331         (scalarPolynomialMap (k := k) (K := K) (Fin n) f) ∈
332       geometricReducedOrderSupportIdeal (k := k) (K := K) W
333     rw [← scalarPolynomialMap_baseLift]
334     exact Ideal.le_radical
335       (Ideal.mem_map_of_mem
336         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n)) hbaseLift)
337   have hfy :
338       MvPolynomial.eval y
339         (scalarPolynomialMap (k := k) (K := K) (Fin n) f) = 0 :=
340     hy _ hmapBase
341   have heval :
342       MvPolynomial.eval (zeroSectionPoint y)
343         ((scalarPolynomialMap (k := k) (K := K) (PhaseVar n) P) ^ m) = 0 := by
344     rw [← map_pow,
345       ← eval_fibreZeroSpecialization K y
346         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) (P ^ m)),
347       ← scalarPolynomialMap_fibreZeroSpecialization]
348     exact hfy
349   have hp0 :
350       MvPolynomial.eval (zeroSectionPoint y)
351         (scalarPolynomialMap (k := k) (K := K) (PhaseVar n) P) = 0 := by
352     exact (pow_eq_zero_iff'.mp (by simpa only [map_pow] using heval)).1
353   simpa [eval_scalarPolynomialMap] using hp0

355 /-- Exact geometric conormal consumer for the scalar-extended reduced order
356 support.  The sole substantive hypothesis left to the caller is the
357 base-relative Poisson condition on the post-base-change radical ideal. -/
358 theorem equationConormalClosure_geometricReducedOrderBase_subset_support
359     [CharZero K]
360     (W : RightIdeal (WeylAlg k n))
361     (hJ : IsBaseRelativePoisson
362       (geometricReducedOrderSupportIdeal (k := k) (K := K) W)) :
363     equationConormalClosure
364         (geometricReducedOrderBaseIdeal (k := k) (K := K) W) ⊆
365       MvPolynomial.zeroLocus K
366         (geometricReducedOrderSupportIdeal (k := k) (K := K) W) := by
367   apply zeroLocus_vanishingIdeal_mono_of_subset_zeroLocus
368   intro q hq
369   let y : Fin n → K := fun i ↦ q (Sum.inl i)
370   let ξ : Fin n → K := fun i ↦ q (Sum.inr i)
371   have hzero := zeroSection_mem_of_mem_geometricReducedOrderBaseZeroSet
372     W y hq.1
373   have hpoint := affineConormal_coordinatePoint_isCommonZero
374     (geometricReducedOrderSupportIdeal (k := k) (K := K) W) hJ y
375     hzero
376     (geometricReducedOrderBaseIdeal (k := k) (K := K) W)
377     (fun f ↦ f.2) ξ hq.2
378   have hsplit : Sum.elim y ξ = q := by
379     funext i
380     rcases i with i | i <;> rfl
381   rw [← hsplit]
382   exact hpoint
