18 theorem exists_constant_coordinate_equation_and_pure_axis
19     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
20     {m : ℕ} (hm : 0 < m)
21     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
22     (halg : IsAlgebraic k (componentCoordinate P ⟨0, hm⟩)) :
23     ∃ (c : k) (y : Fin m → k),
24       MvPolynomial.X ⟨0, hm⟩ - MvPolynomial.C c ∈ P.asIdeal ∧
25       (∀ f ∈ P.asIdeal, MvPolynomial.eval y f = 0) ∧
26       Sum.elim y (fun j ↦ if j = ⟨0, hm⟩ then 1 else 0) ∈
27         Stafford38.Geometry.AffineConormalClosure.equationConormalLocus
28           P.asIdeal := by
29   obtain ⟨c, hc⟩ | htrans :=
30     coordinate_constant_or_transcendental P ⟨0, hm⟩
31   · obtain ⟨M, hMmax, hPM⟩ := Ideal.exists_le_maximal P.asIdeal P.isPrime.ne_top
32     obtain ⟨y, hMy⟩ :=
33       MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hMmax
34     have hy : ∀ f ∈ P.asIdeal, MvPolynomial.eval y f = 0 := by
35       intro f hf
36       have hfM : f ∈ M := hPM hf
37       rw [hMy, MvPolynomial.mem_vanishingIdeal_singleton_iff] at hfM
38       exact hfM
39     refine ⟨c, y, hc, hy, ?_⟩
40     exact constantCoordinate_phasePoint_mem_equationConormalLocus
41       P.asIdeal y hy ⟨0, hm⟩ c hc
42   · exact False.elim (htrans halg)

48 /-- A constant-coordinate component contributes the coordinate axis directly
49  to the ground-field smooth conormal fibre closure. No Laurent arc is needed. -/
50 theorem coordinate_axis_mem_smooth_fibre_closure_of_coordinate_algebraic
51     {k : Type*} [Field k] [IsAlgClosed k] [CharZero k]
52     {m : ℕ} (hm : 0 < m)
53     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
54     (halg : IsAlgebraic k (componentCoordinate P ⟨0, hm⟩)) :
55     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
56       MvPolynomial.zeroLocus k
57         (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection P.asIdeal)) := by
58   obtain ⟨c, y, hc, hy, hpoint⟩ :=
59     exists_constant_coordinate_equation_and_pure_axis hm P halg
60   intro f hf
61   have hfull := fibreLift_mem_vanishingIdeal_equationConormal P.asIdeal P.isPrime f hf
62   have hz := hfull (Sum.elim y (fun i => if i = ⟨0, hm⟩ then 1 else 0)) hpoint
63   simpa [fibreLift, MvPolynomial.eval_rename, Function.comp_def] using hz
