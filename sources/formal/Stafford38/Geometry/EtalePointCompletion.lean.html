18 /-- At a prime over `q`, formal unramifiedness identifies the maximal ideal of the
19 local target with the image of `q`. -/
20 theorem maximalIdeal_eq_map_of_formallyEtale
21     {R B : Type*} [CommRing R] [CommRing B] [Algebra R B]
22     [Algebra.EssFiniteType R B]
23     (q : Ideal R) [q.IsPrime] (p : Ideal B) [p.IsPrime]
24     (hq : Ideal.under R p = q)
25     [Algebra.FormallyEtale R (Localization.AtPrime p)] :
26     IsLocalRing.maximalIdeal (Localization.AtPrime p) =
27       q.map (algebraMap R (Localization.AtPrime p)) := by
28   letI : p.LiesOver q := (Ideal.liesOver_iff p q).mpr hq.symm
29   letI : Algebra (Localization.AtPrime q) (Localization.AtPrime p) :=
30     Localization.AtPrime.algebraOfLiesOver q p
31   letI : IsScalarTower R (Localization.AtPrime q) (Localization.AtPrime p) := inferInstance
32   letI : Algebra.FormallyUnramified R (Localization.AtPrime p) := inferInstance
33   letI : Algebra.EssFiniteType (Localization.AtPrime q) (Localization.AtPrime p) :=
34     Algebra.EssFiniteType.of_comp R (Localization.AtPrime q) (Localization.AtPrime p)
35   letI : Algebra.FormallyUnramified (Localization.AtPrime q) (Localization.AtPrime p) :=
36     Algebra.FormallyUnramified.localization_base q.primeCompl
37   have hmax := Algebra.FormallyUnramified.map_maximalIdeal
38     (R := Localization.AtPrime q) (S := Localization.AtPrime p)
39   calc
40     IsLocalRing.maximalIdeal (Localization.AtPrime p) =
41         (IsLocalRing.maximalIdeal (Localization.AtPrime q)).map
42           (algebraMap (Localization.AtPrime q) (Localization.AtPrime p)) := hmax.symm
43     _ = (q.map (algebraMap R (Localization.AtPrime q))).map
44           (algebraMap (Localization.AtPrime q) (Localization.AtPrime p)) := by
45       rw [IsLocalization.AtPrime.map_eq_maximalIdeal q (Localization.AtPrime q)]
46     _ = q.map (algebraMap R (Localization.AtPrime p)) := by
47       rw [Ideal.map_map]
48       congr 1
49       exact (IsScalarTower.algebraMap_eq R (Localization.AtPrime q)
50         (Localization.AtPrime p)).symm

52 /-- Passing from a prime quotient to the residue quotient of its localization does not
53 change the residue algebra when the prime is maximal. -/
54 noncomputable def quotientAtPrimeAlgEquiv
55     {R B : Type*} [CommRing R] [CommRing B] [Algebra R B]
56     (p : Ideal B) [p.IsMaximal] :
57     (B ⧸ p) ≃ₐ[R]
58       (Localization.AtPrime p) ⧸ IsLocalRing.maximalIdeal (Localization.AtPrime p) := by
59   let T := Localization.AtPrime p
60   let J := IsLocalRing.maximalIdeal T
61   let f : B →ₐ[R] T := IsScalarTower.toAlgHom R B T
62   have hUnder : Ideal.under B J = p := IsLocalization.AtPrime.under_maximalIdeal T p
63   have hle : p ≤ J.comap f := by
64     change p ≤ J.under B
65     rw [hUnder]
66   let g : B ⧸ p →ₐ[R] T ⧸ J := Ideal.quotientMapₐ J f hle
67   have hinj : Function.Injective g := by
68     apply Ideal.quotientMap_injective' (H := hle)
69     change J.under B ≤ p
70     rw [hUnder]
71   have hsurj : Function.Surjective g := by
72     change Function.Surjective
73       (Ideal.quotientMap J (algebraMap B T) hle)
74     have hMaxUnder : (J.under B).IsMaximal := by
75       simpa [hUnder] using (inferInstance : p.IsMaximal)
76     exact IsLocalization.surjective_quotientMap_of_maximal_of_localization
77       p.primeCompl T (I := J) (J := p)
78       (H := by rw [hUnder]) hMaxUnder
79   exact AlgEquiv.ofBijective g ⟨hinj, hsurj⟩

81 theorem quotientAtPrimeAlgEquiv_mk
82     {R B : Type*} [CommRing R] [CommRing B] [Algebra R B]
83     (p : Ideal B) [p.IsMaximal] (b : B) :
84     quotientAtPrimeAlgEquiv (R := R) (B := B) p (Ideal.Quotient.mk p b) =
85       Ideal.Quotient.mkₐ R (IsLocalRing.maximalIdeal (Localization.AtPrime p))
86         (algebraMap B (Localization.AtPrime p) b) := by
87   let T := Localization.AtPrime p
88   let J := IsLocalRing.maximalIdeal T
89   let f : B →ₐ[R] T := IsScalarTower.toAlgHom R B T
90   have hUnder : Ideal.under B J = p := IsLocalization.AtPrime.under_maximalIdeal T p
91   have hle : p ≤ J.comap f := by
92     change p ≤ J.under B
93     rw [hUnder]
94   change (Ideal.quotientMapₐ J f hle) (Ideal.Quotient.mk p b) =
95     Ideal.Quotient.mkₐ R J (f b)
96   exact Ideal.quotient_map_mkₐ J f hle

98 theorem reverseStage_apply_mk
99     {R T : Type*} [CommRing R] [CommRing T] [Algebra R T]
100     (I : Ideal R) (J : Ideal T) (hJ : J = I.map (algebraMap R T))
101     (ψ : T →ₐ[R] AdicCompletion I R) (b : T) :
102     Stafford38.Geometry.FormallyEtaleCompletionEquivalence.reverseStage I J hJ ψ 1
103         (Ideal.Quotient.mkₐ R (J ^ 1) b) = AdicCompletion.evalₐ I 1 (ψ b) := by
104   let u : T →ₐ[R] R ⧸ I ^ 1 := (AdicCompletion.evalₐ I 1).comp ψ
105   have hker : (I ^ 1).map (algebraMap R T) ≤ RingHom.ker (u : T →+* R ⧸ I ^ 1) := by
106     apply (Ideal.map_le_iff_le_comap).2
107     intro a ha
108     change AdicCompletion.evalₐ I 1 (ψ (algebraMap R T a)) = 0
109     rw [ψ.commutes]
110     have ha' : a ∈ I := by simpa only [Submodule.pow_one] using ha
111     simp [Ideal.Quotient.eq_zero_iff_mem, ha']
112   have hpow : (I ^ 1).map (algebraMap R T) = J ^ 1 := by
113     rw [Ideal.map_pow, hJ]
114   have hkill : ∀ x ∈ J ^ 1, u x = 0 := by
115     intro x hx
116     apply RingHom.mem_ker.mp
117     exact hker (hpow.symm ▸ hx)
118   have hcomp := Ideal.Quotient.liftₐ_comp (R₁ := R) (J ^ 1) u hkill
119   change ((Ideal.Quotient.liftₐ (R₁ := R) (J ^ 1) u hkill).comp
120       (Ideal.Quotient.mkₐ R (J ^ 1))) b = _
121   exact DFunLike.congr_fun hcomp b

123 /-- The completion of the coordinate ring at a rational point agrees with the
124 completion of the etale local chart above that point. -/
125 noncomputable def completionEquivOfEtaleResidue
126     {R B : Type*} [CommRing R] [CommRing B] [Algebra R B]
127     [Algebra.EssFiniteType R B]
128     (q : Ideal R) [q.IsPrime] (p : Ideal B) [p.IsPrime] [p.IsMaximal]
129     (hq : Ideal.under R p = q)
130     (e : (B ⧸ p) ≃ₐ[R] (R ⧸ q))
131     [Algebra.FormallyEtale R (Localization.AtPrime p)] :
132     AdicCompletion q R ≃+*
133       AdicCompletion (IsLocalRing.maximalIdeal (Localization.AtPrime p))
134         (Localization.AtPrime p) := by
135   let J : Ideal (Localization.AtPrime p) :=
136     IsLocalRing.maximalIdeal (Localization.AtPrime p)
137   have hJ : J = q.map (algebraMap R (Localization.AtPrime p)) :=
138     maximalIdeal_eq_map_of_formallyEtale q p hq
139   haveI : Algebra.FormallyUnramified R (Localization.AtPrime p) := inferInstance
140   haveI : Algebra.FormallySmooth R (Localization.AtPrime p) := inferInstance
141   let locRes : (B ⧸ p) ≃ₐ[R]
142       (Localization.AtPrime p) ⧸ IsLocalRing.maximalIdeal (Localization.AtPrime p) :=
143     quotientAtPrimeAlgEquiv (R := R) (B := B) p
144   let residueEquiv := locRes.symm.trans e
145   let qeq := Ideal.quotientEquivAlgOfEq R (Submodule.pow_one q)
146   let jMap := Ideal.quotientEquivAlgOfEq R (Submodule.pow_one J)
147   let residueEquiv1 :=
148     jMap.trans residueEquiv |>.trans qeq.symm
149   have hjMap (b : Localization.AtPrime p) :
150       jMap (Ideal.Quotient.mkₐ R (J ^ 1) b) = Ideal.Quotient.mkₐ R J b := by
151     dsimp only [jMap]
152     change Ideal.quotientEquivAlgOfEq R (Submodule.pow_one J)
153         (Ideal.Quotient.mk (J ^ 1) b) = Ideal.Quotient.mk J b
154     exact Ideal.quotientEquivAlgOfEq_mk (R₁ := R) (h := Submodule.pow_one J) b
155   let r : Localization.AtPrime p →ₐ[R] R ⧸ q :=
156     residueEquiv.toAlgHom.comp (Ideal.Quotient.mkₐ R J)
157   let liftExists :=
158     Algebra.FormallySmooth.exists_adicCompletionEvalOneₐ_comp_eq (R := R) (I := q) r
159   let ψ := Classical.choose liftExists
160   have hψ := Classical.choose_spec liftExists
161   have hG :
162       Stafford38.Geometry.FormallyEtaleCompletionEquivalence.reverseStage q J hJ ψ 1 =
163         residueEquiv1.toAlgHom := by
164     apply Ideal.Quotient.algHom_ext R
165     apply AlgHom.ext
166     intro b
167     change Stafford38.Geometry.FormallyEtaleCompletionEquivalence.reverseStage q J hJ ψ 1
168         (Ideal.Quotient.mkₐ R (J ^ 1) b) = residueEquiv1
169           (Ideal.Quotient.mkₐ R (J ^ 1) b)
170     rw [reverseStage_apply_mk]
171     apply qeq.injective
172     calc
173       qeq (AdicCompletion.evalₐ q 1 (ψ b)) =
174           AdicCompletion.evalOneₐ q (ψ b) := by
175         change (qeq.toAlgHom) _ = _
176         rw [Ideal.quotientEquivAlgOfEq_coe_eq_factorₐ]
177         exact AdicCompletion.factorₐ_evalₐ_one q (ψ b)
178       _ = r b := DFunLike.congr_fun hψ b
179       _ = residueEquiv (Ideal.Quotient.mk J b) := by
180         rfl
181       _ = qeq (residueEquiv1 (Ideal.Quotient.mkₐ R (J ^ 1) b)) := by
182         dsimp only [residueEquiv1]
183         simp only [AlgEquiv.trans_apply, AlgEquiv.apply_symm_apply]
184         exact (congrArg residueEquiv (hjMap b)).symm
185   have hF :
186       Stafford38.Geometry.FormallyEtaleCompletionEquivalence.forwardStage q J hJ 1 =
187         residueEquiv1.symm.toAlgHom := by
188     apply Ideal.Quotient.algHom_ext R
189     apply AlgHom.ext
190     intro a
191     change Ideal.Quotient.mkₐ R (J ^ 1) (algebraMap R (Localization.AtPrime p) a) =
192       residueEquiv1.symm (Ideal.Quotient.mkₐ R (q ^ 1) a)
193     have hloc : locRes.symm
194         (Ideal.Quotient.mkₐ R J (algebraMap R (Localization.AtPrime p) a)) =
195           Ideal.Quotient.mkₐ R p (algebraMap R B a) := by
196       apply locRes.injective
197       rw [AlgEquiv.apply_symm_apply]
198       change Ideal.Quotient.mkₐ R J (algebraMap R (Localization.AtPrime p) a) =
199         quotientAtPrimeAlgEquiv (R := R) (B := B) p
200           (Ideal.Quotient.mk p (algebraMap R B a))
201       rw [quotientAtPrimeAlgEquiv_mk]
202       congr 1
203     have he : e (Ideal.Quotient.mkₐ R p (algebraMap R B a)) =
204         Ideal.Quotient.mkₐ R q a := by
205       exact e.commutes a
206     apply residueEquiv1.injective
207     rw [AlgEquiv.apply_symm_apply]
208     change qeq.symm
209       (residueEquiv (jMap
210         (Ideal.Quotient.mkₐ R (J ^ 1) (algebraMap R (Localization.AtPrime p) a)))) = _
211     rw [hjMap]
212     dsimp only [residueEquiv]
213     rw [AlgEquiv.trans_apply]
214     rw [hloc, he]
215     apply qeq.injective
216     rw [AlgEquiv.apply_symm_apply]
217     exact (Ideal.quotientEquivAlgOfEq_mk (R₁ := R)
218       (h := Submodule.pow_one q) a).symm
219   have hres :
220       (Stafford38.Geometry.FormallyEtaleCompletionEquivalence.forwardStage q J hJ 1).comp
221           (Stafford38.Geometry.FormallyEtaleCompletionEquivalence.reverseStage q J hJ ψ 1) =
222         AlgHom.id R ((Localization.AtPrime p) ⧸ (J ^ 1)) := by
223     rw [hF, hG]
224     apply AlgHom.ext
225     intro x
226     simp
227   exact Stafford38.Geometry.FormallyEtaleCompletionEquivalence.formalEtaleCompletionEquiv
228     q J hJ ψ hres
