15 /-- Independent coordinate check: the quotient equivalence built from the actual
16 residue map sends each polynomial generator to the value of that same generator
17 at the selected ground point. -/
18 theorem residueQuotient_generator_value
19     {k σ B : Type*} [Field k] [CommRing B] [Algebra k B]
20     [Algebra (MvPolynomial σ k) B]
21     [IsScalarTower k (MvPolynomial σ k) B]
22     (M : Ideal B) (eM : (B ⧸ M) ≃ₐ[k] k) [Fintype σ] (i : σ) :
23     residuePointQuotientEquiv (σ := σ) M eM
24       (Ideal.Quotient.mk
25         (Stafford38.Geometry.AffinePointCompletion.pointIdeal
26           (residueCoordinates (σ := σ) M eM))
27         (MvPolynomial.X i)) = residueCoordinates (σ := σ) M eM i := by
28   rw [residuePointQuotientEquiv_apply]
29   simp [residueEvaluation, residueCoordinates]

31 /-- The completed map sends the formal variable to the centered original
32 polynomial coordinate in the completion at the actual local point. This checks
33 that the selected residue point and the original coordinate map share one chart. -/
34 theorem prescribedCompletion_centeredGenerator
35     {k σ B : Type*} [Field k] [Fintype σ] [CommRing B] [Algebra k B]
36     [Algebra (MvPolynomial σ k) B]
37     [IsScalarTower k (MvPolynomial σ k) B]
38     (M : Ideal B) [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
39     [Algebra.EssFiniteType (MvPolynomial σ k) B]
40     [Algebra.FormallyEtale (MvPolynomial σ k) (Localization.AtPrime M)] (i : σ) :
41     Stafford38.Geometry.PrescribedAffineResidueCompletion.powerSeriesCompletionAtGroundPoint
42       M eM (MvPowerSeries.X i) =
43     AdicCompletion.of (IsLocalRing.maximalIdeal (Localization.AtPrime M))
44       (Localization.AtPrime M)
45       (algebraMap (MvPolynomial σ k) (Localization.AtPrime M)
46         (MvPolynomial.X i - MvPolynomial.C
47           (residueCoordinates (σ := σ) M eM i))) := by
48   let x := residueCoordinates (σ := σ) M eM
49   let q := Stafford38.Geometry.AffinePointCompletion.pointIdeal x
50   letI : q.IsMaximal :=
51     Stafford38.Geometry.PrescribedAffineResidueCompletion.residuePoint_isMaximal
52       (σ := σ) M eM
53   letI : q.IsPrime := inferInstance
54   let ρ := residueEvaluation (σ := σ) M eM
55   letI : Algebra (MvPolynomial σ k) k := ρ.toRingHom.toAlgebra
56   letI : IsScalarTower (MvPolynomial σ k) k k :=
57     IsScalarTower.of_algebraMap_eq (fun _ => by simp)
58   let eQ := residuePointQuotientEquiv (σ := σ) M eM
59   letI : IsScalarTower (MvPolynomial σ k) k (B ⧸ M) :=
60     IsScalarTower.of_algebraMap_eq (fun a => by
61       apply eM.injective
62       rw [← Ideal.Quotient.mk_algebraMap (MvPolynomial σ k) M a, eM.commutes]
63       change eM (Ideal.Quotient.mk M (algebraMap (MvPolynomial σ k) B a)) = ρ a
64       rfl)
65   letI : IsScalarTower (MvPolynomial σ k) k
66       (MvPolynomial σ k ⧸ q) :=
67     IsScalarTower.of_algebraMap_eq (fun a => by
68       apply eQ.injective
69       calc
70         eQ (algebraMap (MvPolynomial σ k) (MvPolynomial σ k ⧸ q) a) = ρ a := by
71           change eQ (Ideal.Quotient.mk q a) = ρ a
72           simpa [ρ] using
73             Stafford38.Geometry.PrescribedAffineResidueCompletion.residuePointQuotientEquiv_apply
74               (σ := σ) M eM a
75         _ = eQ (algebraMap k (MvPolynomial σ k ⧸ q) (ρ a)) := by
76           rw [eQ.commutes]
77           rfl)
78   have hq : Ideal.under (MvPolynomial σ k) M = q := by
79     simpa [x, q] using
80       Stafford38.Geometry.PrescribedAffineResidueCompletion.residuePoint_under_eq
81         (σ := σ) M eM
82   let e :=
83     Stafford38.Geometry.PrescribedAffineResidueCompletion.residueEquivOverCoordinates
84       (σ := σ) M eM
85   have hAffine :
86       Stafford38.Geometry.AffinePointCompletion.powerSeriesCompletionAtPoint x
87           (MvPowerSeries.X i) =
88         AdicCompletion.of q (MvPolynomial σ k)
89           (MvPolynomial.X i - MvPolynomial.C (x i)) := by
90     simpa [Stafford38.Geometry.AffinePointCompletion.powerSeriesCompletionAtPoint] using
91       Stafford38.Geometry.SmoothLocalParameterCompletion.affinePointCompletion_variable_explicit
92         x i
93   change (Stafford38.Geometry.EtalePointCompletion.completionEquivOfEtaleResidue
94     q M hq e)
95       (Stafford38.Geometry.AffinePointCompletion.powerSeriesCompletionAtPoint x
96         (MvPowerSeries.X i)) = _
97   rw [hAffine]
98   exact Stafford38.Geometry.PrescribedAffineResidueCompletion.etaleCompletion_apply_of
99     q M hq e (MvPolynomial.X i - MvPolynomial.C (x i))
