39 /-- The nonconstant component interface with the weakest certificate needed by
40 the terminal conormal-axis consumer. -/
41 def HigherDimensionalCanonicalResidueExtensionNonconstantFiniteGradientProduction :
42     Prop :=
43   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
44     (n N : ℕ) (d : PresentedWeyl k (n + 1)),
45     0 < n →
46     Disjoint
47       (orderCharacteristicSupport k
48         (canonicalRightIdeal (presentedCoordinate k n) d N))
49       (PrimeSpectrum.zeroLocus
50         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
51           Set (SymbolRing k (n + 1)))) →
52     ∀ P : PrimeSpectrum (MvPolynomial (Fin (n + 1)) k),
53       P.asIdeal ∈
54         (reducedOrderBaseIdeal k
55           (canonicalRightIdeal (presentedCoordinate k n) d N)).minimalPrimes →
56       Transcendental k
57         (componentCoordinate P ⟨0, Nat.zero_lt_succ n⟩) →
58       ∃ (K : Type u) (_ : Field K) (_ : Algebra k K),
59         Nonempty
60           (FiniteGradientBoundaryCertificateOver
61             (k := k) (K := K) (n + 1) (Nat.zero_lt_succ n)
62             (reducedOrderBaseIdeal k
63               (canonicalRightIdeal (presentedCoordinate k n) d N)))

69 theorem higherDimensionalCanonicalResidueExtensionConormalAxisProduction_of_nonconstantFiniteGradient

80     exists_minimalPrime_coordinate_constant_or_transcendental
81       I hproper ⟨0, Nat.zero_lt_succ n⟩
82   · obtain ⟨c, hc⟩ := hconstant
83     exact
84       exists_residueExtensionConormalAxis_of_minimalPrime_coordinate_constant
85         (Nat.zero_lt_succ n) I P.asIdeal
86         (reducedOrderBaseIdeal_isRadical k
87           (canonicalRightIdeal (presentedCoordinate k n) d N))
88         hP c hc
89   · obtain ⟨K, fieldK, algebraK, W⟩ :=
90       hnonconstant k n N d hn hdisjoint P hP htranscendental
91     letI : Field K := fieldK
