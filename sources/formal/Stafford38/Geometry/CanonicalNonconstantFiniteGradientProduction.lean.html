41 /-- The nonconstant component interface with the weakest certificate needed by
42 the terminal conormal-axis consumer. -/
43 def HigherDimensionalCanonicalResidueExtensionNonconstantFiniteGradientProduction :
44     Prop :=
45   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k]
46     (n N : ℕ) (d : WeylAlg k (n + 1)),
47     0 < n →
48     Disjoint
49       (orderCharacteristicSupport k
50         (canonicalRightIdeal (presentedCoordinate k n) d N))
51       (PrimeSpectrum.zeroLocus
52         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
53           Set (SymbolRing k (n + 1)))) →
54     ∀ P : PrimeSpectrum (MvPolynomial (Fin (n + 1)) k),
55       P.asIdeal ∈
56         (reducedOrderBaseIdeal k
57           (canonicalRightIdeal (presentedCoordinate k n) d N)).minimalPrimes →
58       Transcendental k
59         (componentCoordinate P ⟨0, Nat.zero_lt_succ n⟩) →
60       ∃ (K : Type u) (_ : Field K) (_ : Algebra k K),
61         Nonempty
62           (FiniteGradientBoundaryCertificateOver
63             (k := k) (K := K) (n + 1) (Nat.zero_lt_succ n)
64             (reducedOrderBaseIdeal k
65               (canonicalRightIdeal (presentedCoordinate k n) d N)))
