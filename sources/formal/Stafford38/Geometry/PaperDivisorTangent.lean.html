49 /-- The paper's order gap makes the corrected normalized transverse tangent
50 vanish in the axis row. This is derived, not assumed as an output condition. -/
51 lemma normalized_axis_constantCoeff_eq_zero
52     {ι κ : Type*} [Fintype κ]
53     (q : ι → PowerSeries k) (Z : Matrix ι κ (PowerSeries k))
54     (lambda : κ → PowerSeries k) (tau : ι → PowerSeries k)
55     (axis : ι) (b c : ℕ) (u : PowerSeries k)
56     (hbc : c + 1 < b)
57     (hq : q axis = (PowerSeries.X : PowerSeries k) ^ b * u)
58     (hZ : ∀ j, ∃ w : PowerSeries k,
59       Z axis j = (PowerSeries.X : PowerSeries k) ^ b * w)
60     (hfactor : ∀ i, PowerSeries.derivative (R := k) (q i) - Z.mulVec lambda i =
61       (PowerSeries.X : PowerSeries k) ^ c * tau i) :
62     PowerSeries.constantCoeff (tau axis) = 0 := by
63   have hd : PowerSeries.coeff c (PowerSeries.derivative (R := k) (q axis)) = 0 := by
64     rw [PowerSeries.coeff_derivative, hq, PowerSeries.coeff_X_pow_mul']
65     simp [Nat.not_le_of_gt hbc]
66   have hz : PowerSeries.coeff c (Z.mulVec lambda axis) = 0 := by
67     change PowerSeries.coeff c (∑ j, Z axis j * lambda j) = 0
68     rw [map_sum]
69     apply Finset.sum_eq_zero
70     intro j _
71     obtain ⟨w, hw⟩ := hZ j
72     rw [hw, mul_assoc, PowerSeries.coeff_X_pow_mul']
73     simp [Nat.not_le_of_gt (by omega : c < b)]
74   have he := congrArg (PowerSeries.coeff c) (hfactor axis)
75   rw [map_sub, hd, hz, sub_zero] at he
76   simpa [PowerSeries.coeff_X_pow_mul', PowerSeries.coeff_zero_eq_constantCoeff_apply]
77     using he.symm

126 /-- A left inverse makes the actual column lattice a direct summand.  The
127 projector is the composite B C, whose idempotence follows from C B = 1. -/
128 lemma normalizedTangentLattice_isComplemented
129     {n : ℕ} {κ : Type*} [Fintype κ] [DecidableEq κ]
130     (q : Fin (n + 1) → PowerSeries k)
131     (Z : Matrix (Fin (n + 1)) κ (PowerSeries k))
132     (tau : Fin (n + 1) → PowerSeries k)
133     (C : Matrix (FormalTangentColumn κ) (Fin (n + 1)) (PowerSeries k))
134     (hCB : C * formalTangentMatrix q Z tau = 1) :
135     IsComplemented (normalizedTangentLattice q Z tau) := by
136   let B := formalTangentMatrix q Z tau
137   let E : (Fin (n + 1) → PowerSeries k) →ₗ[PowerSeries k]
138       (Fin (n + 1) → PowerSeries k) := B.mulVecLin.comp C.mulVecLin
139   have hCBlin : C.mulVecLin.comp B.mulVecLin = LinearMap.id := by
140     rw [← Matrix.mulVecLin_mul, hCB, Matrix.mulVecLin_one]
141   have hEidem : IsIdempotentElem E := by
142     change E.comp E = E
143     dsimp [E]
144     rw [LinearMap.comp_assoc, ← LinearMap.comp_assoc C.mulVecLin B.mulVecLin]
145     rw [hCBlin]
146     simp
147   have hproj : LinearMap.IsProj (LinearMap.range E) E :=
148     (LinearMap.isProj_range_iff_isIdempotentElem E).mpr hEidem
149   have hrange : LinearMap.range E = normalizedTangentLattice q Z tau := by
150     apply le_antisymm
151     · rintro x ⟨v, rfl⟩
152       exact ⟨C.mulVecLin v, rfl⟩
153     · rintro x ⟨v, rfl⟩
154       refine ⟨B.mulVecLin v, ?_⟩
155       change B.mulVecLin (C.mulVecLin (B.mulVecLin v)) =
156         B.mulVecLin v
157       have hfix : C.mulVecLin (B.mulVecLin v) = v := by
158         have h := congrArg (fun f : (FormalTangentColumn κ → PowerSeries k) →ₗ[
159             PowerSeries k] (FormalTangentColumn κ → PowerSeries k) => f v) hCBlin
160         simpa only [LinearMap.comp_apply, LinearMap.id_apply] using h
161       rw [hfix]
162   exact ⟨LinearMap.ker E, by simpa [hrange] using hproj.isCompl⟩

217 /-- From containment of the projective tangent cone in a column span, the
218 point belongs to that span and the affine Zariski tangent space is contained
219 in the dehomogenized column span. The reverse span containment is unnecessary
220 for the conormal argument. -/
221 theorem matrix_tangent_dictionary_of_projectiveCone_le_span
222     {K : Type*} [Field K] {n : ℕ} {κ : Type*} [Fintype κ]
223     (q : Fin (n + 1) → K) (hq0 : q 0 ≠ 0)
224     (T : Submodule K (Fin n → K))
225     (B : Matrix (Fin (n + 1)) κ K)
226     (hspan : projectiveTangentCone q T ≤
227       Submodule.span K (Set.range fun j => fun i => B i j)) :
228     (∃ c : κ → K, ∀ i, q i = ∑ j, B i j * c j) ∧
229       T ≤ dehomogenizedTangentSpan q B := by
230   let d := dehomogenizedTangentLinearMap q
231   let H := Submodule.span K (Set.range fun j => fun i => B i j)
232   have hqcone : q ∈ projectiveTangentCone q T := by
233     change dehomogenizedTangentColumn q q ∈ T
234     convert T.zero_mem using 1
235     ext i
236     simp [dehomogenizedTangentColumn]
237   have hqspan : q ∈ H := hspan hqcone
238   obtain ⟨c, hc⟩ := (Submodule.mem_span_range_iff_exists_fun K).mp hqspan
239   have hposition : ∃ c : κ → K, ∀ i, q i = ∑ j, B i j * c j := by
240     refine ⟨c, fun i => ?_⟩
241     have hi := congrFun hc.symm i
242     simpa [Finset.sum_apply, smul_eq_mul, mul_comm] using hi
243   have hmap : H.map d = dehomogenizedTangentSpan q B := by
244     rw [dehomogenizedTangentSpan, Submodule.map_span]
245     congr 1
246     ext v
247     simp only [Set.mem_image, Set.mem_range]
248     constructor
249     · rintro ⟨w, ⟨j, rfl⟩, rfl⟩
250       exact ⟨j, rfl⟩
251     · rintro ⟨j, rfl⟩
252       exact ⟨fun i => B i j, ⟨j, rfl⟩, rfl⟩
253   have hdsurj : Function.Surjective d := by
254     intro v
255     refine ⟨Fin.cases 0 (fun i => v i * q 0), ?_⟩
256     ext i
257     change (v i * q 0 * q 0 - q i.succ * 0) / q 0 ^ 2 = v i
258     field_simp
259     ring
260   have htangent : T ≤ dehomogenizedTangentSpan q B := by
261     intro v hv
262     obtain ⟨w, hw⟩ := hdsurj v
263     have hwcone : w ∈ projectiveTangentCone q T := by
264       change d w ∈ T
265       rw [hw]
266       exact hv
267     have hwH : w ∈ H := hspan hwcone
268     have hdH : d w ∈ H.map d := ⟨w, hwH, rfl⟩
269     rw [hmap] at hdH
270     simpa [hw] using hdH
271   exact ⟨hposition, htangent⟩

273 /-- If the point is in a column span and its affine tangent space is in the
274 dehomogenized column span, then the full projective tangent cone is in the
275 original column span. The only extra direction in the kernel of chart
276 dehomogenization is the radial point direction. -/
277 theorem projectiveTangentCone_le_span_of_position_and_tangent
278     {K : Type*} [Field K] {n : ℕ} {κ : Type*}
279     (q : Fin (n + 1) → K) (hq0 : q 0 ≠ 0)
280     (T : Submodule K (Fin n → K))
281     (B : Matrix (Fin (n + 1)) κ K)
282     (hqspan : q ∈ Submodule.span K (Set.range fun j => fun i => B i j))
283     (htangent : T ≤ dehomogenizedTangentSpan q B) :
284     projectiveTangentCone q T ≤
285       Submodule.span K (Set.range fun j => fun i => B i j) := by
286   let d := dehomogenizedTangentLinearMap q
287   let H := Submodule.span K (Set.range fun j => fun i => B i j)
288   have hmap : H.map d = dehomogenizedTangentSpan q B := by
289     rw [dehomogenizedTangentSpan, Submodule.map_span]
290     congr 1
291     ext v
292     simp only [Set.mem_image, Set.mem_range]
293     constructor
294     · rintro ⟨w, ⟨j, rfl⟩, rfl⟩
295       exact ⟨j, rfl⟩
296     · rintro ⟨j, rfl⟩
297       exact ⟨fun i => B i j, ⟨j, rfl⟩, rfl⟩
298   intro x hx
299   have hdx : d x ∈ T := hx
300   have hdxspan : d x ∈ H.map d := by
301     rw [hmap]
302     exact htangent hdx
303   rcases hdxspan with ⟨y, hy, hdy⟩
304   let c : K := (x 0 - y 0) / q 0
305   have hkernel : x - y = c • q := by
306     ext i
307     refine Fin.cases ?_ (fun j => ?_) i
308     · dsimp [c]
309       field_simp [hq0]
310     · have hj := congrFun hdy.symm j
311       change dehomogenizedTangentColumn q x j =
312         dehomogenizedTangentColumn q y j at hj
313       simp [dehomogenizedTangentColumn] at hj
314       field_simp [hq0] at hj
315       dsimp [c]
316       field_simp [hq0]
317       linear_combination hj
318   have hxdecomp : x = y + c • q := by
319     have := congrArg (fun v : Fin (n + 1) → K => v + y) hkernel
320     simpa [add_comm] using this
321   have hxinH : x ∈ H := by
322     rw [hxdecomp]
323     exact H.add_mem hy (H.smul_mem c hqspan)
324   exact hxinH

326 /-- The radial point direction and an exact affine tangent span determine the
327 full projective tangent cone. The inclusion uses the existing radial-kernel
328 comparison; the reverse inclusion follows from the column tangent equations. -/
329 theorem projectiveTangentCone_eq_span_of_position_and_tangent
330     {K : Type*} [Field K] {n : ℕ} {κ : Type*}
331     (q : Fin (n + 1) → K) (hq0 : q 0 ≠ 0)
332     (T : Submodule K (Fin n → K))
333     (B : Matrix (Fin (n + 1)) κ K)
334     (hqspan : q ∈ Submodule.span K (Set.range fun j => fun i => B i j))
335     (htangent : T = dehomogenizedTangentSpan q B) :
336     projectiveTangentCone q T =
337       Submodule.span K (Set.range fun j => fun i => B i j) := by
338   apply le_antisymm
339   · exact projectiveTangentCone_le_span_of_position_and_tangent
340       q hq0 T B hqspan htangent.le
341   · apply Submodule.span_le.mpr
342     rintro _ ⟨j, rfl⟩
343     change dehomogenizedTangentColumn q (fun i => B i j) ∈ T
344     rw [htangent]
345     exact Submodule.subset_span ⟨j, rfl⟩
