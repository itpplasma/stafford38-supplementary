14 /-- A finite relative base cancels from the same-witness total-degree bound.
15 If the residue field is algebraic over that base, the component field has
16 relative transcendence degree at most one; one transcendental parameter makes
17 the bound exact. -/
18 theorem eq_one_of_same_total_bound
19     {k E F κ : Type u}
20     [Field k] [Field E] [Field F] [Field κ]
21     [Algebra k E] [Algebra E F] [Algebra k F] [IsScalarTower k E F]
22     [Algebra E κ] [Algebra k κ] [IsScalarTower k E κ]
23     (hEfinite : Algebra.trdeg k E < Cardinal.aleph0)
24     (hκalg : Algebra.IsAlgebraic E κ)
25     (hbound : Algebra.trdeg k F ≤ Algebra.trdeg k κ + 1)
26     (htrans : Algebra.Transcendental E F) :
27     Algebra.trdeg E F = 1 := by
28   have hκ : Algebra.trdeg k κ = Algebra.trdeg k E := by
29     have hsum : Algebra.trdeg k E + Algebra.trdeg E κ = Algebra.trdeg k κ :=
30       trdeg_add_eq k E (A := κ)
31     rw [trdeg_eq_zero_iff.mpr hκalg, add_zero] at hsum
32     exact hsum.symm
33   have htotal : Algebra.trdeg k E + Algebra.trdeg E F ≤
34       Algebra.trdeg k E + 1 := by
35     calc
36       Algebra.trdeg k E + Algebra.trdeg E F = Algebra.trdeg k F :=
37         trdeg_add_eq k E (A := F)
38       _ ≤ Algebra.trdeg k κ + 1 := hbound
39       _ = Algebra.trdeg k E + 1 := by rw [hκ]
40   have hrel : Algebra.trdeg E F ≤ 1 := by
41     apply (Cardinal.add_le_add_iff_of_lt_aleph0 hEfinite).mp
42     simpa only [add_comm (Algebra.trdeg E F) (Algebra.trdeg k E),
43       add_comm 1 (Algebra.trdeg k E)] using htotal
44   have hpos : 1 ≤ Algebra.trdeg E F :=
45     Cardinal.one_le_iff_ne_zero.mpr (trdeg_ne_zero_iff.mpr htrans)
46   exact le_antisymm hrel hpos

48 /-- The finite-base hypothesis in `eq_one_of_same_total_bound` follows
49 directly from an explicitly supplied finite transcendence basis. -/
50 theorem eq_one_of_same_total_bound_of_finite_basis
51     {k E F κ ι : Type u}
52     [Field k] [Field E] [Field F] [Field κ] [Fintype ι]
53     [Algebra k E] [Algebra E F] [Algebra k F] [IsScalarTower k E F]
54     [Algebra E κ] [Algebra k κ] [IsScalarTower k E κ]
55     (basis : ι → E) (hbasis : IsTranscendenceBasis k basis)
56     (hκalg : Algebra.IsAlgebraic E κ)
57     (hbound : Algebra.trdeg k F ≤ Algebra.trdeg k κ + 1)
58     (htrans : Algebra.Transcendental E F) :
59     Algebra.trdeg E F = 1 := by
60   have hEfinite : Algebra.trdeg k E < Cardinal.aleph0 := by
61     rw [← hbasis.cardinalMk_eq_trdeg]
62     rw [Cardinal.mk_fintype]
63     exact Cardinal.natCast_lt_aleph0
64   exact eq_one_of_same_total_bound hEfinite hκalg hbound htrans

66 /-- A nonzero parameter in a local domain proves the required lower bound
67 after embedding that domain into the component field. -/
68 theorem eq_one_of_same_total_bound_of_parameter
69     {k E V F κ : Type u}
70     [Field k] [Field E] [CommRing V] [IsDomain V] [IsLocalRing V]
71     [Field F] [Field κ]
72     [Algebra k E] [Algebra E V] [Algebra E F] [Algebra V F]
73     [IsScalarTower E V F] [Algebra k F] [IsScalarTower k E F]
74     [Algebra E κ] [Algebra k κ] [IsScalarTower k E κ]
75     (hVF : Function.Injective (algebraMap V F))
76     (x : V) (hx0 : x ≠ 0) (hxM : x ∈ IsLocalRing.maximalIdeal V)
77     (hEfinite : Algebra.trdeg k E < Cardinal.aleph0)
78     (hκalg : Algebra.IsAlgebraic E κ)
79     (hbound : Algebra.trdeg k F ≤ Algebra.trdeg k κ + 1) :
80     Algebra.trdeg E F = 1 := by
81   let f : V →ₐ[E] F := IsScalarTower.toAlgHom E V F
82   have hnot : ¬ IsAlgebraic E (f x) := by
83     intro hfx
84     have hxalg : IsAlgebraic E x :=
85       (isAlgebraic_algHom_iff f hVF).mp (by simpa [f] using hfx)
86     exact Stafford38.Geometry.DVRParameterSmoothness.not_isAlgebraic_of_nonzero_mem_maximalIdeal
87       hx0 hxM hxalg
88   have htransx : Transcendental E (f x) := hnot
89   have htrans : Algebra.Transcendental E F := ⟨f x, htransx⟩
90   exact eq_one_of_same_total_bound hEfinite hκalg hbound htrans

101     (basis : ι → E) (hbasis : IsTranscendenceBasis k basis)
102     (hVF : Function.Injective (algebraMap V F))
103     (x : V) (hx0 : x ≠ 0) (hxM : x ∈ IsLocalRing.maximalIdeal V)
104     (hκalg : Algebra.IsAlgebraic E κ)
