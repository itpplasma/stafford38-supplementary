21 /-- Chain rule for a derivation out of a polynomial algebra into an arbitrary
22 field-valued point. -/
23 theorem derivation_eq_sum_pderiv
24     {S : Type*} [Field S] [Algebra k S]
25     {n : ℕ} [Algebra (MvPolynomial (Fin n) k) S]
26     [IsScalarTower k (MvPolynomial (Fin n) k) S]
27     (D : Derivation k (MvPolynomial (Fin n) k) S)
28     (q : Fin n → S)
29     (halg : ∀ p, algebraMap (MvPolynomial (Fin n) k) S p = aeval q p)
30     (f : MvPolynomial (Fin n) k) :
31     D f = ∑ i, aeval q (pderiv i f) * D (X i) := by
32   have hD : D = MvPolynomial.mkDerivation k (fun i => D (X i)) := by
33     apply MvPolynomial.derivation_ext
34     intro i
35     simp [MvPolynomial.mkDerivation_X]
36   calc
37     D f = MvPolynomial.mkDerivation k (fun i => D (X i)) f :=
38       congrArg (fun E : Derivation k (MvPolynomial (Fin n) k) S => E f) hD
39     _ = ∑ i, aeval q (pderiv i f) * D (X i) :=
40       Stafford38.Geometry.GenericPointKaehlerConormal.mkDerivation_eq_sum_pderiv
41         q (fun i => D (X i)) halg f

43 /-- If an ambient polynomial derivation annihilates an ideal at a point, its
44 actual coordinate-value vector lies in the equation-defined Zariski tangent
45 space of that ideal after scalar extension. -/
46 theorem derivationVector_mem_zariskiTangentSpace_of_vanishes
47     {S : Type*} [Field S] [Algebra k S]
48     {n : ℕ} [Algebra (MvPolynomial (Fin n) k) S]
49     [IsScalarTower k (MvPolynomial (Fin n) k) S]
50     (D : Derivation k (MvPolynomial (Fin n) k) S)
51     (I : Ideal (MvPolynomial (Fin n) k))
52     (q : Fin n → S)
53     (halg : ∀ p, algebraMap (MvPolynomial (Fin n) k) S p = aeval q p)
54     (hD : ∀ f ∈ I, D f = 0)
55     (hq : ∀ f ∈ I, aeval q f = 0) :
56     (fun i ↦ D (X i)) ∈ zariskiTangentSpace q
57       (I.map (MvPolynomial.map (algebraMap k S))) := by
58   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator]
59   intro φ hφ
60   induction hφ using Submodule.span_induction with
61   | mem f hf =>
62       rcases hf with ⟨g, rfl⟩
63       have hmap : ∀ h : MvPolynomial (Fin n) S,
64           h ∈ I.map (MvPolynomial.map (algebraMap k S)) →
65             MvPolynomial.eval q h = 0 ∧
66               differentialCovector q h (fun i ↦ D (X i)) = 0 := by
67         intro h hh
68         rw [Ideal.map] at hh
69         induction hh using Submodule.span_induction with
70         | mem h hh =>
71             rcases hh with ⟨g, hg, rfl⟩
72             constructor
73             · rw [MvPolynomial.eval_map]
74               exact hq g hg
75             · have hderiv := derivation_eq_sum_pderiv D q halg g
76               rw [hD g hg] at hderiv
77               simpa [differentialCovector, differentialAt,
78                 MvPolynomial.pderiv_map, MvPolynomial.eval_map, aeval_def] using hderiv.symm
79         | zero =>
80             simp [differentialCovector, differentialAt]
81         | add h₁ h₂ hh₁ hh₂ ih₁ ih₂ =>
82             constructor
83             · rw [MvPolynomial.eval_add, ih₁.1, ih₂.1, add_zero]
84             · have hi := congrArg₂ (· + ·) ih₁.2 ih₂.2
85               change ∑ i, MvPolynomial.eval q
86                   (MvPolynomial.pderiv i (h₁ + h₂)) * D (X i) = 0
87               simp only [map_add, MvPolynomial.eval_add, add_mul]
88               rw [Finset.sum_add_distrib]
89               simpa [differentialCovector, differentialAt] using hi
90         | smul a h hh ih =>
91             constructor
92             · simpa [smul_eq_mul, ih.1]
93             · have ihD : ∑ i, MvPolynomial.eval q
94                   (MvPolynomial.pderiv i h) * D (X i) = 0 := by
95                 simpa [differentialCovector, differentialAt] using ih.2
96               have hprod : differentialCovector q (a * h) (fun i ↦ D (X i)) =
97                   MvPolynomial.eval q h *
98                       differentialCovector q a (fun i ↦ D (X i)) +
99                     MvPolynomial.eval q a *
100                       differentialCovector q h (fun i ↦ D (X i)) := by
101                 have h := congrArg (fun ψ : Module.Dual S (Fin n → S) =>
102                     ψ (fun i ↦ D (X i))) (differentialCovector_mul q a h)
103                 simpa [smul_eq_mul] using h
104               rw [show (a • h : MvPolynomial (Fin n) S) = a * h by rfl]
105               rw [hprod]
106               simp [ih.1, ih.2]
107       exact (hmap g.1 g.2).2
108   | zero => simp
109   | add φ ψ hφ hψ ihφ ihψ => simpa [ihφ, ihψ]
110   | smul a φ hφ ihφ => simpa [ihφ]

112 /-- The converse coordinate calculation: an ambient tangent vector defines the
113 canonical polynomial derivation, and that derivation vanishes on every equation
114 of the ideal. -/
115 theorem mkDerivation_vanishes_of_mem_zariskiTangentSpace
116     {S : Type*} [Field S] [Algebra k S]
117     {n : ℕ} [Algebra (MvPolynomial (Fin n) k) S]
118     [IsScalarTower k (MvPolynomial (Fin n) k) S]
119     (I : Ideal (MvPolynomial (Fin n) k))
120     (q v : Fin n → S)
121     (halg : ∀ p, algebraMap (MvPolynomial (Fin n) k) S p = aeval q p)
122     (hv : v ∈ zariskiTangentSpace q
123       (I.map (MvPolynomial.map (algebraMap k S)))) :
124     ∀ f ∈ I, MvPolynomial.mkDerivation k v f = 0 := by
125   intro f hf
126   rw [zariskiTangentSpace, Submodule.mem_dualCoannihilator] at hv
127   have hfmap : MvPolynomial.map (algebraMap k S) f ∈
128       I.map (MvPolynomial.map (algebraMap k S)) := Ideal.mem_map_of_mem _ hf
129   have hcov := hv (differentialCovector q (MvPolynomial.map (algebraMap k S) f))
130     (Submodule.subset_span ⟨⟨MvPolynomial.map (algebraMap k S) f, hfmap⟩, rfl⟩)
131   have hformula := Stafford38.Geometry.GenericPointKaehlerConormal.mkDerivation_eq_sum_pderiv
132     q v halg f
133   have hsum : (∑ i, aeval q (pderiv i f) * v i) =
134       differentialCovector q (MvPolynomial.map (algebraMap k S) f) v := by
135     simp [differentialCovector, differentialAt, MvPolynomial.pderiv_map,
136       MvPolynomial.eval_map, aeval_def]
137   calc
138     MvPolynomial.mkDerivation k v f =
139         differentialCovector q (MvPolynomial.map (algebraMap k S) f) v := by
140       rw [hformula, hsum]
141     _ = 0 := hcov

143 /-- Precomposing a genuine derivation on a target algebra along a polynomial
144 chart map produces an actual vector in the ambient equation tangent kernel,
145 provided the equations vanish in that target algebra. -/
146 theorem chartDerivation_mem_zariskiTangentSpace
147     {n : ℕ} {B S : Type*} [CommRing B] [Field S] [Algebra k B] [Algebra k S]
148     [Algebra (MvPolynomial (Fin n) k) B]
149     [IsScalarTower k (MvPolynomial (Fin n) k) B]
150     [Algebra B S] [Algebra (MvPolynomial (Fin n) k) S]
151     [IsScalarTower (MvPolynomial (Fin n) k) B S]
152     [IsScalarTower k B S]
153     [IsScalarTower k (MvPolynomial (Fin n) k) S]
154     (D : Derivation k B S)
155     (I : Ideal (MvPolynomial (Fin n) k))
156     (q : Fin n → S)
157     (halg : ∀ p, algebraMap (MvPolynomial (Fin n) k) S p = aeval q p)
158     (hI : ∀ f ∈ I, algebraMap (MvPolynomial (Fin n) k) B f = 0) :
159     (fun i ↦ D.compAlgebraMap (MvPolynomial (Fin n) k) (X i)) ∈
160       zariskiTangentSpace q (I.map (MvPolynomial.map (algebraMap k S))) := by
161   apply derivationVector_mem_zariskiTangentSpace_of_vanishes
162     (D := D.compAlgebraMap (MvPolynomial (Fin n) k)) I q halg
163   · intro f hf
164     change D (algebraMap (MvPolynomial (Fin n) k) B f) = 0
165     rw [hI f hf, map_zero]
166   · intro f hf
167     rw [← halg f, IsScalarTower.algebraMap_eq (MvPolynomial (Fin n) k) B S]
168     simp [hI f hf]
