9 # A height bound for an integral extension of a one-variable polynomial ring
10 
11 An integral map from `E[X]` into a domain forces every maximal ideal of the
12 target to have height at most one.  Injectivity is not needed: if the kernel
13 is nonzero, the image has dimension zero, while the integral-spectrum argument
14 still gives the same bound.  This packages the local dimension step used after
15 a one-variable Noether normalization.
16 -/

24 /-- If a domain `A` is integral over the image of a one-variable polynomial ring
25 over a field, then every maximal ideal of `A` has height at most one. -/
26 theorem height_le_one_of_integral_polynomial_map
27     {E A : Type*} [Field E] [CommRing A] [IsDomain A] [Algebra E A]
28     (f : Polynomial E →ₐ[E] A) (hint : f.toRingHom.IsIntegral)
29     (p : PrimeSpectrum A) [p.asIdeal.IsMaximal] : p.asIdeal.height ≤ 1 := by
30   letI : Algebra (Polynomial E) A := f.toRingHom.toAlgebra
31   letI : Algebra.IsIntegral (Polynomial E) A := ⟨hint⟩
32   let g : PrimeSpectrum A → PrimeSpectrum (Polynomial E) :=
33     PrimeSpectrum.comap f.toRingHom
34   have hg : StrictMono g := by
35     intro x y hxy
36     change x.asIdeal < y.asIdeal at hxy
37     change (x.asIdeal.comap f.toRingHom) < (y.asIdeal.comap f.toRingHom)
38     exact Ideal.IsIntegral.comap_lt_comap hxy
39   have hgmax : (g p).asIdeal.IsMaximal := by
40     change Ideal.IsMaximal ((p.asIdeal).comap f.toRingHom)
41     exact Ideal.isMaximal_comap_of_isIntegral_of_isMaximal' f.toRingHom hint _
42   letI : (g p).asIdeal.IsMaximal := hgmax
43   have hpoly : (g p).asIdeal.height = 1 :=
44     IsPrincipalIdealRing.height_eq_one_of_isMaximal _ (Polynomial.not_isField E)
45   rw [p.height_eq_orderHeight]
46   exact (Order.height_le_height_apply_of_strictMono g hg p).trans_eq
47     (by rw [← (g p).height_eq_orderHeight, hpoly])
48 
49 end

51 end Stafford38.Geometry.IntegralPolynomialExtensionDimension
