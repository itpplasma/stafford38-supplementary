33 theorem coordinate_axis_mem_smooth_fibre_closure
34     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
35     {m : ℕ} (hm : 0 < m)

37     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) :
38     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
39       MvPolynomial.zeroLocus k
40         (MvPolynomial.vanishingIdeal k (smoothConormalFibreProjection I.asIdeal)) := by
41   obtain ⟨K, hK, hAlg, y, xi, hgeneric, hres⟩ :=
42     exists_groundConormalAxis_of_prime_coordinate_avoidance hm I havoid
43   letI := hK
44   letI := hAlg
45   intro P hP
46   have hfull := fibreLift_mem_vanishingIdeal_equationConormal I.asIdeal I.isPrime P hP
47   have hvan : ∀ q ∈ groundEquationConormalLocus (k := k) (K := K) I.asIdeal,
48       MvPolynomial.eval₂ (groundLaurentMap (k := k) (K := K)) q (fibreLift P) = 0 := by
49     letI : Algebra k (LaurentSeries K) :=
50       (groundLaurentMap (k := k) (K := K)).toAlgebra
51     intro q hq
52     exact scalarExtension_vanishing I.asIdeal (fibreLift P) hfull q hq
53   have hpoly := residueFibreLift_mem_extensionValuedVanishingIdeal_of_ground_vanishing
54     I.asIdeal P hvan
55   have hclosure := residue_mem_residueExtensionFibreClosure_of_laurent_generic
56     (K := K) (groundEquationConormalLocus (k := k) (K := K) I.asIdeal) y xi hgeneric
57   have hzero := hclosure _ hpoly
58   have hzero' : MvPolynomial.eval₂ (algebraMap k K) (residueColumn xi) P = 0 := by
59     simpa [scalarPolynomialMap] using hzero
60   rw [hres] at hzero'
61   apply (FaithfulSMul.algebraMap_injective k K)
62   rw [map_zero]
63   change (algebraMap k K) (MvPolynomial.eval
64     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) P) = 0
65   rw [MvPolynomial.eval₂_comp]
