27 /-- The affine generic-chart coordinate algebra lies in the actual valuation
28 ring of the same retained visible-frame witness. -/
29 theorem chartGenericPointSubalgebra_le_valuationSubring
30     {k : Type u} [Field k] [CharZero k]
31     {m : ℕ} (hm : 0 < m)
32     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
33     (w : GeneralDivisorialVisibleFrameWitness hm P) :
34     letI : Algebra (CoordinateZeroLocalRing w.column.W.coefficientField)
35       (ComponentFractionField P) := w.column.W.ambientAlgebra
36     ∀ z : ComponentFractionField P, z ∈ Algebra.adjoin k (Set.range
37       (chartGenericPoint P w.column.chart
38         (chartAffineCoordinateEquiv w.column.chart))) →
39         z ∈ w.column.W.place.valuation.toSubring := by
40   classical
41   let C := w.column
42   let W := C.W
43   let F := ComponentFractionField P
44   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
45   letI : IsDiscreteValuationRing W.place.valuation.toSubring := W.place.isDiscrete
46   letI : IsLocalRing W.place.valuation.toSubring := W.place.isDiscrete.toIsLocalRing
47   letI : Algebra W.coefficientField W.place.valuation.toSubring :=
48     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
49   letI : Algebra W.place.valuation.toSubring F :=
50     W.place.valuation.toSubring.subtype.toAlgebra
51   letI := C.groundCoeff.toAlgebra
52   letI := C.groundTower
53   let e := chartAffineCoordinateEquiv C.chart
54   let A := Algebra.adjoin k (Set.range (chartGenericPoint P C.chart e))
55   have hcoeffmem : ∀ a : k, algebraMap k F a ∈ W.place.valuation.toSubring := by
56     intro a
57     rw [← DFunLike.congr_fun C.groundCoeff_commutes a]
58     exact (C.groundCoeff a).property
59   let Vsub : Subalgebra k F :=
60     {W.place.valuation.toSubring with algebraMap_mem' := hcoeffmem}
61   have hgen : A ≤ Vsub := by
62     apply Algebra.adjoin_le
63     rintro z ⟨i, rfl⟩
64     change chartGenericPoint P C.chart e i ∈ Vsub
65     rw [chartGenericPoint_eq_normalized_lift (P := P)
66       (V := W.place.valuation.toSubring) C.chart e C.q C.scale
67       C.chart_one C.q_commonScale]
68     exact (C.q (e i).1).property
69   intro z hz
70   exact hgen hz

72 /-- Every coordinate of the retained normalized projective lift lies in the
73 actual affine generic-chart subalgebra of the same component function field.
74 This is the reusable membership fact behind the projective-column binding. -/
75 theorem normalizedCoordinate_mem_chartGenericPointSubalgebra
76     {k : Type u} [Field k] [CharZero k]
77     {m : ℕ} (hm : 0 < m)
78     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
79     (w : GeneralDivisorialVisibleFrameWitness hm P) :
80     let C := w.column
81     let W := C.W
82     let F := ComponentFractionField P
83     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
84     let V := W.place.valuation.toSubring
85     letI : Algebra V F := V.subtype.toAlgebra
86     ∀ a : Fin (m + 1),
87       ((C.q a : V) : F) ∈ Algebra.adjoin k (Set.range
88         (chartGenericPoint P C.chart (chartAffineCoordinateEquiv C.chart))) := by
89   classical
90   dsimp only
91   let C := w.column
92   let W := C.W
93   let F := ComponentFractionField P
94   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
95   let V := W.place.valuation.toSubring
96   letI : IsDiscreteValuationRing V := W.place.isDiscrete
97   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
98   letI : Algebra W.coefficientField V :=
99     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
100   letI : Algebra V F := V.subtype.toAlgebra
101   letI := C.groundCoeff.toAlgebra
102   letI := C.groundTower
103   let e := chartAffineCoordinateEquiv C.chart
104   let Q := Algebra.adjoin k (Set.range (chartGenericPoint P C.chart e))
105   let qF : Fin (m + 1) → F := fun a ↦ ((C.q a : V) : F)
106   have hval (v : V) : algebraMap V F v = (v : F) := rfl
107   have hqF_chart : qF C.chart = 1 := by
108     change ((C.q C.chart : V) : F) = 1
109     exact congrArg (fun v : V => (v : F)) C.chart_one
110   have hgeneric := chartGenericPoint_eq_normalized_lift
111     (P := P) (V := V) C.chart e C.q C.scale C.chart_one C.q_commonScale
112   intro a
113   change qF a ∈ Q
114   by_cases ha : a = C.chart
115   · subst a
116     rw [hqF_chart]
117     exact one_mem Q
118   · let i : Fin m := e.symm ⟨a, ha⟩
119     have hi := congrFun hgeneric i
120     have hvalue : chartGenericPoint P C.chart e i = qF a := by
121       calc
122         chartGenericPoint P C.chart e i = algebraMap V F (C.q (e i).1) := hi
123         _ = qF a := by
124           rw [hval]
125           simp [qF, i]
126     rw [← hvalue]
127     exact Algebra.subset_adjoin (R := k) (A := F) (Set.mem_range_self i)

129 /-- The retained q-column, reinterpreted coordinatewise in its actual generic
130 chart algebra, has the same values in the component field and preserves the
131 chosen chart normalization.  The affine rows agree with the canonical
132 `chartGenericPoint` rows, so later selected-row statements use the very same
133 residue-basis indices. -/
134 theorem exists_actual_normalizedProjectiveColumn
135     {k : Type u} [Field k] [CharZero k]
136     {m : ℕ} (hm : 0 < m)
137     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
138     (w : GeneralDivisorialVisibleFrameWitness hm P) :
139     let C := w.column
140     let W := C.W
141     let F := ComponentFractionField P
142     letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
143     let V := W.place.valuation.toSubring
144     letI : Algebra V F := V.subtype.toAlgebra
145     let e := chartAffineCoordinateEquiv C.chart
146     let Q := Algebra.adjoin k (Set.range (chartGenericPoint P C.chart e))
147     ∃ qQ : Fin (m + 1) → Q,
148       qQ C.chart = 1 ∧
149       (∀ a, ((qQ a : Q) : F) = ((C.q a : V) : F)) ∧
150       (∀ i : Fin m, ((qQ (e i).1 : Q) : F) =
151         chartGenericPoint P C.chart e i) := by
152   classical
153   dsimp only
154   let C := w.column
155   let W := C.W
156   let F := ComponentFractionField P
157   letI : Algebra (CoordinateZeroLocalRing W.coefficientField) F := W.ambientAlgebra
158   let V := W.place.valuation.toSubring
159   letI : IsDiscreteValuationRing V := W.place.isDiscrete
160   letI : IsLocalRing V := W.place.isDiscrete.toIsLocalRing
161   letI : Algebra W.coefficientField V :=
162     (relativeCoefficientMap W.coefficientField W.place).toAlgebra
163   letI : Algebra V F := V.subtype.toAlgebra
164   letI := C.groundCoeff.toAlgebra
165   letI := C.groundTower
166   let e := chartAffineCoordinateEquiv C.chart
167   let Q := Algebra.adjoin k (Set.range (chartGenericPoint P C.chart e))
168   let qF : Fin (m + 1) → F := fun a ↦ ((C.q a : V) : F)
169   have hval (v : V) : algebraMap V F v = (v : F) := rfl
170   have hqmem := normalizedCoordinate_mem_chartGenericPointSubalgebra hm P w
171   let qQ : Fin (m + 1) → Q := fun a ↦ ⟨qF a, hqmem a⟩
172   have hgeneric := chartGenericPoint_eq_normalized_lift
173     (P := P) (V := V) C.chart e C.q C.scale C.chart_one C.q_commonScale
174   have hchart : qQ C.chart = 1 := by
175     apply Subtype.ext
176     change qF C.chart = 1
177     exact congrArg (fun v : V => (v : F)) C.chart_one
178   have hcoe : ∀ a, ((qQ a : Q) : F) = ((C.q a : V) : F) := by
179     intro a
180     rfl
181   have hrows : ∀ i : Fin m, ((qQ (e i).1 : Q) : F) =
182       chartGenericPoint P C.chart e i := by
183     intro i
184     have hi := congrFun hgeneric i
185     calc
186       ((qQ (e i).1 : Q) : F) = qF (e i).1 := rfl
187       _ = algebraMap V F (C.q (e i).1) := (hval _).symm
188       _ = chartGenericPoint P C.chart e i := hi.symm
189   exact ⟨qQ, hchart, hcoe, hrows⟩
