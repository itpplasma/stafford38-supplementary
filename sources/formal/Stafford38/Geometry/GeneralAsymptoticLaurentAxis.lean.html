10 # Laurent conormal axes for coordinate-avoiding prime varieties
11 
12 This combines the constant-coordinate and divisorial cases for an arbitrary
13 prime affine ideal. The conclusion is a residue-extension Laurent witness;
14 its comparison with ground-field projective directions is separate.

31 theorem exists_groundConormalAxis_of_prime_coordinate_avoidance
32     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
33     {m : ℕ} (hm : 0 < m)
34     (P : PrimeSpectrum (MvPolynomial (Fin m) k))
35     (havoid : ∀ y ∈ MvPolynomial.zeroLocus k P.asIdeal, y ⟨0, hm⟩ ≠ 0) :
36     ∃ (K : Type u) (_ : Field K) (_ : Algebra k K)
37       (y : Fin m → LaurentSeries K) (xi : Fin m → PowerSeries K),
38       Sum.elim y
39           (fun i ↦ algebraMap (PowerSeries K) (LaurentSeries K) (xi i)) ∈
40         groundEquationConormalLocus (k := k) (K := K) P.asIdeal ∧
41       residueColumn xi =
42         (fun i : Fin m ↦ if i = ⟨0, hm⟩ then 1 else 0) := by
43   by_cases halg : IsAlgebraic k (componentCoordinate P ⟨0, hm⟩)
44   · obtain ⟨c, y, hc, hy, _⟩ :=
45       exists_constant_coordinate_equation_and_pure_axis hm P halg
46     exact exists_residueExtensionConormalAxis_of_ambientCoordinate_constant
47       hm P.asIdeal ⟨y, hy⟩ ⟨c, hc⟩
48   · apply exists_groundConormalAxis_of_minimalPrime_unit_transcendental
49       hm P.asIdeal P.isPrime.isRadical P
50     · simp [Ideal.minimalPrimes_eq_subsingleton_self]
51     · exact exists_coordinate_inverse_of_avoidance P.asIdeal ⟨0, hm⟩ havoid
52     · exact halg
