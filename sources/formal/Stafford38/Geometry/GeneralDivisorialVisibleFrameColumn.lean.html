17 /-!
18 # Divisorial visible frames for arbitrary prime affine components
19 
20 An invertible, transcendental coordinate on a prime affine component gives
21 a normalized visible divisor frame. This removes the canonical Weyl-support
22 hypotheses from the boundary producer. Identification of the resulting Laurent
23 direction with the smooth projective conormal closure is performed downstream
24 in the general asymptotic-conormal construction.
25 -/

49 /-- Transport residue algebraicity across an equality of ground algebra
50 structures. -/
51 theorem algebraic_adjoin_transfer_of_algebra_eq
52     {k L : Type u} [Field k] [Field L]
53     (A B : Algebra k L) (hAB : A = B) (s : Set L)
54     (hA : letI : Algebra k L := A;
55       Algebra.IsAlgebraic (IntermediateField.adjoin k s : IntermediateField k L) L) :
56     letI : Algebra k L := B;
57       Algebra.IsAlgebraic (IntermediateField.adjoin k s : IntermediateField k L) L := by
58   cases hAB
59   exact hA

61 /-- The normalized Lane-C projective column and the actual retained ground
62 coefficient map. -/
63 structure GeneralDivisorialVisibleFrameColumn
64     {k : Type u} [Field k] [CharZero k]
65     {m : ℕ} (hm : 0 < m)
66     (P : PrimeSpectrum (MvPolynomial (Fin m) k)) where
67   W : Data k (ComponentFractionField P) (componentCoordinate P ⟨0, hm⟩)
68   chart : Fin (m + 1)
69   q :
70     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
71       (ComponentFractionField P) := W.ambientAlgebra
72     Fin (m + 1) → W.place.valuation.toSubring
73   scale : ComponentFractionField P
74   scale_ne : scale ≠ 0
75   chart_one :
76     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
77       (ComponentFractionField P) := W.ambientAlgebra
78     q chart = 1
79   q0_ne :
80     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
81       (ComponentFractionField P) := W.ambientAlgebra
82     q 0 ≠ 0
83   q_commonScale : ∀ a,
84     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
85       (ComponentFractionField P) := W.ambientAlgebra
86     (q a : ComponentFractionField P) = scale * componentProjectivePoint P a
87   q_parameter :
88     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
89       (ComponentFractionField P) := W.ambientAlgebra
90     q (Fin.succ ⟨0, hm⟩) = q 0 * W.place.parameter
91   groundCoeff :
92     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
93       (ComponentFractionField P) := W.ambientAlgebra
94     k →+* W.place.valuation.toSubring
95   groundCoeff_commutes :
96     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
97       (ComponentFractionField P) := W.ambientAlgebra
98     W.place.valuation.toSubring.subtype.comp groundCoeff =
99       algebraMap k (ComponentFractionField P)
100   groundCoeff_eq_retained :
101     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
102       (ComponentFractionField P) := W.ambientAlgebra
103     groundCoeff = retainedComponentCoefficientMap P ⟨0, hm⟩ W
104   groundTower :
105     letI : Algebra (CoordinateZeroLocalRing W.coefficientField)
106       (ComponentFractionField P) := W.ambientAlgebra
107     let V := W.place.valuation.toSubring
108     letI : Algebra k V := groundCoeff.toAlgebra
109     letI : Algebra V (ComponentFractionField P) := V.subtype.toAlgebra
110     IsScalarTower k V (ComponentFractionField P)
