7 /-!
8 # One-variable prime components and the conormal axis
9 
10 The completed boundary chart is unavailable in one affine variable, but no
11 boundary argument is needed there.  A prime component of the affine line
12 whose zero set avoids the origin is nonzero.  Transporting it to the ordinary
13 polynomial ring makes it a nonzero prime in a PID, hence a maximal ideal.
14 Over an algebraically closed field it is therefore the ideal of one point
15 `c != 0`, and in particular contains `X - c`.
16 
17 The final theorem feeds this component equation into the existing
18 constant-coordinate consumer and produces the exact Laurent conormal axis.
19 No projective chart, completion, Gabber input, or filtered cancellation is
20 used here.
21 -/

34 /-- The standard identification of a one-variable multivariate polynomial
35 ring with the ordinary polynomial ring. -/
36 private abbrev finOnePolynomialEquiv (k : Type u) [Field k] :
37     MvPolynomial (Fin 1) k ≃ₐ[k] Polynomial k :=
38   MvPolynomial.uniqueAlgEquiv k (Fin 1)
39 
40 /-- A prime component of the affine line whose entire zero set avoids the
41 origin is maximal.  Algebraic closedness is not needed for this PID step. -/
42 theorem primeComponent_isMaximal_of_axis_avoidance
43     {k : Type u} [Field k]
44     (P : Ideal (MvPolynomial (Fin 1) k)) [P.IsPrime]
45     (havoid : ∀ y : Fin 1 → k,
46       (∀ f ∈ P, MvPolynomial.eval y f = 0) → y 0 ≠ 0) :
47     P.IsMaximal := by
48   have hPne : P ≠ ⊥ := by
49     intro hP
50     have hzero : ∀ f ∈ P,
51         MvPolynomial.eval (fun _ : Fin 1 ↦ (0 : k)) f = 0 := by
52       intro f hf
53       have hf0 : f = 0 := by
54         rw [hP] at hf
55         simpa using hf
56       rw [hf0]
57       simp
58     exact (havoid (fun _ ↦ 0) hzero) rfl
59   let e := finOnePolynomialEquiv k
60   let Q : Ideal (Polynomial k) := Ideal.map e P
61   letI : Q.IsPrime := by
62     dsimp [Q]
63     infer_instance
64   have hQne : Q ≠ ⊥ := by
65     intro hQ
66     apply hPne
67     apply (Ideal.map_eq_bot_iff_of_injective e.injective).mp
68     exact hQ
69   have hQmax : Q.IsMaximal := IsPrime.to_maximal_ideal hQne
70   exact (Ideal.isMaximal_map_iff_of_bijective e e.bijective).mp hQmax

72 /-- Componentwise one-variable production: axis avoidance forces the prime
73 component to be the ideal of a nonzero point and supplies its linear
74 coordinate equation. -/
75 theorem exists_coordinate_sub_constant_mem_of_prime_axis_avoidance
76     {k : Type u} [Field k] [IsAlgClosed k]
77     (P : Ideal (MvPolynomial (Fin 1) k)) [P.IsPrime]
78     (havoid : ∀ y : Fin 1 → k,
79       (∀ f ∈ P, MvPolynomial.eval y f = 0) → y 0 ≠ 0) :
80     ∃ (c : k) (y : Fin 1 → k),
81       c ≠ 0 ∧ y 0 = c ∧
82       P = MvPolynomial.vanishingIdeal k {y} ∧
83       MvPolynomial.X (0 : Fin 1) - MvPolynomial.C c ∈ P := by
84   have hmax := primeComponent_isMaximal_of_axis_avoidance P havoid
85   obtain ⟨y, hPy⟩ :=
86     MvPolynomial.isMaximal_iff_eq_vanishingIdeal_singleton.mp hmax
87   let c : k := y 0
88   have hy : ∀ f ∈ P, MvPolynomial.eval y f = 0 := by
89     intro f hf
90     rw [hPy] at hf
91     exact (MvPolynomial.mem_vanishingIdeal_singleton_iff y f).mp hf
92   have hc : c ≠ 0 := havoid y hy
93   have hcoordinate :
94       MvPolynomial.X (0 : Fin 1) - MvPolynomial.C c ∈ P := by
95     rw [hPy, MvPolynomial.mem_vanishingIdeal_singleton_iff]
96     simp [c]
97   exact ⟨c, y, hc, rfl, hPy, hcoordinate⟩

99 /-- Ambient-to-component form.  If `P` is a prime component above `I`, then
100 avoidance of the coordinate origin by `V(I)` also holds on `V(P)`, so the
101 linear equation is produced in the component ideal `P`. -/
102 theorem exists_coordinate_sub_constant_mem_of_primeComponent
103     {k : Type u} [Field k] [IsAlgClosed k]
104     (I P : Ideal (MvPolynomial (Fin 1) k)) [P.IsPrime]
105     (hIP : I ≤ P)
106     (havoid : ∀ y : Fin 1 → k,
107       (∀ f ∈ I, MvPolynomial.eval y f = 0) → y 0 ≠ 0) :
108     ∃ (c : k) (y : Fin 1 → k),
109       c ≠ 0 ∧ y 0 = c ∧
110       P = MvPolynomial.vanishingIdeal k {y} ∧
111       MvPolynomial.X (0 : Fin 1) - MvPolynomial.C c ∈ P := by
112   apply exists_coordinate_sub_constant_mem_of_prime_axis_avoidance P
113   intro y hyP
114   apply havoid y
115   intro f hf
116   exact hyP f (hIP hf)

118 /-- Direct rank-one replacement for the impossible boundary-chart branch.
119 The fibre residue is the unique pure coordinate axis. -/
120 theorem exists_laurentConormalAxis_of_prime_axis_avoidance
121     {k : Type u} [Field k] [IsAlgClosed k]
122     (P : Ideal (MvPolynomial (Fin 1) k)) [P.IsPrime]
123     (havoid : ∀ y : Fin 1 → k,
124       (∀ f ∈ P, MvPolynomial.eval y f = 0) → y 0 ≠ 0) :
125     ∃ (yL : Fin 1 → LaurentSeries k)
126       (xi : Fin 1 → PowerSeries k),
127       Sum.elim yL
128           (fun j ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi j)) ∈
129         equationConormalLocus
130           (P.map (scalarPolynomialMap
131             (k := k) (K := LaurentSeries k) (Fin 1))) ∧
132       residueColumn xi = (fun _ : Fin 1 ↦ 1) := by
133   obtain ⟨c, y, _hc, _hyc, hPy, hcoordinate⟩ :=
134     exists_coordinate_sub_constant_mem_of_prime_axis_avoidance P havoid
135   have hy : ∀ f ∈ P, MvPolynomial.eval y f = 0 := by
136     intro f hf
137     rw [hPy] at hf
138     exact (MvPolynomial.mem_vanishingIdeal_singleton_iff y f).mp hf
139   obtain ⟨yL, xi, hphase, hresidue⟩ :=
140     exists_laurentConormalAxis_of_coordinate_sub_constant_mem
141       P y hy (0 : Fin 1) c hcoordinate
142   refine ⟨yL, xi, hphase, ?_⟩
143   rw [hresidue]
144   funext j
145   fin_cases j
146   simp
