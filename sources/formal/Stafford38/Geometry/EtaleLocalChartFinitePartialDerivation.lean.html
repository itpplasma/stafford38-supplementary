11 /-!
12 # Full chart partials before restriction to a tilted arc
13 
14 The statements here are conditional on an already-given formally-etale
15 polynomial chart.  The ambient projective-coordinate index is independent of
16 the local chart dimension.  In particular, all transverse partials are
17 computed in the multivariate completion before using the existing tilted-arc
18 map.
19 -/

37 /-- The `B`-valued partial derivation dual to a parameter of an existing
38 formally-etale polynomial chart. -/
39 noncomputable def parameterDerivation (i : Fin (d + 1)) : Derivation k B B :=
40   Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
41     (k := k) (σ := Fin (d + 1)) (B := B) (L := B) i

43 theorem parameterDerivation_apply_parameter (i j : Fin (d + 1)) :
44     parameterDerivation (k := k) (d := d) (B := B) i
45       (algebraMap ChartBase B (MvPolynomial.X j)) = if j = i then (1 : B) else 0 := by
46   exact Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation_apply_parameter
47     (k := k) (σ := Fin (d + 1)) (B := B) (L := B) i j

49 /-- Extending a precompletion parameter derivation to a field-valued point
50 agrees with the existing Kähler-basis coordinate derivation.  The equality is
51 deduced from the checked uniqueness theorem using all polynomial parameter
52 values. -/
53 theorem parameterDerivation_scalarExtension_eq_coordinate
54     {L : Type w} [CommRing L] [Algebra B L] [Algebra k L]
55     [IsScalarTower k B L] (i : Fin (d + 1)) :
56     (Algebra.linearMap B L).compDer
57         (parameterDerivation (k := k) (d := d) (B := B) i) =
58       Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
59         (k := k) (σ := Fin (d + 1)) (B := B) (L := L) i := by
60   let D : Derivation k B L := (Algebra.linearMap B L).compDer
61     (parameterDerivation (k := k) (d := d) (B := B) i)
62   have hD : ∀ j, D (algebraMap ChartBase B (MvPolynomial.X j)) =
63       if j = i then (1 : L) else 0 := by
64     intro j
65     change algebraMap B L
66       (parameterDerivation (k := k) (d := d) (B := B) i
67         (algebraMap ChartBase B (MvPolynomial.X j))) = _
68     rw [parameterDerivation_apply_parameter]
69     simp
70   change D = _
71   calc
72     D = Stafford38.Geometry.EtaleCotangentBasis.derivationFromValues
73         (k := k) (σ := Fin (d + 1)) (B := B) (L := L)
74         (fun j => D (algebraMap ChartBase B (MvPolynomial.X j))) :=
75       (Stafford38.Geometry.EtaleCotangentBasis.derivation_eq_from_parameter_values D).symm
76     _ = Stafford38.Geometry.EtaleCotangentBasis.derivationFromValues
77         (k := k) (σ := Fin (d + 1)) (B := B) (L := L)
78         (fun j => if j = i then (1 : L) else 0) := by
79       congr 1
80       funext j
81       exact hD j
82     _ = Stafford38.Geometry.EtaleCotangentBasis.coordinateDerivation
83         (k := k) (σ := Fin (d + 1)) (B := B) (L := L) i := rfl

85 /-- A selected row equal to its centered chart coordinate has transverse
86 partials equal to the identity and distinguished-parameter partial zero.
87 The map `rows` may land in any ambient `Fin (n+1)`; no dimension equality with
88 `d+1` is assumed. -/
89 theorem selectedRow_parameterDerivatives {n : ℕ}
90     (q : Fin (n + 1) → B) (rows : Fin d ↪ Fin (n + 1))
91     (β : Fin d → k)
92     (hq : ∀ j, q (rows j) = algebraMap k B (β j) +
93       algebraMap ChartBase B (MvPolynomial.X j.succ)) :
94     (∀ i j, parameterDerivation (k := k) (d := d) (B := B) i.succ
95         (q (rows j)) = if i = j then (1 : B) else 0) ∧
96       (∀ j, parameterDerivation (k := k) (d := d) (B := B) 0
97         (q (rows j)) = 0) := by
98   constructor
99   · intro i j
100     rw [hq j]
101     simp [Derivation.map_add, parameterDerivation_apply_parameter, eq_comm]
102   · intro j
103     rw [hq j]
104     simp [Derivation.map_add, parameterDerivation_apply_parameter]

106 /-- The same selected-row formula in the full multivariate completion. -/
107 theorem selectedRow_completedPartials {n : ℕ}
108     (q : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)
109     (rows : Fin d ↪ Fin (n + 1)) (β : Fin d → k)
110     (hq : ∀ j, q (rows j) = MvPowerSeries.C (β j) +
111       MvPowerSeries.X j.succ) :
112     (∀ i j, MvPowerSeries.pderiv i.succ (q (rows j)) =
113         if i = j then 1 else 0) ∧
114       (∀ j, MvPowerSeries.pderiv 0 (q (rows j)) = 0) := by
115   constructor
116   · intro i j
117     rw [hq j]
118     by_cases h : i = j
119     · subst j
120       simp
121     · have h' : j.succ ≠ i.succ := by simpa using Ne.symm h
122       simp [h, MvPowerSeries.pderiv_X_of_ne, h']
123   · intro j
124     rw [hq j]
125     simp

127 /-- The canonical tilted arc sends a selected affine chart row to
128 `β_j + α_j t`, so its first derivative is `α_j`. -/
129 theorem tiltedArc_X_succ (α : Fin d → k) (j : Fin d) :
130     tiltedArc (k := k) α (MvPowerSeries.X j.succ) =
131       PowerSeries.C (α j) * PowerSeries.X := by
132   change MvPowerSeries.rename (Equiv.equivPUnit Unit).symm
133     ((MvPowerSeries.substAlgHom
134       (tiltedArcHasSubstPUnit (k := k) α)) (MvPowerSeries.X j.succ)) = _
135   rw [MvPowerSeries.substAlgHom_X]
136   change MvPowerSeries.rename (Equiv.equivPUnit Unit).symm
137     (MvPowerSeries.C (tiltedArcCoefficients (k := k) α j.succ) *
138       MvPowerSeries.X (PUnit.unit : PUnit.{1})) = _
139   rw [map_mul, MvPowerSeries.rename_C, MvPowerSeries.rename_X]
140   rw [tiltedArcCoefficients_succ]
141   rfl

143 theorem tiltedArc_derivative_parameter (α : Fin d → k) (i : Fin (d + 1)) :
144     PowerSeries.derivative (R := k) (tiltedArc (k := k) α (MvPowerSeries.X i)) =
145       PowerSeries.C (tiltedArcCoefficients (k := k) α i) := by
146   cases i using Fin.cases with
147   | zero =>
148       rw [tiltedArc_X_t]
149       simp
150   | succ j =>
151       rw [tiltedArc_X_succ]
152       simp only [tiltedArcCoefficients_succ, Derivation.leibniz,
153         PowerSeries.derivative_C, PowerSeries.derivative_X]
154       simp

156 theorem selectedRow_tiltedArc_derivative {n : ℕ}
157     (q : Fin (n + 1) → MvPowerSeries (Fin (d + 1)) k)
158     (rows : Fin d ↪ Fin (n + 1)) (β α : Fin d → k)
159     (hq : ∀ j, q (rows j) = MvPowerSeries.C (β j) +
160       MvPowerSeries.X j.succ) :
161     ∀ j, PowerSeries.derivative (R := k)
162       (tiltedArc (k := k) α (q (rows j))) = PowerSeries.C (α j) := by
163   intro j
164   rw [hq j]
165   have hx := tiltedArc_X_succ (k := k) α j
166   have hC : tiltedArc (k := k) α (MvPowerSeries.C (β j)) =
167       PowerSeries.C (β j) := by
168     change punitToPowerSeries (k := k)
169       (tiltedArcMapPUnit (k := k) α (MvPowerSeries.C (β j))) = _
170     have hCP : tiltedArcMapPUnit (k := k) α (MvPowerSeries.C (β j)) =
171         MvPowerSeries.C (β j) := by
172       simp [tiltedArcMapPUnit]
173     rw [hCP]
174     change MvPowerSeries.rename (Equiv.equivPUnit Unit).symm
175       (MvPowerSeries.C (β j)) = _
176     rw [MvPowerSeries.rename_C]
177     rfl
178   change PowerSeries.derivative (R := k)
179     (tiltedArcMap (k := k) α
180       (MvPowerSeries.C (β j) + MvPowerSeries.X j.succ)) = _
181   rw [(tiltedArcMap (k := k) α).map_add]
182   change PowerSeries.derivative (R := k)
183     (tiltedArc (k := k) α (MvPowerSeries.C (β j)) +
184       tiltedArc (k := k) α (MvPowerSeries.X j.succ)) = _
185   rw [hC, hx]
186   simp only [Derivation.map_add, PowerSeries.derivative_C,
187     Derivation.leibniz, PowerSeries.derivative_C, PowerSeries.derivative_X,
188     zero_add]
189   simp
