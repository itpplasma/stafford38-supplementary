29 /-- The arc extended to the actual common open restricts to the original
30 point-local arc on the canonical map from the finite-type chart algebra.
31 This is a derived composition identity: it uses the existing localization
32 comparison and does not posit a second common-open map. -/
33 theorem genericArcToExtraAway_comp_commonOpenMap
34     {L : Type*} [Field L]
35     (M : Ideal A) [M.IsPrime] (f : Q)
36     (e : Localization.Away f ≃ₐ[Q]
37       Localization.Away (algebraMap Q A f)) (g : Q)
38     (ρ : A →+* L)
39     (hf : IsUnit (ρ (algebraMap Q A f)))
40     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b))
41     (hg : IsUnit (ρ (algebraMap Q A g))) :
42     (genericArcToGenericOpenExtraAwayB M f e ρ hf hunitM g hg).comp
43       (genericOpenExtraAwayBMap M f e g) = ρ := by
44   let T := Localization.AtPrime M
45   let U := genericOpenExtraAwayB M f e g
46   let θ := pointLocalToCommonOpen (A := A) M f e g
47   let ρU := genericArcToGenericOpenExtraAwayB M f e ρ hf hunitM g hg
48   have hlocal := pointLocalToCommonOpen_comp_algebraMap (A := A) M f e g
49   have harc := genericArcToExtraAway_comp_pointLocalToCommonOpen
50     (A := A) M f e g ρ hf hunitM hg
51   ext a
52   have hmap : genericOpenExtraAwayBMap M f e g a =
53       θ (algebraMap A T a) := by
54     have h := congrArg (fun φ : A →+* U => φ a) hlocal
55     simpa only [RingHom.comp_apply, θ, T, U] using h.symm
56   calc
57     ρU (genericOpenExtraAwayBMap M f e g a) =
58         ρU (θ (algebraMap A T a)) := congrArg ρU hmap
59     _ = localPointToField M ρ hunitM (algebraMap A T a) := by
60       have h := congrArg (fun φ : T →+* L => φ (algebraMap A T a)) harc
61       simpa only [RingHom.comp_apply, ρU, θ] using h
62     _ = ρ a := by
63       exact localPointToField_apply M ρ hunitM a

65 /-- Nonvanishing at the original point-local arc is preserved by the same
66 common-open map, including for a homogenized numerator. -/
67 theorem genericArcToExtraAway_ne_zero_of_pointArc_ne_zero
68     {L : Type*} [Field L]
69     (M : Ideal A) [M.IsPrime] (f : Q)
70     (e : Localization.Away f ≃ₐ[Q]
71       Localization.Away (algebraMap Q A f)) (g : Q)
72     (ρ : A →+* L)
73     (hf : IsUnit (ρ (algebraMap Q A f)))
74     (hunitM : ∀ b, b ∉ M → IsUnit (ρ b))
75     (hg : IsUnit (ρ (algebraMap Q A g)))
76     (a : A) (ha : ρ a ≠ 0) :
77     genericArcToGenericOpenExtraAwayB M f e ρ hf hunitM g hg
78       (genericOpenExtraAwayBMap M f e g a) ≠ 0 := by
79   have hcomp := genericArcToExtraAway_comp_commonOpenMap
80     (A := A) M f e g ρ hf hunitM hg
81   have h := congrArg (fun φ : A →+* L => φ a) hcomp
82   have h' : genericArcToGenericOpenExtraAwayB M f e ρ hf hunitM g hg
83       (genericOpenExtraAwayBMap M f e g a) = ρ a := by
84     simpa only [RingHom.comp_apply] using h
85   rw [h']
86   exact ha

108 /-- A single selected tilt that avoids a product certifies nonvanishing of
109 each of its three actual point-local factors after mapping to the Laurent
110 series field. -/
111 theorem pointArc_factors_ne_zero_of_tiltedProduct
112     (M : Ideal B) [M.IsMaximal] (eM : (B ⧸ M) ≃ₐ[k] k)
113     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k)
114       (Localization.AtPrime M)]
115     (α : Fin d → k) (a b c : B)
116     (havoid : tiltedArc (k := k) α
117       (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
118           (algebraMap B (Localization.AtPrime M) a) *
119        localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
120           (algebraMap B (Localization.AtPrime M) b) *
121        localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
122           (algebraMap B (Localization.AtPrime M) c)) ≠ 0) :
123     (originalToPointLocalFinSuccArcLaurentSeries
124       (k := k) (d := d) (A := B) M eM α a ≠ 0) ∧
125     (originalToPointLocalFinSuccArcLaurentSeries
126       (k := k) (d := d) (A := B) M eM α b ≠ 0) ∧
127     (originalToPointLocalFinSuccArcLaurentSeries
128       (k := k) (d := d) (A := B) M eM α c ≠ 0) := by
129   have hmul :
130       tiltedArc (k := k) α
131           (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
132             (algebraMap B (Localization.AtPrime M) a)) *
133       tiltedArc (k := k) α
134           (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
135             (algebraMap B (Localization.AtPrime M) b)) *
136       tiltedArc (k := k) α
137           (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
138             (algebraMap B (Localization.AtPrime M) c)) ≠ 0 := by
139     simpa only [tiltedArc, map_mul] using havoid
140   have ha : tiltedArc (k := k) α
141       (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
142         (algebraMap B (Localization.AtPrime M) a)) ≠ 0 := by
143     intro hz
144     apply hmul
145     simp [hz]
146   have hb : tiltedArc (k := k) α
147       (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
148         (algebraMap B (Localization.AtPrime M) b)) ≠ 0 := by
149     intro hz
150     apply hmul
151     simp [hz]
152   have hc : tiltedArc (k := k) α
153       (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
154         (algebraMap B (Localization.AtPrime M) c)) ≠ 0 := by
155     intro hz
156     apply hmul
157     simp [hz]
158   have hla : algebraMap (PowerSeries k) (LaurentSeries k)
159       (tiltedArc (k := k) α
160         (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
161           (algebraMap B (Localization.AtPrime M) a))) ≠ 0 :=
162     (by simpa only [map_zero] using
163       (IsFractionRing.injective (PowerSeries k) (LaurentSeries k)).ne ha)
164   have hlb : algebraMap (PowerSeries k) (LaurentSeries k)
165       (tiltedArc (k := k) α
166         (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
167           (algebraMap B (Localization.AtPrime M) b))) ≠ 0 :=
168     (by simpa only [map_zero] using
169       (IsFractionRing.injective (PowerSeries k) (LaurentSeries k)).ne hb)
170   have hlc : algebraMap (PowerSeries k) (LaurentSeries k)
171       (tiltedArc (k := k) α
172         (localToFinSuccPowerSeries (k := k) (B := B) (d := d) M eM
173           (algebraMap B (Localization.AtPrime M) c))) ≠ 0 :=
174     (by simpa only [map_zero] using
175       (IsFractionRing.injective (PowerSeries k) (LaurentSeries k)).ne hc)
176   constructor
177   · rw [originalPointArc_eq_laurent_tilt]
178     exact hla
179   constructor
180   · rw [originalPointArc_eq_laurent_tilt]
181     exact hlb
182   · rw [originalPointArc_eq_laurent_tilt]
183     exact hlc

189 /-- The point-local tilt-avoidance witness supplies the actual away-unit
190 inputs while keeping the homogenized numerator `r` separate from the
191 common-open chart denominator `g`. The bad product is `map(f) * r`; the
192 zeroth column is `map(g)`. The denominator need not avoid the closed center:
193 its image is a unit because the same arc extends from the chosen point-local
194 ring. -/
195 theorem commonOpen_factors_ne_zero_of_tiltedProduct
196     (M : Ideal A) [M.IsMaximal]
197     (f : Q) (e : Localization.Away f ≃ₐ[Q]
198       Localization.Away (algebraMap Q A f)) (g : Q)
199     (eM : (A ⧸ M) ≃ₐ[k] k) (α : Fin d → k)
200     [Algebra.FormallyEtale (MvPolynomial (Option (Fin d)) k)
201       (Localization.AtPrime M)]
202     (r bad q₀ q₁ : A)
203     (hbad : bad = algebraMap Q A f * r)
204     (hq₀ : q₀ = algebraMap Q A g)
205     (havoid : tiltedArc (k := k) α
206       (localToFinSuccPowerSeries (k := k) (B := A) (d := d) M eM
207           (algebraMap A (Localization.AtPrime M) bad) *
208        localToFinSuccPowerSeries (k := k) (B := A) (d := d) M eM
209           (algebraMap A (Localization.AtPrime M) q₀) *
210        localToFinSuccPowerSeries (k := k) (B := A) (d := d) M eM
211           (algebraMap A (Localization.AtPrime M) q₁)) ≠ 0) :
212     let ρA := originalToPointLocalFinSuccArcLaurentSeries
213       (k := k) (d := d) (A := A) M eM α
214     ∃ (hf : IsUnit (ρA (algebraMap Q A f)))
215       (hg : IsUnit (ρA (algebraMap Q A g))),
216       let hunitM : ∀ b, b ∉ M → IsUnit (ρA b) := fun b hb => by
217         let T := Localization.AtPrime M
218         have hu : IsUnit (algebraMap A T b) :=
219           IsLocalization.map_units T (M := M.primeCompl) ⟨b, hb⟩
220         change IsUnit
221           ((pointLocalFinSuccArcLaurentSeries
222             (k := k) (d := d) (A := A) M eM α).toRingHom
223               (algebraMap A T b))
224         exact IsUnit.map _ hu
225       let ρU := genericArcToGenericOpenExtraAwayB M f e ρA hf hunitM g hg
226       (ρU.comp (genericOpenExtraAwayBMap M f e g) = ρA) ∧
227       ((ρU.comp (genericOpenExtraAwayBMap M f e g)).comp
228           (algebraMap k A) = algebraMap k (LaurentSeries k)) ∧
229       (ρU (genericOpenExtraAwayBMap M f e g bad) ≠ 0) ∧
230       (ρU (genericOpenExtraAwayBMap M f e g r) ≠ 0) ∧
231       (ρU (genericOpenExtraAwayBMap M f e g q₀) ≠ 0) ∧
232       (ρU (genericOpenExtraAwayBMap M f e g q₁) ≠ 0) := by
233   let ρA := originalToPointLocalFinSuccArcLaurentSeries
234     (k := k) (d := d) (A := A) M eM α
235   have hpoints := pointArc_factors_ne_zero_of_tiltedProduct
236     (k := k) (d := d) (B := A) M eM α bad q₀ q₁ havoid
237   rcases hpoints with ⟨hbadPoint, hq₀Point, hq₁Point⟩
238   have hbadFactor : ρA bad =
239       ρA (algebraMap Q A f) * ρA r := by
240     calc
241       ρA bad = ρA (algebraMap Q A f * r) := congrArg ρA hbad
242       _ = ρA (algebraMap Q A f) * ρA r := map_mul ρA _ _
243   have hq₀Chart : ρA q₀ = ρA (algebraMap Q A g) := congrArg ρA hq₀
244   have hf0 : ρA (algebraMap Q A f) ≠ 0 := by
245     intro hz
246     apply hbadPoint
247     rw [hbadFactor, hz]
248     simp
249   have hr0 : ρA r ≠ 0 := by
250     intro hz
251     apply hbadPoint
252     rw [hbadFactor, hz]
253     simp
254   have hg0 : ρA (algebraMap Q A g) ≠ 0 := by
255     intro hz
256     apply hq₀Point
257     rw [hq₀Chart, hz]
258   have hf : IsUnit (ρA (algebraMap Q A f)) := isUnit_iff_ne_zero.mpr hf0
259   have hg : IsUnit (ρA (algebraMap Q A g)) := isUnit_iff_ne_zero.mpr hg0
260   have hunitM : ∀ b, b ∉ M → IsUnit (ρA b) := by
261     intro b hb
262     let T := Localization.AtPrime M
263     have hu : IsUnit (algebraMap A T b) :=
264       IsLocalization.map_units T (M := M.primeCompl) ⟨b, hb⟩
265     change IsUnit
266       ((pointLocalFinSuccArcLaurentSeries
267         (k := k) (d := d) (A := A) M eM α).toRingHom (algebraMap A T b))
268     exact IsUnit.map _ hu
269   let ρU := genericArcToGenericOpenExtraAwayB M f e ρA hf hunitM g hg
270   have hcomp := genericArcToExtraAway_comp_commonOpenMap
271     (A := A) M f e g ρA hf hunitM hg
272   have hbaseA : ρA.comp (algebraMap k A) =
273       algebraMap k (LaurentSeries k) := by
274     apply RingHom.ext
275     intro c
276     exact originalToPointLocalFinSuccArcLaurentSeries_base
277       (k := k) (d := d) (A := A) M eM α c
278   have hground : (ρU.comp (genericOpenExtraAwayBMap M f e g)).comp
279       (algebraMap k A) = algebraMap k (LaurentSeries k) := by
280     calc
281       (ρU.comp (genericOpenExtraAwayBMap M f e g)).comp (algebraMap k A) =
282           ρA.comp (algebraMap k A) := by
283             exact congrArg (fun φ : A →+* LaurentSeries k =>
284               φ.comp (algebraMap k A)) hcomp
285       _ = algebraMap k (LaurentSeries k) := hbaseA
286   have hmap (a : A) : ρU (genericOpenExtraAwayBMap M f e g a) = ρA a := by
287     have h := congrArg (fun φ : A →+* LaurentSeries k => φ a) hcomp
288     simpa only [RingHom.comp_apply, ρU] using h
289   refine ⟨hf, hg, hcomp, hground, ?_, ?_, ?_, ?_⟩
290   · rw [hmap]
291     exact hbadPoint
292   · rw [hmap]
293     exact hr0
294   · rw [hmap]
295     exact hq₀Point
296   · rw [hmap]
297     exact hq₁Point
