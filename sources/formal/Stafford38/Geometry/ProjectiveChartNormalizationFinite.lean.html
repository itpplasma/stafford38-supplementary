8 /-!
9 # Finiteness of the normalization in a chosen fraction-field model
10 
11 The chart-to-valuation construction uses a concrete function field, while
12 Noether normalization is most convenient in `FractionRing R`. This adapter
13 transports the finite normalization to that concrete field without changing
14 the base domain.
15 -/

27 /-- The integral closure of a finite-type domain in any chosen realization of
28 its fraction field is finite over the original domain. -/
29 theorem finite_integralClosure_in_fractionField
30     (k R K : Type u) [Field k] [CharZero k] [CommRing R] [IsDomain R]
31     [Algebra k R] [Algebra.FiniteType k R]
32     [Field K] [Algebra R K] [IsFractionRing R K] :
33     Module.Finite R (integralClosure R K) := by
34   let C₀ : Subalgebra R (FractionRing R) := integralClosure R (FractionRing R)
35   have hC₀ : Module.Finite R C₀ :=
36     Stafford38.Geometry.NormalizationHeightOne.finite_normalization_of_fg_domain
37       k R C₀
38   let e : FractionRing R ≃ₐ[R] K :=
39     IsLocalization.algEquiv R⁰ (FractionRing R) K
40   exact Module.Finite.equiv e.mapIntegralClosure.toLinearEquiv
41 
42 /-- The integral closure is finite type over the ground field, before any
43 localization of the ground field or of the domain. -/
44 theorem integralClosure_finiteType_over_base
45     (k R K : Type u) [Field k] [CharZero k] [CommRing R] [IsDomain R]
46     [Algebra k R] [Algebra.FiniteType k R]
47     [Field K] [Algebra R K] [IsFractionRing R K] :
48     let C : Subalgebra R K := integralClosure R K
49     letI : Algebra k C := Algebra.compHom C (algebraMap k R)
50     letI : IsScalarTower k R C := IsScalarTower.of_algebraMap_eq fun _ => rfl
51     Algebra.FiniteType k C := by
52   let C : Subalgebra R K := integralClosure R K
53   letI : Algebra k C := Algebra.compHom C (algebraMap k R)
54   haveI : IsScalarTower k R C := IsScalarTower.of_algebraMap_eq fun _ => rfl
55   haveI : Module.Finite R C :=
56     finite_integralClosure_in_fractionField k R K
57   exact Algebra.FiniteType.trans
58     (inferInstance : Algebra.FiniteType k R)
59     (inferInstance : Algebra.FiniteType R C)
