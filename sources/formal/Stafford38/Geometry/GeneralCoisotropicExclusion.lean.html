33 abbrev CoordinateAxisClosureInput
34     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k] :=
35   ∀ {m : ℕ} (hm : 0 < m) (I : PrimeSpectrum (MvPolynomial (Fin m) k)),
36     (∀ y ∈ MvPolynomial.zeroLocus k I.asIdeal, y ⟨0, hm⟩ ≠ 0) →
37     (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) ∈
38       MvPolynomial.zeroLocus k
39         (MvPolynomial.vanishingIdeal k
40           (smoothConormalFibreProjection I.asIdeal))

42 theorem exists_zero_base_coordinate_using
43     {k : Type u} [Field k] [IsAlgClosed k] [CharZero k]
44     (axisClosure : CoordinateAxisClosureInput (k := k))
45     {m : ℕ} (hm : 0 < m)
46     (J : Ideal (SymbolRing k m)) (hproper : J ≠ ⊤)
47     (hrad : J.IsRadical)
48     (hhom : J.IsHomogeneous (orderDecomposition (k := k) (n := m)))
49     (hpoisson : IsBaseRelativePoisson J)
50     (P : MvPolynomial (Fin m) k) (hP : fibreLift P ∈ J)
51     (haxis : MvPolynomial.eval
52       (fun i : Fin m => if i = ⟨0, hm⟩ then (1 : k) else 0) P ≠ 0) :
53     ∃ q ∈ MvPolynomial.zeroLocus k J, q (.inl ⟨0, hm⟩) = 0 := by
54   classical
55   by_contra hnone
56   push_neg at hnone
57   let B := J.comap (baseLift (k := k) (n := m)).toRingHom
58   obtain ⟨Q, hQ⟩ := Ideal.nonempty_minimalPrimes
59     (show B ≠ ⊤ from Ideal.comap_ne_top _ hproper)
60   let qprime : PrimeSpectrum (MvPolynomial (Fin m) k) := ⟨Q, hQ.1.1⟩
61   have havoid : ∀ y ∈ MvPolynomial.zeroLocus k Q, y ⟨0, hm⟩ ≠ 0 := by
62     intro y hy
63     have hyB : ∀ f ∈ B, MvPolynomial.eval y f = 0 := by
64       intro f hf
65       exact hy f (hQ.1.2 hf)
66     have hzero : zeroSectionPoint y ∈ MvPolynomial.zeroLocus k J :=
67       zeroSection_commonZero_of_isHomogeneous J hhom y hyB
68     exact hnone (zeroSectionPoint y) hzero
69   have hclosure := axisClosure hm qprime havoid
70   apply haxis
71   apply hclosure P
72   intro ξ hξ
73   obtain ⟨y, hysmooth, hconormal⟩ := hξ
74   have hy : y ∈ MvPolynomial.zeroLocus k Q :=
75     smoothAffinePoint_mem_zeroLocus Q hysmooth
76   have hpoint : Sum.elim y ξ ∈ equationConormalClosure Q := by
77     intro f hf
78     exact hf (Sum.elim y ξ) ⟨hy, hconormal⟩
79   have hJpoint := equationConormalClosure_minimalPrime_subset_zeroLocus
80     J hrad hhom hpoisson Q hQ hpoint
81   have hzero := hJpoint (fibreLift P) hP
82   simpa [fibreLift, MvPolynomial.eval_rename, Function.comp_def] using hzero
