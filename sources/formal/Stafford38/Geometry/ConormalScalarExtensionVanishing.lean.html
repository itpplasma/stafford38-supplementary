26 def groundGradientSpan (I : Ideal (MvPolynomial (Fin n) k))
27     (y : Fin n → K) : Submodule K (Fin n → K) :=
28   Submodule.span K
29     (Set.range fun h : I => fun i =>
30       MvPolynomial.eval₂ (algebraMap k K) y (MvPolynomial.pderiv i h.1))

32 theorem differential_mem_groundGradientSpan
33     (I : Ideal (MvPolynomial (Fin n) k)) (y : Fin n → K)
34     (hy : ∀ h ∈ I, MvPolynomial.eval₂ (algebraMap k K) y h = 0)
35     (g : MvPolynomial (Fin n) K)
36     (hg : g ∈ I.map (scalarPolynomialMap (k := k) (K := K) (Fin n))) :
37     (fun i => differentialAt y g i) ∈ groundGradientSpan I y := by
38   classical
39   change g ∈ Ideal.span
40     (scalarPolynomialMap (k := k) (K := K) (Fin n) '' I) at hg
41   induction hg using Submodule.span_induction with
42   | mem g hg =>
43       obtain ⟨h, hh, rfl⟩ := hg
44       apply Submodule.subset_span
45       refine ⟨⟨h, hh⟩, ?_⟩
46       funext i
47       exact (differentialAt_scalarPolynomialMap y h i).symm
48   | zero =>
49       change (0 : Fin n → K) ∈ groundGradientSpan I y
50       exact (groundGradientSpan I y).zero_mem
51   | add g h _ _ hg hh =>
52       have heq : (fun i => differentialAt y (g + h) i) =
53           (fun i => differentialAt y g i) + (fun i => differentialAt y h i) := by
54         funext i
55         simp [differentialAt]
56       rw [heq]
57       exact (groundGradientSpan I y).add_mem hg hh
58   | smul a g hg ih =>
59       have hgeval : MvPolynomial.eval y g = 0 := by
60         have hy' : y ∈ MvPolynomial.zeroLocus K
61             (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n))) :=
62           (mem_zeroLocus_map_iff I y).mpr hy
63         exact hy' g hg
64       have heq : (fun i => differentialAt y (a * g) i) =
65           MvPolynomial.eval y a • (fun i => differentialAt y g i) := by
66         funext i
67         simp [differentialAt, MvPolynomial.pderiv_mul, hgeval]
68       rw [smul_eq_mul, heq]
69       exact (groundGradientSpan I y).smul_mem _ ih

71 theorem conormal_covector_mem_groundGradientSpan
72     (I : Ideal (MvPolynomial (Fin n) k)) (q : PhaseVar n → K)
73     (hq : q ∈ equationConormalLocus
74       (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n)))) :
75     (fun i => q (.inr i)) ∈ groundGradientSpan I (fun i => q (.inl i)) := by
76   classical
77   let y : Fin n → K := fun i => q (.inl i)
78   let ξ : Fin n → K := fun i => q (.inr i)
79   have hq' := (mem_equationConormalLocus_map_iff I q).mp hq
80   obtain ⟨c, hc⟩ := coordinate_mem_affineConormalSpace_exists_finsupp
81     y ξ _ hq'.2
82   have heq : ξ = ∑ g ∈ c.support, c g • (fun i => differentialAt y g.1 i) := by
83     funext i
84     simpa [Finset.sum_apply] using (hc i)
85   change ξ ∈ groundGradientSpan I y
86   rw [heq]
87   apply Submodule.sum_mem
88   intro g hg
89   exact (groundGradientSpan I y).smul_mem _
90     (differential_mem_groundGradientSpan I y hq'.1 g.1 g.2)

92 theorem exists_ground_gradient_finsupp
93     (I : Ideal (MvPolynomial (Fin n) k)) (q : PhaseVar n → K)
94     (hq : q ∈ equationConormalLocus
95       (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n)))) :
96     ∃ c : I →₀ K, ∀ i,
97       q (.inr i) = ∑ h ∈ c.support, c h *
98         MvPolynomial.eval₂ (algebraMap k K) (fun j => q (.inl j))
99           (MvPolynomial.pderiv i h.1) := by
100   classical
101   have hm := conormal_covector_mem_groundGradientSpan I q hq
102   rw [groundGradientSpan, Finsupp.mem_span_range_iff_exists_finsupp] at hm
103   obtain ⟨c, hc⟩ := hm
104   refine ⟨c, fun i => ?_⟩
105   have hi := congrFun hc i
106   simpa [Finsupp.linearCombination_apply, Finsupp.sum, mul_comm] using hi.symm

108 def finiteGradientSubstitution {ι : Type*} [Fintype ι]
109     (h : ι → MvPolynomial (Fin n) k) :
110     SymbolRing k n →+* MvPolynomial (Fin n ⊕ ι) k :=
111   MvPolynomial.eval₂Hom MvPolynomial.C
112     (Sum.elim (fun i => MvPolynomial.X (.inl i))
113       (fun i => ∑ a : ι, MvPolynomial.X (.inr a) *
114         MvPolynomial.rename Sum.inl (MvPolynomial.pderiv i (h a))))

116 theorem eval₂_finiteGradientSubstitution {L : Type*} [Field L] [Algebra k L]
117     {ι : Type*} [Fintype ι] (h : ι → MvPolynomial (Fin n) k)
118     (P : SymbolRing k n) (z : Fin n ⊕ ι → L) :
119     MvPolynomial.eval₂ (algebraMap k L) z (finiteGradientSubstitution h P) =
120       MvPolynomial.eval₂ (algebraMap k L)
121         (Sum.elim (fun i => z (.inl i))
122           (fun i => ∑ a : ι, z (.inr a) *
123             MvPolynomial.eval₂ (algebraMap k L) (fun j => z (.inl j))
124               (MvPolynomial.pderiv i (h a)))) P := by
125   induction P using MvPolynomial.induction_on with
126   | C a => simp [finiteGradientSubstitution]
127   | add P Q hP hQ =>
128       simp only [map_add, MvPolynomial.eval₂_add]
129       rw [hP, hQ]
130   | mul_X P i hP =>
131       simp only [map_mul, MvPolynomial.eval₂_mul]
132       rw [hP]
133       rcases i with i | i
134       · simp [finiteGradientSubstitution]
135       · simp [finiteGradientSubstitution, MvPolynomial.eval₂_rename]
136         left
137         congr 1

139 theorem scalarExtension_vanishing
140     (I : Ideal (MvPolynomial (Fin n) k)) (P : SymbolRing k n)
141     (hP : ∀ q ∈ equationConormalLocus I, MvPolynomial.eval q P = 0)
142     (q : PhaseVar n → K)
143     (hq : q ∈ equationConormalLocus
144       (I.map (scalarPolynomialMap (k := k) (K := K) (Fin n)))) :
145     MvPolynomial.eval₂ (algebraMap k K) q P = 0 := by
146   classical
147   obtain ⟨c, hc⟩ := exists_ground_gradient_finsupp I q hq
148   let ι := {h : I // h ∈ c.support}
149   let equations : ι → MvPolynomial (Fin n) k := fun a => a.1.1
150   let ρ := finiteGradientSubstitution equations
151   let J : Ideal (MvPolynomial (Fin n ⊕ ι) k) :=
152     I.map (MvPolynomial.rename Sum.inl)
153   have hvan : ρ P ∈ MvPolynomial.vanishingIdeal k
154       (MvPolynomial.zeroLocus k J) := by
155     rw [MvPolynomial.mem_vanishingIdeal_iff]
156     intro z hz
157     let y : Fin n → k := fun i => z (.inl i)
158     let ξ : Fin n → k := fun i => ∑ a : ι, z (.inr a) * differentialAt y (equations a) i
159     have hy : y ∈ MvPolynomial.zeroLocus k I := by
160       intro f hf
161       have hm : MvPolynomial.rename Sum.inl f ∈ J :=
162         Ideal.mem_map_of_mem (MvPolynomial.rename Sum.inl) hf
163       have := hz _ hm
164       change MvPolynomial.eval₂ (algebraMap k k) z
165         (MvPolynomial.rename Sum.inl f) = 0 at this
166       rw [MvPolynomial.eval₂_rename] at this
167       have he : z ∘ Sum.inl = y := by rfl
168       rw [he] at this
169       simpa [MvPolynomial.aeval_def] using this
170     have hξ : coordinateCovector ξ ∈ affineConormalSpace y I := by
171       rw [affineConormalSpace_eq_equationCovectorSpan]
172       have heq : coordinateCovector ξ =
173           ∑ a : ι, z (.inr a) • differentialCovector y (equations a) := by
174         ext v
175         simp [ξ, coordinateCovector, differentialCovector,
176           Finset.sum_mul, Finset.mul_sum]
177         rw [Finset.sum_comm]
178         simp [mul_assoc]
179       rw [heq]
180       apply Submodule.sum_mem
181       intro a ha
182       apply Submodule.smul_mem
183       apply Submodule.subset_span
184       exact ⟨a.1, rfl⟩
185     have heval := eval₂_finiteGradientSubstitution equations P z
186     have hzero := hP (Sum.elim y ξ) ⟨hy, hξ⟩
187     simpa [ρ, y, ξ, MvPolynomial.aeval_def] using heval.trans hzero
188   letI : Finite ι := Finite.of_fintype ι
189   have hrad : ρ P ∈ J.radical := by
190     rw [← MvPolynomial.vanishingIdeal_zeroLocus_eq_radical (K := k) J]
191     exact hvan
192   obtain ⟨m, hm⟩ := (Ideal.mem_radical_iff.mp hrad)
193   let z : Fin n ⊕ ι → K := Sum.elim (fun i => q (.inl i)) (fun a => c a.1)
194   have hzJ : z ∈ MvPolynomial.zeroLocus K J := by
195     change z ∈ MvPolynomial.zeroLocus K
196       (I.map (MvPolynomial.rename Sum.inl))
197     rw [Ideal.map, MvPolynomial.zeroLocus_span]
198     rintro _ ⟨f, hf, rfl⟩
199     change MvPolynomial.eval₂ (algebraMap k K) z
200       (MvPolynomial.rename Sum.inl f) = 0
201     rw [MvPolynomial.eval₂_rename]
202     have he : z ∘ Sum.inl = fun i => q (.inl i) := by
203       funext i
204       rfl
205     rw [he]
206     exact ((mem_equationConormalLocus_map_iff I q).mp hq).1 f hf
207   have hpw : (MvPolynomial.eval₂ (algebraMap k K) z (ρ P)) ^ m = 0 := by
208     simpa only [MvPolynomial.aeval_def, map_pow] using hzJ _ hm
209   have hrho : MvPolynomial.eval₂ (algebraMap k K) z (ρ P) = 0 :=
210     eq_zero_of_pow_eq_zero hpw
211   rw [eval₂_finiteGradientSubstitution equations P z] at hrho
212   have hpoint :
213       Sum.elim (fun i => z (.inl i))
214         (fun i => ∑ a : ι, z (.inr a) *
215           MvPolynomial.eval₂ (algebraMap k K) (fun j => z (.inl j))
216             (MvPolynomial.pderiv i (equations a))) = q := by
217     funext v
218     rcases v with i | i
219     · rfl
220     · have hi := (hc i).symm
221       rw [← Finset.sum_attach] at hi
222       simpa [z, equations, ι, Finsupp.sum] using hi
223   rw [hpoint] at hrho
224   exact hrho
