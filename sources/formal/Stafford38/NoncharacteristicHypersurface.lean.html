9 /-!
10 # Finiteness of the principal-symbol hypersurface on the hyperplane
11 
12 The normal principal symbol is monic in the normal covariable.  After the
13 normal base coordinate is set to zero, its hypersurface coordinate ring is
14 finite over the tangential cotangent-coordinate ring.
15 -/

132 /-- The literal ring `R/(P,x₀)` for the canonical principal symbol is finite
133  over the coordinate ring of `T^*H`. -/
134 theorem canonical_principal_hypersurface_finite
135     {n N : ℕ} {d : WeylAlg k (n + 1)}
136     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
137     Module.Finite (oldTangentialCoeffRing (k := k) n)
138       (SymbolRing k (n + 1) ⧸
139         canonicalPrincipalHypersurfaceIdeal (k := k) n N d) := by
140   let T := oldTangentialCoeffRing (k := k) n
141   let U := Polynomial T
142   let A := Polynomial U
143   let R := SymbolRing k (n + 1)
144   let P := presentedPrincipalComponent k (@orderWeight (n + 1)) N d
145   let K := canonicalPrincipalHypersurfaceIdeal (k := k) n N d
146   let E := R ⧸ K
147   let e := normalMomentumPolynomialEquiv (k := k) n
148   let fA : A →+* E := (Ideal.Quotient.mk K).comp e.symm
149   let g : A := Polynomial.map
150     (normalCoefficientRingEquiv (k := k) n).toRingHom
151     (canonicalNormalPolynomial (k := k) (n := n) (N := N) d)
152   have hg : g.Monic := by
153     exact Polynomial.Monic.map _
154       (canonicalNormalPolynomial_monic (k := k) (n := n) (N := N)
155         (d := d) hd)
156   have heP : e P = g := by
157     change Polynomial.map (normalCoefficientRingEquiv (k := k) n).toRingHom
158       (normalSymbolAlgEquiv (k := k) n P) = _
159     rfl
160   have hxK : (MvPolynomial.X (.inl (0 : Fin (n + 1))) : R) ∈ K := by
161     exact Ideal.mem_sup_right (Ideal.subset_span (by simp [K]))
162   have hPK : P ∈ K := by
163     exact Ideal.mem_sup_left (Ideal.subset_span (by simp [K, P]))
164   letI : Module A E := Module.compHom E fA
165   let fU : U →+* E := fA.comp Polynomial.C
166   letI : Module U E := Module.compHom E fU
167   have hdouble (t : T) :
168       e.symm (Polynomial.C (Polynomial.C t)) =
169         tangentialSymbolMap (k := k) n t := by
170     have h := congrArg (fun f : T →+* A => f t)
171       (normalMomentumPolynomialEquiv_tangentialMap (k := k) n)
172     change e (tangentialSymbolMap (k := k) n t) =
173       Polynomial.C (Polynomial.C t) at h
174     rw [← h, e.symm_apply_apply]
175   have hT : fU.comp Polynomial.C =
176       (Ideal.Quotient.mk K).comp (tangentialSymbolMap (k := k) n) := by
177     apply RingHom.ext
178     intro t
179     change Ideal.Quotient.mk K (e.symm (Polynomial.C (Polynomial.C t))) = _
180     rw [hdouble t]
181     simp only [RingHom.comp_apply]
182   have hT_apply (t : T) : fU (Polynomial.C t) =
183       Ideal.Quotient.mk K (tangentialSymbolMap (k := k) n t) := by
184     have h := congrArg (fun f : T →+* E => f t) hT
185     exact h
186   letI : Module T E :=
187     restrictedSymbolRing_tangentialModule (k := k) n K
188   letI : IsScalarTower U A E := ⟨by
189     intro u a z
190     rw [Polynomial.smul_eq_C_mul]
191     change fA (Polynomial.C u * a) * z =
192       fU u * (fA a * z)
193     rw [map_mul]
194     exact mul_assoc _ _ _⟩
195   letI : IsScalarTower T U E := ⟨by
196     intro t u z
197     rw [Polynomial.smul_eq_C_mul]
198     change fU (Polynomial.C t * u) * z =
199       (Ideal.Quotient.mk K (tangentialSymbolMap (k := k) n t)) *
200         (fU u * z)
201     rw [map_mul]
202     rw [hT_apply t]
203     exact mul_assoc _ _ _⟩
204   have hfiniteA : Module.Finite A E := by
205     let cyclic : A →ₗ[A] E := LinearMap.toSpanSingleton A E (1 : E)
206     have hsurj : Function.Surjective cyclic := by
207       intro z
208       obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective (I := K) z
209       refine ⟨e r, ?_⟩
210       change Ideal.Quotient.mk K (e.symm (e r) * 1) = Ideal.Quotient.mk K r
211       rw [e.symm_apply_apply, mul_one]
212     exact Module.Finite.of_surjective cyclic hsurj
213   have hkill : ∀ z : E, g • z = 0 := by
214     intro z
215     induction z using Submodule.Quotient.induction_on with
216     | _ r =>
217         change Ideal.Quotient.mk K (e.symm g * r) = 0
218         rw [← heP, e.symm_apply_apply]
219         apply Ideal.Quotient.eq_zero_iff_mem.mpr
220         exact Ideal.mul_mem_right r K hPK
221   have hfiniteU : Module.Finite U E :=
222     AlgebraicAnalysis.MonicAnnihilatorFinite.finite_of_monic_annihilator
223       g hg hkill
224   have hXkill : ∀ z : E, (Polynomial.X : U) • z = 0 := by
225     intro z
226     induction z using Submodule.Quotient.induction_on with
227     | _ r =>
228         change Ideal.Quotient.mk K
229           (e.symm (Polynomial.C (Polynomial.X : U)) * r) = 0
230         rw [← normalMomentumPolynomialEquiv_normalCoordinate (k := k) n,
231           e.symm_apply_apply]
232         apply Ideal.Quotient.eq_zero_iff_mem.mpr
233         exact Ideal.mul_mem_right r K hxK
234   exact AlgebraicAnalysis.MonicAnnihilatorFinite.finite_of_variable_annihilates
235     hXkill
