12 # The Weyl action on a polynomial localization
13 
14 This file constructs the Weyl action on the genuine finite-order differential
15 operators of a localization of a polynomial ring.  No identification of the
16 resulting differential-operator algebra with another presentation is used.
17 -/

38 abbrev A := MvPolynomial (Fin n) k
39 abbrev D (B : Type u) [CommRing B] [Algebra k B] :=
40   algebra (k := k) (R := B)
41 
42 def coordinateEnd (i : Fin n) : End (k := k) (R := B) :=
43   multiplication (algebraMap (A (k := k) (n := n)) B (MvPolynomial.X i))
44 
45 def momentumEnd (i : Fin n) : End (k := k) (R := B) :=
46   (localizedPderiv S B i).toLinearMap
47 
48 theorem coordinateEnd_mem_order_zero (i : Fin n) :
49     coordinateEnd (B := B) i ∈ order (k := k) (R := B) 0 := by
50   rw [mem_order_zero_iff_eq_multiplication]
51   ext x
52   simp [coordinateEnd, multiplication_apply]
53 
54 theorem momentumEnd_mem_order_one (i : Fin n) :
55     momentumEnd S B i ∈ order (k := k) (R := B) 1 := by
56   rw [show 1 = 0 + 1 by rfl, mem_order_succ_iff]
57   intro a
58   rw [mem_order_zero_iff_eq_multiplication]
59   apply LinearMap.ext
60   intro x
61   change localizedPderiv S B i (a * x) - a * localizedPderiv S B i x = _
62   rw [(localizedPderiv S B i).leibniz]
63   simp [DifferentialOperators.commutator_apply, multiplication_apply,
64     momentumEnd, mul_comm, add_comm]
65 
66 def differentialGenerator : (Fin n ⊕ Fin n) → D (k := k) B
67   | .inl i => ⟨coordinateEnd (B := B) i, 0,
68       coordinateEnd_mem_order_zero (B := B) i⟩
69   | .inr i => ⟨momentumEnd S B i, 1, momentumEnd_mem_order_one S B i⟩
70 
71 theorem differentialGenerator_commutator (i j : Fin n ⊕ Fin n) :
72     AlgebraicAnalysis.ringCommutator (differentialGenerator S B i)
73         (differentialGenerator S B j) =
74       algebraMap k (D (k := k) B) (Matrix.J (Fin n) k i j) := by
75   apply Subtype.ext
76   apply LinearMap.ext
77   intro f
78   cases i with
79   | inl i =>
80       cases j with
81       | inl j => simp [AlgebraicAnalysis.ringCommutator, differentialGenerator,
82           coordinateEnd, Module.End.mul_apply, Matrix.J, mul_comm, mul_assoc,
83           mul_left_comm]
84       | inr j =>
85           by_cases h : i = j
86           · subst j
87             simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd,
88               momentumEnd, Module.End.mul_apply, Matrix.J,
89               localizedPderiv_apply_algebraMap_X]
90           · simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd,
91               momentumEnd, Module.End.mul_apply, Matrix.J, h,
92               Ne.symm h, localizedPderiv_apply_algebraMap_X]
93   | inr i =>
94       cases j with
95       | inl j =>
96           by_cases h : i = j
97           · subst j
98             simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd,
99               momentumEnd, Module.End.mul_apply, Matrix.J,
100               localizedPderiv_apply_algebraMap_X]
101           · simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd,
102               momentumEnd, differentialGenerator, coordinateEnd,
103               Matrix.J, h, localizedPderiv_apply_algebraMap_X]
104       | inr j =>
105           have hc := localizedPderiv_comm S B i j
106           have hcf := LinearMap.congr_fun hc f
107           simpa [AlgebraicAnalysis.ringCommutator, differentialGenerator, momentumEnd,
108             Module.End.mul_apply, Matrix.J] using (sub_eq_zero.mpr hcf)
109 
110 def localizedWeylAction :
111     WeylAlg k n →ₐ[k] D (k := k) B :=
112   freeWeylLift (Matrix.J (Fin n) k) (differentialGenerator S B)
113     (differentialGenerator_commutator S B)
114 
115 @[simp] theorem localizedWeylAction_generator (i : Fin n ⊕ Fin n) :
116     localizedWeylAction S B
117         (freeWeylGenerator (Matrix.J (Fin n) k) i) =
118       differentialGenerator S B i :=
119   freeWeylLift_generator _ _ _ i

121 theorem localizedWeylAction_mem (a : WeylAlg k n) :
122     (localizedWeylAction S B a : End (k := k) (R := B)) ∈ D (k := k) B :=
123   (localizedWeylAction S B a).property

126 theorem multiplication_algebraMap_mem_range (f : A (k := k) (n := n)) :
127     ∃ P : D (k := k) B, (P : End (k := k) (R := B)) =
128       multiplication (algebraMap (A (k := k) (n := n)) B f) := by
129   induction f using MvPolynomial.induction_on with
130   | C c =>
131       refine ⟨algebraMap k (D (k := k) B) c, ?_⟩
132       ext x
133       have hc : algebraMap (A (k := k) (n := n)) B (MvPolynomial.C c) =
134           algebraMap k B c := by
135         calc
136           _ = algebraMap (A (k := k) (n := n)) B
137               (algebraMap k (A (k := k) (n := n)) c) := by
138                 rw [MvPolynomial.algebraMap_eq]
139           _ = algebraMap k B c := by
140             exact (IsScalarTower.algebraMap_apply k
141               (A (k := k) (n := n)) B c).symm
142       rw [hc]
143       simp [multiplication_apply, Algebra.smul_def]
144   | add f g hf hg =>
145       obtain ⟨P, hP⟩ := hf
146       obtain ⟨Q, hQ⟩ := hg
147       refine ⟨P + Q, ?_⟩
148       ext x
149       simp [hP, hQ, multiplication_apply, map_add, add_mul]
150   | mul_X f i hf =>
151       obtain ⟨P, hP⟩ := hf
152       refine ⟨(differentialGenerator S B (.inl i)) * P, ?_⟩
153       ext x
154       change algebraMap (A (k := k) (n := n)) B (MvPolynomial.X i) *
155           (P : End (k := k) (R := B)) x = _
156       rw [hP]
157       simp [multiplication_apply, map_mul, mul_assoc]
158       ring
