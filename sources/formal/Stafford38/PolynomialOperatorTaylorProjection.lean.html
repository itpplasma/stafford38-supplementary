25 abbrev PolynomialRing := MvPolynomial (Fin n) k
26 abbrev Operator := Module.End k (PolynomialRing k n)
27 abbrev OperatorEnd := Module.End k (Operator k n)

29 /-- The actual finite Taylor projection in coordinate `i`. -/
30 def projection (i : Fin n) (m : ℕ) (P : Operator k n) : Operator k n :=
31   ∑ j ∈ Finset.range (m + 1),
32     ((-1 : k) ^ j / (j.factorial : k)) •
33       ((coordinateCommutator k n i ^ j) P * momentumEnd k n i ^ j)
34 
35 private theorem coordinateCommutator_mul (i : Fin n) (P Q : Operator k n) :
36     coordinateCommutator k n i (P * Q) =
37       coordinateCommutator k n i P * Q + P * coordinateCommutator k n i Q := by
38   apply LinearMap.ext
39   intro f
40   simp [coordinateCommutator, Stafford38.DifferentialOperators.commutator_apply,
41     Module.End.mul_apply]
42 
43 @[simp] theorem coordinateCommutator_momentum (i : Fin n) :
44     coordinateCommutator k n i (momentumEnd k n i) = 1 := by
45   apply LinearMap.ext
46   intro f
47   simp [coordinateCommutator, Stafford38.DifferentialOperators.commutator_apply,
48     momentumEnd, Module.End.mul_apply]
49 
50 private theorem coordinateCommutator_momentum_pow (i : Fin n) : ∀ j : ℕ,
51     coordinateCommutator k n i (momentumEnd k n i ^ j) =
52       (j : k) • momentumEnd k n i ^ (j - 1)
53   | 0 => by
54       apply LinearMap.ext
55       intro f
56       simp [coordinateCommutator, Stafford38.DifferentialOperators.commutator_apply]
57   | j + 1 => by
58       rw [pow_succ, coordinateCommutator_mul, coordinateCommutator_momentum]
59       rw [coordinateCommutator_momentum_pow]
60       cases j with
61       | zero => simp
62       | succ j => simp [pow_succ, add_mul, smul_mul_assoc, add_smul]
63 
64 private theorem iterate_succ_apply (T : OperatorEnd k n) (j : ℕ) (P : Operator k n) :
65     (T ^ (j + 1)) P = T ((T ^ j) P) := by
66   rw [pow_succ']
67   rfl

69 /-- Applying the coordinate commutator to the Taylor projection kills all
70 adjacent terms; the last term vanishes by the stated nilpotence hypothesis. -/
71 theorem coordinateCommutator_projection_eq_zero (i : Fin n) (m : ℕ)
72     (P : Operator k n)
73     (hnil : (coordinateCommutator k n i ^ (m + 1)) P = 0) :
74     coordinateCommutator k n i (projection k n i m P) = 0 := by
75   simp only [projection, map_sum]
76   have hterm (j : ℕ) :
77       coordinateCommutator k n i
78           (((-1 : k) ^ j / (j.factorial : k)) •
79             ((coordinateCommutator k n i ^ j) P * momentumEnd k n i ^ j)) =
80         ((-1 : k) ^ j / (j.factorial : k)) •
81           (((coordinateCommutator k n i ^ (j + 1)) P * momentumEnd k n i ^ j) +
82             ((j : k) •
83               ((coordinateCommutator k n i ^ j) P *
84                 momentumEnd k n i ^ (j - 1)))) := by
85     rw [map_smul, coordinateCommutator_mul,
86       coordinateCommutator_momentum_pow, iterate_succ_apply]
87     simp only [smul_add]
88     rw [mul_smul_comm, smul_smul]
89   simp_rw [hterm]
90   simp only [smul_add, Finset.sum_add_distrib]
91   rw [Finset.sum_range_succ]
92   rw [hnil]
93   simp only [zero_mul, smul_zero, add_zero]
94   -- Shift the derivative-of-the-right-factor sum by one.  Its coefficient is
95   -- the negative of the next term in the first sum.
96   rw [Finset.sum_range_succ']
97   simp only [Nat.cast_zero, zero_smul, zero_add]
98   have hcancel (j : ℕ) :
99       ((-1 : k) ^ j / (j.factorial : k)) •
100           ((coordinateCommutator k n i ^ (j + 1)) P * momentumEnd k n i ^ j) +
101         ((-1 : k) ^ (j + 1) / ((j + 1).factorial : k)) •
102           ((j + 1 : k) •
103             ((coordinateCommutator k n i ^ (j + 1)) P *
104               momentumEnd k n i ^ j)) = 0 := by
105     have hj : ((j + 1 : ℕ) : k) ≠ 0 := by exact_mod_cast Nat.succ_ne_zero j
106     have hfac : ((j.factorial : ℕ) : k) ≠ 0 := by
107       exact_mod_cast Nat.factorial_ne_zero j
108     have hc :
109         ((-1 : k) ^ j / (j.factorial : k)) +
110           ((-1 : k) ^ (j + 1) / ((j + 1).factorial : k)) * (j + 1 : k) = 0 := by
111       rw [Nat.factorial_succ, Nat.cast_mul, Nat.cast_succ]
112       field_simp [hfac, hj]
113       ring
114     rw [smul_smul, ← add_smul, hc, zero_smul]
115   -- `sum_range_succ` exposes exactly the paired adjacent coefficients.
116   simp only [smul_zero, add_zero]
117   rw [← Finset.sum_add_distrib]
118   apply Finset.sum_eq_zero
119   intro j hj
120   simpa [Nat.succ_sub_one] using hcancel j
121 
122 end
123 end Stafford38.PolynomialOperatorTaylorProjection

143 private def rightMomentum (i : Fin n) :
144     Operator k n →ₗ[k] Operator k n :=
145   LinearMap.mulRight k (momentumEnd k n i)
146 
147 private theorem rightMomentum_pow_apply (i : Fin n) (j : ℕ) (P : Operator k n) :
148     (rightMomentum k n i ^ j) P = P * momentumEnd k n i ^ j := by
149   induction j with
150   | zero => simp [rightMomentum]
151   | succ j ih =>
152       rw [pow_succ', Module.End.mul_apply, ih]
153       simp [rightMomentum, pow_succ, mul_assoc]
154 
155 private theorem projectorMapG_apply_eq_projection (i : Fin n) (m : ℕ)
156     (P : Operator k n) :
157     projectorMapG m (rightMomentum k n i) (coordinateCommutator k n i) P =
158       projection k n i m P := by
159   unfold projectorMapG projection
160   rw [LinearMap.sum_apply]
161   apply Finset.sum_congr rfl
162   intro j hj
163   change ((-1 : k) ^ j / (j.factorial : k)) •
164       ((rightMomentum k n i ^ j * coordinateCommutator k n i ^ j) P) = _
165   rw [Module.End.mul_apply, rightMomentum_pow_apply]

167 /-- Finite Taylor reconstruction for the actual polynomial operator.  Every
168 power of the partial derivative occurs as a right factor. -/
169 theorem reconstruction (i : Fin n) (m : ℕ) (P : Operator k n)
170     (hnil : (coordinateCommutator k n i ^ (m + 1)) P = 0) :
171     P = ∑ a ∈ Finset.range (m + 1),
172       ((1 : k) / (a.factorial : k)) •
173         (projection k n i m ((coordinateCommutator k n i ^ a) P) *
174           momentumEnd k n i ^ a) := by
175   have h := reconstruction_all m (rightMomentum k n i)
176     (coordinateCommutator k n i) P hnil
177   refine h.trans ?_
178   apply Finset.sum_congr rfl
179   intro a ha
180   congr 1
181   rw [Module.End.mul_apply, Module.End.mul_apply, rightMomentum_pow_apply,
182     projectorMapG_apply_eq_projection]
183 
184 end
185 end Stafford38.PolynomialOperatorTaylorProjection
