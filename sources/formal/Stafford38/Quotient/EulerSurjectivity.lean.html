9 /-!
10 # Euler-normality surjectivity on a right quotient
11 
12 This file isolates the quotient argument used in the Stafford proof. A right
13 ideal is a submodule of the regular right module, encoded as an
14 `Aᵐᵒᵖ`-submodule of `A`.
15 -/
16 
17 namespace Stafford38.EulerSurjectivity
18 
19 variable {A : Type*} [Ring A]
20 
21 /-- A right ideal, represented as a submodule of the regular right module. -/
22 abbrev RightIdeal (A : Type*) [Ring A] := Submodule Aᵐᵒᵖ A
23 
24 /-- The additive quotient by a right ideal, with its induced right action. -/
25 abbrev RightQuotient (I : RightIdeal A) := A ⧸ I
26 
27 /-- The quotient class of a ring element. -/
28 def qmk (I : RightIdeal A) (a : A) : RightQuotient I :=
29   Submodule.Quotient.mk a
30 
31 /-- Right multiplication on the quotient. -/
32 def rightMul (I : RightIdeal A) (a : A) (q : RightQuotient I) :
33     RightQuotient I :=
34   MulOpposite.op a • q
35 
36 lemma qmk_right_mul (I : RightIdeal A) (a b : A) :
37     qmk I (a * b) = rightMul I b (qmk I a) := by
38   exact Submodule.Quotient.mk_smul I (MulOpposite.op b) a
39 
40 /-- Elements whose quotient classes have representatives in `R`. -/
41 def ReducesTo (I : RightIdeal A) (R : Subring A) (a : A) : Prop :=
42   ∃ r : R, qmk I a = qmk I (r : A)
43 
44 /-- Right multipliers preserving the classes represented by `R`. -/
45 def preservers (I : RightIdeal A) (R : Subring A) : Subring A where
46   carrier := {a | ∀ r : R, ReducesTo I R ((r : A) * a)}
47   zero_mem' := by
48     intro r
49     exact ⟨0, by simp [ReducesTo, qmk]⟩
50   one_mem' := by
51     intro r
52     exact ⟨r, by simp [ReducesTo]⟩
53   add_mem' := by
54     intro a b ha hb r
55     obtain ⟨s, hs⟩ := ha r
56     obtain ⟨t, ht⟩ := hb r
57     refine ⟨s + t, ?_⟩
58     simpa [ReducesTo, mul_add, qmk, Submodule.Quotient.mk_add] using
59       congrArg₂ (· + ·) hs ht
60   neg_mem' := by
61     intro a ha r
62     obtain ⟨s, hs⟩ := ha r
63     refine ⟨-s, ?_⟩
64     simpa [ReducesTo, qmk] using congrArg Neg.neg hs
65   mul_mem' := by
66     intro a b ha hb r
67     obtain ⟨s, hs⟩ := ha r
68     obtain ⟨t, ht⟩ := hb s
69     refine ⟨t, ?_⟩
70     have hright := congrArg (rightMul I b) hs
71     rw [← qmk_right_mul, ← qmk_right_mul] at hright
72     simpa [mul_assoc] using hright.trans ht
73 
74 lemma subring_le_preservers (I : RightIdeal A) (R : Subring A) :
75     R ≤ preservers I R := by
76   intro a ha r
77   exact ⟨r * ⟨a, ha⟩, rfl⟩
78 
79 /--
80 The quotient form of the positive Euler argument.
81 
82 The relation `1 + U*x ∈ I`, normality `x*R ⊆ R*x`, the Euler element
83 `E = x*p ∈ R`, and generation of `A` by `R` and `p` imply that right
84 multiplication by `x` is onto `A/I`.
85 -/
86 theorem rightMul_surjective_of_euler_normal
87     (I : RightIdeal A) (R : Subring A) (x p : A)
88     (hnormal : ∀ r : R, ∃ s : R, x * (r : A) = (s : A) * x)
89     (hE : x * p ∈ R)
90     (hresidue : ∃ U : R, 1 + (U : A) * x ∈ I)
91     (hgenerate : Subring.closure ((R : Set A) ∪ {p}) = ⊤) :
92     Function.Surjective (rightMul I x) := by
93   obtain ⟨U, hU⟩ := hresidue
94 
95   have hp : p ∈ preservers I R := by
96     intro r
97     obtain ⟨s, hs⟩ := hnormal r
98     refine ⟨-(U * s * ⟨x * p, hE⟩), ?_⟩
99     apply (Submodule.Quotient.eq I).2
100     have hmem := I.smul_mem (MulOpposite.op ((r : A) * p)) hU
101     change (1 + (U : A) * x) * ((r : A) * p) ∈ I at hmem
102     convert hmem using 1
103     all_goals simp only [Subring.coe_neg, Subring.coe_mul]
104     calc
105       (r : A) * p - -((U : A) * (s : A) * (x * p)) =
106           (r : A) * p + (U : A) * ((s : A) * x) * p := by noncomm_ring
107       _ = (r : A) * p + (U : A) * (x * (r : A)) * p := by rw [hs]
108       _ = (1 + (U : A) * x) * ((r : A) * p) := by noncomm_ring
109 
110   have hall : preservers I R = ⊤ := by
111     apply top_unique
112     rw [← hgenerate]
113     apply Subring.closure_le.2
114     intro a ha
115     rcases ha with ha | rfl
116     · exact subring_le_preservers I R ha
117     · simpa using hp
118 
119   intro q
120   obtain ⟨a, rfl⟩ := Submodule.Quotient.mk_surjective I q
121   have ha : a ∈ preservers I R := by
122     rw [hall]
123     trivial
124   obtain ⟨r, hr⟩ := ha (1 : R)
125   have har : qmk I a = qmk I (r : A) := by simpa using hr
126   obtain ⟨s, hs⟩ := hnormal r
127   refine ⟨qmk I (-((U : A) * (s : A))), ?_⟩
128   change rightMul I x (qmk I (-((U : A) * (s : A)))) = qmk I a
129   rw [har, ← qmk_right_mul]
130   apply (Submodule.Quotient.eq I).2
131   have hmem := I.smul_mem (MulOpposite.op (r : A)) hU
132   change (1 + (U : A) * x) * (r : A) ∈ I at hmem
133   have hmemneg := I.neg_mem hmem
134   change -((1 + (U : A) * x) * (r : A)) ∈ I at hmemneg
135   convert hmemneg using 1
136   calc
137     (-((U : A) * (s : A))) * x - (r : A) =
138         -((r : A) + (U : A) * ((s : A) * x)) := by noncomm_ring
139     _ = -((r : A) + (U : A) * (x * (r : A))) := by rw [hs]
140     _ = -((1 + (U : A) * x) * (r : A)) := by noncomm_ring
141
