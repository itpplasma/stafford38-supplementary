48 private theorem actionSpan_mul_right_generator
49     (i : Fin n) (a : WeylAlg k n) :
50     weylEnd S B a * (localizedPderiv S B i).toLinearMap ∈ actionSpan S B := by
51   change (localizedWeylAction S B a : Module.End k B) *
52       (differentialGenerator S B (.inr i) : Module.End k B) ∈ actionSpan S B
53   rw [← localizedWeylAction_generator S B (.inr i)]
54   have hm := congrArg (fun q : D (k := k) B =>
55       (q : Module.End k B))
56     ((localizedWeylAction S B).map_mul a
57       (freeWeylGenerator (Matrix.J (Fin n) k) (.inr i)))
58   have hm' : weylEnd S B
59         (a * freeWeylGenerator (Matrix.J (Fin n) k) (.inr i)) =
60       weylEnd S B a * weylEnd S B
61         (freeWeylGenerator (Matrix.J (Fin n) k) (.inr i)) := by
62     simpa [weylEnd] using hm
63   change weylEnd S B a * weylEnd S B
64       (freeWeylGenerator (Matrix.J (Fin n) k) (.inr i)) ∈ actionSpan S B
65   rw [← hm']
66   exact actionSpan_mem S B _

68 private theorem actionSpan_right_stable
69     (i : Fin n) {Q : Module.End k B} (hQ : Q ∈ actionSpan S B) :
70     Q * (localizedPderiv S B i).toLinearMap ∈ actionSpan S B := by
71   refine Submodule.span_induction ?_ ?_ ?_ ?_ hQ
72   · rintro Q ⟨a, rfl⟩
73     exact actionSpan_mul_right_generator S B i a
74   · simp
75   · intro x y hx hy hxr hyr
76     rw [add_mul]
77     exact (actionSpan S B).add_mem hxr hyr
78   · intro c x hx hxr
79     rw [smul_mul_assoc]
80     exact (actionSpan S B).smul_mem c hxr

82 theorem actual_differential_mem_actionSpan
83     (P : Module.End k B)
84     (hP : P ∈ D (k := k) B) : P ∈ actionSpan S B := by
85   refine mem_submodule_of_coordinates
86     (fun i : Fin n => algebraMap (A (k := k) (n := n)) B (MvPolynomial.X i))
87     (fun i => localizedPderiv S B i)
88     (H := (actionSpan S B).restrictScalars k)
89     ?_ ?_ ?_ ?_ P hP
90   · intro i j
91     exact localizedPderiv_apply_algebraMap_X S B i j
92   · intro Q hQ
93     apply eq_multiplication_of_commute_coordinate B S Q
94     intro i b
95     have hi := LinearMap.congr_fun (hQ i) b
96     exact sub_eq_zero.mp (by simpa [commutator_apply] using hi)
97   · intro b
98     rw [← actionSpan_smul_eq_multiplication S B b]
99     exact (actionSpan S B).smul_mem b (actionSpan_mem S B 1)
100   · intro i Q hQ
101     exact actionSpan_right_stable S B i hQ

108 private theorem multiplication_algebraMap_in_action
109     (f : A (k := k) (n := n)) :
110     ∃ a : WeylAlg k n,
111       weylEnd S B a = multiplication (algebraMap (A (k := k) (n := n)) B f) := by
112   induction f using MvPolynomial.induction_on with
113   | C c =>
114       refine ⟨algebraMap k (WeylAlg k n) c, ?_⟩
115       have hc : algebraMap (A (k := k) (n := n)) B (MvPolynomial.C c) =
116           algebraMap k B c := by
117         calc
118           _ = algebraMap (A (k := k) (n := n)) B
119               (algebraMap k (A (k := k) (n := n)) c) := by
120                 rw [MvPolynomial.algebraMap_eq]
121           _ = algebraMap k B c := by
122             exact (IsScalarTower.algebraMap_apply k
123               (A (k := k) (n := n)) B c).symm
124       ext b
125       simp [weylEnd, multiplication_apply, Algebra.smul_def, hc]
126   | add f g hf hg =>
127       obtain ⟨a, ha⟩ := hf
128       obtain ⟨b, hb⟩ := hg
129       refine ⟨a + b, ?_⟩
130       have hadd : weylEnd S B (a + b) = weylEnd S B a + weylEnd S B b := by
131         simp [weylEnd]
132       rw [hadd, ha, hb]
133       ext z
134       simp [multiplication_apply]
135       ring
136   | mul_X f i hf =>
137       obtain ⟨a, ha⟩ := hf
138       let Xweyl := freeWeylGenerator (Matrix.J (Fin n) k) (.inl i)
139       refine ⟨Xweyl * a, ?_⟩
140       have hX : weylEnd S B Xweyl =
141           multiplication (algebraMap (A (k := k) (n := n)) B
142             (MvPolynomial.X i)) := by
143         ext z
144         simp [Xweyl, weylEnd, differentialGenerator, coordinateEnd,
145           multiplication_apply]
146       have hmul := congrArg (fun q : D (k := k) B => (q : Module.End k B))
147         ((localizedWeylAction S B).map_mul Xweyl a)
148       rw [show weylEnd S B (Xweyl * a) = weylEnd S B Xweyl * weylEnd S B a by

155 /-- A common denominator on the left clears every intrinsic differential
156 operator on a localization of a polynomial ring. -/
157 theorem actual_differential_left_denominator_clearing
158     (P : Module.End k B) (hP : P ∈ D (k := k) B) :
159     ∃ s : S, ∃ a : WeylAlg k n,
160       multiplication (algebraMap (A (k := k) (n := n)) B (s : A)) * P =
161         weylEnd S B a := by
162   have hspan := actual_differential_mem_actionSpan S B P hP
163   refine Submodule.span_induction ?_ ?_ ?_ ?_ hspan
164   · rintro Q ⟨a, rfl⟩
165     refine ⟨1, a, ?_⟩
166     ext z
167     simp [multiplication_apply, Module.End.mul_apply]
168   · refine ⟨1, 0, ?_⟩
169     ext z
170     simp [multiplication_apply, weylEnd]
171   · intro Q T hQ hT hcQ hcT
172     obtain ⟨s, a, ha⟩ := hcQ
173     obtain ⟨t, b, hb⟩ := hcT
174     obtain ⟨pt, hpt⟩ := multiplication_algebraMap_in_action S B (t : A)
175     obtain ⟨ps, hps⟩ := multiplication_algebraMap_in_action S B (s : A)
176     refine ⟨s * t, pt * a + ps * b, ?_⟩
177     have hact_mul (c d : WeylAlg k n) :
178         weylEnd S B (c * d) = weylEnd S B c * weylEnd S B d := by
179       have h := congrArg (fun q : D (k := k) B => (q : Module.End k B))
180         ((localizedWeylAction S B).map_mul c d)
181       simpa [weylEnd] using h
182     have hact_add (c d : WeylAlg k n) :
183         weylEnd S B (c + d) = weylEnd S B c + weylEnd S B d := by
184       simp [weylEnd]
185     rw [hact_add, hact_mul, hact_mul, hpt, hps, ← ha, ← hb]
186     ext z
187     simp [multiplication_apply, Module.End.mul_apply, map_mul]
188     ring
189   · intro c Q hQ hcQ
190     obtain ⟨s, a, ha⟩ := hcQ
191     obtain ⟨⟨r, u⟩, hu⟩ := IsLocalization.surj S c
192     obtain ⟨pr, hpr⟩ := multiplication_algebraMap_in_action S B r
193     refine ⟨u * s, pr * a, ?_⟩
194     have hact_mul : weylEnd S B (pr * a) =
195         weylEnd S B pr * weylEnd S B a := by
196       have h := congrArg (fun q : D (k := k) B => (q : Module.End k B))
197         ((localizedWeylAction S B).map_mul pr a)
198       simpa [weylEnd] using h
199     rw [hact_mul, hpr, ← ha]
200     ext z
201     simp only [Module.End.mul_apply, multiplication_apply, LinearMap.smul_apply,
202       Algebra.smul_def, Submonoid.coe_mul, map_mul]
203     change algebraMap (MvPolynomial (Fin n) k) B (u : MvPolynomial (Fin n) k) *
204         algebraMap (MvPolynomial (Fin n) k) B (s : MvPolynomial (Fin n) k) *
205           (c * Q z) =
206       algebraMap (MvPolynomial (Fin n) k) B r *
207         (algebraMap (MvPolynomial (Fin n) k) B
208           (s : MvPolynomial (Fin n) k) * Q z)
209     rw [← hu]
210     ring
