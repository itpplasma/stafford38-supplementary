9 /-!
10 # The distinguished coordinate hyperplane is noncharacteristic
11 
12 For an ideal in the symbol ring, restriction to `x₀ = 0` has coordinate ring
13 `R / (J + (x₀))`.  As a module this is the cokernel of multiplication by
14 `x₀` on `R / J`.  We use that cokernel presentation to express finiteness of
15 the projection to the cotangent space of the hyperplane.
16 -/

35 /-- The coordinate-ring map from the cotangent space of the hyperplane to the
36 ambient symbol ring, omitting both the normal coordinate and normal covariable. -/
37 def tangentialSymbolMap (n : ℕ) :
38     oldTangentialCoeffRing (k := k) n →+* SymbolRing k (n + 1) :=
39   ((tangentialPolynomialActionHom (k := k) n).comp Polynomial.C).comp
40     (oldSymbolTangentialAlgEquiv (k := k) n).toRingHom
41 
42 /-- The natural tangential-coordinate action on the coordinate ring of a
43 closed subset of the cotangent bundle restricted to `x₀ = 0`. -/
44 noncomputable instance restrictedSymbolRing_tangentialModule
45     (n : ℕ) (J : Ideal (SymbolRing k (n + 1))) :
46     Module (oldTangentialCoeffRing (k := k) n)
47       (SymbolRing k (n + 1) ⧸ J) :=
48   Module.compHom (SymbolRing k (n + 1) ⧸ J)
49     ((Ideal.Quotient.mk J).comp (tangentialSymbolMap (k := k) n))

51 /-- Multiplication by the normal base coordinate on the restricted
52 characteristic coordinate ring, viewed as a linear map over the tangential
53 coordinate ring. -/
54 def restrictedCoordinateAction (n : ℕ)
55     (J : Ideal (SymbolRing k (n + 1))) :
56     (SymbolRing k (n + 1) ⧸ J) →ₗ[oldTangentialCoeffRing (k := k) n]
57       (SymbolRing k (n + 1) ⧸ J) where
58   toFun z := (MvPolynomial.X (.inl (0 : Fin (n + 1))) :
59       SymbolRing k (n + 1)) • z
60   map_add' := smul_add _
61   map_smul' := by
62     intro a z
63     change (MvPolynomial.X (.inl (0 : Fin (n + 1))) :
64         SymbolRing k (n + 1)) •
65           (((Ideal.Quotient.mk J) (tangentialSymbolMap (k := k) n a)) • z) =
66       ((Ideal.Quotient.mk J) (tangentialSymbolMap (k := k) n a) •
67         ((MvPolynomial.X (.inl (0 : Fin (n + 1))) :
68           SymbolRing k (n + 1)) • z))
69     exact smul_comm _ _ _

71 /-- `J` is noncharacteristic for the distinguished coordinate hyperplane when
72 the coordinate ring of its restricted zero locus is finite over the
73 coordinate ring of `T^*H`.  The quotient-module presentation here is
74 canonically the ring `R / (J + (x₀))`. -/
75 def IsNoncharacteristic (n : ℕ)
76     (J : Ideal (SymbolRing k (n + 1))) : Prop :=
77   Module.Finite (oldTangentialCoeffRing (k := k) n)
78     ((SymbolRing k (n + 1) ⧸ J) ⧸
79       LinearMap.range (restrictedCoordinateAction (k := k) n J))

81 /-- The coordinate ring obtained by restricting `V(J)` to `x₀ = 0`. -/
82 def restrictedCoordinateIdeal (n : ℕ) (J : Ideal (SymbolRing k (n + 1))) :
83     Ideal (SymbolRing k (n + 1)) :=
84   J ⊔ Ideal.span ({(MvPolynomial.X (.inl (0 : Fin (n + 1))) :
85     SymbolRing k (n + 1))} : Set (SymbolRing k (n + 1)))

87 /-- The tangential-coordinate action on the literal restricted coordinate ring. -/
88 noncomputable instance restrictedCoordinateQuotient_tangentialModule
89     (n : ℕ) (J : Ideal (SymbolRing k (n + 1))) :
90     Module (oldTangentialCoeffRing (k := k) n)
91       (SymbolRing k (n + 1) ⧸ restrictedCoordinateIdeal (k := k) n J) :=
92   Module.compHom _
93     ((Ideal.Quotient.mk (restrictedCoordinateIdeal (k := k) n J)).comp
94       (tangentialSymbolMap (k := k) n))

96 /-- The cokernel presentation maps onto the literal coordinate ring of the
97 restricted zero locus. -/
98 noncomputable def cokerToRestrictedCoordinateRing (n : ℕ)
99     (J : Ideal (SymbolRing k (n + 1))) :
100     ((SymbolRing k (n + 1) ⧸ J) ⧸
101       LinearMap.range (restrictedCoordinateAction (k := k) n J)) →ₗ[
102         oldTangentialCoeffRing (k := k) n]
103       (SymbolRing k (n + 1) ⧸ restrictedCoordinateIdeal (k := k) n J) := by
104   let R := SymbolRing k (n + 1)
105   let T := oldTangentialCoeffRing (k := k) n
106   let K := restrictedCoordinateIdeal (k := k) n J
107   let ψ : R ⧸ J →+* R ⧸ K :=
108     Ideal.Quotient.lift J (Ideal.Quotient.mk K) (by
109       intro a ha
110       apply Ideal.Quotient.eq_zero_iff_mem.mpr
111       exact (le_sup_left : J ≤ K) ha)
112   let g : R ⧸ J →ₗ[T] R ⧸ K :=
113     { toFun := ψ
114       map_add' := ψ.map_add
115       map_smul' := by
116         intro a z
117         obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective (I := J) z
118         change ψ ((Ideal.Quotient.mk J (tangentialSymbolMap (k := k) n a)) *
119             Ideal.Quotient.mk J r) =
120           (Ideal.Quotient.mk K (tangentialSymbolMap (k := k) n a)) *
121             ψ (Ideal.Quotient.mk J r)
122         rw [map_mul]
123         simp [ψ] }
124   have hkill : (LinearMap.range (restrictedCoordinateAction (k := k) n J)) ≤ g.ker := by
125     rintro z ⟨w, rfl⟩
126     rw [LinearMap.mem_ker]
127     obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective (I := J) w
128     change ψ ((MvPolynomial.X (.inl (0 : Fin (n + 1))) : R) •
129         Ideal.Quotient.mk J r) = 0
130     change ψ (Ideal.Quotient.mk J
131       ((MvPolynomial.X (.inl (0 : Fin (n + 1))) : R) * r)) = 0
132     change Ideal.Quotient.mk K
133       ((MvPolynomial.X (.inl (0 : Fin (n + 1))) : R) * r) = 0
134     apply Ideal.Quotient.eq_zero_iff_mem.mpr
135     exact (le_sup_right : Ideal.span ({_} : Set R) ≤ K)
136       (Ideal.mul_mem_right r _ (Ideal.subset_span (by simp)))
137   exact Submodule.liftQ _ g hkill

139 /-- The map from the cokernel presentation sends the class of `r` to its class
140 in the literal restricted coordinate ring. -/
141 theorem cokerToRestrictedCoordinateRing_mk (n : ℕ)
142     (J : Ideal (SymbolRing k (n + 1))) :
143     ∀ r : SymbolRing k (n + 1),
144       cokerToRestrictedCoordinateRing (k := k) n J
145         (Submodule.Quotient.mk (Ideal.Quotient.mk J r)) =
146       Ideal.Quotient.mk (restrictedCoordinateIdeal (k := k) n J) r := by
147   intro r
148   let R := SymbolRing k (n + 1)
149   let K := restrictedCoordinateIdeal (k := k) n J
150   have hJ : ∀ a ∈ J, (Ideal.Quotient.mk K) a = 0 := by
151     intro a ha
152     apply Ideal.Quotient.eq_zero_iff_mem.mpr
153     exact (le_sup_left : J ≤ K) ha
154   unfold cokerToRestrictedCoordinateRing
155   rw [Submodule.liftQ_apply]
156   change (Ideal.Quotient.lift J (Ideal.Quotient.mk K) hJ)
157       (Ideal.Quotient.mk J r) = Ideal.Quotient.mk K r
158   rw [Ideal.Quotient.lift_mk]

160 /-- Every point of the literal restricted coordinate ring has a representative
161 in the cokernel presentation. -/
162 theorem cokerToRestrictedCoordinateRing_surjective (n : ℕ)
163     (J : Ideal (SymbolRing k (n + 1))) :
164     Function.Surjective (cokerToRestrictedCoordinateRing (k := k) n J) := by
165   intro z
166   obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective
167     (I := restrictedCoordinateIdeal (k := k) n J) z
168   exact ⟨Submodule.Quotient.mk (Ideal.Quotient.mk J r),
169     cokerToRestrictedCoordinateRing_mk (k := k) n J r⟩

171 /-- The only classes killed by restriction to `x₀ = 0` are the classes
172 generated by multiplication by `x₀`. -/
173 theorem cokerToRestrictedCoordinateRing_injective (n : ℕ)
174     (J : Ideal (SymbolRing k (n + 1))) :
175     Function.Injective (cokerToRestrictedCoordinateRing (k := k) n J) := by
176   let R := SymbolRing k (n + 1)
177   let T := oldTangentialCoeffRing (k := k) n
178   let K := restrictedCoordinateIdeal (k := k) n J
179   let x : R := MvPolynomial.X (.inl (0 : Fin (n + 1)))
180   let f := restrictedCoordinateAction (k := k) n J
181   intro z₁ z₂ heq
182   have hzero : cokerToRestrictedCoordinateRing (k := k) n J (z₁ - z₂) = 0 := by
183     rw [map_sub, heq, sub_self]
184   have hz : z₁ - z₂ = 0 := by
185     obtain ⟨q, hq⟩ := Submodule.mkQ_surjective f.range (z₁ - z₂)
186     obtain ⟨r, rfl⟩ := Ideal.Quotient.mk_surjective (I := J) q
187     have hmapzero : cokerToRestrictedCoordinateRing (k := k) n J
188         (Submodule.Quotient.mk (Ideal.Quotient.mk J r)) = 0 := by
189       calc
190         _ = cokerToRestrictedCoordinateRing (k := k) n J (z₁ - z₂) :=
191           congrArg (cokerToRestrictedCoordinateRing (k := k) n J) hq
192         _ = 0 := hzero
193     have htarget : Ideal.Quotient.mk K r = 0 := by
194       rw [← cokerToRestrictedCoordinateRing_mk (k := k) n J r]
195       exact hmapzero
196     have hrK : r ∈ K := Ideal.Quotient.eq_zero_iff_mem.mp htarget
197     rcases Submodule.mem_sup.mp hrK with ⟨j, hj, s, hs, hjs⟩
198     rcases Ideal.mem_span_singleton'.mp hs with ⟨a, has⟩
199     have hrange : (Ideal.Quotient.mk J r) ∈ f.range := by
200       refine ⟨Ideal.Quotient.mk J a, ?_⟩
201       change Ideal.Quotient.mk J (x * a) = Ideal.Quotient.mk J r
202       rw [← hjs, map_add, Ideal.Quotient.eq_zero_iff_mem.mpr hj, zero_add]
203       rw [← has, mul_comm]
204     have := (Submodule.Quotient.mk_eq_zero f.range).mpr hrange
205     exact hq.symm.trans this
206   exact sub_eq_zero.mp hz

208 /-- The cokernel presentation and the literal restricted coordinate ring are
209 canonically linearly equivalent over the hyperplane cotangent coordinates. -/
210 noncomputable def cokerToRestrictedCoordinateRingEquiv (n : ℕ)
211     (J : Ideal (SymbolRing k (n + 1))) :
212     ((SymbolRing k (n + 1) ⧸ J) ⧸
213       LinearMap.range (restrictedCoordinateAction (k := k) n J)) ≃ₗ[
214         oldTangentialCoeffRing (k := k) n]
215       (SymbolRing k (n + 1) ⧸ restrictedCoordinateIdeal (k := k) n J) :=
216   LinearEquiv.ofBijective (cokerToRestrictedCoordinateRing (k := k) n J)
217     ⟨cokerToRestrictedCoordinateRing_injective (k := k) n J,
218       cokerToRestrictedCoordinateRing_surjective (k := k) n J⟩

220 /-- The cokernel definition is equivalent to finiteness of the literal
221 restricted coordinate ring `R/(J+(x₀))`. -/
222 theorem isNoncharacteristic_iff_finite_restrictedCoordinateQuotient (n : ℕ)
223     (J : Ideal (SymbolRing k (n + 1))) :
224     IsNoncharacteristic (k := k) n J ↔
225       Module.Finite (oldTangentialCoeffRing (k := k) n)
226         (SymbolRing k (n + 1) ⧸ restrictedCoordinateIdeal (k := k) n J) := by
227   unfold IsNoncharacteristic
228   let e := cokerToRestrictedCoordinateRingEquiv (k := k) n J
229   constructor
230   · intro h
231     exact Module.Finite.of_surjective e.toLinearMap e.surjective
232   · intro h
233     exact Module.Finite.of_surjective e.symm.toLinearMap e.symm.surjective

235 /-- For the canonical characteristic ideal, restriction to `x₀ = 0` is finite
236 over `T^*H`. -/
237 theorem canonical_isNoncharacteristic_annihilator
238     {n N : ℕ} {d : WeylAlg k (n + 1)}
239     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
240     IsNoncharacteristic (k := k) n
241       (Module.annihilator (SymbolRing k (n + 1))
242         (OrderAssociatedGradedModule k
243           (canonicalRightIdeal (presentedCoordinate k n) d N))) := by
244   let I := canonicalRightIdeal (presentedCoordinate k n) d N
245   let E := OrderAssociatedGradedModule k I
246   let R := SymbolRing k (n + 1)
247   let T := oldTangentialCoeffRing (k := k) n
248   let J := Module.annihilator R E
249   let Q := R ⧸ J
250   let x : R := MvPolynomial.X (.inl (0 : Fin (n + 1)))
251   let fE := oldCoordinateMap (k := k) n E
252   let fQ := restrictedCoordinateAction (k := k) n J
253   let e0 := orderAssociatedGradedLinearEquivCharacteristic k I
254   have hAnn : J = orderInitialIdeal k I := by
255     dsimp [J]
256     exact annihilator_orderAssociatedGradedModule k I
257   let eQ : E ≃ₗ[R] Q :=
258     e0.trans (Ideal.quotientEquivAlgOfEq R hAnn.symm).toLinearEquiv
259   letI : Module T E := oldCoeffModule (k := k) n E
260   let eT : E ≃ₗ[T] Q :=
261     { toFun := eQ
262       invFun := eQ.symm
263       left_inv := eQ.left_inv
264       right_inv := eQ.right_inv
265       map_add' := eQ.map_add
266       map_smul' := by
267         intro a z
268         change eQ ((tangentialSymbolMap (k := k) n a) • z) =
269           ((Ideal.Quotient.mk J) (tangentialSymbolMap (k := k) n a)) • eQ z
270         exact eQ.map_smul _ _ }
271   let hfinite := canonical_finite_old_coordinate_kernel_cokernel
272     (k := k) (n := n) (N := N) (d := d) hd
273   have hfiniteCoker : Module.Finite T (E ⧸ fE.range) := by
274     simpa only [fE] using hfinite.2
275   let F : E →ₗ[T] Q ⧸ fQ.range :=
276     { toFun := fun z => Submodule.Quotient.mk (eT z)
277       map_add' := by intro z w; simp [eT.map_add]
278       map_smul' := by
279         intro a z
280         change Submodule.Quotient.mk (eT (a • z)) =
281           a • Submodule.Quotient.mk (eT z)
282         rw [eT.map_smul]
283         exact (Submodule.mkQ fQ.range).map_smul a (eT z) }
284   have hkill : fE.range ≤ F.ker := by
285     rintro z ⟨w, rfl⟩
286     rw [LinearMap.mem_ker]
287     change Submodule.Quotient.mk (eT (fE w)) = 0
288     rw [oldCoordinateMap_apply]
289     change Submodule.Quotient.mk (eT (x • w)) = 0
290     change Submodule.Quotient.mk (eQ (x • w)) = 0
291     rw [eQ.map_smul]
292     rw [Submodule.Quotient.mk_eq_zero]
293     change fQ (eQ w) ∈ fQ.range
294     exact ⟨eQ w, rfl⟩
295   let Fbar : E ⧸ fE.range →ₗ[T] Q ⧸ fQ.range :=
296     Submodule.liftQ fE.range F hkill
297   have hsurj : Function.Surjective Fbar := by
298     intro z
299     obtain ⟨q, rfl⟩ := Submodule.mkQ_surjective fQ.range z
300     obtain ⟨w, hw⟩ := eQ.surjective q
301     refine ⟨Submodule.Quotient.mk w, ?_⟩
302     change Fbar (Submodule.Quotient.mk w) = Submodule.Quotient.mk q
303     dsimp [Fbar]
304     change Submodule.Quotient.mk (eT w) = Submodule.Quotient.mk q
305     simpa [eT] using
306       congrArg (fun q' : Q => (Submodule.Quotient.mk q' : Q ⧸ fQ.range)) hw
307   unfold IsNoncharacteristic
308   exact Module.Finite.of_surjective Fbar hsurj

310 /-- The actual coordinate ring of the restricted canonical characteristic
311 variety is finite over the cotangent-coordinate ring of the hyperplane. -/
312 theorem canonical_finite_restrictedCoordinateQuotient
313     {n N : ℕ} {d : WeylAlg k (n + 1)}
314     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d) :
315     Module.Finite (oldTangentialCoeffRing (k := k) n)
316       (SymbolRing k (n + 1) ⧸ restrictedCoordinateIdeal (k := k) n
317         (Module.annihilator (SymbolRing k (n + 1))
318           (OrderAssociatedGradedModule k
319             (canonicalRightIdeal (presentedCoordinate k n) d N)))) := by
320   let J := Module.annihilator (SymbolRing k (n + 1))
321     (OrderAssociatedGradedModule k
322       (canonicalRightIdeal (presentedCoordinate k n) d N))
323   have hfinite : IsNoncharacteristic (k := k) n J := by
324     simpa [J] using canonical_isNoncharacteristic_annihilator (k := k) hd
325   unfold IsNoncharacteristic at hfinite
326   exact Module.Finite.of_surjective
327     (cokerToRestrictedCoordinateRing (k := k) n J)
328     (cokerToRestrictedCoordinateRing_surjective (k := k) n J)

330 /-- In the canonical support, vanishing of every tangential covariable forces
331 the distinguished normal covariable to vanish as well.  The conclusion does
332 not require the base point to lie on `x₀ = 0`. -/
333 theorem canonicalSupport_conormal_subset_zeroSection
334     {n N : ℕ} (d : WeylAlg k (n + 1))
335     (hd : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d)
336     {p : PrimeSpectrum (SymbolRing k (n + 1))}
337     (hp : p ∈ orderCharacteristicSupport k
338       (canonicalRightIdeal (presentedCoordinate k n) d N))
339     (htan : ∀ i : Fin (n + 1), i ≠ 0 →
340       MvPolynomial.X (.inr i) ∈ p.asIdeal) :
341     MvPolynomial.X (.inr (0 : Fin (n + 1))) ∈ p.asIdeal := by
342   exact CanonicalNormalAxisSupport.normalMomentum_mem_of_mem_canonicalSupport
343     d hd hp htan
