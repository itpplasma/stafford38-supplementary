35 /-- The one concrete theorem still needed by the final algebraic assembly. -/
36 def CanonicalSupportVanishing : Prop :=
37   ∀ (k : Type u) [Field k] [CharZero k] (n N : ℕ)
38     (d : WeylAlg k (n + 1)),
39     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
40       orderCharacteristicSupport k
41         (canonicalRightIdeal (presentedCoordinate k n) d N) = ∅

43 private theorem rankZero_certificate
44     (k : Type u) [Field k] (d : WeylAlg k 0) (hd : d ≠ 0) :
45     ∃ F R S : WeylAlg k 0,
46       (1 : WeylAlg k 0) = d * R + F * d * S := by
47   let e := presentedIteratedEquiv k 0
48   let c : k := e d
49   have hc : c ≠ 0 := by
50     intro hczero
51     change e d = 0 at hczero
52     apply hd
53     calc
54       d = e.symm (e d) := (e.symm_apply_apply d).symm
55       _ = e.symm 0 := congrArg e.symm hczero
56       _ = 0 := map_zero e.symm
57   refine ⟨0, e.symm (c⁻¹ : k), 0, ?_⟩
58   have hmul : c * c⁻¹ = (1 : k) := mul_inv_cancel₀ hc
59   have h : d * e.symm (c⁻¹ : k) = 1 := by
60     calc d * e.symm (c⁻¹ : k)
61         = e.symm c * e.symm (c⁻¹ : k) := by
62           congr 1
63           exact (e.symm_apply_apply d).symm
64       _ = e.symm (c * c⁻¹) := (map_mul e.symm _ _).symm
65       _ = e.symm 1 := congrArg e.symm hmul
66       _ = 1 := map_one e.symm
67   simpa using h.symm

69 /-- Once canonical support vanishing is proved, the exact theorem exported by
70 `Stafford38.Statement` follows with no further mathematical hypothesis. -/
71 theorem universalStatement_of_canonicalSupportVanishing
72     (hvanish : CanonicalSupportVanishing.{u}) :
73     Stafford38Challenge.UniversalStatement.{u} := by
74   intro k _ _ n d hd
75   cases n with
76   | zero =>
77       exact rankZero_certificate k d hd
78   | succ n =>
79       rcases scalar_or_normalized_symplectic_image k (Nat.succ_pos n) hd with
80         hscalar | hchart
81       · rcases hscalar with ⟨c, hc, rfl⟩
82         refine ⟨0, algebraMap k (WeylAlg k (n + 1)) c⁻¹, 0, ?_⟩
83         simp only [zero_mul, zero_add]
84         rw [← map_mul]
85         rw [mul_inv_cancel₀ hc, map_one]
86         simp
87       · rcases hchart with ⟨N, hN, M, Ninv, c, hM, hNinv, hMN, hNM,
88           hc, hpiece, haxis⟩
89         let e := standardSymplecticAlgEquivOfInverse
90           k M Ninv hM hNinv hMN hNM
91         let d' : WeylAlg k (n + 1) :=
92           normalizedSymplecticImage k M hM c d
93         have hd' : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d' := by
94           refine ⟨hpiece, ?_⟩
95           rw [← coeff_principal_pure_eq_normalForm k]
96           rw [← coeff_axisPolynomial k]
97           exact haxis
98         have hsupp := hvanish k n N d' hN hd'
99         rcases exists_fixedSource_certificate_of_orderCharacteristicSupport_eq_empty
100             k n N d' hsupp with ⟨R, S, hcert⟩
101         let a : WeylAlg k (n + 1) :=
102           algebraMap k (WeylAlg k (n + 1)) c⁻¹
103         have hd'eq : d' = a * e d := by
104           simp [d', a, e, normalizedSymplecticImage, Algebra.smul_def,
105             standardSymplecticAlgEquivOfInverse] <;> rfl
106         have ha_comm : ∀ z : WeylAlg k (n + 1), a * z = z * a := by
107           intro z
108           exact Algebra.commutes c⁻¹ z
109         have hchartCert :
110             (1 : WeylAlg k (n + 1)) =
111               e d * (a * R) +
112                 (presentedCoordinate k n) ^ N * e d * (a * S) := by
113           rw [hcert, hd'eq]
114           rw [ha_comm (e d)]
115           simp only [mul_assoc]
116         refine ⟨e.symm ((presentedCoordinate k n) ^ N),
117           e.symm (a * R), e.symm (a * S), ?_⟩
118         have htransport := congrArg e.symm hchartCert
119         simpa using htransport
