11 /-!
12 # Conditional Laurent-direction skeleton for canonical support vanishing
13 
14 This file retains a conditional Laurent-direction assembly between the literal
15 canonical quotient and `CanonicalSupportVanishing`. Its interfaces are
16 discharged downstream; they do not mark gaps in the unconditional theorem.
17 Over an algebraically
18 closed field, one order-zero coordinate predecessor of the unit supplies axis
19 avoidance, a Gabber/scalar-extension input makes the canonical fibre symbol
20 vanish on the Laurent equation-conormal locus, and the asymptotic producer supplies a
21 Laurent-generic conormal whose regular fibre residue is the forbidden pure
22 momentum direction. A fourth input explicitly isolates descent from algebraic
23 closure back to an arbitrary characteristic-zero field.
24 
25 No finite phase-space limit is asserted: the Laurent base coordinates may have
26 poles, and only the fibre residue is specialized.
27 -/

54 /-- Universal production of strict filtered coordinate cancellation for the
55 literal canonical quotient. -/
56 def CanonicalCoordinateCancellation : Prop :=
57   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
58     (d : WeylAlg k (n + 1)),
59     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
60       CoordinateCancellation k n N d
61 
62 /-- Universal production of the weakest filtered input used by terminal axis
63 avoidance: one order-zero coordinate predecessor of the quotient unit. -/
64 def CanonicalStrictUnitCoordinatePreimage : Prop :=
65   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
66     (d : WeylAlg k (n + 1)),
67     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
68       StrictUnitCoordinatePreimage k n N d

70 /-- The pure momentum axis in the fibre-coordinate space. -/
71 def pureMomentumFibreAxis (k : Type u) [Field k] (n : ℕ) :
72     Fin (n + 1) → k :=
73   fun i ↦ if i = 0 then 1 else 0
74 
75 /-- Packaged Laurent symbol control from Gabber, scalar extension, fibre-only
76 symbol extraction, and the checked conormal containment. -/
77 def CanonicalLaurentSymbolControl : Prop :=
78   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
79     (d : WeylAlg k (n + 1)),
80     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
81       ∃ P : MvPolynomial (Fin (n + 1)) k,
82         fibreLift P = presentedPrincipalComponent k orderWeight N d ∧
83         ∀ q ∈ equationConormalLocus
84             ((reducedOrderBaseIdeal k
85               (canonicalRightIdeal (presentedCoordinate k n) d N)).map
86                 (scalarPolynomialMap
87                   (k := k) (K := LaurentSeries k) (Fin (n + 1)))),
88           MvPolynomial.eval₂ (algebraMap k (LaurentSeries k)) q
89             (fibreLift P) = 0
90 
91 /-- The remaining global geometric producer.  Its base coordinates stay in
92 Laurent series; only its regular fibre covector specializes to the pure
93 momentum axis. -/
94 def CanonicalAsymptoticLaurentProducer : Prop :=
95   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
96     (d : WeylAlg k (n + 1)),
97     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
98     Disjoint
99       (orderCharacteristicSupport k
100         (canonicalRightIdeal (presentedCoordinate k n) d N))
101       (PrimeSpectrum.zeroLocus
102         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
103           Set (SymbolRing k (n + 1)))) →
104     (orderCharacteristicSupport k
105       (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty →
106     ∃ (y : Fin (n + 1) → LaurentSeries k)
107       (xi : Fin (n + 1) → PowerSeries k),
108       Sum.elim y
109           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈
110         equationConormalLocus
111           ((reducedOrderBaseIdeal k
112             (canonicalRightIdeal (presentedCoordinate k n) d N)).map
113               (scalarPolynomialMap
114                 (k := k) (K := LaurentSeries k) (Fin (n + 1)))) ∧
115       residueColumn xi = pureMomentumFibreAxis k n

117 /-- Canonical support vanishing restricted to algebraically closed
118 characteristic-zero fields. -/
119 def AlgebraicallyClosedCanonicalSupportVanishing : Prop :=
120   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
121     (d : WeylAlg k (n + 1)),
122     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
123       orderCharacteristicSupport k
124         (canonicalRightIdeal (presentedCoordinate k n) d N) = ∅

126 /-- The still-separate Weyl/base-change descent theorem.  This input must be
127 proved by transporting the canonical quotient and its initial ideal through
128 algebraic closure; geometric support descent alone does not supply it. -/
129 def CanonicalSupportDescent : Prop :=
130   AlgebraicallyClosedCanonicalSupportVanishing.{u} →
131     Stafford38.UniversalAssembly.CanonicalSupportVanishing.{u}

133 /-- The conditional contradiction skeleton over algebraically closed fields.
134 Its asymptotic input must be proved from divisor geometry without using the
135 desired support-vanishing conclusion. -/
136 theorem algebraicallyClosedCanonicalSupportVanishing_of_three_inputs
137     (hunit : CanonicalStrictUnitCoordinatePreimage.{u})
138     (hcontrol : CanonicalLaurentSymbolControl.{u})
139     (hasymptotic : CanonicalAsymptoticLaurentProducer.{u}) :
140     AlgebraicallyClosedCanonicalSupportVanishing.{u} := by
141   intro k _ _ _ n N d hN hd
142   let I := canonicalRightIdeal (presentedCoordinate k n) d N
143   have hdisjoint : Disjoint
144       (orderCharacteristicSupport k I)
145       (PrimeSpectrum.zeroLocus
146         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
147           Set (SymbolRing k (n + 1)))) := by
148     exact
149       canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
150         k n N (hunit k n N d hN hd)
151   by_contra hsupp
152   have hnonempty : (orderCharacteristicSupport k I).Nonempty :=
153     Set.nonempty_iff_ne_empty.mpr hsupp
154   rcases hcontrol k n N d hN hd with ⟨P, hP, hvanishes⟩
155   rcases hasymptotic k n N d hN hd hdisjoint hnonempty with
156     ⟨y, xi, hgeneric, hresidue⟩
157   apply false_of_fibreOnly_symbol_one_on_residue_and_vanishes_on_laurentConormal
158     (reducedOrderBaseIdeal k I) P (pureMomentumFibreAxis k n) y xi
159     hgeneric hresidue hvanishes
160   have hcanonical := canonical_orderPrincipalComponent_eval_pureMomentumAxis
161     n N hd
162   have hsplit : Sum.elim (fun _ : Fin (n + 1) ↦ (0 : k))
163       (pureMomentumFibreAxis k n) =
164       axisPoint k (.inr (0 : Fin (n + 1))) := by
165     funext i
166     rcases i with i | i
167     · simp [axisPoint]
168     · simp [axisPoint, pureMomentumFibreAxis]
169   rw [← hP] at hcanonical
170   have heval := eval₂_fibreLift (K := k) P
171     (fun _ : Fin (n + 1) ↦ (0 : k)) (pureMomentumFibreAxis k n)
172   rw [hsplit] at heval
173   simpa only [← MvPolynomial.aeval_def, MvPolynomial.aeval_eq_eval] using
174     heval.symm.trans hcanonical

176 /-- Consequently the exact Stafford statement follows from the three
177 algebraically closed inputs and the separate support-descent theorem. -/
178 theorem universalStatement_of_four_inputs
179     (hunit : CanonicalStrictUnitCoordinatePreimage.{u})
180     (hcontrol : CanonicalLaurentSymbolControl.{u})
181     (hasymptotic : CanonicalAsymptoticLaurentProducer.{u})
182     (hdescent : CanonicalSupportDescent.{u}) :
183     Stafford38Challenge.UniversalStatement.{u} :=
184   Stafford38.UniversalAssembly.universalStatement_of_canonicalSupportVanishing
185     (hdescent
186       (algebraicallyClosedCanonicalSupportVanishing_of_three_inputs
187         hunit hcontrol hasymptotic))
