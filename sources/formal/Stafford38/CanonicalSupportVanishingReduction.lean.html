9 /-!
10 # Conditional Laurent-direction skeleton for canonical support vanishing
11 
12 This file retains a conditional Laurent-direction assembly between the literal
13 canonical quotient and `CanonicalSupportVanishing`. Its interfaces are
14 discharged downstream; they do not mark gaps in the unconditional theorem.
15 Over an algebraically
16 closed field, one order-zero coordinate predecessor of the unit supplies axis
17 avoidance, a Gabber/scalar-extension input makes the canonical fibre symbol
18 vanish on the Laurent equation-conormal locus, and the asymptotic producer supplies a
19 Laurent-generic conormal whose regular fibre residue is the forbidden pure
20 momentum direction. A fourth input explicitly isolates descent from algebraic
21 closure back to an arbitrary characteristic-zero field.
22 
23 No finite phase-space limit is asserted: the Laurent base coordinates may have
24 poles, and only the fibre residue is specialized.
25 -/

52 /-- Universal production of strict filtered coordinate cancellation for the
53 literal canonical quotient. -/
54 def CanonicalCoordinateCancellation : Prop :=
55   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
56     (d : PresentedWeyl k (n + 1)),
57     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
58       CoordinateCancellation k n N d

60 /-- Universal production of the weakest filtered input used by terminal axis
61 avoidance: one order-zero coordinate predecessor of the quotient unit. -/
62 def CanonicalStrictUnitCoordinatePreimage : Prop :=
63   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
64     (d : PresentedWeyl k (n + 1)),
65     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
66       StrictUnitCoordinatePreimage k n N d

68 /-- The pure momentum axis in the fibre-coordinate space. -/
69 def pureMomentumFibreAxis (k : Type u) [Field k] (n : ℕ) :
70     Fin (n + 1) → k :=
71   fun i ↦ if i = 0 then 1 else 0
72 
73 /-- Packaged Laurent symbol control from Gabber, scalar extension, fibre-only
74 symbol extraction, and the checked conormal containment. -/
75 def CanonicalLaurentSymbolControl : Prop :=
76   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
77     (d : PresentedWeyl k (n + 1)),
78     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
79       ∃ P : MvPolynomial (Fin (n + 1)) k,
80         fibreLift P = presentedPrincipalComponent k orderWeight N d ∧
81         ∀ q ∈ equationConormalLocus
82             ((reducedOrderBaseIdeal k
83               (canonicalRightIdeal (presentedCoordinate k n) d N)).map
84                 (scalarPolynomialMap
85                   (k := k) (K := LaurentSeries k) (Fin (n + 1)))),
86           MvPolynomial.eval₂ (algebraMap k (LaurentSeries k)) q
87             (fibreLift P) = 0
88 
89 /-- The remaining global geometric producer.  Its base coordinates stay in
90 Laurent series; only its regular fibre covector specializes to the pure
91 momentum axis. -/
92 def CanonicalAsymptoticLaurentProducer : Prop :=
93   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
94     (d : PresentedWeyl k (n + 1)),
95     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
96     Disjoint
97       (orderCharacteristicSupport k
98         (canonicalRightIdeal (presentedCoordinate k n) d N))
99       (PrimeSpectrum.zeroLocus
100         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
101           Set (SymbolRing k (n + 1)))) →
102     (orderCharacteristicSupport k
103       (canonicalRightIdeal (presentedCoordinate k n) d N)).Nonempty →
104     ∃ (y : Fin (n + 1) → LaurentSeries k)
105       (xi : Fin (n + 1) → PowerSeries k),
106       Sum.elim y
107           (fun i ↦ algebraMap (PowerSeries k) (LaurentSeries k) (xi i)) ∈
108         equationConormalLocus
109           ((reducedOrderBaseIdeal k
110             (canonicalRightIdeal (presentedCoordinate k n) d N)).map
111               (scalarPolynomialMap
112                 (k := k) (K := LaurentSeries k) (Fin (n + 1)))) ∧
113       residueColumn xi = pureMomentumFibreAxis k n

115 /-- Canonical support vanishing restricted to algebraically closed
116 characteristic-zero fields. -/
117 def AlgebraicallyClosedCanonicalSupportVanishing : Prop :=
118   ∀ (k : Type u) [Field k] [CharZero k] [IsAlgClosed k] (n N : ℕ)
119     (d : PresentedWeyl k (n + 1)),
120     0 < N → IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d →
121       orderCharacteristicSupport k
122         (canonicalRightIdeal (presentedCoordinate k n) d N) = ∅

124 /-- The still-separate Weyl/base-change descent theorem.  This input must be
125 proved by transporting the canonical quotient and its initial ideal through
126 algebraic closure; geometric support descent alone does not supply it. -/
127 def CanonicalSupportDescent : Prop :=
128   AlgebraicallyClosedCanonicalSupportVanishing.{u} →
129     Stafford38.UniversalAssembly.CanonicalSupportVanishing.{u}

131 /-- The conditional contradiction skeleton over algebraically closed fields.
132 Its asymptotic input must be proved from divisor geometry without using the
133 desired support-vanishing conclusion. -/
134 theorem algebraicallyClosedCanonicalSupportVanishing_of_three_inputs
135     (hunit : CanonicalStrictUnitCoordinatePreimage.{u})
136     (hcontrol : CanonicalLaurentSymbolControl.{u})
137     (hasymptotic : CanonicalAsymptoticLaurentProducer.{u}) :
138     AlgebraicallyClosedCanonicalSupportVanishing.{u} := by
139   intro k _ _ _ n N d hN hd
140   let I := canonicalRightIdeal (presentedCoordinate k n) d N
141   have hdisjoint : Disjoint
142       (orderCharacteristicSupport k I)
143       (PrimeSpectrum.zeroLocus
144         ({MvPolynomial.X (.inl (0 : Fin (n + 1)))} :
145           Set (SymbolRing k (n + 1)))) := by
146     exact
147       canonical_orderCharacteristicSupport_disjoint_coordinate_zeroLocus_of_strictUnit
148         k n N (hunit k n N d hN hd)
149   by_contra hsupp
150   have hnonempty : (orderCharacteristicSupport k I).Nonempty :=
151     Set.nonempty_iff_ne_empty.mpr hsupp
152   rcases hcontrol k n N d hN hd with ⟨P, hP, hvanishes⟩
153   rcases hasymptotic k n N d hN hd hdisjoint hnonempty with
154     ⟨y, xi, hgeneric, hresidue⟩
155   apply false_of_fibreOnly_symbol_one_on_residue_and_vanishes_on_laurentConormal
156     (reducedOrderBaseIdeal k I) P (pureMomentumFibreAxis k n) y xi
157     hgeneric hresidue hvanishes
158   have hcanonical := canonical_orderPrincipalComponent_eval_pureMomentumAxis
159     n N hd
160   have hsplit : Sum.elim (fun _ : Fin (n + 1) ↦ (0 : k))
161       (pureMomentumFibreAxis k n) =
162       axisPoint k (.inr (0 : Fin (n + 1))) := by
163     funext i
164     rcases i with i | i
165     · simp [axisPoint]
166     · simp [axisPoint, pureMomentumFibreAxis]
167   rw [← hP] at hcanonical
168   have heval := eval₂_fibreLift (K := k) P
169     (fun _ : Fin (n + 1) ↦ (0 : k)) (pureMomentumFibreAxis k n)
170   rw [hsplit] at heval
171   simpa only [← MvPolynomial.aeval_def, MvPolynomial.aeval_eq_eval] using
172     heval.symm.trans hcanonical

174 /-- Consequently the exact Stafford statement follows from the three
175 algebraically closed inputs and the separate support-descent theorem. -/
176 theorem universalStatement_of_four_inputs
177     (hunit : CanonicalStrictUnitCoordinatePreimage.{u})
178     (hcontrol : CanonicalLaurentSymbolControl.{u})
179     (hasymptotic : CanonicalAsymptoticLaurentProducer.{u})
180     (hdescent : CanonicalSupportDescent.{u}) :
181     Stafford38.UniversalStatement.{u} :=
182   Stafford38.UniversalAssembly.universalStatement_of_canonicalSupportVanishing
183     (hdescent
184       (algebraicallyClosedCanonicalSupportVanishing_of_three_inputs
185         hunit hcontrol hasymptotic))
