11 /-!
12 # The polynomial representation of the Weyl algebra
13 
14 The canonical coordinates act by multiplication and the canonical momenta by
15 formal partial differentiation.  The target is the intrinsic algebra of
16 finite-order differential operators, so this construction does not assume an
17 identification of that algebra with the Weyl algebra.
18 -/

29 variable (k : Type*) [Field k] (n : ℕ)
30 
31 abbrev PolynomialRing := MvPolynomial (Fin n) k
32 
33 def coordinateEnd (i : Fin n) : End (k := k) (R := PolynomialRing k n) :=
34   multiplication (MvPolynomial.X i)
35 
36 def momentumEnd (i : Fin n) : End (k := k) (R := PolynomialRing k n) :=
37   (MvPolynomial.pderiv i).toLinearMap
38 
39 theorem coordinateEnd_mem_order_zero (i : Fin n) :
40     coordinateEnd k n i ∈ order (k := k) (R := PolynomialRing k n) 0 := by
41   rw [mem_order_zero_iff_eq_multiplication]
42   ext f
43   simp [coordinateEnd, multiplication_apply]
44 
45 theorem momentumEnd_mem_order_one (i : Fin n) :
46     momentumEnd k n i ∈ order (k := k) (R := PolynomialRing k n) 1 := by
47   rw [show 1 = 0 + 1 by rfl, mem_order_succ_iff]
48   intro f
49   rw [mem_order_zero_iff_eq_multiplication]
50   apply LinearMap.ext
51   intro g
52   simp [commutator_apply, momentumEnd, multiplication_apply]
53   exact mul_comm _ _
54 
55 def differentialGenerator :
56     (Fin n ⊕ Fin n) → algebra (k := k) (R := PolynomialRing k n)
57   | .inl i => ⟨coordinateEnd k n i, 0, coordinateEnd_mem_order_zero k n i⟩
58   | .inr i => ⟨momentumEnd k n i, 1, momentumEnd_mem_order_one k n i⟩
59 
60 theorem pderiv_comm (i j : Fin n) (f : PolynomialRing k n) :
61     MvPolynomial.pderiv i (MvPolynomial.pderiv j f) =
62       MvPolynomial.pderiv j (MvPolynomial.pderiv i f) := by
63   by_cases hij : i = j
64   · subst j
65     rfl
66   induction f using MvPolynomial.induction_on with
67   | C a => simp
68   | add f g hf hg => simp [hf, hg]
69   | mul_X f a hf =>
70       classical
71       simp only [MvPolynomial.pderiv_mul, map_add]
72       rw [hf]
73       simp only [MvPolynomial.pderiv_X]
74       by_cases hai : a = i <;> by_cases haj : a = j <;>
75         simp [hai, haj, hij, Ne.symm hij]
76 
77 theorem differentialGenerator_commutator (i j : Fin n ⊕ Fin n) :
78     AlgebraicAnalysis.ringCommutator (differentialGenerator k n i)
79         (differentialGenerator k n j) =
80       algebraMap k (algebra (k := k) (R := PolynomialRing k n))
81         (Matrix.J (Fin n) k i j) := by
82   apply Subtype.ext
83   apply LinearMap.ext
84   intro f
85   cases i with
86   | inl i =>
87       cases j with
88       | inl j => simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd,
89           Module.End.mul_apply, Matrix.J, mul_comm, mul_assoc, mul_left_comm]
90       | inr j =>
91           by_cases h : i = j
92           · subst j
93             simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd, momentumEnd,
94               Module.End.mul_apply, Matrix.J]
95           · simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd, momentumEnd,
96               Module.End.mul_apply, Matrix.J, h]
97   | inr i =>
98       cases j with
99       | inl j =>
100           by_cases h : i = j
101           · subst j
102             simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd, momentumEnd,
103               Module.End.mul_apply, Matrix.J]
104           · simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, coordinateEnd, momentumEnd,
105               Module.End.mul_apply, Matrix.J, h]
106       | inr j =>
107           simp [AlgebraicAnalysis.ringCommutator, differentialGenerator, momentumEnd, Module.End.mul_apply,
108             Matrix.J, pderiv_comm k n i j]

110 /-- The polynomial action, valued in genuine finite-order differential operators. -/
111 def polynomialDifferentialAction :
112     WeylAlg k n →ₐ[k] algebra (k := k) (R := PolynomialRing k n) :=
113   freeWeylLift (Matrix.J (Fin n) k) (differentialGenerator k n)
114     (differentialGenerator_commutator k n)
115 
116 @[simp] theorem polynomialDifferentialAction_generator (i : Fin n ⊕ Fin n) :
117     polynomialDifferentialAction k n
118         (freeWeylGenerator (Matrix.J (Fin n) k) i) = differentialGenerator k n i :=
119   freeWeylLift_generator _ _ _ i
120 
121 theorem polynomialDifferentialAction_mem (d : WeylAlg k n) :
122     (polynomialDifferentialAction k n d : End (k := k) (R := PolynomialRing k n)) ∈
123       algebra (k := k) (R := PolynomialRing k n) :=
124   (polynomialDifferentialAction k n d).property
125 
126 end
127 end Stafford38.PolynomialDifferentialOperators
