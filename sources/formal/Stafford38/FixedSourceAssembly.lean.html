30 theorem universalFixedSourceStatement_of_canonicalSupportVanishing
31     (hvanish : CanonicalSupportVanishing.{u}) :
32     UniversalFixedSourceStatement.{u} := by
33   intro k _ _ n d hd
34   rcases exists_top_bernstein_piece k hd with ⟨N, hpiece, hprincipal⟩
35   have hdegree : bernsteinDegree k d = N :=
36     bernsteinDegree_eq_of_piece_of_principal_ne_zero k hpiece hprincipal
37   by_cases hNzero : N = 0
38   · have hpiece0 : d ∈ bernsteinPiece k (n + 1) 0 := by
39       simpa [hNzero] using hpiece
40     rcases eq_algebraMap_of_mem_bernsteinPiece_zero k hpiece0 with ⟨c, rfl⟩
41     have hc : c ≠ 0 := by
42       intro hc
43       subst c
44       simp at hd
45     let ell : WeylAlg k (n + 1) := presentedCoordinate k n
46     have hell : IsLinearWeylCoordinate k n ell := by
47       refine ⟨1, 1, ?_, ?_, ?_, ?_, ?_⟩
48       · simp
49       · simp
50       · simp
51       · simp
52       · change freeWeylGenerator (standardForm k (n + 1))
53             (.inl (0 : Fin (n + 1))) =
54           Stafford.linearCombination 1
55             (freeWeylGenerator (standardForm k (n + 1)))
56             (.inl (0 : Fin (n + 1)))
57         exact (freeWeylLinearCombination_one (standardForm k (n + 1))
58           (.inl (0 : Fin (n + 1)))).symm
59 
60     refine ⟨ell, algebraMap k _ c⁻¹, 0, hell, ?_⟩
61     simp only [hdegree, hNzero, pow_zero]
62     rw [← map_mul, mul_inv_cancel₀ hc, map_one]
63     simp
64   · have hN : 0 < N := Nat.pos_of_ne_zero hNzero
65     rcases exists_normalized_symplectic_image k
66         (.inr ⟨0, Nat.succ_pos n⟩) hpiece (principal_isHomogeneous k d)
67         hprincipal hN with
68       ⟨M, Ninv, c, hM, hNinv, hMN, hNM, hc, hpiece', haxis⟩
69     let e := standardSymplecticAlgEquivOfInverse k M Ninv hM hNinv hMN hNM
70     let d' : WeylAlg k (n + 1) := normalizedSymplecticImage k M hM c d
71     have hd' : IsPBWMonicAt k (.inr (0 : Fin (n + 1))) N d' := by
72       refine ⟨hpiece', ?_⟩
73       rw [← coeff_principal_pure_eq_normalForm k]
74       rw [← coeff_axisPolynomial k]
75       exact haxis
76     have hsupp := hvanish k n N d' hN hd'
77     rcases exists_fixedSource_certificate_of_orderCharacteristicSupport_eq_empty
78       k n N d' hsupp with ⟨R, S, hcert⟩
79     let a : WeylAlg k (n + 1) := algebraMap k _ c⁻¹
80     have hd'eq : d' = a * e d := by
81       simp [d', a, e, normalizedSymplecticImage, Algebra.smul_def,
82         standardSymplecticAlgEquivOfInverse]
83       all_goals rfl
84     have ha_comm : ∀ z : WeylAlg k (n + 1), a * z = z * a := by
85       intro z
86       exact Algebra.commutes c⁻¹ z
87     have hchartCert :
88         (1 : WeylAlg k (n + 1)) = e d * (a * R) +
89           (presentedCoordinate k n) ^ N * e d * (a * S) := by
90       rw [hcert, hd'eq]
91       rw [ha_comm (e d)]
92       simp only [mul_assoc]
93     let ell : WeylAlg k (n + 1) := e.symm (presentedCoordinate k n)
94     have hell : IsLinearWeylCoordinate k n ell := by
95       refine ⟨M, Ninv, hM, hNinv, hMN, hNM, ?_⟩
96       change (standardSymplecticAlgEquivOfInverse k M Ninv hM hNinv hMN hNM).symm
97           (freeWeylGenerator (standardForm k (n + 1))
98             (.inl (0 : Fin (n + 1)))) =
99         Stafford.linearCombination Ninv
100           (freeWeylGenerator (standardForm k (n + 1)))
101           (.inl (0 : Fin (n + 1)))
102       exact standardSymplecticAlgEquivOfInverse_symm_generator
103         k M Ninv hM hNinv hMN hNM (.inl (0 : Fin (n + 1)))
104     refine ⟨ell, e.symm (a * R), e.symm (a * S), hell, ?_⟩
105     have ht := congrArg e.symm hchartCert
106     simpa [ell, hdegree] using ht

108 /-- Pointwise form of the exact-degree fixed-source theorem. -/
109 theorem fixedSource_positive
110     (hvanish : CanonicalSupportVanishing.{u})
111     (k : Type u) [Field k] [CharZero k] (n : ℕ)
112     (d : WeylAlg k (n + 1)) (hd : d ≠ 0) :
113     ∃ ell R S : WeylAlg k (n + 1),
114       IsLinearWeylCoordinate k n ell ∧
115         (1 : WeylAlg k (n + 1)) =
116           d * R + ell ^ bernsteinDegree k d * d * S :=
117   universalFixedSourceStatement_of_canonicalSupportVanishing hvanish k n d hd
