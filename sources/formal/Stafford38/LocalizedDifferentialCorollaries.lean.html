51 private theorem multiplicationD_isUnit {b : B} (hb : IsUnit b) :
52     IsUnit (multiplicationD (k := k) (B := B) b) := by
53   rcases hb with ⟨u, hu⟩
54   refine ⟨⟨multiplicationD (k := k) (B := B) b,
55       multiplicationD (k := k) (B := B) u.inv, ?_, ?_⟩, rfl⟩
56   · have h : b * u.inv = 1 := by rw [← hu]; exact u.val_inv

58   · have h : u.inv * b = 1 := by rw [← hu]; exact u.inv_val

62 theorem s38_localized_differential
63     (hS : S38 (WeylAlg k n)) : S38 (D (k := k) B) := by
64   apply s38_of_leftUnitClearing hS
65   intro q hq
66   obtain ⟨s, a, ha⟩ := localized_differential_left_denominator_clearing S B q
67   let b : B := algebraMap (A (k := k) (n := n)) B (s : A)
68   have hb : IsUnit b := IsLocalization.map_units B s
69   refine ⟨a, multiplicationD (k := k) (B := B) b,
70     multiplicationD_isUnit (k := k) (B := B) hb, ?_⟩
71   apply Subtype.ext
72   change multiplication (k := k) b * (q : Module.End k B) = weylEnd S B a
73   exact ha
