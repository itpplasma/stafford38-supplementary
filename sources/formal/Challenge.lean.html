10 # Stafford's Conjecture 3.8
11 
12 J. T. Stafford, *Module structure of Weyl algebras*, J. London Math. Soc. (2)
13 18 (1978), 429–442, states on page 438:
14 
15 > Conjecture 3.8. Let `d ∈ Aₙ`. Then there exists `f ∈ Aₙ` such that
16 > `Aₙ = d Aₙ + f d Aₙ`.
17 
18 Here `Aₙ = Aₙ(k)` is the `n`th Weyl algebra over a field `k` of characteristic
19 zero, generated by `x₁, …, xₙ, ∂₁, …, ∂ₙ` with `[∂ᵢ, xⱼ] = δᵢⱼ` and all other
20 pairs of generators commuting. Both `d Aₙ` and `f d Aₙ` are right ideals, and
21 `f` stands to the left of `d`. Membership of `1` in their sum is the identity
22 `1 = d R + f d S` for some `R, S ∈ Aₙ`, with the factors in this order.
23 
24 The printed conjecture omits the hypothesis `d ≠ 0`, without which the
25 conclusion fails for `d = 0`; the discussion following it in the article uses
26 that hypothesis. This file states the intended nonzero form, for every
27 characteristic-zero field and every rank `n ≥ 0`. For `n = 0` the Weyl algebra
28 is the field itself and a nonzero `d` is a unit.
29 
30 The presentation and statement below repeat the exact definitions used by
31 `Stafford38.ChallengeDefinitions` in the solution. Palomar requires the selected
32 Challenge to import only authenticated libraries. The independent Comparator
33 checks that the definitions and theorem type match; this file contains the
34 deliberate placeholder filled by the solution.
35 -/

42 /-- Linear combination of an indexed family by the rows of a matrix. -/
43 def linearCombination (M : Matrix ι ι k) (z : ι → A) (i : ι) : A :=
44   ∑ j, algebraMap k A (M i j) * z j
45 
46 variable {k ι : Type*} [Field k]
47 
48 /-- The commutator relations presenting a Weyl-type algebra. -/
49 def freeWeylRelation (omega : Matrix ι ι k)
50     (a b : FreeAlgebra k ι) : Prop :=
51   ∃ i j,
52     a = FreeAlgebra.ι k i * FreeAlgebra.ι k j -
53       FreeAlgebra.ι k j * FreeAlgebra.ι k i ∧
54     b = algebraMap k (FreeAlgebra k ι) (omega i j)
55 
56 /-- The free algebra modulo the commutator relations encoded by `omega`. -/
57 abbrev FreeWeyl (k : Type*) [Field k] (ι : Type*)
58     (omega : Matrix ι ι k) := RingQuot (freeWeylRelation omega)
59 
60 /-- The canonical generators of the presented Weyl-type algebra. -/
61 def freeWeylGenerator (omega : Matrix ι ι k) (i : ι) : FreeWeyl k ι omega :=
62   RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i)
63 
64 end Stafford

84 /-- The Weyl commutator relation for the standard symplectic matrix. -/

117 /-- Insert an old phase variable after the newest coordinate or momentum. -/
118 def oldIndex {n : ℕ} : PhaseVar n → PhaseVar (n + 1)
119   | .inl i => .inl i.succ
120   | .inr i => .inr i.succ
121 
122 @[simp] theorem oldIndex_inl {n : ℕ} (i : Fin n) :
123     oldIndex (.inl i) = .inl i.succ := rfl
124 
125 @[simp] theorem oldIndex_inr {n : ℕ} (i : Fin n) :
126     oldIndex (.inr i) = .inr i.succ := rfl

128 /-- Insert an old free-algebra word into the next-rank free algebra. -/
129 def freeOldMap (k : Type u) [Field k] (n : ℕ) :
130     FreeAlgebra k (PhaseVar n) →ₐ[k] FreeAlgebra k (PhaseVar (n + 1)) :=
131   FreeAlgebra.lift k (fun i => FreeAlgebra.ι k (oldIndex i))
132 
133 /-- Ordered PBW words, with old pairs followed by the newest pair. -/
134 def freeOrderedMonomial (k : Type u) [Field k] :
135     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → FreeAlgebra k (PhaseVar n)
136   | 0, _, _ => 1
137   | n + 1, a, p =>
138       freeOldMap k n
139           (freeOrderedMonomial k n (fun i => a i.succ) (fun i => p i.succ)) *
140         FreeAlgebra.ι k (.inl (0 : Fin (n + 1))) ^ a 0 *
141         FreeAlgebra.ι k (.inr (0 : Fin (n + 1))) ^ p 0
142 
143 /-- The image of an ordered PBW word in the shared Weyl quotient. -/
144 def orderedMonomial (k : Type u) [Field k] (n : ℕ)
145     (a p : Fin n → ℕ) : WeylAlg k n :=
146   RingQuot.mkAlgHom k (relation (k := k) (n := n) (Matrix.J (Fin n) k))
147     (freeOrderedMonomial k n a p)

149 /-- Total degree of a split coordinate/momentum exponent. -/
150 def phaseDegree {n : ℕ} (a p : Fin n → ℕ) : ℕ :=
151   (∑ i, a i) + ∑ i, p i
152 
153 /-- Span of the ordered Weyl words of total degree at most `N`. -/
154 def bernsteinPiece (k : Type u) [Field k] (n N : ℕ) :
155     Submodule k (WeylAlg k n) :=
156   Submodule.span k
157     {z | ∃ a p : Fin n → ℕ,
158       phaseDegree a p ≤ N ∧ z = orderedMonomial k n a p}
159 
160 /-- The intrinsic Bernstein degree of a presented element: its least piece. -/
161 noncomputable def bernsteinDegree (k : Type u) [Field k] {n : ℕ}
162     (d : WeylAlg k n) : ℕ :=
163   sInf {N : ℕ | d ∈ bernsteinPiece k n N}
164 
165 /-- A source coordinate obtained from an invertible symplectic linear change
166 of the canonical Weyl generators. -/
167 def IsLinearWeylCoordinate (k : Type u) [Field k] (n : ℕ)
168     (ell : WeylAlg k (n + 1)) : Prop :=
169   ∃ (M N : Matrix (PhaseVar (n + 1)) (PhaseVar (n + 1)) k)
170       (_hM : M * standardForm k (n + 1) * Matrix.transpose M =
171         standardForm k (n + 1))
172       (_hN : N * standardForm k (n + 1) * Matrix.transpose N =
173         standardForm k (n + 1))
174       (_hMN : M * N = 1) (_hNM : N * M = 1),
175       ell = Stafford.linearCombination N (generator k (n + 1))
176         (.inl (0 : Fin (n + 1)))
177 
178 /-- The exact fixed-source challenge, using the canonical intrinsic degree. -/
179 def UniversalFixedSourceStatement : Prop :=
180   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
181     (d : WeylAlg k (n + 1)), d ≠ 0 →
182       ∃ ell R S : WeylAlg k (n + 1),
183         IsLinearWeylCoordinate k n ell ∧
184           (1 : WeylAlg k (n + 1)) =
185             d * R + ell ^ bernsteinDegree k d * d * S
186 
187 end Stafford38FixedSourceChallenge
