11 /-!
12 # Stafford's Conjecture 3.8
13 
14 J. T. Stafford, *Module structure of Weyl algebras*, J. London Math. Soc. (2)
15 18 (1978), 429–442, states on page 438:
16 
17 > Conjecture 3.8. Let `d ∈ Aₙ`. Then there exists `f ∈ Aₙ` such that
18 > `Aₙ = d Aₙ + f d Aₙ`.
19 
20 Here `Aₙ = Aₙ(k)` is the `n`th Weyl algebra over a field `k` of characteristic
21 zero, generated by `x₁, …, xₙ, ∂₁, …, ∂ₙ` with `[∂ᵢ, xⱼ] = δᵢⱼ` and all other
22 pairs of generators commuting. Both `d Aₙ` and `f d Aₙ` are right ideals, and
23 `f` stands to the left of `d`. Membership of `1` in their sum is the identity
24 `1 = d R + f d S` for some `R, S ∈ Aₙ`, with the factors in this order.
25 
26 The printed conjecture omits the hypothesis `d ≠ 0`, without which the
27 conclusion fails for `d = 0`; the discussion following it in the article uses
28 that hypothesis. This file states the intended nonzero form, for every
29 characteristic-zero field and every rank `n ≥ 0`. For `n = 0` the Weyl algebra
30 is the field itself and a nonzero `d` is a unit.
31 
32 Besides this statement, the file states the results the accompanying paper
33 claims: Theorem 1.1 and Corollary 6.3 as printed, the stable cyclicity of
34 Corollary 6.3, and the consequences listed in Section 7 that have no printed
35 proof (the left-handed identity, Corollary 6.3 for left modules, the identity
36 for differential operators on localizations of `k[x_1, …, x_n]`, and the
37 identity with `F = x^r` for `d = p^r − ∑_j a_j x^{m_j}`).
38 
39 The presentation and statements below repeat the exact definitions used by
40 `Stafford38.ChallengeDefinitions` and `Stafford38.PaperClaimDefinitions` in the
41 solution. Palomar requires the selected Challenge to import only authenticated
42 libraries. The independent Comparator checks that the definitions and theorem
43 types match; this file contains the deliberate placeholders filled by the
44 solution.
45 -/

58 /-- The commutator relations presenting a Weyl-type algebra. -/
59 def freeWeylRelation (omega : Matrix ι ι k)
60     (a b : FreeAlgebra k ι) : Prop :=
61   ∃ i j,
62     a = FreeAlgebra.ι k i * FreeAlgebra.ι k j -
63       FreeAlgebra.ι k j * FreeAlgebra.ι k i ∧
64     b = algebraMap k (FreeAlgebra k ι) (omega i j)
65 
66 /-- The free algebra modulo the commutator relations encoded by `omega`. -/
67 abbrev FreeWeyl (k : Type*) [Field k] (ι : Type*)
68     (omega : Matrix ι ι k) := RingQuot (freeWeylRelation omega)
69 
70 /-- The canonical generators of the presented Weyl-type algebra. -/
71 def freeWeylGenerator (omega : Matrix ι ι k) (i : ι) : FreeWeyl k ι omega :=
72   RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i)

250 /-- An `ω`-preserving `g` (`g J gᵀ = J`) preserves the Weyl relations
251 `[u_i, u_j] = J_ij`. -/
252 theorem linearImage_rel (k : Type*) [Field k] {n : ℕ}
253     (g : Matrix.symplecticGroup (Fin n) k) ⦃a b : FreeAlgebra k (PhaseVar n)⦄
254     (h : relation (Matrix.J (Fin n) k) a b) :
255     FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)) a =
256       FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)) b := by
257   obtain ⟨i, j, rfl, rfl⟩ := h
258   have hcomm : ∀ p q : PhaseVar n,
259       generator k n p * generator k n q - generator k n q * generator k n p =
260         algebraMap k (WeylAlg k n) (Matrix.J (Fin n) k p q) := by
261     intro p q
262     have := RingQuot.mkAlgHom_rel k
263       (show relation (Matrix.J (Fin n) k) _ _ from ⟨p, q, rfl, rfl⟩)
264     simpa [generator, Stafford.freeWeylGenerator, map_sub, map_mul] using this
265   have hg := (SymplecticGroup.mem_iff).mp g.2
266   -- Reversed simp lemmas (`← l`) would become helpers named after the compiling
267   -- module, so challenge and solution copies of this proof would differ.
268   have smul_def' : ∀ (c : k) (x : WeylAlg k n), algebraMap k (WeylAlg k n) c * x = c • x :=
269     fun c x => (Algebra.smul_def c x).symm
270   have sum_sub' : ∀ (s : Finset (PhaseVar n)) (f f' : PhaseVar n → WeylAlg k n),
271       ∑ p ∈ s, f p - ∑ p ∈ s, f' p = ∑ p ∈ s, (f p - f' p) :=
272     fun s f f' => (Finset.sum_sub_distrib f f').symm
273   have sum_algebraMap : ∀ (s : Finset (PhaseVar n)) (f : PhaseVar n → k),
274       ∑ p ∈ s, algebraMap k (WeylAlg k n) (f p) = algebraMap k (WeylAlg k n) (∑ p ∈ s, f p) :=
275     fun s f => (map_sum (algebraMap k (WeylAlg k n)) f s).symm
276   simp only [map_sub, map_mul, FreeAlgebra.lift_ι_apply, AlgHom.commutes, linearImage,
277     Stafford.linearCombination, smul_def', Finset.sum_mul, Finset.mul_sum,
278     smul_mul_smul_comm]
279   rw [Finset.sum_comm, ← Finset.sum_sub_distrib]
280   simp_rw [sum_sub']
281   have hterm : ∀ p q : PhaseVar n,
282       ((g : Matrix _ _ k) i p * (g : Matrix _ _ k) j q) • (generator k n p * generator k n q) -
283         ((g : Matrix _ _ k) j q * (g : Matrix _ _ k) i p) • (generator k n q * generator k n p) =
284       algebraMap k (WeylAlg k n)
285         ((g : Matrix _ _ k) i p * Matrix.J (Fin n) k p q * (g : Matrix _ _ k) j q) := by
286     intro p q
287     rw [mul_comm ((g : Matrix _ _ k) j q)]
288     rw [show ∀ (c : k) (x y : WeylAlg k n), c • x - c • y = c • (x - y) from
289       fun c x y => (smul_sub c x y).symm, hcomm, Algebra.smul_def, ← map_mul]
290     congr 1
291     ring
292   simp_rw [hterm, sum_algebraMap]
293   congr 1
294   conv_rhs => rw [← hg]
295   simp only [Matrix.mul_apply, Matrix.transpose_apply, Finset.sum_mul]
296   rw [Finset.sum_comm]

313 /-- **Theorem 1.1.**  For every field of characteristic zero `k`, every `n ≥ 0`
314 and every nonzero `d ∈ A_n(k)`, there exists an `F ∈ A_n(k)` such that

341 /-- Stafford's *stably cyclic* [Sta78, p. 435], as quoted in the proof of
342 Corollary 6.3 (tex line 1619): "given `l ≥ 1` and `a_1, …, a_{l+1} ∈ T` that
343 generate `T`, there exist `f_i ∈ A_n` such that `a_i + a_{l+1} f_i`,

382 /-- `P` is a `k`-linear differential operator of order at most `N` on the
383 commutative `k`-algebra `B` (Grothendieck): every iterated commutator
384 `[…[[P, b₀], b₁], …, b_N]` with multiplication operators vanishes, where
385 `[Q, b] = Q ∘ b − b ∘ Q`. -/
386 def IsDifferentialOperatorOfOrder (k : Type u) (B : Type v) [CommRing k] [CommRing B]
387     [Algebra k B] (N : ℕ) (P : Module.End k B) : Prop :=
388   ∀ b : List B, b.length = N + 1 →
389     b.foldl (fun Q c => Q * LinearMap.mulLeft k c - LinearMap.mulLeft k c * Q) P = 0
390 
391 /-- `P` is a `k`-linear differential operator of finite order on `B`. -/
392 def IsDifferentialOperator (k : Type u) (B : Type v) [CommRing k] [CommRing B]
393     [Algebra k B] (P : Module.End k B) : Prop :=
394   ∃ N, IsDifferentialOperatorOfOrder k B N P

396 /-- The identity `1 = dR + FdS` in the ring `D_k(B)` of finite-order
397 `k`-linear differential operators on `B`: for every nonzero `d ∈ D_k(B)` there
398 are `F, R, S ∈ D_k(B)` with `1 = dR + FdS`. -/
399 def HasDifferentialIdentity (k : Type u) (B : Type v) [CommRing k] [CommRing B]
400     [Algebra k B] : Prop :=
401   ∀ d : Module.End k B, IsDifferentialOperator k B d → d ≠ 0 →
402     ∃ F R S : Module.End k B, IsDifferentialOperator k B F ∧ IsDifferentialOperator k B R ∧
403       IsDifferentialOperator k B S ∧ (1 : Module.End k B) = d * R + F * d * S
