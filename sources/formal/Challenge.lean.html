9 /-!
10 # Stafford's Conjecture 3.8
11 
12 J. T. Stafford, *Module structure of Weyl algebras*, J. London Math. Soc. (2)
13 18 (1978), 429–442, states on page 438:
14 
15 > Conjecture 3.8. Let `d ∈ Aₙ`. Then there exists `f ∈ Aₙ` such that
16 > `Aₙ = d Aₙ + f d Aₙ`.
17 
18 Here `Aₙ = Aₙ(k)` is the `n`th Weyl algebra over a field `k` of characteristic
19 zero, generated by `x₁, …, xₙ, ∂₁, …, ∂ₙ` with `[∂ᵢ, xⱼ] = δᵢⱼ` and all other
20 pairs of generators commuting. Both `d Aₙ` and `f d Aₙ` are right ideals, and
21 `f` stands to the left of `d`. Membership of `1` in their sum is the identity
22 `1 = d R + f d S` for some `R, S ∈ Aₙ`, with the factors in this order.
23 
24 The printed conjecture omits the hypothesis `d ≠ 0`, without which the
25 conclusion fails for `d = 0`; the discussion following it in the article uses
26 that hypothesis. This file states the intended nonzero form, for every
27 characteristic-zero field and every rank `n ≥ 0`. For `n = 0` the Weyl algebra
28 is the field itself and a nonzero `d` is a unit.
29 
30 The presentation and statement below repeat the exact definitions used by
31 `Stafford38.ChallengeDefinitions` in the solution. Palomar requires the selected
32 Challenge to import only authenticated libraries. The independent Comparator
33 checks that the definitions and theorem type match; this file contains the
34 deliberate placeholder filled by the solution.
35 -/

42 /-- Linear combination of an indexed family by the rows of a matrix. -/
43 def linearCombination (M : Matrix ι ι k) (z : ι → A) (i : ι) : A :=
44   ∑ j, algebraMap k A (M i j) * z j

48 /-- The commutator relations presenting a Weyl-type algebra. -/
49 def freeWeylRelation (omega : Matrix ι ι k)
50     (a b : FreeAlgebra k ι) : Prop :=
51   ∃ i j,
52     a = FreeAlgebra.ι k i * FreeAlgebra.ι k j -
53       FreeAlgebra.ι k j * FreeAlgebra.ι k i ∧
54     b = algebraMap k (FreeAlgebra k ι) (omega i j)
55 
56 /-- The free algebra modulo the commutator relations encoded by `omega`. -/
57 abbrev FreeWeyl (k : Type*) [Field k] (ι : Type*)
58     (omega : Matrix ι ι k) := RingQuot (freeWeylRelation omega)
59 
60 /-- The canonical generators of the presented Weyl-type algebra. -/
61 def freeWeylGenerator (omega : Matrix ι ι k) (i : ι) : FreeWeyl k ι omega :=
62   RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i)
63 
64 end Stafford

139 /-- Insert an old free-algebra word into the next-rank free algebra. -/
140 def freeOldMap (k : Type u) [Field k] (n : ℕ) :

144 /-- Ordered PBW words, with old pairs followed by the newest pair. -/
145 def freeOrderedMonomial (k : Type u) [Field k] :
146     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → FreeAlgebra k (PhaseVar n)
147   | 0, _, _ => 1
148   | n + 1, a, p =>
149       freeOldMap k n
150           (freeOrderedMonomial k n (fun i => a i.succ) (fun i => p i.succ)) *
151         FreeAlgebra.ι k (.inl (0 : Fin (n + 1))) ^ a 0 *
152         FreeAlgebra.ι k (.inr (0 : Fin (n + 1))) ^ p 0

160 /-- Total degree of a split coordinate/momentum exponent. -/
161 def phaseDegree {n : ℕ} (a p : Fin n → ℕ) : ℕ :=
162   (∑ i, a i) + ∑ i, p i

164 /-- Span of the ordered Weyl words of total degree at most `N`. -/
165 def bernsteinPiece (k : Type u) [Field k] (n N : ℕ) :
166     Submodule k (WeylAlg k n) :=
167   Submodule.span k
168     {z | ∃ a p : Fin n → ℕ,
169       phaseDegree a p ≤ N ∧ z = orderedMonomial k n a p}

185 /-- A source coordinate obtained from an invertible symplectic linear change
186 of the canonical Weyl generators. -/
187 def IsLinearWeylCoordinate (k : Type u) [Field k] (n : ℕ)
188     (ell : WeylAlg k (n + 1)) : Prop :=
189   ∃ (M N : Matrix (PhaseVar (n + 1)) (PhaseVar (n + 1)) k)
190       (_hM : M * standardForm k (n + 1) * Matrix.transpose M =
191         standardForm k (n + 1))
192       (_hN : N * standardForm k (n + 1) * Matrix.transpose N =
193         standardForm k (n + 1))
194       (_hMN : M * N = 1) (_hNM : N * M = 1),
195       ell = linearCombination k N (generator k (n + 1))
196         (.inl (0 : Fin (n + 1)))

198 /-- Shared template for the fixed-source proposition, parameterized only by

201     (degree : ∀ (k : Type u) [Field k] {n : ℕ}, WeylAlg k n → ℕ) : Prop :=

240 /-- An `ω`-preserving `g` (`g J gᵀ = J`) preserves the Weyl relations
241 `[u_i, u_j] = J_ij`. -/
242 theorem linearImage_rel (k : Type*) [Field k] {n : ℕ}
243     (g : Matrix.symplecticGroup (Fin n) k) ⦃a b : FreeAlgebra k (PhaseVar n)⦄
244     (h : relation (Matrix.J (Fin n) k) a b) :
245     FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)) a =
246       FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)) b := by
247   obtain ⟨i, j, rfl, rfl⟩ := h
248   have hcomm : ∀ p q : PhaseVar n,
249       generator k n p * generator k n q - generator k n q * generator k n p =
250         algebraMap k (WeylAlg k n) (Matrix.J (Fin n) k p q) := by
251     intro p q
252     have := RingQuot.mkAlgHom_rel k
253       (show relation (Matrix.J (Fin n) k) _ _ from ⟨p, q, rfl, rfl⟩)
254     simpa [generator, Stafford.freeWeylGenerator, map_sub, map_mul] using this
255   have hg := (SymplecticGroup.mem_iff).mp g.2
256   simp only [map_sub, map_mul, FreeAlgebra.lift_ι_apply, AlgHom.commutes, linearImage,
257     Stafford.linearCombination, ← Algebra.smul_def, Finset.sum_mul, Finset.mul_sum,
258     smul_mul_smul_comm]
259   rw [Finset.sum_comm, ← Finset.sum_sub_distrib]
260   simp_rw [← Finset.sum_sub_distrib]
261   have hterm : ∀ p q : PhaseVar n,
262       ((g : Matrix _ _ k) i p * (g : Matrix _ _ k) j q) • (generator k n p * generator k n q) -
263         ((g : Matrix _ _ k) j q * (g : Matrix _ _ k) i p) • (generator k n q * generator k n p) =
264       algebraMap k (WeylAlg k n)
265         ((g : Matrix _ _ k) i p * Matrix.J (Fin n) k p q * (g : Matrix _ _ k) j q) := by
266     intro p q
267     rw [mul_comm ((g : Matrix _ _ k) j q)]
268     rw [show ∀ (c : k) (x y : WeylAlg k n), c • x - c • y = c • (x - y) from
269       fun c x y => (smul_sub c x y).symm, hcomm, Algebra.smul_def, ← map_mul]
270     congr 1
271     ring
272   simp_rw [hterm, ← map_sum]
273   congr 1
274   conv_rhs => rw [← hg]
275   simp only [Matrix.mul_apply, Matrix.transpose_apply, Finset.sum_mul]
276   rw [Finset.sum_comm]

278 /-- **`Φ_g`**: for a linear `ω`-preserving automorphism `g` of the generator
279 space (`g J gᵀ = J`, i.e. `g ∈ Sp(2n, k)`), the algebra automorphism of
280 `A_n(k)` extending `u_i ↦ g(u_i) = ∑_j g_ij u_j`.  Its inverse is
281 `Φ_g⁻¹ = Φ_{g⁻¹}`. -/

283     WeylAlg k n →ₐ[k] WeylAlg k n :=
284   RingQuot.liftAlgHom k
285     ⟨FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)),
286       fun _ _ h => linearImage_rel k g h⟩

288 /-- A right module `M` is cyclic: `M = mA` for some `m ∈ M`. -/
289 def IsCyclicModule (A : Type u) (M : Type v) [Ring A] [AddCommGroup M] [Module Aᵐᵒᵖ M] :
290     Prop :=
291   ∃ m : M, ∀ x : M, ∃ a : A, x = MulOpposite.op a • m

293 /-- **Theorem 1.1.**  For every field of characteristic zero `k`, every `n ≥ 0`
294 and every nonzero `d ∈ A_n(k)`, there exists an `F ∈ A_n(k)` such that
295 `A_n(k) = dA_n(k) + FdA_n(k)`, or equivalently, there exist `F, S, R ∈ A_n(k)`
296 such that `1 = dR + FdS`.  Moreover, if `n ≥ 1`, one may choose
297 `F = Φ_g^{-1}(x₁)^{deg_B d}` for a linear `ω`-preserving automorphism `g`. -/
298 def PaperTheorem_1_1 : Prop :=
299   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (d : WeylAlg k n), d ≠ 0 →
300     (∃ F : WeylAlg k n, (⊤ : Submodule (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n)) =
301         Submodule.span (WeylAlg k n)ᵐᵒᵖ {d} ⊔ Submodule.span (WeylAlg k n)ᵐᵒᵖ {F * d}) ∧
302       (∃ F S R : WeylAlg k n, (1 : WeylAlg k n) = d * R + F * d * S) ∧
303       (∀ hn : 1 ≤ n, ∃ g : Matrix.symplecticGroup (Fin n) k, ∃ R S : WeylAlg k n,
304         (1 : WeylAlg k n) = d * R +
305           phiG k n g⁻¹ (x₁ k hn) ^ Stafford38FixedSourceChallenge.bernsteinDegree k d * d * S)
306
