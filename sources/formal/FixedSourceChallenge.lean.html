9 /-!
10 # Stafford's exact fixed-source strengthening
11 
12 This file states the exact fixed-source challenge and repeats the definitions
13 used by `Stafford38.ChallengeDefinitions` in the solution. It imports only
14 authenticated libraries, as required by Palomar; Comparator checks the exact
15 definitions and statement type against the solution.
16 The Bernstein filtration is the span of images of the ordered PBW words of
17 bounded total word degree, and `bernsteinDegree` is its least membership
18 level. Thus the exponent is attached to the presented element itself; it is
19 not an extra degree or supplied normal-form datum.
20 -/

27 /-- Linear combination of an indexed family by the rows of a matrix. -/
28 def linearCombination (M : Matrix ι ι k) (z : ι → A) (i : ι) : A :=
29   ∑ j, algebraMap k A (M i j) * z j

31 variable {k ι : Type*} [Field k]
32 
33 /-- The commutator relations presenting a Weyl-type algebra. -/
34 def freeWeylRelation (omega : Matrix ι ι k)
35     (a b : FreeAlgebra k ι) : Prop :=
36   ∃ i j,
37     a = FreeAlgebra.ι k i * FreeAlgebra.ι k j -
38       FreeAlgebra.ι k j * FreeAlgebra.ι k i ∧
39     b = algebraMap k (FreeAlgebra k ι) (omega i j)
40 
41 /-- The free algebra modulo the commutator relations encoded by `omega`. -/
42 abbrev FreeWeyl (k : Type*) [Field k] (ι : Type*)
43     (omega : Matrix ι ι k) := RingQuot (freeWeylRelation omega)
44 
45 /-- The canonical generators of the presented Weyl-type algebra. -/
46 def freeWeylGenerator (omega : Matrix ι ι k) (i : ι) : FreeWeyl k ι omega :=
47   RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i)
48 
49 end Stafford

113 /-- Insert an old phase variable after the newest coordinate or momentum. -/
114 def oldIndex {n : ℕ} : PhaseVar n → PhaseVar (n + 1)
115   | .inl i => .inl i.succ
116   | .inr i => .inr i.succ
117 
118 @[simp] theorem oldIndex_inl {n : ℕ} (i : Fin n) :
119     oldIndex (.inl i) = .inl i.succ := rfl
120 
121 @[simp] theorem oldIndex_inr {n : ℕ} (i : Fin n) :
122     oldIndex (.inr i) = .inr i.succ := rfl

124 /-- Insert an old free-algebra word into the next-rank free algebra. -/
125 def freeOldMap (k : Type u) [Field k] (n : ℕ) :
126     FreeAlgebra k (PhaseVar n) →ₐ[k] FreeAlgebra k (PhaseVar (n + 1)) :=
127   FreeAlgebra.lift k (fun i => FreeAlgebra.ι k (oldIndex i))
128 
129 /-- Ordered PBW words, with old pairs followed by the newest pair. -/
130 def freeOrderedMonomial (k : Type u) [Field k] :
131     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → FreeAlgebra k (PhaseVar n)
132   | 0, _, _ => 1
133   | n + 1, a, p =>
134       freeOldMap k n
135           (freeOrderedMonomial k n (fun i => a i.succ) (fun i => p i.succ)) *
136         FreeAlgebra.ι k (.inl (0 : Fin (n + 1))) ^ a 0 *
137         FreeAlgebra.ι k (.inr (0 : Fin (n + 1))) ^ p 0
138 
139 /-- The image of an ordered PBW word in the shared Weyl quotient. -/
140 def orderedMonomial (k : Type u) [Field k] (n : ℕ)
141     (a p : Fin n → ℕ) : WeylAlg k n :=
142   RingQuot.mkAlgHom k (relation (k := k) (n := n) (Matrix.J (Fin n) k))
143     (freeOrderedMonomial k n a p)

145 /-- Total degree of a split coordinate/momentum exponent. -/
146 def phaseDegree {n : ℕ} (a p : Fin n → ℕ) : ℕ :=
147   (∑ i, a i) + ∑ i, p i
148 
149 /-- Span of the ordered Weyl words of total degree at most `N`. -/
150 def bernsteinPiece (k : Type u) [Field k] (n N : ℕ) :
151     Submodule k (WeylAlg k n) :=
152   Submodule.span k
153     {z | ∃ a p : Fin n → ℕ,
154       phaseDegree a p ≤ N ∧ z = orderedMonomial k n a p}
155 
156 /-- The intrinsic Bernstein degree of a presented element: its least piece. -/
157 noncomputable def bernsteinDegree (k : Type u) [Field k] {n : ℕ}
158     (d : WeylAlg k n) : ℕ :=
159   sInf {N : ℕ | d ∈ bernsteinPiece k n N}

170 /-- A source coordinate obtained from an invertible symplectic linear change
171 of the canonical Weyl generators. -/
172 def IsLinearWeylCoordinate (k : Type u) [Field k] (n : ℕ)
173     (ell : WeylAlg k (n + 1)) : Prop :=
174   ∃ (M N : Matrix (PhaseVar (n + 1)) (PhaseVar (n + 1)) k)
175       (_hM : M * standardForm k (n + 1) * Matrix.transpose M =
176         standardForm k (n + 1))
177       (_hN : N * standardForm k (n + 1) * Matrix.transpose N =
178         standardForm k (n + 1))
179       (_hMN : M * N = 1) (_hNM : N * M = 1),
180       ell = linearCombination k N (generator k (n + 1))
181         (.inl (0 : Fin (n + 1)))

183 /-- Shared template for the fixed-source proposition, parameterized only by
184 the chosen degree function. -/
185 def UniversalFixedSourceStatementWithDegree
186     (degree : ∀ (k : Type u) [Field k] {n : ℕ}, WeylAlg k n → ℕ) : Prop :=
187   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
188     (d : WeylAlg k (n + 1)), d ≠ 0 →
189       ∃ ell R S : WeylAlg k (n + 1),
190         IsLinearWeylCoordinate k n ell ∧
191           (1 : WeylAlg k (n + 1)) =
192             d * R + ell ^ degree k d * d * S
193 
194 /-- The exact fixed-source challenge, using the intrinsic degree. -/
195 abbrev UniversalFixedSourceStatement : Prop :=
196   UniversalFixedSourceStatementWithDegree.{u} bernsteinDegree

204 /-- Every element of a right module is killed by some nonzero right scalar. -/
205 def IsRightTorsion {A : Type u} {M : Type v} [Ring A]
206     [AddCommGroup M] [Module Aᵐᵒᵖ M] : Prop :=

219 /-- The linear map `g` on the generator space `V = span(x_1, …, ∂_n)`, given by

221 abbrev linearImage (k : Type*) [Field k] {n : ℕ}
222     (g : Matrix (PhaseVar n) (PhaseVar n) k) (i : PhaseVar n) : WeylAlg k n :=
223   Stafford.linearCombination g (generator k n) i

225 /-- An `ω`-preserving `g` (`g J gᵀ = J`) preserves the Weyl relations
226 `[u_i, u_j] = J_ij`. -/
227 theorem linearImage_rel (k : Type*) [Field k] {n : ℕ}
228     (g : Matrix.symplecticGroup (Fin n) k) ⦃a b : FreeAlgebra k (PhaseVar n)⦄
229     (h : relation (Matrix.J (Fin n) k) a b) :
230     FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)) a =
231       FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)) b := by
232   obtain ⟨i, j, rfl, rfl⟩ := h
233   have hcomm : ∀ p q : PhaseVar n,
234       generator k n p * generator k n q - generator k n q * generator k n p =
235         algebraMap k (WeylAlg k n) (Matrix.J (Fin n) k p q) := by
236     intro p q
237     have := RingQuot.mkAlgHom_rel k
238       (show relation (Matrix.J (Fin n) k) _ _ from ⟨p, q, rfl, rfl⟩)
239     simpa [generator, Stafford.freeWeylGenerator, map_sub, map_mul] using this
240   have hg := (SymplecticGroup.mem_iff).mp g.2
241   -- Reversed simp lemmas (`← l`) would become helpers named after the compiling
242   -- module, so challenge and solution copies of this proof would differ.
243   have smul_def' : ∀ (c : k) (x : WeylAlg k n), algebraMap k (WeylAlg k n) c * x = c • x :=
244     fun c x => (Algebra.smul_def c x).symm
245   have sum_sub' : ∀ (s : Finset (PhaseVar n)) (f f' : PhaseVar n → WeylAlg k n),
246       ∑ p ∈ s, f p - ∑ p ∈ s, f' p = ∑ p ∈ s, (f p - f' p) :=
247     fun s f f' => (Finset.sum_sub_distrib f f').symm
248   have sum_algebraMap : ∀ (s : Finset (PhaseVar n)) (f : PhaseVar n → k),
249       ∑ p ∈ s, algebraMap k (WeylAlg k n) (f p) = algebraMap k (WeylAlg k n) (∑ p ∈ s, f p) :=
250     fun s f => (map_sum (algebraMap k (WeylAlg k n)) f s).symm
251   simp only [map_sub, map_mul, FreeAlgebra.lift_ι_apply, AlgHom.commutes, linearImage,
252     Stafford.linearCombination, smul_def', Finset.sum_mul, Finset.mul_sum,
253     smul_mul_smul_comm]
254   rw [Finset.sum_comm, ← Finset.sum_sub_distrib]
255   simp_rw [sum_sub']
256   have hterm : ∀ p q : PhaseVar n,
257       ((g : Matrix _ _ k) i p * (g : Matrix _ _ k) j q) • (generator k n p * generator k n q) -
258         ((g : Matrix _ _ k) j q * (g : Matrix _ _ k) i p) • (generator k n q * generator k n p) =
259       algebraMap k (WeylAlg k n)
260         ((g : Matrix _ _ k) i p * Matrix.J (Fin n) k p q * (g : Matrix _ _ k) j q) := by
261     intro p q
262     rw [mul_comm ((g : Matrix _ _ k) j q)]
263     rw [show ∀ (c : k) (x y : WeylAlg k n), c • x - c • y = c • (x - y) from
264       fun c x y => (smul_sub c x y).symm, hcomm, Algebra.smul_def, ← map_mul]
265     congr 1
266     ring
267   simp_rw [hterm, sum_algebraMap]
268   congr 1
269   conv_rhs => rw [← hg]
270   simp only [Matrix.mul_apply, Matrix.transpose_apply, Finset.sum_mul]
271   rw [Finset.sum_comm]

273 /-- **`Φ_g`**: for a linear `ω`-preserving automorphism `g` of the generator
274 space (`g J gᵀ = J`, i.e. `g ∈ Sp(2n, k)`), the algebra automorphism of
275 `A_n(k)` extending `u_i ↦ g(u_i) = ∑_j g_ij u_j`.  Its inverse is
276 `Φ_g⁻¹ = Φ_{g⁻¹}`. -/
277 noncomputable def phiG (k : Type*) [Field k] (n : ℕ) (g : Matrix.symplecticGroup (Fin n) k) :
278     WeylAlg k n →ₐ[k] WeylAlg k n :=
279   RingQuot.liftAlgHom k
280     ⟨FreeAlgebra.lift k (linearImage k (g : Matrix (PhaseVar n) (PhaseVar n) k)),
281       fun _ _ h => linearImage_rel k g h⟩

283 /-- A right module `M` is cyclic: `M = mA` for some `m ∈ M`. -/
284 def IsCyclicModule (A : Type u) (M : Type v) [Ring A] [AddCommGroup M] [Module Aᵐᵒᵖ M] :
285     Prop :=
286   ∃ m : M, ∀ x : M, ∃ a : A, x = MulOpposite.op a • m

288 /-- **Theorem 1.1.**  For every field of characteristic zero `k`, every `n ≥ 0`
289 and every nonzero `d ∈ A_n(k)`, there exists an `F ∈ A_n(k)` such that
290 `A_n(k) = dA_n(k) + FdA_n(k)`, or equivalently, there exist `F, S, R ∈ A_n(k)`
291 such that `1 = dR + FdS`.  Moreover, if `n ≥ 1`, one may choose
292 `F = Φ_g^{-1}(x₁)^{deg_B d}` for a linear `ω`-preserving automorphism `g`. -/
293 def PaperTheorem_1_1 : Prop :=
294   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (d : WeylAlg k n), d ≠ 0 →
295     (∃ F : WeylAlg k n, (⊤ : Submodule (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n)) =
296         Submodule.span (WeylAlg k n)ᵐᵒᵖ {d} ⊔ Submodule.span (WeylAlg k n)ᵐᵒᵖ {F * d}) ∧
297       (∃ F S R : WeylAlg k n, (1 : WeylAlg k n) = d * R + F * d * S) ∧
298       (∀ hn : 1 ≤ n, ∃ g : Matrix.symplecticGroup (Fin n) k, ∃ R S : WeylAlg k n,
299         (1 : WeylAlg k n) = d * R +
300           phiG k n g⁻¹ (x₁ k hn) ^ Stafford38FixedSourceChallenge.bernsteinDegree k d * d * S)
301 
302 /-- **Corollary 6.3** (Stafford 1978).  Every finitely generated torsion right
303 `A_n(k)`-module is cyclic. -/
304 def PaperCorollary_6_3 : Prop :=
305   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (M : Type v) [AddCommGroup M]
306     [Module (WeylAlg k n)ᵐᵒᵖ M] [Module.Finite (WeylAlg k n)ᵐᵒᵖ M],
307     Stafford38.TorsionCyclicity.IsRightTorsion (A := WeylAlg k n) (M := M) →
308       IsCyclicModule (WeylAlg k n) M
