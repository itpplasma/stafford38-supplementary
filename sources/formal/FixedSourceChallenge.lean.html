9 /-!
10 # Stafford's exact fixed-source strengthening
11 
12 This file states the exact fixed-source challenge and repeats the definitions
13 used by `Stafford38.ChallengeDefinitions` in the solution. It imports only
14 authenticated libraries, as required by Palomar; Comparator checks the exact
15 definitions and statement type against the solution.
16 The Bernstein filtration is the span of images of the ordered PBW words of
17 bounded total word degree, and `bernsteinDegree` is its least membership
18 level. Thus the exponent is attached to the presented element itself; it is
19 not an extra degree or supplied normal-form datum.
20 -/

27 /-- Linear combination of an indexed family by the rows of a matrix. -/
28 def linearCombination (M : Matrix ι ι k) (z : ι → A) (i : ι) : A :=
29   ∑ j, algebraMap k A (M i j) * z j
30 
31 variable {k ι : Type*} [Field k]
32 
33 /-- The commutator relations presenting a Weyl-type algebra. -/
34 def freeWeylRelation (omega : Matrix ι ι k)
35     (a b : FreeAlgebra k ι) : Prop :=
36   ∃ i j,
37     a = FreeAlgebra.ι k i * FreeAlgebra.ι k j -
38       FreeAlgebra.ι k j * FreeAlgebra.ι k i ∧
39     b = algebraMap k (FreeAlgebra k ι) (omega i j)
40 
41 /-- The free algebra modulo the commutator relations encoded by `omega`. -/
42 abbrev FreeWeyl (k : Type*) [Field k] (ι : Type*)
43     (omega : Matrix ι ι k) := RingQuot (freeWeylRelation omega)
44 
45 /-- The canonical generators of the presented Weyl-type algebra. -/
46 def freeWeylGenerator (omega : Matrix ι ι k) (i : ι) : FreeWeyl k ι omega :=
47   RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i)
48 
49 end Stafford

56 /-- Stafford 3.8 for one element, in written operator order. -/
57 def Stafford38 (e : A) : Prop := ∃ F R S : A, (1 : A) = e * R + F * e * S

66 /-- Index type of the coordinate and momentum generators. -/
67 abbrev PhaseVar (n : ℕ) := Fin n ⊕ Fin n
68 
69 /-- The Weyl commutator relation for the standard symplectic matrix. -/
70 abbrev relation {k : Type*} [Field k] {n : ℕ}
71     (omega : Matrix (PhaseVar n) (PhaseVar n) k)
72     (a b : FreeAlgebra k (PhaseVar n)) : Prop :=
73   Stafford.freeWeylRelation omega a b
74 
75 /-- The standard presented Weyl algebra, shared with the substantive proof. -/
76 abbrev WeylAlg (k : Type*) [Field k] (n : ℕ) :=
77   Stafford.FreeWeyl k (PhaseVar n) (Matrix.J (Fin n) k)
78 
79 /-- Stafford's Conjecture 3.8 in its intended nonzero form. -/
80 def UniversalStatement : Prop :=
81   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (d : WeylAlg k n),
82     d ≠ 0 → Stafford.Reduction.Stafford38 d

84 /-- A named generator in the canonical presentation. -/

102 /-- Insert an old phase variable after the newest coordinate or momentum. -/
103 def oldIndex {n : ℕ} : PhaseVar n → PhaseVar (n + 1)
104   | .inl i => .inl i.succ
105   | .inr i => .inr i.succ
106 
107 @[simp] theorem oldIndex_inl {n : ℕ} (i : Fin n) :
108     oldIndex (.inl i) = .inl i.succ := rfl
109 
110 @[simp] theorem oldIndex_inr {n : ℕ} (i : Fin n) :
111     oldIndex (.inr i) = .inr i.succ := rfl
112 
113 /-- Insert an old free-algebra word into the next-rank free algebra. -/
114 def freeOldMap (k : Type u) [Field k] (n : ℕ) :
115     FreeAlgebra k (PhaseVar n) →ₐ[k] FreeAlgebra k (PhaseVar (n + 1)) :=
116   FreeAlgebra.lift k (fun i => FreeAlgebra.ι k (oldIndex i))
117 
118 /-- Ordered PBW words, with old pairs followed by the newest pair. -/
119 def freeOrderedMonomial (k : Type u) [Field k] :
120     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → FreeAlgebra k (PhaseVar n)
121   | 0, _, _ => 1
122   | n + 1, a, p =>
123       freeOldMap k n
124           (freeOrderedMonomial k n (fun i => a i.succ) (fun i => p i.succ)) *
125         FreeAlgebra.ι k (.inl (0 : Fin (n + 1))) ^ a 0 *
126         FreeAlgebra.ι k (.inr (0 : Fin (n + 1))) ^ p 0
127 
128 /-- The image of an ordered PBW word in the shared Weyl quotient. -/
129 def orderedMonomial (k : Type u) [Field k] (n : ℕ)
130     (a p : Fin n → ℕ) : WeylAlg k n :=
131   RingQuot.mkAlgHom k (relation (k := k) (n := n) (Matrix.J (Fin n) k))
132     (freeOrderedMonomial k n a p)
133 
134 /-- Total degree of a split coordinate/momentum exponent. -/
135 def phaseDegree {n : ℕ} (a p : Fin n → ℕ) : ℕ :=
136   (∑ i, a i) + ∑ i, p i
137 
138 /-- Span of the ordered Weyl words of total degree at most `N`. -/
139 def bernsteinPiece (k : Type u) [Field k] (n N : ℕ) :
140     Submodule k (WeylAlg k n) :=
141   Submodule.span k
142     {z | ∃ a p : Fin n → ℕ,
143       phaseDegree a p ≤ N ∧ z = orderedMonomial k n a p}
144 
145 /-- The intrinsic Bernstein degree of a presented element: its least piece. -/
146 noncomputable def bernsteinDegree (k : Type u) [Field k] {n : ℕ}
147     (d : WeylAlg k n) : ℕ :=
148   sInf {N : ℕ | d ∈ bernsteinPiece k n N}
149 
150 /-- A source coordinate obtained from an invertible symplectic linear change
151 of the canonical Weyl generators. -/
152 def IsLinearWeylCoordinate (k : Type u) [Field k] (n : ℕ)
153     (ell : WeylAlg k (n + 1)) : Prop :=
154   ∃ (M N : Matrix (PhaseVar (n + 1)) (PhaseVar (n + 1)) k)
155       (_hM : M * standardForm k (n + 1) * Matrix.transpose M =
156         standardForm k (n + 1))
157       (_hN : N * standardForm k (n + 1) * Matrix.transpose N =
158         standardForm k (n + 1))
159       (_hMN : M * N = 1) (_hNM : N * M = 1),
160       ell = Stafford.linearCombination N (generator k (n + 1))
161         (.inl (0 : Fin (n + 1)))
162 
163 /-- The exact fixed-source challenge, using the canonical intrinsic degree. -/
164 def UniversalFixedSourceStatement : Prop :=
165   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
166     (d : WeylAlg k (n + 1)), d ≠ 0 →
167       ∃ ell R S : WeylAlg k (n + 1),
168         IsLinearWeylCoordinate k n ell ∧
169           (1 : WeylAlg k (n + 1)) =
170             d * R + ell ^ bernsteinDegree k d * d * S
171 
172 end Stafford38FixedSourceChallenge
