21 /-- Every finitely generated torsion right module over a characteristic-zero
22 Weyl algebra is generated by one element. -/
23 def torsionCyclicStatement : Prop :=
24   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
25     (M : Type v) [AddCommGroup M]
26     [Module (WeylAlg k n)ᵐᵒᵖ M]
27     [Module.Finite (WeylAlg k n)ᵐᵒᵖ M],
28     IsRightTorsion (A := WeylAlg k n) (M := M) →
29       ∃ z : M, Submodule.span (WeylAlg k n)ᵐᵒᵖ ({z} : Set M) = ⊤
30 
31 end Stafford38CorollaryChallenge
