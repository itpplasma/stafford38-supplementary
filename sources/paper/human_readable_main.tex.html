53 \begin{abstract}
54 Let $A_n(k)$ be the Weyl algebra over a field of characteristic zero.  We prove
55 that every nonzero $d\in A_n(k)$ admits $F,R,S\in A_n(k)$ such that
56 $1=dR+FdS$.  Thus Stafford's Conjecture 3.8 holds.  \AIadd{For $n\ge1$,} \AIreplace{The}{the} multiplier $F$ may be
57 chosen as $x^N$, after a linear symplectic change of variables, where $N$ is
58 the Bernstein degree of $d$.  The proof studies a cyclic quotient through its
59 characteristic variety and excludes nonzero quotients by conormal geometry at
60 infinity.
61 \end{abstract}

96 For every characteristic-zero field $k$, every $n\geq1$, and every
97 nonzero $d\in A_n(k)$, there exist $F,R,S\in A_n(k)$ such that

124 \textit{Proof sketch:}
125 Give each standard generator Bernstein degree one and
126 let $N=\deg_B d$.  A linear symplectic change of variables makes $d$ monic of
127 degree $N$ in a momentum $p$, whose conjugate coordinate we denote by $x$.  It is
128 enough to prove that the cyclic right module
129 \begin{equation} Q=A_n/(dA_n+x^NdA_n).
130  \label{eq:Q-intro}\end{equation}
131 vanishes.  An Euler identity makes $x$ surjective on $Q$, while monicity makes
132 $H=\{x=0\}$ noncharacteristic and places $\Char(Q)$ in a hypersurface
133 $V(P)$ with $P(dx)\ne0$.  A filtered-page length argument shows that
134 $\Char(Q)$ has no point over $H$.  Gabber's theorem and the conormal
135 argument of \AIreplace{Sections 4 and 5}{Sections~\ref{sec:conormal-review}
136 and~\ref{sec:involutive-review}} then force $\Char(Q)$ to be empty.  The same
137 argument gives the more precise identity $A_n=dA_n+x^NdA_n$ in the chosen
138 symplectic coordinates.

215 \begin{convention}\label{conv:pair}
216 We single out the first Weyl pair $(x_1, \partial_1)$ and write

230 \begin{definition}[Filtration {[Def.~2.6 and ~2.9 \cite{SchnellNotes}]}]\label{def:filtration}
231 A filtration $F_\bullet$ on $A_n(k)$ is a sequence of linear subspaces
232 \[\{0\} = F_{-1} \subseteq F_0 \subseteq F_1\subseteq...\subseteq A_n(k),\]
233 such that $F_i\cdot F_j \subseteq F_{i+j}$ and $A_n(k) = \bigcup F_j.$
234 The associated graded algebra, also called symbol ring, is defined as
235 \[\mathrm{gr}^F A_n(k) = \bigoplus_{j = 0}^\infty F_j/F_{j-1}.\]
236 Let $a\neq 0 \in A_n(k)$ and let $j$ be the smallest integer, such that $a \in F_j$, called a \emph{filtration degree} $\deg_F a$. The class of $a$ in $F_j/F_{j-1}$ is called its \emph{symbol} $\sigma_j(a)$.
237 \end{definition}
238 
239 In the proof, we use two filtrations of $A_n(k)$.
240 \begin{itemize}[nosep,leftmargin=1.4em]
241 \item The \emph{Bernstein filtration} $B_\bullet$ assigns degree $1$ to every
242   $x_i$ and every $\partial_i$. The Bernstein degree of $d$ is
243   written $\deg_B d$.
244 \item The \emph{order filtration} $F_\bullet$ assigns degree $0$ to every $x_i$
245   and degree $1$ to every $\partial_i$; $F_m$ is spanned by the monomials
246   $x^\alpha\partial^\beta$ with $|\beta|\le m$.  %We write $\sigma_m(a)$ for the order-$m$ principal symbol.
247 \end{itemize}

256 \subsection{Cotangent space and symplectic structure}\label{sec:cotangent}

259 Let $X=\mathbb A^n_k$ be affine $n$-space over a field $k$ of characteristic zero, with coordinates
260 $x_1,\ldots,x_n$. Its \emph{cotangent bundle} $T^*X$ is the space of pairs
261 $(y,\xi)$ consisting of a point $y\in X$ together with a covector $\xi$
262 at $y$; since $X$ is affine space, this cotangent space is the same
263 vector space $(k^n)^*$ at every point, and we may identify
264 \[
265  T^*X \;\cong\; X\times(k^n)^*.
266 \]
267 We write $\xi_1,\ldots,\xi_n$ for the coordinates on $(k^n)^*$ dual to
268 $x_1,\ldots,x_n$, so that a covector is $\sum_i\xi_i\,dx_i$, and identify
269 $\gr^BA_n(k)=k[x,\xi]$ with the coordinate ring of $T^*X$. We write
270 $\pi\colon T^*X\to X$ for the projection forgetting the covector, and for
271 $y\in X$ write $T_y^*X=\pi^{-1}(y)\cong(k^n)^*$ for the \emph{fibre} of
272 $T^*X$ over $y$.

284 \begin{definition}\label{def:good-filtration}
285 A filtration of a right $A_n$-module $M$ compatible with $F_\bullet$
286 is an exhaustive increasing chain of $k$-subspaces $M_\bullet$ with
287 $M_iF_j\subseteq M_{i+j}$ and $M_i=0$ for $i<0$. It is \emph{good} if the associated graded module
288 $\gr M=\bigoplus_iM_i/M_{i-1}$ is finitely generated over $\gr^FA_n$.
289 \end{definition}
290 As an example, the filtration $M_i:=qF_i$ of a cyclic module over $A_n$ induced by its generator $q$ is good, since $\gr Q$ is generated by the element $\bar{q}$ denoting the class of $q$ in $Q_0/Q_{-1}.$
291 
292 For a finitely generated right $A_n$-module $M$ with a good order filtration
293 $M_\bullet$, the graded module $\gr M$ is a finitely generated $k[x,\xi]$-module.
294 \begin{definition}\label{def:char-variety}
295     The \emph{characteristic variety} (also called symbol variety) of a finitely generated $A_n$-module $M$ with a good order filtration
296 $M_\bullet$ is defined by
297 \[
298   \Char(M):=V\bigl(\Ann_{k[x,\xi]}\gr M\bigr)\subseteq
299   T^*\bbA^n=\Spec k[x,\xi],
300 \]
301 $\Ann_{k[x,\xi]}\gr M = \{f \in k[x,\xi]: f\cdot m = 0\,\forall m \in \gr M\}$ are the \emph{annihilators} of $\gr M$ over $k[x,\xi]$, and $V$ denotes their \emph{vanishing locus}, i.e. the set of points $(x, \xi)$ for which all annihilator polynomials vanish. This set of points lies in the \emph{cotangent bundle} $T^*\bbA^n$ of the affine $n$-space $\bbA^n.$
302 
303 $\Char(M)$ is independent of the choice of the good filtration.
304 \end{definition}

311 \begin{definition}[Symplectic form and Poisson bracket]\label{def:symplectic-poisson}
312 A bilinear form $\omega$ on a $k$-vector space is \emph{symplectic} if it
313 is antisymmetric and nondegenerate.
314 
315 The standard symplectic form on the vector space $k^{2n}$, with basis
316 $x_1,\ldots,x_n,\partial_1,\ldots,\partial_n$, is determined by the
317 relations
318 \[ \omega(x_i,\partial_j)=\delta_{ij},\qquad
319  \omega(x_i,x_j)=\omega(\partial_i,\partial_j)=0,\]
320 analogously to the Weyl relations. Identifying $T^*X\cong X\times(k^n)^*$ via the coordinates $x_1,\ldots,x_n$ and their duals
321 $\xi_1,\ldots,\xi_n$, on $T^*X$ this $\omega$ takes the familiar form
322 \[ \omega=\sum_{i=1}^n dx_i\wedge d\xi_i.\]
323 
324 The associated \emph{Poisson bracket} on $k[x,\xi]$, the coordinate ring
325 of $T^*X$, is
326 \[ \{f,g\}=\sum_{i=1}^n\left(
327    \frac{\partial f}{\partial x_i}\frac{\partial g}{\partial \xi_i}
328    -\frac{\partial f}{\partial \xi_i}\frac{\partial g}{\partial x_i}
329  \right).\]
330 \end{definition}
331 With this convention of the Poisson bracket, the order-filtration symbol of a commutator is $\sigma^F([a,b])$ is $\{\sigma^F(b),\sigma^F(a)\}$.

357 \section{Mapping to a monic normal form}\label{sec:monic}
358 Given a fixed nonzero $d \in A_n(k)$ of Bernstein degree $N$, our first goal is to find a map to replace $d$ by a normal form $\tilde{d}$ that is monic in $p$, i.e. of the form $\tilde d=p^N+\sum_{j<N}p^ja_j$ with coefficients $a_j$ in $C$. \\
359 
360 If $N=0$, then $d \in B_0 = k$ is a nonzero scalar, hence a unit, and $1 = dR+FdS$ holds trivially with $F=1=\ell^0$, $R=d^{-1}$, $S=0$. We therefore assume $N\geq1$.
361 
362 
363 We want to construct a well-behaved automorphism of $\gr^BA_n(k)=k[x,\xi]$
364 from simpler linear algebra. We proceed in three stages:
365 define an automorphism of the vector space $\mathbb V=k^{2n}$, extend it to
366 an automorphism of $A_n(k)$ itself, and observe the induced action this
367 extension leaves on the associated graded ring.
368 
369 We can identify $\mathbb V = k^{2n}$ with the Bernstein-degree-one part $B_1/B_0$ of
370 $A_n(k)$, i.e.\ linear combinations of $x_1,\ldots,x_n,\partial_1,\ldots,
371 \partial_n$; the coordinates of a vector are then the coefficients of
372 these generators. By this identification, $\mathbb V$ carries the standard symplectic form $\omega$ (Def. \ref{def:symplectic-poisson}). 
373 
374 \begin{lemma}\label{lem:sympcompl}
375 Let $(\mathbb V,\omega)$ be a symplectic $k$-space of dimension $2n$ and $0\neq v\in \mathbb V$.
376 Then $v$ is the first vector of a symplectic basis of $\mathbb V$; consequently, there
377 exists a $k$-linear automorphism $g$ preserving $\omega$ (i.e. $\omega(gu,gv)=\omega(u,v)$ for all
378 $u,v\in\mathbb V$.) with $g(e)=v$ for any prescribed nonzero vector $e$.
379 \end{lemma}
380 
381 \begin{proof}
382 Since $\omega$ is nondegenerate and $v\neq0$, the functional $\omega(v,-)$
383 is not identically zero, so there is $w'\in\mathbb V$ with
384 $\omega(v,w')=c\neq0$; setting $w:=w'/c$ gives $\omega(v,w)=1$.
385 
386 Let $U=\operatorname{span}(v,w)$. For $0\neq av+bw\in U$: if $a\neq0$,
387 $\omega(av+bw,w)=a\neq0$; if $b\neq0$, $\omega(av+bw,v)=-b\neq0$. Either
388 way $av+bw$ pairs nontrivially with something in $U$, so $\omega|_U$ is
389 nondegenerate.
390 
391 Since $\omega|_U$ is nondegenerate, $U\cap U^\perp=0$, and as
392 $\dim U+\dim U^\perp=\dim\mathbb V$ for a nondegenerate form, this gives
393 $\mathbb V=U\oplus U^\perp$. If $u\in U^\perp$ pairs to zero with all of
394 $U^\perp$, it also pairs to zero with $U$ (as $u\in U^\perp$), hence with
395 all of $\mathbb V$, forcing $u=0$; so $\omega|_{U^\perp}$ is nondegenerate
396 too.
397 
398 By induction on $n$ (trivial for $n=1$, where $U=\mathbb V$), a
399 symplectic basis $e_2,\ldots,f_n$ of $U^\perp$ exists; setting $e_1:=v$,
400 $f_1:=w$ gives a symplectic basis of $\mathbb V=U\oplus U^\perp$, since
401 cross-pairings between $U$ and $U^\perp$ vanish by definition of
402 $U^\perp$.
403 
404 Consequently: given nonzero $v,e\in\mathbb V$, we can complete both vectors to a symplectic basis as above. We then define the linear map $g$ to send $e$ to $v$, and each subsequent $e$-basis vector to the corresponding $v$-basis vector, thus preserving all pairwise values of $\omega$ on basis vectors. 1685 $[dx_1]=[1:0:\dots:0]$ (homogeneity makes the vanishing independent of the
1686 representative). Therefore $[dx_1]\in\operatorname{Dir}(Y)$.

1905 \begin{proof}
1906 Suppose we have a nonempty closed subset $W\subseteq T^*X$ that is fibre-conical and involutive. Suppose further, that $W\cap\pi^{-1}(H)=\varnothing$. 
1907 We define the (radical) vanishing ideal $J=I(W)\subseteq K[x,\xi]$ consisting of all polynomials that vanish at every point $(y,\xi)\in W$, and $B=J\cap K[x]$, consisting of the functions $f(x)$ with $f(y)=0$ for every $(y,\xi) \in W$. \\
1908 
1909 Because $W$ is fibre-conical, the $\xi$-homogeneous components of any
1910 polynomial in $J$ again lie in $J$: if $g=\sum_dg_d\in J$ and
1911 $(y,\xi)\in W$, then $\sum_da^dg_d(y,\xi)=g(y,a\xi)=0$ for all $a\in K^\times$,
1912 and a polynomial in $a$ with infinitely many roots is zero, so every
1913 $g_d(y,\xi)=0$. Hence $J$ is $\xi$-homogeneous.\\
1914 
1915 Let $g \in J$ be written in its $\xi$-homogeneous components form, that all lie in $J$. In particular, also $g_0 \in J$. Since $g_0$ has $\xi$-degree $0$, it is a polynomial in $x$ alone, so $g_0 \in B$. Let $Y = V(B)$. For $y \in Y$ we therefore have $g(y, 0) = g_0(y) = 0$, so every $g \in J$ vanishes at $(y,0)$, i.e. $(y,0)\in V(J)$. Since $W$ is closed, $V(J) = V(I(W)) = W$, hence $(y, 0) \in W$, or $Y \times \{0\} \subseteq W$. 
1916 Furthermore, $Y$ is the projection of $W$ and is therefore per definition of $Y$ closed: $\pi(W)\subseteq Y$ since for ($y',\xi')\in W$ and $f \in B\subseteq J$, we have $f(y') = f(y',\xi') = 0$ so $y' \in V(B) = Y$, and $Y\subseteq \pi(W)$, since if $y \in Y$, then also $(y,0)\in W$, so $y = \pi(y,0) \in \pi(W)$.\\
1917 Lastly, assume there exists a $y \in Y\cap H$, then $(y,0)\in W$ and $\pi(y,0) = y \in H$, which would put $(y,0)\in W\cap \pi^{-1}(H)$, which is wrong by our assumption. Therefore, $Y \cap H = \varnothing$.\\
1918 
1919 Let $f\in B$. Then $f\circ\pi\in J$.
1920 The Hamiltonian derivation of $f$ applied to a $a \in J$ is
1921 \[
1922  H_f g=\sum_i\frac{\partial f}{\partial x_i}
1923  \frac{\partial g}{\partial\xi_i} = \{f,g\}
1924 \]
1925 since the second term of the Poisson bracket vanishes because $f$ does not depend on $\xi$. Since $f, g \in J$ and $\{J, J\} \subseteq J$ by the involutivity of $W$, it follows that $H_f g \in J$ and by induction also $H^m_fg \in J$. 
1926 For a fixed $y \in Y$ and $g \in J$, we define the polynomial $\varphi(t)=g(y,t\,df_y)\in K[t]$. Its derivative is $\varphi'(t) = \sum_i\partial_{x_i}f(y)\frac{\partial g}{\partial \xi_i}{(y, tdf_y)} = H_fg(y, tdf_y)$, so by iteration, $\varphi^{(m)}(t) = H^m_fg(y, tdf_y)$ for $m\geq 0.$ 
1927 Evaluating at $t = 0$, $\varphi^{(m)}(0) = H_f^mg(y,0).$ Since we have established that $(y,0)\in W$ and $H^m_fg \in J$, and all polynomials in $J$ vanish on $W$, we have $\varphi^{(m)}(0) = 0$ for all $m\geq 0$ -- so $\varphi(t)$ is a polynomial it $t$ all of whose derivatives vanish at $t = 0$.  If we write $\varphi(t) = \sum_m c_mt^m$, then $\varphi^{(m)}(0) = m!c_m$ and since $m!\neq 0$ because $K$ has characteristic zero, $c_m = 0$ for all $m$, and therefore, $\varphi(t) = 0$ everywhere. Remembering the definition of $\phi = g(y,t\,df_y) = 0$, so $(y,t\,df_y) \in V(J) = W$.
1928 More generally, since any linear combination of elements in $B$ is again in $B$, we can apply this result to $f = \sum_a c_af_a \in B$ and evaluating at $t = 1$, it follows that 
1929 \[(y, df_y) = (y, d(\sum_ac_af_a)_y) = (y, \sum_a c_ad(f_a)_y) \in W,\]
1930 so the whole linear span $\{y\}\times \operatorname{span}\{df_y:f \in B\}$ is a subset of $W$.\\
1931 
1932 
1933 Since $V(B)=Y=\pi(W)\neq\varnothing$, $B \neq K[x]$, so $B$ must have at least one minimal prime. The ideal $B$ is radical (by construction since it is a vanishing ideal) in the Noetherian ring $K[x]$, so $B$ has only finitely many minimal primes $\mathfrak q_1,\dots,\mathfrak q_r$ and $B=\bigcap_i\mathfrak q_i$. We choose $\mathfrak q=\mathfrak q_1$ and set $Y_0=V(\mathfrak q)$. Since $\mathfrak q$ is prime, $1\notin\mathfrak q$, so $Y_0\neq\varnothing$ by the Nullstellensatz (Theorem~\ref{thm:nss}); moreover $Y_0$ is closed and irreducible since $\mathfrak q$ is prime \cite[Cor.~I.1.4]{hartshorne}, and $Y_0\subseteq Y$, hence
1934 $Y_0\cap H=\varnothing$, as we have already established $Y\cap H=\varnothing$.
1935 
1936 Furthermore, if $r = 1$, we choose $s = 1$, then $sh \in \mathfrak q = B$, and if $r\geq 2$, choose for every $i\geq 2$ elemens $s_i \in \mathfrak q_i \setminus \mathfrak q$. Then $s = s_2s_3\dots s_r$ does not lie in $\mathfrak q$ since no factor lies in $\mathfrak q,$ but does lie in every other $\mathfrak q_i$ since they are ideals and one of the factors lies in them. On the other hand, if $h \in \mathfrak q$, then $sh \in \mathfrak q$ since $h\in \mathfrak q$, and $sh \in \mathfrak q_i$ since $s \in \mathfrak q_i$ for all $i\geq 2$. Therefore, $sh \in \bigcap_i \mathfrak q_i = B,$ i.e. $s\mathfrak q \subseteq B.$ We now fix such a $h \in \mathfrak q,$ and define $Y_0 = V(\mathfrak q).$ For a $y \in Y_0\cap\{s\neq0\}$, $(y, dh_y) \in W:$
1937 Since $sh \in B,$ and $d(sh)_y = s(y)dh_y+h(y)ds_y = s(y)dh_y$ since $h(y) = 0$, so $(y, s(y)dh_y) \in W$ and since $s(y)\neq 0,$ also $(y, dh_y)\in W$. 
1938 For $g\in J$, the polynomial $G(x)=g(x,dh_x)$ therefore vanishes on $Y_0\cap\{s\neq0\}$, and $sG$ vanishes on all of $Y_0$, so $sG \in \mathfrak q$ and therefore $G \in \mathfrak q$, which means that $G$ vanishes on all of $Y_0$. Since this is true for every $g \in J$, we have $g(y, dh_y = G(y) = 0$ for all $g\in J$, so $(y,dh_y) \in V(J)= W.$  As before, we can argue that since $\mathfrak q$ is an ideal, it is closed under $K$-linear combinations, so for $h_1,\dots ,h_k \in \mathfrak q$ and $c_a \in K$, also $(y, \sum_a c_a (dh_a)_y) \in W$, or put differently, $\{y\} \times \operatorname{span}\{dh_y: h \in \mathfrak q\} \subseteq W$ for all $y \in Y_0$. At a smooth point $y$ of $Y_0$, by Definition \ref{def:conormal}, the conormal space is 
1939 \[N_y^*Y_0 = \operatorname{span}\{df_y : f \in \mathfrak q\}\]
1940 since $\mathfrak q$ is prime and hence radical, and so $\mathfrak q = I(Y_0).$ We therefore have that $N_y^*Y_0 \subseteq \{y\} \times W$ for every smooth $y \in Y_0$, and hence $N(Y_0) \subseteq W.$
1941 
1942 By assumption, $P$ vanishes on $W$ and hence on $N(Y_0)$. Because we have established, that $Y_0$ is closed, irreducible, and $Y_0 \cap H = \varnothing $, $Y_0$ therefore fulfills all requirements of Theorem \ref{thm:asymptotic-conormal-formal}, so $P(1,0,\dots,0) = 0,$ contradicting our initial assumption. Therefore, $W \cap \pi^{-1}(H) \neq \varnothing.$
1943 \end{proof}

1955 \begin{lemma}\label{lem:base-change}
1956 Let $K/k$ be a field extension and put
1957 $I_K=\tilde dA_n(K)+x_1^N\tilde dA_n(K)$, $Q_K=Q\otimes_kK=A_n(K)/I_K$, with the
1958 order filtration $Q_{K,i}=q\,F_iA_n(K)$. Then
1959 \[
1960  \gr Q_K\cong\gr Q\otimes_kK,\qquad
1961  \Ann_{K[x,\xi]}\gr Q_K=\bigl(\Ann_{k[x,\xi]}\gr Q\bigr)\,K[x,\xi],
1962 \]
1963 and $Q=0\iff Q_K=0$. In particular, $\Char(Q_K)$ is the preimage of $\Char(Q)$
1964 under $\Spec K[x,\xi]\to\Spec k[x,\xi]$.
1965 \end{lemma}
1966 \begin{proof}
1967 By the PBW basis, $F_iA_n(K)=F_iA_n(k)\otimes_kK$ and $I_K=I\otimes_kK$, so
1968 $Q_{K,i}=Q_i\otimes_kK$. As $K$ is flat over $k$,
1969 $Q_{K,i}/Q_{K,i-1}=(Q_i/Q_{i-1})\otimes_kK$, and summing over $i$ gives
1970 $\gr Q_K=\gr Q\otimes_kK$. The module $\gr Q$ is finitely generated (cyclic,
1971 generated by the class of $q$) over the Noetherian ring $k[x,\xi]$, and
1972 $k[x,\xi]\to K[x,\xi]$ is flat, so the annihilator of the finite module
1973 commutes with this base change \cite[Tag~05DT]{Stacks}. Finally, $Q\otimes_kK=0$
1974 implies $Q=0$, because $K$ is a nonzero free $k$-module.
1975 \end{proof}

1977 \noindent\textbf{Proof of Theorem~\ref{thm:main}.}
1978 Let $d\in A_n(k)$ be nonzero of Bernstein degree $N$ and let
1979 $\tilde d=c^{-1}\Phi_g(d)$ be its monic normal form
1980 (Remark~\ref{rem:transport}). If $n=0$, then $A_0(k)=k$ and every nonzero $d$ is
1981 a unit anyways, so let $n\ge1$. As in Section \ref{sec:euler} we define
1982 \[ I=\tilde dA_n(k)+x_1^N\tilde dA_n(k),\qquad Q=A_n(k)/I .\]
1983 It suffices to show $Q=0$: then $1\in I$, i.e.\ there are
1984 $R,S\in A_n(k)$ with $1=\tilde dR+x_1^N\tilde dS$.
1985 
1986 Let $K=\overline k$, and let
1987 \[W=\Char(Q_K)=V\bigl(\Ann_{K[x,\xi]}\gr Q_K\bigr)\subseteq K^n\times K^n,\]
1988 with vanishing ideal $I(W)=\sqrt{\Ann\gr Q_K}$ by the Nullstellensatz (Theorem \ref{thm:nss}). We check
1989 the hypotheses of Theorem~\ref{thm:coisotropic-exclusion} for $W$ and the
1990 polynomial $P$ of Section~\ref{sec:char}:
1991 
1992 \begin{itemize}
1993 \item $W$ is Zariski closed, since it is the zero set of an ideal of polynomials.
1994 \item The module $\gr Q_K$ is graded by the order filtration, which assigns degree
1995 $0$ to the $x_i$ and degree $1$ to the $\xi_i$; so $\Ann\gr Q_K$  is homogeneous in $\xi$. Hence if $(y,\xi)\in W$
1996 and $g\in\Ann\gr Q_K$, then $g(y,t\xi)=\sum_dt^dg_d(y,\xi)=0$ for every
1997 $t\in K^\times$, where $g_d$ denotes the $\xi$-homogeneous component of degree $d$ of $g$. So
1998 $(y,t\xi)\in W$, i.e.\ $W$ is fibre-conical.
1999 \item By Gabber's theorem (Theorem~\ref{ithm:gabber}) $I(W)$ is involutive. 
2000 \item By eq.~\eqref{eq:P-annihilates}, $P\in
2001 \Ann_{k[x,\xi]}\gr Q$, hence $P\in\Ann_{K[x,\xi]}\gr Q_K$ by
2002 Lemma~\ref{lem:base-change}, so $P$ vanishes on $W$.
2003 \item By eq.~\eqref{eq:P-e1}, $P(1,0,\dots,0)=1\neq 0$.
2004 \item  $W=\Char(Q_K)$ is the preimage of $\Char(Q)$ under
2005 $\Spec K[x,\xi]\to\Spec k[x,\xi]$ (Lemma~\ref{lem:base-change}), and
2006 $\pi^{-1}(H)=V(x_1)$ over $K$ is the preimage of $\pi^{-1}(H)=V(x_1)$ over $k$.
2007 By Proposition~\ref{prop:char-avoids}, $\Char(Q)\cap\pi^{-1}(H)=\varnothing$, and
2008 the preimage of an empty set is also empty, therefore $W\cap\pi^{-1}(H)=\varnothing$.
2009 
2010 \end{itemize}
2011 By Theorem~\ref{thm:coisotropic-exclusion}, $W=\varnothing$. As $K$ is
2012 algebraically closed, by the Nullstellensatz (Theorem \ref{thm:nss})  $1 \in \Ann_{K[x,\xi]}\gr Q_K$,
2013 so $\gr Q_K=0$. Since the filtration of $Q_K$ is exhaustive with $Q_{K,-1}=0$,
2014 induction on $m$ implies $Q_{K,m}=0$ for all $m$, so $Q_K=0$, and
2015 $Q=0$ by Lemma~\ref{lem:base-change}. Therefore, $A_n(k) = \tilde dA_n(k) + x_1^N\tilde dA_n(k)$, or equivalently $1=\tilde dR+x_1^N\tilde dS$
2016 with $R,S\in A_n(k)$.

2018 Lastly, we apply the inverse of the automorphism $\Phi_g$ of
2019 Remark~\ref{rem:transport}. Since $\tilde d=c^{-1}\Phi_g(d)$ with $c\in k^\times$
2020 central, this gives
2021 \[
2022  1=d\bigl(c^{-1}\Phi_g^{-1}(R)\bigr)+F\,d\bigl(c^{-1}\Phi_g^{-1}(S)\bigr),
2023  \qquad F=\Phi_g^{-1}(x_1)^N,
2024 \]
2025 which proves Theorem~\ref{thm:main}. \qed

2042 Lastly, as Stafford already showed, from $1 = dR+FdS$ follows cyclicity of the torsion right module:
2043 \begin{corollary}\label{cor:cyclic}
2044 Every finitely generated torsion right $A_n(k)$-module is cyclic.
2045 \end{corollary}
2046 
2047 \begin{proof}
2048     Suppose for every $d\neq 0 \in A_n$, there exists $F, R, S \in A_n$, such that $1 = dR + FdS$. Let us denote a torsion module by $T$. Stafford's theorem (Theorem 3.7 \mbox{\cite[p.~438]{Stafford1978}}) established, that every finitely generated torsion module can be generated by 2 generators, so let us denote the generators of $T$ with $t_1$ and $t_2$. We can define a homomorphism
2049     \[\phi:A_n \oplus A_n\rightarrow T \quad \phi(r,s)\mapsto t_1r+t_2s.\]
2050     This map is surjective, since $t_1, t_2$ generate all of $T$. By the first isomorphism theorem (for $\phi:M\rightarrow N$, $im(\phi) \cong M/ker(\phi)$), it follows
2051     \[T \cong (A_n\oplus A_n)/M\]
2052     with $M = ker(\phi)$. \\
2053     $T$ is torsion and therefore $\forall t\in T: \exists m \in A_n\backslash\{0\}: tm = 0.$ Specifically, let us denote with $d_1$ and $d_2$ the annihilators of $t_1$ and $t_2$, i.e.
2054     \[t_1d_1 = t_2d_2 = 0.\]
2055     $A_n$ is a right Ore domain, which means it has the property that for any nonzero $d_1, d_2$ there exists elements $r_1, r_2 \in A_n$ such that $d_1r_1 = d_2r_2 \neq 0.$
2056     Defining therefore $d := d_1r_1 = d_2r_2$ yields us a common annihilator of $t_1$ and $t_2$.\\
2057     $(d, 0)$ and $(0,d)$ are elements in $M = ker(\phi)$, since $\phi(d, 0) = t_1d = 0$ and $\phi(0, d) = t_2d = 0.$\\
2058     Because $(0,d), (d, 0) \in M$, also
2059     $(d, 0)S + (0, d)R = (dS, dR) = (dS, 1-FdS) =: (a, 1-Fa) \in M$, i.e. $(a, 1-Fa)$ is an element of $M$. \\
2060     On the other hand, $A_n\oplus A_n$ can be generated by $(a, 1-Fa)$ and $(1, -F)$. We can check this by finding $u,v$ such that for all $(x,y)\in A_n\oplus A_n$ we can write
2061     \[(x,y) = (a, 1-Fa)u + (1, -F)v.\]
2062     Suitable choices are $u = y+Fx$ and $v = x-a(y+Fx)$.\\
2063     Lastly, since  $A_n\oplus A_n$ can be generated by $(a, 1-Fa)$ and $(1, -F)$ and $(a, 1-Fa)$ is an element of $M$, it follows that $T \cong (A_n\oplus A_n)/M$ can be generated by the image of $(1, -F)$, i.e. $T$ is cyclic.
2064 \end{proof}
